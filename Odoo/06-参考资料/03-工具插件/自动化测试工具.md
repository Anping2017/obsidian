from odoo.tests.common import TransactionCase
from odoo.exceptions import ValidationError

class TestMyModel(TransactionCase):
    """测试MyModel模型"""
    
    def setUp(self):
        """测试前准备"""
        super().setUp()
        self.test_model = self.env['my.model']
        self.test_partner = self.env['res.partner'].create({
            'name': 'Test Partner',
            'email': 'test@example.com'
        })
    
    def test_create_record(self):
        """测试记录创建"""
        # 创建测试数据
        record = self.test_model.create({
            'name': 'Test Record',
            'partner_id': self.test_partner.id,
            'amount': 100.0
        })
        
        # 验证记录创建
        self.assertTrue(record.id)
        self.assertEqual(record.name, 'Test Record')
        self.assertEqual(record.partner_id.id, self.test_partner.id)
        self.assertEqual(record.amount, 100.0)
    
    def test_validation_error(self):
        """测试验证错误"""
        with self.assertRaises(ValidationError):
            self.test_model.create({
                'name': '',  # 空名称应该触发验证错误
                'amount': -100.0  # 负数金额应该触发验证错误
            })
    
    def test_computed_field(self):
        """测试计算字段"""
        record = self.test_model.create({
            'name': 'Test Record',
            'amount': 100.0
        })
        
        # 验证计算字段
        expected_total = record.amount * 1.1
        self.assertEqual(record.total_amount, expected_total)
    
    def test_business_method(self):
        """测试业务方法"""
        record = self.test_model.create({
            'name': 'Test Record',
            'state': 'draft'
        })
        
        # 执行业务方法
        record.action_confirm()
        
        # 验证状态变化
        self.assertEqual(record.state, 'confirmed')
    
    def test_search_domain(self):
        """测试搜索域"""
        # 创建测试数据
        self.test_model.create({'name': 'Record 1', 'state': 'draft'})
        self.test_model.create({'name': 'Record 2', 'state': 'confirmed'})
        
        # 测试搜索
        draft_records = self.test_model.search([('state', '=', 'draft')])
        confirmed_records = self.test_model.search([('state', '=', 'confirmed')])
        
        self.assertEqual(len(draft_records), 1)
        self.assertEqual(len(confirmed_records), 1)
    
    def test_constraints(self):
        """测试约束"""
        record = self.test_model.create({
            'name': 'Test Record',
            'amount': 100.0
        })
        
        # 测试约束
        with self.assertRaises(ValidationError):
            record.write({'amount': -50.0})
    
    def test_onchange_method(self):
        """测试onchange方法"""
        record = self.test_model.new({
            'name': 'Test Record',
            'amount': 100.0
        })
        
        # 触发onchange

        # 验证onchange结果

# 负载测试
import time
import threading
from concurrent.futures import ThreadPoolExecutor

class TestPerformance(TransactionCase):
    """性能测试"""
    
    def test_concurrent_requests(self):
        """测试并发请求"""
        def make_request():
            """模拟请求"""
            start_time = time.time()
            
            # 执行数据库操作
            record = self.test_model.create({
                'name': f'Concurrent Record {threading.current_thread().ident}',
                'amount': 100.0
            })
            
            end_time = time.time()
            return end_time - start_time
        
        # 并发执行请求
        with ThreadPoolExecutor(max_workers=10) as executor:
            futures = [executor.submit(make_request) for _ in range(100)]
            results = [future.result() for future in futures]
        
        # 分析结果
        avg_time = sum(results) / len(results)
        max_time = max(results)
        min_time = min(results)
        
        # 验证性能
        self.assertLess(avg_time, 1.0)  # 平均响应时间小于1秒
        self.assertLess(max_time, 5.0)  # 最大响应时间小于5秒
    
    def test_memory_usage(self):
        """测试内存使用"""
        import psutil
        import os
        
        process = psutil.Process(os.getpid())
        initial_memory = process.memory_info().rss / 1024 / 1024  # MB
        
        # 执行内存密集型操作
        records = []
        for i in range(10000):
            record = self.test_model.create({
                'name': f'Memory Test Record {i}',
                'amount': i * 10
            })
            records.append(record)
        
        peak_memory = process.memory_info().rss / 1024 / 1024  # MB
        memory_increase = peak_memory - initial_memory
        
        # 验证内存使用
        self.assertLess(memory_increase, 500)  # 内存增长小于500MB
        
        # 清理
        records.unlink()
    
    def test_database_performance(self):
        """测试数据库性能"""
        # 测试查询性能
        start_time = time.time()
        
        # 执行复杂查询

        end_time = time.time()

#### 2. 测试报告
```python
# 测试报告
import unittest
import xmlrunner
import coverage

class TestRunner:
    """测试运行器"""
    
    def __init__(self):
        self.coverage = coverage.Coverage()
    
    def run_tests_with_coverage(self):
        """运行测试并生成覆盖率报告"""
        # 开始覆盖率收集
        self.coverage.start()
        
        # 运行测试
        loader = unittest.TestLoader()
        suite = loader.discover('tests', pattern='test_*.py')
        
        runner = xmlrunner.XMLTestRunner(output='test-reports')
        result = runner.run(suite)
        
        # 停止覆盖率收集
        self.coverage.stop()
        
        # 生成覆盖率报告
        self.coverage.report()
        self.coverage.html_report(directory='coverage-report')
        
        return result
```
