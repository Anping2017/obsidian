```python
# 数据库调试配置
class DatabaseDebug:
    def __init__(self, env):
        self.env = env
    
    def enable_sql_logging(self):
        """启用SQL日志"""
        self.env.cr.execute("SET log_statement = 'all'")
        self.env.cr.execute("SET log_min_duration_statement = 1000")
    
    def analyze_query_performance(self, query):
        """分析查询性能"""
        self.env.cr.execute(f"EXPLAIN ANALYZE {query}")
        return self.env.cr.fetchall()
    
    def get_slow_queries(self):
        """获取慢查询"""
        self.env.cr.execute("""
            SELECT query, mean_time, calls, total_time
            FROM pg_stat_statements
            WHERE mean_time > 1000
            ORDER BY mean_time DESC
            LIMIT 10
        """)
        return self.env.cr.fetchall()
```

# .git/hooks/pre-commit

# 代码生成工具
class CodeGenerator:
    def __init__(self, module_name, model_name):
        self.module_name = module_name
        self.model_name = model_name
    
    def generate_model(self):
        """生成模型代码"""
        template = f'''
from odoo import models, fields, api
from odoo.exceptions import ValidationError

class {self.model_name.title().replace('_', '')}(models.Model):
    _name = '{self.model_name}'
    _description = '{self.model_name.title().replace("_", " ")}'
    _inherit = ['mail.thread', 'mail.activity.mixin']
    
    name = fields.Char('名称', required=True, index=True, tracking=True)
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '完成'),
        ('cancelled', '已取消')
    ], string='状态', default='draft', tracking=True)
    
    @api.constrains('name')
    def _check_name(self):
        for record in self:
            if not record.name:
                raise ValidationError("名称不能为空")
    
    def action_confirm(self):
        self.write({{'state': 'confirmed'}})
        return True
    
    def action_done(self):
        self.write({{'state': 'done'}})
        return True
    
    def action_cancel(self):
        self.write({{'state': 'cancelled'}})
        return True
'''
        return template
    
    def generate_view(self):
        """生成视图代码"""
        template = f'''
<odoo>
    <record id="view_{self.model_name}_form" model="ir.ui.view">
        <field name="name">{self.model_name}.form</field>
        <field name="model">{self.model_name}</field>
        <field name="arch" type="xml">
            <form>
                <header>
                    <button name="action_confirm" type="object" string="确认" 
                            class="btn-primary" 
                            attrs="{{'invisible': [('state', '!=', 'draft')]}}"/>
                    <button name="action_done" type="object" string="完成" 
                            class="btn-primary"

                            class="btn-secondary"

                    <field name="state" widget="statusbar" 
                           statusbar_visible="draft,confirmed,done"/>
                </header>
                <sheet>
                    <div class="oe_title">
                        <h1><field name="name"/></h1>
                    </div>
                </sheet>
            </form>
        </field>
    </record>
    
    <record id="view_{self.model_name}_tree" model="ir.ui.view">
        <field name="name">{self.model_name}.tree</field>

class TestDataGenerator:
    def __init__(self, env):
        self.env = env
    
    def generate_partners(self, count=100):
        """生成合作伙伴测试数据"""
        partners = []
        for i in range(count):
            partner = self.env['res.partner'].create({
                'name': f'Test Partner {i}',
                'email': f'test{i}@example.com',
                'phone': f'+86-138-{i:08d}',
                'is_company': i % 3 == 0
            })
            partners.append(partner)
        return partners
    
    def generate_products(self, count=50):
        """生成产品测试数据"""
        products = []
        for i in range(count):
            product = self.env['product.product'].create({
                'name': f'Test Product {i}',
                'list_price': 100 + i * 10,
                'standard_price': 80 + i * 8,
                'type': 'product' if i % 2 == 0 else 'service'
            })
            products.append(product)
        return products
    
    def generate_sale_orders(self, count=20):
        """生成销售订单测试数据"""
        partners = self.env['res.partner'].search([], limit=10)
        products = self.env['product.product'].search([], limit=20)
        
        orders = []
        for i in range(count):
            order = self.env['sale.order'].create({
                'partner_id': partners[i % len(partners)].id,
                'order_line': [(0, 0, {
                    'product_id': products[i % len(products)].id,
                    'product_uom_qty': 1 + (i % 5),
                    'price_unit': 100 + i * 10
                })]
            })
            orders.append(order)
        return orders

#### 3. 性能分析工具
```python
# 性能分析工具
import cProfile
import pstats
import io
from functools import wraps

class PerformanceProfiler:
    def __init__(self):
        self.profiler = cProfile.Profile()
    
    def profile_method(self, func):
        """方法性能分析装饰器"""
        @wraps(func)
        def wrapper(*args, **kwargs):
            self.profiler.enable()
            result = func(*args, **kwargs)
            self.profiler.disable()
            return result
        return wrapper
    
    def get_stats(self):
        """获取性能统计"""
        s = io.StringIO()
        ps = pstats.Stats(self.profiler, stream=s)
        ps.sort_stats('cumulative')
        ps.print_stats(20)  # 显示前20个最耗时的函数
        return s.getvalue()
    
    def save_stats(self, filename):
        """保存性能统计到文件"""
        self.profiler.dump_stats(filename)

# 使用示例
profiler = PerformanceProfiler()

class MyModel(models.Model):
    _name = 'my.model'
    
    @profiler.profile_method
    def slow_method(self):
        # 耗时操作
        time.sleep(1)
        return "完成"
    
    def analyze_performance(self):
        # 执行方法
        self.slow_method()
        
        # 获取统计信息
        stats = profiler.get_stats()
        print(stats)
        
        # 保存到文件
        profiler.save_stats('performance.prof')
```
