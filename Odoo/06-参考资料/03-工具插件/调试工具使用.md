# 调试工具使用

## Odoo 17 调试工具使用指南

本文档介绍了Odoo 17开发中常用的调试工具，包括Python调试器、Odoo内置调试功能、前端调试工具、数据库调试工具等，帮助开发者提高调试效率。

### Python调试工具

#### 1. pdb调试器
```python
# 基本pdb使用
import pdb

class MyModel(models.Model):
    _name = 'my.model'
    _description = 'My Model'
    
    def debug_method(self):
        # 设置断点
        pdb.set_trace()
        
        # 执行代码
        result = self._calculate_value()
        
        # 继续执行
        return result
    
    def _calculate_value(self):
        # 条件断点
        if self.amount > 1000:
            pdb.set_trace()  # 只在特定条件下停止
        
        return self.amount * 1.1

# 高级pdb使用
def advanced_debug_example(self):
    # 设置断点并执行命令
    pdb.set_trace()
    
    # 在pdb中可以执行的命令：
    # l (list) - 显示当前代码
    # n (next) - 执行下一行
    # s (step) - 进入函数内部
    # c (continue) - 继续执行
    # p variable_name - 打印变量值
    # pp variable_name - 美化打印变量
    # w (where) - 显示调用栈
    # u (up) - 向上移动调用栈
    # d (down) - 向下移动调用栈
    # h (help) - 显示帮助
    # q (quit) - 退出调试器
    
    values = [1, 2, 3, 4, 5]
    for i, value in enumerate(values):
        if value > 3:
            result = value * 2
            break
    
    return result
```

#### 3. 远程调试
```python
# 使用pdb进行远程调试
import pdb
import sys

class RemoteDebugModel(models.Model):
    _name = 'remote.debug.model'
    _description = 'Remote Debug Model'
    
    def remote_debug_method(self):
        # 设置远程调试
        # 在服务器上运行：python -m pdb -c continue script.py
        # 在客户端连接：telnet localhost 4444
        
        # 或者使用rpdb
        try:
            import rpdb
            rpdb.set_trace(port=4444)
        except ImportError:
            # 如果没有rpdb，使用标准pdb
            pdb.set_trace()
        
        # 执行需要调试的代码
        result = self._complex_calculation()
        return result
    
    def _complex_calculation(self):
        # 复杂的计算逻辑
        data = self.env['sale.order'].search([])
        
        total = 0
        for order in data:
            if order.state == 'sale':
                total += order.amount_total
        
        return total
```

#### 2. 性能调试
```python
import time
from functools import wraps

def performance_debug(func):
    """性能调试装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start_time = time.time()
        _logger.info(f"开始执行 {func.__name__}")
        
        try:
            result = func(*args, **kwargs)
            execution_time = time.time() - start_time
            _logger.info(f"{func.__name__} 执行完成，耗时：{execution_time:.2f}秒")
            return result
        except Exception as e:
            execution_time = time.time() - start_time
            _logger.error(f"{func.__name__} 执行失败，耗时：{execution_time:.2f}秒，错误：{e}")
            raise
    
    return wrapper

class PerformanceDebugModel(models.Model):
    _name = 'performance.debug.model'
    _description = 'Performance Debug Model'
    
    @performance_debug
    def slow_method(self):
        # 模拟耗时操作
        time.sleep(1)
        return "完成"
    
    def memory_debug(self):
        import psutil
        import os
        
        process = psutil.Process(os.getpid())
        memory_before = process.memory_info().rss / 1024 / 1024  # MB
        
        _logger.info(f"执行前内存使用：{memory_before:.2f} MB")
        
        # 执行操作
        data = self._load_large_data()
        
        memory_after = process.memory_info().rss / 1024 / 1024  # MB
        _logger.info(f"执行后内存使用：{memory_after:.2f} MB")
        _logger.info(f"内存增长：{memory_after - memory_before:.2f} MB")
        
        return data
    
    def _load_large_data(self):
        # 加载大量数据
        return list(range(1000000))
```

#### 3. 数据库调试
```python
class DatabaseDebugModel(models.Model):
    _name = 'database.debug.model'
    _description = 'Database Debug Model'
    
    def debug_database_queries(self):
        # 启用SQL日志
        self.env.cr.execute("SET log_statement = 'all'")
        
        # 执行查询
        records = self.search([('state', '=', 'draft')])
        
        # 分析查询计划
        self.env.cr.execute("EXPLAIN ANALYZE SELECT * FROM database_debug_model WHERE state = 'draft'")
        explain_result = self.env.cr.fetchall()
        
        _logger.info("查询计划：")
        for row in explain_result:
            _logger.info(row[0])
        
        return records
    
    def debug_slow_queries(self):
        # 检查慢查询
        self.env.cr.execute("""
            SELECT query, mean_time, calls, total_time
            FROM pg_stat_statements
            WHERE mean_time > 1000
            ORDER BY mean_time DESC
            LIMIT 10
        """)
        
        slow_queries = self.env.cr.fetchall()
        
        _logger.info("慢查询列表：")
        for query in slow_queries:
            _logger.info(f"查询：{query[0][:100]}...")
            _logger.info(f"平均时间：{query[1]}ms")
            _logger.info(f"调用次数：{query[2]}")
            _logger.info(f"总时间：{query[3]}ms")
        
        return slow_queries
```

```javascript
    // 重写rpc调用以添加调试信息
```

#### 1. PostgreSQL调试
```sql
-- 启用查询日志
SET log_statement = 'all';
SET log_min_duration_statement = 1000; -- 记录超过1秒的查询

-- 查看当前连接
SELECT pid, usename, application_name, client_addr, state, query
FROM pg_stat_activity
WHERE state = 'active';

-- 查看锁等待
SELECT blocked_locks.pid AS blocked_pid,
       blocked_activity.usename AS blocked_user,
       blocking_locks.pid AS blocking_pid,
       blocking_activity.usename AS blocking_user,
       blocked_activity.query AS blocked_statement
FROM pg_catalog.pg_locks blocked_locks
JOIN pg_catalog.pg_stat_activity blocked_activity ON blocked_activity.pid = blocked_locks.pid
JOIN pg_catalog.pg_locks blocking_locks ON blocking_locks.locktype = blocked_locks.locktype
JOIN pg_catalog.pg_stat_activity blocking_activity ON blocking_activity.pid = blocking_locks.pid
WHERE NOT blocked_locks.granted;

-- 查看表大小
SELECT schemaname, tablename, 
       pg_size_pretty(pg_total_relation_size(schemaname||'.'||tablename)) as size
FROM pg_tables
WHERE schemaname = 'public'
ORDER BY pg_total_relation_size(schemaname||'.'||tablename) DESC;
```

```python
class QueryDebugModel(models.Model):
    _name = 'query.debug.model'
    _description = 'Query Debug Model'
    
    def analyze_query_performance(self):
        # 分析查询性能
        query = "SELECT * FROM sale_order WHERE state = 'sale'"
        
        # 执行EXPLAIN ANALYZE
        self.env.cr.execute(f"EXPLAIN ANALYZE {query}")
        explain_result = self.env.cr.fetchall()
        
        _logger.info("查询性能分析：")
        for row in explain_result:
            _logger.info(row[0])
        
        # 执行查询并测量时间
        start_time = time.time()
        self.env.cr.execute(query)
        results = self.env.cr.fetchall()
        execution_time = time.time() - start_time
        
        _logger.info(f"查询执行时间：{execution_time:.2f}秒")
        _logger.info(f"返回记录数：{len(results)}")
        
        return results
    
    def debug_index_usage(self):
        # 检查索引使用情况
        self.env.cr.execute("""
            SELECT schemaname, tablename, indexname, idx_scan, idx_tup_read, idx_tup_fetch
            FROM pg_stat_user_indexes
            WHERE schemaname = 'public'
            ORDER BY idx_scan DESC
        """)
        
        index_usage = self.env.cr.fetchall()
        
        _logger.info("索引使用情况：")
        for index in index_usage:
            _logger.info(f"表：{index[1]}, 索引：{index[2]}, 扫描次数：{index[3]}")
        
        return index_usage
```

#### 1. 调试配置文件
```python
# 调试配置文件
import os
import logging

class DebugConfig:
    def __init__(self):
        self.debug_mode = os.getenv('ODOO_DEBUG', 'False').lower() == 'true'
        self.log_level = os.getenv('ODOO_LOG_LEVEL', 'INFO')
        self.enable_pdb = os.getenv('ODOO_ENABLE_PDB', 'False').lower() == 'true'
        self.enable_sql_log = os.getenv('ODOO_ENABLE_SQL_LOG', 'False').lower() == 'true'
    
    def setup_logging(self):
        """设置日志配置"""
        if self.debug_mode:
            logging.basicConfig(
                level=getattr(logging, self.log_level),
                format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
                handlers=[
                    logging.FileHandler('/var/log/odoo/debug.log'),
                    logging.StreamHandler()
                ]
            )
    
    def setup_database_debug(self, env):
        """设置数据库调试"""
        if self.enable_sql_log:
            env.cr.execute("SET log_statement = 'all'")
            env.cr.execute("SET log_min_duration_statement = 1000")
    
    def setup_pdb(self):
        """设置pdb调试"""
        if self.enable_pdb:
            import pdb
            pdb.set_trace()
```

```python
# 调试中间件
class DebugMiddleware:
    def __init__(self, app):
        self.app = app
    
    def __call__(self, environ, start_response):
        # 记录请求信息
        _logger.info(f"请求：{environ['REQUEST_METHOD']} {environ['PATH_INFO']}")
        
        start_time = time.time()
        
        def new_start_response(status, headers):
            # 记录响应信息
            end_time = time.time()
            _logger.info(f"响应：{status}，耗时：{end_time - start_time:.2f}秒")
            return start_response(status, headers)
        
        return self.app(environ, new_start_response)
```
