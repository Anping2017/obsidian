# 字段类型

## Odoo 17 字段类型详解

Odoo提供了丰富的字段类型来满足不同的业务需求。本文档详细介绍各种字段类型的使用方法和最佳实践。

### 基础字段类型

#### 文本字段
```python
from odoo import models, fields

class MyModel(models.Model):
    _name = 'my.model'
    
    # 字符字段
    name = fields.Char('名称', required=True, size=100)
    code = fields.Char('编码', size=20, index=True)
    
    # 文本字段
    description = fields.Text('描述')
    notes = fields.Text('备注', help='详细说明信息')
    
    # HTML字段
    html_content = fields.Html('HTML内容')
    
    # 字段属性
    title = fields.Char(
        string='标题',           # 显示标签
        required=True,           # 必填
        readonly=True,           # 只读
        invisible=True,          # 隐藏
        default='默认值',        # 默认值
        help='字段帮助文本',     # 帮助文本
        index=True,              # 创建索引
        store=True,              # 存储到数据库
        copy=True,               # 复制时包含
        translate=True,          # 可翻译
        size=100,                # 字段大小
        trim=True,               # 去除空格
    )
```

#### 数值字段
```python
class MyModel(models.Model):
```

### 关系字段类型

#### Many2one字段
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    # 多对一关系
    partner_id = fields.Many2one('res.partner', '合作伙伴')
    user_id = fields.Many2one('res.users', '用户', default=lambda self: self.env.user)
    company_id = fields.Many2one('res.company', '公司', default=lambda self: self.env.company)
    
    # 字段属性
    partner_id = fields.Many2one(
        'res.partner',           # 关联模型
        string='合作伙伴',       # 显示标签
        required=True,           # 必填
        readonly=False,          # 可编辑
        domain=[('is_company', '=', True)],  # 过滤条件
        context={'default_is_company': True},  # 默认上下文
        ondelete='cascade',      # 删除行为
        help='选择合作伙伴'
    )
    
    # 访问关联字段
    def get_partner_name(self):
        return self.partner_id.name if self.partner_id else ''
    
    def get_partner_email(self):
        return self.partner_id.email if self.partner_id else ''
```

#### One2many字段
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    # 一对多关系
    line_ids = fields.One2many('my.model.line', 'model_id', '明细行')
    attachment_ids = fields.One2many('ir.attachment', 'res_id', '附件')
    
    # 字段属性
    line_ids = fields.One2many(
        'my.model.line',         # 关联模型
        'model_id',              # 关联字段
        string='明细行',         # 显示标签
        readonly=False,          # 可编辑
        copy=True,               # 复制时包含
        help='明细行信息'
    )
    
    # 计算字段
    total_amount = fields.Float('总金额', compute='_compute_total')
    
    @api.depends('line_ids.amount')
    def _compute_total(self):
        for record in self:
            record.total_amount = sum(record.line_ids.mapped('amount'))

class MyModelLine(models.Model):
```

#### Many2many字段
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    # 多对多关系
    tag_ids = fields.Many2many('my.model.tag', 'model_tag_rel', 'model_id', 'tag_id', '标签')
    category_ids = fields.Many2many('product.category', 'model_category_rel', 'model_id', 'category_id', '分类')
    
    # 字段属性
    tag_ids = fields.Many2many(
        'my.model.tag',          # 关联模型
        'model_tag_rel',         # 关联表名
        'model_id',              # 本模型字段
        'tag_id',                # 关联模型字段
        string='标签',           # 显示标签
        readonly=False,          # 可编辑
        copy=True,               # 复制时包含
        help='选择标签'
    )
    
    # 操作多对多关系
    def add_tag(self, tag_id):
        self.tag_ids = [(4, tag_id)]  # 添加
    
    def remove_tag(self, tag_id):
        self.tag_ids = [(3, tag_id)]  # 移除
    
    def set_tags(self, tag_ids):
        self.tag_ids = [(6, 0, tag_ids)]  # 替换所有

class MyModelTag(models.Model):

    name = fields.Char('标签名称', required=True)
    color = fields.Integer('颜色')
```

#### 计算字段
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    # 计算字段
    total_amount = fields.Float('总金额', compute='_compute_total', store=True)
    char_count = fields.Integer('字符数', compute='_compute_char_count')
    display_name = fields.Char('显示名称', compute='_compute_display_name')
    
    # 存储的计算字段
    @api.depends('line_ids.amount')
    def _compute_total(self):
        for record in self:
            record.total_amount = sum(record.line_ids.mapped('amount'))
    
    # 不存储的计算字段
    @api.depends('name', 'description')
    def _compute_char_count(self):
        for record in self:
            name_len = len(record.name or '')
            desc_len = len(record.description or '')
            record.char_count = name_len + desc_len
    
    # 显示名称计算
    @api.depends('name', 'code')
    def _compute_display_name(self):
        for record in self:
            if record.name and record.code:
                record.display_name = f"[{record.code}] {record.name}"
            elif record.name:
                record.display_name = record.name
            else:
                record.display_name = f"记录 {record.id}"
```

### 字段约束和验证

#### 约束验证
```python
from odoo.exceptions import ValidationError

class MyModel(models.Model):
    _name = 'my.model'
    
    name = fields.Char('名称', required=True)
    email = fields.Char('邮箱')
    amount = fields.Float('金额', digits=(16, 2))
    date = fields.Date('日期')
    
    @api.constrains('name')
    def _check_name(self):
        for record in self:
            if len(record.name) < 3:
                raise ValidationError("名称长度不能少于3个字符")
    
    @api.constrains('email')
    def _check_email(self):
        for record in self:
            if record.email and '@' not in record.email:
                raise ValidationError("邮箱格式不正确")
    
    @api.constrains('amount')
    def _check_amount(self):
        for record in self:
            if record.amount < 0:
                raise ValidationError("金额不能为负数")
    
    @api.constrains('name', 'amount')
    def _check_name_amount(self):
        for record in self:
            if record.name == 'VIP' and record.amount < 1000:
                raise ValidationError("VIP客户金额不能少于1000")
```

#### 继承字段

```python
class BaseModel(models.Model):
```

### 字段最佳实践

#### 1. 字段命名
```python
# 好的命名
name = fields.Char('名称')
partner_id = fields.Many2one('res.partner', '合作伙伴')
line_ids = fields.One2many('sale.order.line', 'order_id', '订单行')

# 避免的命名
n = fields.Char('名称')  # 太简短
partner = fields.Many2one('res.partner', '合作伙伴')  # 缺少_id后缀
lines = fields.One2many('sale.order.line', 'order_id', '订单行')  # 缺少_ids后缀
```

#### 2. 字段属性
```python
# 合理使用字段属性
name = fields.Char(
    string='名称',
    required=True,      # 必填字段
    index=True,         # 经常查询的字段
    translate=True,     # 需要翻译的字段
    help='字段说明'     # 提供帮助信息
)

# 避免过度使用属性
name = fields.Char(
    string='名称',
    required=True,
    readonly=True,      # 通常不需要
    invisible=True,     # 通常不需要
    copy=False,         # 通常不需要
    store=False,        # 通常不需要
)
```

#### 3. 性能优化
```python
# 使用适当的字段类型
# 好的做法
amount = fields.Float('金额', digits=(16, 2))  # 指定精度
date = fields.Date('日期')  # 使用Date而不是Datetime

# 避免的做法
amount = fields.Float('金额')  # 没有指定精度
date = fields.Datetime('日期')  # 不需要时间时使用Datetime
```
