```python
class DataQueryExamples(models.Model):
    _name = 'data.query.examples'
    
    name = fields.Char('名称')
    amount = fields.Float('金额')
    date = fields.Date('日期')
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '完成'),
    ], string='状态')
    partner_id = fields.Many2one('res.partner', string='客户')
    
    def query_examples(self):
        # 1. 基础查询
        all_records = self.search([])
        
        # 2. 条件查询
        draft_records = self.search([('state', '=', 'draft')])
        
        # 3. 多条件查询
        confirmed_records = self.search([
            ('state', '=', 'confirmed'),
            ('amount', '>', 1000)
        ])
        
        # 4. 复杂条件查询
        complex_records = self.search([
            '|',  # OR条件
            ('state', '=', 'draft'),
            '&',  # AND条件
            ('state', '=', 'confirmed'),
            ('amount', '>', 1000)
        ])
        
        # 5. 模糊查询
        fuzzy_records = self.search([
            ('name', 'ilike', '%test%')
        ])
        
        # 6. 范围查询
        range_records = self.search([
            ('amount', '>=', 100),
            ('amount', '<=', 1000)
        ])
        
        # 7. 包含查询
        in_records = self.search([
            ('state', 'in', ['draft', 'confirmed'])
        ])
        
        # 8. 空值查询
        null_records = self.search([
            ('name', '=', False)
        ])
        
        # 9. 非空查询
        not_null_records = self.search([
            ('name', '!=', False)
        ])
        
        # 10. 排序查询
        sorted_records = self.search([], order='amount desc')
        
        # 11. 限制数量
        limited_records = self.search([], limit=10)
        
        # 12. 偏移查询
        offset_records = self.search([], offset=10, limit=10)
        
        # 13. 计数查询
        count = self.search_count([('state', '=', 'draft')])
        
        return {
            'all': all_records,
            'draft': draft_records,
            'confirmed': confirmed_records,
            'complex': complex_records,
            'fuzzy': fuzzy_records,
            'range': range_records,
            'in': in_records,
            'null': null_records,
            'not_null': not_null_records,
            'sorted': sorted_records,
            'limited': limited_records,
            'offset': offset_records,
            'count': count,
        }
```

```python
            # 记录存在

        records = self.search([])
        filtered_records = records.filtered(lambda r: r.amount > 1000)
        
        # 6. 使用sorted排序记录
        records = self.search([])
        sorted_records = records.sorted(lambda r: r.amount, reverse=True)
        
        # 7. 使用mapped获取字段值
        records = self.search([])
        names = records.mapped('name')
        amounts = records.mapped('amount')
        
        # 8. 使用with_context设置上下文

        # 9. 使用sudo提升权限

        # 10. 使用ensure_one确保单记录
        record = self.search([], limit=1)
        record.ensure_one()  # 确保只有一条记录
        
        return {
            'browse': records,
```

```python
class DataCreateExamples(models.Model):
    _name = 'data.create.examples'
    
    name = fields.Char('名称')
    amount = fields.Float('金额')
    line_ids = fields.One2many('data.create.examples.line', 'parent_id', string='明细行')
    
    def create_examples(self):
        # 1. 单条记录创建
        record = self.create({
            'name': '测试记录',
            'amount': 1000,
        })
        
        # 2. 批量创建
        vals_list = []
        for i in range(10):
            vals_list.append({
                'name': f'记录{i}',
                'amount': i * 100,
            })
        records = self.create(vals_list)
        
        # 3. 带关联记录的创建
        record_with_lines = self.create({
            'name': '带明细的记录',
            'amount': 2000,
            'line_ids': [
                (0, 0, {'name': '明细1', 'amount': 500}),
                (0, 0, {'name': '明细2', 'amount': 1500}),
            ]
        })
        
        # 4. 使用default_get获取默认值
        defaults = self.default_get(['name', 'amount'])
        record_with_defaults = self.create(defaults)
        
        # 5. 使用copy复制记录
        original_record = self.search([], limit=1)
        if original_record:
            copied_record = original_record.copy({
                'name': '复制的记录',
            })
        
        return {
            'single': record,
            'batch': records,
            'with_lines': record_with_lines,
            'with_defaults': record_with_defaults,
            'copied': copied_record if 'copied_record' in locals() else None,
        }
```

```python
class DataUpdateExamples(models.Model):
    _name = 'data.update.examples'
    
    name = fields.Char('名称')
    amount = fields.Float('金额')
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '完成'),
    ], string='状态')
    
    def update_examples(self):
        # 1. 单条记录更新
        record = self.search([], limit=1)
        if record:
            record.write({
                'name': '更新的名称',
                'amount': 1500,
            })
        
        # 2. 批量更新
        records = self.search([('state', '=', 'draft')])
        records.write({
            'state': 'confirmed',
            'amount': 2000,
        })
        
        # 3. 条件更新
        self.search([('amount', '<', 1000)]).write({
            'amount': 1000,
        })
        
        # 4. 使用with_context更新
        records = self.search([])
        records.with_context(force_update=True).write({
            'name': '强制更新',
        })
        
        # 5. 使用sudo更新
        records = self.search([])
        records.sudo().write({
            'name': '管理员更新',
        })
        
        # 6. 使用env.cr执行SQL更新
        self.env.cr.execute("""
            UPDATE data_update_examples 
            SET amount = amount * 1.1 
            WHERE state = 'confirmed'
        """)
        
        return {
            'single': record,
            'batch': records,
            'conditional': records,
            'with_context': records,
            'sudo': records,
        }
```

```python
class DataDeleteExamples(models.Model):
    _name = 'data.delete.examples'
    
    name = fields.Char('名称')

    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '完成'),
    ], string='状态')
    
    def delete_examples(self):
        # 1. 单条记录删除
        record = self.search([], limit=1)
        if record:
            record.unlink()
        
        # 2. 批量删除
        records = self.search([('state', '=', 'draft')])
        records.unlink()
        
        # 3. 条件删除
        self.search([('amount', '<', 100)]).unlink()
        
        # 4. 使用sudo删除
        records = self.search([])
        records.sudo().unlink()
        
        # 5. 使用env.cr执行SQL删除
        self.env.cr.execute("""
            DELETE FROM data_delete_examples 
            WHERE state = 'cancelled'
        """)
        
        # 6. 软删除（设置active=False）
        records = self.search([])
        records.write({'active': False})
        
        return {
            'single': record,
            'batch': records,
            'conditional': records,
            'sudo': records,
            'soft_delete': records,
        }
```

```python
class AdvancedDeleteExamples(models.Model):
    _name = 'advanced.delete.examples'
    
    def advanced_delete_examples(self):
        # 1. 使用filtered选择性删除
        records = self.search([])
        old_records = records.filtered(lambda r: r.create_date < fields.Date.today() - timedelta(days=30))

        # 2. 使用mapped删除关联记录
        records = self.search([])

        partners.filtered(lambda p: not p.active).unlink()
        
        # 3. 使用with_context删除

        records.with_context(force_delete=True).unlink()
        
        # 4. 使用env.ref删除引用记录

        records.unlink()
        
        # 6. 使用env.company删除公司相关记录
        company = self.env.company
        records = self.search([('company_id', '=', company.id)])
        records.unlink()
        
        return {
            'filtered': old_records,
            'mapped': partners,
            'with_context': records,
            'with_ref': partner,
            'with_user': records,
            'with_company': records,
        }
```

```python
class DataStatisticsExamples(models.Model):
    _name = 'data.statistics.examples'
    
    name = fields.Char('名称')
    amount = fields.Float('金额')
    date = fields.Date('日期')
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '完成'),
    ], string='状态')
    
    def statistics_examples(self):
        # 1. 计数统计
        total_count = self.search_count([])
        draft_count = self.search_count([('state', '=', 'draft')])
        
        # 2. 求和统计
        records = self.search([])
        total_amount = sum(records.mapped('amount'))
        
        # 3. 平均值统计
        avg_amount = total_amount / len(records) if records else 0
        
        # 4. 最大值统计
        max_amount = max(records.mapped('amount')) if records else 0
        
        # 5. 最小值统计
        min_amount = min(records.mapped('amount')) if records else 0
        
        # 6. 分组统计
        grouped_data = self.read_group(
            domain=[],
            fields=['amount:sum', 'state'],
            groupby=['state']
        )
        
        # 7. 日期统计
        date_grouped = self.read_group(
            domain=[],
            fields=['amount:sum', 'date'],
            groupby=['date:day']
        )
        
        # 8. 使用SQL统计
        self.env.cr.execute("""
            SELECT state, COUNT(*), SUM(amount), AVG(amount)
            FROM data_statistics_examples
            GROUP BY state
        """)
        sql_stats = self.env.cr.fetchall()
        
        return {
            'total_count': total_count,
            'draft_count': draft_count,
            'total_amount': total_amount,
            'avg_amount': avg_amount,
            'max_amount': max_amount,
            'min_amount': min_amount,
            'grouped_data': grouped_data,
            'date_grouped': date_grouped,
            'sql_stats': sql_stats,
        }
```

```python
class AdvancedStatisticsExamples(models.Model):
    _name = 'advanced.statistics.examples'
    
    def advanced_statistics_examples(self):
        # 1. 使用mapped进行复杂统计
        records = self.search([])
        amounts = records.mapped('amount')
        
        # 计算标准差
        mean = sum(amounts) / len(amounts) if amounts else 0
        variance = sum((x - mean) ** 2 for x in amounts) / len(amounts) if amounts else 0
        std_dev = variance ** 0.5
        
        # 2. 使用filtered进行条件统计
        high_amount_records = records.filtered(lambda r: r.amount > 1000)
        high_amount_count = len(high_amount_records)
        high_amount_total = sum(high_amount_records.mapped('amount'))
        
        # 3. 使用sorted进行排序统计
        sorted_records = records.sorted(lambda r: r.amount)
        median_amount = sorted_records[len(sorted_records)//2].amount if sorted_records else 0
        
        # 4. 使用with_context进行上下文统计
        context_stats = self.with_context(statistics_mode=True).read_group(
            domain=[],
            fields=['amount:sum'],
            groupby=['state']
        )
        
        # 5. 使用sudo进行权限统计
        sudo_stats = self.sudo().read_group(
            domain=[],
            fields=['amount:sum'],
            groupby=['state']
        )
        
        # 6. 使用env.cr进行复杂SQL统计
        self.env.cr.execute("""
            WITH stats AS (
                SELECT 
                    state,
                    COUNT(*) as count,
                    SUM(amount) as total,
                    AVG(amount) as average,
                    MIN(amount) as minimum,
                    MAX(amount) as maximum
                FROM advanced_statistics_examples
                GROUP BY state
            )
            SELECT * FROM stats ORDER BY total DESC
        """)
        complex_stats = self.env.cr.fetchall()
        
        return {
            'std_dev': std_dev,
            'high_amount_count': high_amount_count,
            'high_amount_total': high_amount_total,
            'median_amount': median_amount,
            'context_stats': context_stats,
            'sudo_stats': sudo_stats,
            'complex_stats': complex_stats,
        }
```

```python
class BatchOperationExamples(models.Model):
    _name = 'batch.operation.examples'
    
    def batch_operation_examples(self):
        # 1. 批量创建
        vals_list = []
        for i in range(1000):
            vals_list.append({
                'name': f'记录{i}',
                'amount': i * 10,
            })
        records = self.create(vals_list)
        
        # 2. 批量更新
        records.write({'state': 'confirmed'})
        
        # 3. 批量删除
        records.unlink()
        
        # 4. 使用env.cr执行批量SQL
        self.env.cr.execute("""
            INSERT INTO batch_operation_examples (name, amount, create_date, write_date)
            SELECT 
                '批量记录' || generate_series,
                generate_series * 10,
                NOW(),
                NOW()
            FROM generate_series(1, 1000)
        """)
        
        # 5. 使用env.cr执行批量更新
        self.env.cr.execute("""
            UPDATE batch_operation_examples 
            SET amount = amount * 1.1 
            WHERE amount > 1000
        """)
        
        # 6. 使用env.cr执行批量删除
        self.env.cr.execute("""
            DELETE FROM batch_operation_examples 
            WHERE amount < 100
        """)
        
        return {
            'created': records,
            'updated': records,
            'deleted': records,
        }
```
