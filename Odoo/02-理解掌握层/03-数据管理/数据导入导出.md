class DataExportExamples(models.Model):
    _name = 'data.export.examples'
    
    name = fields.Char('名称')
    amount = fields.Float('金额')
    date = fields.Date('日期')
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '完成'),
    ], string='状态')
    partner_id = fields.Many2one('res.partner', string='客户')
    
    def export_data(self, export_format='csv'):
        """导出数据"""
        # 获取要导出的记录
        records = self.search([])
        
        if export_format == 'csv':
            return self._export_csv(records)
        elif export_format == 'excel':
            return self._export_excel(records)
        elif export_format == 'json':
            return self._export_json(records)
        else:
            raise ValueError(f'不支持的导出格式: {export_format}')
    
    def _export_csv(self, records):
        """导出CSV格式"""
        import csv
        import io
        
        output = io.StringIO()
        writer = csv.writer(output)
        
        # 写入标题行
        headers = ['ID', '名称', '金额', '日期', '状态', '客户']
        writer.writerow(headers)
        
        # 写入数据行
        for record in records:
            row = [
                record.id,
                record.name or '',
                record.amount or 0,
                record.date.strftime('%Y-%m-%d') if record.date else '',
                dict(record._fields['state'].selection)[record.state] if record.state else '',
                record.partner_id.name if record.partner_id else '',
            ]
            writer.writerow(row)
        
        output.seek(0)
        return output.getvalue()
    
    def _export_excel(self, records):
        """导出Excel格式"""
        try:
            import xlsxwriter
            import io
            
            output = io.BytesIO()
            workbook = xlsxwriter.Workbook(output)
            worksheet = workbook.add_worksheet('数据导出')
            
            # 设置标题格式
            header_format = workbook.add_format({
                'bold': True,
                'bg_color': '#CCCCCC',
                'border': 1
            })
            
            # 设置数据格式
            data_format = workbook.add_format({
                'border': 1
            })
            
            # 写入标题行
            headers = ['ID', '名称', '金额', '日期', '状态', '客户']
            for col, header in enumerate(headers):
                worksheet.write(0, col, header, header_format)
            
            # 写入数据行
            for row, record in enumerate(records, 1):
                worksheet.write(row, 0, record.id, data_format)
                worksheet.write(row, 1, record.name or '', data_format)
                worksheet.write(row, 2, record.amount or 0, data_format)
                worksheet.write(row, 3, record.date.strftime('%Y-%m-%d') if record.date else '', data_format)
                worksheet.write(row, 4, dict(record._fields['state'].selection)[record.state] if record.state else '', data_format)
                worksheet.write(row, 5, record.partner_id.name if record.partner_id else '', data_format)
            
            # 调整列宽
            worksheet.set_column('A:A', 10)
            worksheet.set_column('B:B', 20)
            worksheet.set_column('C:C', 15)
            worksheet.set_column('D:D', 15)
            worksheet.set_column('E:E', 15)
            worksheet.set_column('F:F', 20)
            
            workbook.close()
            output.seek(0)
            return output.getvalue()
            
        except ImportError:
            raise ImportError('需要安装xlsxwriter库: pip install xlsxwriter')
    
    def _export_json(self, records):
        """导出JSON格式"""
        import json
        
        data = []
        for record in records:
            data.append({
                'id': record.id,
                'name': record.name,
                'amount': record.amount,
                'date': record.date.isoformat() if record.date else None,
                'state': record.state,
                'partner_id': record.partner_id.id if record.partner_id else None,
                'partner_name': record.partner_id.name if record.partner_id else None,
            })
        
        return json.dumps(data, ensure_ascii=False, indent=2)

class CustomExportExamples(models.Model):
    _name = 'custom.export.examples'
    
    def custom_export(self, domain=None, fields=None, format_type='csv'):
        """自定义导出"""
        # 设置默认参数
        if domain is None:
            domain = []
        if fields is None:
            fields = ['name', 'amount', 'date', 'state']
        
        # 查询数据
        records = self.search(domain)
        
        if format_type == 'csv':
            return self._custom_export_csv(records, fields)
        elif format_type == 'excel':
            return self._custom_export_excel(records, fields)
        else:
            raise ValueError(f'不支持的格式: {format_type}')
    
    def _custom_export_csv(self, records, fields):
        """自定义CSV导出"""
        import csv
        import io
        
        output = io.StringIO()
        writer = csv.writer(output)
        
        # 写入标题行
        field_labels = []
        for field_name in fields:
            if field_name in self._fields:
                field_labels.append(self._fields[field_name].string)
            else:
                field_labels.append(field_name)
        writer.writerow(field_labels)
        
        # 写入数据行
        for record in records:
            row = []
            for field_name in fields:
                if field_name in self._fields:
                    field = self._fields[field_name]
                    value = getattr(record, field_name)
                    
                    if field.type == 'many2one':
                        row.append(value.name if value else '')
                    elif field.type == 'selection':
                        row.append(dict(field.selection)[value] if value else '')
                    elif field.type == 'date':
                        row.append(value.strftime('%Y-%m-%d') if value else '')
                    elif field.type == 'datetime':
                        row.append(value.strftime('%Y-%m-%d %H:%M:%S') if value else '')
                    else:
                        row.append(str(value) if value else '')
                else:
                    row.append('')
            writer.writerow(row)
        
        output.seek(0)
        return output.getvalue()
    
    def _custom_export_excel(self, records, fields):
        """自定义Excel导出"""
        try:
            import xlsxwriter
            import io
            
            output = io.BytesIO()
            workbook = xlsxwriter.Workbook(output)
            worksheet = workbook.add_worksheet('自定义导出')
            
            # 设置格式
            header_format = workbook.add_format({
                'bold': True,
                'bg_color': '#CCCCCC',
                'border': 1
            })
            
            data_format = workbook.add_format({
                'border': 1
            })
            
            # 写入标题行
            field_labels = []
            for field_name in fields:
                if field_name in self._fields:
                    field_labels.append(self._fields[field_name].string)
                else:
                    field_labels.append(field_name)
            
            for col, label in enumerate(field_labels):
                worksheet.write(0, col, label, header_format)
            
            # 写入数据行
            for row, record in enumerate(records, 1):
                for col, field_name in enumerate(fields):
                    if field_name in self._fields:
                        field = self._fields[field_name]
                        value = getattr(record, field_name)
                        
                        if field.type == 'many2one':
                            cell_value = value.name if value else ''
                        elif field.type == 'selection':
                            cell_value = dict(field.selection)[value] if value else ''
                        elif field.type == 'date':
                            cell_value = value.strftime('%Y-%m-%d') if value else ''
                        elif field.type == 'datetime':
                            cell_value = value.strftime('%Y-%m-%d %H:%M:%S') if value else ''
                        else:
                            cell_value = str(value) if value else ''
                        
                        worksheet.write(row, col, cell_value, data_format)
            
            workbook.close()
            output.seek(0)
            return output.getvalue()
            
        except ImportError:
            raise ImportError('需要安装xlsxwriter库: pip install xlsxwriter')

class DataImportExamples(models.Model):
    _name = 'data.import.examples'
    
    name = fields.Char('名称')
    amount = fields.Float('金额')
    date = fields.Date('日期')
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '完成'),
    ], string='状态')
    partner_id = fields.Many2one('res.partner', string='客户')
    
    def import_data(self, file_content, file_format='csv'):
        """导入数据"""
        if file_format == 'csv':
            return self._import_csv(file_content)
        elif file_format == 'excel':
            return self._import_excel(file_content)
        elif file_format == 'json':
            return self._import_json(file_content)
        else:
            raise ValueError(f'不支持的导入格式: {file_format}')
    
    def _import_csv(self, file_content):
        """导入CSV格式"""
        import csv
        import io
        
        result = {
            'success_count': 0,
            'error_count': 0,
            'errors': []
        }
        
        try:
            # 解析CSV内容
            csv_reader = csv.reader(io.StringIO(file_content))
            headers = next(csv_reader)  # 跳过标题行
            
            # 验证标题
            expected_headers = ['名称', '金额', '日期', '状态', '客户']
            if headers != expected_headers:
                result['errors'].append(f'标题不匹配，期望: {expected_headers}, 实际: {headers}')
                return result
            
            # 处理数据行
            for row_num, row in enumerate(csv_reader, start=2):
                try:
                    if len(row) >= 5:
                        # 解析数据
                        name = row[0].strip()
                        amount = float(row[1]) if row[1] else 0
                        date_str = row[2].strip()
                        state_str = row[3].strip()
                        partner_name = row[4].strip()
                        
                        # 验证必填字段
                        if not name:
                            result['errors'].append(f'第{row_num}行: 名称不能为空')
                            result['error_count'] += 1
                            continue
                        
                        # 处理日期
                        date = None
                        if date_str:
                            try:
                                from datetime import datetime
                                date = datetime.strptime(date_str, '%Y-%m-%d').date()
                            except ValueError:
                                result['errors'].append(f'第{row_num}行: 日期格式错误')
                                result['error_count'] += 1
                                continue
                        
                        # 处理状态
                        state = None
                        if state_str:
                            state_mapping = {
                                '草稿': 'draft',
                                '已确认': 'confirmed',
                                '完成': 'done',
                            }
                            state = state_mapping.get(state_str)
                            if not state:
                                result['errors'].append(f'第{row_num}行: 状态值无效')
                                result['error_count'] += 1
                                continue
                        
                        # 处理客户
                        partner_id = None
                        if partner_name:
                            partner = self.env['res.partner'].search([('name', '=', partner_name)], limit=1)
                            if partner:
                                partner_id = partner.id
                            else:
                                result['errors'].append(f'第{row_num}行: 客户不存在')
                                result['error_count'] += 1
                                continue
                        
                        # 创建记录
                        self.create({
                            'name': name,
                            'amount': amount,
                            'date': date,
                            'state': state,
                            'partner_id': partner_id,
                        })
                        result['success_count'] += 1
                    else:
                        result['errors'].append(f'第{row_num}行: 数据不完整')
                        result['error_count'] += 1
                        
                except Exception as e:
                    result['errors'].append(f'第{row_num}行: {str(e)}')
                    result['error_count'] += 1
            
        except Exception as e:
            result['errors'].append(f'文件解析错误: {str(e)}')
            result['error_count'] += 1
        
        return result
    
    def _import_excel(self, file_content):
        """导入Excel格式"""
        try:
            import openpyxl
            import io
            
            result = {
                'success_count': 0,
                'error_count': 0,
                'errors': []
            }
            
            # 解析Excel内容
            workbook = openpyxl.load_workbook(io.BytesIO(file_content))
            worksheet = workbook.active
            
            # 获取标题行
            headers = []
            for cell in worksheet[1]:
                headers.append(cell.value)
            
            # 验证标题
            expected_headers = ['名称', '金额', '日期', '状态', '客户']
            if headers != expected_headers:
                result['errors'].append(f'标题不匹配，期望: {expected_headers}, 实际: {headers}')
                return result
            
            # 处理数据行
            for row_num, row in enumerate(worksheet.iter_rows(min_row=2), start=2):
                try:
                    if len(row) >= 5:
                        # 解析数据
                        name = str(row[0].value).strip() if row[0].value else ''
                        amount = float(row[1].value) if row[1].value else 0
                        date_value = row[2].value
                        state_str = str(row[3].value).strip() if row[3].value else ''
                        partner_name = str(row[4].value).strip() if row[4].value else ''
                        
                        # 验证必填字段
                        if not name:
                            result['errors'].append(f'第{row_num}行: 名称不能为空')
                            result['error_count'] += 1
                            continue
                        
                        # 处理日期
                        date = None
                        if date_value:
                            if hasattr(date_value, 'date'):
                                date = date_value.date()
                            else:
                                try:
                                    from datetime import datetime
                                    date = datetime.strptime(str(date_value), '%Y-%m-%d').date()
                                except ValueError:
                                    result['errors'].append(f'第{row_num}行: 日期格式错误')
                                    result['error_count'] += 1
                                    continue
                        
                        # 处理状态
                        state = None
                        if state_str:
                            state_mapping = {
                                '草稿': 'draft',
                                '已确认': 'confirmed',
                                '完成': 'done',
                            }
                            state = state_mapping.get(state_str)
                            if not state:
                                result['errors'].append(f'第{row_num}行: 状态值无效')
                                result['error_count'] += 1
                                continue
                        
                        # 处理客户
                        partner_id = None
                        if partner_name:
                            partner = self.env['res.partner'].search([('name', '=', partner_name)], limit=1)
                            if partner:
                                partner_id = partner.id
                            else:
                                result['errors'].append(f'第{row_num}行: 客户不存在')
                                result['error_count'] += 1
                                continue
                        
                        # 创建记录
                        self.create({
                            'name': name,
                            'amount': amount,
                            'date': date,
                            'state': state,
                            'partner_id': partner_id,
                        })
                        result['success_count'] += 1
                    else:
                        result['errors'].append(f'第{row_num}行: 数据不完整')
                        result['error_count'] += 1
                        
                except Exception as e:
                    result['errors'].append(f'第{row_num}行: {str(e)}')
                    result['error_count'] += 1
            
        except ImportError:
            raise ImportError('需要安装openpyxl库: pip install openpyxl')
        except Exception as e:
            result['errors'].append(f'文件解析错误: {str(e)}')
            result['error_count'] += 1
        
        return result
    
    def _import_json(self, file_content):
        """导入JSON格式"""
        import json
        
        result = {
            'success_count': 0,
            'error_count': 0,
            'errors': []
        }
        
        try:
            # 解析JSON内容
            data = json.loads(file_content)
            
            if not isinstance(data, list):
                result['errors'].append('JSON数据必须是数组格式')
                return result
            
            # 处理数据
            for row_num, item in enumerate(data, start=1):
                try:
                    # 验证必填字段
                    if not item.get('name'):
                        result['errors'].append(f'第{row_num}行: 名称不能为空')
                        result['error_count'] += 1
                        continue
                    
                    # 处理日期
                    date = None
                    if item.get('date'):
                        try:
                            from datetime import datetime
                            date = datetime.strptime(item['date'], '%Y-%m-%d').date()
                        except ValueError:
                            result['errors'].append(f'第{row_num}行: 日期格式错误')
                            result['error_count'] += 1
                            continue
                    
                    # 处理状态
                    state = None
                    if item.get('state'):
                        state_mapping = {
                            'draft': 'draft',
                            'confirmed': 'confirmed',
                            'done': 'done',
                        }
                        state = state_mapping.get(item['state'])
                        if not state:
                            result['errors'].append(f'第{row_num}行: 状态值无效')
                            result['error_count'] += 1
                            continue
                    
                    # 处理客户
                    partner_id = None
                    if item.get('partner_id'):
                        partner = self.env['res.partner'].browse(item['partner_id'])
                        if partner.exists():
                            partner_id = partner.id
                        else:
                            result['errors'].append(f'第{row_num}行: 客户ID不存在')
                            result['error_count'] += 1
                            continue
                    
                    # 创建记录
                    self.create({
                        'name': item['name'],
                        'amount': item.get('amount', 0),
                        'date': date,
                        'state': state,
                        'partner_id': partner_id,
                    })
                    result['success_count'] += 1
                    
                except Exception as e:
                    result['errors'].append(f'第{row_num}行: {str(e)}')
                    result['error_count'] += 1
            
        except json.JSONDecodeError as e:
            result['errors'].append(f'JSON解析错误: {str(e)}')
            result['error_count'] += 1
        except Exception as e:
            result['errors'].append(f'导入错误: {str(e)}')
            result['error_count'] += 1
        
        return result

class CustomImportExamples(models.Model):
    _name = 'custom.import.examples'
    
    def custom_import(self, file_content, file_format='csv', field_mapping=None):
        """自定义导入"""
        if field_mapping is None:
            field_mapping = {
                'name': '名称',
                'amount': '金额',
                'date': '日期',
                'state': '状态',
                'partner_id': '客户',
            }
        
        if file_format == 'csv':
            return self._custom_import_csv(file_content, field_mapping)
        elif file_format == 'excel':
            return self._custom_import_excel(file_content, field_mapping)
        else:
            raise ValueError(f'不支持的格式: {file_format}')
    
    def _custom_import_csv(self, file_content, field_mapping):
        """自定义CSV导入"""
        import csv
        import io
        
        result = {
            'success_count': 0,
            'error_count': 0,
            'errors': []
        }
        
        try:
            # 解析CSV内容
            csv_reader = csv.reader(io.StringIO(file_content))
            headers = next(csv_reader)  # 跳过标题行
            
            # 创建字段映射
            field_index_map = {}
            for field_name, header_name in field_mapping.items():
                if header_name in headers:
                    field_index_map[field_name] = headers.index(header_name)
                else:
                    result['errors'].append(f'缺少字段: {header_name}')
                    return result
            
            # 处理数据行
            for row_num, row in enumerate(csv_reader, start=2):
                try:
                    if len(row) >= len(headers):
                        # 解析数据
                        data = {}
                        for field_name, index in field_index_map.items():
                            if index < len(row):
                                data[field_name] = row[index].strip()
                            else:
                                data[field_name] = ''
                        
                        # 验证和转换数据
                        validated_data = self._validate_import_data(data, row_num)
                        if validated_data:
                            # 创建记录
                            self.create(validated_data)
                            result['success_count'] += 1
                        else:
                            result['error_count'] += 1
                    else:
                        result['errors'].append(f'第{row_num}行: 数据不完整')
                        result['error_count'] += 1
                        
                except Exception as e:
                    result['errors'].append(f'第{row_num}行: {str(e)}')
                    result['error_count'] += 1
            
        except Exception as e:
            result['errors'].append(f'文件解析错误: {str(e)}')
            result['error_count'] += 1
        
        return result
    
    def _validate_import_data(self, data, row_num):
        """验证导入数据"""
        errors = []
        validated_data = {}
        
        # 验证名称
        if not data.get('name'):
            errors.append(f'第{row_num}行: 名称不能为空')
        else:
            validated_data['name'] = data['name']
        
        # 验证金额
        try:
            amount = float(data.get('amount', 0))
            validated_data['amount'] = amount
        except ValueError:
            errors.append(f'第{row_num}行: 金额格式错误')
        
        # 验证日期
        if data.get('date'):
            try:
                from datetime import datetime
                date = datetime.strptime(data['date'], '%Y-%m-%d').date()
                validated_data['date'] = date
            except ValueError:
                errors.append(f'第{row_num}行: 日期格式错误')
        
        # 验证状态
        if data.get('state'):
            state_mapping = {
                '草稿': 'draft',
                '已确认': 'confirmed',
                '完成': 'done',
            }
            state = state_mapping.get(data['state'])
            if state:
                validated_data['state'] = state
            else:
                errors.append(f'第{row_num}行: 状态值无效')
        
        # 验证客户
        if data.get('partner_id'):
            partner = self.env['res.partner'].search([('name', '=', data['partner_id'])], limit=1)
            if partner:
                validated_data['partner_id'] = partner.id
            else:
                errors.append(f'第{row_num}行: 客户不存在')
        
        # 如果有错误，记录并返回None
        if errors:
            self.env['import.error.log'].create({
                'row_number': row_num,
                'errors': '\n'.join(errors),
                'data': str(data),
            })
            return None
        
        return validated_data
```

class BatchImportExportExamples(models.Model):
    _name = 'batch.import.export.examples'
    
    def batch_export(self, domain=None, batch_size=1000):
        """批量导出"""
        if domain is None:
            domain = []
        
        # 获取总记录数
        total_count = self.search_count(domain)
        
        # 分批处理
        batches = []
        offset = 0
        
        while offset < total_count:
            # 获取当前批次数据
            records = self.search(domain, limit=batch_size, offset=offset)
            
            # 导出当前批次
            batch_data = self._export_batch(records)
            batches.append(batch_data)
            
            offset += batch_size
        
        return {
            'total_count': total_count,
            'batch_count': len(batches),
            'batches': batches,
        }
    
    def _export_batch(self, records):
        """导出批次数据"""
        data = []
        for record in records:
            data.append({
                'id': record.id,
                'name': record.name,
                'amount': record.amount,
                'date': record.date.isoformat() if record.date else None,
                'state': record.state,
                'partner_id': record.partner_id.id if record.partner_id else None,
            })
        return data
    
    def batch_import(self, data_list, batch_size=100):
        """批量导入"""
        result = {
            'success_count': 0,
            'error_count': 0,
            'errors': []
        }
        
        # 分批处理
        for i in range(0, len(data_list), batch_size):
            batch = data_list[i:i + batch_size]
            
            try:
                # 批量创建
                records = self.create(batch)
                result['success_count'] += len(records)
            except Exception as e:
                result['error_count'] += len(batch)
                result['errors'].append(f'批次{i//batch_size + 1}: {str(e)}')
        
        return result
```
