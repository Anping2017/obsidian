class DataBackupExamples(models.Model):
    _name = 'data.backup.examples'
    
    def standard_backup(self, backup_type='full'):
        """标准备份功能"""
        if backup_type == 'full':
            return self._full_backup()
        elif backup_type == 'incremental':
            return self._incremental_backup()
        elif backup_type == 'differential':
            return self._differential_backup()
        else:
            raise ValueError(f'不支持的备份类型: {backup_type}')
    
    def _full_backup(self):
        """完整备份"""
        import os
        import shutil
        from datetime import datetime
        
        # 创建备份目录
        backup_dir = os.path.join(os.path.dirname(__file__), 'backups')
        os.makedirs(backup_dir, exist_ok=True)
        
        # 生成备份文件名
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        backup_filename = f'full_backup_{timestamp}.sql'
        backup_path = os.path.join(backup_dir, backup_filename)
        
        # 执行数据库备份
        db_name = self.env.cr.dbname
        backup_command = f"pg_dump -h localhost -U {self.env.cr.dbuser} -d {db_name} > {backup_path}"
        
        try:
            import subprocess
            result = subprocess.run(backup_command, shell=True, capture_output=True, text=True)
            
            if result.returncode == 0:
                return {
                    'status': 'success',
                    'message': '完整备份成功',
                    'backup_path': backup_path,
                    'backup_size': os.path.getsize(backup_path),
                }
            else:
                return {
                    'status': 'error',
                    'message': f'备份失败: {result.stderr}',
                }
        except Exception as e:
            return {
                'status': 'error',
                'message': f'备份异常: {str(e)}',
            }
    
    def _incremental_backup(self):
        """增量备份"""
        import os
        from datetime import datetime, timedelta
        
        # 创建备份目录
        backup_dir = os.path.join(os.path.dirname(__file__), 'backups')
        os.makedirs(backup_dir, exist_ok=True)
        
        # 生成备份文件名
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        backup_filename = f'incremental_backup_{timestamp}.sql'
        backup_path = os.path.join(backup_dir, backup_filename)
        
        # 获取上次备份时间
        last_backup_time = self._get_last_backup_time()
        
        # 执行增量备份
        db_name = self.env.cr.dbname
        backup_command = f"pg_dump -h localhost -U {self.env.cr.dbuser} -d {db_name} --since='{last_backup_time}' > {backup_path}"
        
        try:
            import subprocess
            result = subprocess.run(backup_command, shell=True, capture_output=True, text=True)
            
            if result.returncode == 0:
                return {
                    'status': 'success',
                    'message': '增量备份成功',
                    'backup_path': backup_path,
                    'backup_size': os.path.getsize(backup_path),
                    'last_backup_time': last_backup_time,
                }
            else:
                return {
                    'status': 'error',
                    'message': f'增量备份失败: {result.stderr}',
                }
        except Exception as e:
            return {
                'status': 'error',
                'message': f'增量备份异常: {str(e)}',
            }
    
    def _differential_backup(self):
        """差异备份"""
        import os
        from datetime import datetime
        
        # 创建备份目录
        backup_dir = os.path.join(os.path.dirname(__file__), 'backups')
        os.makedirs(backup_dir, exist_ok=True)
        
        # 生成备份文件名
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        backup_filename = f'differential_backup_{timestamp}.sql'
        backup_path = os.path.join(backup_dir, backup_filename)
        
        # 获取上次完整备份时间
        last_full_backup_time = self._get_last_full_backup_time()
        
        # 执行差异备份
        db_name = self.env.cr.dbname
        backup_command = f"pg_dump -h localhost -U {self.env.cr.dbuser} -d {db_name} --since='{last_full_backup_time}' > {backup_path}"
        
        try:
            import subprocess
            result = subprocess.run(backup_command, shell=True, capture_output=True, text=True)
            
            if result.returncode == 0:
                return {
                    'status': 'success',
                    'message': '差异备份成功',
                    'backup_path': backup_path,
                    'backup_size': os.path.getsize(backup_path),
                    'last_full_backup_time': last_full_backup_time,
                }
            else:
                return {
                    'status': 'error',
                    'message': f'差异备份失败: {result.stderr}',

            return {
                'status': 'error',
                'message': f'差异备份异常: {str(e)}',
            }
    
    def _get_last_backup_time(self):
        """获取上次备份时间"""
        # 从备份记录中获取上次备份时间
        last_backup = self.env['backup.record'].search([], order='create_date desc', limit=1)
        if last_backup:
            return last_backup.create_date.strftime('%Y-%m-%d %H:%M:%S')
        else:
            # 如果没有备份记录，返回7天前
            from datetime import datetime, timedelta
            return (datetime.now() - timedelta(days=7)).strftime('%Y-%m-%d %H:%M:%S')
    
    def _get_last_full_backup_time(self):
        """获取上次完整备份时间"""
        # 从备份记录中获取上次完整备份时间
        last_full_backup = self.env['backup.record'].search([
            ('backup_type', '=', 'full')
        ], order='create_date desc', limit=1)
        
        if last_full_backup:
            return last_full_backup.create_date.strftime('%Y-%m-%d %H:%M:%S')
        else:
            # 如果没有完整备份记录，返回30天前
            from datetime import datetime, timedelta
            return (datetime.now() - timedelta(days=30)).strftime('%Y-%m-%d %H:%M:%S')

### 自定义备份功能
```python
class CustomBackupExamples(models.Model):
    _name = 'custom.backup.examples'
    
    def custom_backup(self, models=None, include_attachments=True, compress=True):
        """自定义备份"""
        if models is None:
            models = ['res.partner', 'res.users', 'product.product']
        
        backup_data = {
            'models': {},
            'attachments': [],
            'metadata': {
                'backup_time': fields.Datetime.now(),
                'models_count': len(models),
                'include_attachments': include_attachments,
                'compress': compress,
            }
        }
        
        # 备份指定模型数据
        for model_name in models:
            if model_name in self.env:
                model = self.env[model_name]
                records = model.search([])
                
                model_data = []
                for record in records:
                    record_data = record.read()
                    model_data.append(record_data[0] if record_data else {})
                
                backup_data['models'][model_name] = model_data
        
        # 备份附件
        if include_attachments:
            attachments = self.env['ir.attachment'].search([])
            for attachment in attachments:
                attachment_data = {
                    'name': attachment.name,
                    'datas': attachment.datas,
                    'mimetype': attachment.mimetype,
                    'res_model': attachment.res_model,
                    'res_id': attachment.res_id,
                }
                backup_data['attachments'].append(attachment_data)
        
        # 保存备份数据
        backup_filename = self._save_backup_data(backup_data, compress)
        
        return {
            'status': 'success',
            'message': '自定义备份成功',
            'backup_filename': backup_filename,
            'models_backed_up': list(backup_data['models'].keys()),
            'attachments_count': len(backup_data['attachments']),
        }
    
    def _save_backup_data(self, backup_data, compress=True):
        """保存备份数据"""
        import json
        import os
        from datetime import datetime
        
        # 创建备份目录
        backup_dir = os.path.join(os.path.dirname(__file__), 'backups')
        os.makedirs(backup_dir, exist_ok=True)
        
        # 生成备份文件名
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        if compress:
            backup_filename = f'custom_backup_{timestamp}.json.gz'
        else:
            backup_filename = f'custom_backup_{timestamp}.json'
        
        backup_path = os.path.join(backup_dir, backup_filename)
        
        # 保存备份数据
        if compress:
            import gzip
            with gzip.open(backup_path, 'wt', encoding='utf-8') as f:
                json.dump(backup_data, f, ensure_ascii=False, indent=2, default=str)
        else:
            with open(backup_path, 'w', encoding='utf-8') as f:
                json.dump(backup_data, f, ensure_ascii=False, indent=2, default=str)
        
        return backup_filename
```

class CustomRestoreExamples(models.Model):
    _name = 'custom.restore.examples'
    
    def custom_restore(self, backup_filename, models=None, include_attachments=True):
        """自定义恢复"""
        import json
        import os
        import gzip
        
        # 构建备份文件路径
        backup_dir = os.path.join(os.path.dirname(__file__), 'backups')
        backup_path = os.path.join(backup_dir, backup_filename)
        
        if not os.path.exists(backup_path):
            return {
                'status': 'error',
                'message': f'备份文件不存在: {backup_path}',
            }
        
        try:
            # 读取备份数据
            if backup_filename.endswith('.gz'):
                with gzip.open(backup_path, 'rt', encoding='utf-8') as f:
                    backup_data = json.load(f)
            else:
                with open(backup_path, 'r', encoding='utf-8') as f:
                    backup_data = json.load(f)
            
            # 恢复模型数据
            restored_models = []
            if models is None:
                models = list(backup_data['models'].keys())
            
            for model_name in models:
                if model_name in backup_data['models']:
                    model_data = backup_data['models'][model_name]
                    restored_count = self._restore_model_data(model_name, model_data)
                    restored_models.append({
                        'model': model_name,
                        'count': restored_count
                    })
            
            # 恢复附件
            restored_attachments = 0
            if include_attachments and 'attachments' in backup_data:
                restored_attachments = self._restore_attachments(backup_data['attachments'])
            
            return {
                'status': 'success',
                'message': '自定义恢复成功',
                'restored_models': restored_models,
                'restored_attachments': restored_attachments,
                'backup_time': backup_data.get('metadata', {}).get('backup_time'),
            }
            
        except Exception as e:
            return {
                'status': 'error',
                'message': f'恢复异常: {str(e)}',
            }
    
    def _restore_model_data(self, model_name, model_data):
        """恢复模型数据"""
        if model_name not in self.env:
            return 0
        
        model = self.env[model_name]
        restored_count = 0
        
        for record_data in model_data:
            try:
                # 移除ID字段，让系统自动生成
                if 'id' in record_data:
                    del record_data['id']
                
                # 创建记录
                model.create(record_data)
                restored_count += 1
            except Exception as e:
                # 记录错误但继续处理其他记录
                continue
        
        return restored_count
    
    def _restore_attachments(self, attachments_data):
        """恢复附件"""
        restored_count = 0
        
        for attachment_data in attachments_data:
            try:
                # 创建附件记录
                self.env['ir.attachment'].create({
                    'name': attachment_data['name'],
                    'datas': attachment_data['datas'],
                    'mimetype': attachment_data['mimetype'],
                    'res_model': attachment_data['res_model'],
                    'res_id': attachment_data['res_id'],
                })
                restored_count += 1
            except Exception as e:
                # 记录错误但继续处理其他附件
                continue
        
        return restored_count
```

class AutoBackupExamples(models.Model):
    _name = 'auto.backup.examples'
    
    def schedule_backup(self):
        """定时备份"""
        # 检查备份配置
        backup_config = self.env['backup.config'].search([('active', '=', True)])
        
        for config in backup_config:
            if self._should_backup(config):
                self._execute_backup(config)
    
    def _should_backup(self, config):
        """判断是否应该备份"""
        from datetime import datetime, timedelta
        
        # 获取上次备份时间
        last_backup = self.env['backup.record'].search([
            ('config_id', '=', config.id)
        ], order='create_date desc', limit=1)
        
        if not last_backup:
            return True
        
        # 计算时间间隔
        time_since_last_backup = datetime.now() - last_backup.create_date
        
        if config.backup_frequency == 'daily':
            return time_since_last_backup.days >= 1
        elif config.backup_frequency == 'weekly':
            return time_since_last_backup.days >= 7
        elif config.backup_frequency == 'monthly':
            return time_since_last_backup.days >= 30
        else:
            return False
    
    def _execute_backup(self, config):
        """执行备份"""
        try:
            # 执行备份
            if config.backup_type == 'full':
                result = self.env['data.backup.examples'].standard_backup('full')
            elif config.backup_type == 'incremental':
                result = self.env['data.backup.examples'].standard_backup('incremental')
            elif config.backup_type == 'differential':
                result = self.env['data.backup.examples'].standard_backup('differential')
            else:
                result = {'status': 'error', 'message': '不支持的备份类型'}
            
            # 记录备份结果
            self.env['backup.record'].create({
                'config_id': config.id,
                'backup_type': config.backup_type,
                'status': result['status'],
                'message': result['message'],
                'backup_path': result.get('backup_path'),
                'backup_size': result.get('backup_size'),
            })
            
        except Exception as e:
            # 记录备份错误
            self.env['backup.record'].create({
                'config_id': config.id,
                'backup_type': config.backup_type,
                'status': 'error',
                'message': f'备份异常: {str(e)}',
            })
```
