### 数据模型开发

#### 基础模型
```python
# models/my_model.py
from odoo import models, fields, api
from odoo.exceptions import ValidationError, UserError

class MyModel(models.Model):
    _name = 'my.model'
    _description = 'My Custom Model'
    _order = 'name'
    _rec_name = 'name'
    _inherit = ['mail.thread', 'mail.activity.mixin']
    
    # 基础字段
    name = fields.Char('名称', required=True, index=True, tracking=True)
    code = fields.Char('编码', size=20, index=True)
    description = fields.Text('描述')
    active = fields.Boolean('激活', default=True)
    sequence = fields.Integer('序号', default=10)
    
    # 日期字段
    date = fields.Date('日期', default=fields.Date.today)
    datetime = fields.Datetime('日期时间', default=fields.Datetime.now)
    
    # 数值字段
    amount = fields.Float('金额', digits=(16, 2))
    quantity = fields.Integer('数量', default=1)
    
    # 选择字段
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('approved', '已审批'),
        ('done', '完成'),
        ('cancelled', '已取消')
    ], '状态', default='draft', tracking=True)
    
    # 关系字段
    partner_id = fields.Many2one('res.partner', '合作伙伴')
    line_ids = fields.One2many('my.model.line', 'model_id', '明细行')
    tag_ids = fields.Many2many('my.model.tag', 'model_tag_rel', 'model_id', 'tag_id', '标签')
    
    # 计算字段
    total_amount = fields.Float('总金额', compute='_compute_total', store=True)
    line_count = fields.Integer('行数', compute='_compute_line_count')
    
    @api.depends('line_ids.amount')
    def _compute_total(self):
        for record in self:
            record.total_amount = sum(record.line_ids.mapped('amount'))
    
    @api.depends('line_ids')
    def _compute_line_count(self):
        for record in self:
            record.line_count = len(record.line_ids)
    
    @api.model
    def create(self, vals):
        # 创建前处理
        if not vals.get('code'):
            vals['code'] = self._generate_code()
        return super().create(vals)
    
    def write(self, vals):
        # 更新前处理
        if 'state' in vals and vals['state'] == 'done':
            self._check_can_done()
        return super().write(vals)
    
    def unlink(self):
        # 删除前检查
        for record in self:
            if record.state == 'done':
                raise UserError("已完成的记录不能删除")
        return super().unlink()
    
    @api.model
    def _generate_code(self):
        sequence = self.env['ir.sequence'].next_by_code('my.model')
        return f"MODEL-{sequence}"
    
    def _check_can_done(self):
        for record in self:
            if not record.line_ids:
                raise ValidationError("必须有明细行才能完成")
    
    def action_confirm(self):
        """确认操作"""
        self.write({'state': 'confirmed'})
        return True
    
    def action_approve(self):
        """审批操作"""
        if self.state != 'confirmed':
            raise UserError("只能审批已确认的记录")
        self.write({'state': 'approved'})
        return True
    
    def action_done(self):
        """完成操作"""
        if self.state != 'approved':
            raise UserError("只能完成已审批的记录")
        self._check_can_done()
        self.write({'state': 'done'})
        return True
    
    def action_cancel(self):
        """取消操作"""
        if self.state in ['done']:
            raise UserError("已完成的记录不能取消")
        self.write({'state': 'cancelled'})
        return True
    
    def action_reset(self):
        """重置操作"""
        self.write({'state': 'draft'})
        return True
```

```python
# models/my_model_line.py
from odoo import models, fields, api
from odoo.exceptions import ValidationError

class MyModelLine(models.Model):
    _name = 'my.model.line'
    _description = 'My Model Line'
    _order = 'sequence, id'
    
    model_id = fields.Many2one('my.model', '主记录', required=True, ondelete='cascade')
    sequence = fields.Integer('序号', default=10)
    name = fields.Char('名称', required=True)
    description = fields.Text('描述')
    quantity = fields.Float('数量', digits=(16, 3), default=1.0)
    price_unit = fields.Float('单价', digits=(16, 2))
    price_subtotal = fields.Float('小计', compute='_compute_price_subtotal', store=True)
    
    @api.depends('quantity', 'price_unit')
    def _compute_price_subtotal(self):
        for line in self:
            line.price_subtotal = line.quantity * line.price_unit
    
    @api.constrains('quantity')
    def _check_quantity(self):
        for line in self:
            if line.quantity <= 0:
                raise ValidationError("数量必须大于0")
```

### 模块钩子

#### 安装后钩子
```python
# __init__.py
def post_init_hook(cr, registry):
    """模块安装后执行"""
    from odoo import api, SUPERUSER_ID
    
    env = api.Environment(cr, SUPERUSER_ID, {})
    
    # 创建默认数据
    env['my.model.tag'].create({
        'name': '默认标签',
        'color': 0
    })

def uninstall_hook(cr, registry):
    """模块卸载前执行"""
    from odoo import api, SUPERUSER_ID
    
    env = api.Environment(cr, SUPERUSER_ID, {})
    
    # 清理数据
    env['my.model'].search([]).unlink()
```

### 模块测试

#### 单元测试
```python
# tests/test_my_model.py
from odoo.tests.common import TransactionCase
from odoo.exceptions import ValidationError, UserError

class TestMyModel(TransactionCase):
    
    def setUp(self):
        super().setUp()
        self.partner = self.env['res.partner'].create({
            'name': 'Test Partner',
            'email': 'test@example.com'
        })
    
    def test_create_model(self):
        """测试创建模型"""
        model = self.env['my.model'].create({
            'name': 'Test Model',
            'partner_id': self.partner.id
        })
        
        self.assertEqual(model.name, 'Test Model')
        self.assertEqual(model.partner_id, self.partner)
        self.assertTrue(model.active)
        self.assertEqual(model.state, 'draft')
    
    def test_workflow(self):
        """测试工作流"""
        model = self.env['my.model'].create({
            'name': 'Workflow Test',
            'partner_id': self.partner.id
        })
        
        # 确认
        model.action_confirm()
        self.assertEqual(model.state, 'confirmed')
        
        # 审批
        model.action_approve()
        self.assertEqual(model.state, 'approved')
        
        # 完成
        model.action_done()
        self.assertEqual(model.state, 'done')
    
    def test_constraints(self):
        """测试约束"""
        with self.assertRaises(ValidationError):
            self.env['my.model'].create({
                'name': 'AB',  # 长度少于3个字符
                'partner_id': self.partner.id
            })
```
