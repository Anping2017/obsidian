### 工作流管理

#### 状态机模式
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('approved', '已审批'),
        ('done', '完成'),
        ('cancelled', '已取消')
    ], '状态', default='draft')
    
    # 状态转换规则
    STATE_TRANSITIONS = {
        'draft': ['confirmed', 'cancelled'],
        'confirmed': ['approved', 'cancelled'],
        'approved': ['done', 'cancelled'],
        'done': [],
        'cancelled': ['draft']
    }
    
    def _check_state_transition(self, new_state):
        """检查状态转换是否合法"""
        current_state = self.state
        if new_state not in self.STATE_TRANSITIONS.get(current_state, []):
            raise UserError(f"不能从 {current_state} 状态转换到 {new_state}")
    
    def action_confirm(self):
        """确认操作"""
        self._check_state_transition('confirmed')
        self._pre_confirm()
        self.write({'state': 'confirmed'})
        self._post_confirm()
        return True
    
    def _pre_confirm(self):
        """确认前处理"""
        self._validate_required_fields()
        self._check_business_rules()
    
    def _post_confirm(self):
        """确认后处理"""
        self._send_notifications()
        self._create_related_records()
```

class ApprovalWorkflow(models.Model):
    _name = 'approval.workflow'
    
    def create_approval_flow(self, record, approvers):
        """创建审批流程"""
        workflow = self.create({
            'model': record._name,
            'res_id': record.id,
            'state': 'pending'
        })
        
        for approver in approvers:
            self.env['approval.line'].create({
                'workflow_id': workflow.id,
                'approver_id': approver.id,
                'sequence': approver.sequence,
                'state': 'pending'
            })
        
        return workflow
    
    def process_approval(self, approver_id, decision):
        """处理审批"""
        approval_line = self.env['approval.line'].search([
            ('workflow_id', '=', self.id),
            ('approver_id', '=', approver_id),
            ('state', '=', 'pending')
        ])
        
        if not approval_line:
            raise UserError("没有找到待审批记录")
        
        approval_line.write({
            'state': 'approved' if decision == 'approve' else 'rejected',
            'decision_date': fields.Datetime.now(),
            'comments': decision
        })
        
        if decision == 'rejected':
            self.write({'state': 'rejected'})
            return
        
        # 检查是否所有审批都完成
        pending_approvals = self.approval_line_ids.filtered(
            lambda x: x.state == 'pending'
        )
        
        if not pending_approvals:
            self.write({'state': 'approved'})
            self._finalize_approval()

class BusinessRule(models.Model):
    _name = 'business.rule'
    
    name = fields.Char('规则名称', required=True)
    model_id = fields.Many2one('ir.model', '模型', required=True)
    condition = fields.Text('条件', help='Python表达式')
    action = fields.Selection([
        ('block', '阻止'),
        ('warn', '警告'),
        ('auto', '自动处理')
    ], '动作', default='block')
    message = fields.Text('消息')
    auto_action = fields.Text('自动动作', help='Python代码')
    active = fields.Boolean('激活', default=True)
    
    def evaluate_rule(self, record):
        """评估规则"""
        if not self.active:
            return True
        
        try:
            # 构建评估环境
            env = {
                'record': record,
                'fields': fields,
                'api': api,
                'datetime': datetime,
                'date': date,
                'time': time,
                'math': math,
                'len': len,
                'sum': sum,
                'min': min,
                'max': max,
            }
            
            # 评估条件
            result = eval(self.condition, env)
            
            if not result:
                if self.action == 'block':
                    raise UserError(self.message)
                elif self.action == 'warn':
                    # 显示警告但继续执行
                    pass
                elif self.action == 'auto':
                    # 执行自动动作
                    exec(self.auto_action, env)
            
            return result
            
        except Exception as e:
            raise UserError(f"规则评估错误: {str(e)}")

class BusinessRuleEngine(models.Model):
    _name = 'business.rule.engine'
    
    def validate_record(self, record):
        """验证记录"""
        rules = self.env['business.rule'].search([
            ('model_id.model', '=', record._name),
            ('active', '=', True)
        ])
        
        for rule in rules:
            rule.evaluate_rule(record)
        
        return True
```

### 事件驱动架构

#### 事件定义

class BusinessEvent(models.Model):
    _name = 'business.event'
    
    name = fields.Char('事件名称', required=True)
    model_id = fields.Many2one('ir.model', '模型', required=True)
    event_type = fields.Selection([
        ('create', '创建'),
        ('write', '更新'),
        ('unlink', '删除'),
        ('state_change', '状态变更'),
        ('custom', '自定义')
    ], '事件类型', required=True)
    condition = fields.Text('触发条件', help='Python表达式')
    action = fields.Text('执行动作', help='Python代码')
    active = fields.Boolean('激活', default=True)
    
    def trigger_event(self, record, event_type, vals=None):
        """触发事件"""
        if not self.active:
            return
        
        if self.event_type != event_type:
            return
        
        # 检查触发条件
        if self.condition:
            env = {
                'record': record,
                'vals': vals or {},
                'event_type': event_type,
                'fields': fields,
                'api': api,
            }
            
            if not eval(self.condition, env):
                return
        
        # 执行动作
        if self.action:
            env = {
                'record': record,
                'vals': vals or {},
                'event_type': event_type,
                'env': self.env,
                'fields': fields,
                'api': api,
            }
            
            exec(self.action, env)

class EventEngine(models.Model):
    _name = 'event.engine'
    
    def process_event(self, record, event_type, vals=None):
        """处理事件"""
        events = self.env['business.event'].search([
            ('model_id.model', '=', record._name),
            ('active', '=', True)
        ])
        
        for event in events:
            event.trigger_event(record, event_type, vals)
```

class SaleOrder(models.Model):

    def create(self, vals):
        """创建后触发事件"""
        record = super().create(vals)
        self.env['event.engine'].process_event(record, 'create', vals)
        return record
    
    def write(self, vals):
        """更新后触发事件"""
        result = super().write(vals)
        for record in self:
            self.env['event.engine'].process_event(record, 'write', vals)
        return result
    
    def unlink(self):
        """删除前触发事件"""
        for record in self:
            self.env['event.engine'].process_event(record, 'unlink')
        return super().unlink()
