# 看板视图

## Odoo 17 看板视图开发

看板视图是Odoo中用于可视化工作流和任务管理的强大工具。本文档详细介绍如何创建和定制看板视图。

### 基础看板视图

#### 简单看板视图
```xml
<record id="view_my_model_kanban" model="ir.ui.view">
    <field name="name">my.model.kanban</field>
    <field name="model">my.model</field>
    <field name="arch" type="xml">
        <kanban>
            <field name="name"/>
            <field name="state"/>
            <field name="partner_id"/>
            <field name="amount"/>
            <templates>
                <t t-name="kanban-box">
                    <div class="oe_kanban_card oe_kanban_global_click">
                        <div class="oe_kanban_content">
                            <div class="o_kanban_record_top">
                                <div class="o_kanban_record_headings">
                                    <strong class="o_kanban_record_title">
                                        <field name="name"/>
                                    </strong>
                                </div>
                            </div>
                            <div class="o_kanban_record_body">
                                <field name="partner_id"/>
                                <field name="amount"/>
                            </div>
                        </div>
                    </div>
                </t>
            </templates>
        </kanban>
    </field>
</record>
```

#### 看板视图JavaScript
```javascript
// static/src/js/kanban_interaction.js
odoo.define('my_module.kanban_interaction', function (require) {
    'use strict';
    
    var KanbanView = require('web.KanbanView');
    var KanbanRecord = require('web.KanbanRecord');
    
    KanbanRecord.include({
        _onCardClick: function (event) {
            // 自定义卡片点击事件
            if (event.target.classList.contains('custom-action')) {
                event.preventDefault();
                event.stopPropagation();
                this._onCustomAction();
                return;
            }
            return this._super.apply(this, arguments);
        },
        
        _onCustomAction: function () {
            // 自定义操作
            this.do_action({
                type: 'ir.actions.act_window',
                res_model: 'my.model',
                res_id: this.record.id,
                view_mode: 'form',
                target: 'new',
            });
        }
    });
    
    KanbanView.include({
        _onGroupBy: function (groupBy) {
            // 自定义分组逻辑
            return this._super.apply(this, arguments);
        }
    });
});
```

### 看板视图配置

#### 看板视图动作
```xml
<record id="action_my_model_kanban" model="ir.actions.act_window">
    <field name="name">我的模型看板</field>
    <field name="res_model">my.model</field>
    <field name="view_mode">kanban,tree,form</field>
    <field name="view_id" ref="view_my_model_kanban"/>
    <field name="context">{
        'search_default_group_by_state': 1,
        'kanban_auto_refresh': 1
    }</field>
    <field name="help" type="html">
        <p class="o_view_nocontent_smiling_face">
            创建新的记录
        </p>
        <p>
            使用看板视图来管理您的工作流程
        </p>
    </field>
</record>
```

#### 看板视图搜索
```xml
<record id="view_my_model_search" model="ir.ui.view">
    <field name="name">my.model.search</field>
    <field name="model">my.model</field>
    <field name="arch" type="xml">
        <search>
            <field name="name"/>
            <field name="partner_id"/>
            <field name="state"/>
            <field name="amount"/>
            
            <filter string="草稿" name="draft" domain="[('state', '=', 'draft')]"/>
            <filter string="已确认" name="confirmed" domain="[('state', '=', 'confirmed')]"/>
            <filter string="已完成" name="done" domain="[('state', '=', 'done')]"/>
            
            <separator/>
            <filter string="高优先级" name="high_priority" domain="[('priority', '=', '3')]"/>
            <filter string="中优先级" name="medium_priority" domain="[('priority', '=', '2')]"/>
            <filter string="低优先级" name="low_priority" domain="[('priority', '=', '1')]"/>
            
            <group expand="0" string="分组">
                <filter string="状态" name="group_state" context="{'group_by': 'state'}"/>
                <filter string="合作伙伴" name="group_partner" context="{'group_by': 'partner_id'}"/>
                <filter string="创建日期" name="group_date" context="{'group_by': 'create_date:day'}"/>
            </group>
        </search>
    </field>
</record>
```
