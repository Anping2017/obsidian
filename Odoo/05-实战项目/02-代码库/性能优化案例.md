# models/query_optimization.py
from odoo import models, fields, api

class QueryOptimizedModel(models.Model):
    """查询优化示例"""
    _name = 'query.optimized.model'
    _description = '查询优化模型'
    
    name = fields.Char('名称')
    partner_id = fields.Many2one('res.partner', '合作伙伴')
    amount = fields.Float('金额')
    date_field = fields.Date('日期')
    
    @api.model
    def get_optimized_data(self):
        """优化的数据获取方法"""
        # 1. 使用read()方法获取特定字段
        records = self.search([]).read(['name', 'amount', 'partner_id'])
        
        # 2. 使用browse()方法避免重复查询
        partner_ids = [r['partner_id'][0] for r in records if r['partner_id']]
        partners = self.env['res.partner'].browse(partner_ids)
        
        # 3. 使用mapped()方法获取关联数据
        partner_names = partners.mapped('name')
        
        return {
            'records': records,
            'partner_names': partner_names
        }
    
    @api.model
    def get_aggregated_data(self):
        """使用read_group进行聚合查询"""
        result = self.read_group(
            domain=[],
            fields=['amount'],
            groupby=['partner_id'],
            lazy=False
        )
        return result
    
    @api.model
    def bulk_update_records(self, record_ids, values):
        """批量更新记录"""
        # 使用SQL进行批量更新
        self.env.cr.execute("""
            UPDATE query_optimized_model 
            SET amount = %s 
            WHERE id IN %s
        """, (values.get('amount'), tuple(record_ids)))
        
        # 刷新缓存
        self.env.flush()
    
    @api.model
    def get_filtered_records(self, filters):
        """优化的过滤查询"""
        domain = []
        
        # 构建优化的查询条件
        if filters.get('partner_id'):
            domain.append(('partner_id', '=', filters['partner_id']))
        
        if filters.get('date_from'):
            domain.append(('date_field', '>=', filters['date_from']))
        
        if filters.get('date_to'):
            domain.append(('date_field', '<=', filters['date_to']))
        
        # 使用limit限制结果数量
        return self.search(domain, limit=1000)

        )

    def execute_with_pool(self, query, params=None):

        pool = self.get_connection_pool()
        conn = pool.getconn()
        
        try:
            cursor = conn.cursor()
            cursor.execute(query, params)
            result = cursor.fetchall()
            conn.commit()
            return result
        except Exception as e:
            conn.rollback()
            raise e
        finally:
            cursor.close()
            pool.putconn(conn)

import json
import pickle

class RedisCacheModel(models.Model):
    """Redis缓存优化示例"""
    _name = 'redis.cache.model'
    _description = 'Redis缓存优化模型'
    
    name = fields.Char('名称')
    data = fields.Text('数据')
    
    @api.model
    def get_redis_client(self):
        """获取Redis客户端"""
        return redis.Redis(
            host='localhost',
            port=6379,
            db=0,
            decode_responses=True
        )
    
    @api.model
    def cache_data(self, key, data, expire_time=3600):
        """缓存数据到Redis"""
        redis_client = self.get_redis_client()
        serialized_data = json.dumps(data)
        redis_client.setex(key, expire_time, serialized_data)
    
    @api.model
    def get_cached_data(self, key):
        """从Redis获取缓存数据"""
        redis_client = self.get_redis_client()
        cached_data = redis_client.get(key)
        
        if cached_data:
            return json.loads(cached_data)
        return None
    
    @api.model
    def get_or_set_cache(self, key, data_func, expire_time=3600):
        """获取或设置缓存"""
        # 尝试从缓存获取
        cached_data = self.get_cached_data(key)
        
        if cached_data is not None:
            return cached_data
        
        # 缓存未命中，执行数据函数
        data = data_func()
        
        # 缓存结果
        self.cache_data(key, data, expire_time)
        
        return data
    
    @api.model
    def invalidate_cache(self, pattern):
        """使缓存失效"""
        redis_client = self.get_redis_client()
        keys = redis_client.keys(pattern)
        if keys:
            redis_client.delete(*keys)
```

// static/src/js/performance_optimization.js
odoo.define('performance_optimization', function (require) {
    'use strict';
    
    var AbstractView = require('web.AbstractView');
    var ListView = require('web.ListView');
    var FormView = require('web.FormView');
    
    // 1. 防抖优化
    function debounce(func, wait) {
        let timeout;
        return function executedFunction(...args) {
            const later = () => {
                clearTimeout(timeout);
                func(...args);
            };
            clearTimeout(timeout);
            timeout = setTimeout(later, wait);
        };
    }
    
    // 2. 节流优化
    function throttle(func, limit) {
        let inThrottle;
        return function() {
            const args = arguments;
            const context = this;
            if (!inThrottle) {
                func.apply(context, args);
                inThrottle = true;
                setTimeout(() => inThrottle = false, limit);
            }
        };
    }
    
    // 3. 虚拟滚动优化
    var VirtualScrollView = ListView.extend({
        events: _.extend({}, ListView.prototype.events, {
            'scroll .o_list_view': 'onScroll',
        }),
        
        init: function() {
            this._super.apply(this, arguments);
            this.scrollTop = 0;
            this.itemHeight = 50;
            this.visibleItems = 20;
        },
        
        onScroll: throttle(function(event) {
            var scrollTop = event.target.scrollTop;
            var startIndex = Math.floor(scrollTop / this.itemHeight);
            var endIndex = Math.min(startIndex + this.visibleItems, this.records.length);
            
            this.renderVisibleItems(startIndex, endIndex);
        }, 16), // 60fps
        
        renderVisibleItems: function(startIndex, endIndex) {
            // 只渲染可见的项目
            var visibleRecords = this.records.slice(startIndex, endIndex);
            this.$('.o_list_view tbody').empty();
            
            _.each(visibleRecords, function(record) {
                var $row = this._renderRow(record);
                this.$('.o_list_view tbody').append($row);
            }, this);
        }
    });
    
    // 4. 懒加载优化
    var LazyLoadView = FormView.extend({
        init: function() {
            this._super.apply(this, arguments);
            this.intersectionObserver = null;
        },
        
        start: function() {
            this._super.apply(this, arguments);
            this.setupIntersectionObserver();
        },
        
        setupIntersectionObserver: function() {
            if ('IntersectionObserver' in window) {
                this.intersectionObserver = new IntersectionObserver(
                    this.onIntersection.bind(this),
                    { threshold: 0.1 }
                );
            }
        },
        
        onIntersection: function(entries) {
            entries.forEach(function(entry) {
                if (entry.isIntersecting) {
                    this.loadLazyContent(entry.target);
                }
            }, this);
        },
        
        loadLazyContent: function(element) {
            var $element = $(element);
            var dataSrc = $element.data('src');
            
            if (dataSrc) {
                $element.attr('src', dataSrc);
                $element.removeAttr('data-src');
                this.intersectionObserver.unobserve(element);
            }
        }
    });
    
    return {
        VirtualScrollView: VirtualScrollView,
        LazyLoadView: LazyLoadView,
        debounce: debounce,
        throttle: throttle
    };
});

# models/async_http_optimization.py
from odoo import models, fields, api
import asyncio
import aiohttp
import json

class AsyncHttpModel(models.Model):
    """异步HTTP优化示例"""
    _name = 'async.http.model'
    _description = '异步HTTP优化模型'
    
    name = fields.Char('名称')
    url = fields.Char('URL')
    response_data = fields.Text('响应数据')
    
    @api.model
    async def fetch_data_async(self, urls):
        """异步获取多个URL的数据"""
        async with aiohttp.ClientSession() as session:
            tasks = []
            for url in urls:
                task = self._fetch_single_url(session, url)
                tasks.append(task)
            
            results = await asyncio.gather(*tasks, return_exceptions=True)
            return results
    
    async def _fetch_single_url(self, session, url):
        """获取单个URL的数据"""
        try:
            async with session.get(url) as response:
                if response.status == 200:
                    data = await response.json()
                    return data
                else:
                    return None
        except Exception as e:
            return None
    
    @api.model
    def process_multiple_requests(self, urls):
        """处理多个HTTP请求"""
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        
        try:
            results = loop.run_until_complete(self.fetch_data_async(urls))
            return results
        finally:
            loop.close()

# models/object_pool_optimization.py
from odoo import models, fields, api
import threading

class ObjectPoolModel(models.Model):
    """对象池优化示例"""
    _name = 'object.pool.model'
    _description = '对象池优化模型'
    
    name = fields.Char('名称')
    
    # 对象池
    _object_pool = []
    _pool_lock = threading.Lock()
    
    @classmethod
    def get_object(cls):
        """从对象池获取对象"""
        with cls._pool_lock:
            if cls._object_pool:
                return cls._object_pool.pop()
            else:
                return cls._create_new_object()
    
    @classmethod
    def return_object(cls, obj):
        """将对象返回到对象池"""
        with cls._pool_lock:
            if len(cls._object_pool) < 10:  # 限制池大小
                cls._object_pool.append(obj)
    
    @classmethod
    def _create_new_object(cls):
        """创建新对象"""
        return cls()
    
    def process_data(self, data):
        """处理数据"""
        # 使用对象处理数据
        result = self._process(data)
        
        # 处理完成后返回对象到池中
        self.return_object(self)
        
        return result
    
    def _process(self, data):
        """具体的处理逻辑"""
        return data

from odoo import models, fields, api
import time
import psutil
import logging

_logger = logging.getLogger(__name__)

class PerformanceMonitoringModel(models.Model):
    """性能监控示例"""
    _name = 'performance.monitoring.model'
    _description = '性能监控模型'
    
    name = fields.Char('名称')
    execution_time = fields.Float('执行时间')
    memory_usage = fields.Float('内存使用')
    cpu_usage = fields.Float('CPU使用')
    
    @api.model
    def monitor_performance(self, func, *args, **kwargs):
        """监控函数性能"""
        start_time = time.time()
        start_memory = psutil.Process().memory_info().rss / 1024 / 1024  # MB
        start_cpu = psutil.Process().cpu_percent()
        
        try:
            result = func(*args, **kwargs)
            return result
        finally:
            end_time = time.time()
            end_memory = psutil.Process().memory_info().rss / 1024 / 1024  # MB
            end_cpu = psutil.Process().cpu_percent()
            
            execution_time = end_time - start_time
            memory_usage = end_memory - start_memory
            cpu_usage = end_cpu - start_cpu
            
            # 记录性能数据
            self.create({
                'name': func.__name__,
                'execution_time': execution_time,
                'memory_usage': memory_usage,
                'cpu_usage': cpu_usage
            })
            
            _logger.info(f"性能监控 - 函数: {func.__name__}, "
                        f"执行时间: {execution_time:.2f}s, "
                        f"内存使用: {memory_usage:.2f}MB, "
                        f"CPU使用: {cpu_usage:.2f}%")
    
    @api.model
    def get_performance_stats(self):
        """获取性能统计"""
        records = self.search([])
        
        if not records:
            return {}
        
        return {
            'avg_execution_time': sum(records.mapped('execution_time')) / len(records),
            'avg_memory_usage': sum(records.mapped('memory_usage')) / len(records),
            'avg_cpu_usage': sum(records.mapped('cpu_usage')) / len(records),

from odoo import models, fields, api

class DatabaseDiagnosisModel(models.Model):
    """数据库性能诊断示例"""
    _name = 'database.diagnosis.model'
    _description = '数据库性能诊断模型'
    
    @api.model
    def analyze_slow_queries(self):
        """分析慢查询"""
        # 启用慢查询日志
        self.env.cr.execute("SET log_min_duration_statement = 1000")  # 1秒
        
        # 查询慢查询统计
        self.env.cr.execute("""
            SELECT query, mean_time, calls, total_time
            FROM pg_stat_statements
            WHERE mean_time > 1000
            ORDER BY mean_time DESC
            LIMIT 10
        """)
        
        return self.env.cr.fetchall()
    
    @api.model
    def analyze_table_sizes(self):
        """分析表大小"""
        self.env.cr.execute("""
            SELECT 
                schemaname,
                tablename,
                pg_size_pretty(pg_total_relation_size(schemaname||'.'||tablename)) as size
            FROM pg_tables
            WHERE schemaname = 'public'
            ORDER BY pg_total_relation_size(schemaname||'.'||tablename) DESC
            LIMIT 20
        """)
        
        return self.env.cr.fetchall()
    
    @api.model
    def analyze_index_usage(self):
        """分析索引使用情况"""
        self.env.cr.execute("""
            SELECT 
                schemaname,
                tablename,
                indexname,
                idx_scan,
                idx_tup_read,
                idx_tup_fetch
            FROM pg_stat_user_indexes
            WHERE idx_scan = 0
            ORDER BY tablename, indexname
        """)
        
        return self.env.cr.fetchall()
    
    @api.model
    def optimize_database(self):
        """数据库优化"""
        # 更新表统计信息
        self.env.cr.execute("ANALYZE")
        
        # 重建索引
        self.env.cr.execute("REINDEX DATABASE odoo_db")
        
        # 清理数据库
        self.env.cr.execute("VACUUM ANALYZE")
        
        return True
```
