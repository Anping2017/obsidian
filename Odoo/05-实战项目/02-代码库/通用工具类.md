```python
# models/base_data_processor.py
from odoo import models, fields, api, _
from odoo.exceptions import UserError, ValidationError
import csv
import json
import io
from datetime import datetime

class BaseDataProcessor(models.AbstractModel):
    """基础数据处理模型"""
    _name = 'base.data.processor'
    _description = '基础数据处理模型'
    
    def process_data_import(self, data_type: str, file_content: bytes, 
                            field_mapping: dict, settings: dict = None) -> dict:
        """
        导入数据
        
        Args:
            data_type: 数据类型 (csv, json, excel)
            file_content: 文件内容
            field_mapping: 字段映射关系
            settings: 导入设置
            
        Returns:
            dict: 导入结果统计
        """
        try:
            if data_type == 'csv':
                return self._process_csv_import(file_content, field_mapping, settings)
            elif data_type == 'json':
                return self._process_json_import(file_content, field_mapping, settings)
            elif data_type == 'excel':
                return self._process_excel_import(file_content, field_mapping, settings)
            else:
                raise ValueError(f"不支持的数据类型: {data_type}")
                
        except Exception as e:
            raise UserError(_("数据导入失败: %s") % str(e))
    
    def _process_csv_import(self, file_content: bytes, field_mapping: dict, 
                           settings: dict) -> dict:
        """处理CSV文件导入"""
        stats = {'success': 0, 'failed': 0, 'errors': []}
        
        try:
            content = file_content.decode('utf-8')
            csv_content = io.StringIO(content)
            reader = csv.DictReader(csv_content)
            
            for row_index, row in enumerate(reader, start=1):
                try:
                    processed_data = self._preprocess_import_data(row, field_mapping)
                    self._validate_import_data(processed_data)
                    if self._create_record_from_data(processed_data):
                        stats['success'] += 1
                    else:
                        stats['failed'] += 1
                        stats['errors'].append(f"第{row_index}行: 创建记录失败")
                        
                except Exception as e:
                    stats['failed'] += 1 
                    stats['errors'].append(f"第{row_index}行: {str(e)}")
                    
        except Exception as e:
            raise UserError(f"CSV处理失败: {str(e)}")
        
        return stats
    
    def _preprocess_import_data(self, raw_data: dict, field_mapping: dict) -> dict:
        """预处理导入数据"""
        processed_data = {}
        
        for field_name, mapping_info in field_mapping.items():
            source_field = mapping_info.get('field')
            data_type = mapping_info.get('type', 'string')
            default_value = mapping_info.get('default')
            
            raw_value = raw_data.get(source_field, default_value)
            
            # 根据字段类型处理数据
            if data_type == 'datetime' and raw_value:
                processed_data[field_name] = self._parse_datetime(raw_value)
            elif data_type == 'integer' and raw_value:
                processed_data[field_name] = int(raw_value)
            elif data_type == 'float' and raw_value:
                processed_data[field_name] = float(raw_value)
            elif data_type == 'boolean':
                processed_data[field_name] = self._parse_boolean(raw_value)
            elif isinstance(raw_value, str) and raw_value.strip() == '':
                processed_data[field_name] = False
            else:
                processed_data[field_name] = raw_value
                
        return processed_data
    
    def _validate_import_data(self, data: dict):
        """验证导入数据"""
        for field_name, field_value in data.items():
            # 基本验证逻辑
            if field_value and len(str(field_value)) > 1000:
                raise ValidationError(f"{field_name} 字段长度超过限制")
    
    def _create_record_from_data(self, data: dict) -> bool:
        """根据数据创建记录"""
        try:
            self.create(data)
            return True
        except Exception:
            return False
    
    def generate_data_template(self, fields_config: list) -> str:
        """生成Excel导入模板"""
        import xlsxwriter
        
        output = io.BytesIO()
        workbook = xlsxwriter.Workbook(output)
        worksheet = workbook.add_worksheet('Import Template')
        
        # 创建表头行
        for col_idx, field_info in enumerate(fields_config):
            worksheet.write(0, col_idx, field_info['name'])
        worksheet.set_row(0, width=100)
        
        # 写入示例数据
        for row_idx, field_info in enumerate(fields_config, start=1):
            worksheet.write(row_idx, 0, field_info['name'])
            worksheet.write(row_idx, 1, field_info['example'])
        
        workbook.close()
        data = output.getvalue()
        
        # 保存文件
        attachment = self.env['ir.attachment'].create({
            'name': 'import_template.xlsx',
            'datas': data.encode('base64'),
            'mimetype': 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
        })
        
        return attachment.public_url
    
    def bulk_delete_by_ids(self, record_ids: list) -> dict:
        """批量删除记录"""
        stats = {'success_count': 0, 'failed_count': 0}
        
        for record_id in record_ids:
            try:
                record = self.browse(record_id)
                record.unlink()
                stats['success_count'] += 1
            except Exception:
                stats['failed_count'] += 1
        return stats
    
    def calculate_statistics(self, field: str, group_by: str = None, 
                            filters: dict = None) -> dict:
        """计算字段统计信息"""
        domain = list(self.env[self._name]._get_domain_from_fields(filters))
        
        if group_by:
            grouped_records = self.env[self._name].read_group(
                domain, [field], [group_by]
            )
            return grouped_records
        
        records = self.env[self._name].search(domain)
        
        # 基本统计信息
        statistics = {
            'count': len(records),
            'sum': sum(records.mapped(field)),
            'avg': sum(records.mapped(field)) / len(records) if records else 0,
            'min': min(records.mapped(field)) if records else 0,
            'max': max(records.mapped(field)) if records else 0,
        }
        
        return statistics
                
    def _parse_datetime(self, value: str) -> datetime:
        """解析日期时间"""
        try:
            return datetime.strptime(value.strip(), '%Y-%m-%d')
        except ValueError:
            try:
                return datetime.strptime(value.strip(), '%Y-%m-%d %H:%M:%S')
            except ValueError:
                raise ValidationError(f"无法解析日期格式: {value}")
    
    def _parse_boolean(self, value) -> bool:
        """解析布尔值"""
        if isinstance(value, bool):
            return value
        
        falsy_values = ['false', 'no', '0', 'f', 'n']
        if str(value).lower().strip() in falsy_values:
            return False
        
        return value and str(value).strip().lower() not in ['', 'null', 'undefined']
```

        else:
            messages.append("密码长度至少8位")

            score += 1
        
        if re.search(r'[A-Z]', password):
            score += 1
            
        if re.search(r'[0-9]', password):

            'level': strength_level,
            'messages': messages,
            'is_valid': score >= 3

# models/base_data_import.py
from odoo import models, fields, api, _
from odoo.exceptions import UserError
import csv
import json
import io
import base64

class BaseDataImport(models.AbstractModel):
    """基础数据导入模型"""
    _name = 'base.data.import'
    _description = '基础数据导入模型'
    
    def batch_insert_from_csv(self, csv_file: bytes, model_name: str, 
                             field_mapping: dict, batch_size: int = 100) -> dict:
        """从CSV文件批量导入数据"""
        result = {
            'success_count': 0,
            'failed_count': 0,
            'error_messages': [],
        }
        
        try:
            csv_content = csv_file.decode('utf-8')
            csv_data = io.StringIO(csv_content)
            
            # 处理行数据
            lines = [line.strip() for line in csv_data.readlines()]
            batch = []
            
            for line_idx, line in enumerate(lines[1:], start=2):  # 跳过标题行
                try:
                    row_data = self._parse_csv_line(line)
                    mapped_data = self._map_field_data(row_data, field_mapping)
                    batch.append(mapped_data)
                    
                    if len(batch) >= batch_size:
                        batch_result = self._insert_batch(batch, model_name)
                        
                        result['success_count'] += batch_result['success']
                        result['failed_count'] += batch_result['failed']
                        
                        batch = []
                        
                except Exception as e:
                    result['failed_count'] += 1
                    result['error_messages'].append(f"第{line_idx}行: {str(e)}")
            
            # 处理剩余数据
            if batch:
                batch_result = self._insert_batch(batch, model_name)
                result['success_count'] += batch_result['success']
                result['failed_count'] += batch_result['failed']
            
        except Exception as e:
            raise UserError(f"CSV批量导入失败: {str(e)}")
        
        return result
    
    def batch_insert_from_excel(self, excel_file: bytes, sheet_name: str, 
                               model_name: str, field_mapping: dict) -> dict:
        """从Excel文件批量导入数据"""
        try:
            import pandas as pd
            
            excel_data = pd.read_excel(io.BytesIO(excel_file), sheet_name=sheet_name)
            
            result = {
                'success_count': 0,
                'failed_count': 0,
                'error_messages': [],
            }
            
            for index, row_data in excel_data.iterrows():
                try:
                    mapped_data = self._map_field_data(row_data.to_dict(), field_mapping)
                    self.env[model_name].create(mapped_data)
                    result['success_count'] += 1
                    
                except Exception as e:
                    result['failed_count'] += 1
                    result['error_messages'].append(f"第{index+1}行: {str(e)}")
            
            return result
            
        except ImportError:
            raise UserError("未安装pandas，请先安装: pip install pandas openpyxl")
    
    def export_to_csv(self, domain: list, fields: list, filename: str = None) -> str:
        """导出为CSV文件"""
        records = self.search(domain)
        
        if not filename:
            filename = f"export_{datetime.now().strftime('%Y%m%d_%H%M%S')}.csv"
        
        output = io.StringIO()
        writer = csv.writer(output)
        
        # 写入文件头
        writer.writerow(fields)
        
        # 写入数据行
        for record in records:
            row_data = []
            for field_name in fields:
                field_value = record[field_name]
                if isinstance(field_value, str):
                    field_value = field_value.encode('utf-8').decode('utf-8')
                row_data.append(field_value or '')
            
            writer.writerow(row_data)
        
        csv_content = output.getvalue()
        
        # 保存导出文件
        attachment = self.env['ir.attachment'].create({
            'name': filename,
            'datas': base64.b64encode(csv_content.encode('utf-8')),
            'mimetype': 'text/csv',
        })
        
        return attachment.public_url
    
    def _parse_csv_line(self, line: str) -> dict:
        """解析CSV行数据"""
        import csv
        import io
        
        csv_data = io.StringIO(line)
        reader = csv.DictReader(csv_data)
        
        return next(reader)
    
    def _map_field_data(self, row_data: dict, field_mapping: dict) -> dict:
        """映射字段数据"""
        mapped_data = {}
        
        for field_name in field_mapping.keys():
            header_field = field_mapping.get(field_name)
            raw_value = row_data.get(header_field, '')
            
            # 数据转换
            mapped_data[field_name] = self._convert_value(raw_value)
        
        return mapped_data
    
    def _convert_value(self, value):
        """转换字段值"""
        value_str = str(value).strip()
        
        if value_str == '':
            return False
        
        # 根据值格式推断类型
        if value_str.isdigit():
            return int(value_str)
        elif self._is_float(value_str):
            return float(value_str)
        elif value_str.lower() in ['true', 'false']:
            return value_str.lower() == 'true'
        
        return value_str
    
    def _insert_batch(self, data_batch: list, model_name: str) -> dict:
        """批量插入数据"""
        result = {'success': 0, 'failed': 0}
        
        try:
            self.env[model_name].create(data_batch)
            result['success'] = len(data_batch)
        except Exception:
            result['failed'] = len(data_batch)
        
        return result
    
    def _is_float(self, value: str) -> bool:
        """判断是否为浮点数"""
        try:
            float(value)
            return True
        except ValueError:
            return False
```

# models/base_file_ops.py
import os
import mimetypes
from slugify import slugify
from odoo import models, api, _
from odoo.exceptions import UserError

class BaseFileOps(models.AbstractModel):
    """基础文件操作模型"""
    _name = 'base.file.ops'
    _description = '基础文件操作模型'
    
    def save_attachment(self, name: str, content: bytes, 
                        mimetype: str = None, res_model: str = None, 
                        res_id: int = None, is_public: bool = False) -> int:
        """保存附件"""
        try:
            # 生成唯一文件名
            filename = self._generate_unique_filename(name)
            
            attachment_vals = {
                'name': filename,
                'datas': content.encode('base64'),
                'mimetype': mimetype or mimetypes.guess_type(filename)[0],
            }
            
            if res_model and res_id:
                attachment_vals.update({
                    'res_model': res_model,
                    'res_id': res_id,
                })
            
            if is_public:
                attachment_vals['public'] = True
            
            attachment = self.env['ir.attachment'].create(attachment_vals)
            return attachment.id
            
        except Exception as e:
            raise UserError(f"保存附件失败: {str(e)}")
    
    def download_file(self, attachment_id: int) -> tuple:
        """下载文件"""
        attachment = self.env['ir.attachment'].browse(attachment_id)
        
        if not attachment.exists():
            raise UserError("文件不存在")
        
        return attachment.datas, attachment.name, attachment.mimetype
    
    def compress_files(self, attachment_ids: list, archive_name: str = None) -> str:
        """压缩文件"""
        import zipfile
        import io
        
        if not archive_name:
            archive_name = f"archive_{datetime.now().strftime('%Y%m%d_%H%M%S')}.zip"
        
        zip_buffer = io.BytesIO()
        
        with zipfile.ZipFile(zip_buffer, 'w', zipfile.ZIP_DEFLATED) as zipf:
            for attachment_id in attachment_ids:
                attachment = self.env['ir.attachment'].browse(attachment_id)
                
                if attachment.exists():
                    attachment_content = attachment.datas.decode('base64') 
                    zipf.writestr(attachment.name, attachment_content)
        
        zip_content = zip_buffer.getvalue()
        
        # 保存压缩文件
        compressed_attachment = self.save_attachment(
            archive_name, zip_content, 
            mimetype='application/zip'
        )
        
        return self.env['ir.attachment'].browse(compressed_attachment).public_url
    
    def generate_pdf_from_html(self, html_content: str, options: dict = None) -> bytes:
        """从HTML生成PDF"""
        try:
            from weasyprint import HTML, CSS
            
            styles = None
            if options and 'styles' in options:
                styles = options['styles']
            
            html_doc = HTML(string=html_content)
            
            pdf_bytes = html_doc.write_pdf(stylesheets=styles)
            
            return pdf_bytes
            
        except ImportError:
            raise UserError("未安装weasyprint，请先安装: pip install weasyprint")
    
    def generate_qrcode(self, data: str, size: int = 200) -> bytes:
        """生成二维码"""
        try:
            import qrcode
            
            qr = qrcode.QRCode(version=1, box_size=10, border=5)
            qr.add_data(data)
            qr.make(fit=True)
            
            img = qr.make_image(fill_color="black", back_color="white")
            
            img_buffer = io.BytesIO()
            img.save(img_buffer, format='PNG')
            img_buffer.seek(0)
            
            return img_buffer.getvalue()
            
        except ImportError:
            raise UserError("未安装qrcode，请先安装: pip install qrcode")
    
    def convert_image_format(self, image_data: bytes, target_format: str) -> bytes:
        """转换图片格式"""
        try:
            from PIL import Image
            
            image = Image.open(io.BytesIO(image_data))
            
            output_buffer = io.BytesIO()
            image.save(output_buffer, format=target_format.upper())
            output_buffer.seek(0)
            
            return output_buffer.getvalue()
            
        except ImportError:
            raise UserError("未安装pillow，请先安装: pip install pillow")
    
    def cleanup_old_files(self, model_name: str, days: int = 30, 
                         keep_count: int = 100) -> int:
        """清理旧文件"""
        cutoff_date = datetime.now() - timedelta(days=days)
        
        attachments = self.env['ir.attachment'].search([
            ('res_model', '=', model_name),
            ('create_date', '<', cutoff_date),
        ], order='create_date DESC')
        
        # 保留较新的文件
        files_to_delete = attachments[keep_count:]
        
        file_count = len(files_to_delete)
        files_to_delete.unlink()
        
        return file_count
    
    def _generate_unique_filename(self, filename: str) -> str:
        """生成唯一文件名"""
        if filename:
            return ''.join(c for c in filename if c.isprintable())[:200]
        
        return f"unknown_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
```
