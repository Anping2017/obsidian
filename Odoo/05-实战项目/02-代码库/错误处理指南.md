```python
# models/exception_handling.py
from odoo import models, fields, api
from odoo.exceptions import ValidationError, UserError, AccessError, MissingError
from odoo.tools import mute_logger
import logging

_logger = logging.getLogger(__name__)

class ExceptionHandlingModel(models.Model):
    """异常处理示例模型"""
    _name = 'exception.handling.model'
    _description = '异常处理模型'
    
    name = fields.Char('名称', required=True)
    amount = fields.Float('金额')
    partner_id = fields.Many2one('res.partner', '合作伙伴')
    
    @api.constrains('amount')
    def _check_amount(self):
        """数据验证异常"""
        for record in self:
            if record.amount < 0:
                raise ValidationError("金额不能为负数")
            if record.amount > 1000000:
                raise ValidationError("金额不能超过100万")
    
    def action_process(self):
        """业务逻辑异常处理"""
        try:
            # 业务逻辑处理
            self._validate_business_rules()
            self._process_data()
            return True
        except ValidationError as e:
            # 数据验证错误，直接抛出给用户
            raise e
        except UserError as e:
            # 用户操作错误，直接抛出给用户
            raise e
        except AccessError as e:
            # 权限错误，记录日志并抛出
            _logger.error(f"权限错误: {e}")
            raise UserError("您没有权限执行此操作")
        except Exception as e:
            # 其他未预期错误，记录日志并抛出用户友好错误
            _logger.error(f"处理失败: {e}", exc_info=True)
            raise UserError("处理过程中发生错误，请稍后重试")
    
    def _validate_business_rules(self):
        """验证业务规则"""
        if not self.partner_id:
            raise UserError("请选择合作伙伴")
        
        if self.amount <= 0:
            raise ValidationError("金额必须大于0")
    
    def _process_data(self):
        """处理数据"""
        # 模拟可能出错的操作
        if self.amount > 500000:
            raise UserError("金额过大，需要特殊审批")
        
        # 正常处理逻辑
        self.write({'state': 'processed'})
```

### 错误处理策略

#### 1. 分层错误处理
```python
# models/layered_error_handling.py
from odoo import models, fields, api
from odoo.exceptions import ValidationError, UserError
import logging

_logger = logging.getLogger(__name__)

class LayeredErrorHandlingModel(models.Model):
    """分层错误处理示例"""
    _name = 'layered.error.handling.model'
    _description = '分层错误处理模型'
    
    name = fields.Char('名称')
    data = fields.Text('数据')
    
    def process_with_layers(self):
        """分层错误处理"""
        # 第一层：数据验证
        try:
            self._validate_input_data()
        except ValidationError:
            # 数据验证错误，直接返回给用户
            raise
        
        # 第二层：业务逻辑处理
        try:
            result = self._process_business_logic()
        except UserError:
            # 业务逻辑错误，直接返回给用户
            raise
        except Exception as e:
            # 其他错误，记录日志并返回通用错误
            _logger.error(f"业务逻辑处理失败: {e}", exc_info=True)
            raise UserError("业务处理失败，请检查数据后重试")
        
        # 第三层：数据持久化
        try:
            self._save_results(result)
        except Exception as e:
            # 数据保存错误，记录日志并回滚
            _logger.error(f"数据保存失败: {e}", exc_info=True)
            self.env.cr.rollback()
            raise UserError("数据保存失败，请重试")
        
        return result
    
    def _validate_input_data(self):
        """验证输入数据"""
        if not self.name:
            raise ValidationError("名称不能为空")
        
        if not self.data:
            raise ValidationError("数据不能为空")
    
    def _process_business_logic(self):
        """处理业务逻辑"""
        # 模拟业务逻辑处理
        if len(self.data) > 1000:
            raise UserError("数据过长，请简化后重试")
        
        return {"processed": True, "length": len(self.data)}
    
    def _save_results(self, result):
        """保存结果"""
        # 模拟数据保存
        self.write({'data': str(result)})
```

#### 2. 重试机制
```python
# models/retry_mechanism.py
from odoo import models, fields, api
from odoo.exceptions import UserError
import time
import random
import logging

_logger = logging.getLogger(__name__)

class RetryMechanismModel(models.Model):
    """重试机制示例"""
    _name = 'retry.mechanism.model'
    _description = '重试机制模型'
    
    name = fields.Char('名称')
    max_retries = fields.Integer('最大重试次数', default=3)
    retry_delay = fields.Float('重试延迟(秒)', default=1.0)
    
    def process_with_retry(self):
        """带重试机制的处理"""
        for attempt in range(self.max_retries + 1):
            try:
                return self._unreliable_operation()
            except Exception as e:
                if attempt == self.max_retries:
                    # 最后一次尝试失败，抛出错误
                    _logger.error(f"操作失败，已重试{self.max_retries}次: {e}")
                    raise UserError(f"操作失败，已重试{self.max_retries}次，请稍后重试")
                else:
                    # 记录重试信息
                    _logger.warning(f"第{attempt + 1}次尝试失败: {e}，{self.retry_delay}秒后重试")
                    time.sleep(self.retry_delay)
        
        return None
    
    def _unreliable_operation(self):
        """模拟不稳定的操作"""
        # 模拟随机失败
        if random.random() < 0.7:  # 70%概率失败
            raise Exception("模拟的操作失败")
        
        return "操作成功"
    
    def process_with_exponential_backoff(self):
        """指数退避重试"""
        base_delay = self.retry_delay
        max_delay = 60  # 最大延迟60秒
        
        for attempt in range(self.max_retries + 1):
            try:
                return self._unreliable_operation()
            except Exception as e:
                if attempt == self.max_retries:
                    _logger.error(f"操作失败，已重试{self.max_retries}次: {e}")
                    raise UserError(f"操作失败，已重试{self.max_retries}次，请稍后重试")
                else:
                    # 指数退避：延迟时间逐渐增加
                    delay = min(base_delay * (2 ** attempt), max_delay)
                    _logger.warning(f"第{attempt + 1}次尝试失败: {e}，{delay}秒后重试")
                    time.sleep(delay)
        
        return None
```

### 日志记录

#### 1. 结构化日志
```python
# models/structured_logging.py
from odoo import models, fields, api
import logging
import json
from datetime import datetime

_logger = logging.getLogger(__name__)

class StructuredLoggingModel(models.Model):
    """结构化日志示例"""
    _name = 'structured.logging.model'
    _description = '结构化日志模型'
    
    name = fields.Char('名称')
    operation_type = fields.Selection([
        ('create', '创建'),
        ('update', '更新'),
        ('delete', '删除'),
        ('process', '处理')
    ])
    
    def log_operation(self, operation, **kwargs):
        """记录操作日志"""
        log_data = {
            'timestamp': datetime.now().isoformat(),
            'model': self._name,
            'record_id': self.id,
            'operation': operation,
            'user_id': self.env.user.id,
            'company_id': self.env.company.id,
            **kwargs
        }
        
        _logger.info(f"操作日志: {json.dumps(log_data, ensure_ascii=False)}")
    
    def create(self, vals):
        """重写创建方法，添加日志"""
        result = super().create(vals)
        result.log_operation('create', values=vals)
        return result
    
    def write(self, vals):
        """重写写入方法，添加日志"""
        old_values = {field: getattr(self, field) for field in vals.keys()}
        result = super().write(vals)
        self.log_operation('update', old_values=old_values, new_values=vals)
        return result
    
    def unlink(self):
        """重写删除方法，添加日志"""
        record_data = {
            'name': self.name,
            'operation_type': self.operation_type
        }
        result = super().unlink()
        self.log_operation('delete', record_data=record_data)
        return result
```

#### 2. 错误日志记录
```python
# models/error_logging.py
from odoo import models, fields, api
import logging
import traceback
from datetime import datetime

_logger = logging.getLogger(__name__)

class ErrorLoggingModel(models.Model):
    """错误日志记录示例"""
    _name = 'error.logging.model'
    _description = '错误日志模型'
    
    name = fields.Char('名称')
    error_count = fields.Integer('错误次数', default=0)
    last_error = fields.Text('最后错误信息')
    last_error_time = fields.Datetime('最后错误时间')
    
    def log_error(self, error, context=None):
        """记录错误日志"""
        error_info = {
            'error_type': type(error).__name__,
            'error_message': str(error),
            'traceback': traceback.format_exc(),
            'context': context or {},
            'timestamp': datetime.now().isoformat(),
            'user_id': self.env.user.id,
            'record_id': self.id
        }
        
        # 记录到日志文件
        _logger.error(f"错误详情: {error_info}")
        
        # 更新记录
        self.write({
            'error_count': self.error_count + 1,
            'last_error': str(error),
            'last_error_time': fields.Datetime.now()
        })
    
    def safe_operation(self, operation_func, *args, **kwargs):
        """安全执行操作"""
        try:
            return operation_func(*args, **kwargs)
        except Exception as e:
            self.log_error(e, {
                'operation': operation_func.__name__,
                'args': args,
                'kwargs': kwargs
            })
            raise
    
    def batch_operation_with_error_handling(self, operations):
        """批量操作，单个失败不影响其他"""
        results = []
        errors = []
        
        for i, operation in enumerate(operations):
            try:
                result = self.safe_operation(operation)
                results.append({'index': i, 'success': True, 'result': result})
            except Exception as e:
                errors.append({'index': i, 'error': str(e)})
                results.append({'index': i, 'success': False, 'error': str(e)})
        
        # 记录批量操作结果
        _logger.info(f"批量操作完成: 成功{len([r for r in results if r['success']])}个，失败{len(errors)}个")
        
        return {
            'results': results,
            'errors': errors,
            'success_count': len([r for r in results if r['success']]),
            'error_count': len(errors)
        }
```

```python
# models/user_friendly_errors.py
from odoo import models, fields, api, _
from odoo.exceptions import UserError, ValidationError
import logging

_logger = logging.getLogger(__name__)

class UserFriendlyErrorModel(models.Model):
    """用户友好错误消息示例"""
    _name = 'user.friendly.error.model'
    _description = '用户友好错误模型'
    
    name = fields.Char('名称', required=True)
    email = fields.Char('邮箱')
    phone = fields.Char('电话')
    amount = fields.Float('金额')
    
    @api.constrains('email')
    def _check_email(self):
        """邮箱验证，提供友好错误消息"""
        for record in self:
            if record.email:
                if '@' not in record.email:
                    raise ValidationError(_("邮箱格式不正确，请检查是否包含@符号"))
                if '.' not in record.email.split('@')[1]:
                    raise ValidationError(_("邮箱格式不正确，请检查域名部分"))
    
    @api.constrains('phone')
    def _check_phone(self):
        """电话验证，提供友好错误消息"""
        for record in self:
            if record.phone:
                if not record.phone.isdigit():
                    raise ValidationError(_("电话号码只能包含数字"))
                if len(record.phone) != 11:
                    raise ValidationError(_("电话号码必须是11位数字"))
                if not record.phone.startswith('1'):
                    raise ValidationError(_("电话号码必须以1开头"))
    
    def action_validate_and_process(self):
        """验证并处理，提供友好的错误消息"""
        try:
            self._validate_business_rules()
            self._process_data()
            return {
                'type': 'ir.actions.client',
                'tag': 'display_notification',
                'params': {
                    'title': _('成功'),
                    'message': _('操作完成 successfully'),
                    'type': 'success',
                }
            }
        except ValidationError as e:
            # 验证错误，直接显示给用户
            raise e
        except UserError as e:
            # 用户错误，直接显示给用户
            raise e
        except Exception as e:
            # 其他错误，记录日志并显示友好消息
            _logger.error(f"处理失败: {e}", exc_info=True)
            raise UserError(_("处理过程中发生错误，请联系系统管理员"))
    
    def _validate_business_rules(self):
        """验证业务规则"""
        if not self.name:
            raise ValidationError(_("请填写名称"))
        
        if self.amount and self.amount < 0:
            raise ValidationError(_("金额不能为负数"))
        
        if self.amount and self.amount > 1000000:
            raise UserError(_("金额过大，请联系财务部门审批"))
    
    def _process_data(self):

        if self.amount and self.amount > 500000:
            raise UserError(_("金额超过50万，需要特殊处理流程"))
```

```python
import logging

class ErrorRecoveryModel(models.Model):
    """错误恢复机制示例"""
    _name = 'error.recovery.model'
    _description = '错误恢复模型'

    state = fields.Selection([
        ('draft', '草稿'),
        ('processing', '处理中'),
        ('completed', '已完成'),
        ('failed', '失败'),
        ('recovered', '已恢复')
    ], default='draft')
    
    error_message = fields.Text('错误信息')
    retry_count = fields.Integer('重试次数', default=0)
    max_retries = fields.Integer('最大重试次数', default=3)
    
    def action_process_with_recovery(self):
        """带恢复机制的处理"""
        try:
            self.write({'state': 'processing'})
```
