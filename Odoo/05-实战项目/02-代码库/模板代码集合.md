```python
# models/main_model.py
from odoo import models, fields, api
from odoo.exceptions import ValidationError, UserError

class MainModel(models.Model):
    """主模型模板"""
    _name = 'main.model'
    _description = 'Main Model'
    _inherit = ['mail.thread', 'mail.activity.mixin']
    _order = 'create_date desc'
    _rec_name = 'name'
    
    # 基础字段
    name = fields.Char('名称', required=True, index=True, tracking=True)
    code = fields.Char('编码', size=10, index=True, copy=False)
    active = fields.Boolean('激活', default=True, tracking=True)
    
    # 状态字段
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '已完成'),
        ('cancelled', '已取消')
    ], string='状态', default='draft', tracking=True, group_expand='_expand_states')
    
    # 关联字段
    partner_id = fields.Many2one('res.partner', '合作伙伴', required=True, tracking=True)
    user_id = fields.Many2one('res.users', '负责人', default=lambda self: self.env.user, tracking=True)
    company_id = fields.Many2one('res.company', '公司', default=lambda self: self.env.company, required=True)
    
    # 行模型关联
    line_ids = fields.One2many('main.model.line', 'main_id', '明细行', copy=True)
    
    # 标签关联
    tag_ids = fields.Many2many('main.model.tag', 'main_model_tag_rel', 'main_id', 'tag_id', '标签')
    
    # 计算字段
    total_amount = fields.Float('总金额', compute='_compute_total_amount', store=True, digits='Product Price')
    line_count = fields.Integer('行数', compute='_compute_line_count', store=True)
    
    # 日期字段
    date_order = fields.Date('订单日期', default=fields.Date.today, required=True, tracking=True)
    date_delivery = fields.Date('交货日期', tracking=True)
    
    # 文本字段
    notes = fields.Text('备注')
    
    # 约束
    @api.constrains('name')
    def _check_name(self):
        for record in self:
            if len(record.name) < 3:
                raise ValidationError("名称长度不能少于3个字符")
    
    @api.constrains('date_delivery')
    def _check_date_delivery(self):
        for record in self:
            if record.date_delivery and record.date_delivery < record.date_order:
                raise ValidationError("交货日期不能早于订单日期")
    
    # 计算字段方法
    @api.depends('line_ids.amount')
    def _compute_total_amount(self):
        for record in self:
            record.total_amount = sum(record.line_ids.mapped('amount'))
    
    @api.depends('line_ids')
    def _compute_line_count(self):
        for record in self:
            record.line_count = len(record.line_ids)
    
    # 状态展开
    def _expand_states(self, states, domain, order):
        return [key for key, val in type(self).state.selection]
    
    # 默认值
    @api.model
    def default_get(self, fields_list):
        res = super().default_get(fields_list)
        res['user_id'] = self.env.user.id
        res['company_id'] = self.env.company.id
        return res
    
    # 创建方法
    @api.model
    def create(self, vals):
        if not vals.get('code'):
            vals['code'] = self._generate_code()
        return super().create(vals)
    
    # 写入方法
    def write(self, vals):
        result = super().write(vals)
        if 'state' in vals:
            self._onchange_state()
        return result
    
    # 业务方法
    def action_confirm(self):
        self.write({'state': 'confirmed'})
        return True
    
    def action_done(self):
        self.write({'state': 'done'})
        return True
    
    def action_cancel(self):
        self.write({'state': 'cancelled'})
        return True
    
    def action_reset_to_draft(self):
        self.write({'state': 'draft'})
        return True
    
    # 私有方法
    def _generate_code(self):
        return self.env['ir.sequence'].next_by_code('main.model.code')
    
    def _onchange_state(self):
        # 状态改变时的处理逻辑
        pass
    
    # 变更方法
    @api.onchange('partner_id')
    def _onchange_partner_id(self):
        if self.partner_id:
            self.user_id = self.partner_id.user_id or self.env.user
    
    @api.onchange('date_order')
    def _onchange_date_order(self):
        if self.date_order and not self.date_delivery:
            # 默认交货日期为订单日期后7天
            from datetime import timedelta
            self.date_delivery = self.date_order + timedelta(days=7)
```

```python
# models/line_model.py
from odoo import models, fields, api
from odoo.exceptions import ValidationError

class MainModelLine(models.Model):
    """主模型行模板"""
    _name = 'main.model.line'
    _description = 'Main Model Line'
    _order = 'sequence, id'
    _rec_name = 'product_id'
    
    # 关联字段
    main_id = fields.Many2one('main.model', '主记录', required=True, ondelete='cascade', index=True)
    product_id = fields.Many2one('product.product', '产品', required=True, domain="[('sale_ok', '=', True)]")
    
    # 基础字段
    name = fields.Char('描述', related='product_id.name', store=True, readonly=False)
    sequence = fields.Integer('序号', default=10)
    quantity = fields.Float('数量', default=1.0, digits='Product Unit of Measure', required=True)
    unit_price = fields.Float('单价', digits='Product Price', required=True)
    amount = fields.Float('金额', compute='_compute_amount', store=True, digits='Product Price')
    
    # 单位字段
    product_uom_id = fields.Many2one('uom.uom', '单位', related='product_id.uom_id', store=True, readonly=True)
    
    # 约束
    @api.constrains('quantity')
    def _check_quantity(self):
        for line in self:
            if line.quantity <= 0:
                raise ValidationError("数量必须大于0")
    
    @api.constrains('unit_price')
    def _check_unit_price(self):
        for line in self:
            if line.unit_price < 0:
                raise ValidationError("单价不能为负数")
    
    # 计算字段
    @api.depends('quantity', 'unit_price')
    def _compute_amount(self):
        for line in self:
            line.amount = line.quantity * line.unit_price
    
    # 变更方法
    @api.onchange('product_id')
    def _onchange_product_id(self):
        if self.product_id:
            self.unit_price = self.product_id.list_price
            self.name = self.product_id.name
            self.product_uom_id = self.product_id.uom_id
    
    @api.onchange('quantity', 'unit_price')
    def _onchange_quantity_price(self):
        # 计算金额的逻辑在计算字段中处理
        pass
```

```python
from odoo import models, fields, api
```

### 向导模板

#### 1. 向导模型模板
```python
# wizards/wizard_model.py
from odoo import models, fields, api
from odoo.exceptions import UserError

class WizardModel(models.TransientModel):
    """向导模型模板"""
    _name = 'wizard.model'
    _description = 'Wizard Model'
    
    # 基础字段
    name = fields.Char('名称', required=True)
    date_from = fields.Date('开始日期', required=True)
    date_to = fields.Date('结束日期', required=True)
    
    # 选择字段
    partner_ids = fields.Many2many('res.partner', string='合作伙伴')
    state = fields.Selection([
        ('draft', '草稿'),
        ('confirmed', '已确认'),
        ('done', '已完成')
    ], string='状态', default='draft')
    
    # 约束
    @api.constrains('date_from', 'date_to')
    def _check_dates(self):
        for record in self:
            if record.date_from > record.date_to:
                raise UserError("开始日期不能晚于结束日期")
    
    # 默认值
    @api.model
    def default_get(self, fields_list):
        res = super().default_get(fields_list)
        res['date_from'] = fields.Date.today()
        res['date_to'] = fields.Date.today()
        return res
    
    # 变更方法
    @api.onchange('date_from')
    def _onchange_date_from(self):
        if self.date_from and not self.date_to:
            self.date_to = self.date_from
    
    # 动作方法
    def action_confirm(self):
        """确认操作"""
        # 处理逻辑
        records = self.env['main.model'].search([
            ('date_order', '>=', self.date_from),
            ('date_order', '<=', self.date_to),
            ('state', '=', self.state)
        ])
        
        if not records:
            raise UserError("没有找到符合条件的记录")
        
        # 执行操作
        records.write({'state': 'confirmed'})
        
        return {
            'type': 'ir.actions.client',
            'tag': 'display_notification',
            'params': {
                'title': '成功',
                'message': f'已处理 {len(records)} 条记录',
                'type': 'success',
            }
        }
    
    def action_cancel(self):
        """取消操作"""
        return {'type': 'ir.actions.act_window_close'}
```
