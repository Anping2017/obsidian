```python
# 系统资源监控工具
import psutil
import time
from datetime import datetime

class SystemMonitor:
    def __init__(self):
        self.process = psutil.Process()
    
    def get_cpu_usage(self):
        """获取CPU使用率"""
        return self.process.cpu_percent(interval=1)
    
    def get_memory_usage(self):
        """获取内存使用情况"""
        memory_info = self.process.memory_info()
        return {
            'rss': memory_info.rss / 1024 / 1024,  # MB
            'vms': memory_info.vms / 1024 / 1024,  # MB
            'percent': self.process.memory_percent()
        }
    
    def get_disk_usage(self):
        """获取磁盘使用情况"""
        disk_usage = psutil.disk_usage('/')
        return {
            'total': disk_usage.total / 1024 / 1024 / 1024,  # GB
            'used': disk_usage.used / 1024 / 1024 / 1024,    # GB
            'free': disk_usage.free / 1024 / 1024 / 1024,    # GB
            'percent': (disk_usage.used / disk_usage.total) * 100
        }
    
    def get_network_usage(self):
        """获取网络使用情况"""
        network_io = psutil.net_io_counters()
        return {
            'bytes_sent': network_io.bytes_sent / 1024 / 1024,  # MB
            'bytes_recv': network_io.bytes_recv / 1024 / 1024,  # MB
            'packets_sent': network_io.packets_sent,
            'packets_recv': network_io.packets_recv
        }
    
    def monitor_performance(self, duration=60):
        """持续监控性能"""
        start_time = time.time()
        performance_data = []
        
        while time.time() - start_time < duration:
            data = {
                'timestamp': datetime.now().isoformat(),
                'cpu': self.get_cpu_usage(),
                'memory': self.get_memory_usage(),
                'disk': self.get_disk_usage(),
                'network': self.get_network_usage()
            }
            performance_data.append(data)
            time.sleep(5)  # 每5秒采集一次
        
        return performance_data
```

```python
# Odoo应用性能监控
import logging
import time
from functools import wraps

_logger = logging.getLogger(__name__)

class OdooPerformanceMonitor:
    def __init__(self, env):
        self.env = env
        self.performance_data = []
    
    def monitor_method(self, method_name=None):
        """方法性能监控装饰器"""
        def decorator(func):
            @wraps(func)
            def wrapper(*args, **kwargs):
                start_time = time.time()
                method = method_name or func.__name__
                
                try:
                    result = func(*args, **kwargs)
                    execution_time = time.time() - start_time
                    
                    # 记录性能数据
                    self._record_performance(method, execution_time, 'success')
                    
                    return result
                except Exception as e:
                    execution_time = time.time() - start_time
                    self._record_performance(method, execution_time, 'error', str(e))
                    raise
            
            return wrapper
        return decorator
    
    def _record_performance(self, method, execution_time, status, error=None):
        """记录性能数据"""
        data = {
            'timestamp': time.time(),
            'method': method,
            'execution_time': execution_time,
            'status': status,
            'error': error
        }
        self.performance_data.append(data)
        
        # 如果执行时间超过阈值，记录警告
        if execution_time > 5.0:  # 5秒阈值
            _logger.warning(f"慢方法检测: {method} 执行时间 {execution_time:.2f}s")
    
    def get_performance_summary(self):
        """获取性能摘要"""
        if not self.performance_data:
            return {}
        
        total_calls = len(self.performance_data)
        successful_calls = len([d for d in self.performance_data if d['status'] == 'success'])
        failed_calls = total_calls - successful_calls
        
        execution_times = [d['execution_time'] for d in self.performance_data]
        avg_time = sum(execution_times) / len(execution_times)
        max_time = max(execution_times)
        min_time = min(execution_times)
        
        return {
            'total_calls': total_calls,
            'successful_calls': successful_calls,
            'failed_calls': failed_calls,
            'success_rate': (successful_calls / total_calls) * 100,
            'avg_execution_time': avg_time,
            'max_execution_time': max_time,
            'min_execution_time': min_time
        }
    
    def get_slow_methods(self, threshold=1.0):
        """获取慢方法列表"""
        slow_methods = [d for d in self.performance_data if d['execution_time'] > threshold]
        return sorted(slow_methods, key=lambda x: x['execution_time'], reverse=True)
```

```python
# 内存使用问题诊断
import gc
import tracemalloc

class MemoryDiagnostic:
    def __init__(self):
        self.memory_snapshots = []
    
    def start_memory_tracing(self):
        """开始内存跟踪"""
        tracemalloc.start()
    
    def take_memory_snapshot(self, label):
        """拍摄内存快照"""
        snapshot = tracemalloc.take_snapshot()
        self.memory_snapshots.append({
            'label': label,
            'snapshot': snapshot,
            'timestamp': time.time()
        })
    
    def analyze_memory_growth(self):
        """分析内存增长"""
        if len(self.memory_snapshots) < 2:
            return "需要至少两个快照才能分析内存增长"
        
        current = self.memory_snapshots[-1]
        previous = self.memory_snapshots[-2]
        
        current_memory = current['snapshot'].statistics('lineno')
        previous_memory = previous['snapshot'].statistics('lineno')
        
        # 计算内存增长
        memory_growth = []
        for current_stat in current_memory:
            for previous_stat in previous_memory:
                if current_stat.traceback.format() == previous_stat.traceback.format():
                    growth = current_stat.size - previous_stat.size
                    if growth > 0:
                        memory_growth.append({
                            'location': current_stat.traceback.format()[0],
                            'growth': growth,
                            'current_size': current_stat.size,
                            'previous_size': previous_stat.size
                        })
        
        return sorted(memory_growth, key=lambda x: x['growth'], reverse=True)
    
    def detect_memory_leaks(self):
        """检测内存泄漏"""
        gc.collect()  # 强制垃圾回收
        
        # 获取当前内存使用
        process = psutil.Process()
        memory_info = process.memory_info()
        
        # 检查内存使用是否异常
        if memory_info.rss > 1024 * 1024 * 1024:  # 1GB阈值
            return {
                'status': 'high_memory_usage',
                'memory_mb': memory_info.rss / 1024 / 1024,
                'recommendations': [
                    '检查是否有内存泄漏',
                    '优化数据结构',
                    '使用生成器替代列表',
                    '及时释放大对象'
                ]
            }
        else:
            return {
                'status': 'normal',
                'memory_mb': memory_info.rss / 1024 / 1024
            }
    
    def get_memory_usage_by_object(self):
        """按对象类型获取内存使用"""
        gc.collect()
        objects = gc.get_objects()
        
        object_counts = {}
        for obj in objects:
            obj_type = type(obj).__name__
            if obj_type not in object_counts:
                object_counts[obj_type] = 0
            object_counts[obj_type] += 1
        
        return sorted(object_counts.items(), key=lambda x: x[1], reverse=True)
```

```python
        """优化Redis缓存"""
        return {
            'redis_config': {
                'maxmemory': '256mb',
                'maxmemory-policy': 'allkeys-lru',
                'timeout': 300,
                'tcp-keepalive': 60
            },
            'cache_strategies': [
                '使用LRU淘汰策略',
                '设置合理的过期时间',
                '监控缓存命中率',
                '使用缓存预热'
            ]
        }
    
    def implement_memory_cache(self):
        """实现内存缓存"""
        class MemoryCache:
            def __init__(self, max_size=1000):
                self.cache = {}
                self.max_size = max_size
                self.access_count = {}
            
            def get(self, key):
                if key in self.cache:
                    self.access_count[key] = self.access_count.get(key, 0) + 1
                    return self.cache[key]
                return None
            
            def set(self, key, value):
                if len(self.cache) >= self.max_size:
                    # LRU淘汰
                    least_used = min(self.access_count.items(), key=lambda x: x[1])
                    del self.cache[least_used[0]]
                    del self.access_count[least_used[0]]
                
                self.cache[key] = value
                self.access_count[key] = 1
        
        return MemoryCache()
```

```python
class PerformanceAlert:

            alerts.append({
                'type': 'response_time',
                'message': f"平均响应时间 {metrics['avg_response_time']:.2f}s 超过阈值 {self.alert_rules['response_time']}s",
                'severity': 'warning'
            })
        
        # 内存使用告警
        if metrics.get('memory_percent', 0) > self.alert_rules['memory_usage']:
            alerts.append({
                'type': 'memory_usage',
                'message': f"内存使用率 {metrics['memory_percent']:.1f}% 超过阈值 {self.alert_rules['memory_usage']}%",
                'severity': 'critical'
            })
        
        # CPU使用告警
        if metrics.get('cpu_percent', 0) > self.alert_rules['cpu_usage']:
            alerts.append({
                'type': 'cpu_usage',
                'message': f"CPU使用率 {metrics['cpu_percent']:.1f}% 超过阈值 {self.alert_rules['cpu_usage']}%",
                'severity': 'critical'
            })
        
        # 错误率告警
        if metrics.get('error_rate', 0) > self.alert_rules['error_rate']:
            alerts.append({
                'type': 'error_rate',
                'message': f"错误率 {metrics['error_rate']:.1f}% 超过阈值 {self.alert_rules['error_rate']}%",
                'severity': 'warning'
            })
        
        return alerts
    
    def send_alert(self, alert):
        """发送告警"""
        # 这里可以实现发送邮件、短信、Slack等告警
        _logger.warning(f"性能告警: {alert['message']}")
        
        # 记录到数据库
        self.env['performance.alert'].create({
            'alert_type': alert['type'],
            'message': alert['message'],
            'severity': alert['severity'],
            'timestamp': fields.Datetime.now()
        })
```
