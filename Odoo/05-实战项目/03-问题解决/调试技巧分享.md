本文档分享了Odoo 17开发中的调试技巧，包括日志调试、断点调试、性能调试、前端调试等方法，帮助开发者快速定位和解决问题。

```python
from odoo import models, fields, api
import logging

_logger = logging.getLogger(__name__)

class DebugModel(models.Model):
    """调试示例模型"""
    _name = 'debug.model'
    _description = 'Debug Model'
    
    name = fields.Char('名称')
    value = fields.Float('值')

        """调试方法示例"""
        # 1. 基础日志记录
        _logger.info("开始执行调试方法")
        
        # 2. 记录变量值
        _logger.info(f"当前记录: {self.name}, 值: {self.value}")
        
        # 3. 记录方法参数
        def process_data(self, data, options=None):
            _logger.info(f"处理数据: {data}, 选项: {options}")
            
            try:
                # 业务逻辑
                result = self._process_logic(data, options)
                _logger.info(f"处理成功: {result}")
                return result

                _logger.error(f"处理失败: {e}", exc_info=True)
                raise
```

```python
# models/structured_debug.py
from odoo import models, fields, api
import logging
import json
from datetime import datetime

_logger = logging.getLogger(__name__)

class StructuredDebugModel(models.Model):
    """结构化调试示例"""
    _name = 'structured.debug.model'
    _description = 'Structured Debug Model'
    
    name = fields.Char('名称')
    data = fields.Text('数据')
    
    def structured_debug(self, operation, **kwargs):
        """结构化调试方法"""
        # 构建调试信息
        debug_info = {
            'timestamp': datetime.now().isoformat(),
            'model': self._name,
            'record_id': self.id,
            'operation': operation,
            'user_id': self.env.user.id,
            'context': self.env.context,
            **kwargs
        }
        
        # 记录结构化日志
        _logger.info(f"调试信息: {json.dumps(debug_info, ensure_ascii=False, indent=2)}")
        
        # 执行操作
        try:
            result = self._execute_operation(operation, **kwargs)
            debug_info['result'] = result
            debug_info['status'] = 'success'
            _logger.info(f"操作成功: {json.dumps(debug_info, ensure_ascii=False)}")
            return result
        except Exception as e:
            debug_info['error'] = str(e)
            debug_info['status'] = 'failed'
            _logger.error(f"操作失败: {json.dumps(debug_info, ensure_ascii=False)}", exc_info=True)
            raise
    
    def _execute_operation(self, operation, **kwargs):
        """执行操作"""
        if operation == 'create':
            return self._create_operation(**kwargs)
        elif operation == 'update':
            return self._update_operation(**kwargs)
        else:
            raise ValueError(f"未知操作: {operation}")
    
    def _create_operation(self, **kwargs):
        """创建操作"""
        _logger.debug("执行创建操作")
        return "创建成功"
    
    def _update_operation(self, **kwargs):
        """更新操作"""
        _logger.debug("执行更新操作")
        return "更新成功"
```

```python
# models/conditional_debug.py
from odoo import models, fields, api
import logging

_logger = logging.getLogger(__name__)

class ConditionalDebugModel(models.Model):
    """条件调试示例"""
    _name = 'conditional.debug.model'
    _description = 'Conditional Debug Model'
    
    name = fields.Char('名称')
    debug_mode = fields.Boolean('调试模式', default=False)
    
    def conditional_debug_method(self):
        """条件调试方法"""
        # 1. 基于字段的条件调试
        if self.debug_mode:
            _logger.info(f"调试模式开启 - 记录: {self.name}")
        
        # 2. 基于环境的条件调试
        if self.env.context.get('debug'):
            _logger.info(f"环境调试模式 - 上下文: {self.env.context}")
        
        # 3. 基于配置的条件调试
        debug_level = self.env['ir.config_parameter'].sudo().get_param('debug.level', 'info')
        if debug_level == 'debug':
            _logger.debug("详细调试信息")
        elif debug_level == 'info':
            _logger.info("一般调试信息")
        
        # 4. 基于用户组的条件调试
        if self.env.user.has_group('base.group_system'):
            _logger.info("系统管理员调试信息")
        
        # 5. 基于方法的条件调试
        self._debug_if_needed()
    
    def _debug_if_needed(self):
        """根据需要调试"""
        # 检查是否需要调试
        if self._should_debug():
            _logger.info("执行详细调试")
            self._detailed_debug()
        else:
            _logger.info("执行简单调试")
            self._simple_debug()
    
    def _should_debug(self):
        """判断是否需要调试"""
        return (
            self.debug_mode or
            self.env.context.get('debug') or
            self.env.user.has_group('base.group_system')
        )
    
    def _detailed_debug(self):
        """详细调试"""
        _logger.info(f"详细调试 - 记录ID: {self.id}")
        _logger.info(f"详细调试 - 记录名称: {self.name}")
        _logger.info(f"详细调试 - 用户: {self.env.user.name}")
        _logger.info(f"详细调试 - 公司: {self.env.company.name}")
    
    def _simple_debug(self):
        """简单调试"""
        _logger.info(f"简单调试 - {self.name}")
```

```python
# models/pdb_debug.py
from odoo import models, fields, api
import pdb

class PdbDebugModel(models.Model):
    """Python调试器示例"""
    _name = 'pdb.debug.model'
    _description = 'Pdb Debug Model'
    
    name = fields.Char('名称')
    value = fields.Float('值')
    
    def pdb_debug_method(self):
        """使用pdb调试"""
        # 设置断点
        pdb.set_trace()
        
        # 执行代码
        result = self._calculate_value()
        
        # 另一个断点
        pdb.set_trace()
        
        return result
    
    def _calculate_value(self):
        """计算值"""
        # 复杂的计算逻辑
        intermediate = self.value * 2
        result = intermediate + 10
        return result
    
    def conditional_pdb_debug(self):
        """条件断点调试"""
        # 只在特定条件下设置断点
        if self.env.context.get('debug_pdb'):
            pdb.set_trace()
        
        # 执行逻辑
        return self._process_data()
    
    def _process_data(self):
        """处理数据"""
        data = []
        for i in range(10):
            # 在循环中设置断点
            if i == 5:
                pdb.set_trace()
            data.append(i * 2)
        return data
```

```python
# models/ipdb_debug.py
from odoo import models, fields, api
try:
    import ipdb as pdb
except ImportError:
    import pdb

class IpdbDebugModel(models.Model):
    """IPython调试器示例"""
    _name = 'ipdb.debug.model'
    _description = 'Ipdb Debug Model'
    
    name = fields.Char('名称')
    data = fields.Text('数据')
    
    def ipdb_debug_method(self):
        """使用ipdb调试"""
        # 设置断点
        pdb.set_trace()
        
        # 执行代码
        result = self._process_complex_data()
        
        return result
    
    def _process_complex_data(self):
        """处理复杂数据"""
        # 解析数据
        import json
        try:
            data = json.loads(self.data)
        except json.JSONDecodeError:
            pdb.set_trace()  # 在错误处设置断点
            return None
        
        # 处理数据
        processed_data = []
        for item in data:
            # 在循环中设置断点
            pdb.set_trace()
            processed_item = self._process_item(item)
            processed_data.append(processed_item)
        
        return processed_data
    
    def _process_item(self, item):
        """处理单个项目"""
        # 复杂的处理逻辑
        if isinstance(item, dict):
            return {k: v * 2 for k, v in item.items()}
        elif isinstance(item, list):
            return [x * 2 for x in item]
        else:
            return item * 2
```

```python
# models/performance_debug.py
from odoo import models, fields, api
import time
import logging
from functools import wraps

_logger = logging.getLogger(__name__)

class PerformanceDebugModel(models.Model):
    """性能调试示例"""
    _name = 'performance.debug.model'
    _description = 'Performance Debug Model'
    
    name = fields.Char('名称')
    data = fields.Text('数据')
    
    def performance_debug_method(self):
        """性能调试方法"""
        # 1. 简单时间测量
        start_time = time.time()
        
        # 执行操作
        result = self._expensive_operation()
        
        end_time = time.time()
        execution_time = end_time - start_time
        
        _logger.info(f"操作执行时间: {execution_time:.2f}秒")
        
        return result
    
    def _expensive_operation(self):
        """昂贵的操作"""
        # 模拟耗时操作
        time.sleep(0.1)
        return "操作完成"
    
    def detailed_performance_debug(self):
        """详细性能调试"""
        # 记录开始时间
        start_time = time.time()
        _logger.info("开始性能调试")
        
        # 步骤1
        step1_start = time.time()
        result1 = self._step1()
        step1_time = time.time() - step1_start
        _logger.info(f"步骤1执行时间: {step1_time:.2f}秒")
        
        # 步骤2
        step2_start = time.time()
        result2 = self._step2(result1)
        step2_time = time.time() - step2_start
        _logger.info(f"步骤2执行时间: {step2_time:.2f}秒")
        
        # 步骤3
        step3_start = time.time()
        result3 = self._step3(result2)
        step3_time = time.time() - step3_start
        _logger.info(f"步骤3执行时间: {step3_time:.2f}秒")
        
        # 总时间
        total_time = time.time() - start_time
        _logger.info(f"总执行时间: {total_time:.2f}秒")
        
        return result3
    
    def _step1(self):
        """步骤1"""
        time.sleep(0.05)
        return "步骤1完成"
    
    def _step2(self, input_data):
        """步骤2"""
        time.sleep(0.03)
        return f"步骤2处理: {input_data}"
    
    def _step3(self, input_data):
        """步骤3"""
        time.sleep(0.02)
        return f"步骤3完成: {input_data}"
    
    # 装饰器方式
    def timed_method(self):
        """使用装饰器计时的方法"""
        return self._timed_operation()
    
    def _timed_operation(self):
        """计时操作"""
        time.sleep(0.1)
        return "计时操作完成"

# 性能调试装饰器
def time_debug(func):
    """时间调试装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start_time = time.time()
        result = func(*args, **kwargs)
        end_time = time.time()
        execution_time = end_time - start_time
        
        _logger.info(f"方法 {func.__name__} 执行时间: {execution_time:.2f}秒")
        return result
    return wrapper

# 使用装饰器
class TimedDebugModel(models.Model):
    """计时调试示例"""
    _name = 'timed.debug.model'
    _description = 'Timed Debug Model'
    
    @time_debug
    def timed_method(self):
        """计时方法"""

        return "方法执行完成"
```

```python
# models/memory_debug.py
from odoo import models, fields, api
import psutil
import logging
import gc

_logger = logging.getLogger(__name__)

class MemoryDebugModel(models.Model):
    """内存调试示例"""
    _name = 'memory.debug.model'
    _description = 'Memory Debug Model'
    
    name = fields.Char('名称')
    data = fields.Text('数据')
    
    def memory_debug_method(self):
        """内存调试方法"""
        # 获取当前进程
        process = psutil.Process()
        
        # 记录初始内存使用
        initial_memory = process.memory_info().rss / 1024 / 1024  # MB
        _logger.info(f"初始内存使用: {initial_memory:.2f} MB")
        
        # 执行操作
        result = self._memory_intensive_operation()
        
        # 记录操作后内存使用
        after_memory = process.memory_info().rss / 1024 / 1024  # MB
        _logger.info(f"操作后内存使用: {after_memory:.2f} MB")
        _logger.info(f"内存增长: {after_memory - initial_memory:.2f} MB")
        
        # 强制垃圾回收
        gc.collect()
        
        # 记录垃圾回收后内存使用
        final_memory = process.memory_info().rss / 1024 / 1024  # MB
        _logger.info(f"垃圾回收后内存使用: {final_memory:.2f} MB")
        
        return result
    
    def _memory_intensive_operation(self):
        """内存密集型操作"""
        # 创建大量数据
        large_data = []
        for i in range(10000):
            large_data.append({
                'id': i,
                'name': f'Item {i}',
                'data': 'x' * 1000  # 1KB数据
            })
        
        # 处理数据
        processed_data = []
        for item in large_data:
            processed_item = {
                'id': item['id'],
                'name': item['name'].upper(),
                'data_length': len(item['data'])
            }
            processed_data.append(processed_item)
        
        return processed_data
    
    def detailed_memory_debug(self):
        """详细内存调试"""
        process = psutil.Process()
        
        # 记录详细内存信息
        memory_info = process.memory_info()
        _logger.info(f"RSS内存: {memory_info.rss / 1024 / 1024:.2f} MB")
        _logger.info(f"VMS内存: {memory_info.vms / 1024 / 1024:.2f} MB")
        
        # 记录内存百分比
        memory_percent = process.memory_percent()
        _logger.info(f"内存使用百分比: {memory_percent:.2f}%")
        
        # 记录系统内存信息
        system_memory = psutil.virtual_memory()
        _logger.info(f"系统总内存: {system_memory.total / 1024 / 1024 / 1024:.2f} GB")
        _logger.info(f"系统可用内存: {system_memory.available / 1024 / 1024 / 1024:.2f} GB")
        _logger.info(f"系统内存使用率: {system_memory.percent:.2f}%")
        
        return "内存调试完成"
```

```python
# models/sql_debug.py
from odoo import models, fields, api
import logging

_logger = logging.getLogger(__name__)

class SqlDebugModel(models.Model):
    """SQL调试示例"""
    _name = 'sql.debug.model'
    _description = 'SQL Debug Model'
    
    name = fields.Char('名称')
    value = fields.Float('值')
    
    def sql_debug_method(self):
        """SQL调试方法"""
        # 1. 启用SQL日志
        self.env.cr.execute("SET log_statement = 'all'")
        
        # 2. 执行查询并记录
        query = "SELECT * FROM sql_debug_model WHERE value > %s"
        params = (100,)
        
        _logger.info(f"执行SQL查询: {query}")
        _logger.info(f"查询参数: {params}")
        
        self.env.cr.execute(query, params)
        results = self.env.cr.fetchall()
        
        _logger.info(f"查询结果: {results}")
        
        return results
    
    def explain_query(self):
        """解释查询计划"""
        query = """
        EXPLAIN ANALYZE 
        SELECT * FROM sql_debug_model 
        WHERE value > 100 
        ORDER BY name
        """
        
        _logger.info(f"解释查询: {query}")
        
        self.env.cr.execute(query)
        explain_result = self.env.cr.fetchall()
        
        _logger.info("查询计划:")
        for row in explain_result:
            _logger.info(f"  {row[0]}")
        
        return explain_result
    
    def slow_query_debug(self):
        """慢查询调试"""
        # 启用慢查询日志
        self.env.cr.execute("SET log_min_duration_statement = 1000")  # 1秒
        
        # 执行可能慢的查询
        start_time = time.time()
        
        query = """
        SELECT sdm1.*, sdm2.name as partner_name
        FROM sql_debug_model sdm1
        LEFT JOIN sql_debug_model sdm2 ON sdm1.partner_id = sdm2.id
        WHERE sdm1.value > 100
        ORDER BY sdm1.name
        """
        
        self.env.cr.execute(query)
        results = self.env.cr.fetchall()
        
        end_time = time.time()
        execution_time = end_time - start_time
        
        _logger.info(f"查询执行时间: {execution_time:.2f}秒")
        
        if execution_time > 1.0:
            _logger.warning(f"慢查询检测: {execution_time:.2f}秒")
            _logger.warning(f"查询: {query}")
        
        return results
```

```python
from odoo import models, fields, api
import logging

_logger = logging.getLogger(__name__)

class TransactionDebugModel(models.Model):
    """事务调试示例"""
    _name = 'transaction.debug.model'
    _description = 'Transaction Debug Model'
    
    name = fields.Char('名称')
    value = fields.Float('值')
    
    def transaction_debug_method(self):
        """事务调试方法"""
        # 1. 检查事务状态
        _logger.info(f"事务状态: {self.env.cr.get_transaction_status()}")
        
        # 2. 记录事务开始
        _logger.info("开始事务操作")
        
        try:
            # 执行操作
            result = self._transaction_operation()
            
            # 提交事务
            self.env.cr.commit()
            _logger.info("事务提交成功")
            
            return result
            
        except Exception as e:
            # 回滚事务
            self.env.cr.rollback()
            _logger.error(f"事务回滚: {e}")
            raise
    
    def _transaction_operation(self):
        """事务操作"""
        # 创建记录
        record1 = self.create({
            'name': '记录1',
            'value': 100
        })
        _logger.info(f"创建记录1: {record1.id}")
        
        # 更新记录
        record1.write({'value': 200})
        _logger.info(f"更新记录1: {record1.value}")
        
        # 创建另一个记录
        record2 = self.create({
            'name': '记录2',
            'value': 300
        })
        _logger.info(f"创建记录2: {record2.id}")
        
        return [record1.id, record2.id]
    
    def nested_transaction_debug(self):
        """嵌套事务调试"""
        _logger.info("开始嵌套事务操作")
        
        # 外层事务
        try:
            result1 = self._outer_transaction()
            
            # 内层事务
            try:
                result2 = self._inner_transaction()
                _logger.info("内层事务成功")
            except Exception as e:
                _logger.error(f"内层事务失败: {e}")
                raise
            
            _logger.info("外层事务成功")
            return [result1, result2]
            
        except Exception as e:
            _logger.error(f"外层事务失败: {e}")
            raise
    
    def _outer_transaction(self):
        """外层事务"""
        record = self.create({
            'name': '外层记录',
            'value': 100
        })
        _logger.info(f"外层事务创建记录: {record.id}")
        return record.id
    
    def _inner_transaction(self):
        """内层事务"""
        record = self.create({
            'name': '内层记录',
            'value': 200
        })
        _logger.info(f"内层事务创建记录: {record.id}")
        return record.id
```
