# 简单计算器编译器

## 🎯 项目概述

**简单计算器编译器**是一个完整的编译器实践项目，实现了一个能够编译简单算术表达式的编译器。该项目涵盖了编译原理的核心概念，包括词法分析、语法分析、语义分析、中间代码生成和目标代码生成。

## 🔍 项目目标

### 学习目标
- **理解编译过程**：掌握编译器的完整工作流程
- **实践编译技术**：实现词法分析、语法分析等核心模块
- **代码生成**：学会生成中间代码和目标代码
- **错误处理**：实现编译错误检测和报告

### 功能目标
- 支持基本算术运算（+、-、*、/）
- 支持括号和优先级
- 支持变量赋值和引用
- 生成可执行的目标代码

## 📊 项目架构

### 编译器模块结构
```c
class SimpleCalculatorCompiler {
private:
    LexicalAnalyzer* lexer;
    SyntaxAnalyzer* parser;
    SemanticAnalyzer* semanticAnalyzer;
    CodeGenerator* codeGenerator;
    ErrorHandler* errorHandler;
    
public:
    SimpleCalculatorCompiler() {
        initializeCompiler();
    }
    
    void initializeCompiler() {
        cout << "初始化简单计算器编译器" << endl;
        
        lexer = new LexicalAnalyzer();
        parser = new SyntaxAnalyzer();
        semanticAnalyzer = new SemanticAnalyzer();
        codeGenerator = new CodeGenerator();
        errorHandler = new ErrorHandler();
        
        // 初始化各个模块
        lexer->initialize();
        parser->initialize();
        semanticAnalyzer->initialize();
        codeGenerator->initialize();
        errorHandler->initialize();
    }
    
    bool compile(const string& sourceCode) {
        cout << "开始编译源代码" << endl;
        cout << "===========================================" << endl;
        
        try {
            // 词法分析
            vector<Token> tokens = lexer->analyze(sourceCode);
            if (tokens.empty()) {
                cout << "词法分析失败" << endl;
                return false;
            }
            
            // 语法分析
            ASTNode* ast = parser->parse(tokens);
            if (!ast) {
                cout << "语法分析失败" << endl;
                return false;
            }
            
            // 语义分析
            if (!semanticAnalyzer->analyze(ast)) {
                cout << "语义分析失败" << endl;
                return false;
            }
            
            // 代码生成
            string targetCode = codeGenerator->generate(ast);
            if (targetCode.empty()) {
                cout << "代码生成失败" << endl;
                return false;
            }
            
            cout << "编译成功" << endl;
            cout << "生成的目标代码:" << endl;
            cout << targetCode << endl;
            
            return true;
            
        } catch (const exception& e) {
            errorHandler->reportError("编译错误: " + string(e.what()));
            return false;
        }
    }
    
    void cleanup() {
        cout << "清理编译器" << endl;
        
        if (lexer) {
            delete lexer;
            lexer = nullptr;
        }
        
        if (parser) {
            delete parser;
            parser = nullptr;
        }
        
        if (semanticAnalyzer) {
            delete semanticAnalyzer;
            semanticAnalyzer = nullptr;
        }
        
        if (codeGenerator) {
            delete codeGenerator;
            codeGenerator = nullptr;
        }
        
        if (errorHandler) {
            delete errorHandler;
            errorHandler = nullptr;
        }
        
        cout << "编译器清理完成" << endl;
    }
    
    ~SimpleCalculatorCompiler() {
        cleanup();
    }
};
```

## 🔧 词法分析器

### 词法分析器实现
```c
class LexicalAnalyzer {
private:
    string sourceCode;
    size_t currentPos;
    vector<Token> tokens;
    
public:
    LexicalAnalyzer() : currentPos(0) {}
    
    void initialize() {
        cout << "初始化词法分析器" << endl;
    }
    
    vector<Token> analyze(const string& code) {
        cout << "开始词法分析" << endl;
        cout << "-------------------------------------------" << endl;
        
        sourceCode = code;
        currentPos = 0;
        tokens.clear();
        
        while (currentPos < sourceCode.length()) {
            char currentChar = sourceCode[currentPos];
            
            if (isspace(currentChar)) {
                currentPos++;
                continue;
            }
            
            if (isdigit(currentChar)) {
                Token token = scanNumber();
                tokens.push_back(token);
            } else if (isalpha(currentChar)) {
                Token token = scanIdentifier();
                tokens.push_back(token);
            } else if (currentChar == '+' || currentChar == '-' || 
                      currentChar == '*' || currentChar == '/') {
                Token token = scanOperator();
                tokens.push_back(token);
            } else if (currentChar == '(' || currentChar == ')') {
                Token token = scanDelimiter();
                tokens.push_back(token);
            } else if (currentChar == '=') {
                Token token = scanAssignment();
                tokens.push_back(token);
            } else {
                cout << "未知字符: " << currentChar << endl;
                currentPos++;
            }
        }
        
        // 添加结束标记
        Token eofToken;
        eofToken.type = TOKEN_EOF;
        eofToken.value = "EOF";
        tokens.push_back(eofToken);
        
        cout << "词法分析完成，共识别 " << tokens.size() << " 个词法单元" << endl;
        return tokens;
    }
    
    Token scanNumber() {
        string value;
        
        while (currentPos < sourceCode.length() && isdigit(sourceCode[currentPos])) {
            value += sourceCode[currentPos];
            currentPos++;
        }
        
        Token token;
        token.type = TOKEN_NUMBER;
        token.value = value;
        
        cout << "识别数字: " << value << endl;
        return token;
    }
    
    Token scanIdentifier() {
        string value;
        
        while (currentPos < sourceCode.length() && 
               (isalnum(sourceCode[currentPos]) || sourceCode[currentPos] == '_')) {
            value += sourceCode[currentPos];
            currentPos++;
        }
        
        Token token;
        token.type = TOKEN_IDENTIFIER;
        token.value = value;
        
        cout << "识别标识符: " << value << endl;
        return token;
    }
    
    Token scanOperator() {
        char op = sourceCode[currentPos];
        currentPos++;
        
        Token token;
        token.type = TOKEN_OPERATOR;
        token.value = string(1, op);
        
        cout << "识别操作符: " << op << endl;
        return token;
    }
    
    Token scanDelimiter() {
        char delim = sourceCode[currentPos];
        currentPos++;
        
        Token token;
        token.type = TOKEN_DELIMITER;
        token.value = string(1, delim);
        
        cout << "识别分隔符: " << delim << endl;
        return token;
    }
    
    Token scanAssignment() {
        currentPos++;
        
        Token token;
        token.type = TOKEN_ASSIGNMENT;
        token.value = "=";
        
        cout << "识别赋值操作符: =" << endl;
        return token;
    }
    
    void printTokens() {
        cout << "词法单元列表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (size_t i = 0; i < tokens.size(); i++) {
            const Token& token = tokens[i];
            cout << "[" << i << "] " << getTokenTypeName(token.type) 
                 << " -> " << token.value << endl;
        }
        
        cout << endl;
    }
    
    string getTokenTypeName(TokenType type) {
        switch (type) {
            case TOKEN_NUMBER: return "数字";
            case TOKEN_IDENTIFIER: return "标识符";
            case TOKEN_OPERATOR: return "操作符";
            case TOKEN_DELIMITER: return "分隔符";
            case TOKEN_ASSIGNMENT: return "赋值";
            case TOKEN_EOF: return "结束";
            default: return "未知";
        }
    }
};

enum TokenType {
    TOKEN_NUMBER,
    TOKEN_IDENTIFIER,
    TOKEN_OPERATOR,
    TOKEN_DELIMITER,
    TOKEN_ASSIGNMENT,
    TOKEN_EOF
};

struct Token {
    TokenType type;
    string value;
};
```

## 🔧 语法分析器

### 语法分析器实现
```c
class SyntaxAnalyzer {
private:
    vector<Token> tokens;
    size_t currentTokenIndex;
    
public:
    SyntaxAnalyzer() : currentTokenIndex(0) {}
    
    void initialize() {
        cout << "初始化语法分析器" << endl;
    }
    
    ASTNode* parse(const vector<Token>& tokenList) {
        cout << "开始语法分析" << endl;
        cout << "-------------------------------------------" << endl;
        
        tokens = tokenList;
        currentTokenIndex = 0;
        
        ASTNode* ast = parseExpression();
        
        if (currentTokenIndex < tokens.size() - 1) {
            cout << "语法分析未完成" << endl;
            return nullptr;
        }
        
        cout << "语法分析完成" << endl;
        return ast;
    }
    
    ASTNode* parseExpression() {
        cout << "解析表达式" << endl;
        
        ASTNode* left = parseTerm();
        
        while (currentTokenIndex < tokens.size() && 
               (getCurrentToken().type == TOKEN_OPERATOR && 
                (getCurrentToken().value == "+" || getCurrentToken().value == "-"))) {
            
            Token op = getCurrentToken();
            currentTokenIndex++;
            
            ASTNode* right = parseTerm();
            
            ASTNode* node = new ASTNode();
            node->type = AST_BINARY_OP;
            node->value = op.value;
            node->left = left;
            node->right = right;
            
            left = node;
        }
        
        return left;
    }
    
    ASTNode* parseTerm() {
        cout << "解析项" << endl;
        
        ASTNode* left = parseFactor();
        
        while (currentTokenIndex < tokens.size() && 
               (getCurrentToken().type == TOKEN_OPERATOR && 
                (getCurrentToken().value == "*" || getCurrentToken().value == "/"))) {
            
            Token op = getCurrentToken();
            currentTokenIndex++;
            
            ASTNode* right = parseFactor();
            
            ASTNode* node = new ASTNode();
            node->type = AST_BINARY_OP;
            node->value = op.value;
            node->left = left;
            node->right = right;
            
            left = node;
        }
        
        return left;
    }
    
    ASTNode* parseFactor() {
        cout << "解析因子" << endl;
        
        Token current = getCurrentToken();
        
        if (current.type == TOKEN_NUMBER) {
            currentTokenIndex++;
            
            ASTNode* node = new ASTNode();
            node->type = AST_NUMBER;
            node->value = current.value;
            
            return node;
        } else if (current.type == TOKEN_IDENTIFIER) {
            currentTokenIndex++;
            
            ASTNode* node = new ASTNode();
            node->type = AST_IDENTIFIER;
            node->value = current.value;
            
            return node;
        } else if (current.type == TOKEN_DELIMITER && current.value == "(") {
            currentTokenIndex++; // 跳过 '('
            
            ASTNode* node = parseExpression();
            
            if (getCurrentToken().type == TOKEN_DELIMITER && getCurrentToken().value == ")") {
                currentTokenIndex++; // 跳过 ')'
                return node;
            } else {
                cout << "缺少右括号" << endl;
                return nullptr;
            }
        } else {
            cout << "语法错误: 意外的词法单元 " << current.value << endl;
            return nullptr;
        }
    }
    
    Token getCurrentToken() {
        if (currentTokenIndex < tokens.size()) {
            return tokens[currentTokenIndex];
        } else {
            Token eofToken;
            eofToken.type = TOKEN_EOF;
            eofToken.value = "EOF";
            return eofToken;
        }
    }
    
    void printAST(ASTNode* node, int depth = 0) {
        if (!node) return;
        
        string indent(depth * 2, ' ');
        cout << indent << "节点类型: " << getASTTypeName(node->type) 
             << ", 值: " << node->value << endl;
        
        if (node->left) {
            cout << indent << "左子树:" << endl;
            printAST(node->left, depth + 1);
        }
        
        if (node->right) {
            cout << indent << "右子树:" << endl;
            printAST(node->right, depth + 1);
        }
    }
    
    string getASTTypeName(ASTType type) {
        switch (type) {
            case AST_NUMBER: return "数字";
            case AST_IDENTIFIER: return "标识符";
            case AST_BINARY_OP: return "二元操作";
            default: return "未知";
        }
    }
};

enum ASTType {
    AST_NUMBER,
    AST_IDENTIFIER,
    AST_BINARY_OP
};

struct ASTNode {
    ASTType type;
    string value;
    ASTNode* left;
    ASTNode* right;
    
    ASTNode() : left(nullptr), right(nullptr) {}
};
```

## 🔧 语义分析器

### 语义分析器实现
```c
class SemanticAnalyzer {
private:
    map<string, int> symbolTable;
    vector<string> errors;
    
public:
    SemanticAnalyzer() {}
    
    void initialize() {
        cout << "初始化语义分析器" << endl;
    }
    
    bool analyze(ASTNode* ast) {
        cout << "开始语义分析" << endl;
        cout << "-------------------------------------------" << endl;
        
        errors.clear();
        
        if (!ast) {
            errors.push_back("AST为空");
            return false;
        }
        
        bool result = analyzeNode(ast);
        
        if (result) {
            cout << "语义分析成功" << endl;
        } else {
            cout << "语义分析失败，发现 " << errors.size() << " 个错误" << endl;
            for (const auto& error : errors) {
                cout << "错误: " << error << endl;
            }
        }
        
        return result;
    }
    
    bool analyzeNode(ASTNode* node) {
        if (!node) return true;
        
        switch (node->type) {
            case AST_NUMBER:
                return analyzeNumber(node);
            case AST_IDENTIFIER:
                return analyzeIdentifier(node);
            case AST_BINARY_OP:
                return analyzeBinaryOp(node);
            default:
                errors.push_back("未知的AST节点类型");
                return false;
        }
    }
    
    bool analyzeNumber(ASTNode* node) {
        // 检查数字是否有效
        try {
            int value = stoi(node->value);
            cout << "分析数字: " << value << endl;
            return true;
        } catch (const exception& e) {
            errors.push_back("无效的数字: " + node->value);
            return false;
        }
    }
    
    bool analyzeIdentifier(ASTNode* node) {
        // 检查标识符是否已定义
        if (symbolTable.find(node->value) == symbolTable.end()) {
            errors.push_back("未定义的变量: " + node->value);
            return false;
        }
        
        cout << "分析标识符: " << node->value << " = " << symbolTable[node->value] << endl;
        return true;
    }
    
    bool analyzeBinaryOp(ASTNode* node) {
        cout << "分析二元操作: " << node->value << endl;
        
        // 分析左操作数
        if (!analyzeNode(node->left)) {
            return false;
        }
        
        // 分析右操作数
        if (!analyzeNode(node->right)) {
            return false;
        }
        
        // 检查操作符是否有效
        if (node->value != "+" && node->value != "-" && 
            node->value != "*" && node->value != "/") {
            errors.push_back("无效的操作符: " + node->value);
            return false;
        }
        
        // 检查除零错误
        if (node->value == "/" && node->right->type == AST_NUMBER) {
            try {
                int rightValue = stoi(node->right->value);
                if (rightValue == 0) {
                    errors.push_back("除零错误");
                    return false;
                }
            } catch (const exception& e) {
                // 忽略转换错误，因为analyzeNumber已经检查过了
            }
        }
        
        return true;
    }
    
    void addVariable(const string& name, int value) {
        symbolTable[name] = value;
        cout << "添加变量: " << name << " = " << value << endl;
    }
    
    void printSymbolTable() {
        cout << "符号表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (const auto& pair : symbolTable) {
            cout << "变量: " << pair.first << " = " << pair.second << endl;
        }
        
        cout << endl;
    }
    
    int getVariableValue(const string& name) {
        if (symbolTable.find(name) != symbolTable.end()) {
            return symbolTable[name];
        }
        return 0;
    }
};
```

## 🔧 代码生成器

### 代码生成器实现
```c
class CodeGenerator {
private:
    vector<string> instructions;
    int tempVarCount;
    
public:
    CodeGenerator() : tempVarCount(0) {}
    
    void initialize() {
        cout << "初始化代码生成器" << endl;
        instructions.clear();
        tempVarCount = 0;
    }
    
    string generate(ASTNode* ast) {
        cout << "开始代码生成" << endl;
        cout << "-------------------------------------------" << endl;
        
        if (!ast) {
            cout << "AST为空，无法生成代码" << endl;
            return "";
        }
        
        string result = generateNode(ast);
        
        cout << "代码生成完成" << endl;
        return result;
    }
    
    string generateNode(ASTNode* node) {
        if (!node) return "";
        
        switch (node->type) {
            case AST_NUMBER:
                return generateNumber(node);
            case AST_IDENTIFIER:
                return generateIdentifier(node);
            case AST_BINARY_OP:
                return generateBinaryOp(node);
            default:
                return "";
        }
    }
    
    string generateNumber(ASTNode* node) {
        string tempVar = "t" + to_string(tempVarCount++);
        string instruction = tempVar + " = " + node->value;
        instructions.push_back(instruction);
        
        cout << "生成数字代码: " << instruction << endl;
        return tempVar;
    }
    
    string generateIdentifier(ASTNode* node) {
        string tempVar = "t" + to_string(tempVarCount++);
        string instruction = tempVar + " = " + node->value;
        instructions.push_back(instruction);
        
        cout << "生成标识符代码: " << instruction << endl;
        return tempVar;
    }
    
    string generateBinaryOp(ASTNode* node) {
        string leftVar = generateNode(node->left);
        string rightVar = generateNode(node->right);
        
        string tempVar = "t" + to_string(tempVarCount++);
        string instruction = tempVar + " = " + leftVar + " " + node->value + " " + rightVar;
        instructions.push_back(instruction);
        
        cout << "生成二元操作代码: " << instruction << endl;
        return tempVar;
    }
    
    void printInstructions() {
        cout << "生成的指令" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (size_t i = 0; i < instructions.size(); i++) {
            cout << "[" << i << "] " << instructions[i] << endl;
        }
        
        cout << endl;
    }
    
    vector<string> getInstructions() {
        return instructions;
    }
};
```

### 错误处理器实现
```c
class ErrorHandler {
private:
    vector<string> errors;
    vector<string> warnings;
    
public:
    ErrorHandler() {}
    
    void initialize() {
        cout << "初始化错误处理器" << endl;
    }
    
    void reportError(const string& error) {
        errors.push_back(error);
        cout << "错误: " << error << endl;
    }
    
    void reportWarning(const string& warning) {
        warnings.push_back(warning);
        cout << "警告: " << warning << endl;
    }
    
    bool hasErrors() {
        return !errors.empty();
    }
    
    bool hasWarnings() {
        return !warnings.empty();
    }
    
    void printErrors() {
        cout << "错误列表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (size_t i = 0; i < errors.size(); i++) {
            cout << "[" << i << "] " << errors[i] << endl;
        }
        
        cout << endl;
    }
    
    void printWarnings() {
        cout << "警告列表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (size_t i = 0; i < warnings.size(); i++) {
            cout << "[" << i << "] " << warnings[i] << endl;
        }
        
        cout << endl;
    }
    
    void clear() {
        errors.clear();
        warnings.clear();
    }
};
```
