# 编译器工具链

## 🎯 项目概述

**编译器工具链**是一个综合性的编译器实践项目，实现了一个完整的编译器工具链，包括预处理器、编译器、汇编器、链接器等工具。该项目涵盖了编译器工具链的各个组件，是理解现代编译器生态系统的重要实践。

## 🔍 项目目标

### 学习目标
- **理解工具链架构**：掌握编译器工具链的整体设计
- **实现各个工具**：实现预处理器、编译器、汇编器、链接器
- **工具集成**：实现工具之间的协作和集成
- **构建系统**：实现自动化构建系统

### 功能目标
- 支持源代码预处理
- 支持多语言编译
- 支持目标代码生成
- 支持库文件链接
- 支持调试信息生成

class CompilerToolchain {
private:
    Preprocessor* preprocessor;
    Compiler* compiler;
    Assembler* assembler;
    Linker* linker;
    BuildSystem* buildSystem;
    
public:
    CompilerToolchain() {
        initializeToolchain();
    }
    
    void initializeToolchain() {
        cout << "初始化编译器工具链" << endl;
        
        preprocessor = new Preprocessor();
        compiler = new Compiler();
        assembler = new Assembler();
        linker = new Linker();
        buildSystem = new BuildSystem();
        
        // 初始化各个工具
        preprocessor->initialize();
        compiler->initialize();
        assembler->initialize();
        linker->initialize();
        buildSystem->initialize();
    }
    
    bool buildProject(const string& projectPath) {
        cout << "构建项目: " << projectPath << endl;
        cout << "===========================================" << endl;
        
        try {
            // 1. 预处理
            vector<string> preprocessedFiles = preprocessor->preprocessProject(projectPath);
            
            // 2. 编译
            vector<string> objectFiles = compiler->compileFiles(preprocessedFiles);
            
            // 3. 汇编
            vector<string> assemblyFiles = assembler->assembleFiles(objectFiles);
            
            // 4. 链接
            string executable = linker->linkFiles(assemblyFiles);
            
            cout << "项目构建成功: " << executable << endl;
            return true;
            
        } catch (const exception& e) {
            cout << "项目构建失败: " << e.what() << endl;
            return false;
        }
    }
    
    void cleanup() {
        cout << "清理编译器工具链" << endl;
        
        if (preprocessor) {
            delete preprocessor;
            preprocessor = nullptr;
        }
        
        if (compiler) {
            delete compiler;
            compiler = nullptr;
        }
        
        if (assembler) {
            delete assembler;
            assembler = nullptr;
        }
        
        if (linker) {
            delete linker;
            linker = nullptr;
        }
        
        if (buildSystem) {
            delete buildSystem;
            buildSystem = nullptr;
        }
        
        cout << "编译器工具链清理完成" << endl;
    }
    
    ~CompilerToolchain() {
        cleanup();
    }
};

class Preprocessor {
private:
    map<string, string> macros;
    vector<string> includePaths;
    bool isRunning;
    
public:
    Preprocessor() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化预处理器" << endl;
        isRunning = true;
        
        // 添加默认包含路径
        includePaths.push_back("./");
        includePaths.push_back("/usr/include");
        includePaths.push_back("/usr/local/include");
    }
    
    void addIncludePath(const string& path) {
        includePaths.push_back(path);
        cout << "添加包含路径: " << path << endl;
    }
    
    void defineMacro(const string& name, const string& value) {
        macros[name] = value;
        cout << "定义宏: " << name << " = " << value << endl;
    }
    
    void undefineMacro(const string& name) {
        macros.erase(name);
        cout << "取消定义宏: " << name << endl;
    }
    
    string preprocessFile(const string& filePath) {
        cout << "预处理文件: " << filePath << endl;
        
        string content = readFile(filePath);
        string preprocessed = preprocessContent(content);
        
        string outputPath = filePath + ".i";
        writeFile(outputPath, preprocessed);
        
        cout << "预处理完成: " << outputPath << endl;
        return outputPath;
    }
    
    vector<string> preprocessProject(const string& projectPath) {
        cout << "预处理项目: " << projectPath << endl;
        cout << "-------------------------------------------" << endl;
        
        vector<string> sourceFiles = findSourceFiles(projectPath);
        vector<string> preprocessedFiles;
        
        for (const string& file : sourceFiles) {
            string preprocessedFile = preprocessFile(file);
            preprocessedFiles.push_back(preprocessedFile);
        }
        
        cout << "项目预处理完成，共处理 " << preprocessedFiles.size() << " 个文件" << endl;
        return preprocessedFiles;
    }
    
    string preprocessContent(const string& content) {
        string result = content;
        
        // 处理宏定义
        result = processMacros(result);
        
        // 处理包含文件
        result = processIncludes(result);
        
        // 处理条件编译
        result = processConditionals(result);
        
        return result;
    }
    
    string processMacros(const string& content) {
        string result = content;
        
        for (const auto& pair : macros) {
            string macroName = pair.first;
            string macroValue = pair.second;
            
            // 简单的宏替换
            size_t pos = 0;
            while ((pos = result.find(macroName, pos)) != string::npos) {
                result.replace(pos, macroName.length(), macroValue);
                pos += macroValue.length();
            }
        }
        
        return result;
    }
    
    string processIncludes(const string& content) {
        string result = content;
        
        size_t pos = 0;
        while ((pos = result.find("#include", pos)) != string::npos) {
            size_t start = result.find("\"", pos);
            size_t end = result.find("\"", start + 1);
            
            if (start != string::npos && end != string::npos) {
                string includeFile = result.substr(start + 1, end - start - 1);
                string includeContent = readIncludeFile(includeFile);
                
                result.replace(pos, end - pos + 1, includeContent);
                pos += includeContent.length();
            } else {
                pos += 8; // 跳过 "#include"
            }
        }
        
        return result;
    }
    
    string processConditionals(const string& content) {
        string result = content;
        
        // 处理 #ifdef
        size_t pos = 0;
        while ((pos = result.find("#ifdef", pos)) != string::npos) {
            size_t start = pos;
            size_t end = result.find("\n", pos);
            
            if (end != string::npos) {
                string line = result.substr(start, end - start);
                string macroName = extractMacroName(line);
                
                if (macros.find(macroName) != macros.end()) {
                    // 宏已定义，保留内容
                    result.erase(start, end - start + 1);
                } else {
                    // 宏未定义，删除内容
                    result.erase(start, end - start + 1);
                }
            } else {
                pos += 6; // 跳过 "#ifdef"
            }
        }
        
        return result;
    }
    
    string extractMacroName(const string& line) {
        size_t start = line.find(" ");
        if (start != string::npos) {
            return line.substr(start + 1);
        }
        return "";
    }
    
    string readFile(const string& filePath) {
        // 简化实现：返回模拟的文件内容
        return "// 模拟文件内容\nint main() { return 0; }";
    }
    
    void writeFile(const string& filePath, const string& content) {
        cout << "写入文件: " << filePath << endl;
    }
    
    string readIncludeFile(const string& fileName) {
        // 在包含路径中查找文件
        for (const string& path : includePaths) {
            string fullPath = path + "/" + fileName;
            if (fileExists(fullPath)) {
                return readFile(fullPath);
            }
        }
        
        cout << "未找到包含文件: " << fileName << endl;
        return "";
    }
    
    bool fileExists(const string& filePath) {
        // 简化实现：模拟文件存在检查
        return filePath.find(".h") != string::npos;
    }
    
    vector<string> findSourceFiles(const string& projectPath) {
        vector<string> files;
        
        // 简化实现：返回模拟的源文件列表
        files.push_back(projectPath + "/main.c");
        files.push_back(projectPath + "/utils.c");
        
        return files;
    }
    
    void printMacros() {
        cout << "定义的宏:" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (const auto& pair : macros) {
            cout << pair.first << " = " << pair.second << endl;
        }
        
        cout << endl;
    }
    
    void printIncludePaths() {
        cout << "包含路径:" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (const string& path : includePaths) {
            cout << path << endl;
        }
        
        cout << endl;
    }
};

class Compiler {
private:
    LexicalAnalyzer* lexer;
    SyntaxAnalyzer* parser;
    SemanticAnalyzer* semanticAnalyzer;
    CodeGenerator* codeGenerator;
    Optimizer* optimizer;
    bool isRunning;
    
public:
    Compiler() : isRunning(false) {
        initializeCompiler();
    }
    
    void initializeCompiler() {
        cout << "初始化编译器" << endl;
        
        lexer = new LexicalAnalyzer();
        parser = new SyntaxAnalyzer();
        semanticAnalyzer = new SemanticAnalyzer();
        codeGenerator = new CodeGenerator();
        optimizer = new Optimizer();
        
        lexer->initialize();
        parser->initialize();
        semanticAnalyzer->initialize();
        codeGenerator->initialize();
        optimizer->initialize();
        
        isRunning = true;
    }
    
    vector<string> compileFiles(const vector<string>& sourceFiles) {
        cout << "编译源文件" << endl;
        cout << "-------------------------------------------" << endl;
        
        vector<string> objectFiles;
        
        for (const string& sourceFile : sourceFiles) {
            string objectFile = compileFile(sourceFile);
            if (!objectFile.empty()) {
                objectFiles.push_back(objectFile);
            }
        }
        
        cout << "编译完成，共生成 " << objectFiles.size() << " 个目标文件" << endl;
        return objectFiles;
    }
    
    string compileFile(const string& sourceFile) {
        cout << "编译文件: " << sourceFile << endl;
        
        try {
            // 读取源文件
            string sourceCode = readFile(sourceFile);
            
            // 词法分析
            vector<Token> tokens = lexer->analyze(sourceCode);
            if (tokens.empty()) {
                cout << "词法分析失败: " << sourceFile << endl;
                return "";
            }
            
            // 语法分析
            ASTNode* ast = parser->parse(tokens);
            if (!ast) {
                cout << "语法分析失败: " << sourceFile << endl;
                return "";
            }
            
            // 语义分析
            if (!semanticAnalyzer->analyze(ast)) {
                cout << "语义分析失败: " << sourceFile << endl;
                return "";
            }
            
            // 代码优化
            ASTNode* optimizedAST = optimizer->optimize(ast);
            
            // 代码生成
            string assemblyCode = codeGenerator->generate(optimizedAST);
            if (assemblyCode.empty()) {
                cout << "代码生成失败: " << sourceFile << endl;
                return "";
            }
            
            // 生成目标文件
            string objectFile = generateObjectFile(sourceFile, assemblyCode);
            
            cout << "编译成功: " << objectFile << endl;
            return objectFile;
            
        } catch (const exception& e) {
            cout << "编译错误: " << e.what() << endl;
            return "";
        }
    }
    
    string generateObjectFile(const string& sourceFile, const string& assemblyCode) {
        string objectFile = sourceFile + ".o";
        
        // 生成目标文件内容
        string objectContent = generateObjectContent(assemblyCode);
        
        // 写入目标文件
        writeFile(objectFile, objectContent);
        
        return objectFile;
    }
    
    string generateObjectContent(const string& assemblyCode) {
        // 简化实现：生成模拟的目标文件内容
        string content = "ELF Header\n";
        content += "Section Headers\n";
        content += "Program Headers\n";
        content += "Assembly Code:\n";
        content += assemblyCode;
        content += "\nSymbol Table\n";
        content += "Relocation Table\n";
        
        return content;
    }
    
    string readFile(const string& filePath) {
        // 简化实现：返回模拟的文件内容
        return "int main() { return 0; }";
    }
    
    void writeFile(const string& filePath, const string& content) {
        cout << "写入目标文件: " << filePath << endl;
    }
    
    void setOptimizationLevel(int level) {
        optimizer->setOptimizationLevel(level);
        cout << "设置优化级别: " << level << endl;
    }
    
    void printCompilerInfo() {
        cout << "编译器信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "版本: 1.0.0" << endl;
        cout << "目标架构: x86_64" << endl;
        cout << "优化级别: " << optimizer->getOptimizationLevel() << endl;
        cout << endl;
    }
};

class Assembler {
private:
    map<string, int> opcodeMap;
    map<string, int> registerMap;
    bool isRunning;
    
public:
    Assembler() : isRunning(false) {
        initializeAssembler();
    }
    
    void initializeAssembler() {
        cout << "初始化汇编器" << endl;
        
        // 初始化操作码映射
        opcodeMap["mov"] = 0x01;
        opcodeMap["add"] = 0x02;
        opcodeMap["sub"] = 0x03;
        opcodeMap["mul"] = 0x04;
        opcodeMap["div"] = 0x05;
        opcodeMap["jmp"] = 0x06;
        opcodeMap["jz"] = 0x07;
        opcodeMap["call"] = 0x08;
        opcodeMap["ret"] = 0x09;
        opcodeMap["push"] = 0x0A;
        opcodeMap["pop"] = 0x0B;
        
        // 初始化寄存器映射
        registerMap["eax"] = 0x00;
        registerMap["ebx"] = 0x01;
        registerMap["ecx"] = 0x02;
        registerMap["edx"] = 0x03;
        registerMap["esp"] = 0x04;
        registerMap["ebp"] = 0x05;
        
        isRunning = true;
    }
    
    vector<string> assembleFiles(const vector<string>& objectFiles) {
        cout << "汇编目标文件" << endl;
        cout << "-------------------------------------------" << endl;
        
        vector<string> assemblyFiles;
        
        for (const string& objectFile : objectFiles) {
            string assemblyFile = assembleFile(objectFile);
            if (!assemblyFile.empty()) {
                assemblyFiles.push_back(assemblyFile);
            }
        }
        
        cout << "汇编完成，共生成 " << assemblyFiles.size() << " 个汇编文件" << endl;
        return assemblyFiles;
    }
    
    string assembleFile(const string& objectFile) {
        cout << "汇编文件: " << objectFile << endl;
        
        try {
            // 读取目标文件
            string objectContent = readFile(objectFile);
            
            // 提取汇编代码
            string assemblyCode = extractAssemblyCode(objectContent);
            
            // 汇编代码
            string machineCode = assembleCode(assemblyCode);
            
            // 生成汇编文件
            string assemblyFile = generateAssemblyFile(objectFile, machineCode);
            
            cout << "汇编成功: " << assemblyFile << endl;
            return assemblyFile;
            
        } catch (const exception& e) {
            cout << "汇编错误: " << e.what() << endl;
            return "";
        }
    }
    
    string extractAssemblyCode(const string& objectContent) {
        // 简化实现：从目标文件中提取汇编代码
        size_t start = objectContent.find("Assembly Code:");
        if (start != string::npos) {
            size_t end = objectContent.find("\nSymbol Table", start);
            if (end != string::npos) {
                return objectContent.substr(start + 15, end - start - 15);
            }
        }
        
        return "mov eax, 0\nret";
    }
    
    string assembleCode(const string& assemblyCode) {
        cout << "汇编代码" << endl;
        
        string machineCode;
        vector<string> lines = splitLines(assemblyCode);
        
        for (const string& line : lines) {
            string trimmedLine = trim(line);
            if (!trimmedLine.empty()) {
                string instruction = assembleInstruction(trimmedLine);
                machineCode += instruction + "\n";
            }
        }
        
        return machineCode;
    }
    
    string assembleInstruction(const string& instruction) {
        vector<string> tokens = splitTokens(instruction);
        
        if (tokens.empty()) {
            return "";
        }
        
        string opcode = tokens[0];
        
        if (opcodeMap.find(opcode) != opcodeMap.end()) {
            int opcodeValue = opcodeMap[opcode];
            
            switch (opcodeValue) {
                case 0x01: // mov
                    return assembleMov(tokens);
                case 0x02: // add
                    return assembleAdd(tokens);
                case 0x03: // sub
                    return assembleSub(tokens);
                case 0x06: // jmp
                    return assembleJmp(tokens);
                case 0x09: // ret
                    return "C3"; // ret 指令的机器码
                default:
                    return "00"; // 默认机器码
            }
        }
        
        return "00";
    }
    
    string assembleMov(const vector<string>& tokens) {
        if (tokens.size() >= 3) {
            string dest = tokens[1];
            string src = tokens[2];
            
            if (registerMap.find(dest) != registerMap.end()) {
                int destReg = registerMap[dest];
                
                if (registerMap.find(src) != registerMap.end()) {
                    int srcReg = registerMap[src];
                    return "8B" + toHex(destReg) + toHex(srcReg);
                } else if (isNumber(src)) {
                    int value = stoi(src);
                    return "B8" + toHex(destReg) + toHex(value);
                }
            }
        }
        
        return "00";
    }
    
    string assembleAdd(const vector<string>& tokens) {
        if (tokens.size() >= 3) {
            string dest = tokens[1];
            string src = tokens[2];
            
            if (registerMap.find(dest) != registerMap.end()) {
                int destReg = registerMap[dest];
                
                if (registerMap.find(src) != registerMap.end()) {
                    int srcReg = registerMap[src];
                    return "01" + toHex(destReg) + toHex(srcReg);
                } else if (isNumber(src)) {
                    int value = stoi(src);
                    return "83" + toHex(destReg) + toHex(value);
                }
            }
        }
        
        return "00";
    }
    
    string assembleSub(const vector<string>& tokens) {
        if (tokens.size() >= 3) {
            string dest = tokens[1];
            string src = tokens[2];
            
            if (registerMap.find(dest) != registerMap.end()) {
                int destReg = registerMap[dest];
                
                if (registerMap.find(src) != registerMap.end()) {
                    int srcReg = registerMap[src];
                    return "29" + toHex(destReg) + toHex(srcReg);
                } else if (isNumber(src)) {
                    int value = stoi(src);
                    return "83" + toHex(destReg) + toHex(value);
                }
            }
        }
        
        return "00";
    }
    
    string assembleJmp(const vector<string>& tokens) {
        if (tokens.size() >= 2) {
            string target = tokens[1];
            
            if (isNumber(target)) {
                int address = stoi(target);
                return "E9" + toHex(address);
            }
        }
        
        return "00";
    }
    
    string generateAssemblyFile(const string& objectFile, const string& machineCode) {
        string assemblyFile = objectFile + ".asm";
        
        // 生成汇编文件内容
        string assemblyContent = generateAssemblyContent(machineCode);
        
        // 写入汇编文件
        writeFile(assemblyFile, assemblyContent);
        
        return assemblyFile;
    }
    
    string generateAssemblyContent(const string& machineCode) {
        string content = "; 汇编文件\n";
        content += "; 机器码: " + machineCode + "\n";
        content += "; 生成时间: " + getCurrentTime() + "\n";
        content += "\n";
        content += "section .text\n";
        content += "global _start\n";
        content += "_start:\n";
        content += machineCode;
        content += "\n";
        
        return content;
    }
    
    string readFile(const string& filePath) {
        // 简化实现：返回模拟的文件内容
        return "Assembly Code:\nmov eax, 0\nret\nSymbol Table\n";
    }
    
    void writeFile(const string& filePath, const string& content) {
        cout << "写入汇编文件: " << filePath << endl;
    }
    
    vector<string> splitLines(const string& text) {
        vector<string> lines;
        stringstream ss(text);
        string line;
        
        while (getline(ss, line)) {
            lines.push_back(line);
        }
        
        return lines;
    }
    
    vector<string> splitTokens(const string& text) {
        vector<string> tokens;
        stringstream ss(text);
        string token;
        
        while (ss >> token) {
            tokens.push_back(token);
        }
        
        return tokens;
    }
    
    string trim(const string& text) {
        size_t start = text.find_first_not_of(" \t");
        size_t end = text.find_last_not_of(" \t");
        
        if (start == string::npos) {
            return "";
        }
        
        return text.substr(start, end - start + 1);
    }
    
    bool isNumber(const string& text) {
        return !text.empty() && all_of(text.begin(), text.end(), ::isdigit);
    }
    
    string toHex(int value) {
        stringstream ss;
        ss << hex << value;
        return ss.str();
    }
    
    string getCurrentTime() {
        time_t now = time(0);
        char* timeStr = ctime(&now);
        return string(timeStr);
    }
    
    void printAssemblerInfo() {
        cout << "汇编器信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "版本: 1.0.0" << endl;
        cout << "目标架构: x86_64" << endl;
        cout << "支持指令数: " << opcodeMap.size() << endl;
        cout << "支持寄存器数: " << registerMap.size() << endl;
        cout << endl;
    }
};

    void collectSymbols(const vector<string>& assemblyFiles) {
        cout << "收集符号" << endl;
        
        for (const string& file : assemblyFiles) {
            string content = readFile(file);
            vector<Symbol> symbols = extractSymbols(content);
            
            for (const Symbol& symbol : symbols) {
                symbolTable[symbol.name] = symbol;
                cout << "收集符号: " << symbol.name << " -> " << symbol.address << endl;
            }
        }
    }
    
    void resolveSymbols(const vector<string>& assemblyFiles) {
        cout << "解析符号引用" << endl;
        
        for (const string& file : assemblyFiles) {
            string content = readFile(file);
            vector<Relocation> fileRelocations = extractRelocations(content);
            
            for (Relocation& relocation : fileRelocations) {
                if (symbolTable.find(relocation.symbolName) != symbolTable.end()) {
                    relocation.resolvedAddress = symbolTable[relocation.symbolName].address;
                    relocations.push_back(relocation);
                    cout << "解析符号: " << relocation.symbolName << " -> " << relocation.resolvedAddress << endl;
                } else {
                    cout << "未找到符号: " << relocation.symbolName << endl;
                }
            }
        }
    }
    
    string generateExecutable(const vector<string>& assemblyFiles) {
        cout << "生成可执行文件" << endl;
        
        string executable = "a.out";
        
        // 生成可执行文件内容
        string executableContent = generateExecutableContent(assemblyFiles);
        
        // 写入可执行文件
        writeFile(executable, executableContent);
        
        return executable;
    }
    
    string generateExecutableContent(const vector<string>& assemblyFiles) {
        string content = "ELF Header\n";
        content += "Program Headers\n";
        content += "Section Headers\n";
        
        // 添加代码段
        content += "Section .text\n";
        for (const string& file : assemblyFiles) {
            string fileContent = readFile(file);
            content += extractCodeSection(fileContent);
        }
        
        // 添加数据段
        content += "Section .data\n";
        content += "Section .bss\n";
        
        // 添加符号表
        content += "Symbol Table\n";
        for (const auto& pair : symbolTable) {
            const Symbol& symbol = pair.second;
            content += symbol.name + " " + to_string(symbol.address) + "\n";
        }
        
        // 添加重定位表
        content += "Relocation Table\n";
        for (const Relocation& relocation : relocations) {
            content += to_string(relocation.offset) + " " + relocation.symbolName + " " + to_string(relocation.resolvedAddress) + "\n";
        }
        
        return content;
    }
    
    vector<Symbol> extractSymbols(const string& content) {
        vector<Symbol> symbols;
        
        // 简化实现：提取模拟的符号
        Symbol mainSymbol;
        mainSymbol.name = "main";
        mainSymbol.address = 0x400000;
        mainSymbol.type = SYMBOL_FUNCTION;
        symbols.push_back(mainSymbol);
        
        return symbols;
    }
    
    vector<Relocation> extractRelocations(const string& content) {
        vector<Relocation> relocations;
        
        // 简化实现：提取模拟的重定位
        Relocation relocation;
        relocation.offset = 0x400010;
        relocation.symbolName = "main";
        relocation.resolvedAddress = 0;
        relocations.push_back(relocation);
        
        return relocations;
    }
    
    string extractCodeSection(const string& content) {
        // 简化实现：提取代码段
        return "mov eax, 0\nret\n";
    }
    
    string readFile(const string& filePath) {
        // 简化实现：返回模拟的文件内容
        return "section .text\nmain:\nmov eax, 0\nret\n";
    }
    
    void writeFile(const string& filePath, const string& content) {
        cout << "写入可执行文件: " << filePath << endl;
    }
    
    void printSymbolTable() {
        cout << "符号表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (const auto& pair : symbolTable) {
            const Symbol& symbol = pair.second;
            cout << symbol.name << " -> " << hex << symbol.address << " (" << getSymbolTypeName(symbol.type) << ")" << endl;
        }
        
        cout << endl;
    }
    
    void printRelocations() {
        cout << "重定位表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (const Relocation& relocation : relocations) {
            cout << hex << relocation.offset << " -> " << relocation.symbolName << " (" << hex << relocation.resolvedAddress << ")" << endl;
        }
        
        cout << endl;
    }
    
    string getSymbolTypeName(SymbolType type) {
        switch (type) {
            case SYMBOL_FUNCTION: return "函数";
            case SYMBOL_VARIABLE: return "变量";
            case SYMBOL_OBJECT: return "对象";
            default: return "未知";
        }
    }
};

enum SymbolType {
    SYMBOL_FUNCTION,
    SYMBOL_VARIABLE,
    SYMBOL_OBJECT
};

struct Symbol {

    int address;
    SymbolType type;
};

    int resolvedAddress;
