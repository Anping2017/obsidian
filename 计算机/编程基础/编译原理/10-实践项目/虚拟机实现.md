# 虚拟机实现

## 🎯 项目概述

**虚拟机实现**是一个重要的编译器实践项目，实现了一个能够执行中间代码的虚拟机。该项目涵盖了指令集设计、虚拟机架构、内存管理、执行引擎等虚拟机核心技术，是理解编译器后端和运行时系统的重要实践。

## 🔍 项目目标

### 学习目标
- **理解虚拟机架构**：掌握虚拟机的整体设计原理
- **实现指令集**：设计并实现虚拟机的指令集
- **内存管理**：实现虚拟机的内存管理系统
- **执行引擎**：实现虚拟机的指令执行引擎

### 功能目标
- 支持基本数据类型（整数、浮点数、字符）
- 支持变量存储和访问
- 支持控制流指令（跳转、条件分支）
- 支持函数调用和返回
- 支持堆栈管理

## 📊 项目架构

### 虚拟机架构
```c
class VirtualMachine {
private:
    vector<int> stack;
    map<string, int> variables;
    vector<Instruction> instructions;
    size_t programCounter;
    bool isRunning;
    
public:
    VirtualMachine() : programCounter(0), isRunning(false) {}
    
    void initialize() {
        cout << "初始化虚拟机" << endl;
        stack.clear();
        variables.clear();
        instructions.clear();
        programCounter = 0;
        isRunning = false;
    }
    
    void loadProgram(const vector<Instruction>& program) {
        cout << "加载程序到虚拟机" << endl;
        instructions = program;
        programCounter = 0;
    }
    
    void execute() {
        cout << "开始执行虚拟机程序" << endl;
        cout << "===========================================" << endl;
        
        isRunning = true;
        
        while (isRunning && programCounter < instructions.size()) {
            Instruction instruction = instructions[programCounter];
            executeInstruction(instruction);
            programCounter++;
        }
        
        cout << "虚拟机程序执行完成" << endl;
    }
    
    void executeInstruction(const Instruction& instruction) {
        cout << "执行指令: " << getInstructionName(instruction.opcode) << endl;
        
        switch (instruction.opcode) {
            case OP_PUSH:
                executePush(instruction);
                break;
            case OP_POP:
                executePop(instruction);
                break;
            case OP_LOAD:
                executeLoad(instruction);
                break;
            case OP_STORE:
                executeStore(instruction);
                break;
            case OP_ADD:
                executeAdd(instruction);
                break;
            case OP_SUB:
                executeSub(instruction);
                break;
            case OP_MUL:
                executeMul(instruction);
                break;
            case OP_DIV:
                executeDiv(instruction);
                break;
            case OP_JMP:
                executeJump(instruction);
                break;
            case OP_JZ:
                executeJumpIfZero(instruction);
                break;
            case OP_CALL:
                executeCall(instruction);
                break;
            case OP_RET:
                executeReturn(instruction);
                break;
            case OP_HALT:
                executeHalt(instruction);
                break;
            default:
                cout << "未知指令: " << instruction.opcode << endl;
                break;
        }
    }
    
    void executePush(const Instruction& instruction) {
        int value = instruction.operand;
        stack.push_back(value);
        cout << "压入栈: " << value << endl;
    }
    
    void executePop(const Instruction& instruction) {
        if (!stack.empty()) {
            int value = stack.back();
            stack.pop_back();
            cout << "弹出栈: " << value << endl;
        } else {
            cout << "栈为空，无法弹出" << endl;
        }
    }
    
    void executeLoad(const Instruction& instruction) {
        string varName = instruction.operandStr;
        if (variables.find(varName) != variables.end()) {
            int value = variables[varName];
            stack.push_back(value);
            cout << "加载变量: " << varName << " = " << value << endl;
        } else {
            cout << "变量未定义: " << varName << endl;
        }
    }
    
    void executeStore(const Instruction& instruction) {
        string varName = instruction.operandStr;
        if (!stack.empty()) {
            int value = stack.back();
            stack.pop_back();
            variables[varName] = value;
            cout << "存储变量: " << varName << " = " << value << endl;
        } else {
            cout << "栈为空，无法存储" << endl;
        }
    }
    
    void executeAdd(const Instruction& instruction) {
        if (stack.size() >= 2) {
            int b = stack.back();
            stack.pop_back();
            int a = stack.back();
            stack.pop_back();
            int result = a + b;
            stack.push_back(result);
            cout << "加法: " << a << " + " << b << " = " << result << endl;
        } else {
            cout << "栈中操作数不足" << endl;
        }
    }
    
    void executeSub(const Instruction& instruction) {
        if (stack.size() >= 2) {
            int b = stack.back();
            stack.pop_back();
            int a = stack.back();
            stack.pop_back();
            int result = a - b;
            stack.push_back(result);
            cout << "减法: " << a << " - " << b << " = " << result << endl;
        } else {
            cout << "栈中操作数不足" << endl;
        }
    }
    
    void executeMul(const Instruction& instruction) {
        if (stack.size() >= 2) {
            int b = stack.back();
            stack.pop_back();
            int a = stack.back();
            stack.pop_back();
            int result = a * b;
            stack.push_back(result);
            cout << "乘法: " << a << " * " << b << " = " << result << endl;
        } else {
            cout << "栈中操作数不足" << endl;
        }
    }
    
    void executeDiv(const Instruction& instruction) {
        if (stack.size() >= 2) {
            int b = stack.back();
            stack.pop_back();
            int a = stack.back();
            stack.pop_back();
            
            if (b != 0) {
                int result = a / b;
                stack.push_back(result);
                cout << "除法: " << a << " / " << b << " = " << result << endl;
            } else {
                cout << "除零错误" << endl;
            }
        } else {
            cout << "栈中操作数不足" << endl;
        }
    }
    
    void executeJump(const Instruction& instruction) {
        int target = instruction.operand;
        programCounter = target - 1; // -1 因为循环会自增
        cout << "跳转到: " << target << endl;
    }
    
    void executeJumpIfZero(const Instruction& instruction) {
        if (!stack.empty()) {
            int value = stack.back();
            stack.pop_back();
            
            if (value == 0) {
                int target = instruction.operand;
                programCounter = target - 1; // -1 因为循环会自增
                cout << "条件跳转: " << value << " == 0, 跳转到 " << target << endl;
            } else {
                cout << "条件跳转: " << value << " != 0, 不跳转" << endl;
            }
        } else {
            cout << "栈为空，无法条件跳转" << endl;
        }
    }
    
    void executeCall(const Instruction& instruction) {
        int target = instruction.operand;
        stack.push_back(programCounter + 1); // 保存返回地址
        programCounter = target - 1; // -1 因为循环会自增
        cout << "调用函数: " << target << endl;
    }
    
    void executeReturn(const Instruction& instruction) {
        if (!stack.empty()) {
            int returnAddress = stack.back();
            stack.pop_back();
            programCounter = returnAddress - 1; // -1 因为循环会自增
            cout << "返回: " << returnAddress << endl;
        } else {
            cout << "栈为空，无法返回" << endl;
        }
    }
    
    void executeHalt(const Instruction& instruction) {
        isRunning = false;
        cout << "程序停止" << endl;
    }
    
    string getInstructionName(OpCode opcode) {
        switch (opcode) {
            case OP_PUSH: return "PUSH";
            case OP_POP: return "POP";
            case OP_LOAD: return "LOAD";
            case OP_STORE: return "STORE";
            case OP_ADD: return "ADD";
            case OP_SUB: return "SUB";
            case OP_MUL: return "MUL";
            case OP_DIV: return "DIV";
            case OP_JMP: return "JMP";
            case OP_JZ: return "JZ";
            case OP_CALL: return "CALL";
            case OP_RET: return "RET";
            case OP_HALT: return "HALT";
            default: return "UNKNOWN";
        }
    }
    
    void printStack() {
        cout << "栈内容: ";
        for (int value : stack) {
            cout << value << " ";
        }
        cout << endl;
    }
    
    void printVariables() {
        cout << "变量表:" << endl;
        for (const auto& pair : variables) {
            cout << "  " << pair.first << " = " << pair.second << endl;
        }
    }
    
    void printProgram() {
        cout << "程序指令:" << endl;
        for (size_t i = 0; i < instructions.size(); i++) {
            const Instruction& instruction = instructions[i];
            cout << "[" << i << "] " << getInstructionName(instruction.opcode);
            
            if (instruction.operand != 0) {
                cout << " " << instruction.operand;
            }
            
            if (!instruction.operandStr.empty()) {
                cout << " " << instruction.operandStr;
            }
            
            cout << endl;
        }
    }
};

enum OpCode {
    OP_PUSH,     // 压入栈
    OP_POP,      // 弹出栈
    OP_LOAD,     // 加载变量
    OP_STORE,    // 存储变量
    OP_ADD,      // 加法
    OP_SUB,      // 减法
    OP_MUL,      // 乘法
    OP_DIV,      // 除法
    OP_JMP,      // 跳转
    OP_JZ,       // 条件跳转
    OP_CALL,     // 函数调用
    OP_RET,      // 函数返回
    OP_HALT      // 停止
};

struct Instruction {
    OpCode opcode;
    int operand;
    string operandStr;
    
    Instruction(OpCode op, int op1 = 0, const string& op2 = "") 
        : opcode(op), operand(op1), operandStr(op2) {}
};
```

```c
class InstructionSet {
private:
    map<string, OpCode> opcodeMap;
    map<OpCode, string> opcodeNames;
    
public:
    InstructionSet() {
        initializeInstructionSet();
    }
    
    void initializeInstructionSet() {
        cout << "初始化指令集" << endl;
        
        // 初始化操作码映射
        opcodeMap["PUSH"] = OP_PUSH;
        opcodeMap["POP"] = OP_POP;
        opcodeMap["LOAD"] = OP_LOAD;
        opcodeMap["STORE"] = OP_STORE;
        opcodeMap["ADD"] = OP_ADD;
        opcodeMap["SUB"] = OP_SUB;
        opcodeMap["MUL"] = OP_MUL;
        opcodeMap["DIV"] = OP_DIV;
        opcodeMap["JMP"] = OP_JMP;
        opcodeMap["JZ"] = OP_JZ;
        opcodeMap["CALL"] = OP_CALL;
        opcodeMap["RET"] = OP_RET;
        opcodeMap["HALT"] = OP_HALT;
        
        // 初始化操作码名称映射
        opcodeNames[OP_PUSH] = "PUSH";
        opcodeNames[OP_POP] = "POP";
        opcodeNames[OP_LOAD] = "LOAD";
        opcodeNames[OP_STORE] = "STORE";
        opcodeNames[OP_ADD] = "ADD";
        opcodeNames[OP_SUB] = "SUB";
        opcodeNames[OP_MUL] = "MUL";
        opcodeNames[OP_DIV] = "DIV";
        opcodeNames[OP_JMP] = "JMP";
        opcodeNames[OP_JZ] = "JZ";
        opcodeNames[OP_CALL] = "CALL";
        opcodeNames[OP_RET] = "RET";
        opcodeNames[OP_HALT] = "HALT";
    }
    
    OpCode getOpCode(const string& name) {
        if (opcodeMap.find(name) != opcodeMap.end()) {
            return opcodeMap[name];
        }
        return OP_HALT; // 默认返回停止指令
    }
    
    string getOpCodeName(OpCode opcode) {
        if (opcodeNames.find(opcode) != opcodeNames.end()) {
            return opcodeNames[opcode];
        }
        return "UNKNOWN";
    }
    
    bool isValidOpCode(const string& name) {
        return opcodeMap.find(name) != opcodeMap.end();
    }
    
    void printInstructionSet() {
        cout << "指令集:" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (const auto& pair : opcodeMap) {
            cout << pair.first << " - " << getInstructionDescription(pair.second) << endl;
        }
        
        cout << endl;
    }
    
    string getInstructionDescription(OpCode opcode) {
        switch (opcode) {
            case OP_PUSH: return "压入栈";
            case OP_POP: return "弹出栈";
            case OP_LOAD: return "加载变量";
            case OP_STORE: return "存储变量";
            case OP_ADD: return "加法";
            case OP_SUB: return "减法";
            case OP_MUL: return "乘法";
            case OP_DIV: return "除法";
            case OP_JMP: return "跳转";
            case OP_JZ: return "条件跳转";
            case OP_CALL: return "函数调用";
            case OP_RET: return "函数返回";
            case OP_HALT: return "停止";
            default: return "未知";
        }
    }
};
```

## 🔧 内存管理器

### 内存管理器实现
```c
class MemoryManager {
private:
    map<string, int> globalVariables;
    vector<map<string, int>> callStack;
    vector<int> dataStack;
    int stackPointer;
    int heapPointer;
    
public:
    MemoryManager() : stackPointer(0), heapPointer(0) {}
    
    void initialize() {
        cout << "初始化内存管理器" << endl;
        globalVariables.clear();
        callStack.clear();
        dataStack.clear();
        stackPointer = 0;
        heapPointer = 0;
    }
    
    void pushFrame() {
        cout << "压入调用帧" << endl;
        callStack.push_back(map<string, int>());
    }
    
    void popFrame() {
        cout << "弹出调用帧" << endl;
        if (!callStack.empty()) {
            callStack.pop_back();
        }
    }
    
    void setGlobalVariable(const string& name, int value) {
        globalVariables[name] = value;
        cout << "设置全局变量: " << name << " = " << value << endl;
    }
    
    int getGlobalVariable(const string& name) {
        if (globalVariables.find(name) != globalVariables.end()) {
            return globalVariables[name];
        }
        return 0;
    }
    
    void setLocalVariable(const string& name, int value) {
        if (!callStack.empty()) {
            callStack.back()[name] = value;
            cout << "设置局部变量: " << name << " = " << value << endl;
        }
    }
    
    int getLocalVariable(const string& name) {
        if (!callStack.empty() && callStack.back().find(name) != callStack.back().end()) {
            return callStack.back()[name];
        }
        return 0;
    }
    
    void setVariable(const string& name, int value) {
        // 先检查局部变量
        if (!callStack.empty() && callStack.back().find(name) != callStack.back().end()) {
            setLocalVariable(name, value);
        } else {
            setGlobalVariable(name, value);
        }
    }
    
    int getVariable(const string& name) {
        // 先检查局部变量
        if (!callStack.empty() && callStack.back().find(name) != callStack.back().end()) {
            return getLocalVariable(name);
        } else {
            return getGlobalVariable(name);
        }
    }
    
    void pushStack(int value) {
        dataStack.push_back(value);
        stackPointer++;
        cout << "压入数据栈: " << value << endl;
    }
    
    int popStack() {
        if (!dataStack.empty()) {
            int value = dataStack.back();
            dataStack.pop_back();
            stackPointer--;
            cout << "弹出数据栈: " << value << endl;
            return value;
        } else {
            cout << "数据栈为空" << endl;
            return 0;
        }
    }
    
    int peekStack() {
        if (!dataStack.empty()) {
            return dataStack.back();
        }
        return 0;
    }
    
    void printMemoryState() {
        cout << "内存状态" << endl;
        cout << "-------------------------------------------" << endl;
        
        cout << "全局变量:" << endl;
        for (const auto& pair : globalVariables) {
            cout << "  " << pair.first << " = " << pair.second << endl;
        }
        
        cout << "调用栈:" << endl;
        for (size_t i = 0; i < callStack.size(); i++) {
            cout << "  帧[" << i << "]:" << endl;
            for (const auto& pair : callStack[i]) {
                cout << "    " << pair.first << " = " << pair.second << endl;
            }
        }
        
        cout << "数据栈: ";
        for (int value : dataStack) {
            cout << value << " ";
        }
        cout << endl;
        
        cout << "栈指针: " << stackPointer << endl;
        cout << "堆指针: " << heapPointer << endl;
    }
};
```

## 🔧 执行引擎

### 执行引擎实现
```c
class ExecutionEngine {
private:
    VirtualMachine* vm;
    MemoryManager* memoryManager;
    InstructionSet* instructionSet;
    bool isRunning;
    
public:
    ExecutionEngine() {
        initializeEngine();
    }
    
    void initializeEngine() {
        cout << "初始化执行引擎" << endl;
        
        vm = new VirtualMachine();
        memoryManager = new MemoryManager();
        instructionSet = new InstructionSet();
        
        vm->initialize();
        memoryManager->initialize();
        
        isRunning = false;
    }
    
    void loadProgram(const vector<Instruction>& program) {
        cout << "加载程序到执行引擎" << endl;
        vm->loadProgram(program);
    }
    
    void execute() {
        cout << "开始执行程序" << endl;
        cout << "===========================================" << endl;
        
        isRunning = true;
        
        while (isRunning) {
            // 获取当前指令
            vector<Instruction> instructions = vm->getInstructions();
            size_t pc = vm->getProgramCounter();
            
            if (pc >= instructions.size()) {
                break;
            }
            
            Instruction instruction = instructions[pc];
            
            // 执行指令
            executeInstruction(instruction);
            
            // 更新程序计数器
            vm->incrementProgramCounter();
        }
        
        cout << "程序执行完成" << endl;
    }
    
    void executeInstruction(const Instruction& instruction) {
        cout << "执行指令: " << instructionSet->getOpCodeName(instruction.opcode) << endl;
        
        switch (instruction.opcode) {
            case OP_PUSH:
                memoryManager->pushStack(instruction.operand);
                break;
            case OP_POP:
                memoryManager->popStack();
                break;
            case OP_LOAD:
                {
                    int value = memoryManager->getVariable(instruction.operandStr);
                    memoryManager->pushStack(value);
                }
                break;
            case OP_STORE:
                {
                    int value = memoryManager->popStack();
                    memoryManager->setVariable(instruction.operandStr, value);
                }
                break;
            case OP_ADD:
                {
                    int b = memoryManager->popStack();
                    int a = memoryManager->popStack();
                    int result = a + b;
                    memoryManager->pushStack(result);
                }
                break;
            case OP_SUB:
                {
                    int b = memoryManager->popStack();
                    int a = memoryManager->popStack();
                    int result = a - b;
                    memoryManager->pushStack(result);
                }
                break;
            case OP_MUL:
                {
                    int b = memoryManager->popStack();
                    int a = memoryManager->popStack();
                    int result = a * b;
                    memoryManager->pushStack(result);
                }
                break;
            case OP_DIV:
                {
                    int b = memoryManager->popStack();
                    int a = memoryManager->popStack();
                    
                    if (b != 0) {
                        int result = a / b;
                        memoryManager->pushStack(result);
                    } else {
                        cout << "除零错误" << endl;
                    }
                }
                break;
            case OP_JMP:
                vm->setProgramCounter(instruction.operand);
                break;
            case OP_JZ:
                {
                    int value = memoryManager->popStack();
                    if (value == 0) {
                        vm->setProgramCounter(instruction.operand);
                    }
                }
                break;
            case OP_CALL:
                {
                    memoryManager->pushFrame();
                    memoryManager->pushStack(vm->getProgramCounter() + 1);
                    vm->setProgramCounter(instruction.operand);
                }
                break;
            case OP_RET:
                {
                    int returnAddress = memoryManager->popStack();
                    memoryManager->popFrame();
                    vm->setProgramCounter(returnAddress);
                }
                break;
            case OP_HALT:
                isRunning = false;
                break;
        }
    }
    
    void printExecutionState() {
        cout << "执行状态" << endl;
        cout << "===========================================" << endl;
        
        vm->printProgram();
        memoryManager->printMemoryState();
    }
    
    void cleanup() {
        cout << "清理执行引擎" << endl;
        
        if (vm) {
            delete vm;
            vm = nullptr;
        }
        
        if (memoryManager) {
            delete memoryManager;
            memoryManager = nullptr;
        }
        
        if (instructionSet) {
            delete instructionSet;
            instructionSet = nullptr;
        }
        
        cout << "执行引擎清理完成" << endl;
    }
    
    ~ExecutionEngine() {
        cleanup();
    }
};
```
