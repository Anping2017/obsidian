# 四元式表示

## 🎯 四元式表示概述

**四元式表示**是中间代码的一种重要表示形式，将每条指令表示为四个字段：操作符、第一个操作数、第二个操作数、结果。四元式表示具有结构清晰、易于处理和优化的特点，是编译器中最常用的中间表示形式之一。

## 🔍 基本概念

### 四元式定义
四元式是一种中间表示形式，每条指令包含四个字段：
- **操作符 (op)**：表示要执行的操作
- **第一个操作数 (arg1)**：第一个操作数
- **第二个操作数 (arg2)**：第二个操作数
- **结果 (result)**：操作结果

### 四元式特点
- **结构清晰**：四个字段结构明确
- **易于处理**：便于进行各种操作
- **便于优化**：容易进行代码优化
- **便于生成**：容易转换为目标代码

## 📊 四元式类型

### 1. 算术运算
```c
struct Quadruple {
    string op;      // 操作符
    string arg1;    // 第一个操作数
    string arg2;    // 第二个操作数
    string result;  // 结果
};

// 基本算术运算
Quadruple q1 = {"+", "a", "b", "t1"};
Quadruple q2 = {"-", "t1", "c", "t2"};
Quadruple q3 = {"*", "t2", "d", "result"};

// 一元运算
Quadruple q4 = {"-", "x", "", "t3"};
Quadruple q5 = {"!", "y", "", "t4"};
```

### 2. 赋值运算
```c
// 直接赋值
Quadruple q1 = {"=", "y", "", "x"};
Quadruple q2 = {"=", "5", "", "z"};

// 复合赋值
Quadruple q3 = {"+=", "x", "1", "x"};
Quadruple q4 = {"*=", "y", "2", "y"};
```

### 3. 比较运算
```c
// 比较运算
Quadruple q1 = {">", "a", "b", "t1"};
Quadruple q2 = {"==", "x", "y", "t2"};
Quadruple q3 = {"<=", "z", "w", "t3"};
```

### 4. 跳转指令
```c
// 无条件跳转
Quadruple q1 = {"goto", "", "", "L1"};

// 条件跳转
Quadruple q2 = {"if_false", "x", "", "L1"};
Quadruple q3 = {"if_true", "y", "", "L2"};

// 标签
Quadruple q4 = {"label", "", "", "L1"};
```

### 5. 函数调用
```c
// 函数调用
Quadruple q1 = {"call", "func", "", "t1"};
Quadruple q2 = {"call", "proc", "", ""};

// 参数传递
Quadruple q3 = {"param", "x", "", ""};
Quadruple q4 = {"param", "y", "", ""};
```

## 🔧 四元式生成器

### 基本四元式生成器
```c
class QuadrupleGenerator {
private:
    vector<Quadruple> quadruples;
    int tempVarCount;
    int labelCount;
    
public:
    QuadrupleGenerator() : tempVarCount(0), labelCount(0) {}
    
    void generateCode(ASTNode* node) {
        switch (node->getType()) {
            case EXPRESSION:
                generateExpressionCode(node);
                break;
            case ASSIGNMENT:
                generateAssignmentCode(node);
                break;
            case IF_STATEMENT:
                generateIfCode(node);
                break;
            case WHILE_STATEMENT:
                generateWhileCode(node);
                break;
            case FUNCTION_CALL:
                generateFunctionCallCode(node);
                break;
        }
    }
    
    void generateExpressionCode(ASTNode* node) {
        if (node->getType() == BINARY_OP) {
            string op = node->getValue();
            ASTNode* left = node->getChild(0);
            ASTNode* right = node->getChild(1);
            
            // 递归生成左右子表达式的代码
            generateExpressionCode(left);
            generateExpressionCode(right);
            
            // 生成当前操作的代码
            string tempVar = getTempVar();
            Quadruple q(op, left->getAttribute("result"), right->getAttribute("result"), tempVar);
            quadruples.push_back(q);
            
            // 设置当前节点的结果
            node->setAttribute("result", tempVar);
        } else if (node->getType() == IDENTIFIER) {
            node->setAttribute("result", node->getValue());
        } else if (node->getType() == NUMBER) {
            node->setAttribute("result", node->getValue());
        }
    }
    
    void generateAssignmentCode(ASTNode* node) {
        string varName = node->getChild(0)->getValue();
        ASTNode* value = node->getChild(1);
        
        // 生成值的代码
        generateExpressionCode(value);
        
        // 生成赋值操作
        Quadruple q("=", value->getAttribute("result"), "", varName);
        quadruples.push_back(q);
    }
    
    void generateIfCode(ASTNode* node) {
        ASTNode* condition = node->getChild(0);
        ASTNode* thenBlock = node->getChild(1);
        ASTNode* elseBlock = node->getChild(2);
        
        // 生成条件表达式的代码
        generateExpressionCode(condition);
        
        // 生成条件跳转
        string falseLabel = getLabel();
        Quadruple q1("if_false", condition->getAttribute("result"), "", falseLabel);
        quadruples.push_back(q1);
        
        // 生成then块的代码
        generateCode(thenBlock);
        
        // 生成跳转到if语句结束的代码
        string endLabel = getLabel();
        Quadruple q2("goto", "", "", endLabel);
        quadruples.push_back(q2);
        
        // 生成else块的代码
        Quadruple q3("label", "", "", falseLabel);
        quadruples.push_back(q3);
        generateCode(elseBlock);
        
        // 生成if语句结束标签
        Quadruple q4("label", "", "", endLabel);
        quadruples.push_back(q4);
    }
    
    void generateWhileCode(ASTNode* node) {
        ASTNode* condition = node->getChild(0);
        ASTNode* body = node->getChild(1);
        
        // 生成循环开始标签
        string startLabel = getLabel();
        Quadruple q1("label", "", "", startLabel);
        quadruples.push_back(q1);
        
        // 生成条件表达式的代码
        generateExpressionCode(condition);
        
        // 生成条件跳转
        string endLabel = getLabel();
        Quadruple q2("if_false", condition->getAttribute("result"), "", endLabel);
        quadruples.push_back(q2);
        
        // 生成循环体的代码
        generateCode(body);
        
        // 生成跳转到循环开始的代码
        Quadruple q3("goto", "", "", startLabel);
        quadruples.push_back(q3);
        
        // 生成循环结束标签
        Quadruple q4("label", "", "", endLabel);
        quadruples.push_back(q4);
    }
    
    void generateFunctionCallCode(ASTNode* node) {
        string funcName = node->getChild(0)->getValue();
        ASTNode* arguments = node->getChild(1);
        
        // 生成参数代码
        for (auto arg : arguments->getChildren()) {
            generateExpressionCode(arg);
            Quadruple q("param", arg->getAttribute("result"), "", "");
            quadruples.push_back(q);
        }
        
        // 生成函数调用
        string resultVar = getTempVar();
        Quadruple q("call", funcName, "", resultVar);
        quadruples.push_back(q);
        
        node->setAttribute("result", resultVar);
    }
    
    string getTempVar() {
        return "t" + to_string(tempVarCount++);
    }
    
    string getLabel() {
        return "L" + to_string(labelCount++);
    }
    
    vector<Quadruple> getQuadruples() {
        return quadruples;
    }
};
```

```c
class AdvancedQuadrupleGenerator {
private:
    QuadrupleGenerator baseGenerator;
    map<string, FunctionInfo> functions;
    vector<BasicBlock> basicBlocks;
    
public:
    void generateAdvancedCode(ASTNode* node) {
        switch (node->getType()) {
            case FUNCTION_DEFINITION:
                generateFunctionDefinitionCode(node);
                break;
            case CLASS_DEFINITION:
                generateClassDefinitionCode(node);
                break;
            case ARRAY_ACCESS:
                generateArrayAccessCode(node);
                break;
            case POINTER_DEREFERENCE:
                generatePointerDereferenceCode(node);
                break;
        }
        
        baseGenerator.generateCode(node);
    }
    
    void generateFunctionDefinitionCode(ASTNode* node) {
        string funcName = node->getChild(1)->getValue();
        ASTNode* parameters = node->getChild(2);
        ASTNode* body = node->getChild(3);
        
        // 创建函数信息
        FunctionInfo funcInfo;
        funcInfo.name = funcName;
        funcInfo.parameters = extractParameters(parameters);
        
        // 生成函数开始标签
        Quadruple q1("func_start", funcName, "", "");
        baseGenerator.getQuadruples().push_back(q1);
        
        // 生成参数处理代码
        generateParameterCode(parameters);
        
        // 生成函数体代码
        generateCode(body);
        
        // 生成函数结束标签
        Quadruple q2("func_end", funcName, "", "");
        baseGenerator.getQuadruples().push_back(q2);
    }
    
    void generateClassDefinitionCode(ASTNode* node) {
        string className = node->getChild(0)->getValue();
        ASTNode* members = node->getChild(1);
        
        // 生成类开始标签
        Quadruple q1("class_start", className, "", "");
        baseGenerator.getQuadruples().push_back(q1);
        
        // 生成成员变量和方法的代码
        for (auto member : members->getChildren()) {
            generateCode(member);
        }
        
        // 生成类结束标签
        Quadruple q2("class_end", className, "", "");
        baseGenerator.getQuadruples().push_back(q2);
    }
    
    void generateArrayAccessCode(ASTNode* node) {
        string arrayName = node->getChild(0)->getValue();
        ASTNode* index = node->getChild(1);
        
        // 生成索引表达式的代码
        generateExpressionCode(index);
        
        // 生成数组访问代码
        string tempVar = getTempVar();
        Quadruple q("array_access", arrayName, index->getAttribute("result"), tempVar);
        baseGenerator.getQuadruples().push_back(q);
        
        node->setAttribute("result", tempVar);
    }
    
    void generatePointerDereferenceCode(ASTNode* node) {
        ASTNode* pointer = node->getChild(0);
        
        // 生成指针表达式的代码
        generateExpressionCode(pointer);
        
        // 生成指针解引用代码
        string tempVar = getTempVar();
        Quadruple q("deref", pointer->getAttribute("result"), "", tempVar);
        baseGenerator.getQuadruples().push_back(q);
        
        node->setAttribute("result", tempVar);
    }
    
    vector<string> extractParameters(ASTNode* node) {
        vector<string> parameters;
        for (auto param : node->getChildren()) {
            parameters.push_back(param->getChild(1)->getValue());
        }
        return parameters;
    }
    
    void generateParameterCode(ASTNode* node) {
        for (auto param : node->getChildren()) {
            string paramName = param->getChild(1)->getValue();
            Quadruple q("param", paramName, "", "");
            baseGenerator.getQuadruples().push_back(q);
        }
    }
};
```

## 📈 四元式优化

### 基本优化技术
```c
class QuadrupleOptimizer {
private:
    vector<Quadruple> quadruples;
    
public:
    QuadrupleOptimizer(vector<Quadruple> quads) : quadruples(quads) {}
    
    vector<Quadruple> optimize() {
        // 常量折叠
        constantFolding();
        
        // 死代码消除
        deadCodeElimination();
        
        // 公共子表达式消除
        commonSubexpressionElimination();
        
        // 强度削弱
        strengthReduction();
        
        return quadruples;
    }
    
    void constantFolding() {
        for (int i = 0; i < quadruples.size(); i++) {
            Quadruple& q = quadruples[i];
            
            if (q.op == "+" && isConstant(q.arg1) && isConstant(q.arg2)) {
                int val1 = stoi(q.arg1);
                int val2 = stoi(q.arg2);
                int result = val1 + val2;
                
                // 替换为常量
                q.op = "=";
                q.arg1 = to_string(result);
                q.arg2 = "";
            }
        }
    }
    
    void deadCodeElimination() {
        set<string> usedVars;
        
        // 从后往前扫描，标记使用的变量
        for (int i = quadruples.size() - 1; i >= 0; i--) {
            Quadruple& q = quadruples[i];
            
            // 如果结果变量被使用，则标记操作数
            if (usedVars.find(q.result) != usedVars.end()) {
                if (!q.arg1.empty()) usedVars.insert(q.arg1);
                if (!q.arg2.empty()) usedVars.insert(q.arg2);
            }
        }
        
        // 移除未使用的代码
        vector<Quadruple> optimized;
        for (auto& q : quadruples) {
            if (usedVars.find(q.result) != usedVars.end() || q.op == "label" || q.op == "goto") {
                optimized.push_back(q);
            }
        }
        
        quadruples = optimized;
    }
    
    void commonSubexpressionElimination() {
        map<string, string> expressionMap;
        
        for (int i = 0; i < quadruples.size(); i++) {
            Quadruple& q = quadruples[i];
            
            if (q.op == "+" || q.op == "-" || q.op == "*" || q.op == "/") {
                string expr = q.op + " " + q.arg1 + " " + q.arg2;
                
                if (expressionMap.find(expr) != expressionMap.end()) {
                    // 替换为已计算的表达式
                    q.op = "=";
                    q.arg1 = expressionMap[expr];
                    q.arg2 = "";
                } else {
                    expressionMap[expr] = q.result;
                }
            }
        }
    }
    
    void strengthReduction() {
        for (int i = 0; i < quadruples.size(); i++) {
            Quadruple& q = quadruples[i];
            
            // 乘法替换为加法
            if (q.op == "*" && isConstant(q.arg2) && stoi(q.arg2) == 2) {
                q.op = "+";
                q.arg2 = q.arg1;
            }
            
            // 除法替换为右移
            if (q.op == "/" && isConstant(q.arg2) && isPowerOfTwo(stoi(q.arg2))) {
                q.op = ">>";
                q.arg2 = to_string(log2(stoi(q.arg2)));
            }
        }
    }
    
    bool isConstant(string str) {
        return !str.empty() && isdigit(str[0]);
    }
    
    bool isPowerOfTwo(int n) {
        return n > 0 && (n & (n - 1)) == 0;
    }
    
    int log2(int n) {
        int result = 0;
        while (n > 1) {
            n >>= 1;
            result++;
        }
        return result;
    }
};
```

```c
private:
    QuadrupleOptimizer baseOptimizer;
    ControlFlowGraph cfg;
    
public:
    vector<Quadruple> optimizeAdvanced(vector<Quadruple> quadruples) {
        // 构建控制流图
        buildControlFlowGraph(quadruples);
        
        // 数据流分析
        performDataFlowAnalysis();
        
        // 循环优化
        optimizeLoops();
        
        // 全局优化
        performGlobalOptimization();
        
        return baseOptimizer.optimize();
    }
    
    void buildControlFlowGraph(vector<Quadruple> quadruples) {
        // 构建基本块
        vector<BasicBlock> blocks;
        BasicBlock currentBlock(0);
        
        for (int i = 0; i < quadruples.size(); i++) {
            Quadruple& q = quadruples[i];
            
            if (q.op == "label") {
                // 开始新块
                if (!currentBlock.instructions.empty()) {
                    blocks.push_back(currentBlock);
                    currentBlock = BasicBlock(blocks.size());
                }
            } else if (q.op == "goto" || q.op == "if_false") {
                // 结束当前块
                currentBlock.instructions.push_back(q);
                blocks.push_back(currentBlock);
                currentBlock = BasicBlock(blocks.size());
            } else {
                currentBlock.instructions.push_back(q);
            }
        }
        
        if (!currentBlock.instructions.empty()) {
            blocks.push_back(currentBlock);
        }
        
        cfg.blocks = blocks;
    }
    
    void performDataFlowAnalysis() {
        // 活跃变量分析
        performLivenessAnalysis();
        
        // 到达定值分析
        performReachingDefinitionAnalysis();
        
        // 可用表达式分析
        performAvailableExpressionAnalysis();
    }
    
    void performLivenessAnalysis() {
        // 计算每个基本块的活跃变量
        for (auto& block : cfg.blocks) {
            set<string> liveVars;
            
            // 从后往前扫描指令
            for (int i = block.instructions.size() - 1; i >= 0; i--) {
                Quadruple& q = block.instructions[i];
                
                // 如果操作数被使用，则标记为活跃
                if (!q.arg1.empty() && liveVars.find(q.arg1) != liveVars.end()) {
                    liveVars.insert(q.arg1);
                }
                if (!q.arg2.empty() && liveVars.find(q.arg2) != liveVars.end()) {
                    liveVars.insert(q.arg2);
                }
                
                // 如果结果变量不再活跃，则移除
                if (liveVars.find(q.result) != liveVars.end()) {
                    liveVars.erase(q.result);
                }
            }
            
            block.liveVars = liveVars;
        }
    }
    
    void performReachingDefinitionAnalysis() {
        // 计算每个基本块的到达定值
        for (auto& block : cfg.blocks) {
            set<string> reachingDefs;
            
            // 从前往后扫描指令
            for (auto& q : block.instructions) {
                // 如果操作数被定义，则标记为到达定值
                if (!q.arg1.empty()) {
                    reachingDefs.insert(q.arg1);
                }
                if (!q.arg2.empty()) {
                    reachingDefs.insert(q.arg2);
                }
                
                // 如果结果变量被定义，则移除其他定义
                if (!q.result.empty()) {
                    reachingDefs.erase(q.result);
                    reachingDefs.insert(q.result);
                }
            }
            
            block.reachingDefs = reachingDefs;
        }
    }
    
    void performAvailableExpressionAnalysis() {
        // 计算每个基本块的可用表达式
        for (auto& block : cfg.blocks) {
            set<string> availableExprs;
            
            // 从前往后扫描指令
            for (auto& q : block.instructions) {
                if (q.op == "+" || q.op == "-" || q.op == "*" || q.op == "/") {
                    string expr = q.op + " " + q.arg1 + " " + q.arg2;
                    availableExprs.insert(expr);
                }
            }
            
            block.availableExprs = availableExprs;
        }
    }
    
    void optimizeLoops() {
        // 循环不变式外提
        loopInvariantCodeMotion();
        
        // 循环展开
        loopUnrolling();
        
        // 循环融合
        loopFusion();
    }
    
    void loopInvariantCodeMotion() {
        // 识别循环不变式
        for (auto& block : cfg.blocks) {
            if (isLoopHeader(block)) {
                // 将循环不变式外提到循环前
                moveInvariantCode(block);
            }
        }
    }
    
    void loopUnrolling() {
        // 展开小循环
        for (auto& block : cfg.blocks) {
            if (isSmallLoop(block)) {
                unrollLoop(block);
            }
        }
    }

        for (int i = 0; i < cfg.blocks.size() - 1; i++) {

            }
        }
    }

        globalDeadCodeElimination();
        
        // 全局公共子表达式消除
        globalCommonSubexpressionElimination();
    }
    
    void globalConstantPropagation() {
        // 在整个程序中传播常量
        map<string, int> constants;
        
        for (auto& block : cfg.blocks) {
            for (auto& q : block.instructions) {
                if (q.op == "=" && isConstant(q.arg1)) {
                    constants[q.result] = stoi(q.arg1);
                }
            }
        }
        
        // 替换常量使用
        for (auto& block : cfg.blocks) {
            for (auto& q : block.instructions) {
                if (constants.find(q.arg1) != constants.end()) {
                    q.arg1 = to_string(constants[q.arg1]);
                }
                if (constants.find(q.arg2) != constants.end()) {
                    q.arg2 = to_string(constants[q.arg2]);
                }
            }
        }
    }
    
    void globalDeadCodeElimination() {
        // 全局死代码消除
        set<string> usedVars;
        
        // 标记所有使用的变量
        for (auto& block : cfg.blocks) {
            for (auto& q : block.instructions) {
                if (q.op == "goto" || q.op == "if_false") {
                    usedVars.insert(q.arg1);
                }
            }
        }
        
        // 移除未使用的代码
        for (auto& block : cfg.blocks) {
            vector<Quadruple> optimized;
            for (auto& q : block.instructions) {
                if (usedVars.find(q.result) != usedVars.end() || q.op == "label" || q.op == "goto") {
                    optimized.push_back(q);
                }
            }
            block.instructions = optimized;
        }
    }

        // 全局公共子表达式消除
        map<string, string> expressionMap;

                    } else {
                        expressionMap[expr] = q.result;
```

### 四元式验证
```c
class QuadrupleValidator {
private:
    vector<Quadruple> quadruples;
    
public:
    QuadrupleValidator(vector<Quadruple> quads) : quadruples(quads) {}
    
    bool validate() {
        // 检查语法正确性
        if (!checkSyntax()) {
            return false;
        }
        
        // 检查语义正确性
        if (!checkSemantics()) {
            return false;
        }
        
        // 检查类型正确性
        if (!checkTypes()) {
            return false;
        }
        
        return true;
    }
    
    bool checkSyntax() {
        for (auto& q : quadruples) {
            if (q.op.empty()) {
                cout << "Error: Empty operation" << endl;
                return false;
            }
            
            if (q.result.empty()) {
                cout << "Error: Empty result" << endl;
                return false;
            }
        }
        
        return true;
    }
    
    bool checkSemantics() {
        // 检查变量定义
        set<string> definedVars;
        
        for (auto& q : quadruples) {
            if (q.op == "=") {
                definedVars.insert(q.result);
            } else if (q.op == "+" || q.op == "-" || q.op == "*" || q.op == "/") {
                if (definedVars.find(q.arg1) == definedVars.end()) {
                    cout << "Error: Undefined variable " << q.arg1 << endl;
                    return false;
                }
                if (definedVars.find(q.arg2) == definedVars.end()) {
                    cout << "Error: Undefined variable " << q.arg2 << endl;
                    return false;
                }
                definedVars.insert(q.result);
            }
        }
        
        return true;
    }
    
    bool checkTypes() {
        // 检查类型兼容性
        map<string, string> varTypes;
        
        for (auto& q : quadruples) {
            if (q.op == "=") {
                if (isConstant(q.arg1)) {
                    varTypes[q.result] = "int";
                } else {
                    varTypes[q.result] = varTypes[q.arg1];
                }
            } else if (q.op == "+" || q.op == "-" || q.op == "*" || q.op == "/") {
                string type1 = varTypes[q.arg1];
                string type2 = varTypes[q.arg2];
                
                if (type1 != type2) {
                    cout << "Error: Type mismatch between " << q.arg1 << " and " << q.arg2 << endl;
                    return false;
                }
                
                varTypes[q.result] = type1;
            }
        }
        
        return true;
    }
    
    bool isConstant(string str) {
        return !str.empty() && isdigit(str[0]);
    }
};
```
