# 三地址码生成

## 🎯 三地址码生成概述

**三地址码生成**是中间代码生成的核心技术，将抽象语法树转换为三地址码形式的中间表示。三地址码具有结构简单、易于优化和代码生成的特点，是编译器中最常用的中间表示形式。

## 🔍 基本概念

### 三地址码定义
三地址码是一种中间表示形式，每条指令最多包含三个地址（操作数），基本形式为：`result = operand1 operator operand2`

### 三地址码特点
- **结构简单**：每条指令最多三个操作数
- **易于优化**：便于进行各种代码优化
- **便于生成**：容易转换为目标代码
- **语义清晰**：保持源程序的语义信息

## 📊 三地址码类型

### 1. 算术运算
```c
// 基本算术运算
t1 = a + b
t2 = t1 * c
result = t2 - d

// 一元运算
t1 = -a
t2 = !b
```

### 2. 赋值运算
```c
// 直接赋值
x = y
x = 5

// 复合赋值
x = x + 1
x = x * 2
```

### 3. 比较运算
```c
// 比较运算
t1 = a > b
t2 = x == y
t3 = z <= w
```

### 4. 跳转指令
```c
// 无条件跳转
goto L1

// 条件跳转
if (x > 0) goto L1
if (y == 0) goto L2
```

### 5. 函数调用
```c
// 函数调用
t1 = call func, arg1, arg2
call proc, arg1, arg2

// 参数传递
param x
param y
```

## 🔧 三地址码生成器

### 基本三地址码生成器
```c
class ThreeAddressCodeGenerator {
private:
    vector<ThreeAddressCode> instructions;
    int tempVarCount;
    int labelCount;
    
public:
    ThreeAddressCodeGenerator() : tempVarCount(0), labelCount(0) {}
    
    void generateCode(ASTNode* node) {
        switch (node->getType()) {
            case EXPRESSION:
                generateExpressionCode(node);
                break;
            case ASSIGNMENT:
                generateAssignmentCode(node);
                break;
            case IF_STATEMENT:
                generateIfCode(node);
                break;
            case WHILE_STATEMENT:
                generateWhileCode(node);
                break;
            case FUNCTION_CALL:
                generateFunctionCallCode(node);
                break;
        }
    }
    
    void generateExpressionCode(ASTNode* node) {
        if (node->getType() == BINARY_OP) {
            string op = node->getValue();
            ASTNode* left = node->getChild(0);
            ASTNode* right = node->getChild(1);
            
            // 递归生成左右子表达式的代码
            generateExpressionCode(left);
            generateExpressionCode(right);
            
            // 生成当前操作的代码
            string tempVar = getTempVar();
            ThreeAddressCode tac(op, left->getAttribute("result"), right->getAttribute("result"), tempVar);
            instructions.push_back(tac);
            
            // 设置当前节点的结果
            node->setAttribute("result", tempVar);
        } else if (node->getType() == IDENTIFIER) {
            node->setAttribute("result", node->getValue());
        } else if (node->getType() == NUMBER) {
            node->setAttribute("result", node->getValue());
        }
    }
    
    void generateAssignmentCode(ASTNode* node) {
        string varName = node->getChild(0)->getValue();
        ASTNode* value = node->getChild(1);
        
        // 生成值的代码
        generateExpressionCode(value);
        
        // 生成赋值操作
        ThreeAddressCode tac("=", value->getAttribute("result"), "", varName);
        instructions.push_back(tac);
    }
    
    void generateIfCode(ASTNode* node) {
        ASTNode* condition = node->getChild(0);
        ASTNode* thenBlock = node->getChild(1);
        ASTNode* elseBlock = node->getChild(2);
        
        // 生成条件表达式的代码
        generateExpressionCode(condition);
        
        // 生成条件跳转
        string falseLabel = getLabel();
        ThreeAddressCode tac1("if_false", condition->getAttribute("result"), "", falseLabel);
        instructions.push_back(tac1);
        
        // 生成then块的代码
        generateCode(thenBlock);
        
        // 生成跳转到if语句结束的代码
        string endLabel = getLabel();
        ThreeAddressCode tac2("goto", "", "", endLabel);
        instructions.push_back(tac2);
        
        // 生成else块的代码
        ThreeAddressCode tac3("label", "", "", falseLabel);
        instructions.push_back(tac3);
        generateCode(elseBlock);
        
        // 生成if语句结束标签
        ThreeAddressCode tac4("label", "", "", endLabel);
        instructions.push_back(tac4);
    }
    
    void generateWhileCode(ASTNode* node) {
        ASTNode* condition = node->getChild(0);
        ASTNode* body = node->getChild(1);
        
        // 生成循环开始标签
        string startLabel = getLabel();
        ThreeAddressCode tac1("label", "", "", startLabel);
        instructions.push_back(tac1);
        
        // 生成条件表达式的代码
        generateExpressionCode(condition);
        
        // 生成条件跳转
        string endLabel = getLabel();
        ThreeAddressCode tac2("if_false", condition->getAttribute("result"), "", endLabel);
        instructions.push_back(tac2);
        
        // 生成循环体的代码
        generateCode(body);
        
        // 生成跳转到循环开始的代码
        ThreeAddressCode tac3("goto", "", "", startLabel);
        instructions.push_back(tac3);
        
        // 生成循环结束标签
        ThreeAddressCode tac4("label", "", "", endLabel);
        instructions.push_back(tac4);
    }
    
    void generateFunctionCallCode(ASTNode* node) {
        string funcName = node->getChild(0)->getValue();
        ASTNode* arguments = node->getChild(1);
        
        // 生成参数代码
        for (auto arg : arguments->getChildren()) {
            generateExpressionCode(arg);
            ThreeAddressCode tac("param", arg->getAttribute("result"), "", "");
            instructions.push_back(tac);
        }
        
        // 生成函数调用
        string resultVar = getTempVar();
        ThreeAddressCode tac("call", funcName, "", resultVar);
        instructions.push_back(tac);
        
        node->setAttribute("result", resultVar);
    }
    
    string getTempVar() {
        return "t" + to_string(tempVarCount++);
    }
    
    string getLabel() {
        return "L" + to_string(labelCount++);
    }
    
    vector<ThreeAddressCode> getInstructions() {
        return instructions;
    }
};
```

```c
class AdvancedThreeAddressCodeGenerator {
private:
    ThreeAddressCodeGenerator baseGenerator;
    map<string, FunctionInfo> functions;
    vector<BasicBlock> basicBlocks;
    
public:
    void generateAdvancedCode(ASTNode* node) {
        switch (node->getType()) {
            case FUNCTION_DEFINITION:
                generateFunctionDefinitionCode(node);
                break;
            case CLASS_DEFINITION:
                generateClassDefinitionCode(node);
                break;
            case ARRAY_ACCESS:
                generateArrayAccessCode(node);
                break;
            case POINTER_DEREFERENCE:
                generatePointerDereferenceCode(node);
                break;
        }
        
        baseGenerator.generateCode(node);
    }
    
    void generateFunctionDefinitionCode(ASTNode* node) {
        string funcName = node->getChild(1)->getValue();
        ASTNode* parameters = node->getChild(2);
        ASTNode* body = node->getChild(3);
        
        // 创建函数信息
        FunctionInfo funcInfo;
        funcInfo.name = funcName;
        funcInfo.parameters = extractParameters(parameters);
        
        // 生成函数开始标签
        ThreeAddressCode tac1("func_start", funcName, "", "");
        baseGenerator.getInstructions().push_back(tac1);
        
        // 生成参数处理代码
        generateParameterCode(parameters);
        
        // 生成函数体代码
        generateCode(body);
        
        // 生成函数结束标签
        ThreeAddressCode tac2("func_end", funcName, "", "");
        baseGenerator.getInstructions().push_back(tac2);
    }
    
    void generateClassDefinitionCode(ASTNode* node) {
        string className = node->getChild(0)->getValue();
        ASTNode* members = node->getChild(1);
        
        // 生成类开始标签
        ThreeAddressCode tac1("class_start", className, "", "");
        baseGenerator.getInstructions().push_back(tac1);
        
        // 生成成员变量和方法的代码
        for (auto member : members->getChildren()) {
            generateCode(member);
        }
        
        // 生成类结束标签
        ThreeAddressCode tac2("class_end", className, "", "");
        baseGenerator.getInstructions().push_back(tac2);
    }
    
    void generateArrayAccessCode(ASTNode* node) {
        string arrayName = node->getChild(0)->getValue();
        ASTNode* index = node->getChild(1);
        
        // 生成索引表达式的代码
        generateExpressionCode(index);
        
        // 生成数组访问代码
        string tempVar = getTempVar();
        ThreeAddressCode tac("array_access", arrayName, index->getAttribute("result"), tempVar);
        baseGenerator.getInstructions().push_back(tac);
        
        node->setAttribute("result", tempVar);
    }
    
    void generatePointerDereferenceCode(ASTNode* node) {
        ASTNode* pointer = node->getChild(0);
        
        // 生成指针表达式的代码
        generateExpressionCode(pointer);
        
        // 生成指针解引用代码
        string tempVar = getTempVar();
        ThreeAddressCode tac("deref", pointer->getAttribute("result"), "", tempVar);
        baseGenerator.getInstructions().push_back(tac);
        
        node->setAttribute("result", tempVar);
    }
    
    vector<string> extractParameters(ASTNode* node) {
        vector<string> parameters;
        for (auto param : node->getChildren()) {
            parameters.push_back(param->getChild(1)->getValue());
        }
        return parameters;
    }
    
    void generateParameterCode(ASTNode* node) {
        for (auto param : node->getChildren()) {
            string paramName = param->getChild(1)->getValue();
            ThreeAddressCode tac("param", paramName, "", "");
            baseGenerator.getInstructions().push_back(tac);
        }
    }
};
```

## 📈 三地址码优化

### 基本优化技术
```c
class ThreeAddressCodeOptimizer {
private:
    vector<ThreeAddressCode> instructions;
    
public:
    ThreeAddressCodeOptimizer(vector<ThreeAddressCode> insts) : instructions(insts) {}
    
    vector<ThreeAddressCode> optimize() {
        // 常量折叠
        constantFolding();
        
        // 死代码消除
        deadCodeElimination();
        
        // 公共子表达式消除
        commonSubexpressionElimination();
        
        // 强度削弱
        strengthReduction();
        
        return instructions;
    }
    
    void constantFolding() {
        for (int i = 0; i < instructions.size(); i++) {
            ThreeAddressCode& tac = instructions[i];
            
            if (tac.op == "+" && isConstant(tac.arg1) && isConstant(tac.arg2)) {
                int val1 = stoi(tac.arg1);
                int val2 = stoi(tac.arg2);
                int result = val1 + val2;
                
                // 替换为常量
                tac.op = "=";
                tac.arg1 = to_string(result);
                tac.arg2 = "";
            }
        }
    }
    
    void deadCodeElimination() {
        set<string> usedVars;
        
        // 从后往前扫描，标记使用的变量
        for (int i = instructions.size() - 1; i >= 0; i--) {
            ThreeAddressCode& tac = instructions[i];
            
            // 如果结果变量被使用，则标记操作数
            if (usedVars.find(tac.result) != usedVars.end()) {
                if (!tac.arg1.empty()) usedVars.insert(tac.arg1);
                if (!tac.arg2.empty()) usedVars.insert(tac.arg2);
            }
        }
        
        // 移除未使用的代码
        vector<ThreeAddressCode> optimized;
        for (auto& tac : instructions) {
            if (usedVars.find(tac.result) != usedVars.end() || tac.op == "label" || tac.op == "goto") {
                optimized.push_back(tac);
            }
        }
        
        instructions = optimized;
    }
    
    void commonSubexpressionElimination() {
        map<string, string> expressionMap;
        
        for (int i = 0; i < instructions.size(); i++) {
            ThreeAddressCode& tac = instructions[i];
            
            if (tac.op == "+" || tac.op == "-" || tac.op == "*" || tac.op == "/") {
                string expr = tac.op + " " + tac.arg1 + " " + tac.arg2;
                
                if (expressionMap.find(expr) != expressionMap.end()) {
                    // 替换为已计算的表达式
                    tac.op = "=";
                    tac.arg1 = expressionMap[expr];
                    tac.arg2 = "";
                } else {
                    expressionMap[expr] = tac.result;
                }
            }
        }
    }
    
    void strengthReduction() {
        for (int i = 0; i < instructions.size(); i++) {
            ThreeAddressCode& tac = instructions[i];
            
            // 乘法替换为加法
            if (tac.op == "*" && isConstant(tac.arg2) && stoi(tac.arg2) == 2) {
                tac.op = "+";
                tac.arg2 = tac.arg1;
            }
            
            // 除法替换为右移
            if (tac.op == "/" && isConstant(tac.arg2) && isPowerOfTwo(stoi(tac.arg2))) {
                tac.op = ">>";
                tac.arg2 = to_string(log2(stoi(tac.arg2)));
            }
        }
    }
    
    bool isConstant(string str) {
        return !str.empty() && isdigit(str[0]);
    }
    
    bool isPowerOfTwo(int n) {
        return n > 0 && (n & (n - 1)) == 0;
    }
    
    int log2(int n) {
        int result = 0;
        while (n > 1) {
            n >>= 1;
            result++;
        }
        return result;
    }
};
```

```c
class AdvancedThreeAddressCodeOptimizer {
private:
    ThreeAddressCodeOptimizer baseOptimizer;
    ControlFlowGraph cfg;
    
public:
    vector<ThreeAddressCode> optimizeAdvanced(vector<ThreeAddressCode> instructions) {
        // 构建控制流图
        buildControlFlowGraph(instructions);
        
        // 数据流分析
        performDataFlowAnalysis();
        
        // 循环优化
        optimizeLoops();
        
        // 全局优化
        performGlobalOptimization();
        
        return baseOptimizer.optimize();
    }
    
    void buildControlFlowGraph(vector<ThreeAddressCode> instructions) {
        // 构建基本块
        vector<BasicBlock> blocks;
        BasicBlock currentBlock(0);
        
        for (int i = 0; i < instructions.size(); i++) {
            ThreeAddressCode& tac = instructions[i];
            
            if (tac.op == "label") {
                // 开始新块
                if (!currentBlock.instructions.empty()) {
                    blocks.push_back(currentBlock);
                    currentBlock = BasicBlock(blocks.size());
                }
            } else if (tac.op == "goto" || tac.op == "if_false") {
                // 结束当前块
                currentBlock.instructions.push_back(tac);
                blocks.push_back(currentBlock);
                currentBlock = BasicBlock(blocks.size());
            } else {
                currentBlock.instructions.push_back(tac);
            }
        }
        
        if (!currentBlock.instructions.empty()) {
            blocks.push_back(currentBlock);
        }
        
        cfg.blocks = blocks;
    }
    
    void performDataFlowAnalysis() {
        // 活跃变量分析
        performLivenessAnalysis();
        
        // 到达定值分析
        performReachingDefinitionAnalysis();
        
        // 可用表达式分析
        performAvailableExpressionAnalysis();
    }
    
    void performLivenessAnalysis() {
        // 计算每个基本块的活跃变量
        for (auto& block : cfg.blocks) {
            set<string> liveVars;
            
            // 从后往前扫描指令
            for (int i = block.instructions.size() - 1; i >= 0; i--) {
                ThreeAddressCode& tac = block.instructions[i];
                
                // 如果操作数被使用，则标记为活跃
                if (!tac.arg1.empty() && liveVars.find(tac.arg1) != liveVars.end()) {
                    liveVars.insert(tac.arg1);
                }
                if (!tac.arg2.empty() && liveVars.find(tac.arg2) != liveVars.end()) {
                    liveVars.insert(tac.arg2);
                }
                
                // 如果结果变量不再活跃，则移除
                if (liveVars.find(tac.result) != liveVars.end()) {
                    liveVars.erase(tac.result);
                }
            }
            
            block.liveVars = liveVars;
        }
    }
    
    void performReachingDefinitionAnalysis() {
        // 计算每个基本块的到达定值
        for (auto& block : cfg.blocks) {
            set<string> reachingDefs;
            
            // 从前往后扫描指令
            for (auto& tac : block.instructions) {
                // 如果操作数被定义，则标记为到达定值
                if (!tac.arg1.empty()) {
                    reachingDefs.insert(tac.arg1);
                }
                if (!tac.arg2.empty()) {
                    reachingDefs.insert(tac.arg2);
                }
                
                // 如果结果变量被定义，则移除其他定义
                if (!tac.result.empty()) {
                    reachingDefs.erase(tac.result);
                    reachingDefs.insert(tac.result);
                }
            }
            
            block.reachingDefs = reachingDefs;
        }
    }
    
    void performAvailableExpressionAnalysis() {
        // 计算每个基本块的可用表达式
        for (auto& block : cfg.blocks) {
            set<string> availableExprs;
            
            // 从前往后扫描指令
            for (auto& tac : block.instructions) {
                if (tac.op == "+" || tac.op == "-" || tac.op == "*" || tac.op == "/") {
                    string expr = tac.op + " " + tac.arg1 + " " + tac.arg2;
                    availableExprs.insert(expr);
                }
            }
            
            block.availableExprs = availableExprs;
        }
    }
    
    void optimizeLoops() {
        // 循环不变式外提
        loopInvariantCodeMotion();
        
        // 循环展开
        loopUnrolling();
        
        // 循环融合
        loopFusion();
    }
    
    void loopInvariantCodeMotion() {
        // 识别循环不变式
        for (auto& block : cfg.blocks) {
            if (isLoopHeader(block)) {
                // 将循环不变式外提到循环前
                moveInvariantCode(block);
            }
        }
    }
    
    void loopUnrolling() {
        // 展开小循环
        for (auto& block : cfg.blocks) {
            if (isSmallLoop(block)) {
                unrollLoop(block);
            }
        }
    }
    
    void loopFusion() {
        // 融合相邻的循环
        for (int i = 0; i < cfg.blocks.size() - 1; i++) {
            if (canFuseLoops(cfg.blocks[i], cfg.blocks[i + 1])) {
                fuseLoops(cfg.blocks[i], cfg.blocks[i + 1]);
            }
        }
    }

        globalConstantPropagation();
        
        // 全局死代码消除
        globalDeadCodeElimination();
        
        // 全局公共子表达式消除
        globalCommonSubexpressionElimination();
    }
    
    void globalConstantPropagation() {
        // 在整个程序中传播常量
        map<string, int> constants;
        
        for (auto& block : cfg.blocks) {
            for (auto& tac : block.instructions) {
                if (tac.op == "=" && isConstant(tac.arg1)) {
                    constants[tac.result] = stoi(tac.arg1);
                }
            }
        }
        
        // 替换常量使用
        for (auto& block : cfg.blocks) {
            for (auto& tac : block.instructions) {
                if (constants.find(tac.arg1) != constants.end()) {
                    tac.arg1 = to_string(constants[tac.arg1]);
                }
                if (constants.find(tac.arg2) != constants.end()) {
                    tac.arg2 = to_string(constants[tac.arg2]);
                }
            }
        }
    }
    
    void globalDeadCodeElimination() {
        // 全局死代码消除
        set<string> usedVars;
        
        // 标记所有使用的变量
        for (auto& block : cfg.blocks) {
            for (auto& tac : block.instructions) {
                if (tac.op == "goto" || tac.op == "if_false") {
                    usedVars.insert(tac.arg1);
                }
            }
        }
        
        // 移除未使用的代码
        for (auto& block : cfg.blocks) {
            vector<ThreeAddressCode> optimized;
            for (auto& tac : block.instructions) {
                if (usedVars.find(tac.result) != usedVars.end() || tac.op == "label" || tac.op == "goto") {
                    optimized.push_back(tac);
                }
            }
            block.instructions = optimized;
```

```c
class ThreeAddressCodeConverter {
private:
    vector<ThreeAddressCode> instructions;
    
public:
    ThreeAddressCodeConverter(vector<ThreeAddressCode> insts) : instructions(insts) {}
    
    string convertToAssembly() {
        string assembly;
        
        for (auto& tac : instructions) {
            if (tac.op == "+") {
                assembly += "add " + tac.result + ", " + tac.arg1 + ", " + tac.arg2 + "\n";
            } else if (tac.op == "-") {
                assembly += "sub " + tac.result + ", " + tac.arg1 + ", " + tac.arg2 + "\n";
            } else if (tac.op == "*") {
                assembly += "mul " + tac.result + ", " + tac.arg1 + ", " + tac.arg2 + "\n";
            } else if (tac.op == "/") {
                assembly += "div " + tac.result + ", " + tac.arg1 + ", " + tac.arg2 + "\n";
            } else if (tac.op == "=") {
                assembly += "mov " + tac.result + ", " + tac.arg1 + "\n";
            } else if (tac.op == "goto") {
```

### 三地址码验证
```c
class ThreeAddressCodeValidator {
private:
    vector<ThreeAddressCode> instructions;
    
public:
    ThreeAddressCodeValidator(vector<ThreeAddressCode> insts) : instructions(insts) {}
    
    bool validate() {
        // 检查语法正确性
        if (!checkSyntax()) {
            return false;
        }
        
        // 检查语义正确性
        if (!checkSemantics()) {
            return false;
        }
        
        // 检查类型正确性
        if (!checkTypes()) {
            return false;
        }
        
        return true;
    }
    
    bool checkSyntax() {
        for (auto& tac : instructions) {
            if (tac.op.empty()) {
                cout << "Error: Empty operation" << endl;
                return false;
            }
            
            if (tac.result.empty()) {
                cout << "Error: Empty result" << endl;
                return false;
            }
        }
        
        return true;
    }
    
    bool checkSemantics() {
        // 检查变量定义
        set<string> definedVars;
        
        for (auto& tac : instructions) {
            if (tac.op == "=") {
                definedVars.insert(tac.result);
            } else if (tac.op == "+" || tac.op == "-" || tac.op == "*" || tac.op == "/") {
                if (definedVars.find(tac.arg1) == definedVars.end()) {
                    cout << "Error: Undefined variable " << tac.arg1 << endl;
                    return false;
                }
                if (definedVars.find(tac.arg2) == definedVars.end()) {
                    cout << "Error: Undefined variable " << tac.arg2 << endl;
                    return false;
                }
                definedVars.insert(tac.result);
            }
        }
        
        return true;
    }
    
    bool checkTypes() {
        // 检查类型兼容性
        map<string, string> varTypes;
        
        for (auto& tac : instructions) {
            if (tac.op == "=") {
                if (isConstant(tac.arg1)) {
                    varTypes[tac.result] = "int";
                } else {
                    varTypes[tac.result] = varTypes[tac.arg1];
                }
            } else if (tac.op == "+" || tac.op == "-" || tac.op == "*" || tac.op == "/") {
                string type1 = varTypes[tac.arg1];
                string type2 = varTypes[tac.arg2];
                
                if (type1 != type2) {
                    cout << "Error: Type mismatch between " << tac.arg1 << " and " << tac.arg2 << endl;
                    return false;
                }
                
                varTypes[tac.result] = type1;
            }
        }
        
        return true;
    }
    
    bool isConstant(string str) {
        return !str.empty() && isdigit(str[0]);
    }
};
```
