# 性能分析工具

## 🎯 性能分析工具概述

**性能分析工具**是编译器开发和优化过程中必不可少的工具集，用于分析编译器的性能瓶颈、内存使用情况、执行时间等关键指标。这些工具帮助开发者识别性能问题，优化编译器性能。

## 🔍 工具分类

### 1. 性能分析器
- **GProf**：GNU性能分析器
- **Valgrind Callgrind**：调用图分析器
- **Intel VTune**：Intel性能分析器
- **Perf**：Linux性能分析器

### 2. 内存分析器
- **Valgrind Massif**：堆内存分析器
- **Valgrind Memcheck**：内存错误检测器
- **AddressSanitizer**：地址消毒器
- **MemorySanitizer**：内存消毒器

### 3. 缓存分析器
- **Valgrind Cachegrind**：缓存分析器
- **Intel Cache Analyzer**：Intel缓存分析器
- **Perf Cache**：Linux缓存分析器
- **LIKWID**：性能监控工具

### 4. 时间分析器
- **Chronometer**：时间测量工具
- **High Resolution Timer**：高精度计时器
- **CPU Time Profiler**：CPU时间分析器
- **Wall Clock Profiler**：墙钟时间分析器

```c
// 示例：使用GProf分析编译器性能
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

// 词法分析器结构
typedef struct {
    char* source;
    size_t position;
    size_t length;
    int line;
    int column;
} LexicalAnalyzer;

// Token类型
typedef enum {
    TOKEN_NUMBER,
    TOKEN_IDENTIFIER,
    TOKEN_OPERATOR,
    TOKEN_DELIMITER,
    TOKEN_EOF
} TokenType;

// Token结构
typedef struct {
    TokenType type;
    char* value;
    int line;
    int column;
} Token;

// 创建词法分析器
LexicalAnalyzer* createLexicalAnalyzer(const char* source) {
    LexicalAnalyzer* lexer = (LexicalAnalyzer*)malloc(sizeof(LexicalAnalyzer));
    lexer->source = strdup(source);
    lexer->position = 0;
    lexer->length = strlen(source);
    lexer->line = 1;
    lexer->column = 1;
    return lexer;
}

// 获取当前字符
char getCurrentChar(LexicalAnalyzer* lexer) {
    if (lexer->position >= lexer->length) {
        return '\0';
    }
    return lexer->source[lexer->position];
}

// 前进一个字符
void advance(LexicalAnalyzer* lexer) {
    if (lexer->position < lexer->length) {
        if (lexer->source[lexer->position] == '\n') {
            lexer->line++;
            lexer->column = 1;
        } else {
            lexer->column++;
        }
        lexer->position++;
    }
}

// 跳过空白字符
void skipWhitespace(LexicalAnalyzer* lexer) {
    while (lexer->position < lexer->length && 
           (lexer->source[lexer->position] == ' ' || 
            lexer->source[lexer->position] == '\t' || 
            lexer->source[lexer->position] == '\n' || 
            lexer->source[lexer->position] == '\r')) {
        advance(lexer);
    }
}

// 扫描数字
Token scanNumber(LexicalAnalyzer* lexer) {
    Token token;
    token.type = TOKEN_NUMBER;
    token.line = lexer->line;
    token.column = lexer->column;
    
    size_t start = lexer->position;
    while (lexer->position < lexer->length && 
           lexer->source[lexer->position] >= '0' && 
           lexer->source[lexer->position] <= '9') {
        advance(lexer);
    }
    
    size_t length = lexer->position - start;
    token.value = (char*)malloc(length + 1);
    strncpy(token.value, lexer->source + start, length);
    token.value[length] = '\0';
    
    return token;
}

// 扫描标识符
Token scanIdentifier(LexicalAnalyzer* lexer) {
    Token token;
    token.type = TOKEN_IDENTIFIER;
    token.line = lexer->line;
    token.column = lexer->column;
    
    size_t start = lexer->position;
    while (lexer->position < lexer->length && 
           ((lexer->source[lexer->position] >= 'a' && lexer->source[lexer->position] <= 'z') ||
            (lexer->source[lexer->position] >= 'A' && lexer->source[lexer->position] <= 'Z') ||
            (lexer->source[lexer->position] >= '0' && lexer->source[lexer->position] <= '9'))) {
        advance(lexer);
    }
    
    size_t length = lexer->position - start;
    token.value = (char*)malloc(length + 1);
    strncpy(token.value, lexer->source + start, length);
    token.value[length] = '\0';
    
    return token;
}

// 扫描操作符
Token scanOperator(LexicalAnalyzer* lexer) {
    Token token;
    token.type = TOKEN_OPERATOR;
    token.line = lexer->line;
    token.column = lexer->column;
    
    char op = getCurrentChar(lexer);
    token.value = (char*)malloc(2);
    token.value[0] = op;
    token.value[1] = '\0';
    
    advance(lexer);
    return token;
}

// 扫描分隔符
Token scanDelimiter(LexicalAnalyzer* lexer) {
    Token token;
    token.type = TOKEN_DELIMITER;
    token.line = lexer->line;
    token.column = lexer->column;
    
    char delim = getCurrentChar(lexer);
    token.value = (char*)malloc(2);
    token.value[0] = delim;
    token.value[1] = '\0';
    
    advance(lexer);
    return token;
}

// 获取下一个Token
Token getNextToken(LexicalAnalyzer* lexer) {
    skipWhitespace(lexer);
    
    if (lexer->position >= lexer->length) {
        Token token;
        token.type = TOKEN_EOF;
        token.value = NULL;
        token.line = lexer->line;
        token.column = lexer->column;
        return token;
    }
    
    char current = getCurrentChar(lexer);
    
    if (current >= '0' && current <= '9') {
        return scanNumber(lexer);
    } else if ((current >= 'a' && current <= 'z') || 
               (current >= 'A' && current <= 'Z')) {
        return scanIdentifier(lexer);
    } else if (current == '+' || current == '-' || 
               current == '*' || current == '/' || 
               current == '=') {
        return scanOperator(lexer);
    } else if (current == '(' || current == ')' || 
               current == '{' || current == '}' || 
               current == ';' || current == ',') {
        return scanDelimiter(lexer);
    } else {
        Token token;
        token.type = TOKEN_EOF;
        token.value = NULL;
        token.line = lexer->line;
        token.column = lexer->column;
        return token;
    }
}

// 词法分析
void lexicalAnalysis(LexicalAnalyzer* lexer) {
    printf("开始词法分析\n");
    
    Token token;
    int tokenCount = 0;
    
    do {
        token = getNextToken(lexer);
        tokenCount++;
        
        if (token.value) {
            free(token.value);
        }
    } while (token.type != TOKEN_EOF);
    
    printf("词法分析完成，共处理 %d 个Token\n", tokenCount);
}

// 清理词法分析器
void freeLexicalAnalyzer(LexicalAnalyzer* lexer) {
    if (lexer) {
        free(lexer->source);
        free(lexer);
    }
}

// 性能测试函数
void performanceTest() {
    printf("性能测试开始\n");
    printf("===========================================\n");
    
    // 测试用例1：简单程序
    printf("测试用例1：简单程序\n");
    const char* source1 = "int main() { return 0; }";
    LexicalAnalyzer* lexer1 = createLexicalAnalyzer(source1);
    
    clock_t start1 = clock();

    clock_t end1 = clock();
    
    double time1 = ((double)(end1 - start1)) / CLOCKS_PER_SEC;
    printf("执行时间: %.6f 秒\n\n", time1);
    
    freeLexicalAnalyzer(lexer1);
    
    // 测试用例2：复杂程序
    printf("测试用例2：复杂程序\n");
    const char* source2 = "int main() { int x = 10; int y = 20; int z = x + y; return z; }";
    LexicalAnalyzer* lexer2 = createLexicalAnalyzer(source2);
    
    clock_t start2 = clock();
    lexicalAnalysis(lexer2);
    clock_t end2 = clock();
    
    double time2 = ((double)(end2 - start2)) / CLOCKS_PER_SEC;
    printf("执行时间: %.6f 秒\n\n", time2);
    
    freeLexicalAnalyzer(lexer2);
    
    // 测试用例3：大程序
    printf("测试用例3：大程序\n");
    const char* source3 = "int main() { int a = 1; int b = 2; int c = 3; int d = 4; int e = 5; int f = 6; int g = 7; int h = 8; int i = 9; int j = 10; int k = a + b + c + d + e + f + g + h + i + j; return k; }";
    LexicalAnalyzer* lexer3 = createLexicalAnalyzer(source3);
    
    clock_t start3 = clock();
    lexicalAnalysis(lexer3);
    clock_t end3 = clock();
    
    double time3 = ((double)(end3 - start3)) / CLOCKS_PER_SEC;

    LexicalAnalyzer* lexer4 = createLexicalAnalyzer(source4);
    
    clock_t start4 = clock();
    lexicalAnalysis(lexer4);
    clock_t end4 = clock();

    printf("执行时间: %.6f 秒\n\n", time4);

    // 测试用例5：嵌套结构
    printf("测试用例5：嵌套结构\n");

    LexicalAnalyzer* lexer5 = createLexicalAnalyzer(source5);
```

### Valgrind Callgrind分析
```c
// 示例：使用Valgrind Callgrind分析调用图
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// 语法分析器结构
typedef struct {
    char* input;
    size_t position;
    size_t length;
    int line;
    int column;
} SyntaxAnalyzer;

// 创建语法分析器
SyntaxAnalyzer* createSyntaxAnalyzer(const char* input) {
    SyntaxAnalyzer* parser = (SyntaxAnalyzer*)malloc(sizeof(SyntaxAnalyzer));
    parser->input = strdup(input);
    parser->position = 0;
    parser->length = strlen(input);
    parser->line = 1;
    parser->column = 1;
    return parser;
}

// 获取当前字符
char getCurrentChar(SyntaxAnalyzer* parser) {
    if (parser->position >= parser->length) {
        return '\0';
    }
    return parser->input[parser->position];
}

// 前进一个字符
void advance(SyntaxAnalyzer* parser) {
    if (parser->position < parser->length) {
        if (parser->input[parser->position] == '\n') {
            parser->line++;
            parser->column = 1;
        } else {
            parser->column++;
        }
        parser->position++;
    }
}

// 跳过空白字符
void skipWhitespace(SyntaxAnalyzer* parser) {
    while (parser->position < parser->length && 
           (parser->input[parser->position] == ' ' || 
            parser->input[parser->position] == '\t' || 
            parser->input[parser->position] == '\n' || 
            parser->input[parser->position] == '\r')) {
        advance(parser);
    }
}

// 解析数字
int parseNumber(SyntaxAnalyzer* parser) {
    if (parser->position >= parser->length) {
        return 0;
    }
    
    int value = 0;
    while (parser->position < parser->length && 
           parser->input[parser->position] >= '0' && 
           parser->input[parser->position] <= '9') {
        value = value * 10 + (parser->input[parser->position] - '0');
        advance(parser);
    }
    
    return value;
}

// 解析因子
int parseFactor(SyntaxAnalyzer* parser) {
    skipWhitespace(parser);
    
    if (parser->position >= parser->length) {
        return 0;
    }
    
    char current = getCurrentChar(parser);
    
    if (current >= '0' && current <= '9') {
        return parseNumber(parser);
    } else if (current == '(') {
        advance(parser); // 跳过 '('
        int result = parseExpression(parser);
        skipWhitespace(parser);
        if (getCurrentChar(parser) == ')') {
            advance(parser); // 跳过 ')'
        }
        return result;
    }
    
    return 0;
}

// 解析项
int parseTerm(SyntaxAnalyzer* parser) {
    int left = parseFactor(parser);
    
    skipWhitespace(parser);
    
    while (parser->position < parser->length) {
        char op = getCurrentChar(parser);
        if (op == '*' || op == '/') {
            advance(parser);
            skipWhitespace(parser);
            
            int right = parseFactor(parser);
            
            if (op == '*') {
                left *= right;
            } else if (op == '/' && right != 0) {
                left /= right;
            }
        } else {
            break;
        }
        
        skipWhitespace(parser);
    }
    
    return left;
}

// 解析表达式
int parseExpression(SyntaxAnalyzer* parser) {
    int left = parseTerm(parser);
    
    skipWhitespace(parser);
    
    while (parser->position < parser->length) {
        char op = getCurrentChar(parser);
        if (op == '+' || op == '-') {
            advance(parser);
            skipWhitespace(parser);
            
            int right = parseTerm(parser);
            
            if (op == '+') {
                left += right;
            } else {
                left -= right;
            }
        } else {
            break;
        }
        
        skipWhitespace(parser);
    }
    
    return left;
}

// 解析程序
int parseProgram(SyntaxAnalyzer* parser) {
    skipWhitespace(parser);
    
    if (parser->position >= parser->length) {
        return 0;
    }
    
    int result = parseExpression(parser);
    
    skipWhitespace(parser);
    
    return result;
}

// 清理语法分析器
void freeSyntaxAnalyzer(SyntaxAnalyzer* parser) {
    if (parser) {
        free(parser->input);
        free(parser);
    }
}

// 递归测试函数
int recursiveTest(int n) {
    if (n <= 0) {
        return 0;
    }
    
    return n + recursiveTest(n - 1);
}

// 循环测试函数
int loopTest(int n) {
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += i;
    }
    return sum;
}

// 嵌套调用测试
int nestedCallTest(int n) {
    if (n <= 0) {
        return 0;
    }
    
    int result = 0;
    for (int i = 0; i < n; i++) {
        result += recursiveTest(i);
    }
    
    return result;
}

// 复杂计算测试
int complexCalculationTest(int n) {
    int result = 0;
    
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            result += i * j;
        }
    }
    
    return result;
}

// 调用图测试
void callGraphTest() {
    printf("调用图测试开始\n");
    printf("===========================================\n");
    
    // 测试递归调用
    printf("测试递归调用\n");
    int recursiveResult = recursiveTest(100);
    printf("递归结果: %d\n\n", recursiveResult);
    
    // 测试循环调用
    printf("测试循环调用\n");
    int loopResult = loopTest(100);
    printf("循环结果: %d\n\n", loopResult);
    
    // 测试嵌套调用
    printf("测试嵌套调用\n");
    int nestedResult = nestedCallTest(10);
    printf("嵌套结果: %d\n\n", nestedResult);
    
    // 测试复杂计算
    printf("测试复杂计算\n");
    int complexResult = complexCalculationTest(50);
    printf("复杂计算结果: %d\n\n", complexResult);
    
    // 测试语法分析
    printf("测试语法分析\n");
    const char* expressions[] = {
        "1 + 2 * 3",
        "(1 + 2) * 3",
        "1 + 2 + 3 + 4 + 5",
        "10 * 20 + 30 * 40",
        "100 / 4 + 200 / 8"
    };
    
    for (int i = 0; i < 5; i++) {
        SyntaxAnalyzer* parser = createSyntaxAnalyzer(expressions[i]);
        int result = parseProgram(parser);
        printf("表达式: %s = %d\n", expressions[i], result);
        freeSyntaxAnalyzer(parser);
    }
    
    printf("\n调用图测试完成\n");
}

// 主函数
int main() {
    printf("Valgrind Callgrind分析示例\n");
    printf("===========================================\n");
    
    callGraphTest();
    
    return 0;
}
```

```c
// 示例：使用Valgrind Massif分析堆内存
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// AST节点结构
typedef struct ASTNode {
    char* type;
    char* value;
    struct ASTNode* left;
    struct ASTNode* right;
} ASTNode;

// 创建AST节点
ASTNode* createASTNode(const char* type, const char* value) {
    ASTNode* node = (ASTNode*)malloc(sizeof(ASTNode));
    node->type = strdup(type);
    node->value = value ? strdup(value) : NULL;
    node->left = NULL;
    node->right = NULL;
    return node;
}

// 设置左子节点
void setLeftChild(ASTNode* parent, ASTNode* child) {
    if (parent) {
        parent->left = child;
    }
}

// 设置右子节点
void setRightChild(ASTNode* parent, ASTNode* child) {
    if (parent) {
        parent->right = child;
    }
}

// 释放AST节点
void freeASTNode(ASTNode* node) {
    if (node) {
        free(node->type);
        if (node->value) {
            free(node->value);
        }
        free(node);
    }
}

// 释放整个AST
void freeAST(ASTNode* node) {
    if (!node) return;
    
    freeAST(node->left);
    freeAST(node->right);
    freeASTNode(node);
}

// 创建大型AST
ASTNode* createLargeAST(int depth) {
    if (depth <= 0) {
        return createASTNode("leaf", "value");
    }
    
    ASTNode* node = createASTNode("node", NULL);
    ASTNode* left = createLargeAST(depth - 1);
    ASTNode* right = createLargeAST(depth - 1);
    
    setLeftChild(node, left);
    setRightChild(node, right);
    
    return node;
}

// 创建字符串数组
char** createStringArray(int size) {
    char** array = (char**)malloc(size * sizeof(char*));
    
    for (int i = 0; i < size; i++) {
        char* str = (char*)malloc(100 * sizeof(char));
        sprintf(str, "String %d", i);
        array[i] = str;
    }
    
    return array;
}

// 释放字符串数组
void freeStringArray(char** array, int size) {
    if (array) {
        for (int i = 0; i < size; i++) {
            if (array[i]) {
                free(array[i]);
            }
        }
        free(array);
    }
}

// 内存碎片测试
void memoryFragmentationTest() {
    printf("内存碎片测试开始\n");

    // 创建不同大小的内存块
    void* blocks[1000];
```

```c
typedef struct {
    int** data;
    int rows;
    int cols;
} Matrix;

// 创建矩阵
Matrix* createMatrix(int rows, int cols) {
    Matrix* matrix = (Matrix*)malloc(sizeof(Matrix));
    matrix->rows = rows;
    matrix->cols = cols;
    matrix->data = (int**)malloc(rows * sizeof(int*));
    
    for (int i = 0; i < rows; i++) {
        matrix->data[i] = (int*)malloc(cols * sizeof(int));
        for (int j = 0; j < cols; j++) {
            matrix->data[i][j] = i * cols + j;
        }
    }
    
    return matrix;
}

// 释放矩阵
void freeMatrix(Matrix* matrix) {
    if (matrix) {
        for (int i = 0; i < matrix->rows; i++) {
            free(matrix->data[i]);
        }
        free(matrix->data);
        free(matrix);
    }
}

// 矩阵乘法（行优先）
Matrix* matrixMultiplyRowMajor(Matrix* a, Matrix* b) {
    if (a->cols != b->rows) {
        return NULL;
    }
    
    Matrix* result = createMatrix(a->rows, b->cols);
    
    for (int i = 0; i < a->rows; i++) {
        for (int j = 0; j < b->cols; j++) {
            result->data[i][j] = 0;
            for (int k = 0; k < a->cols; k++) {
                result->data[i][j] += a->data[i][k] * b->data[k][j];
            }
        }
    }
    
    return result;
```
