# 调试与测试工具

## 🎯 调试与测试工具概述

**调试与测试工具**是编译器开发过程中不可或缺的工具集，用于发现和修复编译器中的错误，验证编译器的正确性和性能。这些工具包括调试器、测试框架、代码覆盖率工具、内存检测工具等。

## 🔍 工具分类

### 1. 调试器
- **GDB**：GNU调试器
- **LLDB**：LLVM调试器
- **Valgrind**：内存调试工具
- **AddressSanitizer**：地址消毒器

### 2. 测试框架
- **Google Test**：C++测试框架
- **CUnit**：C语言测试框架
- **Catch2**：现代C++测试框架
- **Unity**：轻量级C测试框架

### 3. 代码覆盖率工具
- **GCOV**：GCC代码覆盖率工具
- **LLVM Coverage**：LLVM代码覆盖率工具
- **BullseyeCoverage**：商业代码覆盖率工具
- **OpenCppCoverage**：Windows代码覆盖率工具

### 4. 静态分析工具
- **Clang Static Analyzer**：Clang静态分析器
- **Cppcheck**：C/C++静态分析工具
- **PVS-Studio**：商业静态分析工具
- **SonarQube**：代码质量分析平台

### GDB调试器
```c
// 示例：使用GDB调试编译器
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// 词法分析器结构
typedef struct {
    char* source;
    size_t position;
    size_t length;
    int line;
    int column;
} LexicalAnalyzer;

// Token类型
typedef enum {
    TOKEN_NUMBER,
    TOKEN_IDENTIFIER,
    TOKEN_OPERATOR,
    TOKEN_DELIMITER,
    TOKEN_EOF
} TokenType;

// Token结构
typedef struct {
    TokenType type;
    char* value;
    int line;
    int column;
} Token;

// 创建词法分析器
LexicalAnalyzer* createLexicalAnalyzer(const char* source) {
    LexicalAnalyzer* lexer = (LexicalAnalyzer*)malloc(sizeof(LexicalAnalyzer));
    lexer->source = strdup(source);
    lexer->position = 0;
    lexer->length = strlen(source);
    lexer->line = 1;
    lexer->column = 1;
    return lexer;
}

// 获取当前字符
char getCurrentChar(LexicalAnalyzer* lexer) {
    if (lexer->position >= lexer->length) {
        return '\0';
    }
    return lexer->source[lexer->position];
}

// 前进一个字符
void advance(LexicalAnalyzer* lexer) {
    if (lexer->position < lexer->length) {
        if (lexer->source[lexer->position] == '\n') {
            lexer->line++;
            lexer->column = 1;
        } else {
            lexer->column++;
        }
        lexer->position++;
    }
}

// 跳过空白字符
void skipWhitespace(LexicalAnalyzer* lexer) {
    while (lexer->position < lexer->length && 
           (lexer->source[lexer->position] == ' ' || 
            lexer->source[lexer->position] == '\t' || 
            lexer->source[lexer->position] == '\n' || 
            lexer->source[lexer->position] == '\r')) {
        advance(lexer);
    }
}

// 扫描数字
Token scanNumber(LexicalAnalyzer* lexer) {
    Token token;
    token.type = TOKEN_NUMBER;
    token.line = lexer->line;
    token.column = lexer->column;
    
    size_t start = lexer->position;
    while (lexer->position < lexer->length && 
           lexer->source[lexer->position] >= '0' && 
           lexer->source[lexer->position] <= '9') {
        advance(lexer);
    }
    
    size_t length = lexer->position - start;
    token.value = (char*)malloc(length + 1);
    strncpy(token.value, lexer->source + start, length);
    token.value[length] = '\0';
    
    return token;
}

// 扫描标识符
Token scanIdentifier(LexicalAnalyzer* lexer) {
    Token token;
    token.type = TOKEN_IDENTIFIER;
    token.line = lexer->line;
    token.column = lexer->column;
    
    size_t start = lexer->position;
    while (lexer->position < lexer->length && 
           ((lexer->source[lexer->position] >= 'a' && lexer->source[lexer->position] <= 'z') ||
            (lexer->source[lexer->position] >= 'A' && lexer->source[lexer->position] <= 'Z') ||
            (lexer->source[lexer->position] >= '0' && lexer->source[lexer->position] <= '9'))) {
        advance(lexer);
    }
    
    size_t length = lexer->position - start;
    token.value = (char*)malloc(length + 1);
    strncpy(token.value, lexer->source + start, length);
    token.value[length] = '\0';
    
    return token;
}

// 扫描操作符
Token scanOperator(LexicalAnalyzer* lexer) {
    Token token;
    token.type = TOKEN_OPERATOR;
    token.line = lexer->line;
    token.column = lexer->column;
    
    char op = getCurrentChar(lexer);
    token.value = (char*)malloc(2);
    token.value[0] = op;
    token.value[1] = '\0';
    
    advance(lexer);
    return token;
}

// 扫描分隔符
Token scanDelimiter(LexicalAnalyzer* lexer) {
    Token token;
    token.type = TOKEN_DELIMITER;
    token.line = lexer->line;
    token.column = lexer->column;
    
    char delim = getCurrentChar(lexer);
    token.value = (char*)malloc(2);
    token.value[0] = delim;
    token.value[1] = '\0';
    
    advance(lexer);
    return token;
}

// 获取下一个Token
Token getNextToken(LexicalAnalyzer* lexer) {
    skipWhitespace(lexer);
    
    if (lexer->position >= lexer->length) {
        Token token;
        token.type = TOKEN_EOF;
        token.value = NULL;
        token.line = lexer->line;
        token.column = lexer->column;
        return token;
    }
    
    char current = getCurrentChar(lexer);
    
    if (current >= '0' && current <= '9') {
        return scanNumber(lexer);
    } else if ((current >= 'a' && current <= 'z') || 
               (current >= 'A' && current <= 'Z')) {
        return scanIdentifier(lexer);
    } else if (current == '+' || current == '-' || 
               current == '*' || current == '/' || 
               current == '=') {
        return scanOperator(lexer);
    } else if (current == '(' || current == ')' || 
               current == '{' || current == '}' || 
               current == ';' || current == ',') {
        return scanDelimiter(lexer);
    } else {
        // 未知字符
        Token token;
        token.type = TOKEN_EOF;
        token.value = NULL;
        token.line = lexer->line;
        token.column = lexer->column;
        return token;
    }
}

// 打印Token
void printToken(Token token) {
    const char* typeNames[] = {
        "NUMBER", "IDENTIFIER", "OPERATOR", "DELIMITER", "EOF"
    };
    
    printf("Token: %s", typeNames[token.type]);
    if (token.value) {
        printf(", Value: %s", token.value);
    }
    printf(", Line: %d, Column: %d\n", token.line, token.column);
}

// 清理词法分析器
void freeLexicalAnalyzer(LexicalAnalyzer* lexer) {
    if (lexer) {
        free(lexer->source);
        free(lexer);
    }
}

// 主函数
int main() {
    printf("GDB调试器示例\n");
    printf("===========================================\n");
    
    const char* source = "int main() { return 0; }";
    LexicalAnalyzer* lexer = createLexicalAnalyzer(source);
    
    printf("源代码: %s\n", source);
    printf("词法分析结果:\n");
    
    Token token;
    do {
        token = getNextToken(lexer);
        printToken(token);
        
        if (token.value) {
            free(token.value);
        }
    } while (token.type != TOKEN_EOF);
    
    freeLexicalAnalyzer(lexer);
    return 0;
}
```

### Valgrind内存调试
```c
// 示例：使用Valgrind检测内存错误
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// 语法树节点
typedef struct ASTNode {
    char* type;
    char* value;
    struct ASTNode* left;
    struct ASTNode* right;
} ASTNode;

// 创建AST节点
ASTNode* createASTNode(const char* type, const char* value) {
    ASTNode* node = (ASTNode*)malloc(sizeof(ASTNode));
    if (!node) {
        printf("内存分配失败\n");
        return NULL;
    }
    
    node->type = strdup(type);
    node->value = value ? strdup(value) : NULL;
    node->left = NULL;
    node->right = NULL;
    
    return node;
}

// 设置左子节点
void setLeftChild(ASTNode* parent, ASTNode* child) {
    if (parent) {
        parent->left = child;
    }
}

// 设置右子节点
void setRightChild(ASTNode* parent, ASTNode* child) {
    if (parent) {
        parent->right = child;
    }
}

// 打印AST
void printAST(ASTNode* node, int depth) {
    if (!node) return;
    
    for (int i = 0; i < depth; i++) {
        printf("  ");
    }
    
    printf("%s", node->type);
    if (node->value) {
        printf(": %s", node->value);
    }
    printf("\n");
    
    printAST(node->left, depth + 1);
    printAST(node->right, depth + 1);
}

// 释放AST
void freeAST(ASTNode* node) {
    if (!node) return;
    
    free(node->type);
    if (node->value) {
        free(node->value);
    }
    
    freeAST(node->left);
    freeAST(node->right);
    free(node);
}

// 模拟内存泄漏
void simulateMemoryLeak() {
    printf("模拟内存泄漏\n");
    
    // 故意不释放内存
    ASTNode* node1 = createASTNode("number", "42");
    ASTNode* node2 = createASTNode("identifier", "x");
    ASTNode* node3 = createASTNode("add", "+");
    
    setLeftChild(node3, node1);
    setRightChild(node3, node2);
    
    printAST(node3, 0);
    
    // 故意不调用freeAST(node3)
    printf("内存泄漏：未释放AST节点\n");
}

// 模拟野指针
void simulateWildPointer() {
    printf("模拟野指针\n");
    
    ASTNode* node = createASTNode("number", "100");
    ASTNode** ptr = &node;
    
    freeAST(node);
    node = NULL;
    
    // 故意访问已释放的内存
    printf("野指针访问：%s\n", (*ptr)->type); // 这会导致段错误
}

// 模拟缓冲区溢出
void simulateBufferOverflow() {
    printf("模拟缓冲区溢出\n");
    
    char buffer[10];
    char* input = "This is a very long string that will overflow the buffer";
    
    // 故意不检查边界
    strcpy(buffer, input); // 缓冲区溢出
    
    printf("缓冲区内容: %s\n", buffer);
}

// 主函数
int main() {
    printf("Valgrind内存调试示例\n");
    printf("===========================================\n");
    
    // 正常使用
    printf("正常使用:\n");
    ASTNode* root = createASTNode("program", NULL);
    ASTNode* stmt = createASTNode("statement", NULL);
    ASTNode* expr = createASTNode("expression", NULL);
    ASTNode* num = createASTNode("number", "42");
    
    setLeftChild(root, stmt);
    setLeftChild(stmt, expr);
    setLeftChild(expr, num);
    
    printAST(root, 0);
    freeAST(root);
    
    printf("\n");
    
    // 模拟各种内存错误
    printf("模拟内存错误:\n");
    
    // 1. 内存泄漏
    simulateMemoryLeak();
    
    printf("\n");
    
    // 2. 野指针（注释掉以避免程序崩溃）
    // simulateWildPointer();
    
    printf("\n");
    
    // 3. 缓冲区溢出（注释掉以避免程序崩溃）
    // simulateBufferOverflow();
    
    printf("程序结束\n");
    return 0;
}
```

```cpp
// 示例：使用Google Test测试编译器
#include <gtest/gtest.h>
#include <string>
#include <vector>
#include <memory>

// 词法分析器类
class LexicalAnalyzer {
private:
    std::string source;
    size_t position;
    size_t length;
    int line;
    int column;
    
public:
    LexicalAnalyzer(const std::string& source) 
        : source(source), position(0), length(source.length()), line(1), column(1) {}
    
    // Token类型
    enum TokenType {
        TOKEN_NUMBER,
        TOKEN_IDENTIFIER,
        TOKEN_OPERATOR,
        TOKEN_DELIMITER,
        TOKEN_EOF
    };
    
    // Token结构
    struct Token {
        TokenType type;
        std::string value;
        int line;
        int column;
        
        Token(TokenType t, const std::string& v, int l, int c) 
            : type(t), value(v), line(l), column(c) {}
    };
    
    // 获取当前字符
    char getCurrentChar() const {
        if (position >= length) return '\0';
        return source[position];
    }
    
    // 前进一个字符
    void advance() {
        if (position < length) {
            if (source[position] == '\n') {
                line++;
                column = 1;
            } else {
                column++;
            }
            position++;
        }
    }
    
    // 跳过空白字符
    void skipWhitespace() {
        while (position < length && 
               (source[position] == ' ' || 
                source[position] == '\t' || 
                source[position] == '\n' || 
                source[position] == '\r')) {
            advance();
        }
    }
    
    // 扫描数字
    Token scanNumber() {
        int startLine = line;
        int startColumn = column;
        
        std::string value;
        while (position < length && 
               source[position] >= '0' && 
               source[position] <= '9') {
            value += source[position];
            advance();
        }
        
        return Token(TOKEN_NUMBER, value, startLine, startColumn);
    }
    
    // 扫描标识符
    Token scanIdentifier() {
        int startLine = line;
        int startColumn = column;
        
        std::string value;
        while (position < length && 
               ((source[position] >= 'a' && source[position] <= 'z') ||
                (source[position] >= 'A' && source[position] <= 'Z') ||
                (source[position] >= '0' && source[position] <= '9'))) {
            value += source[position];
            advance();
        }
        
        return Token(TOKEN_IDENTIFIER, value, startLine, startColumn);
    }
    
    // 扫描操作符
    Token scanOperator() {
        int startLine = line;
        int startColumn = column;
        
        char op = getCurrentChar();
        advance();
        
        return Token(TOKEN_OPERATOR, std::string(1, op), startLine, startColumn);
    }
    
    // 扫描分隔符
    Token scanDelimiter() {
        int startLine = line;
        int startColumn = column;
        
        char delim = getCurrentChar();
        advance();
        
        return Token(TOKEN_DELIMITER, std::string(1, delim), startLine, startColumn);
    }
    
    // 获取下一个Token
    Token getNextToken() {
        skipWhitespace();
        
        if (position >= length) {
            return Token(TOKEN_EOF, "", line, column);
        }
        
        char current = getCurrentChar();
        
        if (current >= '0' && current <= '9') {
            return scanNumber();
        } else if ((current >= 'a' && current <= 'z') || 
                   (current >= 'A' && current <= 'Z')) {
            return scanIdentifier();
        } else if (current == '+' || current == '-' || 
                   current == '*' || current == '/' || 
                   current == '=') {
            return scanOperator();
        } else if (current == '(' || current == ')' || 
                   current == '{' || current == '}' || 
                   current == ';' || current == ',') {
            return scanDelimiter();
        } else {
            return Token(TOKEN_EOF, "", line, column);
        }
    }
    
    // 获取所有Token
    std::vector<Token> getAllTokens() {
        std::vector<Token> tokens;
        Token token = getNextToken();
        
        while (token.type != TOKEN_EOF) {
            tokens.push_back(token);
            token = getNextToken();
        }
        
        return tokens;
    }
};
```

### GCOV代码覆盖率
```c
// 示例：使用GCOV测量代码覆盖率
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// 语法分析器结构
typedef struct {
    char* input;
    size_t position;
    size_t length;
    int line;
    int column;
} SyntaxAnalyzer;

// 创建语法分析器
SyntaxAnalyzer* createSyntaxAnalyzer(const char* input) {
    SyntaxAnalyzer* parser = (SyntaxAnalyzer*)malloc(sizeof(SyntaxAnalyzer));
    if (!parser) {
        printf("内存分配失败\n");
        return NULL;
    }
    
    parser->input = strdup(input);
    parser->position = 0;
    parser->length = strlen(input);
    parser->line = 1;
    parser->column = 1;
    
    return parser;
}

// 获取当前字符
char getCurrentChar(SyntaxAnalyzer* parser) {
    if (parser->position >= parser->length) {
        return '\0';
    }
    return parser->input[parser->position];
}

// 前进一个字符
void advance(SyntaxAnalyzer* parser) {
    if (parser->position < parser->length) {
        if (parser->input[parser->position] == '\n') {
            parser->line++;
            parser->column = 1;
        } else {
            parser->column++;
        }
        parser->position++;
    }
}

// 跳过空白字符
void skipWhitespace(SyntaxAnalyzer* parser) {
    while (parser->position < parser->length && 
           (parser->input[parser->position] == ' ' || 
            parser->input[parser->position] == '\t' || 
            parser->input[parser->position] == '\n' || 
            parser->input[parser->position] == '\r')) {
        advance(parser);
    }
}

// 解析数字
int parseNumber(SyntaxAnalyzer* parser) {
    printf("解析数字\n");
    
    if (parser->position >= parser->length) {
        printf("错误：输入结束\n");
        return 0;
    }
    
    int value = 0;
    while (parser->position < parser->length && 
           parser->input[parser->position] >= '0' && 
           parser->input[parser->position] <= '9') {
        value = value * 10 + (parser->input[parser->position] - '0');
        advance(parser);
    }
    
    printf("解析到数字: %d\n", value);
    return value;
}

// 解析表达式
int parseExpression(SyntaxAnalyzer* parser) {
    printf("解析表达式\n");
    
    skipWhitespace(parser);
    
    if (parser->position >= parser->length) {
        printf("错误：表达式为空\n");
        return 0;
    }
    
    int left = parseNumber(parser);
    
    skipWhitespace(parser);
    
    if (parser->position >= parser->length) {
        printf("表达式只有数字: %d\n", left);
        return left;
    }
    
    char op = getCurrentChar(parser);
    if (op == '+' || op == '-' || op == '*' || op == '/') {
        advance(parser);
        skipWhitespace(parser);
        
        int right = parseNumber(parser);
        
        switch (op) {
            case '+':
                printf("加法: %d + %d = %d\n", left, right, left + right);
                return left + right;
            case '-':
                printf("减法: %d - %d = %d\n", left, right, left - right);
                return left - right;
            case '*':
                printf("乘法: %d * %d = %d\n", left, right, left * right);
                return left * right;
            case '/':
                if (right == 0) {
                    printf("错误：除零\n");
                    return 0;
                }
                printf("除法: %d / %d = %d\n", left, right, left / right);
                return left / right;
            default:
                printf("错误：未知操作符\n");
                return left;
        }
    }
    
    printf("表达式只有数字: %d\n", left);
    return left;
}

// 解析程序
int parseProgram(SyntaxAnalyzer* parser) {
    printf("解析程序\n");
    
    skipWhitespace(parser);
    
    if (parser->position >= parser->length) {
        printf("错误：程序为空\n");
        return 0;
    }
    
    int result = parseExpression(parser);
    
    skipWhitespace(parser);
    
    if (parser->position < parser->length) {
        printf("警告：程序末尾有多余字符\n");
    }
    
    printf("程序解析完成，结果: %d\n", result);
    return result;
}

// 清理语法分析器
void freeSyntaxAnalyzer(SyntaxAnalyzer* parser) {
    if (parser) {
        free(parser->input);
        free(parser);
    }
}
```
