// 词法分析器示例
class LexicalAnalyzer {
private:
    std::string source;
    size_t position;
    
public:
    LexicalAnalyzer(const std::string& source) 
        : source(source), position(0) {}
    
    Token getNextToken() {
        skipWhitespace();
        
        if (position >= source.length()) {
            return Token(TOKEN_EOF, "", 0, 0);
        }
        
        char current = source[position];
        
        if (current >= '0' && current <= '9') {
            return scanNumber();
        } else if (current == '+' || current == '-' || 
                   current == '*' || current == '/') {
            return scanOperator();
        } else if (current == '(' || current == ')') {
            return scanDelimiter();
        }
        
        return Token(TOKEN_ERROR, "", 0, 0);
    }
    
private:
    Token scanNumber() {
        std::string value;
        while (position < source.length() && 
               source[position] >= '0' && 
               source[position] <= '9') {
            value += source[position++];
        }
        return Token(TOKEN_NUMBER, value, 0, 0);
    }
    
    Token scanOperator() {
        char op = source[position++];
        return Token(TOKEN_OPERATOR, std::string(1, op), 0, 0);
    }
    
    Token scanDelimiter() {
        char delim = source[position++];
        return Token(TOKEN_DELIMITER, std::string(1, delim), 0, 0);
    }
    
    void skipWhitespace() {
        while (position < source.length() && 
               (source[position] == ' ' || 
                source[position] == '\t' || 
                source[position] == '\n')) {
            position++;
        }
    }
};
```

// 语法分析器示例
class SyntaxAnalyzer {
private:
    std::vector<Token> tokens;
    size_t position;
    
public:
    SyntaxAnalyzer(const std::vector<Token>& tokens) 
        : tokens(tokens), position(0) {}
    
    ASTNode* parseProgram() {
        ASTNode* program = new ASTNode("program");
        
        while (position < tokens.size()) {
            if (isTypeSpecifier()) {
                ASTNode* declaration = parseDeclaration();
                program->addChild(declaration);
            } else if (isFunctionDefinition()) {
                ASTNode* function = parseFunctionDefinition();
                program->addChild(function);
            } else {
                error("Unexpected token");
            }
        }
        
        return program;
    }
    
private:
    ASTNode* parseDeclaration() {
        ASTNode* type = parseTypeSpecifier();
        ASTNode* declarator = parseDeclarator();
        
        if (currentToken().type == TOKEN_SEMICOLON) {
            advance();
            return new ASTNode("declaration", {type, declarator});
        } else if (currentToken().type == TOKEN_ASSIGN) {
            advance();
            ASTNode* initializer = parseInitializer();
            advance(); // consume semicolon
            return new ASTNode("declaration_with_init", 
                              {type, declarator, initializer});
        }
        
        error("Expected semicolon or assignment");
        return nullptr;
    }
    
    ASTNode* parseFunctionDefinition() {
        ASTNode* type = parseTypeSpecifier();
        ASTNode* declarator = parseDeclarator();
        ASTNode* body = parseCompoundStatement();
        
        return new ASTNode("function_definition", {type, declarator, body});
    }
    
    ASTNode* parseCompoundStatement() {
        expect(TOKEN_LEFT_BRACE);
        
        ASTNode* compound = new ASTNode("compound_statement");
        
        while (currentToken().type != TOKEN_RIGHT_BRACE) {
            if (isTypeSpecifier()) {
                ASTNode* declaration = parseDeclaration();
                compound->addChild(declaration);
            } else {
                ASTNode* statement = parseStatement();
                compound->addChild(statement);
            }
        }
        
        expect(TOKEN_RIGHT_BRACE);
        return compound;
    }
};
```

// 虚拟机实现示例
class VirtualMachine {
private:
    std::vector<uint8_t> memory;
    std::vector<int> stack;
    std::map<std::string, int> registers;
    size_t pc; // program counter
    size_t sp; // stack pointer
    
public:
    VirtualMachine(size_t memorySize = 1024 * 1024) 
        : memory(memorySize), pc(0), sp(0) {
        registers["eax"] = 0;
        registers["ebx"] = 0;
        registers["ecx"] = 0;
        registers["edx"] = 0;
    }
    
    void execute(const std::vector<Instruction>& instructions) {
        pc = 0;
        
        while (pc < instructions.size()) {
            const Instruction& inst = instructions[pc];
            
            switch (inst.opcode) {
                case OP_LOAD:
                    executeLoad(inst);
                    break;
                case OP_STORE:
                    executeStore(inst);
                    break;
                case OP_ADD:
                    executeAdd(inst);
                    break;
                case OP_SUB:
                    executeSub(inst);
                    break;
                case OP_MUL:
                    executeMul(inst);
                    break;
                case OP_DIV:
                    executeDiv(inst);
                    break;
                case OP_JUMP:
                    executeJump(inst);
                    break;
                case OP_JUMP_IF:
                    executeJumpIf(inst);
                    break;
                case OP_CALL:
                    executeCall(inst);
                    break;
                case OP_RETURN:
                    executeReturn(inst);
                    break;
                default:
                    error("Unknown instruction");
            }
            
            pc++;
        }
    }
    
private:
    void executeLoad(const Instruction& inst) {
        int value = getValue(inst.operands[1]);
        setRegister(inst.operands[0], value);
    }
    
    void executeStore(const Instruction& inst) {
        int value = getRegister(inst.operands[0]);
        setValue(inst.operands[1], value);
    }
    
    void executeAdd(const Instruction& inst) {
        int left = getRegister(inst.operands[1]);
        int right = getRegister(inst.operands[2]);
        setRegister(inst.operands[0], left + right);
    }
    
    void executeJump(const Instruction& inst) {
        pc = inst.operands[0] - 1; // -1 because pc++ at end of loop
    }
    
    void executeJumpIf(const Instruction& inst) {
        int condition = getRegister(inst.operands[0]);
        if (condition != 0) {
            pc = inst.operands[1] - 1; // -1 because pc++ at end of loop
        }
    }
    
    int getRegister(const std::string& reg) {
        return registers[reg];
    }
    
    void setRegister(const std::string& reg, int value) {
        registers[reg] = value;
    }
    
    int getValue(int address) {
        return *reinterpret_cast<int*>(&memory[address]);
    }
    
    void setValue(int address, int value) {
        *reinterpret_cast<int*>(&memory[address]) = value;
    }
};
```

// LLVM IR生成示例
class LLVMCodeGenerator {
private:
    llvm::LLVMContext& context;
    llvm::Module* module;
    llvm::IRBuilder<>* builder;
    
public:
    LLVMCodeGenerator(llvm::LLVMContext& context) 
        : context(context), builder(nullptr) {
        module = new llvm::Module("SimpleCalc", context);
        builder = new llvm::IRBuilder<>(context);
    }
    
    llvm::Function* createMainFunction() {
        llvm::FunctionType* funcType = llvm::FunctionType::get(
            llvm::Type::getInt32Ty(context), false);
        
        llvm::Function* func = llvm::Function::Create(
            funcType, llvm::Function::ExternalLinkage, "main", module);
        
        llvm::BasicBlock* block = llvm::BasicBlock::Create(context, "entry", func);
        builder->SetInsertPoint(block);
        
        return func;
    }
    
    llvm::Value* generateExpression(ASTNode* node) {
        if (node->type == "number") {
            return llvm::ConstantInt::get(
                llvm::Type::getInt32Ty(context), 
                std::stoi(node->value));
        } else if (node->type == "add") {
            llvm::Value* left = generateExpression(node->children[0]);
            llvm::Value* right = generateExpression(node->children[1]);
            return builder->CreateAdd(left, right, "addtmp");
        } else if (node->type == "sub") {
            llvm::Value* left = generateExpression(node->children[0]);
            llvm::Value* right = generateExpression(node->children[1]);
            return builder->CreateSub(left, right, "subtmp");
        } else if (node->type == "mul") {
            llvm::Value* left = generateExpression(node->children[0]);
            llvm::Value* right = generateExpression(node->children[1]);
            return builder->CreateMul(left, right, "multmp");
        } else if (node->type == "div") {
            llvm::Value* left = generateExpression(node->children[0]);
            llvm::Value* right = generateExpression(node->children[1]);
            return builder->CreateSDiv(left, right, "divtmp");
        }
        
        return nullptr;
    }
    
    void generateReturn(llvm::Value* value) {
        builder->CreateRet(value);
    }
    
    void printIR() {
        module->print(llvm::outs(), nullptr);
    }
};
```
