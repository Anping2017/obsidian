# 作用域分析

## 🎯 作用域分析概述

**作用域分析**是语义分析的重要组成部分，负责分析程序中标识符的作用域，确定标识符的可见性和生命周期。作用域分析为符号表管理和类型检查提供基础。

## 🔍 基本概念

### 作用域定义
作用域是程序中标识符可见的区域，在这个区域内可以访问该标识符。

### 作用域类型
```c
enum ScopeType {
    GLOBAL,      // 全局作用域
    FUNCTION,    // 函数作用域
    BLOCK,       // 块作用域
    CLASS,       // 类作用域
    NAMESPACE    // 命名空间作用域
};

struct Scope {
    int id;              // 作用域ID
    ScopeType type;      // 作用域类型
    Scope* parent;       // 父作用域
    vector<Scope*> children; // 子作用域
    map<string, SymbolEntry*> symbols; // 符号表
    
    Scope(int i, ScopeType t, Scope* p = nullptr) 
        : id(i), type(t), parent(p) {}
};
```

## 📊 作用域管理

### 作用域栈
```c
class ScopeStack {
private:
    stack<Scope*> scopeStack;
    int nextScopeId;
    
public:
    ScopeStack() : nextScopeId(0) {
        // 创建全局作用域
        Scope* globalScope = new Scope(nextScopeId++, GLOBAL);
        scopeStack.push(globalScope);
    }
    
    void enterScope(ScopeType type) {
        Scope* currentScope = scopeStack.top();
        Scope* newScope = new Scope(nextScopeId++, type, currentScope);
        currentScope->children.push_back(newScope);
        scopeStack.push(newScope);
    }
    
    void exitScope() {
        if (scopeStack.size() > 1) {
            scopeStack.pop();
        }
    }
    
    Scope* getCurrentScope() {
        return scopeStack.top();
    }
    
    Scope* getGlobalScope() {
        return scopeStack.empty() ? nullptr : scopeStack.top();
    }
    
    void insertSymbol(string name, SymbolEntry* entry) {
        Scope* currentScope = getCurrentScope();
        currentScope->symbols[name] = entry;
    }
    
    SymbolEntry* lookupSymbol(string name) {
        stack<Scope*> tempStack = scopeStack;
        
        while (!tempStack.empty()) {
            Scope* scope = tempStack.top();
            auto it = scope->symbols.find(name);
            if (it != scope->symbols.end()) {
                return it->second;
            }
            tempStack.pop();
        }
        
        return nullptr;
    }
    
    SymbolEntry* lookupSymbolInCurrentScope(string name) {
        Scope* currentScope = getCurrentScope();
        auto it = currentScope->symbols.find(name);
        if (it != currentScope->symbols.end()) {
            return it->second;
        }
        return nullptr;
    }
};
```

### 作用域树
```c
class ScopeTree {
private:
    Scope* root;
    Scope* current;
    int nextScopeId;
    
public:
    ScopeTree() : nextScopeId(0) {
        root = new Scope(nextScopeId++, GLOBAL);
        current = root;
    }
    
    void enterScope(ScopeType type) {
        Scope* newScope = new Scope(nextScopeId++, type, current);
        current->children.push_back(newScope);
        current = newScope;
    }
    
    void exitScope() {
        if (current != root) {
            current = current->parent;
        }
    }
    
    Scope* getCurrentScope() {
        return current;
    }
    
    Scope* getRootScope() {
        return root;
    }
    
    void insertSymbol(string name, SymbolEntry* entry) {
        current->symbols[name] = entry;
    }
    
    SymbolEntry* lookupSymbol(string name) {
        return lookupSymbolInScope(current, name);
    }
    
    SymbolEntry* lookupSymbolInScope(Scope* scope, string name) {
        // 在当前作用域查找
        auto it = scope->symbols.find(name);
        if (it != scope->symbols.end()) {
            return it->second;
        }
        
        // 在父作用域查找
        if (scope->parent != nullptr) {
            return lookupSymbolInScope(scope->parent, name);
        }
        
        return nullptr;
    }
    
    SymbolEntry* lookupSymbolInCurrentScope(string name) {
        auto it = current->symbols.find(name);
        if (it != current->symbols.end()) {
            return it->second;
        }
        return nullptr;
    }
    
    vector<Scope*> getAllScopes() {
        vector<Scope*> scopes;
        collectScopes(root, scopes);
        return scopes;
    }
    
    void collectScopes(Scope* scope, vector<Scope*>& scopes) {
        scopes.push_back(scope);
        for (auto child : scope->children) {
            collectScopes(child, scopes);
        }
    }
};
```

## 🔧 作用域分析器

### 基本作用域分析器
```c
class ScopeAnalyzer {
private:
    ScopeTree scopeTree;
    ASTNode* ast;
    
public:
    ScopeAnalyzer(ASTNode* ast) : ast(ast) {}
    
    void analyzeScopes() {
        traverseAST(ast);
    }
    
    void traverseAST(ASTNode* node) {
        switch (node->getType()) {
            case FUNCTION_DEFINITION:
                handleFunctionDefinition(node);
                break;
            case BLOCK:
                handleBlock(node);
                break;
            case VARIABLE_DECLARATION:
                handleVariableDeclaration(node);
                break;
            case IDENTIFIER:
                handleIdentifier(node);
                break;
        }
        
        // 递归遍历子节点
        for (auto child : node->getChildren()) {
            traverseAST(child);
        }
    }
    
    void handleFunctionDefinition(ASTNode* node) {
        string funcName = node->getChild(1)->getValue();
        
        // 在全局作用域添加函数
        scopeTree.insertSymbol(funcName, new SymbolEntry(funcName, FUNCTION, VOID, 0));
        
        // 进入函数作用域
        scopeTree.enterScope(FUNCTION);
        
        // 处理参数
        ASTNode* parameters = node->getChild(2);
        handleParameters(parameters);
        
        // 处理函数体
        ASTNode* body = node->getChild(3);
        traverseAST(body);
        
        // 退出函数作用域
        scopeTree.exitScope();
    }
    
    void handleBlock(ASTNode* node) {
        // 进入块作用域
        scopeTree.enterScope(BLOCK);
        
        // 处理块内声明
        for (auto child : node->getChildren()) {
            traverseAST(child);
        }
        
        // 退出块作用域
        scopeTree.exitScope();
    }
    
    void handleVariableDeclaration(ASTNode* node) {
        string typeName = node->getChild(0)->getValue();
        string varName = node->getChild(1)->getValue();
        
        // 检查变量是否已声明
        if (scopeTree.lookupSymbolInCurrentScope(varName) != nullptr) {
            reportError("Variable already declared: " + varName);
            return;
        }
        
        // 添加变量到当前作用域
        DataType type = parseDataType(typeName);
        SymbolEntry* entry = new SymbolEntry(varName, VARIABLE, type, scopeTree.getCurrentScope()->id);
        scopeTree.insertSymbol(varName, entry);
    }
    
    void handleIdentifier(ASTNode* node) {
        string name = node->getValue();
        
        // 查找标识符
        SymbolEntry* entry = scopeTree.lookupSymbol(name);
        if (entry == nullptr) {
            reportError("Undeclared identifier: " + name);
            return;
        }
        
        // 设置标识符的属性
        node->setAttribute("symbol", entry);
        node->setAttribute("scope", scopeTree.getCurrentScope()->id);
    }
    
    void handleParameters(ASTNode* node) {
        for (auto param : node->getChildren()) {
            string typeName = param->getChild(0)->getValue();
            string paramName = param->getChild(1)->getValue();
            
            DataType type = parseDataType(typeName);
            SymbolEntry* entry = new SymbolEntry(paramName, VARIABLE, type, scopeTree.getCurrentScope()->id);
            scopeTree.insertSymbol(paramName, entry);
        }
    }
};
```

### 高级作用域分析器
```c
class AdvancedScopeAnalyzer {
private:
    ScopeAnalyzer baseAnalyzer;
    map<string, Scope*> namespaceScopes;
    
public:
    void analyzeScopes(ASTNode* node) {
        switch (node->getType()) {
            case NAMESPACE_DEFINITION:
                handleNamespaceDefinition(node);
                break;
            case CLASS_DEFINITION:
                handleClassDefinition(node);
                break;
            case USING_DECLARATION:
                handleUsingDeclaration(node);
                break;
            case QUALIFIED_NAME:
                handleQualifiedName(node);
                break;
        }
        
        baseAnalyzer.traverseAST(node);
    }
    
    void handleNamespaceDefinition(ASTNode* node) {
        string namespaceName = node->getChild(0)->getValue();
        
        // 创建命名空间作用域
        Scope* namespaceScope = new Scope(getNextScopeId(), NAMESPACE, scopeTree.getCurrentScope());
        namespaceScopes[namespaceName] = namespaceScope;
        
        // 进入命名空间作用域
        scopeTree.enterScope(NAMESPACE);
        
        // 处理命名空间内容
        ASTNode* body = node->getChild(1);
        traverseAST(body);
        
        // 退出命名空间作用域
        scopeTree.exitScope();
    }
    
    void handleClassDefinition(ASTNode* node) {
        string className = node->getChild(0)->getValue();
        
        // 在全局作用域添加类
        SymbolEntry* classEntry = new SymbolEntry(className, TYPE, STRUCT, 0);
        scopeTree.insertSymbol(className, classEntry);
        
        // 进入类作用域
        scopeTree.enterScope(CLASS);
        
        // 处理类成员
        ASTNode* members = node->getChild(1);
        traverseAST(members);
        
        // 退出类作用域
        scopeTree.exitScope();
    }
    
    void handleUsingDeclaration(ASTNode* node) {
        string namespaceName = node->getChild(0)->getValue();
        
        // 查找命名空间
        auto it = namespaceScopes.find(namespaceName);
        if (it != namespaceScopes.end()) {
            // 将命名空间的符号导入当前作用域
            importSymbols(it->second, scopeTree.getCurrentScope());
        } else {
            reportError("Unknown namespace: " + namespaceName);
        }
    }
    
    void handleQualifiedName(ASTNode* node) {
        string namespaceName = node->getChild(0)->getValue();
        string symbolName = node->getChild(1)->getValue();
        
        // 查找命名空间
        auto it = namespaceScopes.find(namespaceName);
        if (it != namespaceScopes.end()) {
            // 在命名空间中查找符号
            SymbolEntry* entry = lookupSymbolInScope(it->second, symbolName);
            if (entry != nullptr) {
                node->setAttribute("symbol", entry);
                node->setAttribute("scope", it->second->id);
            } else {
                reportError("Symbol not found in namespace: " + namespaceName + "::" + symbolName);
            }
        } else {
            reportError("Unknown namespace: " + namespaceName);
        }
    }
    
    void importSymbols(Scope* sourceScope, Scope* targetScope) {
        for (auto& pair : sourceScope->symbols) {
            string name = pair.first;
            SymbolEntry* entry = pair.second;
            
            // 检查是否已存在
            if (targetScope->symbols.find(name) == targetScope->symbols.end()) {
                targetScope->symbols[name] = entry;
            }
        }
    }
};
```

## 📈 作用域分析算法

### 作用域查找算法
```c
class ScopeLookup {
private:
    ScopeTree scopeTree;
    
public:
    SymbolEntry* lookupSymbol(string name, Scope* startScope = nullptr) {
        if (startScope == nullptr) {
            startScope = scopeTree.getCurrentScope();
        }
        
        return lookupSymbolInScope(name, startScope);
    }
    
    SymbolEntry* lookupSymbolInScope(string name, Scope* scope) {
        // 在当前作用域查找
        auto it = scope->symbols.find(name);
        if (it != scope->symbols.end()) {
            return it->second;
        }
        
        // 在父作用域查找
        if (scope->parent != nullptr) {
            return lookupSymbolInScope(name, scope->parent);
        }
        
        return nullptr;
    }
    
    vector<SymbolEntry*> lookupAllSymbols(string name) {
        vector<SymbolEntry*> symbols;
        Scope* currentScope = scopeTree.getCurrentScope();
        
        while (currentScope != nullptr) {
            auto it = currentScope->symbols.find(name);
            if (it != currentScope->symbols.end()) {
                symbols.push_back(it->second);
            }
            currentScope = currentScope->parent;
        }
        
        return symbols;
    }
    
    bool isSymbolVisible(string name, Scope* scope) {
        return lookupSymbolInScope(name, scope) != nullptr;
    }
    
    Scope* findSymbolScope(string name, Scope* startScope = nullptr) {
        if (startScope == nullptr) {
            startScope = scopeTree.getCurrentScope();
        }
        
        Scope* currentScope = startScope;
        while (currentScope != nullptr) {
            auto it = currentScope->symbols.find(name);
            if (it != currentScope->symbols.end()) {
                return currentScope;
            }
            currentScope = currentScope->parent;
        }
        
        return nullptr;
    }
};
```

### 作用域冲突检测
```c
class ScopeConflictDetector {
private:
    ScopeTree scopeTree;
    
public:
    void detectConflicts() {
        vector<Scope*> scopes = scopeTree.getAllScopes();
        
        for (auto scope : scopes) {
            detectConflictsInScope(scope);
        }
    }
    
    void detectConflictsInScope(Scope* scope) {
        // 检查符号冲突
        map<string, vector<SymbolEntry*>> symbolGroups;
        
        for (auto& pair : scope->symbols) {
            string name = pair.first;
            SymbolEntry* entry = pair.second;
            symbolGroups[name].push_back(entry);
        }
        
        // 检查同名符号
        for (auto& pair : symbolGroups) {
            string name = pair.first;
            vector<SymbolEntry*> entries = pair.second;
            
            if (entries.size() > 1) {
                reportError("Multiple declarations of symbol: " + name);
            }
        }
        
        // 检查子作用域冲突
        for (auto child : scope->children) {
            detectConflictsInScope(child);
        }
    }
    
    void detectShadowing() {
        vector<Scope*> scopes = scopeTree.getAllScopes();
        
        for (auto scope : scopes) {
            detectShadowingInScope(scope);
        }
    }
    
    void detectShadowingInScope(Scope* scope) {
        // 检查符号遮蔽
        for (auto& pair : scope->symbols) {
            string name = pair.first;
            SymbolEntry* entry = pair.second;
            
            // 在父作用域中查找同名符号
            Scope* parentScope = scope->parent;
            while (parentScope != nullptr) {
                auto it = parentScope->symbols.find(name);
                if (it != parentScope->symbols.end()) {
                    reportWarning("Symbol shadows declaration in outer scope: " + name);
                    break;
                }
                parentScope = parentScope->parent;
            }
        }
        
        // 递归检查子作用域
        for (auto child : scope->children) {
            detectShadowingInScope(child);
        }
    }
};
```

### 作用域缓存
```c
class ScopeCache {
private:
    map<string, SymbolEntry*> symbolCache;
    map<string, Scope*> scopeCache;
    
public:
    SymbolEntry* getCachedSymbol(string name) {
        auto it = symbolCache.find(name);
        if (it != symbolCache.end()) {
            return it->second;
        }
        
        SymbolEntry* entry = scopeTree.lookupSymbol(name);
        if (entry != nullptr) {
            symbolCache[name] = entry;
        }
        
        return entry;
    }
    
    Scope* getCachedScope(string name) {
        auto it = scopeCache.find(name);
        if (it != scopeCache.end()) {
            return it->second;
        }
        
        Scope* scope = findSymbolScope(name);
        if (scope != nullptr) {
            scopeCache[name] = scope;
        }
        
        return scope;
    }
    
    void clearCache() {
        symbolCache.clear();
        scopeCache.clear();
    }
    
    void invalidateCache(string name) {
        symbolCache.erase(name);
        scopeCache.erase(name);
    }
};
```

### 符号解析
```c
class SymbolResolver {
private:
    ScopeTree scopeTree;
    
public:
    void resolveSymbols(ASTNode* node) {
        switch (node->getType()) {
            case IDENTIFIER:
                resolveIdentifier(node);
                break;
            case QUALIFIED_NAME:
                resolveQualifiedName(node);
                break;
            case FUNCTION_CALL:
                resolveFunctionCall(node);
                break;
        }
        
        // 递归解析子节点
        for (auto child : node->getChildren()) {
            resolveSymbols(child);
        }
    }
    
    void resolveIdentifier(ASTNode* node) {
        string name = node->getValue();
        
        // 查找符号
        SymbolEntry* entry = scopeTree.lookupSymbol(name);
        if (entry == nullptr) {
            reportError("Undeclared identifier: " + name);
            return;
        }
        
        // 设置符号信息
        node->setAttribute("symbol", entry);
        node->setAttribute("scope", scopeTree.getCurrentScope()->id);
    }
    
    void resolveQualifiedName(ASTNode* node) {
        string namespaceName = node->getChild(0)->getValue();
        string symbolName = node->getChild(1)->getValue();
        
        // 查找命名空间
        Scope* namespaceScope = findNamespaceScope(namespaceName);
        if (namespaceScope == nullptr) {
            reportError("Unknown namespace: " + namespaceName);
            return;
        }
        
        // 在命名空间中查找符号
        SymbolEntry* entry = lookupSymbolInScope(symbolName, namespaceScope);
        if (entry == nullptr) {
            reportError("Symbol not found in namespace: " + namespaceName + "::" + symbolName);
            return;
        }
        
        // 设置符号信息
        node->setAttribute("symbol", entry);
        node->setAttribute("scope", namespaceScope->id);
    }
    
    void resolveFunctionCall(ASTNode* node) {
        string funcName = node->getChild(0)->getValue();
        
        // 查找函数
        SymbolEntry* entry = scopeTree.lookupSymbol(funcName);
        if (entry == nullptr) {
            reportError("Undeclared function: " + funcName);
            return;
        }
        
        // 检查是否为函数
        if (entry->type != FUNCTION) {
            reportError("Not a function: " + funcName);
            return;
        }
        
        // 设置函数信息
        node->setAttribute("symbol", entry);
        node->setAttribute("scope", scopeTree.getCurrentScope()->id);
    }
};
```

### 作用域可视化
```c
class ScopeVisualizer {
private:
    ScopeTree scopeTree;
    
public:
    void visualizeScopes() {
        cout << "Scope Tree:" << endl;
        visualizeScope(scopeTree.getRootScope(), 0);
    }
    
    void visualizeScope(Scope* scope, int depth) {
        string indent = string(depth * 2, ' ');
        
        cout << indent << "Scope " << scope->id << " (" << getScopeTypeName(scope->type) << ")" << endl;
        
        // 显示符号
        for (auto& pair : scope->symbols) {
            string name = pair.first;
            SymbolEntry* entry = pair.second;
            cout << indent << "  " << name << " (" << getSymbolTypeName(entry->type) << ")" << endl;
        }
        
        // 显示子作用域
        for (auto child : scope->children) {
            visualizeScope(child, depth + 1);
        }
    }
    
    string getScopeTypeName(ScopeType type) {
        switch (type) {
            case GLOBAL: return "Global";
            case FUNCTION: return "Function";
            case BLOCK: return "Block";
            case CLASS: return "Class";
            case NAMESPACE: return "Namespace";
            default: return "Unknown";
        }
    }
    
    string getSymbolTypeName(SymbolType type) {
        switch (type) {
            case VARIABLE: return "Variable";
            case FUNCTION: return "Function";
            case TYPE: return "Type";
            case CONSTANT: return "Constant";
            default: return "Unknown";
        }
    }
};
```
