# 属性文法与计算

## 🎯 属性文法概述

**属性文法**是在上下文无关文法的基础上，为每个语法符号添加属性的文法。属性文法为语义分析提供了理论基础和实现方法。

## 🔍 基本概念

### 属性文法定义
属性文法G = (V, T, P, S, A, R)，其中：
- **V**：非终结符集合
- **T**：终结符集合
- **P**：产生式集合
- **S**：开始符号
- **A**：属性集合
- **R**：语义规则集合

### 属性类型
```c
enum AttributeType {
    SYNTHESIZED,  // 综合属性
    INHERITED     // 继承属性
};

struct Attribute {
    string name;           // 属性名
    AttributeType type;    // 属性类型
    string value;         // 属性值
    bool computed;        // 是否已计算
};
```

## 📊 综合属性

### 综合属性定义
综合属性是从子节点向父节点传递的属性，其值由子节点的属性值计算得出。

### 综合属性示例
```c
// 算术表达式的综合属性
// E → E + T { E.val = E1.val + T.val; }
// E → T { E.val = T.val; }
// T → T * F { T.val = T1.val * F.val; }
// T → F { T.val = F.val; }
// F → (E) { F.val = E.val; }
// F → id { F.val = lookup(id.name); }
// F → num { F.val = num.val; }

class SynthesizedAttributeCalculator {
public:
    void calculateSynthesizedAttributes(ASTNode* node) {
        // 先计算子节点的综合属性
        for (auto child : node->getChildren()) {
            calculateSynthesizedAttributes(child);
        }
        
        // 计算当前节点的综合属性
        switch (node->getType()) {
            case ADDITION:
                calculateAdditionAttribute(node);
                break;
            case MULTIPLICATION:
                calculateMultiplicationAttribute(node);
                break;
            case IDENTIFIER:
                calculateIdentifierAttribute(node);
                break;
            case NUMBER:
                calculateNumberAttribute(node);
                break;
        }
    }
    
    void calculateAdditionAttribute(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        int leftVal = left->getAttribute("val").getIntValue();
        int rightVal = right->getAttribute("val").getIntValue();
        
        node->setAttribute("val", leftVal + rightVal);
    }
    
    void calculateIdentifierAttribute(ASTNode* node) {
        string name = node->getValue();
        SymbolEntry* entry = symbolTable.lookup(name);
        
        if (entry) {
            node->setAttribute("val", entry->getValue());
        } else {
            reportError("Undeclared identifier: " + name);
        }
    }
};
```

## 📊 继承属性

### 继承属性定义
继承属性是从父节点向子节点传递的属性，其值由父节点或兄弟节点的属性值计算得出。

### 继承属性示例
```c
// 类型声明的继承属性
// D → T L { L.type = T.type; }
// T → int { T.type = int; }
// T → float { T.type = float; }
// L → L1, id { L1.type = L.type; id.type = L.type; }
// L → id { id.type = L.type; }

class InheritedAttributeCalculator {
public:
    void calculateInheritedAttributes(ASTNode* node) {
        switch (node->getType()) {
            case DECLARATION:
                calculateDeclarationAttributes(node);
                break;
            case TYPE:
                calculateTypeAttributes(node);
                break;
            case IDENTIFIER_LIST:
                calculateIdentifierListAttributes(node);
                break;
            case IDENTIFIER:
                calculateIdentifierAttributes(node);
                break;
        }
        
        // 递归计算子节点的继承属性
        for (auto child : node->getChildren()) {
            calculateInheritedAttributes(child);
        }
    }
    
    void calculateDeclarationAttributes(ASTNode* node) {
        ASTNode* type = node->getChild(0);
        ASTNode* list = node->getChild(1);
        
        // 计算类型属性
        calculateInheritedAttributes(type);
        
        // 将类型传递给标识符列表
        list->setAttribute("type", type->getAttribute("type").getStringValue());
        calculateInheritedAttributes(list);
    }
    
    void calculateIdentifierListAttributes(ASTNode* node) {
        if (node->getChildren().size() == 2) {
            // L → L1, id
            ASTNode* list = node->getChild(0);
            ASTNode* id = node->getChild(1);
            
            string type = node->getAttribute("type").getStringValue();
            
            // 传递类型给子列表
            list->setAttribute("type", type);
            calculateInheritedAttributes(list);
            
            // 传递类型给标识符
            id->setAttribute("type", type);
            calculateInheritedAttributes(id);
        } else {
            // L → id
            ASTNode* id = node->getChild(0);
            string type = node->getAttribute("type").getStringValue();
            id->setAttribute("type", type);
            calculateInheritedAttributes(id);
        }
    }
};
```

## 🔧 属性依赖图

### 依赖图构造
```c
class AttributeDependencyGraph {
private:
    map<string, set<string>> dependencies;
    map<string, Attribute> attributes;
    
public:
    void buildDependencyGraph(ASTNode* node) {
        // 分析属性依赖关系
        analyzeDependencies(node);
        
        // 构造依赖图
        constructGraph();
    }
    
    void analyzeDependencies(ASTNode* node) {
        switch (node->getType()) {
            case ADDITION:
                analyzeAdditionDependencies(node);
                break;
            case ASSIGNMENT:
                analyzeAssignmentDependencies(node);
                break;
            case FUNCTION_CALL:
                analyzeFunctionCallDependencies(node);
                break;
        }
        
        // 递归分析子节点
        for (auto child : node->getChildren()) {
            analyzeDependencies(child);
        }
    }
    
    void analyzeAdditionDependencies(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        string nodeId = node->getId();
        string leftId = left->getId();
        string rightId = right->getId();
        
        // E.val 依赖于 E1.val 和 T.val
        dependencies[nodeId + ".val"].insert(leftId + ".val");
        dependencies[nodeId + ".val"].insert(rightId + ".val");
    }
    
    void constructGraph() {
        // 使用拓扑排序确定计算顺序
        vector<string> sortedAttributes = topologicalSort();
        
        // 按顺序计算属性
        for (string attr : sortedAttributes) {
            calculateAttribute(attr);
        }
    }
    
    vector<string> topologicalSort() {
        vector<string> result;
        map<string, int> inDegree;
        
        // 计算入度
        for (auto& pair : dependencies) {
            inDegree[pair.first] = pair.second.size();
        }
        
        // 拓扑排序
        queue<string> q;
        for (auto& pair : inDegree) {
            if (pair.second == 0) {
                q.push(pair.first);
            }
        }
        
        while (!q.empty()) {
            string attr = q.front();
            q.pop();
            result.push_back(attr);
            
            // 更新依赖节点的入度
            for (auto& pair : dependencies) {
                if (pair.second.find(attr) != pair.second.end()) {
                    inDegree[pair.first]--;
                    if (inDegree[pair.first] == 0) {
                        q.push(pair.first);
                    }
                }
            }
        }
        
        return result;
    }
};
```

## 🔄 属性计算算法

### 自底向上计算
```c
class BottomUpAttributeCalculator {
public:
    void calculateAttributes(ASTNode* node) {
        // 先计算子节点属性
        for (auto child : node->getChildren()) {
            calculateAttributes(child);
        }
        
        // 计算当前节点的综合属性
        calculateSynthesizedAttributes(node);
    }
    
    void calculateSynthesizedAttributes(ASTNode* node) {
        switch (node->getType()) {
            case ADDITION:
                node->setAttribute("val", 
                    node->getChild(0)->getAttribute("val").getIntValue() +
                    node->getChild(1)->getAttribute("val").getIntValue());
                break;
            case MULTIPLICATION:
                node->setAttribute("val", 
                    node->getChild(0)->getAttribute("val").getIntValue() *
                    node->getChild(1)->getAttribute("val").getIntValue());
                break;
            case IDENTIFIER:
                node->setAttribute("val", 
                    symbolTable.lookup(node->getValue())->getValue());
                break;
            case NUMBER:
                node->setAttribute("val", stoi(node->getValue()));
                break;
        }
    }
};
```

### 自顶向下计算
```c
class TopDownAttributeCalculator {
public:
    void calculateAttributes(ASTNode* node) {
        // 计算当前节点的继承属性
        calculateInheritedAttributes(node);
        
        // 递归计算子节点属性
        for (auto child : node->getChildren()) {
            calculateAttributes(child);
        }
        
        // 计算当前节点的综合属性
        calculateSynthesizedAttributes(node);
    }
    
    void calculateInheritedAttributes(ASTNode* node) {
        switch (node->getType()) {
            case DECLARATION:
                calculateDeclarationInherited(node);
                break;
            case IDENTIFIER_LIST:
                calculateIdentifierListInherited(node);
                break;
        }
    }
    
    void calculateDeclarationInherited(ASTNode* node) {
        ASTNode* type = node->getChild(0);
        ASTNode* list = node->getChild(1);
        
        // 计算类型属性
        calculateAttributes(type);
        
        // 将类型传递给标识符列表
        list->setAttribute("type", type->getAttribute("type").getStringValue());
    }
};
```

## 📈 属性文法应用

### 类型检查
```c
class TypeChecker {
public:
    void checkTypes(ASTNode* node) {
        switch (node->getType()) {
            case ADDITION:
                checkAdditionTypes(node);
                break;
            case ASSIGNMENT:
                checkAssignmentTypes(node);
                break;
            case FUNCTION_CALL:
                checkFunctionCallTypes(node);
                break;
        }
        
        // 递归检查子节点
        for (auto child : node->getChildren()) {
            checkTypes(child);
        }
    }
    
    void checkAdditionTypes(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        string leftType = left->getAttribute("type").getStringValue();
        string rightType = right->getAttribute("type").getStringValue();
        
        if (leftType != rightType) {
            reportError("Type mismatch in addition: " + leftType + " + " + rightType);
        }
        
        node->setAttribute("type", leftType);
    }
    
    void checkAssignmentTypes(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        string leftType = left->getAttribute("type").getStringValue();
        string rightType = right->getAttribute("type").getStringValue();
        
        if (!isCompatible(leftType, rightType)) {
            reportError("Type mismatch in assignment: " + leftType + " = " + rightType);
        }
    }
};
```

class CodeGenerator {
public:
    void generateCode(ASTNode* node) {
        switch (node->getType()) {

                generateAdditionCode(node);
                break;
            case ASSIGNMENT:
                generateAssignmentCode(node);
                break;

        for (auto child : node->getChildren()) {
            generateCode(child);
        }
    }
    
    void generateAdditionCode(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        generateCode(left);
        generateCode(right);
        
        // 生成加法指令
        string temp = generateTemp();
        emit("add " + temp + ", " + 
             left->getAttribute("temp").getStringValue() + ", " + 
             right->getAttribute("temp").getStringValue());
        
        node->setAttribute("temp", temp);
    }
    
    void generateAssignmentCode(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        generateCode(right);
        
        // 生成赋值指令
        emit("mov " + left->getAttribute("name").getStringValue() + ", " + 
             right->getAttribute("temp").getStringValue());
    }
};

## 🎯 属性文法特点

### 优点
- **理论基础**：提供坚实的理论基础
- **形式化**：形式化描述语义规则
- **可验证**：可以验证语义规则的正确性

### 缺点
- **复杂度高**：属性计算复杂度高
- **实现困难**：实现相对困难
- **调试复杂**：调试相对复杂

### 属性计算优化
```c
class OptimizedAttributeCalculator {
private:
    map<string, Attribute> attributeCache;
    
public:
    Attribute calculateAttribute(ASTNode* node, string attrName) {
        string key = node->getId() + "." + attrName;
        
        if (attributeCache.find(key) != attributeCache.end()) {
            return attributeCache[key];
        }
        
        Attribute attr = computeAttribute(node, attrName);
        attributeCache[key] = attr;
        
        return attr;
    }
    
    void clearCache() {
        attributeCache.clear();
    }
};
```

### 依赖图优化
```c
class OptimizedDependencyGraph {
private:
    map<string, set<string>> dependencies;
    map<string, int> computationCount;
    
public:
    void optimizeDependencies() {
        // 消除冗余依赖
        eliminateRedundantDependencies();
        
        // 合并相同依赖
        mergeSimilarDependencies();
        
        // 优化计算顺序
        optimizeComputationOrder();
    }
    
    void eliminateRedundantDependencies() {
        for (auto& pair : dependencies) {
            set<string>& deps = pair.second;
            
            // 移除传递依赖
            set<string> transitiveDeps;
            for (string dep : deps) {
                if (dependencies.find(dep) != dependencies.end()) {
                    transitiveDeps.insert(dependencies[dep].begin(), dependencies[dep].end());
                }
            }
            
            for (string transitiveDep : transitiveDeps) {
                deps.erase(transitiveDep);
            }
        }
    }
};
```
