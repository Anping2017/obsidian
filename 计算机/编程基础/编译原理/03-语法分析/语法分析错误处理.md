# 语法分析错误处理

## 🎯 语法分析错误处理概述

**语法分析错误处理**是语法分析器的重要组成部分，负责检测、报告和恢复语法错误。良好的错误处理能够提高编译器的用户体验。

## 🔍 语法错误类型

### 错误分类
| 错误类型 | 描述 | 示例 |
|----------|------|------|
| 语法错误 | 不符合语法规则 | `if (x = 5)` |
| 语义错误 | 语法正确但语义错误 | `int x = "hello"` |
| 词法错误 | 词法分析阶段错误 | `int x = 5@` |

### 常见语法错误
```c
// 1. 缺少分号
int x = 5
int y = 10;

// 2. 括号不匹配
if (x > 0 {
    printf("positive");
}

// 3. 缺少关键字
if (x > 0)
    printf("positive");

// 4. 操作符错误
int x = 5 + * 3;

// 5. 类型错误
int x = "hello";
```

## 🔧 错误检测

### 错误检测策略
```c
class SyntaxErrorDetector {
private:
    vector<Token> tokens;
    int position;
    vector<SyntaxError> errors;
    
public:
    void detectErrors() {
        for (int i = 0; i < tokens.size(); i++) {
            Token current = tokens[i];
            Token next = (i + 1 < tokens.size()) ? tokens[i + 1] : Token();
            
            // 检测常见错误
            detectMissingSemicolon(current, next);
            detectMismatchedBrackets(current, next);
            detectInvalidOperators(current, next);
            detectTypeErrors(current, next);
        }
    }
    
    void detectMissingSemicolon(Token current, Token next) {
        if (current.type == IDENTIFIER && next.type == IDENTIFIER) {
            errors.push_back(SyntaxError(
                "Missing semicolon after " + current.value,
                current.line, current.column
            ));
        }
    }
    
    void detectMismatchedBrackets(Token current, Token next) {
        if (current.type == LPAREN && next.type == RBRACE) {
            errors.push_back(SyntaxError(
                "Mismatched brackets: expected ')' but found '}'",
                current.line, current.column
            ));
        }
    }
};
```

### 错误信息结构
```c
struct SyntaxError {
    string message;
    int line;
    int column;
    ErrorSeverity severity;
    string suggestion;
    
    SyntaxError(string msg, int l, int c, ErrorSeverity sev = ERROR, string sug = "")
        : message(msg), line(l), column(c), severity(sev), suggestion(sug) {}
    
    string toString() const {
        return "Error at line " + to_string(line) + ", column " + to_string(column) + 
               ": " + message + (suggestion.empty() ? "" : " Suggestion: " + suggestion);
    }
};

enum ErrorSeverity {
    WARNING,
    ERROR,
    FATAL
};
```

## 🔄 错误恢复

### 错误恢复策略
```c
class ErrorRecovery {
private:
    vector<Token> tokens;
    int position;
    vector<SyntaxError> errors;
    
public:
    void recoverFromError() {
        while (position < tokens.size()) {
            try {
                parseStatement();
            } catch (SyntaxError& error) {
                errors.push_back(error);
                recoverFromSyntaxError(error);
            }
        }
    }
    
    void recoverFromSyntaxError(SyntaxError& error) {
        switch (error.severity) {
            case WARNING:
                // 继续分析
                break;
            case ERROR:
                // 尝试恢复
                recoverFromError();
                break;
            case FATAL:
                // 停止分析
                return;
        }
    }
    
    void recoverFromError() {
        // 跳过当前token
        position++;
        
        // 尝试同步
        synchronize();
    }
    
    void synchronize() {
        // 寻找同步点
        while (position < tokens.size()) {
            Token current = tokens[position];
            
            if (isSynchronizationPoint(current)) {
                break;
            }
            
            position++;
        }
    }
    
    bool isSynchronizationPoint(Token token) {
        return token.type == SEMICOLON || 
               token.type == RBRACE || 
               token.type == END_OF_INPUT;
    }
};
```

#### 2. 短语级恢复
```c
void phraseLevelRecovery() {
    // 尝试修复常见错误
    if (isMissingSemicolon()) {
        insertSemicolon();
    } else if (isMismatchedBrackets()) {
        fixBrackets();
    } else if (isMissingKeyword()) {
        insertKeyword();
    }
}

bool isMissingSemicolon() {
    Token current = tokens[position];
    Token next = (position + 1 < tokens.size()) ? tokens[position + 1] : Token();
    
    return current.type == IDENTIFIER && next.type == IDENTIFIER;
}

void insertSemicolon() {
    Token semicolon(SEMICOLON, ";", tokens[position].line, tokens[position].column + 1);
    tokens.insert(tokens.begin() + position + 1, semicolon);
}
```

#### 3. 错误产生式恢复
```c
void errorProductionRecovery() {
    // 添加错误产生式
    // E → error
    // E → error + T
    // E → error * F
    
    if (isExpressionError()) {
        // 使用错误产生式
        parseErrorExpression();
    }
}

void parseErrorExpression() {
    // 解析错误表达式
    Token errorToken(ERROR, "error", tokens[position].line, tokens[position].column);
    
    // 尝试恢复
    if (position + 1 < tokens.size()) {
        Token next = tokens[position + 1];
        if (next.type == PLUS || next.type == MULTIPLY) {
            // 继续解析
            parseExpression();
        }
    }
}
```

## 📊 错误报告

### 错误报告格式
```c
class ErrorReporter {
private:
    vector<SyntaxError> errors;
    string sourceCode;
    
public:
    void reportErrors() {
        for (auto& error : errors) {
            cout << error.toString() << endl;
            printErrorContext(error);
        }
    }
    
    void printErrorContext(SyntaxError& error) {
        // 打印错误上下文
        vector<string> lines = splitLines(sourceCode);
        
        if (error.line > 0 && error.line <= lines.size()) {
            string line = lines[error.line - 1];
            cout << line << endl;
            
            // 打印错误位置指示器
            string indicator = string(error.column - 1, ' ') + "^";
            cout << indicator << endl;
        }
    }
    
    void addError(SyntaxError error) {
        errors.push_back(error);
    }
    
    int getErrorCount() const {
        return errors.size();
    }
    
    bool hasErrors() const {
        return !errors.empty();
    }
};
```

### 错误上下文显示
```c
void displayErrorContext(SyntaxError& error) {
    // 显示错误前后的代码
    vector<string> lines = splitLines(sourceCode);
    
    int startLine = max(1, error.line - 2);
    int endLine = min((int)lines.size(), error.line + 2);
    
    for (int i = startLine; i <= endLine; i++) {
        cout << setw(4) << i << ": " << lines[i - 1] << endl;
        
        if (i == error.line) {
            string indicator = string(error.column - 1, ' ') + "^";
            cout << "     " << indicator << endl;
        }
    }
}
```

## 🔧 错误处理实现

### LL分析器错误处理
```c
class LLErrorHandler {
private:
    LL1Parser* parser;
    ErrorReporter* reporter;
    
public:
    void handleParseError(Token unexpected) {
        SyntaxError error(
            "Unexpected token: " + unexpected.value,
            unexpected.line, unexpected.column
        );
        
        reporter->addError(error);
        
        // 尝试恢复
        recoverFromError(unexpected);
    }
    
    void recoverFromError(Token unexpected) {
        // 跳过错误token
        parser->advance();
        
        // 寻找同步点
        while (!parser->isAtEnd()) {
            Token current = parser->getCurrentToken();
            
            if (isSynchronizationPoint(current)) {
                break;
            }
            
            parser->advance();
        }
    }
};
```

### LR分析器错误处理
```c
class LRErrorHandler {
private:
    LRParser* parser;
    ErrorReporter* reporter;
    
public:
    void handleParseError(int state, Token unexpected) {
        SyntaxError error(
            "Parse error in state " + to_string(state) + 
            " with token: " + unexpected.value,
            unexpected.line, unexpected.column
        );
        
        reporter->addError(error);
        
        // 尝试恢复
        recoverFromError(state, unexpected);
    }
    
    void recoverFromError(int state, Token unexpected) {
        // 从栈中弹出状态直到找到恢复状态
        while (!parser->isStackEmpty()) {
            int currentState = parser->getTopState();
            
            if (canRecoverFromState(currentState)) {
                break;
            }
            
            parser->popState();
        }
        
        // 跳过错误token
        parser->advance();
    }
    
    bool canRecoverFromState(int state) {
        // 检查状态是否可以恢复
        return parser->hasAction(state, SEMICOLON) ||
               parser->hasAction(state, RBRACE) ||
               parser->hasAction(state, END_OF_INPUT);
    }
};
```

## 🎯 错误处理策略

### 错误处理原则
1. **快速检测**：尽快检测错误
2. **准确报告**：提供准确的错误信息
3. **优雅恢复**：尽可能继续分析
4. **用户友好**：提供有用的错误信息

### 错误处理策略选择
```c
enum ErrorHandlingStrategy {
    PANIC_MODE,        // 恐慌模式
    PHRASE_LEVEL,      // 短语级恢复
    ERROR_PRODUCTION,  // 错误产生式
    ADAPTIVE          // 自适应恢复
};

class ErrorHandlerFactory {
public:
    static ErrorHandler* createHandler(ErrorHandlingStrategy strategy) {
        switch (strategy) {
            case PANIC_MODE:
                return new PanicModeHandler();
            case PHRASE_LEVEL:
                return new PhraseLevelHandler();
            case ERROR_PRODUCTION:
                return new ErrorProductionHandler();
            case ADAPTIVE:
                return new AdaptiveHandler();
            default:
                return new PanicModeHandler();
        }
    }
};
```
