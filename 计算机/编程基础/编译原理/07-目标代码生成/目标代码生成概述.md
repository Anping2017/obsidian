# 目标代码生成概述

## 🎯 目标代码生成概述

**目标代码生成**是编译器的最后一个阶段，负责将中间代码转换为目标机器的机器代码。目标代码生成的质量直接影响最终程序的执行效率和资源使用。

## 🔍 基本概念

### 目标代码生成定义
目标代码生成是将中间表示（如三地址码、四元式、抽象语法树）转换为目标机器可执行的机器代码的过程。

### 目标代码生成特点
- **机器相关**：需要了解目标机器的指令集和架构特性
- **优化关键**：是编译器优化的最后机会
- **资源管理**：需要合理分配寄存器和内存资源
- **代码质量**：直接影响最终程序的性能

```c
    CodeGenerator(TargetMachine* machine, OptimizationLevel level) 
        : targetMachine(machine), optimizationLevel(level) {
        initializeInstructionTemplates();
        initializeRegisterInfo();
    }
    
    void initializeInstructionTemplates() {
        // 初始化指令模板
        instructionTemplates["ADD"] = InstructionTemplate(
            "add %dst, %src1, %src2", 
            {REGISTER, REGISTER, REGISTER},
            {WRITE, READ, READ}
        );
        
        instructionTemplates["SUB"] = InstructionTemplate(
            "sub %dst, %src1, %src2", 
            {REGISTER, REGISTER, REGISTER},
            {WRITE, READ, READ}
        );
        
        instructionTemplates["MUL"] = InstructionTemplate(
            "mul %dst, %src1, %src2", 
            {REGISTER, REGISTER, REGISTER},
            {WRITE, READ, READ}
        );
        
        instructionTemplates["DIV"] = InstructionTemplate(
            "div %dst, %src1, %src2", 
            {REGISTER, REGISTER, REGISTER},
            {WRITE, READ, READ}
        );
        
        instructionTemplates["LOAD"] = InstructionTemplate(
            "ld %dst, %offset(%base)", 
            {REGISTER, IMMEDIATE, REGISTER},
            {WRITE, READ, READ}
        );
        
        instructionTemplates["STORE"] = InstructionTemplate(
            "st %src, %offset(%base)", 
            {REGISTER, IMMEDIATE, REGISTER},
            {READ, READ, READ}
        );
        
        instructionTemplates["BRANCH"] = InstructionTemplate(
            "b %label", 
            {LABEL},
            {READ}
        );
        
        instructionTemplates["CALL"] = InstructionTemplate(
            "call %func", 
            {FUNCTION},
            {READ}
        );
        
        instructionTemplates["RETURN"] = InstructionTemplate(
            "ret", 
            {},
            {}
        );
    }
    
    void initializeRegisterInfo() {
        // 初始化寄存器信息
        registerInfo["R0"] = RegisterInfo(0, GENERAL_PURPOSE, true);
        registerInfo["R1"] = RegisterInfo(1, GENERAL_PURPOSE, true);
        registerInfo["R2"] = RegisterInfo(2, GENERAL_PURPOSE, true);
        registerInfo["R3"] = RegisterInfo(3, GENERAL_PURPOSE, true);
        registerInfo["R4"] = RegisterInfo(4, GENERAL_PURPOSE, true);
        registerInfo["R5"] = RegisterInfo(5, GENERAL_PURPOSE, true);
        registerInfo["R6"] = RegisterInfo(6, GENERAL_PURPOSE, true);
        registerInfo["R7"] = RegisterInfo(7, GENERAL_PURPOSE, true);
        registerInfo["R8"] = RegisterInfo(8, GENERAL_PURPOSE, true);
        registerInfo["R9"] = RegisterInfo(9, GENERAL_PURPOSE, true);
        registerInfo["R10"] = RegisterInfo(10, GENERAL_PURPOSE, true);
        registerInfo["R11"] = RegisterInfo(11, GENERAL_PURPOSE, true);
        registerInfo["R12"] = RegisterInfo(12, GENERAL_PURPOSE, true);
        registerInfo["R13"] = RegisterInfo(13, GENERAL_PURPOSE, true);
        registerInfo["R14"] = RegisterInfo(14, GENERAL_PURPOSE, true);
        registerInfo["R15"] = RegisterInfo(15, GENERAL_PURPOSE, true);
        
        registerInfo["SP"] = RegisterInfo(16, STACK_POINTER, false);
        registerInfo["PC"] = RegisterInfo(17, PROGRAM_COUNTER, false);
        registerInfo["LR"] = RegisterInfo(18, LINK_REGISTER, false);
    }
    
    vector<MachineInstruction> generateCode(IntermediateCode* intermediateCode) {
        vector<MachineInstruction> machineCode;
        
        // 1. 指令选择
        vector<SelectedInstruction> selectedInstructions = selectInstructions(intermediateCode);
        
        // 2. 寄存器分配
        map<string, string> registerAllocation = allocateRegisters(selectedInstructions);
        
        // 3. 指令调度
        vector<SelectedInstruction> scheduledInstructions = scheduleInstructions(selectedInstructions);
        
        // 4. 代码生成
        machineCode = generateMachineCode(scheduledInstructions, registerAllocation);
        
        return machineCode;
    }
    
    vector<SelectedInstruction> selectInstructions(IntermediateCode* intermediateCode) {
        vector<SelectedInstruction> selectedInstructions;
        
        for (auto& inst : intermediateCode->instructions) {
            SelectedInstruction selected = selectInstruction(inst);
            selectedInstructions.push_back(selected);
        }
        
        return selectedInstructions;
    }
    
    SelectedInstruction selectInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        
        switch (inst.op) {
            case ADD:
                selected = selectAddInstruction(inst);
                break;
            case SUB:
                selected = selectSubInstruction(inst);
                break;
            case MUL:
                selected = selectMulInstruction(inst);
                break;
            case DIV:
                selected = selectDivInstruction(inst);
                break;
            case LOAD:
                selected = selectLoadInstruction(inst);
                break;
            case STORE:
                selected = selectStoreInstruction(inst);
                break;
            case BRANCH:
                selected = selectBranchInstruction(inst);
                break;
            case CALL:
                selected = selectCallInstruction(inst);
                break;
            case RETURN:
                selected = selectReturnInstruction(inst);
                break;
            default:
                selected = selectDefaultInstruction(inst);
                break;
        }
        
        return selected;
    }
    
    SelectedInstruction selectAddInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "ADD";
        selected.operands = {inst.result, inst.arg1, inst.arg2};

        selected.template = instructionTemplates["LOAD"];
        return selected;
    }

        return selected;
    }
    
    SelectedInstruction selectBranchInstruction(IntermediateInstruction& inst) {

        return selected;
    }
    
    SelectedInstruction selectCallInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "CALL";
        selected.operands = {inst.result};
        selected.template = instructionTemplates["CALL"];

    }
    
    SelectedInstruction selectReturnInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "RETURN";
        selected.operands = {};
        selected.template = instructionTemplates["RETURN"];
        return selected;
    }
    
    SelectedInstruction selectDefaultInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "UNKNOWN";
        selected.operands = {inst.result, inst.arg1, inst.arg2};
        selected.template = InstructionTemplate("", {}, {});
        return selected;
    }
    
    map<string, string> allocateRegisters(vector<SelectedInstruction>& instructions) {
        // 这里需要实现寄存器分配算法
        map<string, string> allocation;
        
        // 简化实现：为每个变量分配一个寄存器
        int regCount = 0;
        for (auto& inst : instructions) {
            for (auto& operand : inst.operands) {
                if (allocation.find(operand) == allocation.end()) {
                    string reg = "R" + to_string(regCount++);
                    allocation[operand] = reg;
                }
            }
        }
        
        return allocation;
    }
    
    vector<SelectedInstruction> scheduleInstructions(vector<SelectedInstruction>& instructions) {
        // 这里需要实现指令调度算法
        // 简化实现：保持原有顺序
        return instructions;
    }
    
    vector<MachineInstruction> generateMachineCode(vector<SelectedInstruction>& instructions, 
                                                   map<string, string>& registerAllocation) {
        vector<MachineInstruction> machineCode;
        
        for (auto& inst : instructions) {
            MachineInstruction machineInst = generateMachineInstruction(inst, registerAllocation);
            machineCode.push_back(machineInst);
        }
        
        return machineCode;
    }
    
    MachineInstruction generateMachineInstruction(SelectedInstruction& inst, 
                                                 map<string, string>& registerAllocation) {
        MachineInstruction machineInst;
        machineInst.opcode = inst.opcode;
        
        // 根据指令模板生成机器指令
        string templateStr = inst.template.format;
        vector<OperandType> operandTypes = inst.template.operandTypes;
        vector<OperandAccess> operandAccesses = inst.template.operandAccesses;
        
        // 替换模板中的占位符
        for (int i = 0; i < inst.operands.size() && i < operandTypes.size(); i++) {
            string operand = inst.operands[i];
            OperandType type = operandTypes[i];
            OperandAccess access = operandAccesses[i];
            
            if (type == REGISTER) {
                string reg = registerAllocation[operand];
                templateStr = replacePlaceholder(templateStr, "%dst", reg);
                templateStr = replacePlaceholder(templateStr, "%src1", reg);
                templateStr = replacePlaceholder(templateStr, "%src2", reg);
                templateStr = replacePlaceholder(templateStr, "%base", reg);
            } else if (type == IMMEDIATE) {
                templateStr = replacePlaceholder(templateStr, "%offset", operand);
            } else if (type == LABEL) {
                templateStr = replacePlaceholder(templateStr, "%label", operand);
            } else if (type == FUNCTION) {
                templateStr = replacePlaceholder(templateStr, "%func", operand);
            }
        }
        
        machineInst.assembly = templateStr;
        machineInst.binary = assembleInstruction(templateStr);
        
        return machineInst;
    }
    
    string replacePlaceholder(string template, string placeholder, string value) {
        size_t pos = template.find(placeholder);
        if (pos != string::npos) {
            template.replace(pos, placeholder.length(), value);
        }
        return template;
    }

class TargetMachine {
public:
    string architecture;
    int numRegisters;
    int registerSize;
    bool hasFloatingPoint;
    bool hasSIMD;
    
    TargetMachine(string arch, int numRegs, int regSize, bool fp, bool simd) 
        : architecture(arch), numRegisters(numRegs), registerSize(regSize), 
          hasFloatingPoint(fp), hasSIMD(simd) {}
};

class InstructionTemplate {
public:
    string format;
    vector<OperandType> operandTypes;
    vector<OperandAccess> operandAccesses;
    
    InstructionTemplate(string fmt, vector<OperandType> types, vector<OperandAccess> accesses) 
        : format(fmt), operandTypes(types), operandAccesses(accesses) {}
};

class RegisterInfo {
public:
    int number;
    RegisterType type;
    bool isGeneralPurpose;
    
    RegisterInfo(int num, RegisterType t, bool general) 
        : number(num), type(t), isGeneralPurpose(general) {}
};

class SelectedInstruction {
public:
    string opcode;
    vector<string> operands;
    InstructionTemplate template;
};

class MachineInstruction {
public:
    string opcode;
    string assembly;
    vector<uint8_t> binary;
};

    FUNCTION

    WRITE,
    READ_WRITE
};

    GENERAL_PURPOSE,

    PROGRAM_COUNTER,
    LINK_REGISTER
```

## 🎯 代码生成集成

### 代码生成管理器
```c
class CodeGenerationManager {
private:
    map<TargetMachineType, CodeGenerator*> generators;
    map<TargetMachineType, CodeGenerationOptimizer*> optimizers;
    
public:
    CodeGenerationManager() {
        // 初始化各种目标机器的代码生成器
        initializeGenerators();
        initializeOptimizers();
    }
    
    void initializeGenerators() {
        // x86代码生成器
        TargetMachine* x86Machine = new TargetMachine("x86", 16, 32, true, true);
        generators[X86_ARCHITECTURE] = new CodeGenerator(x86Machine, GLOBAL_OPTIMIZATION);
        
        // ARM代码生成器
        TargetMachine* armMachine = new TargetMachine("ARM", 16, 32, true, true);
        generators[ARM_ARCHITECTURE] = new CodeGenerator(armMachine, GLOBAL_OPTIMIZATION);
        
        // MIPS代码生成器
        TargetMachine* mipsMachine = new TargetMachine("MIPS", 32, 32, true, false);
        generators[MIPS_ARCHITECTURE] = new CodeGenerator(mipsMachine, GLOBAL_OPTIMIZATION);
        
        // RISC-V代码生成器
        TargetMachine* riscvMachine = new TargetMachine("RISC-V", 32, 32, true, true);
        generators[RISC_V_ARCHITECTURE] = new CodeGenerator(riscvMachine, GLOBAL_OPTIMIZATION);
        
        // 虚拟机代码生成器
        TargetMachine* vmMachine = new TargetMachine("VM", 256, 32, true, false);
        generators[VIRTUAL_MACHINE] = new CodeGenerator(vmMachine, GLOBAL_OPTIMIZATION);
    }
    
    void initializeOptimizers() {
        // 为每个目标机器初始化优化器
        for (auto& pair : generators) {
            TargetMachineType type = pair.first;
            CodeGenerator* generator = pair.second;
            optimizers[type] = new CodeGenerationOptimizer(generator, GLOBAL_OPTIMIZATION);
        }
    }
    
    vector<MachineInstruction> generateCode(TargetMachineType targetType, 
                                           IntermediateCode* intermediateCode) {
        if (generators.find(targetType) != generators.end()) {
            CodeGenerator* generator = generators[targetType];
            CodeGenerationOptimizer* optimizer = optimizers[targetType];
            
            // 生成代码
            vector<MachineInstruction> machineCode = generator->generateCode(intermediateCode);
            
            // 优化代码
            optimizer->optimizeCodeGeneration(machineCode);
            
            return machineCode;
        }
        
        return vector<MachineInstruction>();
    }
    
    void generateCodeForAllTargets(IntermediateCode* intermediateCode) {
        cout << "为所有目标机器生成代码" << endl;
        cout << "===========================================" << endl;
        
        for (auto& pair : generators) {
            TargetMachineType type = pair.first;
            cout << "目标机器: " << getTargetMachineName(type) << endl;
            
            vector<MachineInstruction> machineCode = generateCode(type, intermediateCode);
            
            cout << "生成指令数: " << machineCode.size() << endl;
            cout << "代码大小: " << calculateCodeSize(machineCode) << " bytes" << endl;
            cout << endl;
        }
    }
    
    string getTargetMachineName(TargetMachineType type) {
        switch (type) {
            case X86_ARCHITECTURE: return "x86";
            case ARM_ARCHITECTURE: return "ARM";
            case MIPS_ARCHITECTURE: return "MIPS";
            case RISC_V_ARCHITECTURE: return "RISC-V";
            case VIRTUAL_MACHINE: return "Virtual Machine";
            default: return "Unknown";
        }
    }
    
    int calculateCodeSize(vector<MachineInstruction>& instructions) {
        int totalSize = 0;
        for (auto& inst : instructions) {
            totalSize += inst.binary.size();
        }
        return totalSize;
    }
    
    void cleanup() {
        for (auto& pair : generators) {
            delete pair.second;
        }
        for (auto& pair : optimizers) {
            delete pair.second;
        }
        generators.clear();
        optimizers.clear();
    }
    
    ~CodeGenerationManager() {
        cleanup();
    }
};
```
