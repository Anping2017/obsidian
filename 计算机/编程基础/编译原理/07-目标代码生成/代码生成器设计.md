# 代码生成器设计

## 🎯 代码生成器设计概述

**代码生成器设计**是目标代码生成的核心环节，负责将中间代码转换为目标机器代码。代码生成器的设计质量直接影响最终代码的性能和正确性。

## 🔍 基本概念

### 代码生成器定义
代码生成器是编译器的最后一个阶段，负责将中间代码转换为目标机器的机器代码或汇编代码。

### 代码生成器特点
- **机器相关**：针对特定目标机器
- **性能关键**：直接影响程序执行效率
- **复杂设计**：需要处理多种优化技术
- **错误处理**：需要处理各种错误情况

```c
class CodeGenerator {
private:
    TargetMachine* targetMachine;
    OptimizationLevel optimizationLevel;
    map<string, InstructionTemplate> instructionTemplates;
    map<string, RegisterInfo> registerInfo;
    map<string, AddressingMode> addressingModes;
    
public:
    CodeGenerator(TargetMachine* machine, OptimizationLevel level) 
        : targetMachine(machine), optimizationLevel(level) {
        initializeInstructionTemplates();
        initializeRegisterInfo();
        initializeAddressingModes();
    }
    
    void initializeInstructionTemplates() {
        // 初始化指令模板
        instructionTemplates["ADD"] = InstructionTemplate(
            "add %{dest}, %{src1}, %{src2}",
            {"dest", "src1", "src2"},
            {"reg", "reg", "reg"}
        );
        
        instructionTemplates["SUB"] = InstructionTemplate(
            "sub %{dest}, %{src1}, %{src2}",
            {"dest", "src1", "src2"},
            {"reg", "reg", "reg"}
        );
        
        instructionTemplates["MUL"] = InstructionTemplate(
            "mul %{dest}, %{src1}, %{src2}",
            {"dest", "src1", "src2"},
            {"reg", "reg", "reg"}
        );
        
        instructionTemplates["DIV"] = InstructionTemplate(
            "div %{dest}, %{src1}, %{src2}",
            {"dest", "src1", "src2"},
            {"reg", "reg", "reg"}
        );
        
        instructionTemplates["LOAD"] = InstructionTemplate(
            "ld %{dest}, %{offset}(%{base})",
            {"dest", "offset", "base"},
            {"reg", "imm", "reg"}
        );
        
        instructionTemplates["STORE"] = InstructionTemplate(
            "st %{src}, %{offset}(%{base})",
            {"src", "offset", "base"},
            {"reg", "imm", "reg"}
        );
        
        instructionTemplates["BRANCH"] = InstructionTemplate(
            "b %{condition} %{label}",
            {"condition", "label"},
            {"cond", "label"}
        );
        
        instructionTemplates["CALL"] = InstructionTemplate(
            "call %{target}",
            {"target"},
            {"label"}
        );
        
        instructionTemplates["RETURN"] = InstructionTemplate(
            "ret",
            {},
            {}
        );
    }
    
    void initializeRegisterInfo() {
        // 初始化寄存器信息
        registerInfo["r0"] = RegisterInfo("r0", "general", true, false);
        registerInfo["r1"] = RegisterInfo("r1", "general", true, false);
        registerInfo["r2"] = RegisterInfo("r2", "general", true, false);
        registerInfo["r3"] = RegisterInfo("r3", "general", true, false);
        registerInfo["r4"] = RegisterInfo("r4", "general", true, false);
        registerInfo["r5"] = RegisterInfo("r5", "general", true, false);
        registerInfo["r6"] = RegisterInfo("r6", "general", true, false);
        registerInfo["r7"] = RegisterInfo("r7", "general", true, false);
        registerInfo["r8"] = RegisterInfo("r8", "general", true, false);
        registerInfo["r9"] = RegisterInfo("r9", "general", true, false);
        registerInfo["r10"] = RegisterInfo("r10", "general", true, false);
        registerInfo["r11"] = RegisterInfo("r11", "general", true, false);
        registerInfo["r12"] = RegisterInfo("r12", "general", true, false);
        registerInfo["r13"] = RegisterInfo("r13", "general", true, false);
        registerInfo["r14"] = RegisterInfo("r14", "general", true, false);
        registerInfo["r15"] = RegisterInfo("r15", "general", true, false);
        
        registerInfo["sp"] = RegisterInfo("sp", "stack", false, true);
        registerInfo["fp"] = RegisterInfo("fp", "frame", false, true);
        registerInfo["lr"] = RegisterInfo("lr", "link", false, true);
        registerInfo["pc"] = RegisterInfo("pc", "program", false, true);
    }
    
    void initializeAddressingModes() {
        // 初始化寻址模式
        addressingModes["REGISTER"] = AddressingMode("REGISTER", "寄存器寻址");
        addressingModes["IMMEDIATE"] = AddressingMode("IMMEDIATE", "立即数寻址");
        addressingModes["MEMORY"] = AddressingMode("MEMORY", "内存寻址");
        addressingModes["INDEXED"] = AddressingMode("INDEXED", "变址寻址");
        addressingModes["INDIRECT"] = AddressingMode("INDIRECT", "间接寻址");
    }
    
    vector<MachineInstruction> generateCode(IntermediateCode* intermediateCode) {
        vector<MachineInstruction> machineCode;
        
        // 1. 指令选择
        vector<SelectedInstruction> selectedInstructions = selectInstructions(intermediateCode);
        
        // 2. 寄存器分配
        map<string, string> registerAllocation = allocateRegisters(selectedInstructions);
        
        // 3. 指令调度
        vector<SelectedInstruction> scheduledInstructions = scheduleInstructions(selectedInstructions);
        
        // 4. 机器代码生成
        machineCode = generateMachineCode(scheduledInstructions, registerAllocation);
        
        return machineCode;
    }
    
    vector<SelectedInstruction> selectInstructions(IntermediateCode* intermediateCode) {
        vector<SelectedInstruction> selectedInstructions;
        
        for (auto& inst : intermediateCode->instructions) {
            SelectedInstruction selected = selectInstruction(inst);
            selectedInstructions.push_back(selected);
        }
        
        return selectedInstructions;
    }
    
    SelectedInstruction selectInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        
        switch (inst.op) {
            case "ADD":
                selected = selectAddInstruction(inst);
                break;
            case "SUB":
                selected = selectSubInstruction(inst);
                break;
            case "MUL":
                selected = selectMulInstruction(inst);
                break;
            case "DIV":
                selected = selectDivInstruction(inst);
                break;
            case "LOAD":
                selected = selectLoadInstruction(inst);
                break;
            case "STORE":
                selected = selectStoreInstruction(inst);
                break;
            case "BRANCH":
                selected = selectBranchInstruction(inst);
                break;
            case "CALL":
                selected = selectCallInstruction(inst);
                break;
            case "RETURN":
                selected = selectReturnInstruction(inst);
                break;
            default:
                selected = selectDefaultInstruction(inst);
                break;
        }
        
        return selected;
    }
    
    SelectedInstruction selectAddInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "ADD";
        selected.template = instructionTemplates["ADD"];
        selected.operands = {inst.result, inst.arg1, inst.arg2};
        selected.addressingModes = {"reg", "reg", "reg"};
        selected.cost = 1;
        return selected;
    }
    
    SelectedInstruction selectSubInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "SUB";
        selected.template = instructionTemplates["SUB"];
        selected.operands = {inst.result, inst.arg1, inst.arg2};
        selected.addressingModes = {"reg", "reg", "reg"};
        selected.cost = 1;
        return selected;
    }
    
    SelectedInstruction selectMulInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "MUL";
        selected.template = instructionTemplates["MUL"];
        selected.operands = {inst.result, inst.arg1, inst.arg2};
        selected.addressingModes = {"reg", "reg", "reg"};
        selected.cost = 3;
        return selected;
    }
    
    SelectedInstruction selectDivInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "DIV";
        selected.template = instructionTemplates["DIV"];
        selected.operands = {inst.result, inst.arg1, inst.arg2};
        selected.addressingModes = {"reg", "reg", "reg"};
        selected.cost = 5;
        return selected;
    }
    
    SelectedInstruction selectLoadInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "LOAD";
        selected.template = instructionTemplates["LOAD"];
        selected.operands = {inst.result, inst.arg1, inst.arg2};
        selected.addressingModes = {"reg", "imm", "reg"};
        selected.cost = 2;
        return selected;
    }
    
    SelectedInstruction selectStoreInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "STORE";
        selected.template = instructionTemplates["STORE"];
        selected.operands = {inst.arg1, inst.arg2, inst.result};
        selected.addressingModes = {"reg", "imm", "reg"};
        selected.cost = 2;
        return selected;
    }
    
    SelectedInstruction selectBranchInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "BRANCH";
        selected.template = instructionTemplates["BRANCH"];
        selected.operands = {inst.arg1, inst.result};
        selected.addressingModes = {"cond", "label"};
        selected.cost = 1;
        return selected;
    }
    
    SelectedInstruction selectCallInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "CALL";
        selected.template = instructionTemplates["CALL"];
        selected.operands = {inst.result};
        selected.addressingModes = {"label"};
        selected.cost = 2;
        return selected;
    }
    
    SelectedInstruction selectReturnInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "RETURN";
        selected.template = instructionTemplates["RETURN"];
        selected.operands = {};
        selected.addressingModes = {};
        selected.cost = 1;
        return selected;
    }
    
    SelectedInstruction selectDefaultInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "UNKNOWN";
        selected.template = InstructionTemplate("", {}, {});
        selected.operands = {};
        selected.addressingModes = {};
        selected.cost = 1;
        return selected;
    }
    
    map<string, string> allocateRegisters(vector<SelectedInstruction>& instructions) {
        map<string, string> registerAllocation;
        
        // 这里需要实现寄存器分配逻辑
        // 简化实现：为每个变量分配一个寄存器
        
        set<string> variables;
        for (auto& inst : instructions) {
            for (auto& operand : inst.operands) {
                if (isVariable(operand)) {
                    variables.insert(operand);
                }
            }
        }
        
        int regCount = 0;
        for (auto& var : variables) {
            if (regCount < 16) {
                registerAllocation[var] = "r" + to_string(regCount);
                regCount++;
            } else {
                registerAllocation[var] = "spill";
            }
        }
        
        return registerAllocation;
    }
    
    bool isVariable(string operand) {
        // 检查操作数是否是变量
        return operand.find("var_") == 0 || operand.find("temp_") == 0;
    }
    
    vector<SelectedInstruction> scheduleInstructions(vector<SelectedInstruction>& instructions) {
        vector<SelectedInstruction> scheduledInstructions;
        
        // 这里需要实现指令调度逻辑
        // 简化实现：保持原有顺序
        
        scheduledInstructions = instructions;
        
        return scheduledInstructions;
    }
    
    vector<MachineInstruction> generateMachineCode(vector<SelectedInstruction>& instructions, 
                                                   map<string, string>& registerAllocation) {
        vector<MachineInstruction> machineCode;
        
        for (auto& inst : instructions) {
            MachineInstruction machineInst = generateMachineInstruction(inst, registerAllocation);
            machineCode.push_back(machineInst);
        }
        
        return machineCode;
    }
    
    MachineInstruction generateMachineInstruction(SelectedInstruction& inst, 
                                                 map<string, string>& registerAllocation) {
        MachineInstruction machineInst;
        
        // 生成机器指令
        string assembly = inst.template.template;
        
        // 替换占位符
        for (int i = 0; i < inst.operands.size(); i++) {
            string placeholder = "%{" + inst.template.placeholders[i] + "}";
            string value = inst.operands[i];
            
            // 如果是变量，使用分配的寄存器
            if (registerAllocation.find(value) != registerAllocation.end()) {
                value = registerAllocation[value];
            }
            
            assembly = replacePlaceholder(assembly, placeholder, value);
        }
        
        machineInst.assembly = assembly;
        machineInst.opcode = inst.opcode;
        machineInst.operands = inst.operands;
        machineInst.cost = inst.cost;
        
        return machineInst;
    }
    
    string replacePlaceholder(string template, string placeholder, string value) {
        size_t pos = template.find(placeholder);
        if (pos != string::npos) {
            template.replace(pos, placeholder.length(), value);
        }
        return template;
    }
};

class InstructionTemplate {

};
```

```c
class CodeGenerationOptimizer {
private:
    TargetMachine* targetMachine;
    OptimizationLevel optimizationLevel;
    map<string, OptimizationPass*> optimizationPasses;
    
public:
    CodeGenerationOptimizer(TargetMachine* machine, OptimizationLevel level) 
        : targetMachine(machine), optimizationLevel(level) {
        initializeOptimizationPasses();
    }
    
    void initializeOptimizationPasses() {
        // 初始化优化通道
        optimizationPasses["INSTRUCTION_SELECTION"] = new InstructionSelectionOptimizer();
        optimizationPasses["REGISTER_ALLOCATION"] = new RegisterAllocationOptimizer();
        optimizationPasses["INSTRUCTION_SCHEDULING"] = new InstructionSchedulingOptimizer();
        optimizationPasses["PEEPHOLE_OPTIMIZATION"] = new PeepholeOptimizer();
        optimizationPasses["MACHINE_SPECIFIC"] = new MachineSpecificOptimizer();
    }
    
    vector<MachineInstruction> optimizeCode(vector<MachineInstruction>& instructions) {
        vector<MachineInstruction> optimizedInstructions = instructions;
        
        // 应用优化通道
        for (auto& pass : optimizationPasses) {
            if (shouldApplyPass(pass.first)) {
                optimizedInstructions = pass.second->optimize(optimizedInstructions);
            }
        }
        
        return optimizedInstructions;
    }
    
    bool shouldApplyPass(string passName) {
        // 根据优化级别决定是否应用优化通道
        switch (optimizationLevel) {
            case NO_OPTIMIZATION:
                return false;
            case BASIC_OPTIMIZATION:
                return passName == "INSTRUCTION_SELECTION" || 
                       passName == "REGISTER_ALLOCATION";
            case ADVANCED_OPTIMIZATION:
                return passName == "INSTRUCTION_SELECTION" || 
                       passName == "REGISTER_ALLOCATION" ||
                       passName == "INSTRUCTION_SCHEDULING";
            case AGGRESSIVE_OPTIMIZATION:
                return true;
            default:
                return false;
        }
    }
    
    void cleanup() {
        for (auto& pass : optimizationPasses) {
            delete pass.second;
        }
        optimizationPasses.clear();
    }
    
    ~CodeGenerationOptimizer() {
        cleanup();
    }
};
```

### 代码生成器错误处理器
```c
class CodeGenerationErrorHandler {
private:
    vector<CodeGenerationError> errors;
    vector<CodeGenerationWarning> warnings;
    
public:
    void addError(CodeGenerationError error) {
        errors.push_back(error);
    }
    
    void addWarning(CodeGenerationWarning warning) {
        warnings.push_back(warning);
    }
    
    void handleInstructionSelectionError(IntermediateInstruction& inst, string reason) {
        CodeGenerationError error;
        error.type = "INSTRUCTION_SELECTION_ERROR";
        error.message = "无法为指令选择目标指令: " + inst.op + " - " + reason;
        error.instruction = inst;
        error.severity = "ERROR";
        
        addError(error);
    }
    
    void handleRegisterAllocationError(string variable, string reason) {
        CodeGenerationError error;
        error.type = "REGISTER_ALLOCATION_ERROR";
        error.message = "无法为变量分配寄存器: " + variable + " - " + reason;
        error.variable = variable;
        error.severity = "ERROR";
        
        addError(error);
    }
    
    void handleInstructionSchedulingError(SelectedInstruction& inst, string reason) {
        CodeGenerationError error;
        error.type = "INSTRUCTION_SCHEDULING_ERROR";
        error.message = "无法调度指令: " + inst.opcode + " - " + reason;
        error.instruction = inst;
        error.severity = "ERROR";
        
        addError(error);
    }
    
    void handleMachineCodeGenerationError(SelectedInstruction& inst, string reason) {
        CodeGenerationError error;
        error.type = "MACHINE_CODE_GENERATION_ERROR";
        error.message = "无法生成机器代码: " + inst.opcode + " - " + reason;
        error.instruction = inst;
        error.severity = "ERROR";
        
        addError(error);
    }
    
    void handleOptimizationWarning(string optimization, string reason) {
        CodeGenerationWarning warning;
        warning.type = "OPTIMIZATION_WARNING";
        warning.message = "优化警告: " + optimization + " - " + reason;
        warning.optimization = optimization;
        warning.severity = "WARNING";
        
        addWarning(warning);
    }
    
    bool hasErrors() {
        return !errors.empty();
    }
    
    bool hasWarnings() {
        return !warnings.empty();
    }
    
    void printErrors() {
        if (hasErrors()) {
            cout << "代码生成错误:" << endl;
            cout << "===========================================" << endl;
            
            for (auto& error : errors) {
                cout << "错误类型: " << error.type << endl;
                cout << "错误信息: " << error.message << endl;
                cout << "严重程度: " << error.severity << endl;
                cout << endl;
            }
        }
    }
    
    void printWarnings() {
        if (hasWarnings()) {
            cout << "代码生成警告:" << endl;
            cout << "===========================================" << endl;
            
            for (auto& warning : warnings) {
                cout << "警告类型: " << warning.type << endl;
                cout << "警告信息: " << warning.message << endl;
                cout << "严重程度: " << warning.severity << endl;
                cout << endl;
            }
        }
    }
    
    void clear() {
        errors.clear();
        warnings.clear();
    }
};

class CodeGenerationError {
public:
    string type;
    string message;
    string severity;
    IntermediateInstruction instruction;
    SelectedInstruction selectedInstruction;
    string variable;
};

class CodeGenerationWarning {
```

```c
        optimizers[ARM64] = new CodeGenerationOptimizer(new ARM64TargetMachine(), ADVANCED_OPTIMIZATION);
        optimizers[MIPS] = new CodeGenerationOptimizer(new MIPSTargetMachine(), ADVANCED_OPTIMIZATION);
        optimizers[RISC_V] = new CodeGenerationOptimizer(new RISC_VTargetMachine(), ADVANCED_OPTIMIZATION);
        optimizers[POWERPC] = new CodeGenerationOptimizer(new PowerPCTargetMachine(), ADVANCED_OPTIMIZATION);
        optimizers[SPARC] = new CodeGenerationOptimizer(new SPARCTargetMachine(), ADVANCED_OPTIMIZATION);
    }
    
    void initializeErrorHandlers() {
        // 初始化各种目标机器的错误处理器
        errorHandlers[X86_64] = new CodeGenerationErrorHandler();
        errorHandlers[ARM64] = new CodeGenerationErrorHandler();
        errorHandlers[MIPS] = new CodeGenerationErrorHandler();
        errorHandlers[RISC_V] = new CodeGenerationErrorHandler();
        errorHandlers[POWERPC] = new CodeGenerationErrorHandler();
        errorHandlers[SPARC] = new CodeGenerationErrorHandler();
    }
    
    vector<MachineInstruction> generateCode(TargetMachineType targetType, IntermediateCode* intermediateCode) {
        if (generators.find(targetType) == generators.end()) {
            cout << "不支持的目标机器类型: " << getTargetMachineName(targetType) << endl;
            return vector<MachineInstruction>();
        }
        
        CodeGenerator* generator = generators[targetType];
        CodeGenerationOptimizer* optimizer = optimizers[targetType];
        CodeGenerationErrorHandler* errorHandler = errorHandlers[targetType];
        
        // 生成代码
        vector<MachineInstruction> machineCode = generator->generateCode(intermediateCode);
        
        // 优化代码
        machineCode = optimizer->optimizeCode(machineCode);
        
        // 检查错误
        if (errorHandler->hasErrors()) {
            errorHandler->printErrors();
            return vector<MachineInstruction>();
        }
        
        // 打印警告
        if (errorHandler->hasWarnings()) {
            errorHandler->printWarnings();
        }
        
        return machineCode;
    }
    
    void generateCodeForAllTargets(IntermediateCode* intermediateCode) {
        cout << "为所有目标机器生成代码" << endl;
        cout << "===========================================" << endl;
        
        for (auto& pair : generators) {
            TargetMachineType targetType = pair.first;
            CodeGenerator* generator = pair.second;
            
            cout << "目标机器: " << getTargetMachineName(targetType) << endl;
            
            vector<MachineInstruction> machineCode = generator->generateCode(intermediateCode);
            
            cout << "生成指令数: " << machineCode.size() << endl;
            cout << "代码大小: " << calculateCodeSize(machineCode) << " 字节" << endl;
            cout << endl;
        }
    }
    
    string getTargetMachineName(TargetMachineType type) {
        switch (type) {
            case X86_64: return "x86-64";
            case ARM64: return "ARM64";
            case MIPS: return "MIPS";
            case RISC_V: return "RISC-V";
            case POWERPC: return "PowerPC";
            case SPARC: return "SPARC";
            default: return "未知";
        }
    }
    
    int calculateCodeSize(vector<MachineInstruction>& instructions) {
        int size = 0;
        
        for (auto& inst : instructions) {
            size += inst.assembly.length();
        }
        
        return size;
    }
    
    void cleanup() {
        for (auto& pair : generators) {
            delete pair.second;
        }
        generators.clear();
        
        for (auto& pair : optimizers) {
            delete pair.second;
        }
        optimizers.clear();
        
        for (auto& pair : errorHandlers) {
            delete pair.second;
        }
        errorHandlers.clear();
    }
    
    ~CodeGenerationManager() {
        cleanup();
    }
};
```
