# 面向对象编译

## 🎯 面向对象编译概述

**面向对象编译**是编译原理在面向对象编程语言中的应用，涉及类、对象、继承、多态、封装等面向对象特性的编译处理。面向对象编译需要处理复杂的类型系统、虚函数表、对象布局等高级概念。

## 🔍 基本概念

### 面向对象编译定义
面向对象编译是指将面向对象编程语言的源代码编译为目标代码的过程，需要处理面向对象语言的特有语法和语义。

### 面向对象编译特点
- **类型系统复杂**：需要处理复杂的类型系统
- **继承关系处理**：需要处理类继承关系
- **多态支持**：需要支持运行时多态
- **对象模型**：需要实现对象的内存模型

```c
        // 这里需要实现成员函数分析逻辑

        MemberFunctionInfo* func2 = new MemberFunctionInfo("setValue", "void", "public", false);

        memberFunctions.push_back(func1);
        memberFunctions.push_back(func2);
        memberFunctions.push_back(func3);
        
        classInfo->setMemberFunctions(memberFunctions);
    }
    
    void analyzeInheritance(string classDefinition, ClassInfo* classInfo) {
        // 分析继承关系
        vector<string> baseClasses;
        
        // 这里需要实现继承关系分析逻辑
        // 简化实现：添加示例基类
        
        baseClasses.push_back("BaseClass");
        
        classInfo->setBaseClasses(baseClasses);
        
        // 创建继承信息
        InheritanceInfo* inheritanceInfo = new InheritanceInfo(classInfo->getName(), baseClasses);
        inheritance[classInfo->getName()] = inheritanceInfo;
    }
    
    void analyzeVirtualFunctions(string classDefinition, ClassInfo* classInfo) {
        // 分析虚函数
        vector<VirtualFunctionInfo*> virtualFunctions;
        
        // 这里需要实现虚函数分析逻辑
        // 简化实现：添加示例虚函数
        
        VirtualFunctionInfo* vfunc1 = new VirtualFunctionInfo("calculate", "int", 0);
        VirtualFunctionInfo* vfunc2 = new VirtualFunctionInfo("display", "void", 1);
        
        virtualFunctions.push_back(vfunc1);
        virtualFunctions.push_back(vfunc2);
        
        classInfo->setVirtualFunctions(virtualFunctions);
        
        // 存储虚函数信息
        for (auto& vfunc : virtualFunctions) {
            this->virtualFunctions[vfunc->getName()] = vfunc;
        }
    }
    
    ClassInfo* getClassInfo(string className) {
        if (classes.find(className) != classes.end()) {
            return classes[className];
        }
        return nullptr;
    }
    
    vector<ClassInfo*> getAllClasses() {
        vector<ClassInfo*> allClasses;
        for (auto& pair : classes) {
            allClasses.push_back(pair.second);
        }
        return allClasses;
    }
    
    void cleanup() {
        for (auto& pair : classes) {
            delete pair.second;
        }
        classes.clear();
        
        for (auto& pair : interfaces) {
            delete pair.second;
        }
        interfaces.clear();
        
        for (auto& pair : inheritance) {
            delete pair.second;
        }
        inheritance.clear();
        
        for (auto& pair : virtualFunctions) {
            delete pair.second;
        }
        virtualFunctions.clear();
    }
    
    ~ClassAnalyzer() {
        cleanup();
    }
};
```

## 🔧 虚函数表生成器

### 虚函数表生成器实现
```c
class VTableGenerator {
private:
    map<string, VTableInfo*> vtables;
    map<string, ClassInfo*> classes;
    
public:
    VTableGenerator(map<string, ClassInfo*> classMap) : classes(classMap) {
        initializeVTableGenerator();
    }
    
    void initializeVTableGenerator() {
        cout << "初始化虚函数表生成器" << endl;
    }
    
    void generateVTables() {
        cout << "开始生成虚函数表" << endl;
        cout << "===========================================" << endl;
        
        for (auto& pair : classes) {
            string className = pair.first;
            ClassInfo* classInfo = pair.second;
            
            VTableInfo* vtable = generateVTableForClass(classInfo);
            vtables[className] = vtable;
            
            cout << "为类 " << className << " 生成虚函数表" << endl;
        }
        
        cout << "虚函数表生成完成" << endl;
    }
    
    VTableInfo* generateVTableForClass(ClassInfo* classInfo) {
        VTableInfo* vtable = new VTableInfo(classInfo->getName());
        
        // 1. 继承基类的虚函数表
        inheritBaseClassVTables(classInfo, vtable);
        
        // 2. 添加当前类的虚函数
        addClassVirtualFunctions(classInfo, vtable);
        
        // 3. 生成虚函数表条目
        generateVTableEntries(vtable);
        
        return vtable;
    }
    
    void inheritBaseClassVTables(ClassInfo* classInfo, VTableInfo* vtable) {
        vector<string> baseClasses = classInfo->getBaseClasses();
        
        for (auto& baseClassName : baseClasses) {
            if (vtables.find(baseClassName) != vtables.end()) {
                VTableInfo* baseVTable = vtables[baseClassName];
                
                // 继承基类虚函数表
                vector<VTableEntry*> baseEntries = baseVTable->getEntries();
                for (auto& entry : baseEntries) {
                    vtable->addEntry(entry);
                }
            }
        }
    }
    
    void addClassVirtualFunctions(ClassInfo* classInfo, VTableInfo* vtable) {
        vector<VirtualFunctionInfo*> virtualFunctions = classInfo->getVirtualFunctions();
        
        for (auto& vfunc : virtualFunctions) {
            VTableEntry* entry = new VTableEntry(vfunc->getName(), vfunc->getReturnType(), vfunc->getVtableIndex());
            vtable->addEntry(entry);
        }
    }
    
    void generateVTableEntries(VTableInfo* vtable) {
        vector<VTableEntry*> entries = vtable->getEntries();
        
        cout << "虚函数表 " << vtable->getClassName() << ":" << endl;
        for (int i = 0; i < entries.size(); i++) {
            VTableEntry* entry = entries[i];
            cout << "  [" << i << "] " << entry->getFunctionName() << " -> " << entry->getReturnType() << endl;
        }
        cout << endl;
    }
    
    VTableInfo* getVTable(string className) {
        if (vtables.find(className) != vtables.end()) {
            return vtables[className];
        }
        return nullptr;
    }
    
    void cleanup() {
        for (auto& pair : vtables) {
            delete pair.second;
        }
        vtables.clear();
    }
    
    ~VTableGenerator() {
        cleanup();
    }
};

class VTableInfo {
private:
    string className;
    vector<VTableEntry*> entries;
    
public:
    VTableInfo(string name) : className(name) {}
    
    string getClassName() { return className; }
    
    void addEntry(VTableEntry* entry) {
        entries.push_back(entry);
    }
    
    vector<VTableEntry*> getEntries() { return entries; }
    
    int getSize() { return entries.size(); }
};

class VTableEntry {
private:
    string functionName;
    string returnType;
    int vtableIndex;
    
public:
    VTableEntry(string name, string ret, int index) 
        : functionName(name), returnType(ret), vtableIndex(index) {}
    
    string getFunctionName() { return functionName; }
    string getReturnType() { return returnType; }
    int getVtableIndex() { return vtableIndex; }
};
```

### 对象布局生成器实现
```c
class ObjectLayoutGenerator {
private:
    map<string, ObjectLayoutInfo*> objectLayouts;
    map<string, ClassInfo*> classes;
    
public:
    ObjectLayoutGenerator(map<string, ClassInfo*> classMap) : classes(classMap) {
        initializeObjectLayoutGenerator();
    }
    
    void initializeObjectLayoutGenerator() {
        cout << "初始化对象布局生成器" << endl;
    }
    
    void generateObjectLayouts() {
        cout << "开始生成对象布局" << endl;
        cout << "===========================================" << endl;
        
        for (auto& pair : classes) {
            string className = pair.first;
            ClassInfo* classInfo = pair.second;
            
            ObjectLayoutInfo* layout = generateObjectLayoutForClass(classInfo);
            objectLayouts[className] = layout;
            
            cout << "为类 " << className << " 生成对象布局" << endl;
        }
        
        cout << "对象布局生成完成" << endl;
    }
    
    ObjectLayoutInfo* generateObjectLayoutForClass(ClassInfo* classInfo) {
        ObjectLayoutInfo* layout = new ObjectLayoutInfo(classInfo->getName());
        
        // 1. 添加虚函数表指针
        addVTablePointer(layout);
        
        // 2. 添加基类成员
        addBaseClassMembers(classInfo, layout);
        
        // 3. 添加当前类成员
        addClassMembers(classInfo, layout);
        
        // 4. 计算布局信息
        calculateLayoutInfo(layout);
        
        return layout;
    }
    
    void addVTablePointer(ObjectLayoutInfo* layout) {
        LayoutEntry* vtablePtr = new LayoutEntry("vtable_ptr", "void*", 0, 8);
        layout->addEntry(vtablePtr);
    }
    
    void addBaseClassMembers(ClassInfo* classInfo, ObjectLayoutInfo* layout) {
        vector<string> baseClasses = classInfo->getBaseClasses();
        
        for (auto& baseClassName : baseClasses) {
            if (objectLayouts.find(baseClassName) != objectLayouts.end()) {
                ObjectLayoutInfo* baseLayout = objectLayouts[baseClassName];
                
                // 添加基类成员
                vector<LayoutEntry*> baseEntries = baseLayout->getEntries();
                for (auto& entry : baseEntries) {
                    if (entry->getName() != "vtable_ptr") {
                        layout->addEntry(entry);
                    }
                }
            }
        }
    }
    
    void addClassMembers(ClassInfo* classInfo, ObjectLayoutInfo* layout) {
        vector<MemberVariableInfo*> memberVariables = classInfo->getMemberVariables();
        
        for (auto& var : memberVariables) {
            int size = getTypeSize(var->getType());
            int offset = layout->getCurrentOffset();
            
            LayoutEntry* entry = new LayoutEntry(var->getName(), var->getType(), offset, size);
            layout->addEntry(entry);
        }
    }
    
    void calculateLayoutInfo(ObjectLayoutInfo* layout) {
        vector<LayoutEntry*> entries = layout->getEntries();
        
        int totalSize = 0;
        for (auto& entry : entries) {
            totalSize += entry->getSize();
        }
        
        layout->setTotalSize(totalSize);
        layout->setAlignment(calculateAlignment(entries));
    }
    
    int getTypeSize(string type) {
        // 获取类型大小
        if (type == "int") return 4;
        if (type == "long") return 8;
        if (type == "float") return 4;
        if (type == "double") return 8;
        if (type == "char") return 1;
        if (type == "bool") return 1;
        if (type == "void*") return 8;
        if (type == "string") return 8;
        
        return 4; // 默认大小
    }
    
    int calculateAlignment(vector<LayoutEntry*> entries) {
        int maxAlignment = 1;
        
        for (auto& entry : entries) {
            int alignment = getTypeAlignment(entry->getType());
            maxAlignment = max(maxAlignment, alignment);
        }
        
        return maxAlignment;
    }
    
    int getTypeAlignment(string type) {
        // 获取类型对齐
        if (type == "int") return 4;
        if (type == "long") return 8;
        if (type == "float") return 4;
        if (type == "double") return 8;
        if (type == "char") return 1;
        if (type == "bool") return 1;
        if (type == "void*") return 8;
        if (type == "string") return 8;
        
        return 4; // 默认对齐
    }
    
    ObjectLayoutInfo* getObjectLayout(string className) {
        if (objectLayouts.find(className) != objectLayouts.end()) {
            return objectLayouts[className];
        }
        return nullptr;
    }
    
    void cleanup() {
        for (auto& pair : objectLayouts) {
            delete pair.second;
        }
        objectLayouts.clear();
    }
    
    ~ObjectLayoutGenerator() {
        cleanup();
    }
};

class ObjectLayoutInfo {
private:
    string className;
    vector<LayoutEntry*> entries;
    int totalSize;
    int alignment;
    
public:
    ObjectLayoutInfo(string name) : className(name), totalSize(0), alignment(1) {}
    
    string getClassName() { return className; }
    
    void addEntry(LayoutEntry* entry) {
        entries.push_back(entry);
    }
    
    vector<LayoutEntry*> getEntries() { return entries; }
    
    int getCurrentOffset() {
        int offset = 0;
        for (auto& entry : entries) {
            offset += entry->getSize();
        }
        return offset;
    }
    
    void setTotalSize(int size) { totalSize = size; }
    int getTotalSize() { return totalSize; }
    
    void setAlignment(int align) { alignment = align; }
    int getAlignment() { return alignment; }
};

class LayoutEntry {
private:
    string name;
    string type;
    int offset;
    int size;
    
public:
    LayoutEntry(string n, string t, int off, int s) 
        : name(n), type(t), offset(off), size(s) {}
    
    string getName() { return name; }
    string getType() { return type; }
    int getOffset() { return offset; }
    int getSize() { return size; }
};
```
