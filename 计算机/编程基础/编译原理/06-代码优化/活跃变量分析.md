# 活跃变量分析

## 🎯 活跃变量分析概述

**活跃变量分析**是数据流分析的重要类型，用于确定程序中每个点处哪些变量是活跃的（即可能在后续执行中被使用）。活跃变量分析是死代码消除、寄存器分配等优化技术的基础。

## 🔍 基本概念

### 活跃变量定义
一个变量在程序点p处是活跃的，当且仅当存在从p开始的执行路径，在该路径上该变量被使用，且在该使用之前没有被重新定义。

### 活跃变量分析特点
- **后向分析**：从程序出口向入口分析
- **可能分析**：提供保守的近似结果
- **迭代计算**：通过迭代计算收敛到不动点
- **优化基础**：为多种优化技术提供信息

## 🔧 活跃变量分析实现

### 基本活跃变量分析
```c
class LiveVariableAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, set<string>> liveIn;
    map<BasicBlock*, set<string>> liveOut;
    map<BasicBlock*, set<string>> useSets;
    map<BasicBlock*, set<string>> defSets;
    
public:
    LiveVariableAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 计算使用集和定义集
        computeUseDefSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeUseDefSets() {
        for (auto block : cfg->getBlocks()) {
            set<string> use;
            set<string> def;
            
            // 从后往前遍历指令
            for (int i = block->instructions.size() - 1; i >= 0; i--) {
                Instruction& inst = block->instructions[i];
                
                if (inst.type == ASSIGNMENT) {
                    // 添加到定义集
                    def.insert(inst.result);
                    
                    // 添加到使用集（如果变量在使用前未定义）
                    if (def.find(inst.arg1) == def.end()) {
                        use.insert(inst.arg1);
                    }
                    if (def.find(inst.arg2) == def.end()) {
                        use.insert(inst.arg2);
                    }
                } else if (inst.type == FUNCTION_CALL) {
                    // 函数调用可能使用全局变量
                    addGlobalVariables(inst, use);
                } else if (inst.type == RETURN) {
                    // 返回语句使用返回值
                    if (!inst.arg1.empty()) {
                        use.insert(inst.arg1);
                    }
                }
            }
            
            useSets[block] = use;
            defSets[block] = def;
        }
    }
    
    void addGlobalVariables(Instruction& inst, set<string>& use) {
        // 添加可能被函数调用使用的全局变量
        // 这里需要根据具体的函数调用语义来实现
    }
    
    void iterativeAnalysis() {
        // 初始化
        initialize();
        
        // 迭代计算
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 1000;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                set<string> oldLiveOut = liveOut[block];
                set<string> newLiveOut = computeLiveOut(block);
                
                if (oldLiveOut != newLiveOut) {
                    liveOut[block] = newLiveOut;
                    liveIn[block] = computeLiveIn(block);
                    changed = true;
                }
            }
        }
    }
    
    void initialize() {
        for (auto block : cfg->getBlocks()) {
            liveIn[block] = set<string>();
            liveOut[block] = set<string>();
        }
    }
    
    set<string> computeLiveOut(BasicBlock* block) {
        set<string> liveOut;
        
        // 合并所有后继块的liveIn
        for (auto succ : block->successors) {
            for (auto var : liveIn[succ]) {
                liveOut.insert(var);
            }
        }
        
        return liveOut;
    }
    
    set<string> computeLiveIn(BasicBlock* block) {
        // IN[B] = USE[B] ∪ (OUT[B] - DEF[B])
        set<string> liveIn = liveOut[block];
        
        // 减去定义集
        for (auto var : defSets[block]) {
            liveIn.erase(var);
        }
        
        // 加上使用集
        for (auto var : useSets[block]) {
            liveIn.insert(var);
        }
        
        return liveIn;
    }
    
    set<string> getLiveIn(BasicBlock* block) {
        return liveIn[block];
    }
    
    set<string> getLiveOut(BasicBlock* block) {
        return liveOut[block];
    }
    
    set<string> getUseSet(BasicBlock* block) {
        return useSets[block];
    }
    
    set<string> getDefSet(BasicBlock* block) {
        return defSets[block];
    }
};
```

## 🔧 高级活跃变量分析

### 精确活跃变量分析
```c
class PreciseLiveVariableAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, map<int, set<string>>> liveIn;
    map<BasicBlock*, map<int, set<string>>> liveOut;
    map<BasicBlock*, map<int, set<string>>> useSets;
    map<BasicBlock*, map<int, set<string>>> defSets;
    
public:
    PreciseLiveVariableAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 计算使用集和定义集
        computeUseDefSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeUseDefSets() {
        for (auto block : cfg->getBlocks()) {
            map<int, set<string>> use;
            map<int, set<string>> def;
            
            // 为每个指令位置计算使用集和定义集
            for (int i = 0; i < block->instructions.size(); i++) {
                Instruction& inst = block->instructions[i];
                
                if (inst.type == ASSIGNMENT) {
                    // 添加到定义集
                    def[i].insert(inst.result);
                    
                    // 添加到使用集（如果变量在使用前未定义）
                    if (def[i].find(inst.arg1) == def[i].end()) {
                        use[i].insert(inst.arg1);
                    }
                    if (def[i].find(inst.arg2) == def[i].end()) {
                        use[i].insert(inst.arg2);
                    }
                }
            }
            
            useSets[block] = use;
            defSets[block] = def;
        }
    }
    
    void iterativeAnalysis() {
        // 初始化
        initialize();
        
        // 迭代计算
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 1000;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                map<int, set<string>> oldLiveOut = liveOut[block];
                map<int, set<string>> newLiveOut = computeLiveOut(block);
                
                if (oldLiveOut != newLiveOut) {
                    liveOut[block] = newLiveOut;
                    liveIn[block] = computeLiveIn(block);
                    changed = true;
                }
            }
        }
    }
    
    void initialize() {
        for (auto block : cfg->getBlocks()) {
            liveIn[block] = map<int, set<string>>();
            liveOut[block] = map<int, set<string>>();
        }
    }
    
    map<int, set<string>> computeLiveOut(BasicBlock* block) {
        map<int, set<string>> liveOut;
        
        // 为每个指令位置计算liveOut
        for (int i = 0; i < block->instructions.size(); i++) {
            set<string> liveOutAtI;
            
            if (i == block->instructions.size() - 1) {
                // 最后一条指令的liveOut是后继块的liveIn
                for (auto succ : block->successors) {
                    for (auto var : liveIn[succ][0]) {
                        liveOutAtI.insert(var);
                    }
                }
            } else {
                // 其他指令的liveOut是下一条指令的liveIn
                liveOutAtI = liveIn[block][i + 1];
            }
            
            liveOut[i] = liveOutAtI;
        }
        
        return liveOut;
    }
    
    map<int, set<string>> computeLiveIn(BasicBlock* block) {
        map<int, set<string>> liveIn;
        
        // 为每个指令位置计算liveIn
        for (int i = 0; i < block->instructions.size(); i++) {
            set<string> liveInAtI = liveOut[block][i];
            
            // 减去定义集
            for (auto var : defSets[block][i]) {
                liveInAtI.erase(var);
            }
            
            // 加上使用集
            for (auto var : useSets[block][i]) {
                liveInAtI.insert(var);
            }
            
            liveIn[i] = liveInAtI;
        }
        
        return liveIn;
    }
    
    set<string> getLiveIn(BasicBlock* block, int position) {
        return liveIn[block][position];
    }
    
    set<string> getLiveOut(BasicBlock* block, int position) {
        return liveOut[block][position];
    }
};
```

## 🔧 寄存器活跃变量分析

### 寄存器活跃变量分析
```c
class RegisterLiveVariableAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, set<string>> liveIn;
    map<BasicBlock*, set<string>> liveOut;
    map<string, int> variableToRegister;
    map<int, set<string>> registerToVariables;
    
public:
    RegisterLiveVariableAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 执行基本活跃变量分析
        LiveVariableAnalysis basicAnalysis(cfg);
        basicAnalysis.analyze();
        
        // 获取基本分析结果
        liveIn = basicAnalysis.getLiveIn();
        liveOut = basicAnalysis.getLiveOut();
        
        // 计算寄存器活跃变量
        computeRegisterLiveVariables();
    }
    
    void computeRegisterLiveVariables() {
        // 为每个基本块计算寄存器活跃变量
        for (auto block : cfg->getBlocks()) {
            set<string> liveVars = liveIn[block];
            
            // 计算每个寄存器的活跃变量
            for (auto var : liveVars) {
                if (variableToRegister.find(var) != variableToRegister.end()) {
                    int reg = variableToRegister[var];
                    registerToVariables[reg].insert(var);
                }
            }
        }
    }
    
    void setVariableToRegister(string var, int reg) {
        variableToRegister[var] = reg;
    }
    
    int getVariableRegister(string var) {
        if (variableToRegister.find(var) != variableToRegister.end()) {
            return variableToRegister[var];
        }
        return -1; // 表示变量未分配寄存器
    }
    
    set<string> getRegisterVariables(int reg) {
        if (registerToVariables.find(reg) != registerToVariables.end()) {
            return registerToVariables[reg];
        }
        return set<string>();
    }
    
    bool isRegisterLive(int reg, BasicBlock* block) {
        set<string> regVars = getRegisterVariables(reg);
        set<string> liveVars = liveIn[block];
        
        for (auto var : regVars) {
            if (liveVars.find(var) != liveVars.end()) {
                return true;
            }
        }
        
        return false;
    }
    
    set<int> getLiveRegisters(BasicBlock* block) {
        set<int> liveRegs;
        
        for (auto var : liveIn[block]) {
            if (variableToRegister.find(var) != variableToRegister.end()) {
                int reg = variableToRegister[var];
                liveRegs.insert(reg);
            }
        }
        
        return liveRegs;
    }
};
```

```c
class MemoryLiveVariableAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, set<string>> liveIn;
    map<BasicBlock*, set<string>> liveOut;
    map<string, MemoryLocation> variableToMemory;
    map<MemoryLocation, set<string>> memoryToVariables;
    
public:
    MemoryLiveVariableAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 执行基本活跃变量分析
        LiveVariableAnalysis basicAnalysis(cfg);
        basicAnalysis.analyze();
        
        // 获取基本分析结果
        liveIn = basicAnalysis.getLiveIn();
        liveOut = basicAnalysis.getLiveOut();
        
        // 计算内存活跃变量
        computeMemoryLiveVariables();
    }
    
    void computeMemoryLiveVariables() {
        // 为每个基本块计算内存活跃变量
        for (auto block : cfg->getBlocks()) {
            set<string> liveVars = liveIn[block];
            
            // 计算每个内存位置的活跃变量
            for (auto var : liveVars) {
                if (variableToMemory.find(var) != variableToMemory.end()) {
                    MemoryLocation mem = variableToMemory[var];
                    memoryToVariables[mem].insert(var);
                }
            }
        }
    }
    
    void setVariableToMemory(string var, MemoryLocation mem) {
        variableToMemory[var] = mem;
    }
    
    MemoryLocation getVariableMemory(string var) {
        if (variableToMemory.find(var) != variableToMemory.end()) {
            return variableToMemory[var];
        }
        return MemoryLocation(); // 表示变量未分配内存
    }
    
    set<string> getMemoryVariables(MemoryLocation mem) {
        if (memoryToVariables.find(mem) != memoryToVariables.end()) {
            return memoryToVariables[mem];
        }
        return set<string>();
    }
    
    bool isMemoryLive(MemoryLocation mem, BasicBlock* block) {
        set<string> memVars = getMemoryVariables(mem);
        set<string> liveVars = liveIn[block];
        
        for (auto var : memVars) {
            if (liveVars.find(var) != liveVars.end()) {
                return true;
            }
        }
        
        return false;
    }
    
    set<MemoryLocation> getLiveMemoryLocations(BasicBlock* block) {
        set<MemoryLocation> liveMem;
        
        for (auto var : liveIn[block]) {
            if (variableToMemory.find(var) != variableToMemory.end()) {
                MemoryLocation mem = variableToMemory[var];
                liveMem.insert(mem);
            }
        }
        
        return liveMem;
    }
};

struct MemoryLocation {
    int baseAddress;
    int offset;
    int size;
    
    MemoryLocation(int base = 0, int off = 0, int sz = 0) 
        : baseAddress(base), offset(off), size(sz) {}
    
    bool operator<(const MemoryLocation& other) const {
        if (baseAddress != other.baseAddress) {
            return baseAddress < other.baseAddress;
        }
        if (offset != other.offset) {
            return offset < other.offset;
        }
        return size < other.size;
    }
    
    bool operator==(const MemoryLocation& other) const {
        return baseAddress == other.baseAddress && 
               offset == other.offset && 
               size == other.size;
    }
};
```

### 死代码消除
```c
class DeadCodeEliminationUsingLiveness {
private:
    ControlFlowGraph* cfg;
    LiveVariableAnalysis* livenessAnalysis;
    
public:
    DeadCodeEliminationUsingLiveness(ControlFlowGraph* cfg) : cfg(cfg) {
        livenessAnalysis = new LiveVariableAnalysis(cfg);
    }
    
    void eliminateDeadCode() {
        // 执行活跃变量分析
        livenessAnalysis->analyze();
        
        // 消除死代码
        for (auto block : cfg->getBlocks()) {
            eliminateDeadCodeInBlock(block);
        }
    }
    
    void eliminateDeadCodeInBlock(BasicBlock* block) {
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                // 检查变量是否活跃
                if (isVariableLive(inst.result, block)) {
                    optimizedInstructions.push_back(inst);
                }
            } else {
                // 非赋值指令都保留
                optimizedInstructions.push_back(inst);
            }
        }
        
        block->instructions = optimizedInstructions;
    }
    
    bool isVariableLive(string var, BasicBlock* block) {
        set<string> liveVars = livenessAnalysis->getLiveOut(block);
        return liveVars.find(var) != liveVars.end();
    }
};
```

### 寄存器分配
```c
class RegisterAllocationUsingLiveness {
private:
    ControlFlowGraph* cfg;
    LiveVariableAnalysis* livenessAnalysis;
    map<string, int> variableToRegister;
    map<int, set<string>> registerToVariables;
    int numRegisters;
    
public:
    RegisterAllocationUsingLiveness(ControlFlowGraph* cfg, int numRegs) 
        : cfg(cfg), numRegisters(numRegs) {
        livenessAnalysis = new LiveVariableAnalysis(cfg);
    }
    
    void allocateRegisters() {
        // 执行活跃变量分析
        livenessAnalysis->analyze();
        
        // 构建冲突图
        map<string, set<string>> conflictGraph = buildConflictGraph();
        
        // 使用图着色算法分配寄存器
        colorVariables(conflictGraph);
    }
    
    map<string, set<string>> buildConflictGraph() {
        map<string, set<string>> conflictGraph;
        
        for (auto block : cfg->getBlocks()) {
            set<string> liveVars = livenessAnalysis->getLiveIn(block);
            
            // 为每个基本块中的活跃变量添加冲突边
            for (auto var1 : liveVars) {
                for (auto var2 : liveVars) {
                    if (var1 != var2) {
                        conflictGraph[var1].insert(var2);
                        conflictGraph[var2].insert(var1);
                    }
                }
            }
        }
        
        return conflictGraph;
    }
    
    void colorVariables(map<string, set<string>>& conflictGraph) {
        // 使用贪心算法进行着色
        for (auto& pair : conflictGraph) {
            string var = pair.first;
            set<string>& neighbors = pair.second;
            
            // 找到可用的寄存器
            int registerNum = findAvailableRegister(neighbors);
            variableToRegister[var] = registerNum;
            
            // 更新寄存器到变量的映射
            registerToVariables[registerNum].insert(var);
        }
    }
    
    int findAvailableRegister(set<string>& neighbors) {
        set<int> usedRegisters;
        
        for (auto neighbor : neighbors) {
            if (variableToRegister.find(neighbor) != variableToRegister.end()) {
                usedRegisters.insert(variableToRegister[neighbor]);
            }
        }
        
        // 找到第一个可用的寄存器
        for (int i = 0; i < numRegisters; i++) {
            if (usedRegisters.find(i) == usedRegisters.end()) {
                return i;
            }
        }
        
        // 如果没有可用寄存器，返回-1表示需要溢出
        return -1;
    }
    
    int getVariableRegister(string var) {
        if (variableToRegister.find(var) != variableToRegister.end()) {
            return variableToRegister[var];
        }
        return -1;
    }
    
    set<string> getRegisterVariables(int reg) {
        if (registerToVariables.find(reg) != registerToVariables.end()) {
            return registerToVariables[reg];
        }
        return set<string>();
    }
};
```

## 🎯 活跃变量分析集成

### 活跃变量分析管理器
```c
class LiveVariableAnalysisManager {
private:
    ControlFlowGraph* cfg;
    map<LivenessAnalysisType, LiveVariableAnalysisBase*> analyses;
    
public:
    LiveVariableAnalysisManager(ControlFlowGraph* cfg) : cfg(cfg) {
        // 初始化各种分析
        analyses[MAY_LIVENESS] = new LiveVariableAnalysis(cfg);
        analyses[MUST_LIVENESS] = new PreciseLiveVariableAnalysis(cfg);
    }
    
    void runAnalysis(LivenessAnalysisType type) {
        if (analyses.find(type) != analyses.end()) {
            analyses[type]->analyze();
        }
    }
    
    void runAllAnalyses() {
        for (auto& pair : analyses) {
            pair.second->analyze();
        }
    }
    
    set<string> getLiveVariables(LivenessAnalysisType type, BasicBlock* block, bool isInSet) {
        if (analyses.find(type) != analyses.end()) {
            LiveVariableAnalysis* analysis = dynamic_cast<LiveVariableAnalysis*>(analyses[type]);
            if (analysis) {
                return isInSet ? analysis->getLiveIn(block) : analysis->getLiveOut(block);
            }
        }
        return set<string>();
    }
    
    void cleanup() {
        for (auto& pair : analyses) {
            delete pair.second;
        }
        analyses.clear();
    }
};

class LiveVariableAnalysisBase {
public:
    virtual void analyze() = 0;
    virtual ~LiveVariableAnalysisBase() = default;
};
```
