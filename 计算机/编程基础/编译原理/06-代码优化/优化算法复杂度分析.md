# 优化算法复杂度分析

## 🎯 优化算法复杂度分析概述

**优化算法复杂度分析**是编译器优化的重要理论基础，用于评估各种优化算法的性能特征和适用场景。通过复杂度分析，可以指导优化算法的选择和实现策略。

## 🔍 基本概念

### 复杂度分析定义
复杂度分析是评估算法在时间和空间上的资源消耗，通常用大O记号表示算法的最坏情况、平均情况和最好情况下的性能。

### 复杂度分析特点
- **理论指导**：为算法选择提供理论依据
- **性能预测**：预测算法在不同规模下的性能
- **优化指导**：指导算法的优化方向
- **比较基准**：比较不同算法的优劣

## 🔧 时间复杂度分析

### 基本时间复杂度分析
```c
class TimeComplexityAnalyzer {
private:
    map<string, ComplexityMetrics> algorithmComplexities;
    
public:
    TimeComplexityAnalyzer() {
        // 初始化各种算法的时间复杂度
        initializeAlgorithmComplexities();
    }
    
    void initializeAlgorithmComplexities() {
        // 数据流分析算法复杂度
        algorithmComplexities["ReachingDefinitionAnalysis"] = 
            ComplexityMetrics(O(n^2), O(n^2), O(n^2));
        algorithmComplexities["LiveVariableAnalysis"] = 
            ComplexityMetrics(O(n^2), O(n^2), O(n^2));
        algorithmComplexities["AvailableExpressionAnalysis"] = 
            ComplexityMetrics(O(n^2), O(n^2), O(n^2));
        
        // 优化算法复杂度
        algorithmComplexities["ConstantFolding"] = 
            ComplexityMetrics(O(n), O(n), O(n));
        algorithmComplexities["DeadCodeElimination"] = 
            ComplexityMetrics(O(n^2), O(n^2), O(n^2));
        algorithmComplexities["CommonSubexpressionElimination"] = 
            ComplexityMetrics(O(n^2), O(n^2), O(n^2));
        
        // 寄存器分配算法复杂度
        algorithmComplexities["GraphColoringAllocation"] = 
            ComplexityMetrics(O(n^2), O(n^2), O(n^2));
        algorithmComplexities["LinearScanAllocation"] = 
            ComplexityMetrics(O(n log n), O(n log n), O(n log n));
        algorithmComplexities["ChowAllocation"] = 
            ComplexityMetrics(O(n^2), O(n^2), O(n^2));
        algorithmComplexities["BriggsAllocation"] = 
            ComplexityMetrics(O(n^2), O(n^2), O(n^2));
    }
    
    ComplexityMetrics getTimeComplexity(string algorithm) {
        if (algorithmComplexities.find(algorithm) != algorithmComplexities.end()) {
            return algorithmComplexities[algorithm];
        }
        return ComplexityMetrics(O(1), O(1), O(1));
    }
    
    void analyzeTimeComplexity(string algorithm, int inputSize) {
        ComplexityMetrics complexity = getTimeComplexity(algorithm);
        
        cout << "算法: " << algorithm << endl;
        cout << "输入规模: " << inputSize << endl;
        cout << "最坏情况: " << complexity.worstCase << endl;
        cout << "平均情况: " << complexity.averageCase << endl;
        cout << "最好情况: " << complexity.bestCase << endl;
        
        // 计算实际运行时间估计
        estimateRunningTime(algorithm, inputSize);
    }
    
    void estimateRunningTime(string algorithm, int inputSize) {
        ComplexityMetrics complexity = getTimeComplexity(algorithm);
        
        // 简化的运行时间估计
        double worstTime = estimateTimeFromComplexity(complexity.worstCase, inputSize);
        double averageTime = estimateTimeFromComplexity(complexity.averageCase, inputSize);
        double bestTime = estimateTimeFromComplexity(complexity.bestCase, inputSize);
        
        cout << "估计运行时间:" << endl;
        cout << "最坏情况: " << worstTime << " ms" << endl;
        cout << "平均情况: " << averageTime << " ms" << endl;
        cout << "最好情况: " << bestTime << " ms" << endl;
    }
    
    double estimateTimeFromComplexity(string complexity, int inputSize) {
        // 简化的复杂度到时间的转换
        if (complexity == "O(1)") return 1.0;
        if (complexity == "O(log n)") return log2(inputSize);
        if (complexity == "O(n)") return inputSize;
        if (complexity == "O(n log n)") return inputSize * log2(inputSize);
        if (complexity == "O(n^2)") return inputSize * inputSize;
        if (complexity == "O(n^3)") return inputSize * inputSize * inputSize;
        return 1.0;
    }
    
    void compareAlgorithms(vector<string> algorithms, int inputSize) {
        cout << "算法复杂度比较 (输入规模: " << inputSize << ")" << endl;
        cout << "===========================================" << endl;
        
        for (auto algorithm : algorithms) {
            ComplexityMetrics complexity = getTimeComplexity(algorithm);
            double estimatedTime = estimateTimeFromComplexity(complexity.averageCase, inputSize);
            
            cout << algorithm << ": " << complexity.averageCase 
                 << " (估计时间: " << estimatedTime << " ms)" << endl;
        }
    }
};

class ComplexityMetrics {
public:
    string worstCase;
    string averageCase;
    string bestCase;
    
    ComplexityMetrics(string worst, string average, string best) 
        : worstCase(worst), averageCase(average), bestCase(best) {}
};
```

```c
        cout << "空间复杂度:" << endl;

        PerformanceMetrics performance = getPerformanceMetrics(algorithm);
        ComplexityMetrics timeComplexity = timeAnalyzer->getTimeComplexity(algorithm);
        SpaceComplexityMetrics spaceComplexity = spaceAnalyzer->getSpaceComplexity(algorithm);
        
        // 计算综合评分
        double timeScore = calculateTimeScore(timeComplexity, inputSize);
        double spaceScore = calculateSpaceScore(spaceComplexity, inputSize);
        double performanceScore = calculatePerformanceScore(performance);
        
        double overallScore = (timeScore + spaceScore + performanceScore) / 3.0;
        
        cout << "综合评估:" << endl;
        cout << "  时间评分: " << timeScore << endl;
        cout << "  空间评分: " << spaceScore << endl;
        cout << "  性能评分: " << performanceScore << endl;
        cout << "  综合评分: " << overallScore << endl;
        
        // 推荐使用场景
        recommendUsageScenarios(algorithm, overallScore);
    }
    
    double calculateTimeScore(ComplexityMetrics timeComplexity, int inputSize) {
        // 简化的时间评分计算
        string complexity = timeComplexity.averageCase;
        if (complexity == "O(1)") return 1.0;
        if (complexity == "O(log n)") return 0.9;
        if (complexity == "O(n)") return 0.8;
        if (complexity == "O(n log n)") return 0.7;
        if (complexity == "O(n^2)") return 0.5;
        if (complexity == "O(n^3)") return 0.3;
        return 0.5;
    }
    
    double calculateSpaceScore(SpaceComplexityMetrics spaceComplexity, int inputSize) {
        // 简化的空间评分计算
        string complexity = spaceComplexity.averageCase;
        if (complexity == "O(1)") return 1.0;
        if (complexity == "O(log n)") return 0.9;
        if (complexity == "O(n)") return 0.8;
        if (complexity == "O(n log n)") return 0.7;
        if (complexity == "O(n^2)") return 0.5;
        if (complexity == "O(n^3)") return 0.3;
        return 0.5;
    }
    
    double calculatePerformanceScore(PerformanceMetrics performance) {
        return (performance.timeEfficiency + performance.spaceEfficiency + 
                performance.optimizationEffect + performance.stability) / 4.0;
    }
    
    void recommendUsageScenarios(string algorithm, double overallScore) {
        cout << "使用场景推荐:" << endl;
        
        if (overallScore >= 0.8) {
            cout << "  推荐用于生产环境" << endl;
            cout << "  适合大规模程序优化" << endl;
        } else if (overallScore >= 0.6) {
            cout << "  推荐用于开发环境" << endl;
            cout << "  适合中等规模程序优化" << endl;
        } else {
            cout << "  推荐用于实验环境" << endl;
            cout << "  适合小规模程序优化" << endl;
        }
    }
    
    void compareAlgorithms(vector<string> algorithms, int inputSize) {
        cout << "算法性能比较 (输入规模: " << inputSize << ")" << endl;
        cout << "===========================================" << endl;
        
        vector<pair<string, double>> algorithmScores;
        
        for (auto algorithm : algorithms) {
            PerformanceMetrics performance = getPerformanceMetrics(algorithm);
            ComplexityMetrics timeComplexity = timeAnalyzer->getTimeComplexity(algorithm);
            SpaceComplexityMetrics spaceComplexity = spaceAnalyzer->getSpaceComplexity(algorithm);
            
            double timeScore = calculateTimeScore(timeComplexity, inputSize);
            double spaceScore = calculateSpaceScore(spaceComplexity, inputSize);
            double performanceScore = calculatePerformanceScore(performance);
            
            double overallScore = (timeScore + spaceScore + performanceScore) / 3.0;
            algorithmScores.push_back({algorithm, overallScore});
        }
        
        // 按综合评分排序
        sort(algorithmScores.begin(), algorithmScores.end(),
             [](const pair<string, double>& a, const pair<string, double>& b) {
                 return a.second > b.second;
             });
        
        for (auto& pair : algorithmScores) {
            cout << pair.first << ": " << pair.second << endl;
        }
    }
    
    ~AlgorithmPerformanceAnalyzer() {
        delete timeAnalyzer;
        delete spaceAnalyzer;
    }
};
```

```c
        cout << "输入规模: " << inputSize << endl;
        cout << "===========================================" << endl;
        
        for (auto algorithm : algorithms) {
            analyzeAlgorithm(algorithm, inputSize);
            cout << endl;
        }
        
        // 生成优化建议
        generateOptimizationRecommendations(algorithms);
    }
    
    void generateOptimizationRecommendations(vector<string> algorithms) {
        cout << "优化建议" << endl;
        cout << "===========================================" << endl;
        
        for (auto algorithm : algorithms) {
            cout << "算法: " << algorithm << endl;
            
            // 根据算法类型生成优化建议
            if (algorithm.find("Analysis") != string::npos) {
                optimizer->optimizeAlgorithm(algorithm, "TimeOptimization");
            } else if (algorithm.find("Allocation") != string::npos) {
                optimizer->optimizeAlgorithm(algorithm, "SpaceOptimization");
            } else {
                optimizer->optimizeAlgorithm(algorithm, "ComprehensiveOptimization");
            }
            
            cout << endl;
        }
    }
    
    string getCurrentTime() {
        // 获取当前时间
        time_t now = time(0);
        char* timeStr = ctime(&now);
        return string(timeStr);
    }
    
    ~ComplexityAnalysisManager() {
        delete timeAnalyzer;
        delete spaceAnalyzer;
        delete performanceAnalyzer;
        delete optimizer;
    }
};
```
