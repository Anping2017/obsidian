# 寄存器分配算法

## 🎯 寄存器分配算法概述

**寄存器分配算法**是编译器优化的重要技术，用于将程序中的变量分配到有限的寄存器中，以提高程序执行效率。寄存器分配算法的目标是最大化寄存器利用率，最小化内存访问次数。

## 🔍 基本概念

### 寄存器分配定义
寄存器分配是将程序中的变量分配到物理寄存器或内存位置的过程，使得程序能够高效执行。

### 寄存器分配特点
- **资源有限**：寄存器数量有限，需要合理分配
- **性能关键**：寄存器访问速度远快于内存访问
- **冲突解决**：需要解决变量间的寄存器冲突
- **溢出处理**：当寄存器不足时需要将变量溢出到内存

### 基本图着色算法
```c
class GraphColoringRegisterAllocation {
private:
    ControlFlowGraph* cfg;
    map<string, set<string>> conflictGraph;
    map<string, int> variableToRegister;
    map<int, bool> registerUsed;
    int numRegisters;
    
public:
    GraphColoringRegisterAllocation(ControlFlowGraph* cfg, int numRegisters) 
        : cfg(cfg), numRegisters(numRegisters) {
        // 初始化寄存器使用状态
        for (int i = 0; i < numRegisters; i++) {
            registerUsed[i] = false;
        }
    }
    
    void allocateRegisters() {
        // 构建冲突图
        buildConflictGraph();
        
        // 执行图着色
        performGraphColoring();
        
        // 处理溢出
        handleSpilling();
    }
    
    void buildConflictGraph() {
        // 初始化冲突图
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    string var = inst.result;
                    if (conflictGraph.find(var) == conflictGraph.end()) {
                        conflictGraph[var] = set<string>();
                    }
                }
            }
        }
        
        // 构建冲突边
        for (auto block : cfg->getBlocks()) {
            buildConflictsInBlock(block);
        }
    }
    
    void buildConflictsInBlock(BasicBlock* block) {
        // 计算基本块内的活跃变量
        set<string> liveVars = computeLiveVariables(block);
        
        // 为每个变量对添加冲突边
        for (auto var1 : liveVars) {
            for (auto var2 : liveVars) {
                if (var1 != var2) {
                    conflictGraph[var1].insert(var2);
                    conflictGraph[var2].insert(var1);
                }
            }
        }
    }
    
    set<string> computeLiveVariables(BasicBlock* block) {
        // 这里需要实现活跃变量分析
        // 简化实现，返回所有变量
        set<string> liveVars;
        
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                liveVars.insert(inst.result);
                if (!inst.arg1.empty()) liveVars.insert(inst.arg1);
                if (!inst.arg2.empty()) liveVars.insert(inst.arg2);
            }
        }
        
        return liveVars;
    }
    
    void performGraphColoring() {
        // 选择着色顺序
        vector<string> coloringOrder = selectColoringOrder();
        
        // 为每个变量分配寄存器
        for (auto var : coloringOrder) {
            int reg = findAvailableRegister(var);
            if (reg != -1) {
                variableToRegister[var] = reg;
                registerUsed[reg] = true;
            } else {
                // 需要溢出
                handleVariableSpilling(var);
            }
        }
    }
    
    vector<string> selectColoringOrder() {
        // 使用启发式方法选择着色顺序
        // 优先选择冲突度高的变量
        vector<pair<string, int>> varConflicts;
        
        for (auto& pair : conflictGraph) {
            string var = pair.first;
            int conflicts = pair.second.size();
            varConflicts.push_back({var, conflicts});
        }
        
        // 按冲突度降序排序
        sort(varConflicts.begin(), varConflicts.end(), 
             [](const pair<string, int>& a, const pair<string, int>& b) {
                 return a.second > b.second;
             });
        
        vector<string> order;
        for (auto& pair : varConflicts) {
            order.push_back(pair.first);
        }
        
        return order;
    }
    
    int findAvailableRegister(string var) {
        // 找到与var不冲突的可用寄存器
        set<string> conflicts = conflictGraph[var];
        
        for (int reg = 0; reg < numRegisters; reg++) {
            if (!registerUsed[reg]) {
                // 检查是否与已分配的变量冲突
                bool hasConflict = false;
                for (auto& pair : variableToRegister) {
                    if (pair.second == reg && conflicts.find(pair.first) != conflicts.end()) {
                        hasConflict = true;
                        break;
                    }
                }
                
                if (!hasConflict) {
                    return reg;
                }
            }
        }
        
        return -1; // 没有可用寄存器
    }
    
    void handleVariableSpilling(string var) {
        // 处理变量溢出
        // 将变量分配到内存位置
        variableToRegister[var] = -1; // -1表示溢出到内存
    }
    
    void handleSpilling() {
        // 处理所有溢出的变量
        for (auto& pair : variableToRegister) {
            if (pair.second == -1) {
                // 为溢出变量分配内存位置
                allocateMemoryLocation(pair.first);
            }
        }
    }
    
    void allocateMemoryLocation(string var) {
        // 为溢出变量分配内存位置
        // 这里需要实现内存分配逻辑
    }
    
    int getVariableRegister(string var) {
        if (variableToRegister.find(var) != variableToRegister.end()) {
            return variableToRegister[var];
        }
        return -1; // 变量未分配寄存器
    }
    
    bool isVariableSpilled(string var) {
        return getVariableRegister(var) == -1;
    }
    
    set<string> getSpilledVariables() {
        set<string> spilled;
        for (auto& pair : variableToRegister) {
```

### 线性扫描算法

```c
class LinearScanRegisterAllocation {
private:
    ControlFlowGraph* cfg;
    map<string, LiveInterval> liveIntervals;
    map<string, int> variableToRegister;
    map<int, bool> registerUsed;
    int numRegisters;
    
public:
    LinearScanRegisterRegisterAllocation(ControlFlowGraph* cfg, int numRegisters) 
        : cfg(cfg), numRegisters(numRegisters) {
        // 初始化寄存器使用状态
        for (int i = 0; i < numRegisters; i++) {
            registerUsed[i] = false;
        }
    }
    
    void allocateRegisters() {
        // 计算活跃区间
        computeLiveIntervals();
        
        // 执行线性扫描
        performLinearScan();
        
        // 处理溢出
        handleSpilling();
    }
    
    void computeLiveIntervals() {
        // 为每个变量计算活跃区间
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    string var = inst.result;
                    
                    if (liveIntervals.find(var) == liveIntervals.end()) {
                        liveIntervals[var] = LiveInterval();
                    }
                    
                    // 更新活跃区间
                    liveIntervals[var].addPoint(inst.position);
                }
            }
        }
    }
    
    void performLinearScan() {
        // 按活跃区间开始位置排序
        vector<pair<string, LiveInterval>> sortedIntervals;
        for (auto& pair : liveIntervals) {
            sortedIntervals.push_back({pair.first, pair.second});
        }
        
        sort(sortedIntervals.begin(), sortedIntervals.end(),
             [](const pair<string, LiveInterval>& a, const pair<string, LiveInterval>& b) {
                 return a.second.getStart() < b.second.getStart();
             });
        
        // 线性扫描分配
        for (auto& pair : sortedIntervals) {
            string var = pair.first;
            LiveInterval interval = pair.second;
            
            int reg = findAvailableRegister(interval);
            if (reg != -1) {
                variableToRegister[var] = reg;
                registerUsed[reg] = true;
            } else {
                // 需要溢出
                handleVariableSpilling(var);
            }
        }
    }
    
    int findAvailableRegister(LiveInterval interval) {
        // 找到与interval不冲突的可用寄存器
        for (int reg = 0; reg < numRegisters; reg++) {
            if (!registerUsed[reg]) {
                // 检查是否与已分配的变量冲突
                bool hasConflict = false;
                for (auto& pair : variableToRegister) {
                    if (pair.second == reg) {
                        LiveInterval existingInterval = liveIntervals[pair.first];
                        if (interval.overlaps(existingInterval)) {
                            hasConflict = true;
                            break;
                        }
                    }
                }
                
                if (!hasConflict) {
                    return reg;
                }
            }
        }
        
        return -1; // 没有可用寄存器
    }
    
    void handleVariableSpilling(string var) {
        // 处理变量溢出
        variableToRegister[var] = -1; // -1表示溢出到内存
    }
    
    void handleSpilling() {
        // 处理所有溢出的变量
        for (auto& pair : variableToRegister) {
            if (pair.second == -1) {
                allocateMemoryLocation(pair.first);
            }
        }
    }
    
    void allocateMemoryLocation(string var) {
        // 为溢出变量分配内存位置
    }
    
    int getVariableRegister(string var) {
        if (variableToRegister.find(var) != variableToRegister.end()) {
            return variableToRegister[var];
        }
        return -1;
    }
    
    bool isVariableSpilled(string var) {
        return getVariableRegister(var) == -1;
    }
    
    set<string> getSpilledVariables() {
        set<string> spilled;
        for (auto& pair : variableToRegister) {
            if (pair.second == -1) {
                spilled.insert(pair.first);
            }
        }
        return spilled;
    }

class LiveInterval {
private:
    int start;
    int end;
    set<int> points;
    
public:
    LiveInterval() : start(INT_MAX), end(INT_MIN) {}
    
    void addPoint(int position) {
        points.insert(position);
        start = min(start, position);
        end = max(end, position);
    }
    
    int getStart() const { return start; }
    int getEnd() const { return end; }
    
    bool overlaps(const LiveInterval& other) const {
        return !(end < other.start || other.end < start);
    }
    
    bool contains(int position) const {
        return position >= start && position <= end;
    }
    
    set<int> getPoints() const { return points; }
};
```

```c
        // 2. 优化溢出位置
        optimizeSpillLocations();
    }
    
    void selectBestSpillVariables() {
        // 选择最佳溢出变量
        // 优先溢出使用频率低的变量
        for (auto& pair : variableToRegister) {
            if (pair.second == -1) {
                // 计算变量使用频率
                int frequency = computeVariableFrequency(pair.first);
                
                // 根据频率决定是否溢出
                if (frequency < getSpillThreshold()) {
                    // 溢出变量
                    spillVariable(pair.first);
                }
            }
        }
    }
    
    int computeVariableFrequency(string var) {
        // 计算变量使用频率
        int frequency = 0;
        
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.result == var || inst.arg1 == var || inst.arg2 == var) {
                    frequency++;
                }
            }
        }
        
        return frequency;
    }
    
    int getSpillThreshold() {
        // 获取溢出阈值
        return 5; // 简化实现
    }
    
    void spillVariable(string var) {
        // 溢出变量
        variableToRegister[var] = -1;
    }
    
    void optimizeSpillLocations() {

        // 选择最佳的内存位置进行溢出
        for (auto& pair : variableToRegister) {

    void optimizeSpilling() {

        reduceSpillCount();
        
        // 2. 优化溢出位置
        optimizeSpillLocations();
    }
    
    void reduceSpillCount() {
        // 减少溢出次数
        // 通过重新分配寄存器来减少溢出
        for (auto& pair : variableToRegister) {
            if (pair.second == -1) {
                // 尝试重新分配寄存器
                int reg = findAlternativeRegister(pair.first);
                if (reg != -1) {
                    variableToRegister[pair.first] = reg;
                    registerUsed[reg] = true;
                }
            }

    }
    
    int findAlternativeRegister(string var) {

        for (int reg = 0; reg < numRegisters; reg++) {

            }
        }
        return -1;
    }
    
    int getVariableRegister(string var) {
        if (variableToRegister.find(var) != variableToRegister.end()) {
```

```c
class RegisterAllocationManager {
private:
    ControlFlowGraph* cfg;
    map<RegisterAllocationAlgorithm, RegisterAllocationBase*> allocators;
    int numRegisters;
    
public:
    RegisterAllocationManager(ControlFlowGraph* cfg, int numRegisters) 
        : cfg(cfg), numRegisters(numRegisters) {
        // 初始化各种分配器
        allocators[GRAPH_COLORING_ALLOCATION] = new GraphColoringRegisterAllocation(cfg, numRegisters);
        allocators[LINEAR_SCAN_ALLOCATION] = new LinearScanRegisterAllocation(cfg, numRegisters);
        allocators[CHOW_ALLOCATION] = new ChowRegisterAllocation(cfg, numRegisters);
        allocators[BRIGGS_ALLOCATION] = new BriggsRegisterAllocation(cfg, numRegisters);
    }
    
    void runAllocation(RegisterAllocationAlgorithm algorithm) {
        if (allocators.find(algorithm) != allocators.end()) {
            allocators[algorithm]->allocateRegisters();
        }
    }
    
    void runAllAllocations() {
        for (auto& pair : allocators) {
            pair.second->allocateRegisters();
        }
    }
    
    int getVariableRegister(RegisterAllocationAlgorithm algorithm, string var) {
        if (allocators.find(algorithm) != allocators.end()) {
            RegisterAllocationBase* allocator = allocators[algorithm];
            return allocator->getVariableRegister(var);
        }
        return -1;
    }
    
    bool isVariableSpilled(RegisterAllocationAlgorithm algorithm, string var) {
        if (allocators.find(algorithm) != allocators.end()) {
            RegisterAllocationBase* allocator = allocators[algorithm];
            return allocator->isVariableSpilled(var);
        }
        return true;
    }
    
    set<string> getSpilledVariables(RegisterAllocationAlgorithm algorithm) {
        if (allocators.find(algorithm) != allocators.end()) {
            RegisterAllocationBase* allocator = allocators[algorithm];
            return allocator->getSpilledVariables();
        }
        return set<string>();
    }
    
    void cleanup() {
        for (auto& pair : allocators) {
            delete pair.second;
        }
        allocators.clear();
    }
};

class RegisterAllocationBase {
public:
    virtual void allocateRegisters() = 0;
    virtual int getVariableRegister(string var) = 0;
    virtual bool isVariableSpilled(string var) = 0;
    virtual set<string> getSpilledVariables() = 0;
    virtual ~RegisterAllocationBase() = default;
};
```
