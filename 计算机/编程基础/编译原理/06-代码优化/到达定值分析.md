# 到达定值分析

## 🎯 到达定值分析概述

**到达定值分析**是数据流分析的重要类型，用于确定程序中每个点处哪些变量的定义可能到达该点。到达定值分析是常量传播、复制传播等优化技术的基础。

## 🔍 基本概念

### 到达定值定义
一个变量v的定义d到达程序点p，当且仅当存在从d到p的执行路径，且在该路径上v没有被重新定义。

### 到达定值分析特点
- **前向分析**：从程序入口向出口分析
- **可能分析**：提供保守的近似结果
- **迭代计算**：通过迭代计算收敛到不动点
- **优化基础**：为多种优化技术提供信息

## 🔧 到达定值分析实现

### 基本到达定值分析
```c
class ReachingDefinitionAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, set<Definition>> reachingIn;
    map<BasicBlock*, set<Definition>> reachingOut;
    map<BasicBlock*, set<Definition>> genSets;
    map<BasicBlock*, set<Definition>> killSets;
    
public:
    ReachingDefinitionAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 计算生成集和杀死集
        computeGenKillSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeGenKillSets() {
        for (auto block : cfg->getBlocks()) {
            set<Definition> gen;
            set<Definition> kill;
            
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    Definition def(inst.result, block, &inst);
                    
                    // 添加到生成集
                    gen.insert(def);
                    
                    // 添加到杀死集（杀死其他对同一变量的定义）
                    for (auto& otherDef : getAllDefinitions(inst.result)) {
                        if (otherDef != def) {
                            kill.insert(otherDef);
                        }
                    }
                }
            }
            
            genSets[block] = gen;
            killSets[block] = kill;
        }
    }
    
    set<Definition> getAllDefinitions(string variable) {
        set<Definition> definitions;
        
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT && inst.result == variable) {
                    Definition def(variable, block, &inst);
                    definitions.insert(def);
                }
            }
        }
        
        return definitions;
    }
    
    void iterativeAnalysis() {
        // 初始化
        initialize();
        
        // 迭代计算
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 1000;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                set<Definition> oldReachingOut = reachingOut[block];
                set<Definition> newReachingOut = computeReachingOut(block);
                
                if (oldReachingOut != newReachingOut) {
                    reachingOut[block] = newReachingOut;
                    reachingIn[block] = computeReachingIn(block);
                    changed = true;
                }
            }
        }
    }
    
    void initialize() {
        for (auto block : cfg->getBlocks()) {
            reachingIn[block] = set<Definition>();
            reachingOut[block] = set<Definition>();
        }
    }
    
    set<Definition> computeReachingOut(BasicBlock* block) {
        // OUT[B] = (IN[B] - KILL[B]) ∪ GEN[B]
        set<Definition> reachingOut = reachingIn[block];
        
        // 减去杀死集
        for (auto def : killSets[block]) {
            reachingOut.erase(def);
        }
        
        // 加上生成集
        for (auto def : genSets[block]) {
            reachingOut.insert(def);
        }
        
        return reachingOut;
    }
    
    set<Definition> computeReachingIn(BasicBlock* block) {
        set<Definition> reachingIn;
        
        // 合并所有前驱块的reachingOut
        for (auto pred : block->predecessors) {
            for (auto def : reachingOut[pred]) {
                reachingIn.insert(def);
            }
        }
        
        return reachingIn;
    }
    
    set<Definition> getReachingIn(BasicBlock* block) {
        return reachingIn[block];
    }
    
    set<Definition> getReachingOut(BasicBlock* block) {
        return reachingOut[block];
    }
    
    set<Definition> getGenSet(BasicBlock* block) {
        return genSets[block];
    }
    
    set<Definition> getKillSet(BasicBlock* block) {
        return killSets[block];
    }
};

struct Definition {
    string variable;
    BasicBlock* block;
    Instruction* instruction;
    
    Definition(string var, BasicBlock* blk, Instruction* inst) 
        : variable(var), block(blk), instruction(inst) {}
    
    bool operator<(const Definition& other) const {
        if (variable != other.variable) {
            return variable < other.variable;
        }
        if (block != other.block) {
            return block < other.block;
        }
        return instruction < other.instruction;
    }
    
    bool operator==(const Definition& other) const {
        return variable == other.variable && 
               block == other.block && 
               instruction == other.instruction;
    }
};
```

## 🔧 高级到达定值分析

### 精确到达定值分析
```c
class PreciseReachingDefinitionAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, map<int, set<Definition>>> reachingIn;
    map<BasicBlock*, map<int, set<Definition>>> reachingOut;
    map<BasicBlock*, map<int, set<Definition>>> genSets;
    map<BasicBlock*, map<int, set<Definition>>> killSets;
    
public:
    PreciseReachingDefinitionAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 计算生成集和杀死集
        computeGenKillSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeGenKillSets() {
        for (auto block : cfg->getBlocks()) {
            map<int, set<Definition>> gen;
            map<int, set<Definition>> kill;
            
            // 为每个指令位置计算生成集和杀死集
            for (int i = 0; i < block->instructions.size(); i++) {
                Instruction& inst = block->instructions[i];
                
                if (inst.type == ASSIGNMENT) {
                    Definition def(inst.result, block, &inst);
                    
                    // 添加到生成集
                    gen[i].insert(def);
                    
                    // 添加到杀死集（杀死其他对同一变量的定义）
                    for (auto& otherDef : getAllDefinitions(inst.result)) {
                        if (otherDef != def) {
                            kill[i].insert(otherDef);
                        }
                    }
                }
            }
            
            genSets[block] = gen;
            killSets[block] = kill;
        }
    }
    
    set<Definition> getAllDefinitions(string variable) {
        set<Definition> definitions;
        
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT && inst.result == variable) {
                    Definition def(variable, block, &inst);
                    definitions.insert(def);
                }
            }
        }
        
        return definitions;
    }
    
    void iterativeAnalysis() {
        // 初始化
        initialize();
        
        // 迭代计算
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 1000;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                map<int, set<Definition>> oldReachingOut = reachingOut[block];
                map<int, set<Definition>> newReachingOut = computeReachingOut(block);
                
                if (oldReachingOut != newReachingOut) {
                    reachingOut[block] = newReachingOut;
                    reachingIn[block] = computeReachingIn(block);
                    changed = true;
                }
            }
        }
    }
    
    void initialize() {
        for (auto block : cfg->getBlocks()) {
            reachingIn[block] = map<int, set<Definition>>();
            reachingOut[block] = map<int, set<Definition>>();
        }
    }
    
    map<int, set<Definition>> computeReachingOut(BasicBlock* block) {
        map<int, set<Definition>> reachingOut;
        
        // 为每个指令位置计算reachingOut
        for (int i = 0; i < block->instructions.size(); i++) {
            set<Definition> reachingOutAtI;
            
            if (i == 0) {
                // 第一条指令的reachingOut是前驱块的reachingOut
                for (auto pred : block->predecessors) {
                    for (auto def : reachingOut[pred][block->instructions.size() - 1]) {
                        reachingOutAtI.insert(def);
                    }
                }
            } else {
                // 其他指令的reachingOut是前一条指令的reachingOut
                reachingOutAtI = reachingOut[block][i - 1];
            }
            
            reachingOut[i] = reachingOutAtI;
        }
        
        return reachingOut;
    }
    
    map<int, set<Definition>> computeReachingIn(BasicBlock* block) {
        map<int, set<Definition>> reachingIn;
        
        // 为每个指令位置计算reachingIn
        for (int i = 0; i < block->instructions.size(); i++) {
            set<Definition> reachingInAtI = reachingOut[block][i];
            
            // 减去杀死集
            for (auto def : killSets[block][i]) {
                reachingInAtI.erase(def);
            }
            
            // 加上生成集
            for (auto def : genSets[block][i]) {
                reachingInAtI.insert(def);
            }
            
            reachingIn[i] = reachingInAtI;
        }
        
        return reachingIn;
    }
    
    set<Definition> getReachingIn(BasicBlock* block, int position) {
        return reachingIn[block][position];
    }
    
    set<Definition> getReachingOut(BasicBlock* block, int position) {
        return reachingOut[block][position];
    }
};
```

### 寄存器到达定值分析
```c
class RegisterReachingDefinitionAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, set<Definition>> reachingIn;
    map<BasicBlock*, set<Definition>> reachingOut;
    map<string, int> variableToRegister;
    map<int, set<Definition>> registerToDefinitions;
    
public:
    RegisterReachingDefinitionAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 执行基本到达定值分析
        ReachingDefinitionAnalysis basicAnalysis(cfg);
        basicAnalysis.analyze();
        
        // 获取基本分析结果
        reachingIn = basicAnalysis.getReachingIn();
        reachingOut = basicAnalysis.getReachingOut();
        
        // 计算寄存器到达定值
        computeRegisterReachingDefinitions();
    }
    
    void computeRegisterReachingDefinitions() {
        // 为每个基本块计算寄存器到达定值
        for (auto block : cfg->getBlocks()) {
            set<Definition> reachingDefs = reachingIn[block];
            
            // 计算每个寄存器的到达定值
            for (auto def : reachingDefs) {
                if (variableToRegister.find(def.variable) != variableToRegister.end()) {
                    int reg = variableToRegister[def.variable];
                    registerToDefinitions[reg].insert(def);
                }
            }
        }
    }
    
    void setVariableToRegister(string var, int reg) {
        variableToRegister[var] = reg;
    }
    
    int getVariableRegister(string var) {
        if (variableToRegister.find(var) != variableToRegister.end()) {
            return variableToRegister[var];
        }
        return -1; // 表示变量未分配寄存器
    }
    
    set<Definition> getRegisterDefinitions(int reg) {
        if (registerToDefinitions.find(reg) != registerToDefinitions.end()) {
            return registerToDefinitions[reg];
        }
        return set<Definition>();
    }
    
    bool isRegisterReachingDefinition(int reg, BasicBlock* block) {
        set<Definition> regDefs = getRegisterDefinitions(reg);
        set<Definition> reachingDefs = reachingIn[block];
        
        for (auto def : regDefs) {
            if (reachingDefs.find(def) != reachingDefs.end()) {
                return true;
            }
        }
        
        return false;
    }
    
    set<int> getReachingRegisters(BasicBlock* block) {
        set<int> reachingRegs;
        
        for (auto def : reachingIn[block]) {
            if (variableToRegister.find(def.variable) != variableToRegister.end()) {
                int reg = variableToRegister[def.variable];
                reachingRegs.insert(reg);
            }
        }
        
        return reachingRegs;
    }
};
```

```c
                }

        }

        variableToMemory[var] = mem;
    }

        if (variableToMemory.find(var) != variableToMemory.end()) {
            return variableToMemory[var];
        }
        return MemoryLocation(); // 表示变量未分配内存
    }
    
    set<Definition> getMemoryDefinitions(MemoryLocation mem) {
        if (memoryToDefinitions.find(mem) != memoryToDefinitions.end()) {
            return memoryToDefinitions[mem];
        }
        return set<Definition>();
    }
    
    bool isMemoryReachingDefinition(MemoryLocation mem, BasicBlock* block) {
        set<Definition> memDefs = getMemoryDefinitions(mem);
        set<Definition> reachingDefs = reachingIn[block];
        
        for (auto def : memDefs) {
            if (reachingDefs.find(def) != reachingDefs.end()) {
                return true;
            }
        }
        
        return false;
    }
    
    set<MemoryLocation> getReachingMemoryLocations(BasicBlock* block) {
        set<MemoryLocation> reachingMem;
        
        for (auto def : reachingIn[block]) {
            if (variableToMemory.find(def.variable) != variableToMemory.end()) {
                MemoryLocation mem = variableToMemory[def.variable];
                reachingMem.insert(mem);
            }
        }
        
        return reachingMem;
    }
};
```

### 常量传播
```c
class ConstantPropagationUsingReachingDefinition {
private:
    ControlFlowGraph* cfg;
    ReachingDefinitionAnalysis* reachingDefinitionAnalysis;
    map<string, ConstantValue> constantMap;
    
public:
    ConstantPropagationUsingReachingDefinition(ControlFlowGraph* cfg) : cfg(cfg) {
        reachingDefinitionAnalysis = new ReachingDefinitionAnalysis(cfg);
    }
    
    void propagateConstants() {
        // 执行到达定值分析
        reachingDefinitionAnalysis->analyze();
        
        // 传播常量
        for (auto block : cfg->getBlocks()) {
            propagateConstantsInBlock(block);
        }
    }
    
    void propagateConstantsInBlock(BasicBlock* block) {
        set<Definition> reachingDefs = reachingDefinitionAnalysis->getReachingIn(block);
        
        // 为每个变量计算常量值
        for (auto def : reachingDefs) {
            string var = def.variable;
            ConstantValue value = evaluateConstantDefinition(def);
            
            if (value.isConstant) {
                constantMap[var] = value;
            }
        }
        
        // 应用常量传播
        applyConstantPropagation(block);
    }
    
    ConstantValue evaluateConstantDefinition(Definition& def) {
        Instruction* inst = def.instruction;
        
        if (inst->type == ASSIGNMENT) {
            if (inst->op == "=") {
                // 简单赋值
                return getConstantValue(inst->arg1);
            } else {
                // 二元运算
                ConstantValue left = getConstantValue(inst->arg1);
                ConstantValue right = getConstantValue(inst->arg2);
                
                if (left.isConstant && right.isConstant) {
                    int result = evaluateBinaryOperation(inst->op, left.value, right.value);
                    return ConstantValue(true, result);
                }
            }
        }
        
        return ConstantValue(false, 0);
    }
    
    ConstantValue getConstantValue(string var) {
        // 检查是否为字面常量
        if (isLiteralConstant(var)) {
            return ConstantValue(true, parseLiteralConstant(var));
        }
        
        // 检查是否为已传播的常量
        if (constantMap.find(var) != constantMap.end()) {
            return constantMap[var];
        }
        
        return ConstantValue(false, 0);
    }
    
    int evaluateBinaryOperation(string op, int left, int right) {
        if (op == "+") return left + right;
        if (op == "-") return left - right;
        if (op == "*") return left * right;
        if (op == "/") return right != 0 ? left / right : 0;
        if (op == "%") return right != 0 ? left % right : 0;
        if (op == "&") return left & right;
        if (op == "|") return left | right;
        if (op == "^") return left ^ right;
        if (op == "<<") return left << right;
        if (op == ">>") return left >> right;
        if (op == "==") return left == right ? 1 : 0;
        if (op == "!=") return left != right ? 1 : 0;
        if (op == "<") return left < right ? 1 : 0;
        if (op == ">") return left > right ? 1 : 0;
        if (op == "<=") return left <= right ? 1 : 0;
        if (op == ">=") return left >= right ? 1 : 0;
        return 0;
    }
    
    bool isLiteralConstant(string str) {
        if (str.empty()) return false;
        return str.find_first_not_of("0123456789") == string::npos ||
               str == "true" || str == "false";
    }
    
    int parseLiteralConstant(string str) {
        if (str == "true") return 1;
        if (str == "false") return 0;
        return stoi(str);
    }
    
    void applyConstantPropagation(BasicBlock* block) {
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                // 替换常量操作数
                if (constantMap.find(inst.arg1) != constantMap.end() && 
                    constantMap[inst.arg1].isConstant) {
                    inst.arg1 = to_string(constantMap[inst.arg1].value);
                }
                if (constantMap.find(inst.arg2) != constantMap.end() && 
                    constantMap[inst.arg2].isConstant) {
                    inst.arg2 = to_string(constantMap[inst.arg2].value);
                }
            }
        }
    }
};
```

### 复制传播
```c
class CopyPropagationUsingReachingDefinition {
private:
    ControlFlowGraph* cfg;
    ReachingDefinitionAnalysis* reachingDefinitionAnalysis;
    map<string, string> copyMap;
    
public:
    CopyPropagationUsingReachingDefinition(ControlFlowGraph* cfg) : cfg(cfg) {
        reachingDefinitionAnalysis = new ReachingDefinitionAnalysis(cfg);
    }
    
    void propagateCopies() {
        // 执行到达定值分析
        reachingDefinitionAnalysis->analyze();
        
        // 传播复制
        for (auto block : cfg->getBlocks()) {
            propagateCopiesInBlock(block);
        }
    }
    
    void propagateCopiesInBlock(BasicBlock* block) {
        set<Definition> reachingDefs = reachingDefinitionAnalysis->getReachingIn(block);
        
        // 为每个变量计算复制值
        for (auto def : reachingDefs) {
            string var = def.variable;
            string copyValue = evaluateCopyDefinition(def);
            
            if (!copyValue.empty()) {
                copyMap[var] = copyValue;
            }
        }
        
        // 应用复制传播
        applyCopyPropagation(block);
    }
    
    string evaluateCopyDefinition(Definition& def) {
        Instruction* inst = def.instruction;
        
        if (inst->type == ASSIGNMENT && inst->op == "=") {
            // 简单复制操作
            return inst->arg1;
        }
        
        return "";
    }
    
    void applyCopyPropagation(BasicBlock* block) {
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                // 替换复制操作数
                if (copyMap.find(inst.arg1) != copyMap.end()) {
                    inst.arg1 = copyMap[inst.arg1];
                }
                if (copyMap.find(inst.arg2) != copyMap.end()) {
                    inst.arg2 = copyMap[inst.arg2];
                }
            }
        }
    }
};
```

## 🎯 到达定值分析集成

### 到达定值分析管理器
```c
class ReachingDefinitionAnalysisManager {
private:
    ControlFlowGraph* cfg;
    map<ReachingDefinitionAnalysisType, ReachingDefinitionAnalysisBase*> analyses;
    
public:
    ReachingDefinitionAnalysisManager(ControlFlowGraph* cfg) : cfg(cfg) {
        // 初始化各种分析
        analyses[MAY_REACHING_DEFINITION] = new ReachingDefinitionAnalysis(cfg);
        analyses[MUST_REACHING_DEFINITION] = new PreciseReachingDefinitionAnalysis(cfg);
    }
    
    void runAnalysis(ReachingDefinitionAnalysisType type) {
        if (analyses.find(type) != analyses.end()) {
            analyses[type]->analyze();
        }
    }
    
    void runAllAnalyses() {
        for (auto& pair : analyses) {
            pair.second->analyze();
        }
    }
    
    set<Definition> getReachingDefinitions(ReachingDefinitionAnalysisType type, BasicBlock* block, bool isInSet) {
        if (analyses.find(type) != analyses.end()) {
            ReachingDefinitionAnalysis* analysis = dynamic_cast<ReachingDefinitionAnalysis*>(analyses[type]);
            if (analysis) {
                return isInSet ? analysis->getReachingIn(block) : analysis->getReachingOut(block);
            }
        }
        return set<Definition>();
    }
    
    void cleanup() {
        for (auto& pair : analyses) {
            delete pair.second;
        }
        analyses.clear();
    }
};

class ReachingDefinitionAnalysisBase {
public:
    virtual void analyze() = 0;
    virtual ~ReachingDefinitionAnalysisBase() = default;
};
```
