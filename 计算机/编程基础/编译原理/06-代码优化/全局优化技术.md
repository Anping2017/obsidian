# 全局优化技术

## 🎯 全局优化技术概述

**全局优化技术**是在整个程序范围内进行的代码优化，需要跨基本块的数据流分析。全局优化能够发现和利用更多的优化机会，显著提高程序性能。

## 🔍 基本概念

### 全局优化定义
全局优化是在整个程序范围内进行的优化，考虑跨基本块的数据流关系和控制流关系，能够进行更深入的代码分析和优化。

### 全局优化特点
- **范围广泛**：覆盖整个程序或函数
- **分析复杂**：需要复杂的数据流分析
- **效果显著**：能够发现更多优化机会
- **实现困难**：需要精确的数据流分析

## 🔧 常量传播优化

### 常量传播实现
```c
class ConstantPropagationOptimizer {
private:
    ControlFlowGraph* cfg;
    map<string, ConstantValue> constantMap;
    map<BasicBlock*, map<string, ConstantValue>> blockConstants;
    
public:
    ConstantPropagationOptimizer(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void optimize() {
        // 初始化常量映射
        initializeConstants();
        
        // 迭代传播常量
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 100;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                if (propagateConstantsInBlock(block)) {
                    changed = true;
                }
            }
        }
        
        // 应用常量传播
        applyConstantPropagation();
    }
    
    void initializeConstants() {
        constantMap.clear();
        blockConstants.clear();
        
        // 初始化每个基本块的常量映射
        for (auto block : cfg->getBlocks()) {
            blockConstants[block] = map<string, ConstantValue>();
        }
    }
    
    bool propagateConstantsInBlock(BasicBlock* block) {
        bool changed = false;
        map<string, ConstantValue>& blockConsts = blockConstants[block];
        
        // 合并前驱块的常量信息
        for (auto pred : block->predecessors) {
            mergeConstants(blockConsts, blockConstants[pred]);
        }
        
        // 在基本块内传播常量
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                ConstantValue result = evaluateConstantExpression(inst, blockConsts);
                if (result.isConstant) {
                    blockConsts[inst.result] = result;
                    changed = true;
                } else {
                    // 如果结果不是常量，移除之前的常量信息
                    if (blockConsts.find(inst.result) != blockConsts.end()) {
                        blockConsts.erase(inst.result);
                        changed = true;
                    }
                }
            }
        }
        
        return changed;
    }
    
    void mergeConstants(map<string, ConstantValue>& target, map<string, ConstantValue>& source) {
        for (auto& pair : source) {
            string var = pair.first;
            ConstantValue value = pair.second;
            
            if (target.find(var) != target.end()) {
                // 如果变量在两个映射中都存在，检查是否一致
                if (target[var] != value) {
                    // 不一致，标记为非常量
                    target[var] = ConstantValue(false, 0);
                }
            } else {
                target[var] = value;
            }
        }
    }
    
    ConstantValue evaluateConstantExpression(Instruction& inst, map<string, ConstantValue>& constants) {
        if (inst.op == "=") {
            // 简单赋值
            return getConstantValue(inst.arg1, constants);
        } else {
            // 二元运算
            ConstantValue left = getConstantValue(inst.arg1, constants);
            ConstantValue right = getConstantValue(inst.arg2, constants);
            
            if (left.isConstant && right.isConstant) {
                int result = evaluateBinaryOperation(inst.op, left.value, right.value);
                return ConstantValue(true, result);
            }
        }
        
        return ConstantValue(false, 0);
    }
    
    ConstantValue getConstantValue(string var, map<string, ConstantValue>& constants) {
        if (constants.find(var) != constants.end()) {
            return constants[var];
        }
        
        // 检查是否为字面常量
        if (isLiteralConstant(var)) {
            return ConstantValue(true, parseLiteralConstant(var));
        }
        
        return ConstantValue(false, 0);
    }
    
    int evaluateBinaryOperation(string op, int left, int right) {
        if (op == "+") return left + right;
        if (op == "-") return left - right;
        if (op == "*") return left * right;
        if (op == "/") return right != 0 ? left / right : 0;
        if (op == "%") return right != 0 ? left % right : 0;
        if (op == "&") return left & right;
        if (op == "|") return left | right;
        if (op == "^") return left ^ right;
        if (op == "<<") return left << right;
        if (op == ">>") return left >> right;
        if (op == "==") return left == right ? 1 : 0;
        if (op == "!=") return left != right ? 1 : 0;
        if (op == "<") return left < right ? 1 : 0;
        if (op == ">") return left > right ? 1 : 0;
        if (op == "<=") return left <= right ? 1 : 0;
        if (op == ">=") return left >= right ? 1 : 0;
        return 0;
    }
    
    bool isLiteralConstant(string str) {
        if (str.empty()) return false;
        return str.find_first_not_of("0123456789") == string::npos ||
               str == "true" || str == "false";
    }
    
    int parseLiteralConstant(string str) {
        if (str == "true") return 1;
        if (str == "false") return 0;
        return stoi(str);
    }
    
    void applyConstantPropagation() {
        for (auto block : cfg->getBlocks()) {
            map<string, ConstantValue>& blockConsts = blockConstants[block];
            
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    // 替换常量操作数
                    if (blockConsts.find(inst.arg1) != blockConsts.end() && 
                        blockConsts[inst.arg1].isConstant) {
                        inst.arg1 = to_string(blockConsts[inst.arg1].value);
                    }
                    if (blockConsts.find(inst.arg2) != blockConsts.end() && 
                        blockConsts[inst.arg2].isConstant) {
                        inst.arg2 = to_string(blockConsts[inst.arg2].value);
                    }
                }
            }
        }
    }
};

struct ConstantValue {
    bool isConstant;
    int value;
    
    ConstantValue(bool isConst, int val) : isConstant(isConst), value(val) {}
    
    bool operator==(const ConstantValue& other) const {
        return isConstant == other.isConstant && value == other.value;
    }
    
    bool operator!=(const ConstantValue& other) const {
        return !(*this == other);
    }
};
```

```c
class CopyPropagationOptimizer {
private:
    ControlFlowGraph* cfg;
    map<string, string> copyMap;
    map<BasicBlock*, map<string, string>> blockCopies;
    
public:
    CopyPropagationOptimizer(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void optimize() {
        // 初始化复制映射
        initializeCopies();
        
        // 迭代传播复制
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 100;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                if (propagateCopiesInBlock(block)) {
                    changed = true;
                }
            }
        }
        
        // 应用复制传播
        applyCopyPropagation();
    }
    
    void initializeCopies() {
        copyMap.clear();
        blockCopies.clear();
        
        // 初始化每个基本块的复制映射
        for (auto block : cfg->getBlocks()) {
            blockCopies[block] = map<string, string>();
        }
    }
    
    bool propagateCopiesInBlock(BasicBlock* block) {
        bool changed = false;
        map<string, string>& blockCopy = blockCopies[block];
        
        // 合并前驱块的复制信息
        for (auto pred : block->predecessors) {
            mergeCopies(blockCopy, blockCopies[pred]);
        }
        
        // 在基本块内传播复制
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                if (inst.op == "=" && isSimpleCopy(inst)) {
                    // 简单复制操作
                    blockCopy[inst.result] = inst.arg1;
                    changed = true;
                } else {
                    // 复杂操作，移除复制信息
                    if (blockCopy.find(inst.result) != blockCopy.end()) {
                        blockCopy.erase(inst.result);
                        changed = true;
                    }
                }
            }
        }
        
        return changed;
    }
    
    void mergeCopies(map<string, string>& target, map<string, string>& source) {
        for (auto& pair : source) {
            string var = pair.first;
            string value = pair.second;
            
            if (target.find(var) != target.end()) {
                // 如果变量在两个映射中都存在，检查是否一致
                if (target[var] != value) {
                    // 不一致，移除复制信息
                    target.erase(var);
                }
            } else {
                target[var] = value;
            }
        }
    }
    
    bool isSimpleCopy(Instruction& inst) {
        // 检查是否为简单的复制操作
        return inst.op == "=" && !isLiteralConstant(inst.arg1);
    }
    
    bool isLiteralConstant(string str) {
        if (str.empty()) return false;
        return str.find_first_not_of("0123456789") == string::npos ||
               str == "true" || str == "false";
    }
    
    void applyCopyPropagation() {
        for (auto block : cfg->getBlocks()) {
            map<string, string>& blockCopy = blockCopies[block];
            
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    // 替换复制操作数
                    if (blockCopy.find(inst.arg1) != blockCopy.end()) {
                        inst.arg1 = blockCopy[inst.arg1];
                    }
                    if (blockCopy.find(inst.arg2) != blockCopy.end()) {
                        inst.arg2 = blockCopy[inst.arg2];
                    }
                }
            }
        }
    }
};
```

```c
class PartialRedundancyEliminationOptimizer {
private:
    ControlFlowGraph* cfg;
    AvailableExpressionAnalysis* availableAnalysis;
    map<BasicBlock*, set<string>> availableExpressions;
    map<BasicBlock*, set<string>> anticipatedExpressions;
    
public:
    PartialRedundancyEliminationOptimizer(ControlFlowGraph* cfg) : cfg(cfg) {
        availableAnalysis = new AvailableExpressionAnalysis(cfg);
    }
    
    void optimize() {
        // 计算可用表达式
        availableAnalysis->analyze();
        availableExpressions = availableAnalysis->getAvailableExpressions();
        
        // 计算预期表达式
        computeAnticipatedExpressions();
        
        // 计算可插入点
        computeInsertionPoints();
        
        // 执行部分冗余消除
        eliminatePartialRedundancy();
    }
    
    void computeAnticipatedExpressions() {
        // 初始化预期表达式
        for (auto block : cfg->getBlocks()) {
            anticipatedExpressions[block] = set<string>();
        }
        
        // 迭代计算预期表达式
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 100;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                if (computeAnticipatedForBlock(block)) {
                    changed = true;
                }
            }
        }
    }
    
    bool computeAnticipatedForBlock(BasicBlock* block) {
        bool changed = false;
        set<string>& anticipated = anticipatedExpressions[block];
        set<string> newAnticipated;
        
        // 计算基本块内生成的表达式
        set<string> generated = getGeneratedExpressions(block);
        
        // 计算基本块内杀死的表达式
        set<string> killed = getKilledExpressions(block);
        
        // 计算后继块的预期表达式
        set<string> successorAnticipated;
        for (auto succ : block->successors) {
            for (auto expr : anticipatedExpressions[succ]) {
                successorAnticipated.insert(expr);
            }
        }
        
        // 计算新的预期表达式
        for (auto expr : generated) {
            newAnticipated.insert(expr);
        }
        
        for (auto expr : successorAnticipated) {
            if (killed.find(expr) == killed.end()) {
                newAnticipated.insert(expr);
            }
        }
        
        if (newAnticipated != anticipated) {
            anticipated = newAnticipated;
            changed = true;
        }
        
        return changed;
    }
    
    set<string> getGeneratedExpressions(BasicBlock* block) {
        set<string> generated;
        
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                string expr = buildExpressionKey(inst);
                generated.insert(expr);
            }
        }
        
        return generated;
    }
    
    set<string> getKilledExpressions(BasicBlock* block) {
        set<string> killed;
        
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                // 杀死包含结果变量的表达式
                string result = inst.result;
                for (auto& expr : availableExpressions[block]) {
                    if (expr.find(result) != string::npos) {
                        killed.insert(expr);
                    }
                }
            }
        }
        
        return killed;
    }
```

## 🎯 全局优化集成

### 全局优化器
```c
class GlobalOptimizer {
private:
    ControlFlowGraph* cfg;
    vector<GlobalOptimizationType> enabledOptimizations;
    GlobalOptimizationDependency dependency;
    
public:
    GlobalOptimizer(ControlFlowGraph* cfg) : cfg(cfg), dependency() {
        // 默认启用所有全局优化
        enabledOptimizations = {
            CONSTANT_PROPAGATION,
            COPY_PROPAGATION,
            PARTIAL_REDUNDANCY_ELIMINATION,
            DEAD_CODE_ELIMINATION,
            FUNCTION_INLINING
        };
    }
    
    void optimize() {
        // 按依赖关系排序优化
        vector<GlobalOptimizationType> sortedOptimizations = sortByDependency();
        
        // 执行优化
        for (auto optType : sortedOptimizations) {
            if (isEnabled(optType)) {
                applyOptimization(optType);
            }
        }
    }
    
    vector<GlobalOptimizationType> sortByDependency() {
        vector<GlobalOptimizationType> sorted;
        set<GlobalOptimizationType> visited;
        
        for (auto optType : enabledOptimizations) {
            if (visited.find(optType) == visited.end()) {
                visitOptimization(optType, sorted, visited);
            }
        }
        
        return sorted;
    }
    
    void visitOptimization(GlobalOptimizationType optType, 
                          vector<GlobalOptimizationType>& sorted, 
                          set<GlobalOptimizationType>& visited) {
        visited.insert(optType);
        
        // 先访问依赖的优化
        for (auto dep : dependency.getDependencies(optType)) {
            if (visited.find(dep) == visited.end()) {
                visitOptimization(dep, sorted, visited);
            }
        }
        
        sorted.push_back(optType);
    }
    
    void applyOptimization(GlobalOptimizationType type) {
        switch (type) {
            case CONSTANT_PROPAGATION:
                ConstantPropagationOptimizer(cfg).optimize();
                break;
            case COPY_PROPAGATION:
                CopyPropagationOptimizer(cfg).optimize();
                break;
            case PARTIAL_REDUNDANCY_ELIMINATION:
                PartialRedundancyEliminationOptimizer(cfg).optimize();
                break;
            case DEAD_CODE_ELIMINATION:
                // 全局死代码消除
                break;
            case FUNCTION_INLINING:
                FunctionInliningOptimizer(cfg).optimize();
                break;
        }
    }
    
    bool isEnabled(GlobalOptimizationType type) {
        return find(enabledOptimizations.begin(), enabledOptimizations.end(), type) 
               != enabledOptimizations.end();
    }
    
    void enableOptimization(GlobalOptimizationType type) {
        if (!isEnabled(type)) {
            enabledOptimizations.push_back(type);
        }
    }
    
    void disableOptimization(GlobalOptimizationType type) {
        enabledOptimizations.erase(
            remove(enabledOptimizations.begin(), enabledOptimizations.end(), type),
            enabledOptimizations.end());
    }
};
```
