# 优化技术对比

## 🎯 优化技术对比概述

**优化技术对比**是编译器原理学习的重要组成部分，通过对比不同优化技术的原理、效果和适用场景，帮助学习者深入理解各种优化技术的本质差异，从而在实际应用中做出正确的选择。

## 🔍 对比维度

### 1. 优化效果维度
- **性能提升**：优化后代码的执行效率提升
- **代码质量**：优化后代码的可读性和维护性
- **资源消耗**：优化过程中消耗的时间和空间

### 2. 适用范围维度
- **局部性**：优化适用的代码范围
- **通用性**：优化技术的通用程度
- **针对性**：优化技术的针对性

### 3. 实现复杂度维度
- **算法复杂度**：优化算法的复杂度
- **实现难度**：优化技术的实现难度
- **维护成本**：优化技术的维护成本

## 📊 局部优化技术对比

### 常量优化技术对比

#### 常量折叠 vs 常量传播对比

| 特性 | 常量折叠 | 常量传播 | 优势分析 |
|------|----------|----------|----------|
| **优化时机** | 编译时 | 编译时 | 都在编译时进行 |
| **优化范围** | 表达式内 | 跨语句 | 常量传播范围更广 |
| **优化效果** | 直接计算 | 间接优化 | 常量折叠效果更直接 |
| **实现复杂度** | 简单 | 中等 | 常量折叠实现更简单 |
| **适用场景** | 常量表达式 | 常量变量 | 根据场景选择 |

```c
// 常量折叠 vs 常量传播对比示例
class ConstantOptimizationComparison {
public:
    // 常量折叠：直接计算常量表达式
    class ConstantFolding {
    public:
        int foldConstant(int a, int b, char op) {
            switch (op) {
                case '+':
                    return a + b;
                case '-':
                    return a - b;
                case '*':
                    return a * b;
                case '/':
                    return b != 0 ? a / b : 0;
                default:
                    return 0;
            }
        }
        
        void optimizeExpression(ASTNode* node) {
            if (node->type == "binary_op") {
                ASTNode* left = node->children[0];
                ASTNode* right = node->children[1];
                
                if (left->type == "number" && right->type == "number") {
                    int result = foldConstant(left->value, right->value, node->op);
                    node->type = "number";
                    node->value = result;
                    node->children.clear();
                }
            }
        }
    };
    
    // 常量传播：将常量值传播到使用处
    class ConstantPropagation {
    private:
        map<string, int> constantValues;
        
    public:
        void propagateConstants(ASTNode* node) {
            if (node->type == "assignment") {
                string varName = node->children[0]->value;
                ASTNode* value = node->children[1];
                
                if (value->type == "number") {
                    constantValues[varName] = value->value;
                }
            } else if (node->type == "identifier") {
                string varName = node->value;
                if (constantValues.find(varName) != constantValues.end()) {
                    node->type = "number";
                    node->value = constantValues[varName];
                }
            }
            
            for (ASTNode* child : node->children) {
                propagateConstants(child);
            }
        }
    };
};
```

#### 死代码消除 vs 无用代码消除对比

| 特性 | 死代码消除 | 无用代码消除 | 优势分析 |
|------|------------|--------------|----------|
| **定义** | 永远不会执行的代码 | 执行但无用的代码 | 死代码消除更彻底 |
| **检测方法** | 控制流分析 | 数据流分析 | 检测方法不同 |
| **优化效果** | 减少代码大小 | 提高执行效率 | 效果不同 |
| **实现复杂度** | 中等 | 复杂 | 无用代码消除更复杂 |
| **适用场景** | 条件分支 | 变量使用 | 根据场景选择 |

```c
// 死代码消除 vs 无用代码消除对比示例
class DeadCodeEliminationComparison {
public:
    // 死代码消除：消除永远不会执行的代码
    class DeadCodeElimination {
    public:
        void eliminateDeadCode(ASTNode* node) {
            if (node->type == "if_statement") {
                ASTNode* condition = node->children[0];
                ASTNode* thenBranch = node->children[1];
                ASTNode* elseBranch = node->children[2];
                
                if (isConstantCondition(condition)) {
                    if (evaluateCondition(condition)) {
                        // 只保留then分支
                        node->children = {thenBranch};
                    } else {
                        // 只保留else分支
                        node->children = {elseBranch};
                    }
                }
            }
            
            for (ASTNode* child : node->children) {
                eliminateDeadCode(child);
            }
        }
        
        bool isConstantCondition(ASTNode* condition) {
            return condition->type == "number";
        }
        
        bool evaluateCondition(ASTNode* condition) {
            return condition->value != 0;
        }
    };
    
    // 无用代码消除：消除执行但无用的代码
    class UselessCodeElimination {
    private:
        map<string, bool> variableUsed;
        
    public:
        void eliminateUselessCode(ASTNode* node) {
            // 标记使用的变量
            markUsedVariables(node);
            
            // 消除无用赋值
            eliminateUselessAssignments(node);
        }
        
        void markUsedVariables(ASTNode* node) {
            if (node->type == "identifier") {
                variableUsed[node->value] = true;
            }
            
            for (ASTNode* child : node->children) {
                markUsedVariables(child);
            }
        }
        
        void eliminateUselessAssignments(ASTNode* node) {
            if (node->type == "assignment") {
                string varName = node->children[0]->value;
                if (variableUsed.find(varName) == variableUsed.end()) {
                    // 消除无用赋值
                    node->type = "empty";
                    node->children.clear();
                }
            }
            
            for (ASTNode* child : node->children) {
                eliminateUselessAssignments(child);
            }
        }
    };
};
```

### 表达式优化技术对比

#### 公共子表达式消除 vs 强度削弱对比

| 特性 | 公共子表达式消除 | 强度削弱 | 优势分析 |
|------|------------------|----------|----------|
| **优化目标** | 重复计算 | 计算效率 | 目标不同 |
| **优化方法** | 缓存结果 | 替换操作 | 方法不同 |
| **优化效果** | 减少计算量 | 提高计算速度 | 效果不同 |
| **实现复杂度** | 复杂 | 简单 | 强度削弱更简单 |
| **适用场景** | 重复表达式 | 特定操作 | 根据场景选择 |

```c
// 公共子表达式消除 vs 强度削弱对比示例
class ExpressionOptimizationComparison {
public:
    // 公共子表达式消除：消除重复计算
    class CommonSubexpressionElimination {
    private:
        map<string, string> expressionCache;
        
    public:
        void eliminateCommonSubexpressions(ASTNode* node) {
            if (node->type == "binary_op") {
                string exprKey = getExpressionKey(node);
                
                if (expressionCache.find(exprKey) != expressionCache.end()) {
                    // 使用缓存的结果
                    string tempVar = expressionCache[exprKey];
                    node->type = "identifier";
                    node->value = tempVar;
                    node->children.clear();
                } else {
                    // 缓存新表达式
                    string tempVar = generateTempVar();
                    expressionCache[exprKey] = tempVar;
                }
            }
            
            for (ASTNode* child : node->children) {
                eliminateCommonSubexpressions(child);
            }
        }
        
        string getExpressionKey(ASTNode* node) {
            // 生成表达式的唯一标识
            return node->toString();
        }
        
        string generateTempVar() {
            static int counter = 0;
            return "t" + to_string(counter++);
        }
    };
    
    // 强度削弱：用更高效的操作替换低效操作
    class StrengthReduction {
    public:
        void reduceStrength(ASTNode* node) {
            if (node->type == "binary_op") {
                if (node->op == "*") {
                    ASTNode* left = node->children[0];
                    ASTNode* right = node->children[1];
                    
                    // 乘法替换为移位
                    if (left->type == "number" && isPowerOfTwo(left->value)) {
                        node->op = "<<";
                        left->value = log2(left->value);
                    } else if (right->type == "number" && isPowerOfTwo(right->value)) {
                        node->op = "<<";
                        right->value = log2(right->value);
                    }
                } else if (node->op == "/") {
                    ASTNode* right = node->children[1];
                    
                    // 除法替换为移位
                    if (right->type == "number" && isPowerOfTwo(right->value)) {
                        node->op = ">>";
                        right->value = log2(right->value);
                    }
                }
            }
            
            for (ASTNode* child : node->children) {
                reduceStrength(child);
            }
        }
        
        bool isPowerOfTwo(int n) {
            return n > 0 && (n & (n - 1)) == 0;
        }
        
        int log2(int n) {
            int result = 0;
            while (n > 1) {
                n >>= 1;
                result++;
            }
            return result;
        }
    };
};
```

### 数据流优化技术对比

#### 活跃变量分析 vs 到达定值分析对比

| 特性 | 活跃变量分析 | 到达定值分析 | 优势分析 |
|------|--------------|--------------|----------|
| **分析方向** | 向后分析 | 向前分析 | 分析方向不同 |
| **分析目标** | 变量活跃性 | 定值可达性 | 目标不同 |
| **应用场景** | 寄存器分配 | 常量传播 | 应用场景不同 |
| **算法复杂度** | O(n×m) | O(n×m) | 复杂度相同 |
| **实现难度** | 中等 | 中等 | 实现难度相同 |

```c
// 活跃变量分析 vs 到达定值分析对比示例
class DataFlowOptimizationComparison {
public:
    // 活跃变量分析：向后分析变量活跃性
    class LiveVariableAnalysis {
    private:
        map<BasicBlock*, set<string>> liveIn;
        map<BasicBlock*, set<string>> liveOut;
        map<BasicBlock*, set<string>> def;
        map<BasicBlock*, set<string>> use;
        
    public:
        void analyze(ControlFlowGraph* cfg) {
            initialize(cfg);
            
            bool changed = true;
            while (changed) {
                changed = false;
                
                for (BasicBlock* block : cfg->getBlocks()) {
                    set<string> oldLiveIn = liveIn[block];
                    set<string> oldLiveOut = liveOut[block];
                    
                    // LiveOut[B] = ∪(LiveIn[S]) for all successors S of B
                    liveOut[block].clear();
                    for (BasicBlock* successor : block->getSuccessors()) {
                        liveOut[block].insert(liveIn[successor].begin(), 
                                            liveIn[successor].end());
                    }
                    
                    // LiveIn[B] = Use[B] ∪ (LiveOut[B] - Def[B])
                    liveIn[block] = use[block];
                    for (string var : liveOut[block]) {
                        if (def[block].find(var) == def[block].end()) {
                            liveIn[block].insert(var);
                        }
                    }
                    
                    if (oldLiveIn != liveIn[block] || oldLiveOut != liveOut[block]) {
                        changed = true;
                    }
                }
            }
        }
        
        void initialize(ControlFlowGraph* cfg) {
            for (BasicBlock* block : cfg->getBlocks()) {
                liveIn[block] = set<string>();
                liveOut[block] = set<string>();
                def[block] = computeDef(block);
                use[block] = computeUse(block);
            }
        }
    };
    
    // 到达定值分析：向前分析定值可达性
    class ReachingDefinitionAnalysis {
    private:
        map<BasicBlock*, set<int>> reachIn;
        map<BasicBlock*, set<int>> reachOut;
        map<BasicBlock*, set<int>> gen;
        map<BasicBlock*, set<int>> kill;
        
    public:
        void analyze(ControlFlowGraph* cfg) {
            initialize(cfg);
            
            bool changed = true;
            while (changed) {
                changed = false;
                
                for (BasicBlock* block : cfg->getBlocks()) {
                    set<int> oldReachIn = reachIn[block];
                    set<int> oldReachOut = reachOut[block];
                    
                    // ReachIn[B] = ∪(ReachOut[P]) for all predecessors P of B
                    reachIn[block].clear();
                    for (BasicBlock* predecessor : block->getPredecessors()) {
                        reachIn[block].insert(reachOut[predecessor].begin(), 
                                            reachOut[predecessor].end());
                    }
                    
                    // ReachOut[B] = Gen[B] ∪ (ReachIn[B] - Kill[B])
                    reachOut[block] = gen[block];
                    for (int def : reachIn[block]) {
                        if (kill[block].find(def) == kill[block].end()) {
                            reachOut[block].insert(def);
                        }
                    }
                    
                    if (oldReachIn != reachIn[block] || oldReachOut != reachOut[block]) {
                        changed = true;
                    }
                }
            }
        }
        
        void initialize(ControlFlowGraph* cfg) {
            for (BasicBlock* block : cfg->getBlocks()) {
                reachIn[block] = set<int>();
                reachOut[block] = set<int>();
                gen[block] = computeGen(block);
                kill[block] = computeKill(block);
            }
        }
    };
};
```

// 函数内联 vs 过程间常量传播对比示例
class InterproceduralOptimizationComparison {
public:
    // 函数内联：将函数调用替换为函数体

            if (node->type == "function_call") {
                string functionName = node->children[0]->value;
                FunctionDefinition* func = getFunctionDefinition(functionName);

                inlineFunctions(child);
            }

            // 判断是否应该内联
            return func->body->getStatementCount() < 10;
        }
        
        ASTNode* createInlinedBody(FunctionDefinition* func, vector<ASTNode*> args) {
            // 创建内联后的函数体
            return func->body; // 简化实现
        }
    };
    
    // 过程间常量传播：跨函数传播常量值
    class InterproceduralConstantPropagation {
    private:
        map<string, int> globalConstants;
        
    public:
        void propagateConstants(ASTNode* node) {
            if (node->type == "function_call") {
                string functionName = node->children[0]->value;
                vector<ASTNode*> args = node->children;
                
                // 传播常量参数
                propagateArguments(functionName, args);
            }
            
            for (ASTNode* child : node->children) {
                propagateConstants(child);
            }
        }
        
        void propagateArguments(string functionName, vector<ASTNode*> args) {
            FunctionDefinition* func = getFunctionDefinition(functionName);
            vector<string> params = func->parameters;
            
            for (int i = 0; i < args.size() && i < params.size(); i++) {
                if (args[i]->type == "number") {
                    globalConstants[params[i]] = args[i]->value;
                }
            }
        }
    };
};
