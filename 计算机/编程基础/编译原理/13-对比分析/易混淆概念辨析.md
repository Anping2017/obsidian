# 易混淆概念辨析

## 🎯 易混淆概念辨析概述

### 有限自动机 vs 正则表达式辨析

| 特性 | 有限自动机 | 正则表达式 | 辨析要点 |
|------|------------|------------|----------|
| **定义** | 状态转换图 | 字符串模式 | 定义不同 |
| **表示方式** | 图形化 | 文本化 | 表示方式不同 |
| **表达能力** | 强 | 中等 | 有限自动机表达能力更强 |
| **实现复杂度** | 高 | 低 | 正则表达式实现更简单 |
| **应用场景** | 词法分析器 | 模式匹配 | 应用场景不同 |

```c
// 有限自动机 vs 正则表达式辨析示例
class FiniteAutomataVsRegexComparison {
public:
    // 有限自动机：状态转换图
    class FiniteAutomata {
    private:
        set<int> states;
        set<char> alphabet;
        map<pair<int, char>, int> transitions;
        int startState;
        set<int> acceptStates;
        
    public:
        bool accepts(const string& input) {
            int currentState = startState;
            
            for (char c : input) {
                auto key = make_pair(currentState, c);
                if (transitions.find(key) != transitions.end()) {
                    currentState = transitions[key];
                } else {
                    return false;
                }
            }
            
            return acceptStates.find(currentState) != acceptStates.end();
        }
    };
    
    // 正则表达式：字符串模式
    class RegularExpression {
    private:
        string pattern;
        
    public:
        bool matches(const string& input) {
            // 简化的正则表达式匹配
            return regex_match(input, regex(pattern));
        }
    };
};
```

### DFA vs NFA辨析

| 特性 | DFA | NFA | 辨析要点 |
|------|-----|-----|----------|
| **状态转换** | 确定性 | 非确定性 | 状态转换不同 |
| **实现复杂度** | 简单 | 复杂 | DFA实现更简单 |
| **执行效率** | 高 | 低 | DFA执行效率更高 |
| **状态数量** | 多 | 少 | DFA状态数量更多 |
| **应用场景** | 生产环境 | 开发环境 | 应用场景不同 |

```c
// DFA vs NFA辨析示例
class DFAVsNFAComparison {
public:
    // DFA：确定性有限自动机
    class DFA {
    private:
        map<pair<int, char>, int> transitions;
        
    public:
        bool accepts(const string& input) {
            int currentState = 0;
            
            for (char c : input) {
                auto key = make_pair(currentState, c);
                if (transitions.find(key) != transitions.end()) {
                    currentState = transitions[key];
                } else {
                    return false;
                }
            }
            
            return currentState == 1; // 接受状态
        }
    };
    
    // NFA：非确定性有限自动机
    class NFA {
    private:
        map<pair<int, char>, set<int>> transitions;
        
    public:
        bool accepts(const string& input) {
            set<int> currentStates = {0};
            
            for (char c : input) {
                set<int> nextStates;
                for (int state : currentStates) {
                    auto key = make_pair(state, c);
                    if (transitions.find(key) != transitions.end()) {
                        for (int nextState : transitions[key]) {
                            nextStates.insert(nextState);
                        }
                    }
                }
                currentStates = nextStates;
            }
            
            return currentStates.find(1) != currentStates.end(); // 接受状态
        }
    };
};
```

### LL(1) vs LR(1)辨析

| 特性 | LL(1) | LR(1) | 辨析要点 |
|------|-------|-------|----------|
| **分析方向** | 自顶向下 | 自底向上 | 分析方向不同 |
| **文法类型** | LL(1)文法 | LR(1)文法 | 文法类型不同 |
| **分析表大小** | 小 | 大 | LL(1)分析表更小 |
| **错误恢复** | 困难 | 容易 | LR(1)错误恢复更容易 |
| **实现复杂度** | 简单 | 复杂 | LL(1)实现更简单 |

```c
// LL(1) vs LR(1)辨析示例
class LL1VsLR1Comparison {
public:
    // LL(1)分析器：自顶向下分析
    class LL1Parser {
    private:
        map<pair<string, string>, string> parseTable;
        
    public:
        bool parse(const vector<Token>& tokens) {
            stack<string> parseStack;
            parseStack.push("$");
            parseStack.push("S");
            
            int tokenIndex = 0;
            while (!parseStack.empty()) {
                string top = parseStack.top();
                parseStack.pop();
                
                if (top == "$") {
                    return tokenIndex == tokens.size();
                } else if (top == tokens[tokenIndex].type) {
                    tokenIndex++;
                } else {
                    auto key = make_pair(top, tokens[tokenIndex].type);
                    if (parseTable.find(key) != parseTable.end()) {
                        string production = parseTable[key];
                        // 处理产生式
                    } else {
                        return false;
                    }
                }
            }
            
            return true;
        }
    };
    
    // LR(1)分析器：自底向上分析
    class LR1Parser {
    private:
        map<pair<int, string>, string> actionTable;
        map<pair<int, string>, int> gotoTable;
        
    public:
        bool parse(const vector<Token>& tokens) {
            stack<int> stateStack;
            stateStack.push(0);
            
            int tokenIndex = 0;
            while (tokenIndex < tokens.size()) {
                int currentState = stateStack.top();
                string currentToken = tokens[tokenIndex].type;
                
                auto key = make_pair(currentState, currentToken);
                if (actionTable.find(key) != actionTable.end()) {
                    string action = actionTable[key];
                    if (action[0] == 's') {
                        // 移进
                        int nextState = stoi(action.substr(1));
                        stateStack.push(nextState);
                        tokenIndex++;
                    } else if (action[0] == 'r') {
                        // 归约
                        int production = stoi(action.substr(1));
                        // 处理归约
                    } else if (action == "accept") {
                        return true;
                    }
                } else {
                    return false;
                }
            }
            
            return false;
        }
    };
};
```

### 类型检查 vs 类型推断辨析

| 特性 | 类型检查 | 类型推断 | 辨析要点 |

| **实现复杂度** | 简单 | 复杂 | 类型推断实现更复杂 |
| **应用场景** | 静态类型语言 | 动态类型语言 | 应用场景不同 |

```c
// 类型检查 vs 类型推断辨析示例
class TypeCheckingVsTypeInferenceComparison {
public:
    // 类型检查：验证类型正确性
    class TypeChecker {
    public:
        bool checkType(ASTNode* node, const string& expectedType) {
            string actualType = getNodeType(node);
            return actualType == expectedType;
        }
        
        string getNodeType(ASTNode* node) {
            if (node->type == "number") {
                return "int";
            } else if (node->type == "string") {
                return "string";
            } else if (node->type == "add") {
                string leftType = getNodeType(node->children[0]);
                string rightType = getNodeType(node->children[1]);
                if (leftType == "int" && rightType == "int") {
                    return "int";
                }
            }
            return "unknown";
        }
    };
    
    // 类型推断：推导类型
    class TypeInference {
    public:
        string inferType(ASTNode* node) {
            if (node->type == "number") {
                return "int";
            } else if (node->type == "string") {
                return "string";
            } else if (node->type == "add") {
                string leftType = inferType(node->children[0]);
                string rightType = inferType(node->children[1]);
                if (leftType == "int" && rightType == "int") {
                    return "int";
                } else if (leftType == "string" && rightType == "string") {
                    return "string";
                }
            }
            return "unknown";
        }
    };
};
```

class InstructionSelectionVsSchedulingComparison {
public:
    // 指令选择：选择指令
    class InstructionSelector {
    public:
        string selectInstruction(const string& operation, 
                               const vector<string>& operands) {
            if (operation == "add" && operands.size() == 2) {
                return "add " + operands[0] + ", " + operands[1];
            } else if (operation == "sub" && operands.size() == 2) {
                return "sub " + operands[0] + ", " + operands[1];
            } else if (operation == "mul" && operands.size() == 2) {
                return "mul " + operands[0] + ", " + operands[1];
            } else if (operation == "div" && operands.size() == 2) {
                return "div " + operands[0] + ", " + operands[1];
            }
            return "";
        }
    };
    
    // 指令调度：调度指令
    class InstructionScheduler {
    public:
        vector<string> scheduleInstructions(const vector<string>& instructions) {
            vector<string> scheduled;
            
            // 简单的指令调度算法
            for (const string& instruction : instructions) {
                if (instruction.find("load") != string::npos) {
                    // 加载指令优先调度
                    scheduled.insert(scheduled.begin(), instruction);
                } else if (instruction.find("store") != string::npos) {
                    // 存储指令延后调度
                    scheduled.push_back(instruction);
                } else {
                    // 其他指令按顺序调度
                    scheduled.push_back(instruction);
                }
            }
            
            return scheduled;
        }
    };
};

    private:
        set<string> availableRegisters;
        map<string, string> variableToRegister;
        
    public:
        string allocateRegister(const string& variable) {
            if (variableToRegister.find(variable) != variableToRegister.end()) {
                return variableToRegister[variable];
            }
            
            if (!availableRegisters.empty()) {
                string register = *availableRegisters.begin();
                availableRegisters.erase(register);
                variableToRegister[variable] = register;
                return register;
            }
            
            return ""; // 没有可用寄存器

        void deallocateRegister(const string& variable) {
