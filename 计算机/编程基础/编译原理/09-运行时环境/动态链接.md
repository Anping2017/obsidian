# 动态链接

## 🎯 动态链接概述

**动态链接**是编译原理在程序运行时模块加载中的应用，涉及共享库、符号解析、延迟加载、动态库管理等动态链接技术。动态链接需要处理复杂的模块依赖、符号解析和内存管理。

## 🔍 基本概念

### 动态链接定义
动态链接是指程序运行时动态加载和链接共享库的过程，实现代码共享和模块化。

### 动态链接特点
- **延迟加载**：需要延迟加载模块
- **符号解析**：需要解析符号引用
- **内存共享**：需要共享内存空间
- **版本管理**：需要管理库版本

## 🔧 动态链接器

### 动态链接器实现
```c
class DynamicLinker {
private:
    map<string, DynamicLibrary*> loadedLibraries;
    map<string, void*> symbolTable;
    vector<string> librarySearchPaths;
    bool isRunning;
    
public:
    DynamicLinker() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化动态链接器" << endl;
        isRunning = true;
        
        // 添加默认搜索路径
        librarySearchPaths.push_back("/usr/lib");
        librarySearchPaths.push_back("/usr/local/lib");
        librarySearchPaths.push_back("./");
    }
    
    void addSearchPath(const string& path) {
        librarySearchPaths.push_back(path);
        cout << "添加库搜索路径: " << path << endl;
    }
    
    DynamicLibrary* loadLibrary(const string& libraryName) {
        cout << "加载动态库: " << libraryName << endl;
        
        // 检查是否已经加载
        if (loadedLibraries.find(libraryName) != loadedLibraries.end()) {
            cout << "库已加载: " << libraryName << endl;
            return loadedLibraries[libraryName];
        }
        
        // 查找库文件
        string libraryPath = findLibrary(libraryName);
        if (libraryPath.empty()) {
            cout << "未找到库文件: " << libraryName << endl;
            return nullptr;
        }
        
        // 创建动态库对象
        DynamicLibrary* library = new DynamicLibrary(libraryName, libraryPath);
        
        // 加载库
        if (library->load()) {
            loadedLibraries[libraryName] = library;
            cout << "库加载成功: " << libraryName << endl;
            return library;
        } else {
            cout << "库加载失败: " << libraryName << endl;
            delete library;
            return nullptr;
        }
    }
    
    string findLibrary(const string& libraryName) {
        cout << "查找库文件: " << libraryName << endl;
        
        for (const auto& path : librarySearchPaths) {
            string fullPath = path + "/" + libraryName;
            if (fileExists(fullPath)) {
                cout << "找到库文件: " << fullPath << endl;
                return fullPath;
            }
        }
        
        return "";
    }
    
    bool fileExists(const string& path) {
        // 简化实现：模拟文件存在检查
        return path.find("lib") != string::npos;
    }
    
    void* resolveSymbol(const string& symbolName) {
        cout << "解析符号: " << symbolName << endl;
        
        // 检查符号表
        if (symbolTable.find(symbolName) != symbolTable.end()) {
            cout << "符号已解析: " << symbolName << endl;
            return symbolTable[symbolName];
        }
        
        // 在所有加载的库中查找符号
        for (auto& pair : loadedLibraries) {
            void* symbol = pair.second->getSymbol(symbolName);
            if (symbol) {
                symbolTable[symbolName] = symbol;
                cout << "符号解析成功: " << symbolName << endl;
                return symbol;
            }
        }
        
        cout << "符号解析失败: " << symbolName << endl;
        return nullptr;
    }
    
    void unloadLibrary(const string& libraryName) {
        cout << "卸载动态库: " << libraryName << endl;
        
        auto it = loadedLibraries.find(libraryName);
        if (it != loadedLibraries.end()) {
            it->second->unload();
            delete it->second;
            loadedLibraries.erase(it);
            cout << "库卸载成功: " << libraryName << endl;
        } else {
            cout << "库未加载: " << libraryName << endl;
        }
    }
    
    void printLoadedLibraries() {
        cout << "已加载的库" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pair : loadedLibraries) {
            cout << "库名: " << pair.first << endl;
            cout << "路径: " << pair.second->getPath() << endl;
            cout << "状态: " << (pair.second->isLoaded() ? "已加载" : "未加载") << endl;
        }
        
        cout << endl;
    }
    
    void printSymbolTable() {
        cout << "符号表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pair : symbolTable) {
            cout << "符号: " << pair.first << " -> " << pair.second << endl;
        }
        
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理动态链接器" << endl;
        
        // 卸载所有库
        for (auto& pair : loadedLibraries) {
            pair.second->unload();
            delete pair.second;
        }
        loadedLibraries.clear();
        
        symbolTable.clear();
        librarySearchPaths.clear();
        
        isRunning = false;
        
        cout << "动态链接器清理完成" << endl;
    }
    
    ~DynamicLinker() {
        cleanup();
    }
};

class DynamicLibrary {
private:
    string name;
    string path;
    bool loaded;
    map<string, void*> symbols;
    
public:
    DynamicLibrary(const string& n, const string& p) : name(n), path(p), loaded(false) {}
    
    bool load() {
        cout << "加载库: " << name << endl;
        
        // 模拟加载过程
        cout << "打开库文件: " << path << endl;
        cout << "解析库头" << endl;
        cout << "加载符号表" << endl;
        
        // 模拟符号加载
        symbols["function1"] = (void*)0x1000;
        symbols["function2"] = (void*)0x2000;
        symbols["variable1"] = (void*)0x3000;
        
        loaded = true;
        cout << "库加载完成: " << name << endl;
        
        return true;
    }
    
    void unload() {
        cout << "卸载库: " << name << endl;
        
        symbols.clear();
        loaded = false;
        
        cout << "库卸载完成: " << name << endl;
    }
    
    void* getSymbol(const string& symbolName) {
        if (symbols.find(symbolName) != symbols.end()) {
            cout << "找到符号: " << symbolName << " -> " << symbols[symbolName] << endl;
            return symbols[symbolName];
        }
        
        cout << "未找到符号: " << symbolName << endl;
        return nullptr;
    }
    
    string getName() { return name; }
    string getPath() { return path; }
    bool isLoaded() { return loaded; }
};
```

## 🔧 符号解析器

### 符号解析器实现
```c
class SymbolResolver {
private:
    map<string, SymbolInfo> symbolTable;
    vector<SymbolReference> unresolvedReferences;
    bool isRunning;
    
public:
    SymbolResolver() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化符号解析器" << endl;
        isRunning = true;
    }
    
    void addSymbol(const string& name, void* address, SymbolType type) {
        cout << "添加符号: " << name << " -> " << address << endl;
        
        SymbolInfo info;
        info.name = name;
        info.address = address;
        info.type = type;
        info.resolved = true;
        
        symbolTable[name] = info;
    }
    
    void addReference(const string& name, SymbolType type) {
        cout << "添加符号引用: " << name << endl;
        
        SymbolReference ref;
        ref.name = name;
        ref.type = type;
        ref.resolved = false;
        
        unresolvedReferences.push_back(ref);
    }
    
    bool resolveSymbol(const string& name) {
        cout << "解析符号: " << name << endl;
        
        // 检查是否已经解析
        if (symbolTable.find(name) != symbolTable.end()) {
            cout << "符号已解析: " << name << endl;
            return true;
        }
        
        // 尝试解析符号
        void* address = resolveSymbolAddress(name);
        if (address) {
            addSymbol(name, address, getSymbolType(name));
            cout << "符号解析成功: " << name << endl;
            return true;
        } else {
            cout << "符号解析失败: " << name << endl;
            return false;
        }
    }
    
    void* resolveSymbolAddress(const string& name) {
        // 简化实现：模拟符号地址解析
        if (name.find("function") != string::npos) {
            return (void*)0x1000;
        } else if (name.find("variable") != string::npos) {
            return (void*)0x2000;
        } else if (name.find("class") != string::npos) {
            return (void*)0x3000;
        }
        
        return nullptr;
    }
    
    SymbolType getSymbolType(const string& name) {
        if (name.find("function") != string::npos) {
            return FUNCTION;
        } else if (name.find("variable") != string::npos) {
            return VARIABLE;
        } else if (name.find("class") != string::npos) {
            return CLASS;
        }
        
        return UNKNOWN;
    }
    
    string getTypeName(SymbolType type) {
        switch (type) {
            case FUNCTION: return "函数";
            case VARIABLE: return "变量";
            case CLASS: return "类";
            case UNKNOWN: return "未知";
            default: return "未知";
        }
    }
    
    void resolveAllReferences() {
        cout << "解析所有符号引用" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& ref : unresolvedReferences) {
            if (!ref.resolved) {
                if (resolveSymbol(ref.name)) {
                    ref.resolved = true;
                }
            }
        }
        
        cout << "符号引用解析完成" << endl;
    }
    
    void printSymbolTable() {
        cout << "符号表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pair : symbolTable) {
            const auto& symbol = pair.second;
            cout << "符号: " << symbol.name << endl;
            cout << "  地址: " << symbol.address << endl;
            cout << "  类型: " << getTypeName(symbol.type) << endl;
            cout << "  已解析: " << (symbol.resolved ? "是" : "否") << endl;
        }
        
        cout << endl;
    }
    
    void printUnresolvedReferences() {
        cout << "未解析的符号引用" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& ref : unresolvedReferences) {
            if (!ref.resolved) {
                cout << "符号: " << ref.name << endl;
                cout << "  类型: " << getTypeName(ref.type) << endl;
            }
        }
        
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理符号解析器" << endl;
        
        symbolTable.clear();
        unresolvedReferences.clear();
        
        isRunning = false;
        
        cout << "符号解析器清理完成" << endl;
    }
    
    ~SymbolResolver() {
        cleanup();
    }
};

enum SymbolType {
    FUNCTION,
    VARIABLE,
    CLASS,
    UNKNOWN
};

struct SymbolInfo {
    string name;
    void* address;
    SymbolType type;
    bool resolved;
};

struct SymbolReference {
    string name;
    SymbolType type;
    bool resolved;
};
```

## 🔧 延迟加载器

### 延迟加载器实现
```c
class LazyLoader {
private:
    map<string, LazyLoadInfo> lazyLoadTable;
    vector<string> loadQueue;
    bool isRunning;
    
public:
    LazyLoader() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化延迟加载器" << endl;
        isRunning = true;
    }
    
    void registerLazyLoad(const string& name, function<void()> loadFunction) {
        cout << "注册延迟加载: " << name << endl;
        
        LazyLoadInfo info;
        info.name = name;
        info.loadFunction = loadFunction;
        info.loaded = false;
        info.priority = 0;
        
        lazyLoadTable[name] = info;
    }
    
    void setLoadPriority(const string& name, int priority) {
        if (lazyLoadTable.find(name) != lazyLoadTable.end()) {
            lazyLoadTable[name].priority = priority;
            cout << "设置加载优先级: " << name << " -> " << priority << endl;
        }
    }
    
    void loadOnDemand(const string& name) {
        cout << "按需加载: " << name << endl;
        
        if (lazyLoadTable.find(name) != lazyLoadTable.end()) {
            LazyLoadInfo& info = lazyLoadTable[name];
            
            if (!info.loaded) {
                cout << "执行延迟加载: " << name << endl;
                info.loadFunction();
                info.loaded = true;
                cout << "延迟加载完成: " << name << endl;
            } else {
                cout << "已加载: " << name << endl;
            }
        } else {
            cout << "未注册延迟加载: " << name << endl;
        }
    }
    
    void loadAll() {
        cout << "加载所有延迟加载项" << endl;
        cout << "-------------------------------------------" << endl;
        
        // 按优先级排序
        vector<pair<string, LazyLoadInfo*>> sortedItems;
        for (auto& pair : lazyLoadTable) {
            sortedItems.push_back({pair.first, &pair.second});
        }
        
        sort(sortedItems.begin(), sortedItems.end(),
            [](const pair<string, LazyLoadInfo*>& a, const pair<string, LazyLoadInfo*>& b) {
                return a.second->priority > b.second->priority;
            });
        
        // 加载所有项
        for (auto& item : sortedItems) {
            if (!item.second->loaded) {
                cout << "加载: " << item.first << endl;
                item.second->loadFunction();
                item.second->loaded = true;
            }
        }
        
        cout << "所有延迟加载项加载完成" << endl;
    }
    
    void loadByPriority(int priority) {
        cout << "按优先级加载: " << priority << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pair : lazyLoadTable) {
            if (pair.second.priority == priority && !pair.second.loaded) {
                cout << "加载: " << pair.first << endl;
                pair.second.loadFunction();
                pair.second.loaded = true;
            }
        }
        
        cout << "优先级 " << priority << " 加载完成" << endl;
    }
    
    void printLazyLoadTable() {
        cout << "延迟加载表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pair : lazyLoadTable) {
            const auto& info = pair.second;
            cout << "名称: " << info.name << endl;
            cout << "  优先级: " << info.priority << endl;
            cout << "  已加载: " << (info.loaded ? "是" : "否") << endl;
        }
        
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理延迟加载器" << endl;
        
        lazyLoadTable.clear();
        loadQueue.clear();
        
        isRunning = false;
        
        cout << "延迟加载器清理完成" << endl;
    }
    
    ~LazyLoader() {
        cleanup();
    }
};

struct LazyLoadInfo {
    string name;
    function<void()> loadFunction;
    bool loaded;
    int priority;
};
```

## 🔧 动态链接管理器

### 动态链接管理器实现
```c
class DynamicLinkingManager {
private:
    DynamicLinker* linker;
    SymbolResolver* resolver;
    LazyLoader* lazyLoader;
    bool isRunning;
    
public:
    DynamicLinkingManager() {
        initializeManager();
    }
    
    void initializeManager() {
        cout << "初始化动态链接管理器" << endl;
        
        linker = new DynamicLinker();
        resolver = new SymbolResolver();
        lazyLoader = new LazyLoader();
        
        // 初始化各个组件
        linker->initialize();
        resolver->initialize();
        lazyLoader->initialize();
        
        isRunning = true;
    }
    
    void loadLibrary(const string& libraryName) {
        cout << "动态链接管理器加载库" << endl;
        cout << "===========================================" << endl;
        
        // 加载库
        DynamicLibrary* library = linker->loadLibrary(libraryName);
        
        if (library) {
            cout << "库加载成功: " << libraryName << endl;
            
            // 注册延迟加载
            lazyLoader->registerLazyLoad(libraryName, [this, libraryName]() {
                cout << "延迟加载库: " << libraryName << endl;
                // 这里可以添加具体的延迟加载逻辑
            });
            
            // 设置加载优先级
            lazyLoader->setLoadPriority(libraryName, 1);
            
        } else {
            cout << "库加载失败: " << libraryName << endl;
        }
        
        cout << endl;
    }
    
    void* resolveSymbol(const string& symbolName) {
        cout << "动态链接管理器解析符号" << endl;
        cout << "-------------------------------------------" << endl;
        
        // 解析符号
        void* symbol = linker->resolveSymbol(symbolName);
        
        if (symbol) {
            cout << "符号解析成功: " << symbolName << endl;
            
            // 添加到符号表
            resolver->addSymbol(symbolName, symbol, getSymbolType(symbolName));
            
        } else {
            cout << "符号解析失败: " << symbolName << endl;
            
            // 添加到未解析引用
            resolver->addReference(symbolName, getSymbolType(symbolName));
        }
        
        return symbol;
    }
    
    SymbolType getSymbolType(const string& symbolName) {
        if (symbolName.find("function") != string::npos) {
            return FUNCTION;
        } else if (symbolName.find("variable") != string::npos) {
            return VARIABLE;
        } else if (symbolName.find("class") != string::npos) {
            return CLASS;
        }
        
        return UNKNOWN;
    }
    
    void loadOnDemand(const string& name) {
        lazyLoader->loadOnDemand(name);
    }
    
    void printDynamicLinkingInfo() {
        cout << "动态链接管理器信息" << endl;
        cout << "===========================================" << endl;
        
        linker->printLoadedLibraries();
        linker->printSymbolTable();
        resolver->printSymbolTable();
        resolver->printUnresolvedReferences();
        lazyLoader->printLazyLoadTable();
    }
    
    void cleanup() {
        cout << "清理动态链接管理器" << endl;
        
        if (linker) {
            delete linker;
            linker = nullptr;
        }
        
        if (resolver) {
            delete resolver;
            resolver = nullptr;
        }
        
        if (lazyLoader) {
            delete lazyLoader;
            lazyLoader = nullptr;
        }
        
        isRunning = false;
        
        cout << "动态链接管理器清理完成" << endl;
    }
    
    ~DynamicLinkingManager() {
        cleanup();
    }
};
```
