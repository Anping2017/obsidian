# 运行时环境概述

## 🎯 运行时环境概述

**运行时环境**是编译原理在程序运行时的应用，涉及内存管理、垃圾回收、异常处理、动态链接等运行时环境技术。运行时环境需要处理复杂的程序执行、资源管理和错误处理。

## 🔍 基本概念

### 运行时环境定义
运行时环境是指程序在目标机器上运行时所依赖的环境，包括操作系统、硬件平台、运行时库等。

### 运行时环境特点
- **内存管理**：需要管理程序内存
- **垃圾回收**：需要回收无用内存
- **异常处理**：需要处理程序异常
- **动态链接**：需要支持动态链接

## 🔧 内存管理器

### 内存管理器实现
```c
class MemoryManager {
private:
    map<void*, MemoryBlock*> allocatedBlocks;
    vector<MemoryPool*> memoryPools;
    size_t totalAllocated;
    size_t maxAllocated;
    
public:
    MemoryManager() : totalAllocated(0), maxAllocated(0) {}
    
    void initialize() {
        cout << "初始化内存管理器" << endl;
        
        // 创建内存池
        createMemoryPools();
    }
    
    void createMemoryPools() {
        cout << "创建内存池" << endl;
        
        // 创建不同大小的内存池
        memoryPools.push_back(new MemoryPool(16, 1000));    // 16字节池
        memoryPools.push_back(new MemoryPool(32, 1000));    // 32字节池
        memoryPools.push_back(new MemoryPool(64, 1000));     // 64字节池
        memoryPools.push_back(new MemoryPool(128, 1000));   // 128字节池
        memoryPools.push_back(new MemoryPool(256, 1000));   // 256字节池
    }
    
    void* allocate(size_t size) {
        cout << "分配内存: " << size << " 字节" << endl;
        
        // 查找合适的内存池
        MemoryPool* pool = findSuitablePool(size);
        
        if (pool) {
            void* ptr = pool->allocate();
            if (ptr) {
                MemoryBlock* block = new MemoryBlock(ptr, size);
                allocatedBlocks[ptr] = block;
                
                totalAllocated += size;
                if (totalAllocated > maxAllocated) {
                    maxAllocated = totalAllocated;
                }
                
                cout << "内存分配成功: " << ptr << endl;
                return ptr;
            }
        }
        
        // 如果内存池分配失败，使用系统分配
        void* ptr = malloc(size);
        if (ptr) {
            MemoryBlock* block = new MemoryBlock(ptr, size);
            allocatedBlocks[ptr] = block;
            
            totalAllocated += size;
            if (totalAllocated > maxAllocated) {
                maxAllocated = totalAllocated;
            }
            
            cout << "系统内存分配成功: " << ptr << endl;
            return ptr;
        }
        
        cout << "内存分配失败" << endl;
        return nullptr;
    }
    
    void deallocate(void* ptr) {
        cout << "释放内存: " << ptr << endl;
        
        if (allocatedBlocks.find(ptr) != allocatedBlocks.end()) {
            MemoryBlock* block = allocatedBlocks[ptr];
            
            // 从分配表中移除
            allocatedBlocks.erase(ptr);
            
            // 更新统计信息
            totalAllocated -= block->getSize();
            
            // 释放内存
            free(ptr);
            
            delete block;
            
            cout << "内存释放成功" << endl;
        } else {
            cout << "内存指针无效: " << ptr << endl;
        }
    }
    
    MemoryPool* findSuitablePool(size_t size) {
        for (auto& pool : memoryPools) {
            if (pool->getBlockSize() >= size) {
                return pool;
            }
        }
        return nullptr;
    }
    
    void printMemoryStats() {
        cout << "内存统计信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "总分配内存: " << totalAllocated << " 字节" << endl;
        cout << "最大分配内存: " << maxAllocated << " 字节" << endl;
        cout << "当前分配块数: " << allocatedBlocks.size() << endl;
        cout << "内存池数量: " << memoryPools.size() << endl;
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理内存管理器" << endl;
        
        // 释放所有分配的内存
        for (auto& pair : allocatedBlocks) {
            free(pair.first);
            delete pair.second;
        }
        allocatedBlocks.clear();
        
        // 清理内存池
        for (auto& pool : memoryPools) {
            delete pool;
        }
        memoryPools.clear();
        
        totalAllocated = 0;
        maxAllocated = 0;
        
        cout << "内存管理器清理完成" << endl;
    }
    
    ~MemoryManager() {
        cleanup();
    }
};

private:

    size_t size;
    
public:

    void* getPtr() { return ptr; }

class MemoryPool {
private:
    size_t blockSize;
    size_t poolSize;
    vector<void*> freeBlocks;
    void* poolMemory;
    
public:
    MemoryPool(size_t blockSize, size_t poolSize) 
        : blockSize(blockSize), poolSize(poolSize) {
        initializePool();
    }
    
    void initializePool() {
        cout << "初始化内存池: " << blockSize << " 字节" << endl;
        
        // 分配池内存
        poolMemory = malloc(blockSize * poolSize);
        
        // 初始化空闲块列表
        for (size_t i = 0; i < poolSize; i++) {
            void* block = (char*)poolMemory + i * blockSize;
            freeBlocks.push_back(block);
        }
        
        cout << "内存池初始化完成" << endl;
    }
    
    void* allocate() {
        if (!freeBlocks.empty()) {
            void* block = freeBlocks.back();
            freeBlocks.pop_back();
            
            cout << "从内存池分配: " << block << endl;
            return block;
        }
        
        cout << "内存池已满" << endl;
        return nullptr;
    }
    
    void deallocate(void* block) {
        freeBlocks.push_back(block);
        cout << "释放到内存池: " << block << endl;
    }
    
    size_t getBlockSize() { return blockSize; }
    size_t getPoolSize() { return poolSize; }
    size_t getFreeBlocks() { return freeBlocks.size(); }
    
    ~MemoryPool() {
        if (poolMemory) {
            free(poolMemory);
        }
    }
};
```

```c
class GarbageCollector {
private:
    vector<GCObject*> rootObjects;
    vector<GCObject*> allObjects;
    map<GCObject*, int> referenceCounts;
    bool isRunning;
    
public:
    GarbageCollector() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化垃圾回收器" << endl;
        
        isRunning = true;
    }
    
    void registerObject(GCObject* obj) {
        cout << "注册对象: " << obj << endl;
        
        allObjects.push_back(obj);
        referenceCounts[obj] = 0;
    }
    
    void addReference(GCObject* obj) {
        if (referenceCounts.find(obj) != referenceCounts.end()) {
            referenceCounts[obj]++;
            cout << "增加引用计数: " << obj << " -> " << referenceCounts[obj] << endl;
        }
    }
    
    void removeReference(GCObject* obj) {
        if (referenceCounts.find(obj) != referenceCounts.end()) {
            referenceCounts[obj]--;
            cout << "减少引用计数: " << obj << " -> " << referenceCounts[obj] << endl;
            
            if (referenceCounts[obj] <= 0) {
                cout << "对象可回收: " << obj << endl;
            }
        }
    }
    
    void collect() {
        cout << "执行垃圾回收" << endl;
        cout << "-------------------------------------------" << endl;
        
        // 标记阶段
        markPhase();
        
        // 清除阶段
        sweepPhase();
        
        cout << "垃圾回收完成" << endl;
        cout << endl;
    }
    
    void markPhase() {
        cout << "标记阶段" << endl;
        
        // 从根对象开始标记
        for (auto& root : rootObjects) {
            markObject(root);
        }
    }
    
    void markObject(GCObject* obj) {
        if (obj && !obj->isMarked()) {
            obj->setMarked(true);
            cout << "标记对象: " << obj << endl;
            
            // 标记引用的对象
            vector<GCObject*> references = obj->getReferences();
            for (auto& ref : references) {
                markObject(ref);
            }
        }
    }
    
    void sweepPhase() {
        cout << "清除阶段" << endl;
        
        vector<GCObject*> toDelete;
        
        for (auto& obj : allObjects) {
            if (!obj->isMarked()) {
                toDelete.push_back(obj);
            } else {
                obj->setMarked(false);
            }
        }
        
        // 删除未标记的对象
        for (auto& obj : toDelete) {
            cout << "删除对象: " << obj << endl;
            delete obj;
        }
        
        // 从列表中移除已删除的对象
        allObjects.erase(
            remove_if(allObjects.begin(), allObjects.end(),
                [&toDelete](GCObject* obj) {
                    return find(toDelete.begin(), toDelete.end(), obj) != toDelete.end();
                }),
            allObjects.end()
        );
    }
    
    void addRootObject(GCObject* obj) {
        rootObjects.push_back(obj);
        cout << "添加根对象: " << obj << endl;
    }
    
    void removeRootObject(GCObject* obj) {
        rootObjects.erase(
            remove(rootObjects.begin(), rootObjects.end(), obj),
            rootObjects.end()
        );
        cout << "移除根对象: " << obj << endl;
    }
    
    void printGCStats() {
        cout << "垃圾回收统计信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "总对象数: " << allObjects.size() << endl;
        cout << "根对象数: " << rootObjects.size() << endl;
        cout << "引用计数对象数: " << referenceCounts.size() << endl;
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理垃圾回收器" << endl;
        
        // 清理所有对象
        for (auto& obj : allObjects) {
            delete obj;
        }
        allObjects.clear();
        rootObjects.clear();
        referenceCounts.clear();
        
        isRunning = false;
        
        cout << "垃圾回收器清理完成" << endl;
    }
    
    ~GarbageCollector() {
        cleanup();
    }
};

class GCObject {
private:
    bool marked;
    vector<GCObject*> references;
    
public:
    GCObject() : marked(false) {}
    
    void setMarked(bool m) { marked = m; }
    bool isMarked() { return marked; }
    
    void addReference(GCObject* obj) {
        references.push_back(obj);
    }
    
    void removeReference(GCObject* obj) {
        references.erase(
            remove(references.begin(), references.end(), obj),
            references.end()
        );
    }
    
    vector<GCObject*> getReferences() { return references; }
};
```

```c
class DynamicLinker {
private:
    map<string, DynamicLibrary*> loadedLibraries;
    map<string, void*> symbolTable;
    bool isInitialized;
    
public:
    DynamicLinker() : isInitialized(false) {}
    
    void initialize() {
        cout << "初始化动态链接器" << endl;
        
        isInitialized = true;
    }
    
    void linkLibrary(string libraryName) {
        cout << "链接库: " << libraryName << endl;
        
        if (loadedLibraries.find(libraryName) != loadedLibraries.end()) {
            cout << "库已加载: " << libraryName << endl;
            return;
        }
        
        DynamicLibrary* library = new DynamicLibrary(libraryName);
        
        // 加载库
        if (library->load()) {
            loadedLibraries[libraryName] = library;
            
            // 解析符号
            resolveSymbols(library);
            
            cout << "库链接成功: " << libraryName << endl;
        } else {
            cout << "库链接失败: " << libraryName << endl;
            delete library;
        }
    }
    
    void unlinkLibrary(string libraryName) {
        cout << "取消链接库: " << libraryName << endl;
        
        if (loadedLibraries.find(libraryName) != loadedLibraries.end()) {
            DynamicLibrary* library = loadedLibraries[libraryName];
            
            // 卸载库
            library->unload();
            
            // 从加载列表中移除
            loadedLibraries.erase(libraryName);
            
            delete library;
            
            cout << "库取消链接成功: " << libraryName << endl;
        } else {
            cout << "库未加载: " << libraryName << endl;
        }
    }
    
    void resolveSymbols(DynamicLibrary* library) {
        cout << "解析符号: " << library->getName() << endl;
        
        vector<string> symbols = library->getSymbols();
        
        for (auto& symbol : symbols) {
            void* address = library->getSymbolAddress(symbol);
            if (address) {
                symbolTable[symbol] = address;
                cout << "解析符号: " << symbol << " -> " << address << endl;
            }
        }
    }
    
    void* getSymbol(string symbolName) {
        if (symbolTable.find(symbolName) != symbolTable.end()) {
            return symbolTable[symbolName];
        }
        
        cout << "符号未找到: " << symbolName << endl;
        return nullptr;
    }
    
    void printLoadedLibraries() {
        cout << "已加载的库" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pair : loadedLibraries) {
            cout << "库名: " << pair.first << endl;
        }
        
        cout << endl;
    }
    
    void printSymbolTable() {
        cout << "符号表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pair : symbolTable) {
            cout << "符号: " << pair.first << " -> " << pair.second << endl;
        }
        
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理动态链接器" << endl;
        
        // 卸载所有库
        for (auto& pair : loadedLibraries) {
            pair.second->unload();
            delete pair.second;
        }
        loadedLibraries.clear();
        symbolTable.clear();
        
        isInitialized = false;
        
        cout << "动态链接器清理完成" << endl;
    }
    
    ~DynamicLinker() {
        cleanup();
    }
};
```
