# 垃圾回收

## 🎯 垃圾回收概述

**垃圾回收**是编译原理在程序运行时自动内存管理中的应用，涉及标记清除、引用计数、分代回收、增量回收等垃圾回收技术。垃圾回收需要处理复杂的内存回收策略、性能优化和并发安全。

## 🔍 基本概念

### 垃圾回收定义
垃圾回收是指程序运行时自动识别和回收不再使用的内存对象的过程，避免内存泄漏和提高程序安全性。

### 垃圾回收特点
- **自动回收**：需要自动识别无用对象
- **性能优化**：需要优化回收性能
- **并发安全**：需要支持并发回收
- **内存整理**：需要整理内存碎片

### 标记清除垃圾回收器实现

```c
class MarkSweepGC {
private:
    vector<GCObject*> rootObjects;
    vector<GCObject*> allObjects;
    bool isRunning;
    
public:
    MarkSweepGC() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化标记清除垃圾回收器" << endl;
        isRunning = true;
    }
    
    void registerObject(GCObject* obj) {
        cout << "注册对象: " << obj << endl;
        allObjects.push_back(obj);
    }
    
    void addRootObject(GCObject* obj) {
        rootObjects.push_back(obj);
        cout << "添加根对象: " << obj << endl;
    }
    
    void removeRootObject(GCObject* obj) {
        rootObjects.erase(
            remove(rootObjects.begin(), rootObjects.end(), obj),
            rootObjects.end()
        );
        cout << "移除根对象: " << obj << endl;
    }
    
    void collect() {
        cout << "执行标记清除垃圾回收" << endl;
        cout << "===========================================" << endl;
        
        // 标记阶段
        markPhase();
        
        // 清除阶段
        sweepPhase();
        
        cout << "标记清除垃圾回收完成" << endl;
        cout << endl;
    }
    
    void markPhase() {
        cout << "标记阶段" << endl;
        cout << "-------------------------------------------" << endl;
        
        // 从根对象开始标记
        for (auto& root : rootObjects) {
            markObject(root);
        }
        
        cout << "标记阶段完成" << endl;
    }
    
    void markObject(GCObject* obj) {
        if (obj && !obj->isMarked()) {
            obj->setMarked(true);
            cout << "标记对象: " << obj << endl;
            
            // 标记引用的对象
            vector<GCObject*> references = obj->getReferences();
            for (auto& ref : references) {
                markObject(ref);
            }
        }
    }
    
    void sweepPhase() {
        cout << "清除阶段" << endl;
        cout << "-------------------------------------------" << endl;
        
        vector<GCObject*> toDelete;
        
        for (auto& obj : allObjects) {
            if (!obj->isMarked()) {
                toDelete.push_back(obj);
            } else {
                obj->setMarked(false);
            }
        }
        
        // 删除未标记的对象
        for (auto& obj : toDelete) {
            cout << "删除对象: " << obj << endl;
            delete obj;
        }
        
        // 从列表中移除已删除的对象
        allObjects.erase(
            remove_if(allObjects.begin(), allObjects.end(),
                [&toDelete](GCObject* obj) {
                    return find(toDelete.begin(), toDelete.end(), obj) != toDelete.end();
                }),
            allObjects.end()
        );
        
        cout << "清除阶段完成" << endl;
    }
    
    void printGCStats() {
        cout << "垃圾回收统计信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "总对象数: " << allObjects.size() << endl;
        cout << "根对象数: " << rootObjects.size() << endl;
        cout << "已标记对象数: " << countMarkedObjects() << endl;
        cout << endl;
    }
    
    int countMarkedObjects() {
        int count = 0;
        for (auto& obj : allObjects) {
            if (obj->isMarked()) {
                count++;
            }
        }
        return count;
    }
    
    void cleanup() {
        cout << "清理垃圾回收器" << endl;
        
        // 清理所有对象
        for (auto& obj : allObjects) {
            delete obj;
        }
        allObjects.clear();
        rootObjects.clear();
        
        isRunning = false;
        
        cout << "垃圾回收器清理完成" << endl;
    }
    
    ~MarkSweepGC() {
        cleanup();
    }
};
```

## 🔧 引用计数垃圾回收器

### 引用计数垃圾回收器实现

```c
class ReferenceCountingGC {
private:
    map<GCObject*, int> referenceCounts;
    vector<GCObject*> allObjects;
    bool isRunning;
    
public:
    ReferenceCountingGC() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化引用计数垃圾回收器" << endl;
        isRunning = true;
    }
    
    void registerObject(GCObject* obj) {
        cout << "注册对象: " << obj << endl;
        allObjects.push_back(obj);
        referenceCounts[obj] = 0;
    }
    
    void addReference(GCObject* obj) {
        if (referenceCounts.find(obj) != referenceCounts.end()) {
            referenceCounts[obj]++;
            cout << "增加引用计数: " << obj << " -> " << referenceCounts[obj] << endl;
        }
    }
    
    void removeReference(GCObject* obj) {
        if (referenceCounts.find(obj) != referenceCounts.end()) {
            referenceCounts[obj]--;
            cout << "减少引用计数: " << obj << " -> " << referenceCounts[obj] << endl;
            
            if (referenceCounts[obj] <= 0) {
                cout << "对象可回收: " << obj << endl;
                collectObject(obj);
            }
        }
    }
    
    void collectObject(GCObject* obj) {
        cout << "回收对象: " << obj << endl;
        
        // 减少引用对象的计数
        vector<GCObject*> references = obj->getReferences();
        for (auto& ref : references) {
            removeReference(ref);
        }
        
        // 从引用计数表中移除
        referenceCounts.erase(obj);
        
        // 从对象列表中移除
        allObjects.erase(
            remove(allObjects.begin(), allObjects.end(), obj),
            allObjects.end()
        );
        
        // 删除对象
        delete obj;
        
        cout << "对象回收完成" << endl;
    }
    
    void collect() {
        cout << "执行引用计数垃圾回收" << endl;
        cout << "===========================================" << endl;
        
        // 查找引用计数为0的对象
        vector<GCObject*> toDelete;
        
        for (auto& pair : referenceCounts) {
            if (pair.second <= 0) {
                toDelete.push_back(pair.first);
            }
        }
        
        // 回收对象
        for (auto& obj : toDelete) {
            collectObject(obj);
        }
        
        cout << "引用计数垃圾回收完成" << endl;
        cout << endl;
    }
    
    void printGCStats() {
        cout << "垃圾回收统计信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "总对象数: " << allObjects.size() << endl;
        cout << "引用计数对象数: " << referenceCounts.size() << endl;
        
        int totalReferences = 0;
        for (auto& pair : referenceCounts) {
            totalReferences += pair.second;
        }
        cout << "总引用数: " << totalReferences << endl;
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理垃圾回收器" << endl;
        
        // 清理所有对象
        for (auto& obj : allObjects) {
            delete obj;
        }
        allObjects.clear();
        referenceCounts.clear();
        
        isRunning = false;
        
        cout << "垃圾回收器清理完成" << endl;
    }
    
    ~ReferenceCountingGC() {
        cleanup();
    }
};
```

## 🔧 分代垃圾回收器

### 分代垃圾回收器实现

```c
class GenerationalGC {
private:
    vector<GCObject*> youngGeneration;
    vector<GCObject*> oldGeneration;
    vector<GCObject*> rootObjects;
    int youngGenerationSize;
    int oldGenerationSize;
    bool isRunning;
    
public:
    GenerationalGC() : youngGenerationSize(0), oldGenerationSize(0), isRunning(false) {}
    
    void initialize() {
        cout << "初始化分代垃圾回收器" << endl;
        isRunning = true;
    }
    
    void registerObject(GCObject* obj) {
        cout << "注册对象: " << obj << endl;
        youngGeneration.push_back(obj);
        youngGenerationSize++;
    }
    
    void addRootObject(GCObject* obj) {
        rootObjects.push_back(obj);
        cout << "添加根对象: " << obj << endl;
    }
    
    void promoteObject(GCObject* obj) {
        cout << "提升对象到老年代: " << obj << endl;
        
        // 从年轻代移除
        youngGeneration.erase(
            remove(youngGeneration.begin(), youngGeneration.end(), obj),
            youngGeneration.end()
        );
        youngGenerationSize--;
        
        // 添加到老年代
        oldGeneration.push_back(obj);
        oldGenerationSize++;
    }
    
    void collectYoungGeneration() {
        cout << "回收年轻代" << endl;
        cout << "-------------------------------------------" << endl;
        
        // 标记年轻代对象
        markYoungGeneration();
        
        // 清除年轻代对象
        sweepYoungGeneration();
        
        cout << "年轻代回收完成" << endl;
    }
    
    void markYoungGeneration() {
        cout << "标记年轻代对象" << endl;
        
        // 从根对象开始标记
        for (auto& root : rootObjects) {
            markObject(root, true);
        }
    }
    
    void markObject(GCObject* obj, bool isYoung) {
        if (obj && !obj->isMarked()) {
            obj->setMarked(true);
            cout << "标记对象: " << obj << endl;
            
            // 标记引用的对象
            vector<GCObject*> references = obj->getReferences();
            for (auto& ref : references) {
                if (isYoung && find(youngGeneration.begin(), youngGeneration.end(), ref) != youngGeneration.end()) {
                    markObject(ref, true);
                } else if (!isYoung && find(oldGeneration.begin(), oldGeneration.end(), ref) != oldGeneration.end()) {
                    markObject(ref, false);
                }
            }
        }
    }
    
    void sweepYoungGeneration() {
        cout << "清除年轻代对象" << endl;
        
        vector<GCObject*> toDelete;
        vector<GCObject*> toPromote;
        
        for (auto& obj : youngGeneration) {
            if (!obj->isMarked()) {
                toDelete.push_back(obj);
            } else {
                // 检查是否需要提升到老年代
                if (obj->getAge() > 3) {
                    toPromote.push_back(obj);
                }
                obj->setMarked(false);
            }
        }
        
        // 删除未标记的对象
        for (auto& obj : toDelete) {
            cout << "删除年轻代对象: " << obj << endl;
            delete obj;
        }
        
        // 提升对象到老年代
        for (auto& obj : toPromote) {
            promoteObject(obj);
        }
        
        // 从列表中移除已删除的对象
        youngGeneration.erase(
            remove_if(youngGeneration.begin(), youngGeneration.end(),
                [&toDelete](GCObject* obj) {
                    return find(toDelete.begin(), toDelete.end(), obj) != toDelete.end();
                }),
            youngGeneration.end()
        );
    }
    
    void collectOldGeneration() {
        cout << "回收老年代" << endl;
        cout << "-------------------------------------------" << endl;
        
        // 标记老年代对象
        markOldGeneration();
        
        // 清除老年代对象
        sweepOldGeneration();
        
        cout << "老年代回收完成" << endl;
    }
    
    void markOldGeneration() {
        cout << "标记老年代对象" << endl;
        
        // 从根对象开始标记
        for (auto& root : rootObjects) {
            markObject(root, false);
        }
    }
    
    void sweepOldGeneration() {
        cout << "清除老年代对象" << endl;
        
        vector<GCObject*> toDelete;
        
        for (auto& obj : oldGeneration) {
            if (!obj->isMarked()) {
                toDelete.push_back(obj);
            } else {
                obj->setMarked(false);
            }
        }
        
        // 删除未标记的对象
        for (auto& obj : toDelete) {
            cout << "删除老年代对象: " << obj << endl;
            delete obj;
        }
        
        // 从列表中移除已删除的对象
        oldGeneration.erase(
            remove_if(oldGeneration.begin(), oldGeneration.end(),
                [&toDelete](GCObject* obj) {
                    return find(toDelete.begin(), toDelete.end(), obj) != toDelete.end();
                }),
            oldGeneration.end()
        );
    }
    
    void collect() {
        cout << "执行分代垃圾回收" << endl;
        cout << "===========================================" << endl;
        
        // 回收年轻代
        collectYoungGeneration();
        
        // 如果老年代使用率过高，回收老年代
        if (oldGenerationSize > 1000) {
            collectOldGeneration();
        }
        
        cout << "分代垃圾回收完成" << endl;
        cout << endl;
    }
    
    void printGCStats() {
        cout << "分代垃圾回收统计信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "年轻代对象数: " << youngGeneration.size() << endl;
        cout << "老年代对象数: " << oldGeneration.size() << endl;
        cout << "根对象数: " << rootObjects.size() << endl;
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理垃圾回收器" << endl;
        
        // 清理所有对象
        for (auto& obj : youngGeneration) {
            delete obj;
        }
        youngGeneration.clear();
        
        for (auto& obj : oldGeneration) {
            delete obj;
        }
        oldGeneration.clear();
        
        rootObjects.clear();
        
        isRunning = false;
        
        cout << "垃圾回收器清理完成" << endl;
    }
    
    ~GenerationalGC() {
        cleanup();
    }
};
```

## 🔧 增量垃圾回收器

### 增量垃圾回收器实现
```c
class IncrementalGC {
private:
    vector<GCObject*> rootObjects;
    vector<GCObject*> allObjects;
    vector<GCObject*> workList;
    int workLimit;
    bool isRunning;
    
public:
    IncrementalGC() : workLimit(100), isRunning(false) {}
    
    void initialize() {
        cout << "初始化增量垃圾回收器" << endl;
        isRunning = true;
    }
    
    void registerObject(GCObject* obj) {
        cout << "注册对象: " << obj << endl;
        allObjects.push_back(obj);
    }
    
    void addRootObject(GCObject* obj) {
        rootObjects.push_back(obj);
        cout << "添加根对象: " << obj << endl;
    }
    
    void collectIncremental() {
        cout << "执行增量垃圾回收" << endl;
        cout << "-------------------------------------------" << endl;
        
        // 如果工作列表为空，重新初始化
        if (workList.empty()) {
            initializeWorkList();
        }
        
        // 处理工作列表中的对象
        processWorkList();
        
        cout << "增量垃圾回收完成" << endl;
    }
    
    void initializeWorkList() {
        cout << "初始化工作列表" << endl;
        
        // 将所有对象添加到工作列表
        workList = allObjects;
        
        // 清除所有标记
        for (auto& obj : allObjects) {
            obj->setMarked(false);
        }
    }
    
    void processWorkList() {
        cout << "处理工作列表" << endl;
        
        int processed = 0;
        
        while (!workList.empty() && processed < workLimit) {
            GCObject* obj = workList.back();
            workList.pop_back();
            
            if (!obj->isMarked()) {
                // 标记对象
                obj->setMarked(true);
                cout << "标记对象: " << obj << endl;
                
                // 将引用的对象添加到工作列表
                vector<GCObject*> references = obj->getReferences();
                for (auto& ref : references) {
                    if (!ref->isMarked()) {
                        workList.push_back(ref);
                    }
                }
            }
            
            processed++;
        }
        
        cout << "处理了 " << processed << " 个对象" << endl;
    }
    
    void sweepIncremental() {
        cout << "执行增量清除" << endl;
        cout << "-------------------------------------------" << endl;
        
        vector<GCObject*> toDelete;
        
        for (auto& obj : allObjects) {
            if (!obj->isMarked()) {
                toDelete.push_back(obj);
            } else {
                obj->setMarked(false);
            }
        }
        
        // 删除未标记的对象
        for (auto& obj : toDelete) {
            cout << "删除对象: " << obj << endl;
            delete obj;
        }
        
        // 从列表中移除已删除的对象
        allObjects.erase(
            remove_if(allObjects.begin(), allObjects.end(),
                [&toDelete](GCObject* obj) {
                    return find(toDelete.begin(), toDelete.end(), obj) != toDelete.end();
                }),
            allObjects.end()
        );
        
        cout << "增量清除完成" << endl;
    }
    
    void collect() {
        cout << "执行增量垃圾回收" << endl;
        cout << "===========================================" << endl;
        
        // 增量标记
        collectIncremental();
        
        // 如果工作列表为空，执行清除
        if (workList.empty()) {
            sweepIncremental();
        }
        
        cout << "增量垃圾回收完成" << endl;
        cout << endl;
    }
    
    void printGCStats() {
        cout << "增量垃圾回收统计信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "总对象数: " << allObjects.size() << endl;
        cout << "根对象数: " << rootObjects.size() << endl;
        cout << "工作列表大小: " << workList.size() << endl;
        cout << "工作限制: " << workLimit << endl;
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理垃圾回收器" << endl;
        
        // 清理所有对象
        for (auto& obj : allObjects) {
            delete obj;
        }
        allObjects.clear();
        rootObjects.clear();
        workList.clear();
        
        isRunning = false;
        
        cout << "垃圾回收器清理完成" << endl;
    }
    
    ~IncrementalGC() {
        cleanup();
    }
};
```

```c
            case GENERATIONAL:
                generationalGC->registerObject(obj);
                break;
            case MARK_COMPACT:
            case COPYING:
                incrementalGC->registerObject(obj);
                break;
        }
    }
    
    void addRootObject(GCObject* obj) {
        switch (currentStrategy) {
            case MARK_SWEEP:
                markSweepGC->addRootObject(obj);
                break;
            case REFERENCE_COUNTING:
                // 引用计数不需要根对象
                break;
            case GENERATIONAL:
                generationalGC->addRootObject(obj);
                break;
            case MARK_COMPACT:
            case COPYING:
                incrementalGC->addRootObject(obj);
                break;
        }
    }
    
    void printGCStats() {
        cout << "垃圾回收管理器统计信息" << endl;
        cout << "===========================================" << endl;
        
        switch (currentStrategy) {
            case MARK_SWEEP:
                markSweepGC->printGCStats();
                break;
            case REFERENCE_COUNTING:
                referenceCountingGC->printGCStats();
                break;
            case GENERATIONAL:
                generationalGC->printGCStats();
                break;
            case MARK_COMPACT:
            case COPYING:
                incrementalGC->printGCStats();
                break;
        }
    }
    
    void cleanup() {
        cout << "清理垃圾回收管理器" << endl;
        
        if (markSweepGC) {
            delete markSweepGC;
            markSweepGC = nullptr;
        }
        
        if (referenceCountingGC) {
            delete referenceCountingGC;
            referenceCountingGC = nullptr;
        }
        
        if (generationalGC) {
            delete generationalGC;
            generationalGC = nullptr;
        }
        
        if (incrementalGC) {
            delete incrementalGC;
            incrementalGC = nullptr;
        }
        
        cout << "垃圾回收管理器清理完成" << endl;
    }
    
    ~GarbageCollectionManager() {
        cleanup();
    }
};
```
