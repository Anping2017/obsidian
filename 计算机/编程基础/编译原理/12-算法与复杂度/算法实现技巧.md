# 算法实现技巧

## 🎯 算法实现技巧概述

**算法实现技巧**是编译器原理学习的重要组成部分，涵盖了编译算法实现过程中的各种技巧和优化方法。掌握这些技巧有助于提高算法效率、降低复杂度、减少内存使用，并提高代码的可读性和可维护性。

## 🔍 实现技巧分类

### 1. 数据结构选择
- **数组**：固定大小、连续内存、快速访问
- **链表**：动态大小、灵活插入删除、内存分散
- **哈希表**：快速查找、空间换时间、冲突处理
- **树结构**：层次关系、有序访问、平衡维护

### 2. 算法优化
- **时间复杂度优化**：减少循环嵌套、使用更高效算法
- **空间复杂度优化**：原地操作、复用空间、压缩存储
- **常数因子优化**：减少函数调用、内联优化、位运算

### 3. 内存管理
- **内存池**：预分配内存、减少分配开销
- **对象池**：复用对象、减少构造析构
- **缓存友好**：局部性原理、预取优化

**技巧**：使用压缩状态转移表减少内存使用

```c
// 压缩状态转移表实现
class CompactDFA {
private:
    // 使用稀疏矩阵存储状态转移
    map<pair<int, int>, int> transitions;
    vector<int> acceptStates;
    int startState;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(实际转移数)
    int getTransition(int state, int symbol) {
        auto key = make_pair(state, symbol);
        auto it = transitions.find(key);
        return (it != transitions.end()) ? it->second : -1;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void addTransition(int from, int symbol, int to) {
        transitions[make_pair(from, symbol)] = to;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool isAcceptState(int state) {
        return find(acceptStates.begin(), acceptStates.end(), state) 
               != acceptStates.end();
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    bool simulate(const string& input) {
        int currentState = startState;
        
        for (char c : input) {
            int symbol = getSymbolIndex(c);
            if (symbol == -1) return false;
            
            currentState = getTransition(currentState, symbol);
            if (currentState == -1) return false;
        }
        
        return isAcceptState(currentState);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    int getSymbolIndex(char c) {
        if (c >= '0' && c <= '9') return c - '0';
        if (c >= 'a' && c <= 'z') return c - 'a' + 10;
        if (c >= 'A' && c <= 'Z') return c - 'A' + 36;
        return -1;
    }
};
```

#### 位向量优化
**技巧**：使用位向量表示状态集合，提高集合操作效率

```c
// 位向量状态集合实现
class BitVectorStateSet {
private:
    vector<uint64_t> bits;
    int numStates;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(n/64)
    BitVectorStateSet(int states) : numStates(states) {
        int numWords = (states + 63) / 64;
        bits.resize(numWords, 0);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void add(int state) {
        int wordIndex = state / 64;
        int bitIndex = state % 64;
        bits[wordIndex] |= (1ULL << bitIndex);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool contains(int state) {
        int wordIndex = state / 64;
        int bitIndex = state % 64;
        return (bits[wordIndex] & (1ULL << bitIndex)) != 0;
    }
    
    // 时间复杂度：O(n/64)
    // 空间复杂度：O(1)
    void unionWith(const BitVectorStateSet& other) {
        for (int i = 0; i < bits.size(); i++) {
            bits[i] |= other.bits[i];
        }
    }
    
    // 时间复杂度：O(n/64)
    // 空间复杂度：O(1)
    void intersectWith(const BitVectorStateSet& other) {
        for (int i = 0; i < bits.size(); i++) {
            bits[i] &= other.bits[i];
        }
    }
    
    // 时间复杂度：O(n/64)
    // 空间复杂度：O(1)
    bool isEmpty() {
        for (uint64_t word : bits) {
            if (word != 0) return false;
        }
        return true;
    }
    
    // 时间复杂度：O(n/64)
    // 空间复杂度：O(1)
    int count() {
        int count = 0;
        for (uint64_t word : bits) {
            count += __builtin_popcountll(word);
        }
        return count;
    }
};
```

#### 缓存优化
**技巧**：使用缓存避免重复计算

```c
// 缓存优化的NFA到DFA转换
class CachedNFAToDFAConverter {
private:
    map<set<int>, int> stateMap;
    map<set<int>, set<int>> epsilonClosureCache;
    map<pair<set<int>, int>, set<int>> moveCache;
    
public:
    // 时间复杂度：O(2^|Q| × |Σ|)
    // 空间复杂度：O(2^|Q| × |Σ|)
    DFA convert(NFA nfa) {
        DFA dfa;
        
        // 初始状态：ε-闭包
        set<int> initialState = getEpsilonClosure(nfa.startState);
        dfaStates.push_back(initialState);
        stateMap[initialState] = 0;
        
        queue<int> stateQueue;
        stateQueue.push(0);
        
        while (!stateQueue.empty()) {
            int currentStateIndex = stateQueue.front();
            stateQueue.pop();
            
            set<int> currentState = dfaStates[currentStateIndex];
            
            // 对每个输入符号
            for (char symbol : alphabet) {
                set<int> nextState = getMove(currentState, symbol);
                
                if (!nextState.empty()) {
                    int nextStateIndex = getOrCreateState(nextState);
                    dfa.addTransition(currentStateIndex, symbol, nextStateIndex);
                    
                    if (nextStateIndex == dfaStates.size() - 1) {
                        stateQueue.push(nextStateIndex);
                    }
                }
            }
        }
        
        return dfa;
    }
    
    // 时间复杂度：O(|Q|)
    // 空间复杂度：O(|Q|)
    set<int> getEpsilonClosure(int state) {
        if (epsilonClosureCache.find({state}) != epsilonClosureCache.end()) {
            return epsilonClosureCache[{state}];
        }
        
        set<int> closure;
        stack<int> stateStack;
        stateStack.push(state);
        
        while (!stateStack.empty()) {
            int currentState = stateStack.top();
            stateStack.pop();
            
            if (closure.find(currentState) == closure.end()) {
                closure.insert(currentState);
                
                for (int nextState : nfa.getEpsilonTransitions(currentState)) {
                    stateStack.push(nextState);
                }
            }
        }
        
        epsilonClosureCache[{state}] = closure;
        return closure;
    }
    
    // 时间复杂度：O(|Q|)
    // 空间复杂度：O(|Q|)
    set<int> getMove(const set<int>& states, int symbol) {
        auto key = make_pair(states, symbol);
        if (moveCache.find(key) != moveCache.end()) {
            return moveCache[key];
        }
        
        set<int> result;
        for (int state : states) {
            for (int nextState : nfa.getTransitions(state, symbol)) {
                result.insert(nextState);
            }
        }
        
        moveCache[key] = result;
        return result;
    }
};
```

**技巧**：预编译正则表达式，避免重复解析

```c
// 预编译正则表达式实现
class PrecompiledRegex {
private:
    map<string, NFA> compiledRegexes;
    map<string, DFA> compiledDFAs;
    
public:
    // 时间复杂度：O(|r|)
    // 空间复杂度：O(|r|)
    NFA compileRegex(const string& regex) {
        if (compiledRegexes.find(regex) != compiledRegexes.end()) {
            return compiledRegexes[regex];
        }
        
        NFA nfa = thompsonConstruct(regex);
        compiledRegexes[regex] = nfa;
        return nfa;
    }
    
    // 时间复杂度：O(2^|Q| × |Σ|)
    // 空间复杂度：O(2^|Q| × |Σ|)
    DFA compileDFA(const string& regex) {
        if (compiledDFAs.find(regex) != compiledDFAs.end()) {
            return compiledDFAs[regex];
        }
        
        NFA nfa = compileRegex(regex);
        DFA dfa = nfaToDFA(nfa);
        compiledDFAs[regex] = dfa;
        return dfa;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    bool match(const string& regex, const string& input) {
        DFA dfa = compileDFA(regex);
        return dfa.simulate(input);
    }
    
    // 时间复杂度：O(|r|)
    // 空间复杂度：O(|r|)
    NFA thompsonConstruct(const string& regex) {
        stack<NFA> nfaStack;
        
        for (char c : regex) {
            switch (c) {
                case '|':
                    if (nfaStack.size() >= 2) {
                        NFA nfa2 = nfaStack.top(); nfaStack.pop();
                        NFA nfa1 = nfaStack.top(); nfaStack.pop();
                        nfaStack.push(unionNFA(nfa1, nfa2));
                    }
                    break;
                    
                case '*':
                    if (!nfaStack.empty()) {
                        NFA nfa = nfaStack.top(); nfaStack.pop();
                        nfaStack.push(kleeneStar(nfa));
                    }
                    break;
                    
                default:
                    nfaStack.push(singleSymbolNFA(c));
                    break;
            }
        }
        
        return nfaStack.empty() ? NFA() : nfaStack.top();
    }
};
```

```c
// 惰性求值正则表达式实现
class LazyRegex {
private:
    string pattern;
    mutable NFA* nfa;
    mutable DFA* dfa;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    LazyRegex(const string& p) : pattern(p), nfa(nullptr), dfa(nullptr) {}
    
    // 时间复杂度：O(|r|)
    // 空间复杂度：O(|r|)
    NFA getNFA() const {
        if (nfa == nullptr) {
            nfa = new NFA(thompsonConstruct(pattern));
        }
        return *nfa;
    }
    
    // 时间复杂度：O(2^|Q| × |Σ|)
    // 空间复杂度：O(2^|Q| × |Σ|)
    DFA getDFA() const {
        if (dfa == nullptr) {
            dfa = new DFA(nfaToDFA(getNFA()));
        }
        return *dfa;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    bool match(const string& input) const {
        return getDFA().simulate(input);
    }
    
    // 析构函数
    ~LazyRegex() {
        delete nfa;
        delete dfa;
    }
};
```

```c
// 稀疏预测分析表实现
class SparseParseTable {
private:
    map<pair<string, string>, Production> table;
    set<string> terminals;
    set<string> nonterminals;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void addEntry(const string& nonterminal, const string& terminal, 
                  const Production& production) {
        table[make_pair(nonterminal, terminal)] = production;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    Production* getEntry(const string& nonterminal, const string& terminal) {
        auto key = make_pair(nonterminal, terminal);
        auto it = table.find(key);
        return (it != table.end()) ? &it->second : nullptr;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool hasEntry(const string& nonterminal, const string& terminal) {
        return table.find(make_pair(nonterminal, terminal)) != table.end();
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    bool parse(const vector<Token>& tokens) {
        stack<string> parseStack;
        parseStack.push("$");
        parseStack.push(startSymbol);
        
        int tokenIndex = 0;
        
        while (!parseStack.empty()) {
            string top = parseStack.top();
            parseStack.pop();
            
            if (top == "$") {
                return tokenIndex == tokens.size();
            }
            
            if (terminals.find(top) != terminals.end()) {
                if (tokenIndex < tokens.size() && 
                    top == tokens[tokenIndex].type) {
                    tokenIndex++;
                } else {
                    return false;
                }
            } else {
                string lookahead = tokenIndex < tokens.size() ? 
                    tokens[tokenIndex].type : "$";
                
                Production* prod = getEntry(top, lookahead);
                if (prod != nullptr) {
                    for (int i = prod->rhs.size() - 1; i >= 0; i--) {
                        parseStack.push(prod->rhs[i]);
                    }
                } else {
                    return false;
                }
            }
        }
        
        return false;
    }
};
```

#### 递归下降优化
**技巧**：使用尾递归优化和记忆化

```c
// 优化的递归下降分析器
class OptimizedRecursiveDescentParser {
private:
    vector<Token> tokens;
    int currentToken;
    map<pair<int, string>, bool> memoization;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    bool parse(const vector<Token>& inputTokens) {
        tokens = inputTokens;
        currentToken = 0;
        memoization.clear();
        
        try {
            return parseProgram();
        } catch (const ParseError& e) {
            return false;
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    bool parseProgram() {
        while (currentToken < tokens.size()) {
            if (isTypeSpecifier()) {
                if (!parseDeclaration()) return false;
            } else if (isFunctionDefinition()) {
                if (!parseFunctionDefinition()) return false;
            } else {
                return false;
            }
        }
        return true;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    bool parseDeclaration() {
        if (!parseTypeSpecifier()) return false;
        if (!parseDeclarator()) return false;
        
        if (match("=")) {
            if (!parseInitializer()) return false;
        }
        
        return expect(";");
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    bool parseFunctionDefinition() {
        if (!parseTypeSpecifier()) return false;
        if (!parseDeclarator()) return false;
        return parseCompoundStatement();
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    bool parseCompoundStatement() {
        if (!expect("{")) return false;
        
        while (currentToken < tokens.size() && 
               tokens[currentToken].type != "}") {
            if (isTypeSpecifier()) {
                if (!parseDeclaration()) return false;
            } else {
                if (!parseStatement()) return false;
            }
        }
        
        return expect("}");
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool match(const string& expected) {
        if (currentToken < tokens.size() && 
            tokens[currentToken].type == expected) {
            currentToken++;
            return true;
        }
        return false;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool expect(const string& expected) {
        if (match(expected)) {
            return true;
        } else {
            throw ParseError("Expected " + expected);
        }
    }
};
```

```c
class BitwiseActionTable {
private:

    vector<uint32_t> gotoTable;

    int numSymbols;
    
public:
    // 时间复杂度：O(1)

    BitwiseActionTable(int states, int symbols)

        int index = state * numSymbols + symbol;
        actionTable[index] = (static_cast<uint32_t>(type) << 24) | value;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void setGoto(int state, int symbol, int value) {
        int index = state * numSymbols + symbol;
        gotoTable[index] = value;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    ActionType getActionType(int state, int symbol) {
        int index = state * numSymbols + symbol;
        return static_cast<ActionType>(actionTable[index] >> 24);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)

    }

    // 空间复杂度：O(1)
```

#### 哈希表优化
**技巧**：使用开放地址法处理哈希冲突

```c
// 开放地址法哈希表实现
class OpenAddressingHashTable {
private:
    struct HashEntry {
        string key;
        Symbol value;
        bool occupied;
        
        HashEntry() : occupied(false) {}
    };
    
    vector<HashEntry> table;
    int size;
    int capacity;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(n)
    OpenAddressingHashTable(int cap) : capacity(cap), size(0) {
        table.resize(capacity);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    int hash(const string& key) {
        int hash = 0;
        for (char c : key) {
            hash = (hash * 31 + c) % capacity;
        }
        return hash;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void insert(const string& key, const Symbol& value) {
        if (size >= capacity * 0.75) {
            resize();
        }
        
        int index = hash(key);
        while (table[index].occupied && table[index].key != key) {
            index = (index + 1) % capacity;
        }
        
        if (!table[index].occupied) {
            size++;
        }
        
        table[index].key = key;
        table[index].value = value;
        table[index].occupied = true;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    Symbol* lookup(const string& key) {
        int index = hash(key);
        while (table[index].occupied) {
            if (table[index].key == key) {
                return &table[index].value;
            }
            index = (index + 1) % capacity;
        }
        return nullptr;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void resize() {
        vector<HashEntry> oldTable = table;
        capacity *= 2;
        table.resize(capacity);
        size = 0;
        
        for (const HashEntry& entry : oldTable) {
            if (entry.occupied) {
                insert(entry.key, entry.value);
            }
        }
    }
};
```

```c
// 优化的类型推断实现
class OptimizedTypeInference {
private:
    TypeCache cache;
    map<ASTNode*, vector<Type>> constraintGraph;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    Type inferType(ASTNode* node) {
        Type cachedType = cache.getNodeType(node);
        if (cachedType != Type::UNKNOWN) {
            return cachedType;
        }
        
        Type type = inferTypeRecursive(node);
        cache.setNodeType(node, type);
        return type;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    Type inferTypeRecursive(ASTNode* node) {
        switch (node->type) {
            case "number":
                return Type::INT;
                
            case "string":
                return Type::STRING;
                
            case "identifier":
                return cache.getVariableType(node->value);
                
            case "add":
            case "sub":
            case "mul":
            case "div":
                return inferArithmeticType(node);
                
            case "function_call":
                return inferFunctionCallType(node);
                
            default:
                return Type::UNKNOWN;
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    Type inferArithmeticType(ASTNode* node) {
        Type leftType = inferType(node->children[0]);
        Type rightType = inferType(node->children[1]);
        
        if (leftType == Type::INT && rightType == Type::INT) {
            return Type::INT;
        } else if (leftType == Type::FLOAT || rightType == Type::FLOAT) {
            return Type::FLOAT;
        } else {
            return Type::UNKNOWN;
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    Type inferFunctionCallType(ASTNode* node) {
        string functionName = node->children[0]->value;
        FunctionType* funcType = cache.getFunctionType(functionName);
        
        if (funcType == nullptr) {
            return Type::UNKNOWN;
        }
        
        return funcType->returnType;
    }
};
```

```c
// 增量数据流分析实现
class IncrementalDataFlowAnalysis {
private:
    map<BasicBlock*, set<string>> liveIn;
    map<BasicBlock*, set<string>> liveOut;
    map<BasicBlock*, set<string>> def;
    map<BasicBlock*, set<string>> use;
    set<BasicBlock*> changedBlocks;
    
public:
    // 时间复杂度：O(n × m)
    // 空间复杂度：O(n × m)
    void analyze(ControlFlowGraph* cfg) {
        initialize(cfg);
        
        while (!changedBlocks.empty()) {
            BasicBlock* block = *changedBlocks.begin();
            changedBlocks.erase(changedBlocks.begin());
            
            bool changed = updateBlock(block);
            if (changed) {
                // 将后继块加入待处理集合
                for (BasicBlock* successor : block->getSuccessors()) {
                    changedBlocks.insert(successor);
                }
            }
        }
    }
    
    // 时间复杂度：O(m)
    // 空间复杂度：O(1)
    bool updateBlock(BasicBlock* block) {
        set<string> oldLiveIn = liveIn[block];
        set<string> oldLiveOut = liveOut[block];
        
        // LiveOut[B] = ∪(LiveIn[S]) for all successors S of B
        liveOut[block].clear();
        for (BasicBlock* successor : block->getSuccessors()) {
            liveOut[block].insert(liveIn[successor].begin(), 
                                liveIn[successor].end());
        }
        
        // LiveIn[B] = Use[B] ∪ (LiveOut[B] - Def[B])
        liveIn[block] = use[block];
        for (string var : liveOut[block]) {
            if (def[block].find(var) == def[block].end()) {
                liveIn[block].insert(var);
            }
        }
        
        return oldLiveIn != liveIn[block] || oldLiveOut != liveOut[block];
    }
    
    // 时间复杂度：O(n × m)
    // 空间复杂度：O(n × m)
    void initialize(ControlFlowGraph* cfg) {
        for (BasicBlock* block : cfg->getBlocks()) {
            liveIn[block] = set<string>();
            liveOut[block] = set<string>();
            def[block] = computeDef(block);
            use[block] = computeUse(block);
            changedBlocks.insert(block);
        }
    }
};
```

```c
class BitVectorDataFlowAnalysis {
private:
    map<BasicBlock*, BitVectorStateSet> liveIn;
    map<BasicBlock*, BitVectorStateSet> liveOut;
    map<BasicBlock*, BitVectorStateSet> def;
    map<BasicBlock*, BitVectorStateSet> use;
    int numVariables;
    
public:
    // 时间复杂度：O(n × m/64)
    // 空间复杂度：O(n × m/64)
    void analyze(ControlFlowGraph* cfg) {
        initialize(cfg);
        
        bool changed = true;
        while (changed) {
            changed = false;
            
            for (BasicBlock* block : cfg->getBlocks()) {
                BitVectorStateSet oldLiveIn = liveIn[block];
                BitVectorStateSet oldLiveOut = liveOut[block];
                
                // LiveOut[B] = ∪(LiveIn[S]) for all successors S of B
                liveOut[block].clear();
                for (BasicBlock* successor : block->getSuccessors()) {
                    liveOut[block].unionWith(liveIn[successor]);
                }
                
                // LiveIn[B] = Use[B] ∪ (LiveOut[B] - Def[B])
                liveIn[block] = use[block];
                BitVectorStateSet temp = liveOut[block];
                temp.intersectWith(def[block]);
                liveIn[block].unionWith(temp);
                
                if (oldLiveIn != liveIn[block] || oldLiveOut != liveOut[block]) {
                    changed = true;
                }
            }
        }
    }
    
    // 时间复杂度：O(n × m/64)
    // 空间复杂度：O(n × m/64)
    void initialize(ControlFlowGraph* cfg) {
        for (BasicBlock* block : cfg->getBlocks()) {
            liveIn[block] = BitVectorStateSet(numVariables);
            liveOut[block] = BitVectorStateSet(numVariables);
            def[block] = computeDef(block);
            use[block] = computeUse(block);
        }
    }
};
```

```c
// 活跃区间优化的寄存器分配
class LiveIntervalOptimizer {
private:
    map<string, LiveInterval> intervals;
    map<string, int> registerAssignment;
    int numRegisters;
    
public:
    // 时间复杂度：O(n log n)
    // 空间复杂度：O(n)
    void allocateRegisters(const vector<string>& variables) {
        // 计算活跃区间
        computeLiveIntervals(variables);
        
        // 按开始时间排序
        vector<string> sortedVars = variables;
        sort(sortedVars.begin(), sortedVars.end(), 
             [this](const string& a, const string& b) {
                 return intervals[a].start < intervals[b].start;
             });
        
        // 分配寄存器
        for (const string& var : sortedVars) {
            allocateRegister(var);
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void computeLiveIntervals(const vector<string>& variables) {
        for (const string& var : variables) {
            LiveInterval interval;
            interval.start = getFirstUse(var);
            interval.end = getLastUse(var);
            intervals[var] = interval;
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    void allocateRegister(const string& var) {
        LiveInterval& interval = intervals[var];
        
        // 查找可用寄存器
        for (int reg = 0; reg < numRegisters; reg++) {
            if (isRegisterAvailable(reg, interval)) {
                registerAssignment[var] = reg;
                return;
            }
        }
        
        // 溢出处理
        handleSpill(var);
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    bool isRegisterAvailable(int reg, const LiveInterval& interval) {
        for (const auto& pair : registerAssignment) {
            if (pair.second == reg) {
                const LiveInterval& otherInterval = intervals[pair.first];
                if (intervalsOverlap(interval, otherInterval)) {
                    return false;
                }
            }
        }
        return true;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool intervalsOverlap(const LiveInterval& a, const LiveInterval& b) {
        return !(a.end <= b.start || b.end <= a.start);
    }
};
```

#### 图着色优化
**技巧**：使用图着色优化寄存器分配

```c
// 图着色优化的寄存器分配
class GraphColoringOptimizer {
private:
    map<string, set<string>> interferenceGraph;
    map<string, int> registerAssignment;
    int numRegisters;
    
public:
    // 时间复杂度：O(n²)
    // 空间复杂度：O(n²)
    bool allocateRegisters(const vector<string>& variables) {
        buildInterferenceGraph(variables);
        
        // 简化阶段
        stack<string> variableStack;
        map<string, int> degree;
        
        for (const string& var : variables) {
            degree[var] = interferenceGraph[var].size();
        }
        
        while (!variables.empty()) {
            bool found = false;
            
            for (const string& var : variables) {
                if (degree[var] < numRegisters) {
                    variableStack.push(var);
                    variables.erase(find(variables.begin(), variables.end(), var));
                    
                    for (const string& neighbor : interferenceGraph[var]) {
                        degree[neighbor]--;
                    }
                    
                    found = true;
                    break;
                }
            }
            
            if (!found) {
                // 溢出处理
                string maxVar = *max_element(variables.begin(), variables.end(),
                    [&](const string& a, const string& b) {
                        return degree[a] < degree[b];
                    });
                
                variableStack.push(maxVar);
                variables.erase(find(variables.begin(), variables.end(), maxVar));
                
                for (const string& neighbor : interferenceGraph[maxVar]) {
                    degree[neighbor]--;
                }
            }
        }
        
        // 着色阶段
        while (!variableStack.empty()) {
            string var = variableStack.top();
            variableStack.pop();
            
            set<int> usedColors;
            for (const string& neighbor : interferenceGraph[var]) {
                if (registerAssignment.find(neighbor) != registerAssignment.end()) {
                    usedColors.insert(registerAssignment[neighbor]);
                }
            }
            
            for (int color = 0; color < numRegisters; color++) {
                if (usedColors.find(color) == usedColors.end()) {
                    registerAssignment[var] = color;
                    break;
                }
            }
            
            if (registerAssignment.find(var) == registerAssignment.end()) {
                return false; // 分配失败
            }
        }
        
        return true;
    }
};
```
