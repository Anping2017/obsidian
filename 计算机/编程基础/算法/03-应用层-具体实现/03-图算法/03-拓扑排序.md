# 拓扑排序

## 🎯 核心概念

**拓扑排序(Topological Sorting)**是对有向无环图(DAG)的顶点进行线性排序，使得对于图中的每一条有向边(u,v)，u在排序中都出现在v之前。拓扑排序在任务调度、依赖关系分析、编译系统等领域有重要应用。

## 🔍 算法原理

### 1. 基本思想

```python
def topological_sorting_concept():
    """拓扑排序基本思想"""
    # 1. 找到入度为0的顶点
    # 2. 将该顶点加入结果序列
    # 3. 从图中删除该顶点及其出边
    # 4. 重复上述过程直到图为空
    # 5. 如果图中还有顶点但无入度为0的顶点，则存在环
```

## 🎨 算法实现

### 1. Kahn算法
```python
def kahn_algorithm():
    """Kahn算法实现"""
    
    def kahn_basic(graph):
        """基本Kahn算法"""
        # 计算入度
        in_degree = {vertex: 0 for vertex in graph}
        for vertex in graph:
            for neighbor in graph[vertex]:
                in_degree[neighbor] += 1
        
        # 找到入度为0的顶点
        queue = [vertex for vertex in graph if in_degree[vertex] == 0]
        result = []
        
        while queue:
            current = queue.pop(0)
            result.append(current)
            
            # 更新邻居的入度
            for neighbor in graph[current]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)
        
        # 检查是否存在环
        if len(result) != len(graph):
            return None, "存在环"
        
        return result, "无环"
    
    def kahn_with_cycle_detection(graph):
        """带环检测的Kahn算法"""
        in_degree = {vertex: 0 for vertex in graph}
        for vertex in graph:
            for neighbor in graph[vertex]:
                in_degree[neighbor] += 1
        
        queue = [vertex for vertex in graph if in_degree[vertex] == 0]
        result = []
        processed_count = 0
        
        while queue:
            current = queue.pop(0)
            result.append(current)
            processed_count += 1
            
            for neighbor in graph[current]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)
        
        if processed_count != len(graph):
            return None, "存在环"
        
        return result, "无环"
    
    def kahn_with_priority_queue(graph):
        """使用优先队列的Kahn算法"""
        import heapq
        
        in_degree = {vertex: 0 for vertex in graph}
        for vertex in graph:
            for neighbor in graph[vertex]:
                in_degree[neighbor] += 1
        
        # 使用优先队列保证字典序
        pq = [vertex for vertex in graph if in_degree[vertex] == 0]
        heapq.heapify(pq)
        result = []
        
        while pq:
            current = heapq.heappop(pq)
            result.append(current)
            
            for neighbor in graph[current]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    heapq.heappush(pq, neighbor)
        
        if len(result) != len(graph):
            return None, "存在环"
        
        return result, "无环"
    
    return kahn_basic, kahn_with_cycle_detection, kahn_with_priority_queue
```

### 2. DFS算法
```python
def dfs_algorithm():
    """DFS拓扑排序算法实现"""
    
    def dfs_topological_sort(graph):
        """DFS拓扑排序"""
        visited = set()
        stack = []
        
        def dfs(vertex):
            if vertex in visited:
                return
            
            visited.add(vertex)
            
            for neighbor in graph[vertex]:
                dfs(neighbor)
            
            stack.append(vertex)
        
        for vertex in graph:
            if vertex not in visited:
                dfs(vertex)
        
        return stack[::-1]  # 反转得到拓扑排序
    
    def dfs_with_cycle_detection(graph):
        """带环检测的DFS拓扑排序"""
        WHITE, GRAY, BLACK = 0, 1, 2
        color = {vertex: WHITE for vertex in graph}
        stack = []
        
        def dfs(vertex):
            if color[vertex] == GRAY:
                return False  # 发现环
            if color[vertex] == BLACK:
                return True
            
            color[vertex] = GRAY
            
            for neighbor in graph[vertex]:
                if not dfs(neighbor):
                    return False
            
            color[vertex] = BLACK
            stack.append(vertex)
            return True
        
        for vertex in graph:
            if color[vertex] == WHITE:
                if not dfs(vertex):
                    return None, "存在环"
        
        return stack[::-1], "无环"
    
    def dfs_iterative(graph):
        """迭代DFS拓扑排序"""
        visited = set()
        stack = []
        
        for vertex in graph:
            if vertex not in visited:
                dfs_stack = [vertex]
                
                while dfs_stack:
                    current = dfs_stack.pop()
                    
                    if current in visited:
                        continue
                    
                    visited.add(current)
                    
                    # 检查是否还有未访问的邻居
                    has_unvisited_neighbors = False
                    for neighbor in graph[current]:
                        if neighbor not in visited:
                            has_unvisited_neighbors = True
                            break
                    
                    if has_unvisited_neighbors:
                        dfs_stack.append(current)
                        for neighbor in graph[current]:
                            if neighbor not in visited:
                                dfs_stack.append(neighbor)
                    else:
                        stack.append(current)
        
        return stack[::-1]
    
    return dfs_topological_sort, dfs_with_cycle_detection, dfs_iterative
```

### 3. Tarjan算法
```python
def tarjan_algorithm():
    """Tarjan算法实现"""
    
    def tarjan_scc(graph):
        """Tarjan强连通分量算法"""
        index = 0
        stack = []
        indices = {}
        lowlinks = {}
        on_stack = set()
        sccs = []
        
        def strongconnect(vertex):
            nonlocal index
            indices[vertex] = index
            lowlinks[vertex] = index
            index += 1
            stack.append(vertex)
            on_stack.add(vertex)
            
            for neighbor in graph[vertex]:
                if neighbor not in indices:
                    strongconnect(neighbor)
                    lowlinks[vertex] = min(lowlinks[vertex], lowlinks[neighbor])
                elif neighbor in on_stack:
                    lowlinks[vertex] = min(lowlinks[vertex], indices[neighbor])
            
            if lowlinks[vertex] == indices[vertex]:
                scc = []
                while True:
                    w = stack.pop()
                    on_stack.remove(w)
                    scc.append(w)
                    if w == vertex:
                        break
                sccs.append(scc)
        
        for vertex in graph:
            if vertex not in indices:
                strongconnect(vertex)
        
        return sccs
    
    def tarjan_topological_sort(graph):
        """基于Tarjan算法的拓扑排序"""
        sccs = tarjan_scc(graph)
        
        # 如果存在大小大于1的强连通分量，则存在环
        for scc in sccs:
            if len(scc) > 1:
                return None, "存在环"
        
        # 构建DAG的拓扑排序
        dag_graph = {}
        for scc in sccs:
            vertex = scc[0]
            dag_graph[vertex] = []
            for neighbor in graph[vertex]:
                # 找到邻居所在的强连通分量
                for other_scc in sccs:
                    if neighbor in other_scc and other_scc != scc:
                        dag_graph[vertex].append(other_scc[0])
                        break
        
        # 对DAG进行拓扑排序
        result = dfs_topological_sort(dag_graph)
        return result, "无环"
    
    return tarjan_scc, tarjan_topological_sort

    return tarjan_scc_example, tarjan_cycle_example, tarjan_topological_example
```

```python
def task_scheduling_applications():

        schedule = dfs_topological_sort(graph)
```

```python
def dependency_analysis_applications():
    """依赖关系分析应用"""
    
    def package_dependency_analysis(packages, dependencies):
        """包依赖分析"""
        # 构建包依赖图
        graph = {package: [] for package in packages}
        for package, deps in dependencies.items():
            for dep in deps:
                graph[dep].append(package)
        
        # 检测循环依赖
        result, status = kahn_with_cycle_detection(graph)
        if status == "存在环":
            return None, "存在循环依赖"
        
        # 返回安装顺序
        return result, "无循环依赖"
    
    def library_dependency_analysis(libraries, dependencies):
        """库依赖分析"""
        # 构建库依赖图
        graph = {library: [] for library in libraries}
        for library, deps in dependencies.items():
            for dep in deps:
                graph[dep].append(library)
        
        # 使用Tarjan算法分析强连通分量
        sccs = tarjan_scc(graph)
        
        # 检查是否存在循环依赖
        cyclic_deps = [scc for scc in sccs if len(scc) > 1]
        if cyclic_deps:
            return None, f"存在循环依赖: {cyclic_deps}"
        
        # 返回加载顺序
        load_order = dfs_topological_sort(graph)
        return load_order, "无循环依赖"
    
    def service_dependency_analysis(services, dependencies):
        """服务依赖分析"""
        # 构建服务依赖图
        graph = {service: [] for service in services}
        for service, deps in dependencies.items():
            for dep in deps:
                graph[dep].append(service)
        
        # 拓扑排序确定启动顺序
        startup_order = kahn_basic(graph)
        return startup_order
    
    return package_dependency_analysis, library_dependency_analysis, service_dependency_analysis
```

```python
def compilation_system_applications():
    """编译系统应用"""
    
    def source_code_compilation_order(source_files, includes):
        """源代码编译顺序"""
        # 构建包含依赖图
        graph = {file: [] for file in source_files}
        for file, included_files in includes.items():
            for included_file in included_files:
                if included_file in graph:
                    graph[included_file].append(file)
        
        # 拓扑排序确定编译顺序
        compilation_order = dfs_topological_sort(graph)
        return compilation_order
    
    def makefile_dependency_analysis(targets, dependencies):
        """Makefile依赖分析"""
        # 构建目标依赖图
        graph = {target: [] for target in targets}
        for target, deps in dependencies.items():
            for dep in deps:
                if dep in graph:
                    graph[dep].append(target)
        
        # 检测循环依赖
        result, status = kahn_with_cycle_detection(graph)
        if status == "存在环":
            return None, "Makefile存在循环依赖"
        
        # 返回构建顺序
        return result, "无循环依赖"
    
    def linker_dependency_analysis(object_files, symbols):
        """链接器依赖分析"""
        # 构建符号依赖图
        graph = {obj_file: [] for obj_file in object_files}
        for obj_file, required_symbols in symbols.items():
            for symbol in required_symbols:
                # 找到定义该符号的目标文件
                for other_obj_file in object_files:
                    if other_obj_file != obj_file and symbol in symbols.get(other_obj_file, []):
                        graph[other_obj_file].append(obj_file)
                        break
        
        # 拓扑排序确定链接顺序
        link_order = kahn_basic(graph)
        return link_order
    
    return source_code_compilation_order, makefile_dependency_analysis, linker_dependency_analysis
```
