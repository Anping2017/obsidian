# 最小生成树算法

## 🎯 核心概念

**最小生成树(Minimum Spanning Tree, MST)**是连接图中所有顶点的边权重之和最小的树。最小生成树算法用于解决网络设计、电路布线、聚类分析等问题。

## 🎨 算法实现

### 1. Kruskal算法
```python
def kruskal_algorithm():
    """Kruskal算法实现"""
    
    class UnionFind:
        """并查集数据结构"""
        def __init__(self, n):
            self.parent = list(range(n))
            self.rank = [0] * n
        
        def find(self, x):
            if self.parent[x] != x:
                self.parent[x] = self.find(self.parent[x])
            return self.parent[x]
        
        def union(self, x, y):
            px, py = self.find(x), self.find(y)
            if px == py:
                return False
            
            if self.rank[px] < self.rank[py]:
                self.parent[px] = py
            elif self.rank[px] > self.rank[py]:
                self.parent[py] = px
            else:
                self.parent[py] = px
                self.rank[px] += 1
            return True
    
    def kruskal_basic(edges, num_vertices):
        """基本Kruskal算法"""
        # 按权重排序边
        edges.sort(key=lambda x: x[2])
        
        uf = UnionFind(num_vertices)
        mst = []
        total_weight = 0
        
        for u, v, weight in edges:
            if uf.union(u, v):
                mst.append((u, v, weight))
                total_weight += weight
                if len(mst) == num_vertices - 1:
                    break
        
        return mst, total_weight
    
    def kruskal_with_vertex_names(edges, vertices):
        """带顶点名称的Kruskal算法"""
        # 创建顶点到索引的映射
        vertex_to_index = {v: i for i, v in enumerate(vertices)}
        
        # 转换边格式
        indexed_edges = []
        for u, v, weight in edges:
            indexed_edges.append((vertex_to_index[u], vertex_to_index[v], weight))
        
        # 运行Kruskal算法
        mst_indices, total_weight = kruskal_basic(indexed_edges, len(vertices))
        
        # 转换回顶点名称
        mst = []
        for u_idx, v_idx, weight in mst_indices:
            mst.append((vertices[u_idx], vertices[v_idx], weight))
        
        return mst, total_weight
    
    def kruskal_with_cycle_detection(edges, num_vertices):
        """带环检测的Kruskal算法"""
        edges.sort(key=lambda x: x[2])
        
        uf = UnionFind(num_vertices)
        mst = []
        total_weight = 0
        edges_added = 0
        
        for u, v, weight in edges:
            if uf.find(u) != uf.find(v):  # 不形成环
                uf.union(u, v)
                mst.append((u, v, weight))
                total_weight += weight
                edges_added += 1
                
                if edges_added == num_vertices - 1:
                    break
        
        return mst, total_weight
    
    return kruskal_basic, kruskal_with_vertex_names, kruskal_with_cycle_detection
```

### 2. Prim算法
```python
def prim_algorithm():
    """Prim算法实现"""
    
    def prim_basic(graph, start):
        """基本Prim算法"""
        import heapq
        
        # 优先队列：(权重, 顶点, 父顶点)
        pq = [(0, start, None)]
        visited = set()
        mst = []
        total_weight = 0
        
        while pq and len(visited) < len(graph):
            weight, current, parent = heapq.heappop(pq)
            
            if current in visited:
                continue
            
            visited.add(current)
            
            if parent is not None:
                mst.append((parent, current, weight))
                total_weight += weight
            
            # 添加邻居到优先队列
            for neighbor, edge_weight in graph[current]:
                if neighbor not in visited:
                    heapq.heappush(pq, (edge_weight, neighbor, current))
        
        return mst, total_weight
    
    def prim_with_adjacency_matrix(adj_matrix, start):
        """使用邻接矩阵的Prim算法"""
        n = len(adj_matrix)
        visited = [False] * n
        key = [float('inf')] * n
        parent = [-1] * n
        
        key[start] = 0
        mst = []
        total_weight = 0
        
        for _ in range(n):
            # 找到最小key值的未访问顶点
            u = min((i for i in range(n) if not visited[i]), key=lambda x: key[x])
            visited[u] = True
            
            if parent[u] != -1:
                mst.append((parent[u], u, key[u]))
                total_weight += key[u]
            
            # 更新邻居的key值
            for v in range(n):
                if (not visited[v] and adj_matrix[u][v] > 0 and 
                    adj_matrix[u][v] < key[v]):
                    key[v] = adj_matrix[u][v]
                    parent[v] = u
        
        return mst, total_weight
    
    def prim_with_priority_queue(graph, start):
        """使用优先队列的Prim算法"""
        import heapq
        
        pq = [(0, start, None)]
        visited = set()
        mst = []
        total_weight = 0
        
        while pq:
            weight, current, parent = heapq.heappop(pq)
            
            if current in visited:
                continue
            
            visited.add(current)
            
            if parent is not None:
                mst.append((parent, current, weight))
                total_weight += weight
            
            for neighbor, edge_weight in graph[current]:
                if neighbor not in visited:
                    heapq.heappush(pq, (edge_weight, neighbor, current))
        
        return mst, total_weight
    
    return prim_basic, prim_with_adjacency_matrix, prim_with_priority_queue
```

### 3. Boruvka算法
```python
def boruvka_algorithm():
    """Boruvka算法实现"""
    
    def boruvka_basic(edges, num_vertices):
        """基本Boruvka算法"""
        # 初始化并查集
        parent = list(range(num_vertices))
        rank = [0] * num_vertices
        
        def find(x):
            if parent[x] != x:
                parent[x] = find(parent[x])
            return parent[x]
        
        def union(x, y):
            px, py = find(x), find(y)
            if px == py:
                return False
            if rank[px] < rank[py]:
                parent[px] = py
            elif rank[px] > rank[py]:
                parent[py] = px
            else:
                parent[py] = px
                rank[px] += 1
            return True
        
        mst = []
        total_weight = 0
        components = num_vertices
        
        while components > 1:
            # 为每个连通分量找到最小权重边
            min_edges = {}
            
            for u, v, weight in edges:
                root_u, root_v = find(u), find(v)
                if root_u != root_v:
                    if root_u not in min_edges or weight < min_edges[root_u][2]:
                        min_edges[root_u] = (u, v, weight)
                    if root_v not in min_edges or weight < min_edges[root_v][2]:
                        min_edges[root_v] = (u, v, weight)
            
            # 添加最小权重边
            for u, v, weight in min_edges.values():
                if union(u, v):
                    mst.append((u, v, weight))
                    total_weight += weight
                    components -= 1
        
        return mst, total_weight
    
    def boruvka_with_vertex_names(edges, vertices):
        """带顶点名称的Boruvka算法"""
        vertex_to_index = {v: i for i, v in enumerate(vertices)}
        
        indexed_edges = []
        for u, v, weight in edges:
            indexed_edges.append((vertex_to_index[u], vertex_to_index[v], weight))
        
        mst_indices, total_weight = boruvka_basic(indexed_edges, len(vertices))
        
        mst = []
        for u_idx, v_idx, weight in mst_indices:
            mst.append((vertices[u_idx], vertices[v_idx], weight))
        
        return mst, total_weight
    
    return boruvka_basic, boruvka_with_vertex_names
```
