# 网络流算法

## 概述

网络流算法（Network Flow Algorithms）是图论中的重要算法，用于解决网络中的最大流、最小割等问题。网络流算法在运输优化、网络设计、资源分配等领域有重要应用。

## 核心概念

### 基本定义
- **网络流**：有向图中的流
- **容量**：边的最大流量
- **流量**：边的实际流量
- **最大流**：从源点到汇点的最大流量

### 关键特性
- **流量守恒**：除源点和汇点外，流入等于流出
- **容量限制**：流量不能超过容量
- **非负性**：流量非负
- **可增性**：可以通过增广路径增加流量

### 基本网络流算法

from typing import List, Tuple, Any, Callable, Optional
import time
from collections import deque

class NetworkFlowAlgorithms:
    """网络流算法示例"""

# Ford-Fulkerson算法
def ford_fulkerson(graph: List[List[int]], source: int, sink: int) -> int:
    """Ford-Fulkerson算法"""
    n = len(graph)
    residual_graph = [row[:] for row in graph]
    max_flow = 0
    
    def bfs(start: int, end: int) -> List[int]:
        """BFS找增广路径"""
        parent = [-1] * n
        queue = deque([start])
        parent[start] = start
        
        while queue:
            u = queue.popleft()
            for v in range(n):
                if parent[v] == -1 and residual_graph[u][v] > 0:
                    parent[v] = u
                    queue.append(v)
                    if v == end:
                        break
        
        if parent[end] == -1:
            return []
        
        # 重构路径
        path = []
        v = end
        while v != start:
            path.append(v)
            v = parent[v]
        path.append(start)
        path.reverse()
        
        return path
    
    # 主循环
    while True:
        path = bfs(source, sink)
        if not path:
            break
        
        # 计算路径上的最小容量
        min_capacity = float('inf')
        for i in range(len(path) - 1):
            u, v = path[i], path[i + 1]
            min_capacity = min(min_capacity, residual_graph[u][v])
        
        # 更新残量图
        for i in range(len(path) - 1):
            u, v = path[i], path[i + 1]
            residual_graph[u][v] -= min_capacity
            residual_graph[v][u] += min_capacity
        
        max_flow += min_capacity
    
    return max_flow

# Edmonds-Karp算法
def edmonds_karp(graph: List[List[int]], source: int, sink: int) -> int:
    """Edmonds-Karp算法"""
    n = len(graph)
    residual_graph = [row[:] for row in graph]
    max_flow = 0
    
    def bfs(start: int, end: int) -> Tuple[List[int], int]:
        """BFS找增广路径"""
        parent = [-1] * n
        queue = deque([start])
        parent[start] = start
        
        while queue:
            u = queue.popleft()
            for v in range(n):
                if parent[v] == -1 and residual_graph[u][v] > 0:
                    parent[v] = u
                    queue.append(v)
                    if v == end:
                        break
        
        if parent[end] == -1:
            return [], 0
        
        # 重构路径并计算最小容量
        path = []
        min_capacity = float('inf')
        v = end
        while v != start:
            path.append(v)
            u = parent[v]
            min_capacity = min(min_capacity, residual_graph[u][v])
            v = u
        path.append(start)
        path.reverse()
        
        return path, min_capacity
    
    # 主循环
    while True:
        path, min_capacity = bfs(source, sink)
        if not path:
            break
        
        # 更新残量图
        for i in range(len(path) - 1):
            u, v = path[i], path[i + 1]
            residual_graph[u][v] -= min_capacity
            residual_graph[v][u] += min_capacity
        
        max_flow += min_capacity
    
    return max_flow

# Dinic算法
def dinic(graph: List[List[int]], source: int, sink: int) -> int:
    """Dinic算法"""
    n = len(graph)
    residual_graph = [row[:] for row in graph]
    max_flow = 0
    
    def bfs_level(start: int, end: int) -> List[int]:
        """BFS计算层次"""
        level = [-1] * n
        queue = deque([start])
        level[start] = 0
        
        while queue:
            u = queue.popleft()
            for v in range(n):
                if level[v] == -1 and residual_graph[u][v] > 0:
                    level[v] = level[u] + 1
                    queue.append(v)
                    if v == end:
                        break
        
        return level
    
    def dfs_blocking_flow(u: int, end: int, level: List[int], flow: int) -> int:
        """DFS找阻塞流"""
        if u == end:
            return flow
        
        for v in range(n):
            if level[v] == level[u] + 1 and residual_graph[u][v] > 0:
                pushed = dfs_blocking_flow(v, end, level, min(flow, residual_graph[u][v]))
                if pushed > 0:
                    residual_graph[u][v] -= pushed
                    residual_graph[v][u] += pushed
                    return pushed
        
        return 0
    
    # 主循环
    while True:
        level = bfs_level(source, sink)
        if level[sink] == -1:
            break
        
        # 找阻塞流
        while True:
            pushed = dfs_blocking_flow(source, sink, level, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed
    
    return max_flow

# Push-Relabel算法
def push_relabel(graph: List[List[int]], source: int, sink: int) -> int:
    """Push-Relabel算法"""
    n = len(graph)
    residual_graph = [row[:] for row in graph]
    height = [0] * n
    excess = [0] * n
    
    # 初始化
    height[source] = n
    for v in range(n):
        if residual_graph[source][v] > 0:
            residual_graph[source][v] = 0
            residual_graph[v][source] = graph[source][v]
            excess[v] = graph[source][v]
    
    def push(u: int, v: int) -> None:
        """推送操作"""
        flow = min(excess[u], residual_graph[u][v])
        residual_graph[u][v] -= flow
        residual_graph[v][u] += flow
        excess[u] -= flow
        excess[v] += flow
    
    def relabel(u: int) -> None:
        """重标记操作"""
        min_height = float('inf')
        for v in range(n):
            if residual_graph[u][v] > 0:
                min_height = min(min_height, height[v])
        height[u] = min_height + 1
    
    # 主循环
    while True:
        # 找活跃节点
        active_nodes = []
        for u in range(n):
            if u != source and u != sink and excess[u] > 0:
                active_nodes.append(u)
        
        if not active_nodes:
            break
        
        for u in active_nodes:
            if excess[u] > 0:
                # 尝试推送
                pushed = False
                for v in range(n):
                    if residual_graph[u][v] > 0 and height[u] == height[v] + 1:
                        push(u, v)
                        pushed = True
                        break
                
                # 如果不能推送，则重标记
                if not pushed:
                    relabel(u)
    
    return excess[sink]

# 最小割算法
def min_cut(graph: List[List[int]], source: int, sink: int) -> Tuple[int, List[int]]:
    """最小割算法"""
    n = len(graph)
    residual_graph = [row[:] for row in graph]
    
    # 使用Ford-Fulkerson算法找最大流
    max_flow = ford_fulkerson(residual_graph, source, sink)
    
    # 使用BFS找最小割
    visited = [False] * n
    queue = deque([source])
    visited[source] = True
    
    while queue:
        u = queue.popleft()
        for v in range(n):
            if not visited[v] and residual_graph[u][v] > 0:
                visited[v] = True
                queue.append(v)
    
    # 找到割边
    cut_edges = []
    for u in range(n):
        for v in range(n):
            if visited[u] and not visited[v] and graph[u][v] > 0:
                cut_edges.append((u, v))
    
    return max_flow, cut_edges

# 最大流最小割定理
def max_flow_min_cut_theorem(graph: List[List[int]], source: int, sink: int) -> bool:
    """验证最大流最小割定理"""
    max_flow, cut_edges = min_cut(graph, source, sink)
    
    # 计算割的容量
    cut_capacity = sum(graph[u][v] for u, v in cut_edges)
    
    # 验证定理：最大流 = 最小割
    return max_flow == cut_capacity

## 性能分析

### 网络流算法分析

| 算法类型 | 时间复杂度 | 空间复杂度 | 适用场景 | 学习难度 |
|---------|-----------|-----------|----------|----------|
| Ford-Fulkerson | O(E * f) | O(V) | 一般情况 | 简单 |
| Edmonds-Karp | O(V * E²) | O(V) | 一般情况 | 简单 |
| Dinic | O(V² * E) | O(V) | 一般情况 | 中等 |
| Push-Relabel | O(V² * E) | O(V) | 一般情况 | 困难 |
| 最小割 | O(V * E²) | O(V) | 最小割问题 | 中等 |

### 算法选择分析

| 问题特征 | 推荐算法 | 优势 | 劣势 |
|---------|---------|------|------|
| 一般情况 | Edmonds-Karp | 简单、稳定 | 效率一般 |
| 效率要求高 | Dinic | 高效 | 实现复杂 |
| 理论分析 | Push-Relabel | 理论优美 | 实现复杂 |
| 最小割问题 | 最小割算法 | 功能完整 | 实现复杂 |
