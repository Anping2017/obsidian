# 最短路径算法

## 🎯 核心概念

**最短路径算法**是图算法中的重要分支，用于在加权图中找到从一个顶点到另一个顶点的最短路径。这些算法在路由、网络优化、游戏AI等领域有广泛应用。

## 🎨 算法实现

### 1. Dijkstra算法
```python
def dijkstra_algorithm():
    """Dijkstra算法实现"""
    
    def dijkstra_basic(graph, start):
        """基本Dijkstra算法"""
        import heapq
        
        # 初始化距离数组
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        
        # 优先队列：(距离, 顶点)
        pq = [(0, start)]
        visited = set()
        
        while pq:
            current_distance, current_vertex = heapq.heappop(pq)
            
            if current_vertex in visited:
                continue
            
            visited.add(current_vertex)
            
            # 更新邻居节点的距离
            for neighbor, weight in graph[current_vertex]:
                if neighbor not in visited:
                    new_distance = current_distance + weight
                    if new_distance < distances[neighbor]:
                        distances[neighbor] = new_distance
                        heapq.heappush(pq, (new_distance, neighbor))
        
        return distances
    
    def dijkstra_with_path(graph, start, end):
        """带路径的Dijkstra算法"""
        import heapq
        
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        previous = {vertex: None for vertex in graph}
        
        pq = [(0, start)]
        visited = set()
        
        while pq:
            current_distance, current_vertex = heapq.heappop(pq)
            
            if current_vertex in visited:
                continue
            
            visited.add(current_vertex)
            
            if current_vertex == end:
                break
            
            for neighbor, weight in graph[current_vertex]:
                if neighbor not in visited:
                    new_distance = current_distance + weight
                    if new_distance < distances[neighbor]:
                        distances[neighbor] = new_distance
                        previous[neighbor] = current_vertex
                        heapq.heappush(pq, (new_distance, neighbor))
        
        # 重构路径
        path = []
        current = end
        while current is not None:
            path.append(current)
            current = previous[current]
        path.reverse()
        
        return distances[end], path
    
    def dijkstra_with_priority_queue(graph, start):
        """使用优先队列的Dijkstra算法"""
        import heapq
        
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        
        # 使用优先队列
        pq = [(0, start)]
        visited = set()
        
        while pq:
            current_distance, current_vertex = heapq.heappop(pq)
            
            if current_vertex in visited:
                continue
            
            visited.add(current_vertex)
            
            for neighbor, weight in graph[current_vertex]:
                if neighbor not in visited:
                    new_distance = current_distance + weight
                    if new_distance < distances[neighbor]:
                        distances[neighbor] = new_distance
                        heapq.heappush(pq, (new_distance, neighbor))
        
        return distances
    
    return dijkstra_basic, dijkstra_with_path, dijkstra_with_priority_queue
```

### 2. Bellman-Ford算法
```python
def bellman_ford_algorithm():
    """Bellman-Ford算法实现"""
    
    def bellman_ford_basic(graph, start):
        """基本Bellman-Ford算法"""
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        
        # 松弛操作：进行V-1次
        for _ in range(len(graph) - 1):
            for vertex in graph:
                for neighbor, weight in graph[vertex]:
                    if distances[vertex] + weight < distances[neighbor]:
                        distances[neighbor] = distances[vertex] + weight
        
        return distances
    
    def bellman_ford_with_negative_cycle(graph, start):
        """带负环检测的Bellman-Ford算法"""
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        
        # 松弛操作：进行V-1次
        for _ in range(len(graph) - 1):
            for vertex in graph:
                for neighbor, weight in graph[vertex]:
                    if distances[vertex] + weight < distances[neighbor]:
                        distances[neighbor] = distances[vertex] + weight
        
        # 检测负环：再进行一次松弛
        for vertex in graph:
            for neighbor, weight in graph[vertex]:
                if distances[vertex] + weight < distances[neighbor]:
                    return None, "负环检测"
        
        return distances, "无负环"
    
    def bellman_ford_with_path(graph, start, end):
        """带路径的Bellman-Ford算法"""
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        previous = {vertex: None for vertex in graph}
        
        # 松弛操作
        for _ in range(len(graph) - 1):
            for vertex in graph:
                for neighbor, weight in graph[vertex]:
                    if distances[vertex] + weight < distances[neighbor]:
                        distances[neighbor] = distances[vertex] + weight
                        previous[neighbor] = vertex
        
        # 检测负环
        for vertex in graph:
            for neighbor, weight in graph[vertex]:
                if distances[vertex] + weight < distances[neighbor]:
                    return None, None, "负环检测"
        
        # 重构路径
        path = []
        current = end
        while current is not None:
            path.append(current)
            current = previous[current]
        path.reverse()
        
        return distances[end], path, "无负环"
    
    return bellman_ford_basic, bellman_ford_with_negative_cycle, bellman_ford_with_path
```

### 3. Floyd-Warshall算法
```python
def floyd_warshall_algorithm():
    """Floyd-Warshall算法实现"""
    
    def floyd_warshall_basic(graph):
        """基本Floyd-Warshall算法"""
        vertices = list(graph.keys())
        n = len(vertices)
        
        # 初始化距离矩阵
        dist = [[float('inf')] * n for _ in range(n)]
        for i in range(n):
            dist[i][i] = 0
        
        # 初始化边权重
        for i, vertex in enumerate(vertices):
            for neighbor, weight in graph[vertex]:
                j = vertices.index(neighbor)
                dist[i][j] = weight
        
        # Floyd-Warshall主算法
        for k in range(n):
            for i in range(n):
                for j in range(n):
                    if dist[i][k] + dist[k][j] < dist[i][j]:
                        dist[i][j] = dist[i][k] + dist[k][j]
        
        return dist, vertices
    
    def floyd_warshall_with_path(graph):
        """带路径的Floyd-Warshall算法"""
        vertices = list(graph.keys())
        n = len(vertices)
        
        # 初始化距离矩阵和路径矩阵
        dist = [[float('inf')] * n for _ in range(n)]
        next_vertex = [[None] * n for _ in range(n)]
        
        for i in range(n):
            dist[i][i] = 0
        
        # 初始化边权重
        for i, vertex in enumerate(vertices):
            for neighbor, weight in graph[vertex]:
                j = vertices.index(neighbor)
                dist[i][j] = weight
                next_vertex[i][j] = j
        
        # Floyd-Warshall主算法
        for k in range(n):
            for i in range(n):
                for j in range(n):
                    if dist[i][k] + dist[k][j] < dist[i][j]:
                        dist[i][j] = dist[i][k] + dist[k][j]
                        next_vertex[i][j] = next_vertex[i][k]
        
        return dist, next_vertex, vertices
    
    def reconstruct_path(next_vertex, vertices, start, end):
        """重构路径"""
        start_idx = vertices.index(start)
        end_idx = vertices.index(end)
        
        if next_vertex[start_idx][end_idx] is None:
            return []
        
        path = [start]
        current = start_idx
        
        while current != end_idx:
            current = next_vertex[current][end_idx]
            path.append(vertices[current])
        
        return path
    
    return floyd_warshall_basic, floyd_warshall_with_path, reconstruct_path

def floyd_warshall_examples():
    """Floyd-Warshall算法示例"""
    
    # 示例图
    graph = {
        'A': [('B', 4), ('C', 2)],
        'B': [('A', 4), ('C', 1), ('D', 5)],
        'C': [('A', 2), ('B', 1), ('D', 8), ('E', 10)],
        'D': [('B', 5), ('C', 8), ('E', 2)],
        'E': [('C', 10), ('D', 2)]
    }
    
    def floyd_warshall_example():
        """Floyd-Warshall算法示例"""
        dist, vertices = floyd_warshall_basic(graph)
        print("所有点对之间的最短距离:")
        for i, vertex1 in enumerate(vertices):
            for j, vertex2 in enumerate(vertices):
                if dist[i][j] != float('inf'):
                    print(f"{vertex1} -> {vertex2}: {dist[i][j]}")
        return dist, vertices
    
    def floyd_warshall_path_example():
        """Floyd-Warshall路径示例"""
        dist, next_vertex, vertices = floyd_warshall_with_path(graph)
        start = 'A'
        end = 'E'
        path = reconstruct_path(next_vertex, vertices, start, end)
        print(f"从{start}到{end}的最短路径: {path}")
        return path
    
    return floyd_warshall_example, floyd_warshall_path_example
```

```python
def network_routing_applications():

    def ospf_routing(network_topology, source, destination):

        distances = dijkstra_basic(network_topology, source)

    def bgp_routing(autonomous_systems, source_as, destination_as):
        """BGP路由算法"""
        # 边界网关协议中的路径选择
        # 考虑AS路径长度、本地偏好等因素
        return select_best_path(autonomous_systems, source_as, destination_as)
    
    def load_balancing_routing(servers, request, current_load):
        """负载均衡路由"""
        # 根据服务器负载选择最优路径
        best_server = min(servers, key=lambda s: current_load[s])
        return best_server
    
    return ospf_routing, bgp_routing, load_balancing_routing
```
