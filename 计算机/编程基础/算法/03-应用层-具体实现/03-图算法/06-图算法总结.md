# 图算法总结

## 🎯 核心概念

**图算法**是处理图数据结构的算法集合，包括图遍历、最短路径、最小生成树、拓扑排序、强连通分量、二分图匹配等。图算法在社交网络、网络路由、任务调度、编译器优化等领域有广泛应用。

### 1. 图遍历实现
```python
def graph_traversal_implementations():
    """图遍历实现"""
    
    def dfs_implementation(graph, start):
        """DFS实现"""
        visited = set()
        result = []
        
        def dfs_recursive(vertex):
            if vertex in visited:
                return
            visited.add(vertex)
            result.append(vertex)
            for neighbor in graph.get(vertex, []):
                dfs_recursive(neighbor)
        
        dfs_recursive(start)
        return result
    
    def bfs_implementation(graph, start):
        """BFS实现"""
        visited = set()
        queue = [start]
        result = []
        
        while queue:
            vertex = queue.pop(0)
            if vertex in visited:
                continue
            visited.add(vertex)
            result.append(vertex)
            for neighbor in graph.get(vertex, []):
                if neighbor not in visited:
                    queue.append(neighbor)
        
        return result
    
    def bidirectional_search(graph, start, end):
        """双向搜索实现"""
        if start == end:
            return [start]
        
        start_queue = [start]
        end_queue = [end]
        start_visited = {start: None}
        end_visited = {end: None}
        
        while start_queue and end_queue:
            # 从起点搜索
            current = start_queue.pop(0)
            for neighbor in graph.get(current, []):
                if neighbor in end_visited:
                    # 找到路径
                    path = []
                    node = current
                    while node is not None:
                        path.append(node)
                        node = start_visited[node]
                    path.reverse()
                    
                    node = neighbor
                    while node is not None:
                        path.append(node)
                        node = end_visited[node]
                    
                    return path
                
                if neighbor not in start_visited:
                    start_visited[neighbor] = current
                    start_queue.append(neighbor)
            
            # 从终点搜索
            current = end_queue.pop(0)
            for neighbor in graph.get(current, []):
                if neighbor in start_visited:
                    # 找到路径
                    path = []
                    node = neighbor
                    while node is not None:
                        path.append(node)
                        node = start_visited[node]
                    path.reverse()
                    
                    node = current
                    while node is not None:
                        path.append(node)
                        node = end_visited[node]
                    
                    return path
                
                if neighbor not in end_visited:
                    end_visited[neighbor] = current
                    end_queue.append(neighbor)
        
        return None
    
    def a_star_implementation(graph, start, end, heuristic):
        """A*算法实现"""
        open_set = [(0, start)]
        came_from = {}
        g_score = {start: 0}
        f_score = {start: heuristic(start, end)}
        
        while open_set:
            current = min(open_set, key=lambda x: x[0])[1]
            open_set.remove((f_score[current], current))
            
            if current == end:
                # 重构路径
                path = []
                while current in came_from:
                    path.append(current)
                    current = came_from[current]
                path.append(start)
                path.reverse()
                return path
            
            for neighbor in graph.get(current, []):
                tentative_g_score = g_score[current] + 1
                
                if neighbor not in g_score or tentative_g_score < g_score[neighbor]:
                    came_from[neighbor] = current
                    g_score[neighbor] = tentative_g_score
                    f_score[neighbor] = g_score[neighbor] + heuristic(neighbor, end)
                    
                    if (f_score[neighbor], neighbor) not in open_set:
                        open_set.append((f_score[neighbor], neighbor))
        
        return None
    
    return dfs_implementation, bfs_implementation, bidirectional_search, a_star_implementation
```

def shortest_path_implementations():
    """最短路径实现"""
    
    def dijkstra_implementation(graph, weights, start):
        """Dijkstra算法实现"""
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        visited = set()
        
        while len(visited) < len(graph):
            # 找到未访问的最小距离顶点
            current = min((v for v in graph if v not in visited), 
                         key=lambda v: distances[v])
            visited.add(current)
            
            for neighbor in graph[current]:
                if neighbor not in visited:
                    new_distance = distances[current] + weights.get((current, neighbor), 1)
                    if new_distance < distances[neighbor]:
                        distances[neighbor] = new_distance
        
        return distances
    
    def bellman_ford_implementation(graph, weights, start):
        """Bellman-Ford算法实现"""
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        
        # 松弛操作
        for _ in range(len(graph) - 1):
            for u in graph:
                for v in graph[u]:
                    if distances[u] + weights.get((u, v), 1) < distances[v]:
                        distances[v] = distances[u] + weights.get((u, v), 1)
        
        # 检测负环
        for u in graph:
            for v in graph[u]:
                if distances[u] + weights.get((u, v), 1) < distances[v]:
                    return None  # 存在负环
        
        return distances
    
    def floyd_warshall_implementation(graph, weights):
        """Floyd-Warshall算法实现"""
        vertices = list(graph.keys())
        n = len(vertices)
        
        # 初始化距离矩阵
        dist = [[float('inf')] * n for _ in range(n)]
        for i in range(n):
            dist[i][i] = 0
        
        # 设置初始边权重
        for u in graph:
            for v in graph[u]:
                i, j = vertices.index(u), vertices.index(v)
                dist[i][j] = weights.get((u, v), 1)
        
        # Floyd-Warshall算法
        for k in range(n):
            for i in range(n):
                for j in range(n):
                    if dist[i][k] + dist[k][j] < dist[i][j]:
                        dist[i][j] = dist[i][k] + dist[k][j]
        
        return dist
    
    def spfa_implementation(graph, weights, start):
        """SPFA算法实现"""
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        queue = [start]
        in_queue = {start}
        
        while queue:
            current = queue.pop(0)
            in_queue.remove(current)
            
            for neighbor in graph[current]:
                new_distance = distances[current] + weights.get((current, neighbor), 1)
                if new_distance < distances[neighbor]:
                    distances[neighbor] = new_distance
                    if neighbor not in in_queue:
                        queue.append(neighbor)
                        in_queue.add(neighbor)
        
        return distances
    
    return dijkstra_implementation, bellman_ford_implementation, floyd_warshall_implementation, spfa_implementation

### 3. 最小生成树实现
```python
def mst_implementations():
    """最小生成树实现"""
    
    def kruskal_implementation(graph, weights):
        """Kruskal算法实现"""
        edges = []
        for u in graph:
            for v in graph[u]:
                edges.append((weights.get((u, v), 1), u, v))
        
        edges.sort()
        
        # 并查集
        parent = {vertex: vertex for vertex in graph}
        
        def find(x):
            if parent[x] != x:
                parent[x] = find(parent[x])
            return parent[x]
        
        def union(x, y):
            px, py = find(x), find(y)
            if px != py:
                parent[px] = py
                return True
            return False
        
        mst = []
        for weight, u, v in edges:
            if union(u, v):
                mst.append((u, v, weight))
                if len(mst) == len(graph) - 1:
                    break
        
        return mst
    
    def prim_implementation(graph, weights, start):
        """Prim算法实现"""
        mst = []
        visited = {start}
        edges = []
        
        # 添加起始顶点的边
        for neighbor in graph[start]:
            edges.append((weights.get((start, neighbor), 1), start, neighbor))
        
        edges.sort()
        
        while len(visited) < len(graph):
            weight, u, v = edges.pop(0)
            
            if v not in visited:
                visited.add(v)
                mst.append((u, v, weight))
                
                # 添加新顶点的边
                for neighbor in graph[v]:
                    if neighbor not in visited:
                        edges.append((weights.get((v, neighbor), 1), v, neighbor))
                
                edges.sort()
        
        return mst
    
    def boruvka_implementation(graph, weights):
        """Boruvka算法实现"""
        vertices = list(graph.keys())
        n = len(vertices)
        
        # 初始化连通分量
        components = {i: {vertices[i]} for i in range(n)}
        mst = []
        
        while len(components) > 1:
            # 为每个连通分量找到最小边
            min_edges = {}
            for comp_id, comp_vertices in components.items():
                min_weight = float('inf')
                min_edge = None
                
                for u in comp_vertices:
                    for v in graph[u]:
                        if v not in comp_vertices:
                            weight = weights.get((u, v), 1)
                            if weight < min_weight:
                                min_weight = weight
                                min_edge = (u, v, weight)
                
                if min_edge:
                    min_edges[comp_id] = min_edge
            
            # 合并连通分量
            new_components = {}
            used = set()
            
            for comp_id, (u, v, weight) in min_edges.items():
                if u not in used and v not in used:
                    # 找到包含u和v的连通分量
                    comp_u = None
                    comp_v = None
                    
                    for cid, comp_vertices in components.items():
                        if u in comp_vertices:
                            comp_u = cid
                        if v in comp_vertices:
                            comp_v = cid
                    
                    if comp_u != comp_v:
                        # 合并连通分量
                        new_comp = components[comp_u] | components[comp_v]
                        new_components[len(new_components)] = new_comp
                        mst.append((u, v, weight))
                        used.update(new_comp)
            
            components = new_components
        
        return mst
    
    return kruskal_implementation, prim_implementation, boruvka_implementation

        """社区检测"""
        # 使用强连通分量检测社区
        visited = set()
        communities = []
        
        def dfs_community(vertex, community):
            if vertex in visited:
                return
            visited.add(vertex)
            community.append(vertex)
            for neighbor in graph.get(vertex, []):
                dfs_community(neighbor, community)
        
        for vertex in graph:
            if vertex not in visited:
                community = []
                dfs_community(vertex, community)
                communities.append(community)
        
        return communities
    
    def influence_analysis(graph):
        """影响力分析"""
        # 使用PageRank算法分析影响力
        vertices = list(graph.keys())
        n = len(vertices)
        damping_factor = 0.85
        
        # 初始化PageRank值
        pr = {vertex: 1.0 / n for vertex in vertices}
        
        # 迭代计算
        for _ in range(100):
            new_pr = {}
            for vertex in vertices:
                new_pr[vertex] = (1 - damping_factor) / n
                for neighbor in graph.get(vertex, []):
                    new_pr[vertex] += damping_factor * pr[neighbor] / len(graph.get(neighbor, []))
            pr = new_pr
        
        return pr
    
    return friend_recommendation, community_detection, influence_analysis

def network_routing_applications():
    """网络路由应用"""
    
    def shortest_path_routing(graph, weights, start, end):
        """最短路径路由"""
        # 使用Dijkstra算法
        distances = {vertex: float('inf') for vertex in graph}
        distances[start] = 0
        previous = {}
        visited = set()
        
        while len(visited) < len(graph):
            current = min((v for v in graph if v not in visited), 
                         key=lambda v: distances[v])
            visited.add(current)
            
            for neighbor in graph[current]:
                if neighbor not in visited:
                    new_distance = distances[current] + weights.get((current, neighbor), 1)
                    if new_distance < distances[neighbor]:
                        distances[neighbor] = new_distance
                        previous[neighbor] = current
        
        # 重构路径
        path = []
        current = end
        while current is not None:
            path.append(current)
            current = previous.get(current)
        path.reverse()
        
        return path
    
    def load_balancing(graph, weights, requests):
        """负载均衡"""
        # 使用最短路径算法进行负载均衡
        server_loads = {server: 0 for server in graph}
        routing_table = {}
        
        for request in requests:
            # 找到负载最小的服务器
            min_load_server = min(server_loads.keys(), key=lambda s: server_loads[s])
            
            # 计算到该服务器的路径
            path = shortest_path_routing(graph, weights, request['source'], min_load_server)
            routing_table[request['id']] = path
            
            # 更新服务器负载
            server_loads[min_load_server] += request['load']
        
        return routing_table, server_loads
    
    def network_topology_discovery(graph):
        """网络拓扑发现"""
        # 使用BFS发现网络拓扑
        visited = set()
        topology = {}
        
        def bfs_topology(start):

                current = queue.pop(0)
                for neighbor in graph.get(current, []):
                    if neighbor not in visited:
                        visited.add(neighbor)
                        topology[start].append(neighbor)
                        queue.append(neighbor)
        
        for vertex in graph:
            if vertex not in visited:
                bfs_topology(vertex)
        
        return topology
    
    return shortest_path_routing, load_balancing, network_topology_discovery
```

### 3. 任务调度
```python
def task_scheduling_applications():
    """任务调度应用"""
    
    def dependency_scheduling(tasks, dependencies):
        """依赖调度"""
        # 构建任务依赖图
        graph = {task: [] for task in tasks}
        in_degree = {task: 0 for task in tasks}
        
        for task, deps in dependencies.items():
            for dep in deps:
                graph[dep].append(task)
                in_degree[task] += 1
        
        # 使用拓扑排序
        queue = [task for task in tasks if in_degree[task] == 0]
        schedule = []
        
        while queue:
            current = queue.pop(0)
            schedule.append(current)
            
            for neighbor in graph[current]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)
        
        return schedule
    
    def resource_scheduling(tasks, resources, requirements):
        """资源调度"""
        # 构建任务-资源二分图
        graph = {task: [] for task in tasks}
        for task, req in requirements.items():
            for resource in resources:
                if resource >= req:
                    graph[task].append(resource)
        
        # 使用匈牙利算法求解
        match_left, match_right = hungarian_matching(graph, tasks, resources)
        
        return match_left, match_right
    
    def priority_scheduling(tasks, priorities, deadlines):
        """优先级调度"""
        # 按优先级和截止时间排序
        sorted_tasks = sorted(tasks, key=lambda t: (priorities[t], deadlines[t]))
        
        schedule = []
        current_time = 0
        
        for task in sorted_tasks:
            if current_time < deadlines[task]:
                schedule.append((task, current_time))
                current_time += 1
        
        return schedule
    
    return dependency_scheduling, resource_scheduling, priority_scheduling
```
