# 双向搜索

## 🎯 核心概念

**双向搜索(Bidirectional Search)**是一种搜索算法，它同时从起点和终点开始搜索，直到两个搜索路径相遇。这种方法可以显著减少搜索空间，提高搜索效率。

### 1. 双向BFS实现
```python
def bidirectional_bfs():
    """双向BFS实现"""
    
    def bidirectional_bfs_graph(graph, start, end):
        """图的双向BFS实现"""
        if start == end:
            return [start]
        
        # 正向搜索
        forward_queue = [start]
        forward_visited = {start: [start]}
        
        # 反向搜索
        backward_queue = [end]
        backward_visited = {end: [end]}
        
        while forward_queue and backward_queue:
            # 正向扩展
            if forward_queue:
                forward_node = forward_queue.pop(0)
                for neighbor in graph.get(forward_node, []):
                    if neighbor not in forward_visited:
                        forward_visited[neighbor] = forward_visited[forward_node] + [neighbor]
                        forward_queue.append(neighbor)
                        
                        # 检查是否相遇
                        if neighbor in backward_visited:
                            return (forward_visited[neighbor] + 
                                   backward_visited[neighbor][::-1][1:])
            
            # 反向扩展
            if backward_queue:
                backward_node = backward_queue.pop(0)
                for neighbor in graph.get(backward_node, []):
                    if neighbor not in backward_visited:
                        backward_visited[neighbor] = backward_visited[backward_node] + [neighbor]
                        backward_queue.append(neighbor)
                        
                        # 检查是否相遇
                        if neighbor in forward_visited:
                            return (forward_visited[neighbor] + 
                                   backward_visited[neighbor][::-1][1:])
        
        return None  # 无路径
    
    def bidirectional_bfs_tree(root, target):
        """树的双向BFS实现"""
        if root.val == target:
            return [root.val]
        
        # 正向搜索
        forward_queue = [root]
        forward_visited = {root.val: [root.val]}
        
        # 反向搜索（需要找到目标节点）
        target_node = find_target_node(root, target)
        if not target_node:
            return None
        
        backward_queue = [target_node]
        backward_visited = {target_node.val: [target_node.val]}
        
        while forward_queue and backward_queue:
            # 正向扩展
            if forward_queue:
                forward_node = forward_queue.pop(0)
                for child in [forward_node.left, forward_node.right]:
                    if child and child.val not in forward_visited:
                        forward_visited[child.val] = forward_visited[forward_node.val] + [child.val]
                        forward_queue.append(child)
                        
                        # 检查是否相遇
                        if child.val in backward_visited:
                            return (forward_visited[child.val] + 
                                   backward_visited[child.val][::-1][1:])
            
            # 反向扩展
            if backward_queue:
                backward_node = backward_queue.pop(0)
                for child in [backward_node.left, backward_node.right]:
                    if child and child.val not in backward_visited:
                        backward_visited[child.val] = backward_visited[backward_node.val] + [child.val]
                        backward_queue.append(child)
                        
                        # 检查是否相遇
                        if child.val in forward_visited:
                            return (forward_visited[child.val] + 
                                   backward_visited[child.val][::-1][1:])
        
        return None
    
    def find_target_node(root, target):
        """查找目标节点"""
        if not root:
            return None
        
        if root.val == target:
            return root
        
        left_result = find_target_node(root.left, target)
        if left_result:
            return left_result
        
        return find_target_node(root.right, target)
    
    return bidirectional_bfs_graph, bidirectional_bfs_tree
```

```python
def bidirectional_dfs():
    """双向DFS实现"""
    
    def bidirectional_dfs_graph(graph, start, end, max_depth=10):
        """图的双向DFS实现"""
        if start == end:
            return [start]
        
        # 正向搜索
        forward_path = []
        forward_visited = set()
        
        # 反向搜索
        backward_path = []
        backward_visited = set()
        
        def dfs_forward(node, depth):
            if depth > max_depth:
                return None
            
            forward_path.append(node)
            forward_visited.add(node)
            
            if node == end:
                return forward_path[:]
            
            for neighbor in graph.get(node, []):
                if neighbor not in forward_visited:
                    result = dfs_forward(neighbor, depth + 1)
                    if result:
                        return result
            
            forward_path.pop()
            forward_visited.remove(node)
            return None
        
        def dfs_backward(node, depth):
            if depth > max_depth:
                return None
            
            backward_path.append(node)
            backward_visited.add(node)
            
            if node == start:
                return backward_path[:]
            
            for neighbor in graph.get(node, []):
                if neighbor not in backward_visited:
                    result = dfs_backward(neighbor, depth + 1)
                    if result:
                        return result
            
            backward_path.pop()
            backward_visited.remove(node)
            return None
        
        # 同时进行双向搜索
        forward_result = dfs_forward(start, 0)
        if forward_result:
            return forward_result
        
        backward_result = dfs_backward(end, 0)
        if backward_result:
            return backward_result[::-1]
        
        return None
    
    def bidirectional_dfs_tree(root, target, max_depth=10):
        """树的双向DFS实现"""
        if root.val == target:
            return [root.val]
        
        # 正向搜索
        forward_path = []
        
        def dfs_forward(node, depth):
            if depth > max_depth or not node:
                return None
            
            forward_path.append(node.val)
            
            if node.val == target:
                return forward_path[:]
            
            # 搜索左子树
            result = dfs_forward(node.left, depth + 1)
            if result:
                return result
            
            # 搜索右子树
            result = dfs_forward(node.right, depth + 1)
            if result:
                return result
            
            forward_path.pop()
            return None
        
        # 反向搜索
        backward_path = []
        target_node = find_target_node(root, target)
        
        if not target_node:
            return None
        
        def dfs_backward(node, depth):
            if depth > max_depth or not node:
                return None
            
            backward_path.append(node.val)
            
            if node.val == root.val:
                return backward_path[:]
            
            # 搜索左子树
            result = dfs_backward(node.left, depth + 1)
            if result:
                return result
            
            # 搜索右子树
            result = dfs_backward(node.right, depth + 1)
            if result:
                return result
            
            backward_path.pop()
            return None
        
        # 同时进行双向搜索
        forward_result = dfs_forward(root, 0)
        if forward_result:
            return forward_result
        
        backward_result = dfs_backward(target_node, 0)
        if backward_result:
            return backward_result[::-1]
        
        return None
    
    return bidirectional_dfs_graph, bidirectional_dfs_tree
```

### 3. 双向A*搜索实现
```python
def bidirectional_astar():
    """双向A*搜索实现"""
    
    def bidirectional_astar_graph(graph, start, end, heuristic):
        """图的双向A*搜索实现"""
        if start == end:
            return [start]
        
        import heapq
        
        # 正向搜索
        forward_queue = [(heuristic(start, end), 0, start, [start])]
        forward_visited = set()
        
        # 反向搜索
        backward_queue = [(heuristic(end, start), 0, end, [end])]
        backward_visited = set()
        
        while forward_queue and backward_queue:
            # 正向扩展
            if forward_queue:
                f_cost, g_cost, node, path = heapq.heappop(forward_queue)
                
                if node in forward_visited:
                    continue
                
                forward_visited.add(node)
                
                if node == end:
                    return path
                
                # 检查是否相遇
                if node in backward_visited:
                    # 找到相遇点，合并路径
                    return merge_paths(forward_queue, backward_queue, node)
                
                for neighbor in graph.get(node, []):
                    if neighbor not in forward_visited:
                        new_path = path + [neighbor]
                        new_g_cost = g_cost + 1
                        new_f_cost = new_g_cost + heuristic(neighbor, end)
                        heapq.heappush(forward_queue, (new_f_cost, new_g_cost, neighbor, new_path))
            
            # 反向扩展
            if backward_queue:
                f_cost, g_cost, node, path = heapq.heappop(backward_queue)
                
                if node in backward_visited:
                    continue
                
                backward_visited.add(node)
                
                if node == start:
                    return path[::-1]
                
                # 检查是否相遇
                if node in forward_visited:
                    # 找到相遇点，合并路径
                    return merge_paths(forward_queue, backward_queue, node)
                
                for neighbor in graph.get(node, []):
                    if neighbor not in backward_visited:
                        new_path = path + [neighbor]
                        new_g_cost = g_cost + 1
                        new_f_cost = new_g_cost + heuristic(neighbor, start)
                        heapq.heappush(backward_queue, (new_f_cost, new_g_cost, neighbor, new_path))
        
        return None
    
    def merge_paths(forward_queue, backward_queue, meeting_point):
        """合并两个搜索路径"""
        # 找到包含meeting_point的路径
        for f_cost, g_cost, node, path in forward_queue:
            if node == meeting_point:
                forward_path = path
                break
        
        for f_cost, g_cost, node, path in backward_queue:
            if node == meeting_point:
                backward_path = path
                break
        
        # 合并路径
        return forward_path + backward_path[::-1][1:]
    
    def manhattan_distance(node1, node2):
        """曼哈顿距离启发式函数"""
        # 假设节点是坐标
        if isinstance(node1, tuple) and isinstance(node2, tuple):
            return abs(node1[0] - node2[0]) + abs(node1[1] - node2[1])
        return 0
    
    return bidirectional_astar_graph, merge_paths, manhattan_distance

def bidirectional_astar_examples():
    """双向A*搜索示例"""
    
    # 网格图示例
    grid_graph = {
        (0, 0): [(0, 1), (1, 0)],
        (0, 1): [(0, 0), (0, 2), (1, 1)],
        (0, 2): [(0, 1), (1, 2)],
        (1, 0): [(0, 0), (1, 1), (2, 0)],
        (1, 1): [(0, 1), (1, 0), (1, 2), (2, 1)],
        (1, 2): [(0, 2), (1, 1), (2, 2)],
        (2, 0): [(1, 0), (2, 1)],
        (2, 1): [(1, 1), (2, 0), (2, 2)],
        (2, 2): [(1, 2), (2, 1)]
    }
    
    def bidirectional_astar_example():
        """双向A*搜索示例"""
        start = (0, 0)
        end = (2, 2)
        
        path = bidirectional_astar_graph(grid_graph, start, end, manhattan_distance)
        print(f"双向A*路径: {path}")
        return path
    
    return bidirectional_astar_example
```

```python
def bidirectional_state_space():
    """双向状态空间搜索"""
    
    def bidirectional_state_space_search(initial_state, goal_state, get_neighbors):
        """双向状态空间搜索"""
        if initial_state == goal_state:
            return [initial_state]
        
        # 正向搜索
        forward_queue = [initial_state]
        forward_visited = {initial_state: [initial_state]}
        
        # 反向搜索
        backward_queue = [goal_state]
        backward_visited = {goal_state: [goal_state]}
        
        while forward_queue and backward_queue:
            # 正向扩展
            if forward_queue:
                current_state = forward_queue.pop(0)
                
                for neighbor in get_neighbors(current_state):
                    if neighbor not in forward_visited:
                        forward_visited[neighbor] = forward_visited[current_state] + [neighbor]
                        forward_queue.append(neighbor)

            # 反向扩展
            if backward_queue:
                current_state = backward_queue.pop(0)
                
                for neighbor in get_neighbors(current_state):
                    if neighbor not in backward_visited:

                                   backward_visited[neighbor][::-1][1:])

    def eight_puzzle_neighbors(state):
        """八数码问题的邻居状态"""
        # 找到空格位置
        empty_pos = state.index(0)
        row, col = empty_pos // 3, empty_pos % 3
        
        neighbors = []
        
        # 上下左右移动
        for dr, dc in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
            new_row, new_col = row + dr, col + dc
            
            if 0 <= new_row < 3 and 0 <= new_col < 3:
                new_empty_pos = new_row * 3 + new_col
                new_state = list(state)
                new_state[empty_pos], new_state[new_empty_pos] = new_state[new_empty_pos], new_state[empty_pos]
                neighbors.append(tuple(new_state))
        
        return neighbors
    
    def bidirectional_eight_puzzle(initial_state, goal_state):
        """双向八数码问题求解"""
        return bidirectional_state_space_search(initial_state, goal_state, eight_puzzle_neighbors)
    
    return bidirectional_state_space_search, eight_puzzle_neighbors, bidirectional_eight_puzzle
```

```python
        """获取单词的邻居"""
        neighbors = []
        
        for i in range(len(word)):
            for c in 'abcdefghijklmnopqrstuvwxyz':
                if c != word[i]:
                    new_word = word[:i] + c + word[i+1:]
                    if new_word in word_set:
                        neighbors.append(new_word)
        
        return neighbors
    
    def bidirectional_string_transformation_general(start, end, transformations):

        backward_visited = {end: [end]}

            if forward_queue:
                current_string = forward_queue.pop(0)
                
                for neighbor in get_string_neighbors(current_string, transformations):
                    if neighbor not in forward_visited:
                        forward_visited[neighbor] = forward_visited[current_string] + [neighbor]
                        forward_queue.append(neighbor)

                        if neighbor in backward_visited:
                            return (forward_visited[neighbor] +

            if backward_queue:
                current_string = backward_queue.pop(0)
                
                for neighbor in get_string_neighbors(current_string, transformations):
                    if neighbor not in backward_visited:
                        backward_visited[neighbor] = backward_visited[current_string] + [neighbor]
                        backward_queue.append(neighbor)

                new_string = string.replace(pattern, replacement, 1)
                neighbors.append(new_string)
        
        return neighbors
    
    return bidirectional_word_ladder, get_word_neighbors, bidirectional_string_transformation_general
```

```python
def bidirectional_graph_isomorphism():
    """双向图同构检测"""
    
    def bidirectional_graph_isomorphism_check(graph1, graph2):
        """双向图同构检测"""
        if len(graph1) != len(graph2):
            return False
        
        if not graph1:
            return True
        
        # 正向搜索：从graph1的节点映射到graph2的节点
        forward_queue = [list(graph1.keys())[0]]
        forward_mapping = {}
        
        # 反向搜索：从graph2的节点映射到graph1的节点
        backward_queue = [list(graph2.keys())[0]]
        backward_mapping = {}
        
        while forward_queue and backward_queue:
            # 正向扩展
            if forward_queue:
                node1 = forward_queue.pop(0)
                
                for node2 in graph2:
                    if node2 not in backward_mapping:
                        if is_compatible_mapping(node1, node2, graph1, graph2, forward_mapping, backward_mapping):
                            forward_mapping[node1] = node2
                            backward_mapping[node2] = node1
                            forward_queue.extend([neighbor for neighbor in graph1[node1] 
                                                if neighbor not in forward_mapping])
                            break
            
            # 反向扩展
            if backward_queue:
                node2 = backward_queue.pop(0)
                
                for node1 in graph1:
                    if node1 not in forward_mapping:
                        if is_compatible_mapping(node1, node2, graph1, graph2, forward_mapping, backward_mapping):
                            forward_mapping[node1] = node2
                            backward_mapping[node2] = node1
                            backward_queue.extend([neighbor for neighbor in graph2[node2] 
                                                  if neighbor not in backward_mapping])
                            break
        
        return len(forward_mapping) == len(graph1)
    
    def is_compatible_mapping(node1, node2, graph1, graph2, forward_mapping, backward_mapping):
        """检查映射是否兼容"""
        # 检查度数是否相同
        if len(graph1[node1]) != len(graph2[node2]):
            return False
        
        # 检查已映射的邻居是否兼容
        for neighbor1 in graph1[node1]:
            if neighbor1 in forward_mapping:
                mapped_neighbor2 = forward_mapping[neighbor1]
                if mapped_neighbor2 not in graph2[node2]:
                    return False
        
        for neighbor2 in graph2[node2]:
            if neighbor2 in backward_mapping:
                mapped_neighbor1 = backward_mapping[neighbor2]
                if mapped_neighbor1 not in graph1[node1]:
                    return False
        
        return True
    
    return bidirectional_graph_isomorphism_check, is_compatible_mapping
```

```python
def bidirectional_optimization_examples():
    """双向搜索优化示例"""
    
    def balanced_bidirectional_bfs(graph, start, end):
        """平衡的双向BFS"""
        if start == end:
            return [start]
        
        # 正向搜索
        forward_queue = [start]
        forward_visited = {start: [start]}
        
        # 反向搜索
        backward_queue = [end]
        backward_visited = {end: [end]}
        
        # 搜索轮次
        round_count = 0
        
        while forward_queue and backward_queue:
            round_count += 1
            
            # 交替搜索，平衡两个方向的搜索
            if round_count % 2 == 1:
                # 正向搜索
                current_queue = forward_queue
                current_visited = forward_visited
                other_visited = backward_visited
            else:
                # 反向搜索
                current_queue = backward_queue
                current_visited = backward_visited
                other_visited = forward_visited
            
            # 扩展当前队列
            new_queue = []
            for node in current_queue:
                for neighbor in graph.get(node, []):
                    if neighbor not in current_visited:
                        current_visited[neighbor] = current_visited[node] + [neighbor]
                        new_queue.append(neighbor)
                        
                        # 检查是否相遇
                        if neighbor in other_visited:
                            if round_count % 2 == 1:
                                return (current_visited[neighbor] + 
                                       other_visited[neighbor][::-1][1:])
                            else:
                                return (other_visited[neighbor] + 
                                       current_visited[neighbor][::-1][1:])
            
            if round_count % 2 == 1:
                forward_queue = new_queue
            else:
                backward_queue = new_queue
        
        return None
    
    def optimized_meeting_detection(graph, start, end):
        """优化的相遇检测"""
        if start == end:
            return [start]
        
        # 使用集合进行快速相遇检测
        forward_queue = [start]
        forward_visited = {start: [start]}
        forward_set = {start}
        
        backward_queue = [end]
        backward_visited = {end: [end]}
        backward_set = {end}
        
        while forward_queue and backward_queue:
            # 正向扩展
            if forward_queue:
                node = forward_queue.pop(0)
                for neighbor in graph.get(node, []):
                    if neighbor not in forward_set:
                        forward_set.add(neighbor)
                        forward_visited[neighbor] = forward_visited[node] + [neighbor]
                        forward_queue.append(neighbor)
                        
                        # 快速相遇检测

            # 反向扩展
            if backward_queue:
                node = backward_queue.pop(0)
                for neighbor in graph.get(node, []):
                    if neighbor not in backward_set:
                        backward_set.add(neighbor)
                        backward_visited[neighbor] = backward_visited[node] + [neighbor]
                        backward_queue.append(neighbor)
                        
                        # 快速相遇检测
                        if neighbor in forward_set:
                            return (forward_visited[neighbor] + 
                                   backward_visited[neighbor][::-1][1:])
        
        return None
```
