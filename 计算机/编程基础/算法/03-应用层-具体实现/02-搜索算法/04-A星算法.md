# A星算法

## 🎯 核心概念

**A星算法(A* Algorithm)**是一种启发式搜索算法，它结合了Dijkstra算法的最优性和贪心算法的效率。A*算法通过评估函数f(n) = g(n) + h(n)来选择最优路径，其中g(n)是从起点到当前节点的实际代价，h(n)是从当前节点到终点的启发式估计代价。

### 1. 基本A*实现
```python
def astar_basic():
    """基本A*算法实现"""
    
    def astar_search(graph, start, goal, heuristic):
        """A*搜索算法"""
        import heapq
        
        # 优先级队列：(f_cost, g_cost, node, path)
        open_set = [(0, 0, start, [start])]
        closed_set = set()
        
        # 记录每个节点的最佳g_cost
        g_costs = {start: 0}
        
        while open_set:
            f_cost, g_cost, current, path = heapq.heappop(open_set)
            
            if current in closed_set:
                continue
            
            closed_set.add(current)
            
            # 找到目标
            if current == goal:
                return path
            
            # 扩展邻居节点
            for neighbor in graph.get(current, []):
                if neighbor in closed_set:
                    continue
                
                # 计算新的g_cost
                new_g_cost = g_cost + 1  # 假设边权重为1
                
                # 如果找到更好的路径
                if neighbor not in g_costs or new_g_cost < g_costs[neighbor]:
                    g_costs[neighbor] = new_g_cost
                    h_cost = heuristic(neighbor, goal)
                    f_cost = new_g_cost + h_cost
                    
                    heapq.heappush(open_set, (f_cost, new_g_cost, neighbor, path + [neighbor]))
        
        return None  # 无路径
    
    def manhattan_distance(node1, node2):
        """曼哈顿距离启发式函数"""
        if isinstance(node1, tuple) and isinstance(node2, tuple):
            return abs(node1[0] - node2[0]) + abs(node1[1] - node2[1])
        return 0
    
    def euclidean_distance(node1, node2):
        """欧几里得距离启发式函数"""
        if isinstance(node1, tuple) and isinstance(node2, tuple):
            return ((node1[0] - node2[0])**2 + (node1[1] - node2[1])**2)**0.5
        return 0
    
    def zero_heuristic(node1, node2):
        """零启发式函数（退化为Dijkstra）"""
        return 0
    
    return astar_search, manhattan_distance, euclidean_distance, zero_heuristic
```

```python
        import heapq
        
        # 优先级队列：(f_cost, g_cost, node, path, path_cost)
        open_set = [(0, 0, start, [start], 0)]
        closed_set = set()

        g_costs = {start: 0}

            f_cost, g_cost, current, path, path_cost = heapq.heappop(open_set)
            
            if current in closed_set:
                continue
            
            closed_set.add(current)
            
            # 找到目标
            if current == goal:
                return path, path_cost
            
            # 扩展邻居节点
            for neighbor, weight in graph.get(current, []):
                if neighbor in closed_set:
                    continue
                
                # 计算新的g_cost
                new_g_cost = g_cost + weight
                new_path_cost = path_cost + weight
                
                # 如果找到更好的路径
                if neighbor not in g_costs or new_g_cost < g_costs[neighbor]:
                    g_costs[neighbor] = new_g_cost

                    f_cost = new_g_cost + h_cost
                    
                    heapq.heappush(open_set, (f_cost, new_g_cost, neighbor, path + [neighbor], new_path_cost))
        
        return None, float('inf')  # 无路径
    
    return astar_search_weighted, astar_with_path_cost
```

```python
def astar_game_ai():
    """游戏AI路径规划"""
    
    def astar_game_pathfinding(grid, start, goal, obstacles):
        """游戏AI路径规划"""
        import heapq
        
        # 优先级队列：(f_cost, g_cost, node, path)
        open_set = [(0, 0, start, [start])]
        closed_set = set()
        
        # 记录每个节点的最佳g_cost
        g_costs = {start: 0}
        
        def is_valid_position(pos):
            x, y = pos
            return (0 <= x < len(grid) and 0 <= y < len(grid[0]) and 
                    grid[x][y] not in obstacles)
        
        def get_neighbors(pos):
            x, y = pos
            neighbors = []
            for dx, dy in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
                new_pos = (x + dx, y + dy)
                if is_valid_position(new_pos):
                    neighbors.append(new_pos)
            return neighbors
        
        while open_set:
            f_cost, g_cost, current, path = heapq.heappop(open_set)
            
            if current in closed_set:
                continue
            
            closed_set.add(current)
            
            # 找到目标

                return path

                if neighbor in closed_set:
                    continue

                # 如果找到更好的路径

                    g_costs[neighbor] = new_g_cost
                    h_cost = manhattan_distance(neighbor, goal)

            x, y = pos
```

```python
            for dx in range(-robot_size, robot_size + 1):
                for dy in range(-robot_size, robot_size + 1):
                    check_x, check_y = x + dx, y + dy
                    if (check_x < 0 or check_x >= len(grid) or 
                        check_y < 0 or check_y >= len(grid[0]) or 
                        grid[check_x][check_y] == 1):
                        return False
            return True
        
        def get_neighbors(pos):
            x, y = pos
            neighbors = []
            for dx, dy in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
                new_pos = (x + dx, y + dy)
                if is_valid_position(new_pos):
                    neighbors.append(new_pos)
            return neighbors
        
        while open_set:
            f_cost, g_cost, current, path = heapq.heappop(open_set)
            
            if current in closed_set:
                continue
            
            closed_set.add(current)
            
            # 找到目标
            if current == goal:
                return path
            
            # 扩展邻居节点
            for neighbor in get_neighbors(current):
                if neighbor in closed_set:
                    continue
                
                # 计算新的g_cost
                new_g_cost = g_cost + 1
                
                # 如果找到更好的路径
                if neighbor not in g_costs or new_g_cost < g_costs[neighbor]:
                    g_costs[neighbor] = new_g_cost
                    h_cost = manhattan_distance(neighbor, goal)
                    f_cost = new_g_cost + h_cost
                    
                    heapq.heappush(open_set, (f_cost, new_g_cost, neighbor, path + [neighbor]))
        
        return None
    
    def astar_with_dynamic_obstacles(grid, start, goal, obstacle_timeline):
        """带动态障碍物的A*搜索"""
        import heapq
        
        # 优先级队列：(f_cost, g_cost, node, path, time)
        open_set = [(0, 0, start, [start], 0)]
        closed_set = set()
        
        # 记录每个节点的最佳g_cost
        g_costs = {(start, 0): 0}
        
        def is_valid_position(pos, time):
            x, y = pos
            if (x < 0 or x >= len(grid) or y < 0 or y >= len(grid[0])):
                return False
            
            # 检查静态障碍物
            if grid[x][y] == 1:
                return False
            
            # 检查动态障碍物
            if time in obstacle_timeline and pos in obstacle_timeline[time]:
                return False
            
            return True
        
        def get_neighbors(pos, time):
            x, y = pos
            neighbors = []
            for dx, dy in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
                new_pos = (x + dx, y + dy)
                if is_valid_position(new_pos, time + 1):
                    neighbors.append(new_pos)
            return neighbors
        
        while open_set:
            f_cost, g_cost, current, path, time = heapq.heappop(open_set)
            
            if (current, time) in closed_set:
                continue
            
            closed_set.add((current, time))
            
            # 找到目标
            if current == goal:
                return path
            
            # 扩展邻居节点
            for neighbor in get_neighbors(current, time):
                if (neighbor, time + 1) in closed_set:
                    continue
                
                # 计算新的g_cost
                new_g_cost = g_cost + 1
                
                # 如果找到更好的路径
                if (neighbor, time + 1) not in g_costs or new_g_cost < g_costs[(neighbor, time + 1)]:
                    g_costs[(neighbor, time + 1)] = new_g_cost
                    h_cost = manhattan_distance(neighbor, goal)
                    f_cost = new_g_cost + h_cost
                    
                    heapq.heappush(open_set, (f_cost, new_g_cost, neighbor, path + [neighbor], time + 1))
        
        return None
    
    return astar_robot_pathfinding, astar_with_dynamic_obstacles
```

```python
def astar_network_routing():

    def astar_network_pathfinding(network, start, goal, link_costs):

        g_costs = {start: 0}

            return link_costs.get((node1, node2), 1)
        
        while open_set:
            f_cost, g_cost, current, path = heapq.heappop(open_set)

                if neighbor in closed_set:
                    continue
                
                # 计算新的g_cost
                link_cost = get_link_cost(current, neighbor)
                new_g_cost = g_cost + link_cost
                
                # 如果找到更好的路径
                if neighbor not in g_costs or new_g_cost < g_costs[neighbor]:
                    g_costs[neighbor] = new_g_cost
                    h_cost = 0  # 网络路由通常不使用启发式函数
                    f_cost = new_g_cost + h_cost
                    
                    heapq.heappush(open_set, (f_cost, new_g_cost, neighbor, path + [neighbor]))
        
        return None

            return network.get(node, [])

            return link_costs.get((node1, node2), 1)
        
        def has_sufficient_bandwidth(node1, node2):
            # 检查链路是否有足够带宽
            return True  # 简化实现
        
        while open_set:
            f_cost, g_cost, current, path = heapq.heappop(open_set)
            
            if current in closed_set:
                continue
            
            closed_set.add(current)
            
            # 找到目标
            if current == goal:
                return path
            
            # 扩展邻居节点
            for neighbor in get_neighbors(current):
                if neighbor in closed_set:
                    continue
                
                # 检查带宽要求

                    f_cost = new_g_cost + h_cost
                    
                    heapq.heappush(open_set, (f_cost, new_g_cost, neighbor, path + [neighbor]))
        
        return None
    
    return astar_network_pathfinding, astar_with_bandwidth
```
