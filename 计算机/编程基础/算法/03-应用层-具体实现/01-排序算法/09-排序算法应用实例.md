```python
def basic_grade_sorting():
    """基础成绩排序"""
    class Student:
        def __init__(self, name, grade):
            self.name = name
            self.grade = grade
        
        def __repr__(self):
            return f"{self.name}: {self.grade}"
    
    # 学生数据
    students = [
        Student("Alice", 85),
        Student("Bob", 92),
        Student("Charlie", 78),
        Student("David", 96),
        Student("Eve", 88)
    ]
    
    # 按成绩降序排序（高分在前）
    def sort_by_grade_desc(students):
        """按成绩降序排序"""
        return sorted(students, key=lambda s: s.grade, reverse=True)
    
    # 按成绩升序排序（低分在前）
    def sort_by_grade_asc(students):
        """按成绩升序排序"""
        return sorted(students, key=lambda s: s.grade)
    
    # 按姓名排序
    def sort_by_name(students):
        """按姓名排序"""
        return sorted(students, key=lambda s: s.name)
    
    # 多关键字排序：先按成绩，再按姓名
    def sort_by_grade_then_name(students):
        """先按成绩再按姓名排序"""
        return sorted(students, key=lambda s: (-s.grade, s.name))
    
    return students

def advanced_grade_sorting():
    """高级成绩排序"""
    class Student:
        def __init__(self, name, math, english, science):
            self.name = name
            self.math = math
            self.english = english
            self.science = science

    def sort_by_total(students):
        """按总分排序"""
        return sorted(students, key=lambda s: s.total, reverse=True)
    
    # 按平均分排序
    def sort_by_average(students):
        """按平均分排序"""
        return sorted(students, key=lambda s: s.average, reverse=True)

    def sort_by_math(students):
```

```python
def advanced_file_sorting():
    """高级文件排序"""
    import os
    import re
    
    class FileInfo:
        def __init__(self, path):
            self.path = path
            self.name = os.path.basename(path)
            self.size = os.path.getsize(path)
            self.modified_time = os.path.getmtime(path)
            self.extension = os.path.splitext(path)[1]
    
    files = [
        FileInfo("/path/to/file1.txt"),
        FileInfo("/path/to/file10.txt"),
        FileInfo("/path/to/file2.txt"),
        FileInfo("/path/to/file20.txt"),
        FileInfo("/path/to/file3.txt")
    ]
    
    # 自然排序（处理数字）
    def natural_sort(files):
        """自然排序"""
        def natural_key(file_info):
            # 将文件名中的数字提取出来进行排序
            parts = re.split(r'(\d+)', file_info.name)
            return [int(part) if part.isdigit() else part.lower() for part in parts]
        
        return sorted(files, key=natural_key)
    
    # 按文件类型分组排序
    def sort_by_type_then_name(files):
        """按文件类型分组排序"""
        return sorted(files, key=lambda f: (f.extension, f.name.lower()))
    
    # 按大小范围排序
    def sort_by_size_range(files):
        """按大小范围排序"""
        def size_category(file_info):
            size = file_info.size
            if size < 1024:  # < 1KB
                return 0
            elif size < 1024 * 1024:  # < 1MB
                return 1
            elif size < 1024 * 1024 * 1024:  # < 1GB
                return 2
            else:  # >= 1GB
                return 3
        
        return sorted(files, key=lambda f: (size_category(f), f.size))
    
    return files
```

```python
        """按性价比排序"""

def advanced_product_sorting():
    """高级商品排序"""
    class Product:
        def __init__(self, name, price, rating, sales_count, category, brand):
            self.name = name
            self.price = price
            self.rating = rating
            self.sales_count = sales_count
            self.category = category
            self.brand = brand
    
    products = [
        Product("MacBook Pro", 1999.99, 4.8, 50, "Laptop", "Apple"),
        Product("Dell XPS", 1299.99, 4.6, 80, "Laptop", "Dell"),
        Product("iPhone 13", 799.99, 4.7, 200, "Phone", "Apple"),
        Product("Samsung Galaxy", 699.99, 4.5, 150, "Phone", "Samsung"),
        Product("AirPods", 159.99, 4.4, 300, "Accessory", "Apple")
    ]
    
    # 多维度排序：类别 > 品牌 > 评分
    def multi_dimensional_sort(products):
        """多维度排序"""
        return sorted(products, key=lambda p: (p.category, p.brand, p.rating), reverse=True)
    
    # 价格区间排序
    def sort_by_price_range(products):
        """按价格区间排序"""
        def price_range(product):
            price = product.price
            if price < 100:
                return 0  # 低价
            elif price < 500:
                return 1  # 中价
            elif price < 1000:
                return 2  # 高价
            else:
                return 3  # 超高价
        
        return sorted(products, key=lambda p: (price_range(p), p.price))
    
    # 综合评分排序（考虑评分和销量）
    def comprehensive_rating_sort(products):
        """综合评分排序"""
        def composite_score(product):
            # 综合评分 = 评分 * log(销量) * 0.1
            import math
            return product.rating * math.log(product.sales_count + 1) * 0.1
        
        return sorted(products, key=composite_score, reverse=True)
    
    return products
```

```python
def priority_task_sorting():
    """优先级任务排序"""
    class Task:
        def __init__(self, name, priority, deadline, execution_time):
            self.name = name
            self.priority = priority  # 1-10，10最高
            self.deadline = deadline  # 时间戳
            self.execution_time = execution_time  # 秒
    
    tasks = [
        Task("System Update", 8, 1640995200, 300),  # 高优先级
        Task("User Login", 5, 1640995800, 10),
        Task("File Backup", 3, 1641000000, 600),

        Task("Critical Fix", 10, 1640995300, 120)  # 最高优先级
    ]
    
    # 按优先级排序
    def sort_by_priority(tasks):
        """按优先级排序"""
        return sorted(tasks, key=lambda t: t.priority, reverse=True)
    
    # 按截止时间排序（最早截止时间优先）
    def sort_by_deadline(tasks):
        """按截止时间排序"""
        return sorted(tasks, key=lambda t: t.deadline)
    
    # 最短作业优先
    def shortest_job_first(tasks):
        """最短作业优先"""
        return sorted(tasks, key=lambda t: t.execution_time)
    
    # 最早截止时间优先
    def earliest_deadline_first(tasks):
        """最早截止时间优先"""
        return sorted(tasks, key=lambda t: t.deadline)

def advanced_task_scheduling():
    """高级任务调度"""
    import time
    
    class Task:
        def __init__(self, name, priority, deadline, execution_time, arrival_time):
            self.name = name
            self.priority = priority
            self.deadline = deadline
            self.execution_time = execution_time
            self.arrival_time = arrival_time
            self.remaining_time = execution_time
    
    tasks = [
        Task("Task A", 5, time.time() + 3600, 300, time.time()),
        Task("Task B", 8, time.time() + 1800, 120, time.time() + 60),
        Task("Task C", 3, time.time() + 7200, 600, time.time() + 120),
        Task("Task D", 10, time.time() + 900, 60, time.time() + 180)
    ]
    
    # 轮转调度排序
    def round_robin_sort(tasks, time_quantum=100):
        """轮转调度排序"""
        # 按到达时间排序
        return sorted(tasks, key=lambda t: t.arrival_time)
    
    # 多级反馈队列排序
    def multilevel_feedback_queue(tasks):
        """多级反馈队列排序"""
        # 按优先级分组，同优先级按剩余时间排序
        return sorted(tasks, key=lambda t: (t.priority, t.remaining_time), reverse=True)
    
    # 实时调度排序
    def real_time_scheduling(tasks):
        """实时调度排序"""
        current_time = time.time()
        
        def urgency_score(task):
            # 紧急度 = 优先级 * (截止时间 - 当前时间) / 执行时间
            time_remaining = task.deadline - current_time
            if time_remaining <= 0:
                return float('inf')  # 已过期
            return task.priority * time_remaining / task.execution_time
        
        return sorted(tasks, key=urgency_score, reverse=True)
    
    return tasks
```

```python
        """按相关性排序"""

        """按点击量排序"""

    # 按新鲜度排序
    def sort_by_freshness(results):
        """按新鲜度排序"""
        return sorted(results, key=lambda r: r.freshness, reverse=True)
    
    # 综合排序（相关性 + 点击量 + 新鲜度）
    def comprehensive_search_sort(results):
        """综合搜索排序"""
        def composite_score(result):
            # 综合评分 = 相关性 * 0.5 + 点击量归一化 * 0.3 + 新鲜度归一化 * 0.2
            max_clicks = max(r.click_count for r in results)
            max_freshness = max(r.freshness for r in results)
            
            click_normalized = result.click_count / max_clicks
            freshness_normalized = result.freshness / max_freshness
            
            return (result.relevance_score * 0.5 + 
                   click_normalized * 0.3 + 
                   freshness_normalized * 0.2)
        
        return sorted(results, key=composite_score, reverse=True)
    
    return results
```

```python
            sorted_records = sorted(records, key=lambda r: r.age, reverse=not ascending)
        elif sort_key == 'name':
            sorted_records = sorted(records, key=lambda r: r.name, reverse=not ascending)
        else:
            sorted_records = records
        
        # 分页
        start = (page - 1) * page_size
        end = start + page_size
        return sorted_records[start:end]
    
    return records

def advanced_database_sorting():

        DatabaseRecord(5, "Eve", 26, 52000, "Engineering", "2020-09-30", 8.9)
    ]
    
    # 动态排序
    def dynamic_sort(records, sort_config):
        """动态排序"""
        # sort_config = [('salary', False), ('age', True)]
        def sort_key(record):
            key_values = []
            for field, ascending in sort_config:
                value = getattr(record, field)
                key_values.append(value if ascending else -value)
            return tuple(key_values)
        
        return sorted(records, key=sort_key)
    
    # 条件排序
    def conditional_sort(records, condition_field, condition_value, sort_field):
        """条件排序"""
        # 先筛选，再排序
        filtered_records = [r for r in records if getattr(r, condition_field) == condition_value]
        return sorted(filtered_records, key=lambda r: getattr(r, sort_field))
    
    # 复合排序（多字段加权）
    def weighted_sort(records, weights):
        """加权排序"""
        # weights = {'salary': 0.4, 'performance_score': 0.6}
        def weighted_score(record):
            score = 0
            for field, weight in weights.items():
                value = getattr(record, field)
                # 归一化处理
                max_value = max(getattr(r, field) for r in records)
                normalized_value = value / max_value
                score += normalized_value * weight
            return score
        
        return sorted(records, key=weighted_score, reverse=True)
    
    return records
```

```python
def big_data_sorting_optimization():
    """大数据排序优化实例"""

def memory_efficient_sorting():
    """内存高效排序"""
    import heapq
    import tempfile
    import os
    
    def external_sort(large_data, chunk_size=1000):
        """外部排序（适用于大数据）"""
        # 1. 分块排序
        chunks = []
        for i in range(0, len(large_data), chunk_size):
            chunk = large_data[i:i + chunk_size]
            chunk.sort()
            
            # 写入临时文件
            temp_file = tempfile.NamedTemporaryFile(mode='w', delete=False)
            for item in chunk:
                temp_file.write(f"{item}\n")
            temp_file.close()
            chunks.append(temp_file.name)
        
        # 2. 多路归并
        def merge_chunks(chunk_files):
            """合并排序后的块"""
            # 打开所有文件
            files = [open(f, 'r') for f in chunk_files]
            heap = []
            
            # 初始化堆
            for i, file in enumerate(files):
                line = file.readline()
                if line:
                    heapq.heappush(heap, (int(line.strip()), i, file))
            
            # 归并排序
            result = []
            while heap:
                value, file_index, file = heapq.heappop(heap)
                result.append(value)
                
                # 读取下一个值
                line = file.readline()
                if line:
                    heapq.heappush(heap, (int(line.strip()), file_index, file))
                else:
                    file.close()
            
            return result
        
        # 3. 清理临时文件
        result = merge_chunks(chunks)
        for chunk_file in chunks:
            os.unlink(chunk_file)
        
        return result
    
    return external_sort

def parallel_sorting():
    """并行排序"""
    import multiprocessing as mp
    from concurrent.futures import ProcessPoolExecutor
    
    def parallel_merge_sort(data, num_processes=4):
        """并行归并排序"""
        if len(data) <= 1000:  # 小数据量直接排序
            return sorted(data)
        
        # 分割数据
        chunk_size = len(data) // num_processes
        chunks = [data[i:i + chunk_size] for i in range(0, len(data), chunk_size)]
        
        # 并行排序
        with ProcessPoolExecutor(max_workers=num_processes) as executor:
            sorted_chunks = list(executor.map(sorted, chunks))
        
        # 合并结果
        def merge_two_sorted_lists(list1, list2):
            """合并两个有序列表"""
            result = []
            i = j = 0
            
            while i < len(list1) and j < len(list2):
                if list1[i] <= list2[j]:
                    result.append(list1[i])
                    i += 1
                else:
                    result.append(list2[j])
                    j += 1
            
            result.extend(list1[i:])
            result.extend(list2[j:])
            return result
        
        # 递归合并
        while len(sorted_chunks) > 1:
            new_chunks = []
            for i in range(0, len(sorted_chunks), 2):
                if i + 1 < len(sorted_chunks):
                    merged = merge_two_sorted_lists(sorted_chunks[i], sorted_chunks[i + 1])
                    new_chunks.append(merged)
                else:
                    new_chunks.append(sorted_chunks[i])
            sorted_chunks = new_chunks
        
        return sorted_chunks[0] if sorted_chunks else []
    
    return parallel_merge_sort

def adaptive_sorting():
    """自适应排序"""
    def adaptive_sort(data):
        """根据数据特征选择最优排序算法"""
        n = len(data)
        
        # 分析数据特征
        is_sorted = all(data[i] <= data[i + 1] for i in range(n - 1))
        is_reverse_sorted = all(data[i] >= data[i + 1] for i in range(n - 1))
        
        # 检查数据范围
        min_val, max_val = min(data), max(data)
        range_size = max_val - min_val + 1
        
        # 检查重复元素
        unique_count = len(set(data))
        duplicate_ratio = unique_count / n
        
        # 选择算法
        if is_sorted:
            return data  # 已排序
        elif is_reverse_sorted:
            return data[::-1]  # 逆序
        elif n <= 10:
            return insertion_sort(data)  # 小数据量
        elif range_size <= n and duplicate_ratio < 0.5:
            return counting_sort(data, min_val, max_val)  # 计数排序
        elif duplicate_ratio > 0.8:
            return bucket_sort(data)  # 桶排序
        else:
            return quick_sort(data)  # 快速排序
    
    def insertion_sort(data):
        """插入排序"""
        for i in range(1, len(data)):
            key = data[i]
            j = i - 1
            while j >= 0 and data[j] > key:
                data[j + 1] = data[j]
                j -= 1
            data[j + 1] = key
        return data
    
    def counting_sort(data, min_val, max_val):
        """计数排序"""
        count = [0] * (max_val - min_val + 1)
        for num in data:
            count[num - min_val] += 1
        
        result = []
        for i in range(len(count)):
            result.extend([i + min_val] * count[i])
        return result
    
    def bucket_sort(data):
        """桶排序"""
        if not data:
            return data
        
        min_val, max_val = min(data), max(data)
        bucket_count = len(data)
        buckets = [[] for _ in range(bucket_count)]
        
        for num in data:
            bucket_index = int((num - min_val) / (max_val - min_val) * (bucket_count - 1))
            buckets[bucket_index].append(num)
        
        result = []
        for bucket in buckets:
            result.extend(sorted(bucket))
        return result
    
    def quick_sort(data):
        """快速排序"""
        if len(data) <= 1:
            return data
        
        pivot = data[len(data) // 2]
        left = [x for x in data if x < pivot]
        middle = [x for x in data if x == pivot]
        right = [x for x in data if x > pivot]
        
        return quick_sort(left) + middle + quick_sort(right)
    
    return adaptive_sort
```
