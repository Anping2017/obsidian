# AC自动机

## 🎯 核心概念

**AC自动机(Aho-Corasick Automaton)**是一种多模式串匹配算法，基于Trie树和KMP算法的思想，能够同时匹配多个模式串。AC自动机通过构建失败指针(failure pointer)实现高效的字符串匹配，在文本搜索、病毒检测、关键词过滤等领域有重要应用。

def ac_automaton_construction():
    """AC自动机构建"""
    
    class ACNode:
        """AC自动机节点"""
        def __init__(self):
            self.children = {}  # 子节点
            self.fail = None    # 失败指针
            self.output = []    # 输出函数
            self.is_end = False # 是否为模式串结尾
    
    class ACAutomaton:
        """AC自动机"""
        def __init__(self):
            self.root = ACNode()
        
        def add_pattern(self, pattern):
            """添加模式串"""
            node = self.root
            for char in pattern:
                if char not in node.children:
                    node.children[char] = ACNode()
                node = node.children[char]
            node.is_end = True
            node.output.append(pattern)
        
        def build_failure_pointers(self):
            """构建失败指针"""
            from collections import deque
            
            queue = deque()
            
            # 根节点的失败指针为None
            self.root.fail = None
            
            # 根节点的子节点失败指针指向根节点
            for child in self.root.children.values():
                child.fail = self.root
                queue.append(child)
            
            # BFS构建失败指针
            while queue:
                current = queue.popleft()
                
                for char, child in current.children.items():
                    queue.append(child)
                    
                    # 查找失败指针
                    fail_node = current.fail
                    while fail_node and char not in fail_node.children:
                        fail_node = fail_node.fail
                    
                    if fail_node and char in fail_node.children:
                        child.fail = fail_node.children[char]
                    else:
                        child.fail = self.root
                    
                    # 合并输出函数
                    child.output.extend(child.fail.output)
        
        def search(self, text):
            """搜索模式串"""
            results = []
            current = self.root
            
            for i, char in enumerate(text):
                # 状态转移
                while current and char not in current.children:
                    current = current.fail
                
                if current:
                    current = current.children[char]
                else:
                    current = self.root
                
                # 收集匹配结果
                if current.output:
                    for pattern in current.output:
                        results.append((i - len(pattern) + 1, pattern))
            
            return results
    
    return ACAutomaton

def advanced_ac_automaton():
    """高级AC自动机实现"""
    
    class AdvancedACNode:
        """高级AC自动机节点"""
        def __init__(self):
            self.children = {}
            self.fail = None
            self.output = []
            self.is_end = False
            self.depth = 0
            self.count = 0
    
    class AdvancedACAutomaton:
        """高级AC自动机"""
        def __init__(self):
            self.root = AdvancedACNode()
            self.patterns = []
        
        def add_pattern(self, pattern):
            """添加模式串"""
            node = self.root
            for char in pattern:
                if char not in node.children:
                    node.children[char] = AdvancedACNode()
                node = node.children[char]
                node.depth += 1
            node.is_end = True
            node.output.append(pattern)
            self.patterns.append(pattern)
        
        def build_failure_pointers(self):
            """构建失败指针"""
            from collections import deque
            
            queue = deque()
            self.root.fail = None
            
            for child in self.root.children.values():
                child.fail = self.root
                queue.append(child)
            
            while queue:
                current = queue.popleft()
                
                for char, child in current.children.items():
                    queue.append(child)
                    
                    fail_node = current.fail
                    while fail_node and char not in fail_node.children:
                        fail_node = fail_node.fail
                    
                    if fail_node and char in fail_node.children:
                        child.fail = fail_node.children[char]
                    else:
                        child.fail = self.root
                    
                    child.output.extend(child.fail.output)
        
        def search(self, text):
            """搜索模式串"""
            results = []
            current = self.root
            
            for i, char in enumerate(text):
                while current and char not in current.children:
                    current = current.fail
                
                if current:
                    current = current.children[char]
                else:
                    current = self.root
                
                current.count += 1
                
                if current.output:
                    for pattern in current.output:
                        results.append((i - len(pattern) + 1, pattern))
            
            return results
        
        def get_statistics(self):
            """获取统计信息"""
            stats = {
                "pattern_count": len(self.patterns),
                "node_count": self._count_nodes(self.root),
                "max_depth": self._get_max_depth(self.root)
            }
            return stats
        
        def _count_nodes(self, node):
            """计算节点数量"""
            count = 1
            for child in node.children.values():
                count += self._count_nodes(child)
            return count
        
        def _get_max_depth(self, node):
            """获取最大深度"""
            if not node.children:
                return 0
            
            max_depth = 0
            for child in node.children.values():
                max_depth = max(max_depth, self._get_max_depth(child) + 1)
            return max_depth
    
    return AdvancedACAutomaton

            ac.add_pattern(pattern)

        results = ac.search(text)
        
        # 按位置排序
        results.sort(key=lambda x: x[0])
        
        return results

        ac = ACAutomaton()

            ac.add_pattern(keyword)
        ac.build_failure_pointers()
        
        results = ac.search(text)
        
        # 标记敏感词
        filtered_text = text
        for pos, keyword in results:

        return filtered_text, results
