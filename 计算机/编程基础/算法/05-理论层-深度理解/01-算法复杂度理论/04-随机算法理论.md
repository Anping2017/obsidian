# 随机算法理论

## 概述

随机算法理论（Randomized Algorithm Theory）是计算复杂性理论的重要分支，研究使用随机性来解决问题的算法。随机算法通过引入随机性来改善算法的性能，在某些情况下可以达到比确定性算法更好的时间复杂度或空间复杂度。

## 核心概念

### 基本定义
- **随机算法**：使用随机性的算法
- **概率分析**：分析算法性能的概率方法
- **期望复杂度**：算法运行时间的期望值
- **高概率**：以高概率满足某个性质

### 关键特性
- **随机性**：算法中使用随机数
- **概率性**：结果具有概率性
- **期望性**：性能有期望保证
- **鲁棒性**：对输入分布不敏感

from typing import List, Tuple, Any, Callable, Optional
import time
import random
import math

class RandomizedAlgorithms:
    """随机算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.performance_guarantees = {}
    
    def add_algorithm(self, name: str, algorithm: Callable, guarantee: str) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
        self.performance_guarantees[name] = guarantee
    
    def solve_problem(self, name: str, input_data: Any) -> Tuple[Any, float, str]:
        """解决问题"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        start_time = time.time()
        result = algorithm(input_data)
        end_time = time.time()
        
        return result, end_time - start_time, self.performance_guarantees[name]
    
    def run_multiple_times(self, name: str, input_data: Any, times: int) -> List[Tuple[Any, float]]:
        """多次运行算法"""
        results = []
        for _ in range(times):
            result, time_taken, _ = self.solve_problem(name, input_data)
            results.append((result, time_taken))
        return results

# 随机快速排序
def randomized_quicksort(data: List[int]) -> List[int]:
    """随机快速排序 - 期望O(n log n)"""
    if len(data) <= 1:
        return data
    
    # 随机选择基准
    pivot_index = random.randint(0, len(data) - 1)
    pivot = data[pivot_index]
    
    # 分割
    left = [x for x in data if x < pivot]
    middle = [x for x in data if x == pivot]
    right = [x for x in data if x > pivot]
    
    # 递归排序
    return randomized_quicksort(left) + middle + randomized_quicksort(right)

# 随机选择算法
def randomized_select(data: List[int], k: int) -> int:
    """随机选择第k小元素 - 期望O(n)"""
    if len(data) == 1:
        return data[0]
    
    # 随机选择基准
    pivot_index = random.randint(0, len(data) - 1)
    pivot = data[pivot_index]
    
    # 分割
    left = [x for x in data if x < pivot]
    middle = [x for x in data if x == pivot]
    right = [x for x in data if x > pivot]
    
    # 递归选择
    if k <= len(left):
        return randomized_select(left, k)
    elif k <= len(left) + len(middle):
        return pivot
    else:
        return randomized_select(right, k - len(left) - len(middle))

# 随机化最小生成树
def randomized_mst(graph: List[List[Tuple[int, int]]]) -> List[Tuple[int, int, int]]:
    """随机化最小生成树 - 期望O(E log V)"""
    n = len(graph)
    mst = []
    parent = list(range(n))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # 收集所有边
    edges = []
    for u in range(n):
        for v, weight in graph[u]:
            if u < v:  # 避免重复边
                edges.append((weight, u, v))
    
    # 随机打乱边
    random.shuffle(edges)
    
    # Kruskal算法
    for weight, u, v in edges:
        if union(u, v):
            mst.append((u, v, weight))
            if len(mst) == n - 1:
                break
    
    return mst

# 随机化图着色
def randomized_graph_coloring(graph: List[List[int]], colors: int) -> List[int]:
    """随机化图着色 - 期望O(V + E)"""
    n = len(graph)
    coloring = [0] * n
    
    # 随机着色
    for i in range(n):
        # 找到可用的颜色
        available_colors = set(range(colors))
        for neighbor in graph[i]:
            if coloring[neighbor] in available_colors:
                available_colors.remove(coloring[neighbor])
        
        if available_colors:
            coloring[i] = random.choice(list(available_colors))
        else:
            # 如果没有可用颜色，随机选择一个
            coloring[i] = random.randint(0, colors - 1)
    
    return coloring

# 随机化最大割
def randomized_max_cut(graph: List[List[int]]) -> Tuple[List[int], List[int]]:
    """随机化最大割 - 期望1/2近似"""
    n = len(graph)
    partition1 = []
    partition2 = []
    
    # 随机分配顶点
    for i in range(n):
        if random.random() < 0.5:
            partition1.append(i)
        else:
            partition2.append(i)
    
    return partition1, partition2

# 随机化字符串匹配
def randomized_string_matching(text: str, pattern: str) -> List[int]:
    """随机化字符串匹配 - 期望O(n + m)"""
    n = len(text)
    m = len(pattern)
    matches = []
    
    if m > n:
        return matches
    
    # 计算模式哈希
    pattern_hash = 0
    for i in range(m):
        pattern_hash = (pattern_hash * 256 + ord(pattern[i])) % 1000000007
    
    # 计算文本哈希
    text_hash = 0
    power = 1
    for i in range(m):
        text_hash = (text_hash * 256 + ord(text[i])) % 1000000007
        if i < m - 1:
            power = (power * 256) % 1000000007
    
    # 滑动窗口
    for i in range(n - m + 1):
        if text_hash == pattern_hash:
            # 验证匹配
            if text[i:i+m] == pattern:
                matches.append(i)
        
        if i < n - m:
            # 更新哈希
            text_hash = (text_hash - ord(text[i]) * power) % 1000000007
            text_hash = (text_hash * 256 + ord(text[i + m])) % 1000000007
            text_hash = (text_hash + 1000000007) % 1000000007
    
    return matches

from typing import List, Tuple, Any, Callable, Optional
import time
import random
import math
import statistics

class ProbabilisticAnalysis:
    """概率分析"""
    
    def __init__(self):
        self.analyses = {}
        self.statistics = {}
    
    def add_analysis(self, name: str, analysis: Callable) -> None:
        """添加分析"""
        self.analyses[name] = analysis
    
    def analyze_algorithm(self, name: str, input_data: Any, runs: int = 100) -> Dict[str, Any]:
        """分析算法"""
        if name not in self.analyses:
            raise ValueError(f"未知分析: {name}")
        
        analysis = self.analyses[name]
        results = []
        
        for _ in range(runs):
            result = analysis(input_data)
            results.append(result)
        
        # 计算统计信息
        stats = {
            'mean': statistics.mean(results),
            'median': statistics.median(results),
            'std_dev': statistics.stdev(results) if len(results) > 1 else 0,
            'min': min(results),
            'max': max(results),
            'results': results
        }
        
        self.statistics[name] = stats
        return stats
    
    def calculate_expected_value(self, results: List[float]) -> float:
        """计算期望值"""
        return statistics.mean(results)
    
    def calculate_variance(self, results: List[float]) -> float:
        """计算方差"""
        return statistics.variance(results) if len(results) > 1 else 0
    
    def calculate_probability(self, results: List[float], threshold: float) -> float:
        """计算概率"""
        count = sum(1 for r in results if r >= threshold)
        return count / len(results)
    
    def analyze_convergence(self, results: List[float]) -> Dict[str, Any]:
        """分析收敛性"""
        n = len(results)
        if n < 2:
            return {'converged': False, 'rate': 0}
        
        # 计算收敛率
        differences = [abs(results[i] - results[i-1]) for i in range(1, n)]
        convergence_rate = statistics.mean(differences) if differences else 0
        
        # 判断是否收敛
        converged = convergence_rate < 0.01
        
        return {
            'converged': converged,
            'rate': convergence_rate,
            'differences': differences
        }

class RandomizedSkipList:
    """随机化跳跃表"""
    
    def __init__(self):
        self.head = self._create_node(float('-inf'), None)
        self.levels = 1
    
    def _create_node(self, key: Any, value: Any) -> 'SkipListNode':
        return SkipListNode(key, value)
    
    def _random_level(self) -> int:
        level = 1
        while random.random() < 0.5 and level < 16:
            level += 1
        return level
    
    def insert(self, key: Any, value: Any) -> None:
        """插入元素"""
        new_level = self._random_level()
        
        if new_level > self.levels:
            for _ in range(self.levels, new_level):
                new_head = self._create_node(float('-inf'), None)
                new_head.down = self.head
                self.head = new_head
            self.levels = new_level
        
        current = self.head
        update = [None] * self.levels
        
        for i in range(self.levels - 1, -1, -1):
            while current.next[i] and current.next[i].key < key:
                current = current.next[i]
            update[i] = current
        
        if current.next[0] and current.next[0].key == key:
            current.next[0].value = value
            return
        
        new_node = self._create_node(key, value)
        new_node.next = [None] * new_level
        
        for i in range(new_level):
            new_node.next[i] = update[i].next[i]
            update[i].next[i] = new_node
    
    def search(self, key: Any) -> Optional[Any]:
        """搜索元素"""
        current = self.head
        
        for i in range(self.levels - 1, -1, -1):
            while current.next[i] and current.next[i].key < key:
                current = current.next[i]
        
        if current.next[0] and current.next[0].key == key:
            return current.next[0].value
        
        return None
    
    def delete(self, key: Any) -> bool:
        """删除元素"""
        current = self.head
        update = [None] * self.levels
        
        for i in range(self.levels - 1, -1, -1):
            while current.next[i] and current.next[i].key < key:
                current = current.next[i]
            update[i] = current
        
        if not current.next[0] or current.next[0].key != key:
            return False
        
        node_to_delete = current.next[0]
        
        for i in range(len(node_to_delete.next)):
            update[i].next[i] = node_to_delete.next[i]
        
        return True

        self.key = key
        self.value = value
        self.next = []
        self.down = None

class RandomizedHashTable:
    """随机化哈希表"""
    
    def __init__(self, initial_size: int = 16):
        self.size = initial_size
        self.table = [None] * self.size
        self.count = 0
        self.load_factor = 0.75
    
    def _hash(self, key: Any) -> int:
        """哈希函数"""
        return hash(key) % self.size
    
    def _resize(self) -> None:
        """调整大小"""
        old_table = self.table
        old_size = self.size
        
        self.size *= 2
        self.table = [None] * self.size
        self.count = 0
        
        for item in old_table:
            if item:
                self.insert(item[0], item[1])
    
    def insert(self, key: Any, value: Any) -> None:
        """插入元素"""
        if self.count >= self.size * self.load_factor:
            self._resize()
        
        index = self._hash(key)
        
        # 线性探测
        while self.table[index] is not None:
            if self.table[index][0] == key:
                self.table[index] = (key, value)
                return
            index = (index + 1) % self.size
        
        self.table[index] = (key, value)
        self.count += 1
    
    def search(self, key: Any) -> Optional[Any]:
        """搜索元素"""
        index = self._hash(key)
        
        while self.table[index] is not None:
            if self.table[index][0] == key:
                return self.table[index][1]
            index = (index + 1) % self.size
        
        return None
    
    def delete(self, key: Any) -> bool:
        """删除元素"""
        index = self._hash(key)
        
        while self.table[index] is not None:
            if self.table[index][0] == key:
                self.table[index] = None
                self.count -= 1
                return True
            index = (index + 1) % self.size
        
        return False
