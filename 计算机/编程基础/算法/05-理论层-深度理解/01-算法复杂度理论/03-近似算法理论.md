# 近似算法理论

## 概述

近似算法理论（Approximation Algorithm Theory）是计算复杂性理论的重要分支，研究如何在多项式时间内找到优化问题的近似解。对于NP难问题，精确算法通常需要指数时间，而近似算法可以在多项式时间内找到接近最优的解。

## 核心概念

### 基本定义
- **近似算法**：在多项式时间内找到近似解的算法
- **近似比**：近似解与最优解的比值
- **性能保证**：算法性能的理论保证
- **近似方案**：可以任意接近最优解的算法

### 关键特性
- **多项式时间**：算法运行时间为多项式
- **近似性**：解的质量接近最优
- **保证性**：有理论性能保证
- **实用性**：可以解决实际问题

```python
from typing import List, Tuple, Any, Callable, Optional
import time
import random
import math

class ApproximationAlgorithms:
    """近似算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.performance_guarantees = {}
    
    def add_algorithm(self, name: str, algorithm: Callable, guarantee: float) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
        self.performance_guarantees[name] = guarantee
    
    def solve_problem(self, name: str, input_data: Any) -> Tuple[Any, float, float]:
        """解决问题"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        start_time = time.time()
        result = algorithm(input_data)
        end_time = time.time()
        
        return result, end_time - start_time, self.performance_guarantees[name]
    
    def calculate_approximation_ratio(self, approximate_value: float, optimal_value: float) -> float:
        """计算近似比"""
        if optimal_value == 0:
            return float('inf') if approximate_value > 0 else 1.0
        
        return approximate_value / optimal_value

# 顶点覆盖近似算法
def vertex_cover_approximation(graph: List[List[int]]) -> List[int]:
    """顶点覆盖近似算法 - 2-近似"""
    n = len(graph)
    cover = []
    edges = []
    
    # 构建边列表
    for i in range(n):
        for j in graph[i]:
            if i < j:  # 避免重复边
                edges.append((i, j))
    
    # 贪心选择
    while edges:
        # 选择第一条边
        u, v = edges[0]
        cover.append(u)
        cover.append(v)
        
        # 移除所有与u或v相邻的边
        edges = [(x, y) for x, y in edges if x != u and x != v and y != u and y != v]
    
    return cover

# 集合覆盖近似算法
def set_cover_approximation(universe: set, sets: List[set]) -> List[int]:
    """集合覆盖近似算法 - O(log n)近似"""
    uncovered = universe.copy()
    cover = []
    
    while uncovered:
        # 选择覆盖最多未覆盖元素的集合
        best_set_index = max(
            range(len(sets)),
            key=lambda i: len(sets[i] & uncovered)
        )
        
        cover.append(best_set_index)
        uncovered -= sets[best_set_index]
    
    return cover

# 旅行商问题近似算法
def tsp_approximation(distances: List[List[int]]) -> List[int]:
    """旅行商问题近似算法 - 2-近似（基于MST）"""
    n = len(distances)
    if n == 0:
        return []
    
    # 构建最小生成树
    mst = build_mst(distances)
    
    # 对MST进行前序遍历
    path = preorder_traversal(mst, 0)
    
    # 移除重复访问的顶点
    visited = set()
    final_path = []
    for vertex in path:
        if vertex not in visited:
            final_path.append(vertex)
            visited.add(vertex)
    
    return final_path

def build_mst(distances: List[List[int]]) -> List[List[int]]:
    """构建最小生成树"""
    n = len(distances)
    mst = [[] for _ in range(n)]
    visited = {0}
    edges = []
    
    # 添加所有边
    for i in range(n):
        for j in range(i + 1, n):
            edges.append((distances[i][j], i, j))
    
    edges.sort()  # 按权重排序
    
    # Kruskal算法
    parent = list(range(n))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    for weight, u, v in edges:
        if union(u, v):
            mst[u].append(v)
            mst[v].append(u)
            if len(visited) == n:
                break
    
    return mst

def preorder_traversal(mst: List[List[int]], start: int) -> List[int]:
    """前序遍历"""
    visited = set()
    path = []
    
    def dfs(vertex):
        visited.add(vertex)
        path.append(vertex)
        for neighbor in mst[vertex]:
            if neighbor not in visited:
                dfs(neighbor)
    
    dfs(start)
    return path

# 背包问题近似算法
def knapsack_approximation(items: List[Tuple[int, int]], capacity: int) -> List[int]:
    """背包问题近似算法 - 1/2近似"""
    n = len(items)
    
    # 按价值密度排序
    items_with_index = [(i, items[i][0], items[i][1]) for i in range(n)]
    items_with_index.sort(key=lambda x: x[1] / x[2], reverse=True)
    
    # 贪心选择
    solution = []
    current_weight = 0
    
    for index, value, weight in items_with_index:
        if current_weight + weight <= capacity:
            solution.append(index)
            current_weight += weight
    
    return solution

# 最大割近似算法
def max_cut_approximation(graph: List[List[int]]) -> Tuple[List[int], List[int]]:
    """最大割近似算法 - 1/2近似"""
    n = len(graph)
    partition1 = []
    partition2 = []
    
    # 随机分配顶点
    for i in range(n):
        if random.random() < 0.5:
            partition1.append(i)
        else:
            partition2.append(i)
    
    # 局部优化
    improved = True
    while improved:
        improved = False
        
        for i in range(n):
            # 计算移动到另一个分区的收益
            if i in partition1:
                gain = 0
                for j in graph[i]:
                    if j in partition1:
                        gain -= 1
                    else:
                        gain += 1
                
                if gain > 0:
                    partition1.remove(i)
                    partition2.append(i)
                    improved = True
            else:
                gain = 0
                for j in graph[i]:
                    if j in partition2:
                        gain -= 1
                    else:
                        gain += 1
                
                if gain > 0:
                    partition2.remove(i)
                    partition1.append(i)
                    improved = True
    
    return partition1, partition2
```

### 近似方案

```python
from typing import List, Tuple, Any, Callable, Optional
import time
import random
import math

class ApproximationScheme:
    """近似方案"""
    
    def __init__(self):
        self.schemes = {}
        self.performance_guarantees = {}
    
    def add_scheme(self, name: str, scheme: Callable, guarantee: Callable) -> None:
        """添加近似方案"""
        self.schemes[name] = scheme
        self.performance_guarantees[name] = guarantee
    
    def solve_problem(self, name: str, input_data: Any, epsilon: float) -> Tuple[Any, float, float]:
        """解决问题"""
        if name not in self.schemes:
            raise ValueError(f"未知方案: {name}")
        
        scheme = self.schemes[name]
        start_time = time.time()
        result = scheme(input_data, epsilon)
        end_time = time.time()
        
        guarantee = self.performance_guarantees[name](epsilon)
        
        return result, end_time - start_time, guarantee
    
    def calculate_approximation_ratio(self, approximate_value: float, optimal_value: float) -> float:
        """计算近似比"""
        if optimal_value == 0:
            return float('inf') if approximate_value > 0 else 1.0
        
        return approximate_value / optimal_value

# 背包问题的多项式时间近似方案
def knapsack_ptas(items: List[Tuple[int, int]], capacity: int, epsilon: float) -> List[int]:
    """背包问题的多项式时间近似方案"""
    n = len(items)
    
    # 计算参数
    k = int(1 / epsilon)
    
    # 选择前k个最有价值的物品
    items_with_index = [(i, items[i][0], items[i][1]) for i in range(n)]
    items_with_index.sort(key=lambda x: x[1], reverse=True)
    
    best_solution = []
    best_value = 0
    
    # 尝试所有可能的k个物品的子集
    for i in range(min(k, n)):
        for j in range(i + 1, min(k, n)):
            # 选择物品i和j
            selected_items = [items_with_index[i][0], items_with_index[j][0]]
            current_weight = items_with_index[i][2] + items_with_index[j][2]
            current_value = items_with_index[i][1] + items_with_index[j][1]
            
            if current_weight <= capacity and current_value > best_value:
                best_solution = selected_items
                best_value = current_value
    
    # 如果k个物品的子集不满足，尝试单个物品
    if not best_solution:
        for i in range(min(k, n)):
            if items_with_index[i][2] <= capacity and items_with_index[i][1] > best_value:
                best_solution = [items_with_index[i][0]]
                best_value = items_with_index[i][1]
    
    return best_solution

# 旅行商问题的多项式时间近似方案
def tsp_ptas(distances: List[List[int]], epsilon: float) -> List[int]:
    """旅行商问题的多项式时间近似方案"""
    n = len(distances)
    if n == 0:
        return []
    
    # 计算参数
    k = int(1 / epsilon)
    
    # 选择前k个城市
    cities = list(range(min(k, n)))
    
    # 尝试所有可能的路径
    best_path = []
    best_distance = float('inf')
    
    for perm in itertools.permutations(cities):
        path = list(perm)
        total_distance = 0
        
        for i in range(len(path)):
            current_city = path[i]
            next_city = path[(i + 1) % len(path)]
            total_distance += distances[current_city][next_city]
        
        if total_distance < best_distance:
            best_path = path
            best_distance = total_distance
    
    return best_path

# 最大割的多项式时间近似方案
def max_cut_ptas(graph: List[List[int]], epsilon: float) -> Tuple[List[int], List[int]]:
    """最大割的多项式时间近似方案"""
    n = len(graph)
    
    # 计算参数
    k = int(1 / epsilon)
    
    # 选择前k个顶点
    vertices = list(range(min(k, n)))
    
    # 尝试所有可能的分区
    best_partition1 = []
    best_partition2 = []
    best_cut_size = 0
    
    for i in range(2 ** len(vertices)):
        partition1 = []
        partition2 = []
        
        for j in range(len(vertices)):
            if i & (1 << j):
                partition1.append(vertices[j])
            else:
                partition2.append(vertices[j])
        
        # 计算割的大小
        cut_size = 0
        for v1 in partition1:
            for v2 in graph[v1]:
                if v2 in partition2:
                    cut_size += 1
        
        if cut_size > best_cut_size:
            best_partition1 = partition1
            best_partition2 = partition2
            best_cut_size = cut_size
    
    return best_partition1, best_partition2

# 使用示例
if __name__ == "__main__":

    scheme = ApproximationScheme()
    
    # 添加近似方案
    scheme.add_scheme("knapsack_ptas", knapsack_ptas, lambda e: 1 + e)
    scheme.add_scheme("tsp_ptas", tsp_ptas, lambda e: 1 + e)
    scheme.add_scheme("max_cut_ptas", max_cut_ptas, lambda e: 1 + e)
    
    # 测试背包问题的PTAS
    items = [(60, 10), (100, 20), (120, 30), (80, 15), (90, 25)]
    capacity = 50
    epsilon = 0.1

    print(f"背包问题PTAS结果: {result}")
    print(f"背包问题PTAS时间: {time_taken:.6f}秒")
    print(f"性能保证: {guarantee}")
    
    # 测试旅行商问题的PTAS
    distances = [
        [0, 10, 15, 20, 25],
        [10, 0, 35, 25, 30],
        [15, 35, 0, 30, 35],
        [20, 25, 30, 0, 40],
        [25, 30, 35, 40, 0]
    ]
    epsilon = 0.1
    
    result, time_taken, guarantee = scheme.solve_problem("tsp_ptas", distances, epsilon)
    print(f"旅行商问题PTAS结果: {result}")
    print(f"旅行商问题PTAS时间: {time_taken:.6f}秒")
    print(f"性能保证: {guarantee}")
    
    # 测试最大割的PTAS
    graph = [[1, 2], [0, 2, 3], [0, 1, 3], [1, 2]]
    epsilon = 0.1
    
    result, time_taken, guarantee = scheme.solve_problem("max_cut_ptas", graph, epsilon)
    print(f"最大割PTAS结果: {result}")
    print(f"最大割PTAS时间: {time_taken:.6f}秒")
    print(f"性能保证: {guarantee}")
```

## 性能分析

### 近似比分析

| 算法类型 | 近似比 | 时间复杂度 | 空间复杂度 | 适用场景 |
|---------|--------|-----------|-----------|----------|
| 顶点覆盖 | 2 | O(V + E) | O(V) | 图论问题 |
| 集合覆盖 | O(log n) | O(n²) | O(n) | 集合问题 |
| 旅行商 | 2 | O(V²) | O(V) | 路径规划 |
| 背包问题 | 1/2 | O(n log n) | O(n) | 资源分配 |
| 最大割 | 1/2 | O(V + E) | O(V) | 图分割 |

### 近似方案分析

| 方案类型 | 近似比 | 时间复杂度 | 空间复杂度 | 适用场景 |
|---------|--------|-----------|-----------|----------|
| PTAS | 1 + ε | O(n^(1/ε)) | O(n) | 小规模问题 |
| FPTAS | 1 + ε | O(n/ε) | O(n) | 大规模问题 |
| 随机化 | 期望 | O(n log n) | O(n) | 概率问题 |
