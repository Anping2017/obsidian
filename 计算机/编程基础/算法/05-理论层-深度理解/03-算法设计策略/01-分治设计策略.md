# 分治设计策略

## 概述

分治设计策略（Divide and Conquer Design Strategy）是算法设计的重要方法，通过将复杂问题分解为更小的子问题，递归求解子问题，然后合并子问题的解来得到原问题的解。分治策略是许多经典算法的基础。

## 核心概念

### 基本定义
- **分治策略**：将问题分解为更小的子问题，递归求解，然后合并解
- **分解**：将原问题分解为若干个子问题
- **解决**：递归求解子问题
- **合并**：将子问题的解合并为原问题的解

### 关键特性
- **递归性**：使用递归求解子问题
- **分解性**：将问题分解为更小的子问题
- **合并性**：将子问题的解合并
- **最优性**：子问题的最优解构成原问题的最优解

# 归并排序的分治策略
def merge_sort_strategy(data: List[int]) -> Dict[str, Any]:
    """归并排序的分治策略"""
    def merge_sort(arr: List[int], left: int, right: int) -> None:
        if left < right:
            mid = (left + right) // 2
            
            # 分解：将数组分为两部分
            merge_sort(arr, left, mid)
            merge_sort(arr, mid + 1, right)
            
            # 合并：合并两个已排序的子数组
            merge(arr, left, mid, right)
    
    def merge(arr: List[int], left: int, mid: int, right: int) -> None:
        n1 = mid - left + 1
        n2 = right - mid
        
        L = [0] * n1
        R = [0] * n2
        
        for i in range(n1):
            L[i] = arr[left + i]
        for j in range(n2):
            R[j] = arr[mid + 1 + j]
        
        i = j = 0
        k = left
        
        while i < n1 and j < n2:
            if L[i] <= R[j]:
                arr[k] = L[i]
                i += 1
            else:
                arr[k] = R[j]
                j += 1
            k += 1
        
        while i < n1:
            arr[k] = L[i]
            i += 1
            k += 1
        
        while j < n2:
            arr[k] = R[j]
            j += 1
            k += 1
    
    # 执行归并排序
    sorted_data = data.copy()
    merge_sort(sorted_data, 0, len(sorted_data) - 1)
    
    return {
        'algorithm': 'merge_sort',
        'strategy': 'divide_and_conquer',
        'divide': '将数组分为两部分',
        'conquer': '递归排序两个子数组',
        'combine': '合并两个已排序的子数组',
        'time_complexity': 'O(n log n)',
        'space_complexity': 'O(n)',
        'result': sorted_data
    }

# 快速排序的分治策略
def quicksort_strategy(data: List[int]) -> Dict[str, Any]:
    """快速排序的分治策略"""
    def quicksort(arr: List[int], low: int, high: int) -> None:
        if low < high:
            # 分解：选择基准元素并分区
            pivot_index = partition(arr, low, high)
            
            # 解决：递归排序两个子数组
            quicksort(arr, low, pivot_index - 1)
            quicksort(arr, pivot_index + 1, high)
    
    def partition(arr: List[int], low: int, high: int) -> int:
        pivot = arr[high]
        i = low - 1
        
        for j in range(low, high):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        
        arr[i + 1], arr[high] = arr[high], arr[i + 1]
        return i + 1
    
    # 执行快速排序
    sorted_data = data.copy()
    quicksort(sorted_data, 0, len(sorted_data) - 1)
    
    return {
        'algorithm': 'quicksort',
        'strategy': 'divide_and_conquer',
        'divide': '选择基准元素并分区',
        'conquer': '递归排序两个子数组',
        'combine': '分区操作本身就是合并',
        'time_complexity': 'O(n log n) average, O(n²) worst',
        'space_complexity': 'O(log n)',
        'result': sorted_data
    }

# 二分搜索的分治策略
def binary_search_strategy(data: List[int], target: int) -> Dict[str, Any]:
    """二分搜索的分治策略"""
    def binary_search(arr: List[int], target: int, left: int, right: int) -> int:
        if left > right:
            return -1
        
        mid = (left + right) // 2
        
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            return binary_search(arr, target, mid + 1, right)
        else:
            return binary_search(arr, target, left, mid - 1)
    
    # 执行二分搜索
    result = binary_search(data, target, 0, len(data) - 1)
    
    return {
        'algorithm': 'binary_search',
        'strategy': 'divide_and_conquer',
        'divide': '将搜索区间分为两部分',
        'conquer': '递归搜索目标子区间',
        'combine': '返回搜索结果',
        'time_complexity': 'O(log n)',
        'space_complexity': 'O(log n)',
        'result': result
    }

# 最大子数组和的分治策略
def max_subarray_sum_strategy(data: List[int]) -> Dict[str, Any]:
    """最大子数组和的分治策略"""
    def max_subarray_sum(arr: List[int], left: int, right: int) -> int:
        if left == right:
            return arr[left]
        
        mid = (left + right) // 2
        
        # 分解：将数组分为两部分
        left_sum = max_subarray_sum(arr, left, mid)
        right_sum = max_subarray_sum(arr, mid + 1, right)
        
        # 合并：计算跨越中点的最大子数组和
        cross_sum = max_crossing_sum(arr, left, mid, right)
        
        return max(left_sum, right_sum, cross_sum)
    
    def max_crossing_sum(arr: List[int], left: int, mid: int, right: int) -> int:
        left_sum = float('-inf')
        sum_val = 0
        
        for i in range(mid, left - 1, -1):
            sum_val += arr[i]
            left_sum = max(left_sum, sum_val)
        
        right_sum = float('-inf')
        sum_val = 0
        
        for i in range(mid + 1, right + 1):
            sum_val += arr[i]
            right_sum = max(right_sum, sum_val)
        
        return left_sum + right_sum
    
    # 执行最大子数组和
    result = max_subarray_sum(data, 0, len(data) - 1)
    
    return {
        'algorithm': 'max_subarray_sum',
        'strategy': 'divide_and_conquer',
        'divide': '将数组分为两部分',
        'conquer': '递归求解两个子数组的最大子数组和',
        'combine': '计算跨越中点的最大子数组和',
        'time_complexity': 'O(n log n)',
        'space_complexity': 'O(log n)',
        'result': result
    }

# 最近点对的分治策略
def closest_pair_strategy(points: List[Tuple[float, float]]) -> Dict[str, Any]:
    """最近点对的分治策略"""
    def closest_pair(pts: List[Tuple[float, float]]) -> float:
        if len(pts) <= 1:
            return float('inf')
        
        if len(pts) == 2:
            return distance(pts[0], pts[1])
        
        # 分解：按x坐标排序并分为两部分
        pts.sort(key=lambda p: p[0])
        mid = len(pts) // 2
        mid_point = pts[mid]
        
        left_points = pts[:mid]
        right_points = pts[mid:]
        
        # 解决：递归求解两个子问题
        left_min = closest_pair(left_points)
        right_min = closest_pair(right_points)
        
        # 合并：计算跨越分割线的最近点对
        min_dist = min(left_min, right_min)
        strip_min = closest_in_strip(pts, mid_point, min_dist)
        
        return min(min_dist, strip_min)
    
    def distance(p1: Tuple[float, float], p2: Tuple[float, float]) -> float:
        return ((p1[0] - p2[0]) ** 2 + (p1[1] - p2[1]) ** 2) ** 0.5
    
    def closest_in_strip(pts: List[Tuple[float, float]], mid_point: Tuple[float, float], min_dist: float) -> float:
        strip = [p for p in pts if abs(p[0] - mid_point[0]) < min_dist]
        strip.sort(key=lambda p: p[1])
        
        min_strip_dist = min_dist
        for i in range(len(strip)):
            for j in range(i + 1, len(strip)):
                if strip[j][1] - strip[i][1] >= min_dist:
                    break
                min_strip_dist = min(min_strip_dist, distance(strip[i], strip[j]))
        
        return min_strip_dist
    
    # 执行最近点对
    result = closest_pair(points)
    
    return {
        'algorithm': 'closest_pair',
        'strategy': 'divide_and_conquer',
        'divide': '按x坐标排序并分为两部分',
        'conquer': '递归求解两个子问题的最近点对',
        'combine': '计算跨越分割线的最近点对',
        'time_complexity': 'O(n log n)',
        'space_complexity': 'O(n)',
        'result': result
    }

def matrix_multiplication_strategy(matrix_a: List[List[int]], matrix_b: List[List[int]]) -> Dict[str, Any]:
    """矩阵乘法的分治策略"""
    def matrix_multiply(a: List[List[int]], b: List[List[int]]) -> List[List[int]]:
        n = len(a)
        if n == 1:
            return [[a[0][0] * b[0][0]]]
        
        # 分解：将矩阵分为四个子矩阵
        mid = n // 2
        
        a11 = [row[:mid] for row in a[:mid]]
        a12 = [row[mid:] for row in a[:mid]]
        a21 = [row[:mid] for row in a[mid:]]
        a22 = [row[mid:] for row in a[mid:]]
        
        b11 = [row[:mid] for row in b[:mid]]
        b12 = [row[mid:] for row in b[:mid]]
        b21 = [row[:mid] for row in b[mid:]]
        b22 = [row[mid:] for row in b[mid:]]
        
        # 解决：递归计算子矩阵乘法
        c11 = matrix_add(matrix_multiply(a11, b11), matrix_multiply(a12, b21))
        c12 = matrix_add(matrix_multiply(a11, b12), matrix_multiply(a12, b22))
        c21 = matrix_add(matrix_multiply(a21, b11), matrix_multiply(a22, b21))
        c22 = matrix_add(matrix_multiply(a21, b12), matrix_multiply(a22, b22))
        
        # 合并：组合子矩阵结果
        result = []
        for i in range(n):
            if i < mid:
                result.append(c11[i] + c12[i])
            else:
                result.append(c21[i - mid] + c22[i - mid])
        
        return result
    
    def matrix_add(a: List[List[int]], b: List[List[int]]) -> List[List[int]]:
        n = len(a)
        result = [[0] * n for _ in range(n)]
        for i in range(n):
            for j in range(n):
                result[i][j] = a[i][j] + b[i][j]
        return result
    
    # 执行矩阵乘法
    result = matrix_multiply(matrix_a, matrix_b)
    
    return {
        'algorithm': 'matrix_multiplication',
        'strategy': 'divide_and_conquer',
        'divide': '将矩阵分为四个子矩阵',
        'conquer': '递归计算子矩阵乘法',
        'combine': '组合子矩阵结果',
        'time_complexity': 'O(n³)',
        'space_complexity': 'O(n²)',
        'result': result
    }

def fast_power_strategy(base: int, exponent: int) -> Dict[str, Any]:
    """快速幂的分治策略"""
    def fast_power(base: int, exp: int) -> int:
        if exp == 0:
            return 1
        if exp == 1:
            return base
        
        # 分解：将指数分为两部分
        half = fast_power(base, exp // 2)
        
        # 解决：递归计算幂
        if exp % 2 == 0:
            return half * half
        else:
            return half * half * base
    
    # 执行快速幂
    result = fast_power(base, exponent)
    
    return {
        'algorithm': 'fast_power',
        'strategy': 'divide_and_conquer',
        'divide': '将指数分为两部分',
        'conquer': '递归计算幂',
        'combine': '合并幂的结果',
        'time_complexity': 'O(log n)',
        'space_complexity': 'O(log n)',
        'result': result
    }

# 快速傅里叶变换的分治策略
def fft_strategy(signal: List[complex]) -> Dict[str, Any]:
    """快速傅里叶变换的分治策略"""
    def fft(x: List[complex]) -> List[complex]:
        n = len(x)
        if n <= 1:
            return x
        
        # 分解：将信号分为偶数和奇数部分
        even = fft(x[0::2])
        odd = fft(x[1::2])
        
        # 解决：递归计算FFT
        factor = [complex(0, -2 * 3.14159 * k / n) for k in range(n // 2)]
        factor = [complex.exp(f) for f in factor]
        
        # 合并：组合FFT结果
        result = [0] * n
        for k in range(n // 2):
            result[k] = even[k] + factor[k] * odd[k]
            result[k + n // 2] = even[k] - factor[k] * odd[k]
        
        return result
    
    # 执行FFT
    result = fft(signal)
    
    return {
        'algorithm': 'fft',
        'strategy': 'divide_and_conquer',
        'divide': '将信号分为偶数和奇数部分',
        'conquer': '递归计算FFT',
        'combine': '组合FFT结果',
        'time_complexity': 'O(n log n)',
        'space_complexity': 'O(n)',
        'result': result
    }

## 应用场景

### 1. 排序算法
- **归并排序**：稳定排序，适合大数据量
- **快速排序**：平均性能好，适合一般情况
- **堆排序**：原地排序，适合内存受限

### 2. 搜索算法
- **二分搜索**：有序数组搜索
- **三分搜索**：单峰函数搜索
- **分块搜索**：大范围搜索

### 3. 优化问题
- **最大子数组和**：连续子数组优化
- **最近点对**：几何优化问题
- **最小生成树**：图优化问题

## 优缺点分析

### 优点
- **理论清晰**：分治思想清晰易懂
- **性能稳定**：大多数情况下性能稳定
- **可并行化**：子问题可以并行求解
- **适用广泛**：适用于多种问题类型

### 缺点
- **空间复杂度**：递归调用消耗栈空间
- **实现复杂**：某些算法实现复杂
- **常数因子**：常数因子可能较大
- **不适合小问题**：小问题可能不如简单算法
