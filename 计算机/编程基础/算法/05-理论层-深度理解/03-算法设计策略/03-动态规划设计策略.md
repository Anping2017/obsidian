# 动态规划设计策略

## 概述

动态规划设计策略（Dynamic Programming Design Strategy）是算法设计的重要方法，通过将复杂问题分解为重叠的子问题，存储子问题的解来避免重复计算，从而高效求解原问题。动态规划是解决最优化问题的强大工具。

## 核心概念

### 基本定义
- **动态规划**：通过存储子问题的解来避免重复计算
- **最优子结构**：问题的最优解包含子问题的最优解
- **重叠子问题**：递归过程中会重复求解相同的子问题
- **状态转移**：从子问题的解推导出原问题的解

### 关键特性
- **记忆化**：存储已计算的子问题解
- **自底向上**：从最小子问题开始求解
- **状态定义**：明确定义问题的状态
- **转移方程**：描述状态之间的转移关系

from typing import List, Tuple, Any, Callable, Optional
import time

class DynamicProgrammingDesignStrategy:
    """动态规划设计策略示例"""
    
    def __init__(self):
        self.strategies = {}
        self.algorithms = {}
    
    def add_strategy(self, name: str, strategy: Callable) -> None:
        """添加策略"""
        self.strategies[name] = strategy
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def apply_strategy(self, name: str, input_data: Any) -> Dict[str, Any]:
        """应用策略"""
        if name not in self.strategies:
            raise ValueError(f"未知策略: {name}")
        
        strategy = self.strategies[name]
        return strategy(input_data)
    
    def execute_algorithm(self, name: str, input_data: Any) -> Any:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(input_data)

# 斐波那契数列的动态规划策略
def fibonacci_strategy(n: int) -> Dict[str, Any]:
    """斐波那契数列的动态规划策略"""
    def fibonacci_dp(num: int) -> int:
        if num <= 1:
            return num
        
        # 状态定义：dp[i] 表示第i个斐波那契数
        dp = [0] * (num + 1)
        dp[0] = 0
        dp[1] = 1
        
        # 状态转移：dp[i] = dp[i-1] + dp[i-2]
        for i in range(2, num + 1):
            dp[i] = dp[i - 1] + dp[i - 2]
        
        return dp[num]
    
    # 执行斐波那契数列
    result = fibonacci_dp(n)
    
    return {
        'algorithm': 'fibonacci_dp',
        'strategy': 'dynamic_programming',
        'state_definition': 'dp[i] 表示第i个斐波那契数',
        'state_transition': 'dp[i] = dp[i-1] + dp[i-2]',
        'optimal_substructure': '第i个斐波那契数依赖于前两个数',
        'overlapping_subproblems': '递归过程中会重复计算相同的子问题',
        'time_complexity': 'O(n)',
        'space_complexity': 'O(n)',
        'result': result
    }

# 最长公共子序列的动态规划策略
def lcs_strategy(text1: str, text2: str) -> Dict[str, Any]:
    """最长公共子序列的动态规划策略"""
    def longest_common_subsequence(t1: str, t2: str) -> int:
        m, n = len(t1), len(t2)
        
        # 状态定义：dp[i][j] 表示text1[0:i]和text2[0:j]的最长公共子序列长度
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        
        # 状态转移
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if t1[i - 1] == t2[j - 1]:
                    dp[i][j] = dp[i - 1][j - 1] + 1
                else:
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
        
        return dp[m][n]
    
    # 执行最长公共子序列
    result = longest_common_subsequence(text1, text2)
    
    return {
        'algorithm': 'lcs_dp',
        'strategy': 'dynamic_programming',
        'state_definition': 'dp[i][j] 表示text1[0:i]和text2[0:j]的最长公共子序列长度',
        'state_transition': 'dp[i][j] = dp[i-1][j-1] + 1 if t1[i-1] == t2[j-1] else max(dp[i-1][j], dp[i][j-1])',
        'optimal_substructure': '子序列的最优解包含在全局最优解中',
        'overlapping_subproblems': '递归过程中会重复计算相同的子问题',
        'time_complexity': 'O(mn)',
        'space_complexity': 'O(mn)',
        'result': result
    }

# 0-1背包问题的动态规划策略
def knapsack_strategy(weights: List[int], values: List[int], capacity: int) -> Dict[str, Any]:
    """0-1背包问题的动态规划策略"""
    def knapsack_01(w: List[int], v: List[int], cap: int) -> int:
        n = len(w)
        
        # 状态定义：dp[i][j] 表示前i个物品在容量j下的最大价值
        dp = [[0] * (cap + 1) for _ in range(n + 1)]
        
        # 状态转移
        for i in range(1, n + 1):
            for j in range(cap + 1):
                if w[i - 1] <= j:
                    dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - w[i - 1]] + v[i - 1])
                else:
                    dp[i][j] = dp[i - 1][j]
        
        return dp[n][cap]
    
    # 执行0-1背包
    result = knapsack_01(weights, values, capacity)
    
    return {
        'algorithm': 'knapsack_01_dp',
        'strategy': 'dynamic_programming',
        'state_definition': 'dp[i][j] 表示前i个物品在容量j下的最大价值',
        'state_transition': 'dp[i][j] = max(dp[i-1][j], dp[i-1][j-w[i-1]] + v[i-1])',
        'optimal_substructure': '子问题的最优解包含在全局最优解中',
        'overlapping_subproblems': '递归过程中会重复计算相同的子问题',
        'time_complexity': 'O(nW)',
        'space_complexity': 'O(nW)',
        'result': result
    }

# 最长递增子序列的动态规划策略
def lis_strategy(nums: List[int]) -> Dict[str, Any]:
    """最长递增子序列的动态规划策略"""
    def longest_increasing_subsequence(num_list: List[int]) -> int:
        n = len(num_list)
        
        # 状态定义：dp[i] 表示以第i个元素结尾的最长递增子序列长度
        dp = [1] * n
        
        # 状态转移
        for i in range(1, n):
            for j in range(i):
                if num_list[j] < num_list[i]:
                    dp[i] = max(dp[i], dp[j] + 1)
        
        return max(dp)
    
    # 执行最长递增子序列
    result = longest_increasing_subsequence(nums)
    
    return {
        'algorithm': 'lis_dp',
        'strategy': 'dynamic_programming',
        'state_definition': 'dp[i] 表示以第i个元素结尾的最长递增子序列长度',
        'state_transition': 'dp[i] = max(dp[i], dp[j] + 1) if nums[j] < nums[i]',
        'optimal_substructure': '子序列的最优解包含在全局最优解中',
        'overlapping_subproblems': '递归过程中会重复计算相同的子问题',
        'time_complexity': 'O(n²)',
        'space_complexity': 'O(n)',
        'result': result
    }

# 编辑距离的动态规划策略
def edit_distance_strategy(word1: str, word2: str) -> Dict[str, Any]:
    """编辑距离的动态规划策略"""
    def edit_distance(w1: str, w2: str) -> int:
        m, n = len(w1), len(w2)
        
        # 状态定义：dp[i][j] 表示word1[0:i]转换为word2[0:j]的最小编辑距离
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        
        # 初始化
        for i in range(m + 1):
            dp[i][0] = i
        for j in range(n + 1):
            dp[0][j] = j
        
        # 状态转移
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if w1[i - 1] == w2[j - 1]:
                    dp[i][j] = dp[i - 1][j - 1]
                else:
                    dp[i][j] = min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1]) + 1
        
        return dp[m][n]
    
    # 执行编辑距离
    result = edit_distance(word1, word2)
    
    return {
        'algorithm': 'edit_distance_dp',
        'strategy': 'dynamic_programming',
        'state_definition': 'dp[i][j] 表示word1[0:i]转换为word2[0:j]的最小编辑距离',
        'state_transition': 'dp[i][j] = dp[i-1][j-1] if w1[i-1] == w2[j-1] else min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1]) + 1',
        'optimal_substructure': '子问题的最优解包含在全局最优解中',
        'overlapping_subproblems': '递归过程中会重复计算相同的子问题',
        'time_complexity': 'O(mn)',
        'space_complexity': 'O(mn)',
        'result': result
    }

def matrix_chain_multiplication_strategy(dimensions: List[int]) -> Dict[str, Any]:
    """矩阵链乘法的动态规划策略"""
    def matrix_chain_multiplication(dims: List[int]) -> int:
        n = len(dims) - 1
        
        # 状态定义：dp[i][j] 表示矩阵链A[i:j+1]的最小乘法次数
        dp = [[0] * n for _ in range(n)]
        
        # 状态转移
        for length in range(2, n + 1):
            for i in range(n - length + 1):
                j = i + length - 1
                dp[i][j] = float('inf')
                
                for k in range(i, j):
                    cost = dp[i][k] + dp[k + 1][j] + dims[i] * dims[k + 1] * dims[j + 1]
                    dp[i][j] = min(dp[i][j], cost)
        
        return dp[0][n - 1]
    
    # 执行矩阵链乘法
    result = matrix_chain_multiplication(dimensions)
    
    return {
        'algorithm': 'matrix_chain_multiplication_dp',
        'strategy': 'dynamic_programming',
        'state_definition': 'dp[i][j] 表示矩阵链A[i:j+1]的最小乘法次数',
        'state_transition': 'dp[i][j] = min(dp[i][k] + dp[k+1][j] + dims[i] * dims[k+1] * dims[j+1])',
        'optimal_substructure': '子问题的最优解包含在全局最优解中',
        'overlapping_subproblems': '递归过程中会重复计算相同的子问题',
        'time_complexity': 'O(n³)',
        'space_complexity': 'O(n²)',
        'result': result

        for length in range(2, n + 1):

                    dp[i][j] = dp[i + 1][j - 1] + 2
                else:
                    dp[i][j] = max(dp[i + 1][j], dp[i][j - 1])
        
        return dp[0][n - 1]
    
    # 执行最长回文子序列
    result = longest_palindromic_subsequence(s)
    
    return {
        'algorithm': 'longest_palindromic_subsequence_dp',
        'strategy': 'dynamic_programming',

# 股票买卖的动态规划策略
def stock_trading_strategy(prices: List[int]) -> Dict[str, Any]:
    """股票买卖的动态规划策略"""
    def max_profit(price_list: List[int]) -> int:
        n = len(price_list)
        
        # 状态定义：dp[i][0] 表示第i天持有股票的最大利润，dp[i][1] 表示第i天不持有股票的最大利润
        dp = [[0] * 2 for _ in range(n)]
        
        # 初始化
        dp[0][0] = -price_list[0]  # 第0天买入股票
        dp[0][1] = 0  # 第0天不持有股票
        
        # 状态转移
        for i in range(1, n):
            dp[i][0] = max(dp[i - 1][0], dp[i - 1][1] - price_list[i])  # 持有股票
            dp[i][1] = max(dp[i - 1][1], dp[i - 1][0] + price_list[i])  # 不持有股票
        
        return dp[n - 1][1]
    
    # 执行股票买卖
    result = max_profit(prices)
    
    return {
        'algorithm': 'stock_trading_dp',
        'strategy': 'dynamic_programming',
        'state_definition': 'dp[i][0] 表示第i天持有股票的最大利润，dp[i][1] 表示第i天不持有股票的最大利润',
        'state_transition': 'dp[i][0] = max(dp[i-1][0], dp[i-1][1] - prices[i]), dp[i][1] = max(dp[i-1][1], dp[i-1][0] + prices[i])',
        'optimal_substructure': '子问题的最优解包含在全局最优解中',
        'overlapping_subproblems': '递归过程中会重复计算相同的子问题',
        'time_complexity': 'O(n)',
        'space_complexity': 'O(n)',
        'result': result
    }

## 优缺点分析

### 优点
- **全局最优**：保证找到全局最优解
- **避免重复**：通过记忆化避免重复计算
- **适用广泛**：适用于多种优化问题
- **理论基础**：有坚实的数学理论基础

### 缺点
- **空间复杂度**：需要存储大量中间结果
- **状态设计**：状态设计可能复杂
