# 循环不变式

## 概述

循环不变式（Loop Invariant）是算法正确性证明的重要工具，用于证明循环算法的正确性。循环不变式是在循环的每次迭代前后都保持为真的性质，它帮助我们理解算法的行为并证明算法的正确性。

## 核心概念

### 基本定义
- **循环不变式**：在循环的每次迭代前后都保持为真的性质
- **初始化**：循环开始前不变式为真
- **保持**：每次迭代后不变式仍为真
- **终止**：循环终止时不变式帮助我们证明正确性

### 关键特性
- **不变性**：在循环过程中保持不变
- **有用性**：帮助我们证明算法正确性
- **完整性**：覆盖算法的所有状态
- **简洁性**：表达简洁明了

        """添加证明"""

        """添加不变式"""

        if name not in self.proofs:

        proof = self.proofs[name]
        return proof(input_data)
    
    def verify_invariant(self, name: str, input_data: Any) -> bool:
        """验证不变式"""
        if name not in self.invariants:
            raise ValueError(f"未知不变式: {name}")
        
        invariant = self.invariants[name]
        return invariant(input_data)

# 线性搜索的循环不变式
def linear_search_invariant(data: List[int], target: int) -> Dict[str, Any]:
    """线性搜索的循环不变式证明"""
    n = len(data)
    result = -1
    
    # 循环不变式：result == -1 当且仅当 target 不在 data[0:i] 中
    # 其中 i 是当前检查的索引
    
    for i in range(n):
        # 不变式：result == -1 当且仅当 target 不在 data[0:i] 中
        if data[i] == target:
            result = i
            break
    
    return {
        'algorithm': 'linear_search',
        'invariant': 'result == -1 当且仅当 target 不在 data[0:i] 中',
        'initialization': '循环开始前，i=0，data[0:0]为空，result=-1，不变式为真',
        'maintenance': '每次迭代后，如果找到target则result=i，否则result仍为-1，不变式保持',
        'termination': '循环终止时，如果result=-1则target不在整个数组中，否则result为target的索引',
        'correctness': '算法正确找到target的索引或返回-1'
    }

# 插入排序的循环不变式
def insertion_sort_invariant(data: List[int]) -> Dict[str, Any]:
    """插入排序的循环不变式证明"""
    n = len(data)
    
    # 循环不变式：data[0:i] 已排序
    for i in range(1, n):
        key = data[i]
        j = i - 1
        
        # 内层循环不变式：data[0:j+1] 已排序，且 data[j+1:i+1] 中的元素都大于等于key
        while j >= 0 and data[j] > key:
            data[j + 1] = data[j]
            j -= 1
        
        data[j + 1] = key
    
    return {
        'algorithm': 'insertion_sort',
        'invariant': 'data[0:i] 已排序',
        'initialization': '循环开始前，i=1，data[0:1]只有一个元素，已排序，不变式为真',
        'maintenance': '每次迭代后，将data[i]插入到已排序的data[0:i]中的正确位置，data[0:i+1]仍为已排序',
        'termination': '循环终止时，i=n，data[0:n]已排序，即整个数组已排序',
        'correctness': '算法正确排序整个数组'
    }

# 选择排序的循环不变式
def selection_sort_invariant(data: List[int]) -> Dict[str, Any]:
    """选择排序的循环不变式证明"""
    n = len(data)
    
    # 循环不变式：data[0:i] 已排序，且 data[0:i] 中的元素都小于等于 data[i:n] 中的元素
    for i in range(n - 1):
        min_index = i
        
        # 内层循环不变式：data[min_index] 是 data[i:j] 中的最小元素
        for j in range(i + 1, n):
            if data[j] < data[min_index]:
                min_index = j
        
        # 交换
        data[i], data[min_index] = data[min_index], data[i]
    
    return {
        'algorithm': 'selection_sort',
        'invariant': 'data[0:i] 已排序，且 data[0:i] 中的元素都小于等于 data[i:n] 中的元素',
        'initialization': '循环开始前，i=0，data[0:0]为空，不变式为真',
        'maintenance': '每次迭代后，将data[i:n]中的最小元素与data[i]交换，data[0:i+1]仍为已排序',
        'termination': '循环终止时，i=n-1，data[0:n-1]已排序，且data[n-1]是最大元素，整个数组已排序',
        'correctness': '算法正确排序整个数组'
    }

# 冒泡排序的循环不变式
def bubble_sort_invariant(data: List[int]) -> Dict[str,Any]:
    """冒泡排序的循环不变式证明"""
    n = len(data)
    
    # 循环不变式：data[n-i:n] 已排序，且 data[n-i:n] 中的元素都大于等于 data[0:n-i] 中的元素
    for i in range(n):
        swapped = False
        
        # 内层循环不变式：data[j] 是 data[0:j+1] 中的最大元素
        for j in range(n - i - 1):
            if data[j] > data[j + 1]:
                data[j], data[j + 1] = data[j + 1], data[j]
                swapped = True
        
        if not swapped:
            break
    
    return {
        'algorithm': 'bubble_sort',
        'invariant': 'data[n-i:n] 已排序，且 data[n-i:n] 中的元素都大于等于 data[0:n-i] 中的元素',
        'initialization': '循环开始前，i=0，data[n:n]为空，不变式为真',
        'maintenance': '每次迭代后，将data[0:n-i]中的最大元素冒泡到data[n-i-1]，data[n-i:n]仍为已排序',
        'termination': '循环终止时，i=n，data[0:n]已排序，即整个数组已排序',
        'correctness': '算法正确排序整个数组'
    }

# 二分搜索的循环不变式
def binary_search_invariant(data: List[int], target: int) -> Dict[str, Any]:
    """二分搜索的循环不变式证明"""
    left = 0
    right = len(data) - 1
    result = -1
    
    # 循环不变式：如果target在data中，则target在data[left:right+1]中
    while left <= right:
        mid = (left + right) // 2
        
        if data[mid] == target:
            result = mid
            break
        elif data[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    
    return {
        'algorithm': 'binary_search',
        'invariant': '如果target在data中，则target在data[left:right+1]中',
        'initialization': '循环开始前，left=0，right=n-1，data[left:right+1]是整个数组，不变式为真',
        'maintenance': '每次迭代后，根据比较结果调整left或right，保持target在data[left:right+1]中',
        'termination': '循环终止时，left>right，data[left:right+1]为空，target不在数组中',
        'correctness': '算法正确找到target的索引或返回-1'
    }

# 快速排序的循环不变式
def quicksort_invariant(data: List[int]) -> Dict[str, Any]:
    """快速排序的循环不变式证明"""
    def quicksort_helper(arr: List[int], low: int, high: int) -> None:
        if low < high:
            # 分区
            pivot_index = partition(arr, low, high)
            
            # 递归排序
            quicksort_helper(arr, low, pivot_index - 1)
            quicksort_helper(arr, pivot_index + 1, high)
    
    def partition(arr: List[int], low: int, high: int) -> int:
        pivot = arr[high]
        i = low - 1
        
        # 循环不变式：arr[low:i+1] 中的元素都小于等于pivot，arr[i+1:j] 中的元素都大于pivot
        for j in range(low, high):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        
        arr[i + 1], arr[high] = arr[high], arr[i + 1]
        return i + 1
    
    quicksort_helper(data, 0, len(data) - 1)
    
    return {
        'algorithm': 'quicksort',
        'invariant': 'arr[low:i+1] 中的元素都小于等于pivot，arr[i+1:j] 中的元素都大于pivot',
        'initialization': '循环开始前，i=low-1，j=low，arr[low:i+1]为空，arr[i+1:j]为空，不变式为真',
        'maintenance': '每次迭代后，如果arr[j]<=pivot则交换arr[i+1]和arr[j]，保持不变式',
        'termination': '循环终止时，j=high，arr[low:i+1]中的元素都小于等于pivot，arr[i+1:high]中的元素都大于pivot',
        'correctness': '算法正确排序整个数组'
    }

# 归并排序的循环不变式
def merge_sort_invariant(data: List[int]) -> Dict[str, Any]:
    """归并排序的循环不变式证明"""
    def merge_sort_helper(arr: List[int], left: int, right: int) -> None:
        if left < right:
            mid = (left + right) // 2
            
            # 递归排序
            merge_sort_helper(arr, left, mid)
            merge_sort_helper(arr, mid + 1, right)
            
            # 合并
            merge(arr, left, mid, right)
    
    def merge(arr: List[int], left: int, mid: int, right: int) -> None:
        n1 = mid - left + 1
        n2 = right - mid
        
        L = [0] * n1
        R = [0] * n2
        
        for i in range(n1):
            L[i] = arr[left + i]
        for j in range(n2):
            R[j] = arr[mid + 1 + j]
        
        i = j = 0
        k = left
        
        # 循环不变式：arr[left:k] 已排序，且包含L[0:i]和R[0:j]中的最小元素
        while i < n1 and j < n2:
            if L[i] <= R[j]:
                arr[k] = L[i]
                i += 1
            else:
                arr[k] = R[j]
                j += 1
            k += 1
        
        # 复制剩余元素
        while i < n1:
            arr[k] = L[i]
            i += 1
            k += 1
        
        while j < n2:
            arr[k] = R[j]
            j += 1
            k += 1
    
    merge_sort_helper(data, 0, len(data) - 1)
    
    return {
        'algorithm': 'merge_sort',
        'invariant': 'arr[left:k] 已排序，且包含L[0:i]和R[0:j]中的最小元素',
        'initialization': '循环开始前，i=j=0，k=left，arr[left:k]为空，不变式为真',
        'maintenance': '每次迭代后，将L[i]或R[j]中的较小元素复制到arr[k]，保持不变式',
        'termination': '循环终止时，i=n1或j=n2，arr[left:k]包含所有已排序元素',
        'correctness': '算法正确排序整个数组'
    }

# 堆排序的循环不变式
def heap_sort_invariant(data: List[int]) -> Dict[str, Any]:
    """堆排序的循环不变式证明"""
    def heapify(arr: List[int], n: int, i: int) -> None:
        largest = i
        left = 2 * i + 1
        right = 2 * i + 2
        
        if left < n and arr[left] > arr[largest]:
            largest = left
        
        if right < n and arr[right] > arr[largest]:
            largest = right
        
        if largest != i:
            arr[i], arr[largest] = arr[largest], arr[i]
            heapify(arr, n, largest)
    
    n = len(data)
    
    # 构建最大堆
    for i in range(n // 2 - 1, -1, -1):
        heapify(data, n, i)
    
    # 提取元素
    for i in range(n - 1, 0, -1):
        data[0], data[i] = data[i], data[0]
        heapify(data, i, 0)
    
    return {
        'algorithm': 'heap_sort',
        'invariant': 'data[i:n] 已排序，且 data[0:i] 构成最大堆',
        'initialization': '循环开始前，i=n，data[n:n]为空，不变式为真',
        'maintenance': '每次迭代后，将堆顶元素与data[i-1]交换，然后重新堆化data[0:i-1]',
        'termination': '循环终止时，i=1，data[1:n]已排序，data[0]是最大元素，整个数组已排序',
        'correctness': '算法正确排序整个数组'
    }

# 计数排序的循环不变式
def counting_sort_invariant(data: List[int]) -> Dict[str, Any]:
    """计数排序的循环不变式证明"""
    n = len(data)
    max_val = max(data)
    count = [0] * (max_val + 1)
    output = [0] * n
    
    # 计数
    for i in range(n):
        count[data[i]] += 1
    
    # 累积计数
    for i in range(1, max_val + 1):
        count[i] += count[i - 1]
    
    # 排序
    for i in range(n - 1, -1, -1):
        output[count[data[i]] - 1] = data[i]
        count[data[i]] -= 1
    
    return {
        'algorithm': 'counting_sort',
        'invariant': 'output[0:count[data[i]]] 已排序，且包含data[0:i+1]中的元素',
        'initialization': '循环开始前，i=n-1，output[0:count[data[n-1]]]为空，不变式为真',
        'maintenance': '每次迭代后，将data[i]放置到output[count[data[i]]-1]，保持不变式',
        'termination': '循环终止时，i=-1，output[0:n]包含所有已排序元素',
        'correctness': '算法正确排序整个数组'
    }

def radix_sort_invariant(data: List[int]) -> Dict[str, Any]:
    """基数排序的循环不变式证明"""
    def counting_sort_by_digit(arr: List[int], exp: int) -> None:
        n = len(arr)
        output = [0] * n
        count = [0] * 10
        
        # 计数
        for i in range(n):
            count[(arr[i] // exp) % 10] += 1
        
        # 累积计数
        for i in range(1, 10):
            count[i] += count[i - 1]
        
        # 排序
        for i in range(n - 1, -1, -1):
            output[count[(arr[i] // exp) % 10] - 1] = arr[i]
            count[(arr[i] // exp) % 10] -= 1
        
        for i in range(n):
            arr[i] = output[i]
    
    max_val = max(data)
    exp = 1
    
    # 循环不变式：data 已按前k位数字排序
    while max_val // exp > 0:
        counting_sort_by_digit(data, exp)
        exp *= 10
    
    return {
        'algorithm': 'radix_sort',
        'invariant': 'data 已按前k位数字排序',
        'initialization': '循环开始前，k=0，data未排序，不变式为真',
        'maintenance': '每次迭代后，按第k位数字排序，保持前k-1位的排序',
        'termination': '循环终止时，k为最大数字的位数，data完全排序',
        'correctness': '算法正确排序整个数组'
