# 模运算

## 概述

模运算（Modular Arithmetic）是数论中的基础运算，用于计算整数除以另一个整数后的余数。模运算在密码学、计算机科学、数论研究等领域有重要应用。

## 核心概念

### 基本定义
- **模运算**：计算a mod m的运算
- **同余**：两个数除以同一个数的余数相同
- **模逆元**：在模运算下的乘法逆元
- **中国剩余定理**：解决同余方程组

### 关键特性
- **周期性**：模运算具有周期性
- **结合性**：(a + b) mod m = ((a mod m) + (b mod m)) mod m
- **分配性**：(a * b) mod m = ((a mod m) * (b mod m)) mod m
- **可逆性**：某些数在模运算下有逆元

from typing import List, Tuple, Any, Callable, Optional
import time
import math

class ModularArithmeticAlgorithms:
    """模运算算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.values = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, *args) -> Any:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(*args)
    
    def set_values(self, values: List[int]) -> None:
        """设置值"""
        self.values = values

# 基本模运算
def basic_mod(a: int, m: int) -> int:
    """基本模运算"""
    return a % m

# 模加法
def modular_add(a: int, b: int, m: int) -> int:
    """模加法"""
    return (a + b) % m

# 模减法
def modular_subtract(a: int, b: int, m: int) -> int:
    """模减法"""
    return (a - b) % m

# 模乘法
def modular_multiply(a: int, b: int, m: int) -> int:
    """模乘法"""
    return (a * b) % m

# 模幂运算
def modular_power(base: int, exponent: int, m: int) -> int:
    """模幂运算"""
    result = 1
    base = base % m
    
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % m
        
        base = (base * base) % m
        exponent //= 2
    
    return result

# 模逆元
def modular_inverse(a: int, m: int) -> int:
    """模逆元"""
    def extended_gcd(a: int, b: int) -> Tuple[int, int, int]:
        if b == 0:
            return a, 1, 0
        
        gcd, x1, y1 = extended_gcd(b, a % b)
        x = y1
        y = x1 - (a // b) * y1
        
        return gcd, x, y
    
    gcd, x, y = extended_gcd(a, m)
    if gcd != 1:
        raise ValueError("模逆元不存在")
    
    return (x % m + m) % m

# 模除法
def modular_divide(a: int, b: int, m: int) -> int:
    """模除法"""
    b_inv = modular_inverse(b, m)
    return (a * b_inv) % m

# 使用示例

# 中国剩余定理
def chinese_remainder_theorem(remainders: List[int], moduli: List[int]) -> int:
    """中国剩余定理"""
    if len(remainders) != len(moduli):
        raise ValueError("余数和模数数量不匹配")
    
    # 计算所有模数的乘积
    M = 1
    for m in moduli:
        M *= m
    
    result = 0
    for i in range(len(remainders)):
        Mi = M // moduli[i]
        Mi_inv = modular_inverse(Mi, moduli[i])
        result += remainders[i] * Mi * Mi_inv
    
    return result % M

# 费马小定理
def fermat_little_theorem(a: int, p: int) -> bool:
    """费马小定理验证"""
    if p <= 1:
        return False
    
    # 检查p是否为素数
    for i in range(2, int(math.sqrt(p)) + 1):
        if p % i == 0:
            return False
    
    # 费马小定理：a^(p-1) ≡ 1 (mod p)
    return modular_power(a, p - 1, p) == 1

# 欧拉定理
def euler_theorem(a: int, n: int) -> bool:
    """欧拉定理验证"""
    if math.gcd(a, n) != 1:
        return False
    
    # 计算欧拉函数
    phi_n = euler_phi(n)
    
    # 欧拉定理：a^φ(n) ≡ 1 (mod n)
    return modular_power(a, phi_n, n) == 1

# 欧拉函数
def euler_phi(n: int) -> int:
    """计算欧拉函数"""
    if n <= 0:
        return 0
    
    result = n
    p = 2
    
    while p * p <= n:
        if n % p == 0:
            while n % p == 0:
                n //= p
            result -= result // p
        p += 1
    
    if n > 1:
        result -= result // n
    
    return result

# 模运算组合数
def modular_combination(n: int, k: int, m: int) -> int:
    """模运算组合数"""
    if k > n:
        return 0
    
    # 计算n!
    factorial_n = 1
    for i in range(1, n + 1):
        factorial_n = (factorial_n * i) % m
    
    # 计算k!
    factorial_k = 1
    for i in range(1, k + 1):
        factorial_k = (factorial_k * i) % m
    
    # 计算(n-k)!
    factorial_nk = 1
    for i in range(1, n - k + 1):
        factorial_nk = (factorial_nk * i) % m
    
    # 计算逆元
    inv_k = modular_inverse(factorial_k, m)
    inv_nk = modular_inverse(factorial_nk, m)
    
    # 计算组合数
    return (factorial_n * inv_k * inv_nk) % m

# 使用示例
if __name__ == "__main__":
