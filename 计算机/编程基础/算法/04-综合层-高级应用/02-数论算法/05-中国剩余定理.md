# 中国剩余定理

## 概述

中国剩余定理（Chinese Remainder Theorem, CRT）是数论中的重要定理，用于解决同余方程组问题。中国剩余定理在密码学、计算机科学、数论研究等领域有重要应用。

## 核心概念

### 基本定义
- **中国剩余定理**：解决同余方程组的定理
- **同余方程组**：多个同余方程组成的方程组
- **模数互质**：两个模数互质
- **唯一解**：在模数乘积下的唯一解

### 关键特性
- **唯一性**：在模数乘积下有唯一解
- **存在性**：当模数互质时解存在
- **构造性**：可以构造出解
- **可逆性**：可以逆推原问题

# 基本中国剩余定理
def basic_chinese_remainder_theorem(remainders: List[int], moduli: List[int]) -> int:
    """基本中国剩余定理"""
    if len(remainders) != len(moduli):
        raise ValueError("余数和模数数量不匹配")
    
    # 计算所有模数的乘积
    M = 1
    for m in moduli:
        M *= m
    
    result = 0
    for i in range(len(remainders)):
        Mi = M // moduli[i]
        Mi_inv = modular_inverse(Mi, moduli[i])
        result += remainders[i] * Mi * Mi_inv
    
    return result % M

# 扩展欧几里得算法
def extended_gcd(a: int, b: int) -> Tuple[int, int, int]:
    """扩展欧几里得算法"""
    if b == 0:
        return a, 1, 0
    
    gcd, x1, y1 = extended_gcd(b, a % b)
    x = y1
    y = x1 - (a // b) * y1
    
    return gcd, x, y

# 模逆元
def modular_inverse(a: int, m: int) -> int:
    """模逆元"""
    gcd, x, y = extended_gcd(a, m)
    if gcd != 1:
        raise ValueError("模逆元不存在")
    
    return (x % m + m) % m

# 优化的中国剩余定理
def optimized_chinese_remainder_theorem(remainders: List[int], moduli: List[int]) -> int:
    """优化的中国剩余定理"""
    if len(remainders) != len(moduli):
        raise ValueError("余数和模数数量不匹配")
    
    # 检查模数是否互质
    for i in range(len(moduli)):
        for j in range(i + 1, len(moduli)):
            if math.gcd(moduli[i], moduli[j]) != 1:
                raise ValueError("模数不互质")
    
    # 计算所有模数的乘积
    M = 1
    for m in moduli:
        M *= m
    
    result = 0
    for i in range(len(remainders)):
        Mi = M // moduli[i]
        Mi_inv = modular_inverse(Mi, moduli[i])
        result += remainders[i] * Mi * Mi_inv
    
    return result % M

# 非互质模数的中国剩余定理
def non_coprime_chinese_remainder_theorem(remainders: List[int], moduli: List[int]) -> int:
    """非互质模数的中国剩余定理"""
    if len(remainders) != len(moduli):
        raise ValueError("余数和模数数量不匹配")
    
    # 合并同余方程
    result = remainders[0]
    current_mod = moduli[0]
    
    for i in range(1, len(remainders)):
        # 解同余方程：x ≡ result (mod current_mod) 和 x ≡ remainders[i] (mod moduli[i])
        new_result, new_mod = solve_two_congruences(result, current_mod, remainders[i], moduli[i])
        result = new_result
        current_mod = new_mod
    
    return result

# 解两个同余方程
def solve_two_congruences(a1: int, m1: int, a2: int, m2: int) -> Tuple[int, int]:
    """解两个同余方程"""
    # 解同余方程：x ≡ a1 (mod m1) 和 x ≡ a2 (mod m2)
    gcd, x, y = extended_gcd(m1, m2)
    
    if (a2 - a1) % gcd != 0:
        raise ValueError("同余方程无解")
    
    # 计算解
    lcm = m1 * m2 // gcd
    k = (a2 - a1) // gcd
    x0 = a1 + k * x * m1
    
    return x0 % lcm, lcm

# 大数中国剩余定理
def big_number_chinese_remainder_theorem(remainders: List[int], moduli: List[int]) -> int:
    """大数中国剩余定理"""
    if len(remainders) != len(moduli):
        raise ValueError("余数和模数数量不匹配")
    
    # 使用优化的算法
    return optimized_chinese_remainder_theorem(remainders, moduli)
