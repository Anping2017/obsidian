# 最大公约数

## 概述

最大公约数（Greatest Common Divisor, GCD）是数论中的基础概念，指两个或多个整数中最大的能够整除它们的正整数。最大公约数算法在密码学、分数化简、线性代数等领域有重要应用。

## 核心概念

### 基本定义
- **最大公约数**：两个或多个整数的最大公因子
- **互质**：最大公约数为1的两个数
- **欧几里得算法**：计算最大公约数的高效算法
- **扩展欧几里得算法**：求解线性丢番图方程

### 关键特性
- **唯一性**：最大公约数是唯一的
- **非负性**：最大公约数非负
- **交换性**：gcd(a,b) = gcd(b,a)
- **结合性**：gcd(a,gcd(b,c)) = gcd(gcd(a,b),c)

## 最大公约数算法基础

### 基本最大公约数算法

```python
from typing import List, Tuple, Any, Callable, Optional
import time
import math

class GCDAlgorithms:
    """最大公约数算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.numbers = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, a: int, b: int) -> int:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(a, b)
    
    def set_numbers(self, numbers: List[int]) -> None:
        """设置数字"""
        self.numbers = numbers

# 暴力算法
def brute_force_gcd(a: int, b: int) -> int:
    """暴力算法计算最大公约数"""
    a, b = abs(a), abs(b)
    if a == 0:
        return b
    if b == 0:
        return a
    
    gcd = 1
    for i in range(1, min(a, b) + 1):
        if a % i == 0 and b % i == 0:
            gcd = i
    
    return gcd

# 欧几里得算法（递归）
def euclidean_gcd_recursive(a: int, b: int) -> int:
    """欧几里得算法（递归版本）"""
    a, b = abs(a), abs(b)
    if b == 0:
        return a
    return euclidean_gcd_recursive(b, a % b)

# 欧几里得算法（迭代）
def euclidean_gcd_iterative(a: int, b: int) -> int:
    """欧几里得算法（迭代版本）"""
    a, b = abs(a), abs(b)
    while b != 0:
        a, b = b, a % b
    return a

# 扩展欧几里得算法
def extended_euclidean_gcd(a: int, b: int) -> Tuple[int, int, int]:
    """扩展欧几里得算法"""
    a, b = abs(a), abs(b)
    
    if b == 0:
        return a, 1, 0
    
    gcd, x1, y1 = extended_euclidean_gcd(b, a % b)
    x = y1
    y = x1 - (a // b) * y1
    
    return gcd, x, y

# 二进制GCD算法
def binary_gcd(a: int, b: int) -> int:
    """二进制GCD算法"""
    a, b = abs(a), abs(b)
    
    if a == 0:
        return b
    if b == 0:
        return a
    
    # 计算2的幂次
    shift = 0
    while ((a | b) & 1) == 0:
        a >>= 1
        b >>= 1
        shift += 1
    
    # 确保a是奇数
    while (a & 1) == 0:
        a >>= 1
    
    # 主循环
    while b != 0:
        while (b & 1) == 0:
            b >>= 1
        
        if a > b:
            a, b = b, a
        
        b -= a
    
    return a << shift

# 多个数的最大公约数
def gcd_multiple(numbers: List[int]) -> int:
    """计算多个数的最大公约数"""
    if not numbers:
        return 0
    
    result = numbers[0]
    for num in numbers[1:]:
        result = euclidean_gcd_iterative(result, num)
        if result == 1:
            break
    
    return result

# 最小公倍数
def lcm(a: int, b: int) -> int:
    """计算最小公倍数"""
    return abs(a * b) // euclidean_gcd_iterative(a, b)

# 使用示例
if __name__ == "__main__":
    algorithms = GCDAlgorithms()
    
    # 添加算法
    algorithms.add_algorithm("brute_force", brute_force_gcd)
    algorithms.add_algorithm("euclidean_recursive", euclidean_gcd_recursive)
    algorithms.add_algorithm("euclidean_iterative", euclidean_gcd_iterative)
    algorithms.add_algorithm("extended_euclidean", extended_euclidean_gcd)
    algorithms.add_algorithm("binary_gcd", binary_gcd)
    
    # 测试数字
    a, b = 48, 18
    
    # 测试暴力算法
    result = algorithms.execute_algorithm("brute_force", a, b)
    print(f"暴力算法结果: {result}")
    
    # 测试欧几里得算法（递归）
    result = algorithms.execute_algorithm("euclidean_recursive", a, b)
    print(f"欧几里得算法（递归）结果: {result}")
    
    # 测试欧几里得算法（迭代）
    result = algorithms.execute_algorithm("euclidean_iterative", a, b)
    print(f"欧几里得算法（迭代）结果: {result}")
    
    # 测试扩展欧几里得算法
    result = algorithms.execute_algorithm("extended_euclidean", a, b)
    print(f"扩展欧几里得算法结果: {result}")
    
    # 测试二进制GCD算法
    result = algorithms.execute_algorithm("binary_gcd", a, b)
    print(f"二进制GCD算法结果: {result}")
    
    # 测试多个数的最大公约数

    result = gcd_multiple(numbers)
    print(f"多个数的最大公约数: {result}")
```

```python
# 快速GCD算法
def fast_gcd(a: int, b: int) -> int:
    """快速GCD算法"""
    a, b = abs(a), abs(b)
    
    if a == 0:
        return b
    if b == 0:
        return a
    
    # 使用位运算优化
    shift = 0
    while ((a | b) & 1) == 0:
        a >>= 1
        b >>= 1
        shift += 1
    
    while (a & 1) == 0:
        a >>= 1
    
    while b != 0:
        while (b & 1) == 0:
            b >>= 1
        
        if a > b:
            a, b = b, a
        
        b -= a
    
    return a << shift

# 模逆元
def modular_inverse(a: int, m: int) -> int:
    """计算模逆元"""
    gcd, x, y = extended_euclidean_gcd(a, m)
    if gcd != 1:
        raise ValueError("模逆元不存在")
    
    return (x % m + m) % m

# 中国剩余定理
def chinese_remainder_theorem(remainders: List[int], moduli: List[int]) -> int:
    """中国剩余定理"""
    if len(remainders) != len(moduli):
        raise ValueError("余数和模数数量不匹配")
    
    # 计算所有模数的乘积
    M = 1
    for m in moduli:
        M *= m
    
    result = 0
    for i in range(len(remainders)):
        Mi = M // moduli[i]
        Mi_inv = modular_inverse(Mi, moduli[i])
        result += remainders[i] * Mi * Mi_inv
    
    return result % M

# 费马小定理
def fermat_little_theorem(a: int, p: int) -> bool:
    """费马小定理验证"""
    if p <= 1:
        return False
    
    # 检查p是否为素数
    for i in range(2, int(math.sqrt(p)) + 1):
        if p % i == 0:
            return False
    
    # 费马小定理：a^(p-1) ≡ 1 (mod p)
    return pow(a, p - 1, p) == 1

# 欧拉函数
def euler_phi(n: int) -> int:
    """计算欧拉函数"""
    if n <= 0:
        return 0
    
    result = n
    p = 2
    
    while p * p <= n:
        if n % p == 0:
            while n % p == 0:
                n //= p
            result -= result // p
        p += 1
    
    if n > 1:
        result -= result // n
    
    return result

# 使用示例
if __name__ == "__main__":

    advanced_algorithms.add_algorithm("chinese_remainder_theorem", chinese_remainder_theorem)
    advanced_algorithms.add_algorithm("fermat_little_theorem", fermat_little_theorem)
    advanced_algorithms.add_algorithm("euler_phi", euler_phi)

    a, b = 48, 18

    result = advanced_algorithms.execute_algorithm("modular_inverse", a, m)
    print(f"模逆元结果: {result}")
    
    # 测试中国剩余定理

    moduli = [3, 5, 7]
    result = advanced_algorithms.execute_algorithm("chinese_remainder_theorem", remainders, moduli)
    print(f"中国剩余定理结果: {result}")
```

## 性能分析

### 最大公约数算法分析

| 算法类型 | 时间复杂度 | 空间复杂度 | 适用场景 | 学习难度 |
|---------|-----------|-----------|----------|----------|
| 暴力算法 | O(min(a,b)) | O(1) | 小数字 | 简单 |
| 欧几里得算法 | O(log(min(a,b))) | O(1) | 一般情况 | 简单 |
| 扩展欧几里得算法 | O(log(min(a,b))) | O(log(min(a,b))) | 需要系数 | 中等 |
| 二进制GCD算法 | O(log(min(a,b))) | O(1) | 大数字 | 中等 |
| 快速GCD算法 | O(log(min(a,b))) | O(1) | 大数字 | 中等 |

| 问题特征 | 推荐算法 | 优势 | 劣势 |
|---------|---------|------|------|
| 小数字 | 暴力算法 | 简单、直观 | 效率低 |
| 一般情况 | 欧几里得算法 | 高效、简单 | 递归开销 |
