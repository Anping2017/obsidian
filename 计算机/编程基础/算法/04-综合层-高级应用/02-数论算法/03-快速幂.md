# 快速幂

## 概述

快速幂（Fast Exponentiation）是数论中的重要算法，用于高效地计算大数的幂运算。快速幂算法在密码学、数论研究、算法竞赛等领域有重要应用。

## 核心概念

### 基本定义
- **快速幂**：高效计算a^n的算法
- **二进制分解**：将指数分解为二进制
- **模运算**：在模运算下的幂运算
- **矩阵快速幂**：矩阵的快速幂运算

### 关键特性
- **高效性**：时间复杂度O(log n)
- **递归性**：可以使用递归实现
- **迭代性**：可以使用迭代实现
- **模运算**：支持模运算

## 快速幂算法基础

### 基本快速幂算法

```python
from typing import List, Tuple, Any, Callable, Optional
import time
import math

class FastExponentiationAlgorithms:
    """快速幂算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.base = 0
        self.exponent = 0
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, base: int, exponent: int, modulus: int = None) -> int:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(base, exponent, modulus)
    
    def set_values(self, base: int, exponent: int) -> None:
        """设置值"""
        self.base = base
        self.exponent = exponent

# 递归快速幂
def recursive_fast_power(base: int, exponent: int, modulus: int = None) -> int:
    """递归快速幂"""
    if exponent == 0:
        return 1
    
    if exponent == 1:
        return base % modulus if modulus else base
    
    # 递归计算
    half = recursive_fast_power(base, exponent // 2, modulus)
    
    if exponent % 2 == 0:
        result = half * half
    else:
        result = half * half * base
    
    return result % modulus if modulus else result

# 迭代快速幂
def iterative_fast_power(base: int, exponent: int, modulus: int = None) -> int:
    """迭代快速幂"""
    result = 1
    base = base % modulus if modulus else base
    
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % modulus if modulus else result * base
        
        base = (base * base) % modulus if modulus else base * base
        exponent //= 2
    
    return result

# 二进制快速幂
def binary_fast_power(base: int, exponent: int, modulus: int = None) -> int:
    """二进制快速幂"""
    result = 1
    base = base % modulus if modulus else base
    
    # 将指数转换为二进制
    binary_exp = bin(exponent)[2:]  # 去掉'0b'前缀
    
    for bit in binary_exp:
        result = (result * result) % modulus if modulus else result * result
        if bit == '1':
            result = (result * base) % modulus if modulus else result * base
    
    return result

# 矩阵快速幂
def matrix_fast_power(matrix: List[List[int]], exponent: int, modulus: int = None) -> List[List[int]]:
    """矩阵快速幂"""
    n = len(matrix)
    
    if exponent == 0:
        # 返回单位矩阵
        identity = [[0] * n for _ in range(n)]
        for i in range(n):
            identity[i][i] = 1
        return identity
    
    if exponent == 1:
        return matrix
    
    # 递归计算
    half = matrix_fast_power(matrix, exponent // 2, modulus)
    
    # 矩阵乘法
    def matrix_multiply(a: List[List[int]], b: List[List[int]]) -> List[List[int]]:
        result = [[0] * n for _ in range(n)]
        for i in range(n):
            for j in range(n):
                for k in range(n):
                    result[i][j] += a[i][k] * b[k][j]
                    if modulus:
                        result[i][j] %= modulus
        return result
    
    if exponent % 2 == 0:
        return matrix_multiply(half, half)
    else:
        return matrix_multiply(matrix_multiply(half, half), matrix)

# 使用示例
if __name__ == "__main__":
    algorithms = FastExponentiationAlgorithms()
    
    # 添加算法
    algorithms.add_algorithm("recursive", recursive_fast_power)
    algorithms.add_algorithm("iterative", iterative_fast_power)
    algorithms.add_algorithm("binary", binary_fast_power)
    algorithms.add_algorithm("matrix", matrix_fast_power)
    
    # 测试数字
    base, exponent = 2, 10
    modulus = 1000000007
    
    # 测试递归快速幂
    result = algorithms.execute_algorithm("recursive", base, exponent, modulus)
    print(f"递归快速幂结果: {result}")
    
    # 测试迭代快速幂
    result = algorithms.execute_algorithm("iterative", base, exponent, modulus)
    print(f"迭代快速幂结果: {result}")
    
    # 测试二进制快速幂
    result = algorithms.execute_algorithm("binary", base, exponent, modulus)
    print(f"二进制快速幂结果: {result}")
    
    # 测试矩阵快速幂
    matrix = [[1, 1], [1, 0]]
    result = algorithms.execute_algorithm("matrix", matrix, exponent, modulus)
    print(f"矩阵快速幂结果: {result}")
```

```python
# 优化的快速幂
def optimized_fast_power(base: int, exponent: int, modulus: int = None) -> int:
    """优化的快速幂"""
    if modulus:
        base %= modulus
    
    result = 1
    while exponent > 0:
        if exponent & 1:  # 使用位运算
            result = (result * base) % modulus if modulus else result * base
        
        base = (base * base) % modulus if modulus else base * base
        exponent >>= 1  # 使用位运算
    
    return result

# 大数快速幂
def big_number_fast_power(base: int, exponent: int, modulus: int = None) -> int:
    """大数快速幂"""
    if modulus:
        base %= modulus
    
    result = 1
    while exponent > 0:
        if exponent & 1:
            result = (result * base) % modulus if modulus else result * base
        
        base = (base * base) % modulus if modulus else base * base
        exponent >>= 1
    
    return result

# 快速幂求逆元
def fast_power_inverse(a: int, modulus: int) -> int:
    """快速幂求逆元"""
    return optimized_fast_power(a, modulus - 2, modulus)

# 快速幂求组合数
def fast_power_combination(n: int, k: int, modulus: int) -> int:
    """快速幂求组合数"""
    if k > n:
        return 0
    
    # 计算n!
    factorial_n = 1
    for i in range(1, n + 1):
        factorial_n = (factorial_n * i) % modulus
    
    # 计算k!
    factorial_k = 1
    for i in range(1, k + 1):
        factorial_k = (factorial_k * i) % modulus
    
    # 计算(n-k)!
    factorial_nk = 1
    for i in range(1, n - k + 1):
        factorial_nk = (factorial_nk * i) % modulus
    
    # 计算逆元
    inv_k = fast_power_inverse(factorial_k, modulus)
    inv_nk = fast_power_inverse(factorial_nk, modulus)
    
    # 计算组合数
    return (factorial_n * inv_k * inv_nk) % modulus

# 使用示例
if __name__ == "__main__":
    advanced_algorithms = AdvancedFastExponentiationAlgorithms()
    
    # 添加算法
    advanced_algorithms.add_algorithm("optimized", optimized_fast_power)
    advanced_algorithms.add_algorithm("big_number", big_number_fast_power)
    advanced_algorithms.add_algorithm("inverse", fast_power_inverse)
    advanced_algorithms.add_algorithm("combination", fast_power_combination)
    
    # 测试优化的快速幂

    a = 3

    print(f"快速幂求逆元结果: {result}")

    result = advanced_algorithms.execute_algorithm("combination", n, k, modulus)
    print(f"快速幂求组合数结果: {result}")
```

## 性能分析

### 快速幂算法分析
