# 遗传算法

## 概述

遗传算法（Genetic Algorithm, GA）是一种模拟生物进化过程的优化算法，通过模拟自然选择、交叉和变异等生物进化机制来寻找问题的最优解。遗传算法特别适用于复杂的优化问题，能够在解空间中全局搜索，避免陷入局部最优。

## 核心概念

### 基本思想
- **个体表示**：将解编码为染色体
- **适应度函数**：评估个体优劣
- **选择操作**：选择优秀个体
- **交叉操作**：产生新个体
- **变异操作**：增加多样性

### 关键特性
- **全局搜索**：避免局部最优
- **并行性**：可并行处理
- **鲁棒性**：对问题变化适应
- **通用性**：适用于多种问题

## 遗传算法基础

### 基本遗传算法

```python
from typing import List, Tuple, Any, Callable, Optional
import random
import math
import time

class GeneticAlgorithm:
    """基本遗传算法"""
    
    def __init__(self, population_size: int = 100, generations: int = 100,
                 crossover_rate: float = 0.8, mutation_rate: float = 0.1,
                 elitism_rate: float = 0.1):
        self.population_size = population_size
        self.generations = generations
        self.crossover_rate = crossover_rate
        self.mutation_rate = mutation_rate
        self.elitism_rate = elitism_rate
        
        self.population = []
        self.best_individual = None
        self.best_fitness = float('inf')
        self.fitness_history = []
    
    def solve(self, chromosome_length: int, fitness_function: Callable,
              bounds: Optional[List[Tuple[float, float]]] = None,
              encoding_type: str = 'binary') -> Tuple[List[Any], float]:
        """求解优化问题"""
        # 初始化种群
        self._initialize_population(chromosome_length, bounds, encoding_type)
        
        # 进化过程
        for generation in range(self.generations):
            # 评估适应度
            self._evaluate_fitness(fitness_function)
            
            # 记录最佳个体
            self._update_best_individual()
            
            # 选择操作
            selected_population = self._selection()
            
            # 交叉操作
            offspring_population = self._crossover(selected_population)
            
            # 变异操作
            mutated_population = self._mutation(offspring_population)
            
            # 更新种群
            self._update_population(mutated_population)
            
            # 记录适应度历史
            self.fitness_history.append(self.best_fitness)
        
        return self.best_individual, self.best_fitness
    
    def _initialize_population(self, chromosome_length: int,
                             bounds: Optional[List[Tuple[float, float]]],
                             encoding_type: str) -> None:
        """初始化种群"""
        self.population = []
        
        for _ in range(self.population_size):
            if encoding_type == 'binary':
                chromosome = [random.randint(0, 1) for _ in range(chromosome_length)]
            elif encoding_type == 'real':
                chromosome = []
                for i in range(chromosome_length):
                    if bounds and i < len(bounds):
                        lower, upper = bounds[i]
                        chromosome.append(random.uniform(lower, upper))
                    else:
                        chromosome.append(random.uniform(-10, 10))
            elif encoding_type == 'permutation':
                chromosome = list(range(chromosome_length))
                random.shuffle(chromosome)
            else:
                raise ValueError(f"不支持的编码类型: {encoding_type}")
            
            self.population.append(chromosome)
    
    def _evaluate_fitness(self, fitness_function: Callable) -> None:
        """评估适应度"""
        for individual in self.population:
            individual.fitness = fitness_function(individual)
    
    def _update_best_individual(self) -> None:
        """更新最佳个体"""
        for individual in self.population:
            if individual.fitness < self.best_fitness:
                self.best_individual = individual.copy()
                self.best_fitness = individual.fitness
    
    def _selection(self) -> List[List[Any]]:
        """选择操作（轮盘赌选择）"""
        # 计算适应度总和
        total_fitness = sum(individual.fitness for individual in self.population)
        
        # 计算选择概率
        probabilities = []
        for individual in self.population:
            if total_fitness > 0:
                prob = individual.fitness / total_fitness
            else:
                prob = 1.0 / len(self.population)
            probabilities.append(prob)
        
        # 轮盘赌选择
        selected_population = []
        for _ in range(self.population_size):
            rand = random.random()
            cumulative = 0
            for i, prob in enumerate(probabilities):
                cumulative += prob
                if rand <= cumulative:
                    selected_population.append(self.population[i].copy())
                    break
        
        return selected_population
    
    def _crossover(self, selected_population: List[List[Any]]) -> List[List[Any]]:
        """交叉操作（单点交叉）"""
        offspring_population = []
        
        for i in range(0, len(selected_population), 2):
            parent1 = selected_population[i]
            parent2 = selected_population[i + 1] if i + 1 < len(selected_population) else selected_population[i]
            
            if random.random() < self.crossover_rate:
                # 执行交叉
                crossover_point = random.randint(1, len(parent1) - 1)
                child1 = parent1[:crossover_point] + parent2[crossover_point:]
                child2 = parent2[:crossover_point] + parent1[crossover_point:]
                
                offspring_population.extend([child1, child2])
            else:
                # 不执行交叉
                offspring_population.extend([parent1, parent2])
        
        return offspring_population
    
    def _mutation(self, offspring_population: List[List[Any]]) -> List[List[Any]]:
        """变异操作"""
        mutated_population = []
        
        for individual in offspring_population:
            if random.random() < self.mutation_rate:
                # 执行变异
                mutation_point = random.randint(0, len(individual) - 1)
                
                if isinstance(individual[mutation_point], int):
                    # 二进制变异
                    individual[mutation_point] = 1 - individual[mutation_point]
                elif isinstance(individual[mutation_point], float):
                    # 实数变异
                    individual[mutation_point] += random.gauss(0, 0.1)
                else:
                    # 其他类型变异
                    individual[mutation_point] = random.choice(range(len(individual)))
            
            mutated_population.append(individual)
        
        return mutated_population
    
    def _update_population(self, mutated_population: List[List[Any]]) -> None:
        """更新种群"""
        # 精英保留
        elite_size = int(self.population_size * self.elitism_rate)
        elite_individuals = sorted(self.population, key=lambda x: x.fitness)[:elite_size]
        
        # 更新种群
        self.population = elite_individuals + mutated_population[:self.population_size - elite_size]
```

### 高级遗传算法

```python
from typing import List, Tuple, Any, Callable, Optional
import random
import math
import time

class AdvancedGeneticAlgorithm:
    """高级遗传算法"""
    
    def __init__(self, population_size: int = 100, generations: int = 100,
                 crossover_rate: float = 0.8, mutation_rate: float = 0.1,
                 elitism_rate: float = 0.1, tournament_size: int = 3):
        self.population_size = population_size
        self.generations = generations
        self.crossover_rate = crossover_rate
        self.mutation_rate = mutation_rate
        self.elitism_rate = elitism_rate
        self.tournament_size = tournament_size
        
        self.population = []
        self.best_individual = None
        self.best_fitness = float('inf')
        self.fitness_history = []
        self.diversity_history = []
    
    def solve(self, chromosome_length: int, fitness_function: Callable,
              bounds: Optional[List[Tuple[float, float]]] = None,
              encoding_type: str = 'real') -> Tuple[List[Any], float]:
        """求解优化问题"""
        # 初始化种群
        self._initialize_population(chromosome_length, bounds, encoding_type)
        
        # 进化过程
        for generation in range(self.generations):
            # 评估适应度
            self._evaluate_fitness(fitness_function)
            
            # 记录最佳个体
            self._update_best_individual()
            
            # 记录多样性
            diversity = self._calculate_diversity()
            self.diversity_history.append(diversity)
            
            # 自适应参数调整
            self._adaptive_parameter_adjustment(generation)
            
            # 选择操作
            selected_population = self._tournament_selection()
            
            # 交叉操作
            offspring_population = self._uniform_crossover(selected_population)
            
            # 变异操作
            mutated_population = self._gaussian_mutation(offspring_population)
            
            # 更新种群
            self._update_population(mutated_population)
            
            # 记录适应度历史
            self.fitness_history.append(self.best_fitness)
        
        return self.best_individual, self.best_fitness
    
    def _initialize_population(self, chromosome_length: int,
                             bounds: Optional[List[Tuple[float, float]]],
                             encoding_type: str) -> None:
        """初始化种群"""
        self.population = []
        
        for _ in range(self.population_size):
            if encoding_type == 'real':
                chromosome = []
                for i in range(chromosome_length):
                    if bounds and i < len(bounds):
                        lower, upper = bounds[i]
                        chromosome.append(random.uniform(lower, upper))
                    else:
                        chromosome.append(random.uniform(-10, 10))
            else:
                raise ValueError(f"不支持的编码类型: {encoding_type}")
            
            self.population.append(chromosome)
    
    def _evaluate_fitness(self, fitness_function: Callable) -> None:
        """评估适应度"""
        for individual in self.population:
            individual.fitness = fitness_function(individual)
    
    def _update_best_individual(self) -> None:
        """更新最佳个体"""
        for individual in self.population:
            if individual.fitness < self.best_fitness:
                self.best_individual = individual.copy()
                self.best_fitness = individual.fitness
    
    def _calculate_diversity(self) -> float:
        """计算种群多样性"""
        if len(self.population) < 2:
            return 0
        
        total_distance = 0
        count = 0
        
        for i in range(len(self.population)):
            for j in range(i + 1, len(self.population)):
                distance = math.sqrt(sum((a - b) ** 2 for a, b in zip(self.population[i], self.population[j])))
                total_distance += distance
                count += 1
        
        return total_distance / count if count > 0 else 0
    
    def _adaptive_parameter_adjustment(self, generation: int) -> None:
        """自适应参数调整"""
        # 动态调整交叉率和变异率
        progress = generation / self.generations
        
        # 交叉率随进化过程降低
        self.crossover_rate = 0.9 - 0.3 * progress
        
        # 变异率随进化过程降低
        self.mutation_rate = 0.2 - 0.15 * progress
    
    def _tournament_selection(self) -> List[List[Any]]:
        """锦标赛选择"""
        selected_population = []
        
        for _ in range(self.population_size):
            # 随机选择锦标赛参与者
            tournament = random.sample(self.population, self.tournament_size)
            
            # 选择最佳个体
            best_individual = min(tournament, key=lambda x: x.fitness)
            selected_population.append(best_individual.copy())
        
        return selected_population
    
    def _uniform_crossover(self, selected_population: List[List[Any]]) -> List[List[Any]]:
        """均匀交叉"""
        offspring_population = []
        
        for i in range(0, len(selected_population), 2):
            parent1 = selected_population[i]
            parent2 = selected_population[i + 1] if i + 1 < len(selected_population) else selected_population[i]
            
            if random.random() < self.crossover_rate:
                # 执行均匀交叉
                child1 = []
                child2 = []
                
                for j in range(len(parent1)):
                    if random.random() < 0.5:
                        child1.append(parent1[j])
                        child2.append(parent2[j])
                    else:
                        child1.append(parent2[j])
                        child2.append(parent1[j])
                
                offspring_population.extend([child1, child2])
            else:
                # 不执行交叉
                offspring_population.extend([parent1, parent2])
        
        return offspring_population
    
    def _gaussian_mutation(self, offspring_population: List[List[Any]]) -> List[List[Any]]:
        """高斯变异"""
        mutated_population = []
        
        for individual in offspring_population:
            if random.random() < self.mutation_rate:
                # 执行高斯变异
                for i in range(len(individual)):
                    if random.random() < 0.1:  # 每个基因的变异概率
                        individual[i] += random.gauss(0, 0.1)
            
            mutated_population.append(individual)
        
        return mutated_population
    
    def _update_population(self, mutated_population: List[List[Any]]) -> None:
        """更新种群"""
        # 精英保留
        elite_size = int(self.population_size * self.elitism_rate)
        elite_individuals = sorted(self.population, key=lambda x: x.fitness)[:elite_size]
        
        # 更新种群
        self.population = elite_individuals + mutated_population[:self.population_size - elite_size]
```

## 遗传算法变种

### 1. 多目标遗传算法

```python
from typing import List, Tuple, Any, Callable, Optional
import random
import math
import time

class MultiObjectiveGeneticAlgorithm:
    """多目标遗传算法（NSGA-II）"""
    
    def __init__(self, population_size: int = 100, generations: int = 100,
                 crossover_rate: float = 0.8, mutation_rate: float = 0.1):
        self.population_size = population_size
        self.generations = generations
        self.crossover_rate = crossover_rate
        self.mutation_rate = mutation_rate
        
        self.population = []
        self.fronts = []
        self.best_solutions = []
    
    def solve(self, chromosome_length: int, fitness_functions: List[Callable],
              bounds: Optional[List[Tuple[float, float]]] = None) -> List[List[Any]]:
        """求解多目标优化问题"""
        # 初始化种群
        self._initialize_population(chromosome_length, bounds)
        
        # 进化过程
        for generation in range(self.generations):
            # 评估适应度
            self._evaluate_fitness(fitness_functions)
            
            # 非支配排序
            self._non_dominated_sorting()
            
            # 拥挤距离计算
            self._calculate_crowding_distance()
            
            # 选择操作
            selected_population = self._selection()
            
            # 交叉操作
            offspring_population = self._crossover(selected_population)
            
            # 变异操作
            mutated_population = self._mutation(offspring_population)
            
            # 更新种群
            self._update_population(mutated_population)
        
        return self.best_solutions
    
    def _initialize_population(self, chromosome_length: int,
                             bounds: Optional[List[Tuple[float, float]]]) -> None:
        """初始化种群"""
        self.population = []
        
        for _ in range(self.population_size):
            chromosome = []
            for i in range(chromosome_length):
                if bounds and i < len(bounds):
                    lower, upper = bounds[i]
                    chromosome.append(random.uniform(lower, upper))
                else:
                    chromosome.append(random.uniform(-10, 10))
            
            self.population.append(chromosome)
    
    def _evaluate_fitness(self, fitness_functions: List[Callable]) -> None:
        """评估适应度"""
        for individual in self.population:
            individual.fitness = [f(individual) for f in fitness_functions]
    
    def _non_dominated_sorting(self) -> None:
        """非支配排序"""
        self.fronts = []
        remaining_individuals = self.population.copy()
        
        while remaining_individuals:
            current_front = []
            for individual in remaining_individuals:
                is_dominated = False
                for other in remaining_individuals:
                    if self._dominates(other, individual):
                        is_dominated = True
                        break
                
                if not is_dominated:
                    current_front.append(individual)
            
            self.fronts.append(current_front)
            remaining_individuals = [ind for ind in remaining_individuals if ind not in current_front]
    
    def _dominates(self, individual1: List[Any], individual2: List[Any]) -> bool:
        """判断个体1是否支配个体2"""
        fitness1 = individual1.fitness
        fitness2 = individual2.fitness
        
        # 至少有一个目标更好
        better = any(f1 < f2 for f1, f2 in zip(fitness1, fitness2))
        
        # 所有目标都不差
        not_worse = all(f1 <= f2 for f1, f2 in zip(fitness1, fitness2))
        
        return better and not_worse
    
    def _calculate_crowding_distance(self) -> None:
        """计算拥挤距离"""
        for front in self.fronts:
            if len(front) <= 2:
                for individual in front:
                    individual.crowding_distance = float('inf')
                continue
            
            # 初始化拥挤距离
            for individual in front:
                individual.crowding_distance = 0
            
            # 对每个目标函数计算拥挤距离
            num_objectives = len(front[0].fitness)
            for obj_idx in range(num_objectives):
                # 按目标函数值排序
                front.sort(key=lambda x: x.fitness[obj_idx])
                
                # 边界个体距离设为无穷大
                front[0].crowding_distance = float('inf')
                front[-1].crowding_distance = float('inf')
                
                # 计算中间个体的拥挤距离
                obj_range = front[-1].fitness[obj_idx] - front[0].fitness[obj_idx]
                if obj_range > 0:
                    for i in range(1, len(front) - 1):
                        distance = (front[i + 1].fitness[obj_idx] - front[i - 1].fitness[obj_idx]) / obj_range
                        front[i].crowding_distance += distance
    
    def _selection(self) -> List[List[Any]]:
        """选择操作"""
        selected_population = []
        
        # 按前沿和拥挤距离选择
        for front in self.fronts:
            if len(selected_population) + len(front) <= self.population_size:
                selected_population.extend(front)
            else:
                # 按拥挤距离排序
                front.sort(key=lambda x: x.crowding_distance, reverse=True)
                remaining = self.population_size - len(selected_population)
                selected_population.extend(front[:remaining])
                break
        
        return selected_population
    
    def _crossover(self, selected_population: List[List[Any]]) -> List[List[Any]]:
        """交叉操作"""
        offspring_population = []
        
        for i in range(0, len(selected_population), 2):
            parent1 = selected_population[i]
            parent2 = selected_population[i + 1] if i + 1 < len(selected_population) else selected_population[i]
            
            if random.random() < self.crossover_rate:
                # 执行交叉
                child1 = []
                child2 = []
                
                for j in range(len(parent1)):
                    if random.random() < 0.5:
                        child1.append(parent1[j])
                        child2.append(parent2[j])
                    else:
                        child1.append(parent2[j])
                        child2.append(parent1[j])
                
                offspring_population.extend([child1, child2])
            else:
                offspring_population.extend([parent1, parent2])
        
        return offspring_population
    
    def _mutation(self, offspring_population: List[List[Any]]) -> List[List[Any]]:
        """变异操作"""
        mutated_population = []
        
        for individual in offspring_population:
            if random.random() < self.mutation_rate:
                for i in range(len(individual)):
                    if random.random() < 0.1:
                        individual[i] += random.gauss(0, 0.1)
            
            mutated_population.append(individual)
        
        return mutated_population
    
    def _update_population(self, mutated_population: List[List[Any]]) -> None:
        """更新种群"""
        # 合并父代和子代
        combined_population = self.population + mutated_population
        
        # 非支配排序
        self._non_dominated_sorting()
        
        # 选择下一代
        self.population = self._selection()
        
        # 更新最佳解
        self.best_solutions = self.fronts[0] if self.fronts else []

# 使用示例
if __name__ == "__main__":
    # 定义多目标函数
    def objective1(x: List[float]) -> float:
        return x[0] ** 2 + x[1] ** 2
    
    def objective2(x: List[float]) -> float:
        return (x[0] - 1) ** 2 + (x[1] - 1) ** 2
    
    # 多目标优化
    bounds = [(-5, 5), (-5, 5)]
    dimensions = 2
    
    mo_ga = MultiObjectiveGeneticAlgorithm(population_size=50, generations=100)
    
    start_time = time.time()
    best_solutions = mo_ga.solve(
        chromosome_length=dimensions,
        fitness_functions=[objective1, objective2],
        bounds=bounds
    )
    mo_ga_time = time.time() - start_time
    
    print(f"多目标优化解数量: {len(best_solutions)}")
    print(f"多目标优化求解时间: {mo_ga_time:.4f}秒")
    
    # 打印前几个解
    for i, solution in enumerate(best_solutions[:5]):
        print(f"解 {i+1}: {solution}, 目标值: {solution.fitness}")
```

## 性能分析

### 时间复杂度
- **基本遗传算法**：O(generations * population_size * chromosome_length)
- **多目标遗传算法**：O(generations * population_size² * objectives)
- **高级遗传算法**：O(generations * population_size * chromosome_length)

### 空间复杂度
- **基本遗传算法**：O(population_size * chromosome_length)

- **高级遗传算法**：O(population_size * chromosome_length)
