# 旋转变换

## 概述

旋转变换（Rotation Transformation）是计算几何中的重要变换，用于将点、线段、多边形等几何对象绕某个中心点旋转一定角度。旋转变换在计算机图形学、机器人学、游戏开发等领域有广泛应用。

## 核心概念

### 基本定义
- **旋转变换**：将几何对象绕中心点旋转
- **旋转中心**：旋转的中心点
- **旋转角度**：旋转的角度（弧度或度数）
- **旋转矩阵**：用于表示旋转变换的矩阵

### 关键特性
- **保角性**：旋转保持角度不变
- **保距性**：旋转保持距离不变
- **可逆性**：旋转可以逆变换
- **组合性**：多个旋转可以组合

## 旋转变换算法基础

### 基本旋转变换算法

```python
from typing import List, Tuple, Any, Callable, Optional
import time
import math

class RotationTransformation:
    """旋转变换算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.points = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, points: List[Tuple[float, float]], center: Tuple[float, float], angle: float) -> List[Tuple[float, float]]:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(points, center, angle)
    
    def set_points(self, points: List[Tuple[float, float]]) -> None:
        """设置点集"""
        self.points = points

# 单点旋转变换
def rotate_point(point: Tuple[float, float], center: Tuple[float, float], angle: float) -> Tuple[float, float]:
    """单点旋转变换"""
    x, y = point
    cx, cy = center
    
    # 将点平移到原点
    x -= cx
    y -= cy
    
    # 应用旋转变换
    cos_angle = math.cos(angle)
    sin_angle = math.sin(angle)
    
    new_x = x * cos_angle - y * sin_angle
    new_y = x * sin_angle + y * cos_angle
    
    # 平移回原位置
    new_x += cx
    new_y += cy
    
    return (new_x, new_y)

# 多点旋转变换
def rotate_points(points: List[Tuple[float, float]], center: Tuple[float, float], angle: float) -> List[Tuple[float, float]]:
    """多点旋转变换"""
    rotated_points = []
    
    for point in points:
        rotated_point = rotate_point(point, center, angle)
        rotated_points.append(rotated_point)
    
    return rotated_points

# 线段旋转变换
def rotate_segment(segment: Tuple[Tuple[float, float], Tuple[float, float]], center: Tuple[float, float], angle: float) -> Tuple[Tuple[float, float], Tuple[float, float]]:
    """线段旋转变换"""
    p1, p2 = segment
    
    rotated_p1 = rotate_point(p1, center, angle)
    rotated_p2 = rotate_point(p2, center, angle)
    
    return (rotated_p1, rotated_p2)

# 多边形旋转变换
def rotate_polygon(polygon: List[Tuple[float, float]], center: Tuple[float, float], angle: float) -> List[Tuple[float, float]]:
    """多边形旋转变换"""
    return rotate_points(polygon, center, angle)

# 使用示例
if __name__ == "__main__":
    rotation_algorithms = RotationTransformation()
    
    # 添加算法
    rotation_algorithms.add_algorithm("rotate_points", rotate_points)
    rotation_algorithms.add_algorithm("rotate_segment", rotate_segment)
    rotation_algorithms.add_algorithm("rotate_polygon", rotate_polygon)
    
    # 测试点集
    test_points = [(0, 0), (1, 0), (1, 1), (0, 1)]
    center = (0.5, 0.5)
    angle = math.pi / 4  # 45度
    
    # 测试多点旋转
    result = rotation_algorithms.execute_algorithm("rotate_points", test_points, center, angle)
    print(f"多点旋转结果: {result}")
    
    # 测试线段旋转
    test_segment = ((0, 0), (2, 0))
    result = rotation_algorithms.execute_algorithm("rotate_segment", test_segment, center, angle)
    print(f"线段旋转结果: {result}")
    
    # 测试多边形旋转
    result = rotation_algorithms.execute_algorithm("rotate_polygon", test_points, center, angle)
    print(f"多边形旋转结果: {result}")
```

### 高级旋转变换算法

```python
from typing import List, Tuple, Any, Callable, Optional
import time
import math

class AdvancedRotationTransformation:
    """高级旋转变换算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.points = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, points: List[Tuple[float, float]], center: Tuple[float, float], angle: float) -> List[Tuple[float, float]]:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(points, center, angle)
    
    def set_points(self, points: List[Tuple[float, float]]) -> None:
        """设置点集"""
        self.points = points

# 三维旋转变换
def rotate_point_3d(point: Tuple[float, float, float], center: Tuple[float, float, float], 
                   axis: Tuple[float, float, float], angle: float) -> Tuple[float, float, float]:
    """三维点旋转变换"""
    x, y, z = point
    cx, cy, cz = center
    ax, ay, az = axis
    
    # 将点平移到原点
    x -= cx
    y -= cy
    z -= cz
    
    # 归一化旋转轴
    length = math.sqrt(ax**2 + ay**2 + az**2)
    if length > 0:
        ax /= length
        ay /= length
        az /= length
    
    # 计算旋转矩阵
    cos_angle = math.cos(angle)
    sin_angle = math.sin(angle)
    
    # 旋转矩阵
    rotation_matrix = [
        [cos_angle + ax**2 * (1 - cos_angle), ax * ay * (1 - cos_angle) - az * sin_angle, ax * az * (1 - cos_angle) + ay * sin_angle],
        [ay * ax * (1 - cos_angle) + az * sin_angle, cos_angle + ay**2 * (1 - cos_angle), ay * az * (1 - cos_angle) - ax * sin_angle],
        [az * ax * (1 - cos_angle) - ay * sin_angle, az * ay * (1 - cos_angle) + ax * sin_angle, cos_angle + az**2 * (1 - cos_angle)]
    ]
    
    # 应用旋转变换
    new_x = rotation_matrix[0][0] * x + rotation_matrix[0][1] * y + rotation_matrix[0][2] * z
    new_y = rotation_matrix[1][0] * x + rotation_matrix[1][1] * y + rotation_matrix[1][2] * z
    new_z = rotation_matrix[2][0] * x + rotation_matrix[2][1] * y + rotation_matrix[2][2] * z
    
    # 平移回原位置
    new_x += cx
    new_y += cy
    new_z += cz
    
    return (new_x, new_y, new_z)

# 绕X轴旋转
def rotate_around_x(point: Tuple[float, float, float], center: Tuple[float, float, float], angle: float) -> Tuple[float, float, float]:
    """绕X轴旋转"""
    return rotate_point_3d(point, center, (1, 0, 0), angle)

# 绕Y轴旋转
def rotate_around_y(point: Tuple[float, float, float], center: Tuple[float, float, float], angle: float) -> Tuple[float, float, float]:
    """绕Y轴旋转"""
    return rotate_point_3d(point, center, (0, 1, 0), angle)

# 绕Z轴旋转
def rotate_around_z(point: Tuple[float, float, float], center: Tuple[float, float, float], angle: float) -> Tuple[float, float, float]:
    """绕Z轴旋转"""
    return rotate_point_3d(point, center, (0, 0, 1), angle)

# 组合旋转变换
def combined_rotation(points: List[Tuple[float, float]], center: Tuple[float, float], 
                     rotations: List[Tuple[str, float]]) -> List[Tuple[float, float]]:
    """组合旋转变换"""
    result_points = points.copy()
    
    for rotation_type, angle in rotations:
        if rotation_type == "x":
            # 绕X轴旋转（在2D中相当于绕Y轴旋转）
            result_points = rotate_points(result_points, center, angle)
        elif rotation_type == "y":
            # 绕Y轴旋转（在2D中相当于绕X轴旋转）
            result_points = rotate_points(result_points, center, angle)
        elif rotation_type == "z":
            # 绕Z轴旋转（在2D中相当于绕Z轴旋转）
            result_points = rotate_points(result_points, center, angle)
    
    return result_points

# 使用示例
if __name__ == "__main__":
    advanced_algorithms = AdvancedRotationTransformation()
    
    # 添加算法
    advanced_algorithms.add_algorithm("rotate_point_3d", rotate_point_3d)
    advanced_algorithms.add_algorithm("rotate_around_x", rotate_around_x)
    advanced_algorithms.add_algorithm("rotate_around_y", rotate_around_y)
    advanced_algorithms.add_algorithm("rotate_around_z", rotate_around_z)
    advanced_algorithms.add_algorithm("combined_rotation", combined_rotation)
    
    # 测试三维点旋转
    test_point_3d = (1, 0, 0)
    center_3d = (0, 0, 0)
    axis = (0, 0, 1)
    angle = math.pi / 2
    
    result = advanced_algorithms.execute_algorithm("rotate_point_3d", test_point_3d, center_3d, axis, angle)
    print(f"三维点旋转结果: {result}")
    
    # 测试绕X轴旋转
    result = advanced_algorithms.execute_algorithm("rotate_around_x", test_point_3d, center_3d, angle)
    print(f"绕X轴旋转结果: {result}")
    
    # 测试绕Y轴旋转
    result = advanced_algorithms.execute_algorithm("rotate_around_y", test_point_3d, center_3d, angle)
    print(f"绕Y轴旋转结果: {result}")
    
    # 测试绕Z轴旋转
    result = advanced_algorithms.execute_algorithm("rotate_around_z", test_point_3d, center_3d, angle)
    print(f"绕Z轴旋转结果: {result}")
    
    # 测试组合旋转
    test_points = [(0, 0), (1, 0), (1, 1), (0, 1)]
    center = (0.5, 0.5)
    rotations = [("z", math.pi / 4), ("z", math.pi / 4)]
    
    result = advanced_algorithms.execute_algorithm("combined_rotation", test_points, center, rotations)
    print(f"组合旋转结果: {result}")
```
