# 双指针技巧

## 🎯 核心概念

**双指针技巧**是一种高效的算法优化技术，通过使用两个指针在数组或序列中同时移动来解决问题。双指针技巧通常能将O(n²)的暴力解法优化到O(n)，是算法优化的重要方法。

def two_pointers_types():
    """双指针类型"""
    
    def collision_pointers(arr, target):
        """对撞指针"""
        left, right = 0, len(arr) - 1
        
        while left < right:
            current_sum = arr[left] + arr[right]
            if current_sum == target:
                return [left, right]
            elif current_sum < target:
                left += 1
            else:
                right -= 1
        
        return None
    
    def fast_slow_pointers(head):
        """快慢指针"""
        slow = fast = head
        
        while fast and fast.next:
            slow = slow.next
            fast = fast.next.next
            
            if slow == fast:
                return True  # 有环
        
        return False  # 无环
    
    def sliding_window_pointers(arr, k):
        """滑动窗口指针"""
        left = 0
        max_sum = 0
        current_sum = 0
        
        for right in range(len(arr)):
            current_sum += arr[right]
            
            if right >= k - 1:
                max_sum = max(max_sum, current_sum)
                current_sum -= arr[left]
                left += 1
        
        return max_sum
    
    return collision_pointers, fast_slow_pointers, sliding_window_pointers

        def __init__(self):
            self.results = []
        
        def two_sum(self, arr, target):
            """两数之和"""
            left, right = 0, len(arr) - 1
            
            while left < right:
                current_sum = arr[left] + arr[right]
                if current_sum == target:
                    return [left, right]
                elif current_sum < target:
                    left += 1
                else:
                    right -= 1
            
            return None
        
        def three_sum(self, arr, target):
            """三数之和"""
            arr.sort()
            results = []
            
            for i in range(len(arr) - 2):
                if i > 0 and arr[i] == arr[i - 1]:
                    continue
                
                left, right = i + 1, len(arr) - 1
                
                while left < right:
                    current_sum = arr[i] + arr[left] + arr[right]
                    
                    if current_sum == target:
                        results.append([arr[i], arr[left], arr[right]])
                        
                        while left < right and arr[left] == arr[left + 1]:
                            left += 1
                        while left < right and arr[right] == arr[right - 1]:
                            right -= 1
                        
                        left += 1
                        right -= 1
                    elif current_sum < target:
                        left += 1
                    else:
                        right -= 1
            
            return results
        
        def container_with_most_water(self, height):
            """盛最多水的容器"""
            left, right = 0, len(height) - 1
            max_area = 0
            
            while left < right:
                area = min(height[left], height[right]) * (right - left)
                max_area = max(max_area, area)
                
                if height[left] < height[right]:
                    left += 1
                else:
                    right -= 1
            
            return max_area
        
        def remove_duplicates(self, arr):
            """删除重复元素"""
            if not arr:
                return 0
            
            slow = 0
            for fast in range(1, len(arr)):
                if arr[fast] != arr[slow]:
                    slow += 1
                    arr[slow] = arr[fast]

                right -= 1

                fast = fast.next.next
                
                if slow == fast:

            fast = fast.next
        
        slow.next = slow.next.next
        
        return dummy.next
    
    def merge_sorted_arrays(arr1, arr2):
        """合并有序数组"""
        m, n = len(arr1), len(arr2)
        i, j, k = m - 1, n - 1, m + n - 1
        
        while i >= 0 and j >= 0:
            if arr1[i] > arr2[j]:
                arr1[k] = arr1[i]

            else:

def linked_list_problems():
    """链表问题"""
    
    class ListNode:
        """链表节点"""
        def __init__(self, val=0, next=None):
            self.val = val
            self.next = next
    
    class LinkedListTwoPointers:
        """链表双指针"""
        def __init__(self):
            self.results = []
        
        def find_cycle(self, head):
            """检测环"""
            slow = fast = head
            
            while fast and fast.next:
                slow = slow.next
                fast = fast.next.next
                
                if slow == fast:
                    return True
            
            return False
        
        def find_cycle_start(self, head):
            """找到环的起点"""
            slow = fast = head
            
            while fast and fast.next:
                slow = slow.next
                fast = fast.next.next
                
                if slow == fast:
                    break
            
            if not fast or not fast.next:
                return None
            
            slow = head
            while slow != fast:
                slow = slow.next
                fast = fast.next
            
            return slow
        
        def find_middle(self, head):
            """找到中间节点"""
            slow = fast = head
            
            while fast and fast.next:
                slow = slow.next
                fast = fast.next.next
            
            return slow
        
        def remove_nth_from_end(self, head, n):
            """删除倒数第n个节点"""
            dummy = ListNode(0)
            dummy.next = head
            
            slow = fast = dummy
            
            for _ in range(n + 1):
                fast = fast.next
            
            while fast:
                slow = slow.next
                fast = fast.next
            
            slow.next = slow.next.next
            
            return dummy.next
        
        def merge_sorted_lists(self, l1, l2):
            """合并有序链表"""
            dummy = ListNode(0)
            current = dummy
            
            while l1 and l2:
                if l1.val <= l2.val:
                    current.next = l1
                    l1 = l1.next
                else:
                    current.next = l2
                    l2 = l2.next
                current = current.next
            
            current.next = l1 or l2
            
            return dummy.next
        
        def reverse_list(self, head):
            """反转链表"""
            prev = None
            current = head
            
            while current:
                next_temp = current.next
                current.next = prev
                prev = current
                current = next_temp
            
            return prev
        
        def is_palindrome(self, head):
            """回文链表"""
            slow = fast = head
            
            while fast and fast.next:
                slow = slow.next
                fast = fast.next.next
            
            second_half = self.reverse_list(slow)
            first_half = head
            
            while second_half:
                if first_half.val != second_half.val:
                    return False
                first_half = first_half.next
                second_half = second_half.next
            
            return True
        
        def reorder_list(self, head):
            """重排链表"""
            if not head or not head.next:
                return
            
            slow = fast = head
            while fast and fast.next:
                slow = slow.next
                fast = fast.next.next
            
            second_half = self.reverse_list(slow.next)
            slow.next = None
            
            first_half = head
            while second_half:
                temp1 = first_half.next
                temp2 = second_half.next
                
                first_half.next = second_half
                second_half.next = temp1
                
                first_half = temp1
                second_half = temp2
    
    return LinkedListTwoPointers

                right -= 1

                s[left], s[right] = s[right], s[left]
                left += 1
                right -= 1
        
        def valid_palindrome(self, s):
            """验证回文串"""
            left, right = 0, len(s) - 1
            
            while left < right:
                while left < right and not s[left].isalnum():
                    left += 1
                while left < right and not s[right].isalnum():
                    right -= 1
                
                if s[left].lower() != s[right].lower():
                    return False
                
                left += 1
                right -= 1
            
            return True
        
        def two_sum_sorted(self, s, target):
            """有序字符串两数之和"""
            left, right = 0, len(s) - 1
            
            while left < right:
                current_sum = int(s[left]) + int(s[right])
                if current_sum == target:
                    return [left, right]
                elif current_sum < target:
                    left += 1
                else:
                    right -= 1
            
            return None
        
        def remove_duplicates(self, s):
            """删除重复字符"""
            if not s:
                return ""
            
            slow = 0
            for fast in range(1, len(s)):
                if s[fast] != s[slow]:
                    slow += 1
                    s[slow] = s[fast]
            
            return s[:slow + 1]
        
        def move_zeros(self, s):
            """移动零字符"""
            slow = 0
            
            for fast in range(len(s)):
                if s[fast] != '0':
                    s[slow], s[fast] = s[fast], s[slow]
                    slow += 1
        
        def sort_string(self, s):
            """字符串排序"""
            left, right = 0, len(s) - 1
            i = 0
            
            while i <= right:
                if s[i] == 'a':
                    s[left], s[i] = s[i], s[left]
                    left += 1
                    i += 1
                elif s[i] == 'c':
                    s[right], s[i] = s[i], s[right]
                    right -= 1
                else:
                    i += 1
        
        def partition_string(self, s, pivot):
            """分割字符串"""

            while left < right:
                while left < right and s[left] < pivot:
                    left += 1

                    right -= 1
                
                if left < right:
                    s[left], s[right] = s[right], s[left]
                    left += 1
                    right -= 1
            
            return left
