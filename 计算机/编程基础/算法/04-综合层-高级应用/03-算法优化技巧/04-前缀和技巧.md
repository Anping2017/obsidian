# 前缀和技巧

## 🎯 核心概念

**前缀和技巧**是一种高效的算法优化技术，通过预处理数组的前缀和来快速计算区间和。前缀和技巧通常能将O(n)的区间查询优化到O(1)，是算法优化的重要方法。

def prefix_sum_types():
    """前缀和类型"""
    
    def one_dimensional_prefix_sum(arr):
        """一维前缀和"""
        prefix = [0] * (len(arr) + 1)
        
        for i in range(len(arr)):
            prefix[i + 1] = prefix[i] + arr[i]
        
        return prefix
    
    def two_dimensional_prefix_sum(matrix):
        """二维前缀和"""
        m, n = len(matrix), len(matrix[0])
        prefix = [[0] * (n + 1) for _ in range(m + 1)]
        
        for i in range(m):
            for j in range(n):
                prefix[i + 1][j + 1] = prefix[i][j + 1] + prefix[i + 1][j] - prefix[i][j] + matrix[i][j]
        
        return prefix
    
    def range_sum_query(prefix, left, right):
        """区间和查询"""
        return prefix[right + 1] - prefix[left]
    
    def range_sum_query_2d(prefix, row1, col1, row2, col2):
        """二维区间和查询"""
        return prefix[row2 + 1][col2 + 1] - prefix[row1][col2 + 1] - prefix[row2 + 1][col1] + prefix[row1][col1]
    
    return one_dimensional_prefix_sum, two_dimensional_prefix_sum, range_sum_query, range_sum_query_2d

def prefix_sum_algorithms():
    """前缀和算法构建"""
    
    class PrefixSum:
        """前缀和类"""
        def __init__(self):
            self.prefix = []
            self.results = []
        
        def build_prefix_sum(self, arr):
            """构建前缀和"""
            self.prefix = [0] * (len(arr) + 1)
            
            for i in range(len(arr)):
                self.prefix[i + 1] = self.prefix[i] + arr[i]
        
        def range_sum(self, left, right):
            """区间和查询"""
            return self.prefix[right + 1] - self.prefix[left]
        
        def range_sum_count(self, arr, target):
            """区间和计数"""
            count = 0
            sum_count = {0: 1}
            current_sum = 0
            
            for num in arr:
                current_sum += num
                
                if current_sum - target in sum_count:
                    count += sum_count[current_sum - target]
                
                sum_count[current_sum] = sum_count.get(current_sum, 0) + 1
            
            return count
        
        def range_sum_closest(self, arr, target):
            """最接近目标的区间和"""
            prefix = [0] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = prefix[i] + arr[i]
            
            min_diff = float('inf')
            result = 0
            
            for i in range(len(arr)):
                for j in range(i, len(arr)):
                    current_sum = prefix[j + 1] - prefix[i]
                    diff = abs(current_sum - target)
                    
                    if diff < min_diff:
                        min_diff = diff
                        result = current_sum
            
            return result
        
        def range_sum_maximum(self, arr):
            """最大区间和"""
            prefix = [0] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = prefix[i] + arr[i]
            
            max_sum = arr[0]
            min_prefix = 0
            
            for i in range(len(arr)):
                max_sum = max(max_sum, prefix[i + 1] - min_prefix)
                min_prefix = min(min_prefix, prefix[i + 1])
            
            return max_sum
        
        def range_sum_minimum(self, arr):
            """最小区间和"""
            prefix = [0] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = prefix[i] + arr[i]
            
            min_sum = arr[0]
            max_prefix = 0
            
            for i in range(len(arr)):
                min_sum = min(min_sum, prefix[i + 1] - max_prefix)
                max_prefix = max(max_prefix, prefix[i + 1])
            
            return min_sum
        
        def range_sum_average(self, arr, k):
            """区间平均值"""
            prefix = [0] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = prefix[i] + arr[i]
            
            max_average = float('-inf')
            
            for i in range(len(arr) - k + 1):
                current_sum = prefix[i + k] - prefix[i]
                current_average = current_sum / k
                max_average = max(max_average, current_average)
            
            return max_average
        
        def range_sum_product(self, arr):
            """区间乘积"""
            prefix = [1] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = prefix[i] * arr[i]
            
            max_product = arr[0]
            
            for i in range(len(arr)):
                for j in range(i, len(arr)):
                    current_product = prefix[j + 1] // prefix[i]
                    max_product = max(max_product, current_product)
            
            return max_product
        
        def range_sum_xor(self, arr):
            """区间异或"""
            prefix = [0] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = prefix[i] ^ arr[i]
            
            max_xor = 0
            
            for i in range(len(arr)):
                for j in range(i, len(arr)):
                    current_xor = prefix[j + 1] ^ prefix[i]
                    max_xor = max(max_xor, current_xor)
            
            return max_xor
        
        def range_sum_gcd(self, arr):
            """区间最大公约数"""
            prefix = [0] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = self.gcd(prefix[i], arr[i])
            
            max_gcd = 0
            
            for i in range(len(arr)):
                for j in range(i, len(arr)):
                    current_gcd = self.gcd(prefix[j + 1], prefix[i])
                    max_gcd = max(max_gcd, current_gcd)
            
            return max_gcd
        
        def gcd(self, a, b):
            """最大公约数"""
            while b:
                a, b = b, a % b
            return a
        
        def range_sum_lcm(self, arr):
            """区间最小公倍数"""
            prefix = [1] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = self.lcm(prefix[i], arr[i])
            
            max_lcm = 0
            
            for i in range(len(arr)):
                for j in range(i, len(arr)):
                    current_lcm = self.lcm(prefix[j + 1], prefix[i])
                    max_lcm = max(max_lcm, current_lcm)
            
            return max_lcm
        
        def lcm(self, a, b):
            """最小公倍数"""
            return a * b // self.gcd(a, b)
    
    return PrefixSum

def two_dimensional_prefix_sum():
    """二维前缀和"""
    
    class TwoDimensionalPrefixSum:
        """二维前缀和"""
        def __init__(self):
            self.prefix = []
            self.results = []
        
        def build_prefix_sum(self, matrix):
            """构建二维前缀和"""
            m, n = len(matrix), len(matrix[0])
            self.prefix = [[0] * (n + 1) for _ in range(m + 1)]
            
            for i in range(m):
                for j in range(n):
                    self.prefix[i + 1][j + 1] = self.prefix[i][j + 1] + self.prefix[i + 1][j] - self.prefix[i][j] + matrix[i][j]
        
        def range_sum_query(self, row1, col1, row2, col2):
            """二维区间和查询"""
            return self.prefix[row2 + 1][col2 + 1] - self.prefix[row1][col2 + 1] - self.prefix[row2 + 1][col1] + self.prefix[row1][col1]
        
        def range_sum_count(self, matrix, target):
            """二维区间和计数"""
            count = 0
            sum_count = {0: 1}
            current_sum = 0
            
            for i in range(len(matrix)):
                for j in range(len(matrix[0])):
                    current_sum += matrix[i][j]
                    
                    if current_sum - target in sum_count:
                        count += sum_count[current_sum - target]
                    
                    sum_count[current_sum] = sum_count.get(current_sum, 0) + 1
            
            return count
        
        def range_sum_closest(self, matrix, target):
            """最接近目标的二维区间和"""
            min_diff = float('inf')
            result = 0
            
            for i in range(len(matrix)):
                for j in range(len(matrix[0])):
                    for k in range(i, len(matrix)):
                        for l in range(j, len(matrix[0])):
                            current_sum = self.range_sum_query(i, j, k, l)
                            diff = abs(current_sum - target)
                            
                            if diff < min_diff:
                                min_diff = diff
                                result = current_sum
            
            return result
        
        def range_sum_maximum(self, matrix):
            """最大二维区间和"""
            max_sum = matrix[0][0]
            
            for i in range(len(matrix)):
                for j in range(len(matrix[0])):
                    for k in range(i, len(matrix)):
                        for l in range(j, len(matrix[0])):
                            current_sum = self.range_sum_query(i, j, k, l)
                            max_sum = max(max_sum, current_sum)
            
            return max_sum
        
        def range_sum_minimum(self, matrix):
            """最小二维区间和"""
            min_sum = matrix[0][0]
            
            for i in range(len(matrix)):
                for j in range(len(matrix[0])):
                    for k in range(i, len(matrix)):
                        for l in range(j, len(matrix[0])):
                            current_sum = self.range_sum_query(i, j, k, l)
                            min_sum = min(min_sum, current_sum)
            
            return min_sum
        
        def range_sum_average(self, matrix, k):
            """二维区间平均值"""
            max_average = float('-inf')
            
            for i in range(len(matrix) - k + 1):
                for j in range(len(matrix[0]) - k + 1):
                    current_sum = self.range_sum_query(i, j, i + k - 1, j + k - 1)
                    current_average = current_sum / (k * k)
                    max_average = max(max_average, current_average)
            
            return max_average
        
        def range_sum_product(self, matrix):
            """二维区间乘积"""
            max_product = matrix[0][0]
            
            for i in range(len(matrix)):
                for j in range(len(matrix[0])):
                    for k in range(i, len(matrix)):
                        for l in range(j, len(matrix[0])):
                            current_product = 1
                            for m in range(i, k + 1):
                                for n in range(j, l + 1):
                                    current_product *= matrix[m][n]
                            max_product = max(max_product, current_product)
            
            return max_product
        
        def range_sum_xor(self, matrix):
            """二维区间异或"""
            max_xor = 0
            
            for i in range(len(matrix)):
                for j in range(len(matrix[0])):
                    for k in range(i, len(matrix)):
                        for l in range(j, len(matrix[0])):
                            current_xor = 0
                            for m in range(i, k + 1):
                                for n in range(j, l + 1):
                                    current_xor ^= matrix[m][n]
                            max_xor = max(max_xor, current_xor)
            
            return max_xor
        
        def range_sum_gcd(self, matrix):
            """二维区间最大公约数"""
            max_gcd = 0
            
            for i in range(len(matrix)):
                for j in range(len(matrix[0])):
                    for k in range(i, len(matrix)):
                        for l in range(j, len(matrix[0])):
                            current_gcd = matrix[i][j]
                            for m in range(i, k + 1):
                                for n in range(j, l + 1):
                                    current_gcd = self.gcd(current_gcd, matrix[m][n])
                            max_gcd = max(max_gcd, current_gcd)
            
            return max_gcd
        
        def gcd(self, a, b):
            """最大公约数"""
            while b:
                a, b = b, a % b
            return a
        
        def range_sum_lcm(self, matrix):
            """二维区间最小公倍数"""
            max_lcm = 0
            
            for i in range(len(matrix)):
                for j in range(len(matrix[0])):
                    for k in range(i, len(matrix)):
                        for l in range(j, len(matrix[0])):
                            current_lcm = matrix[i][j]
                            for m in range(i, k + 1):
                                for n in range(j, l + 1):
                                    current_lcm = self.lcm(current_lcm, matrix[m][n])
                            max_lcm = max(max_lcm, current_lcm)
            
            return max_lcm
        
        def lcm(self, a, b):
            """最小公倍数"""
            return a * b // self.gcd(a, b)
    
    return TwoDimensionalPrefixSum

def multi_dimensional_prefix_sum():

    class MultiDimensionalPrefixSum:

            self.prefix = []
            self.results = []
        
        def build_prefix_sum(self, arr, dimensions):
            """构建多维前缀和"""
            if len(dimensions) == 1:
                self.prefix = [0] * (dimensions[0] + 1)
                
                for i in range(dimensions[0]):
                    self.prefix[i + 1] = self.prefix[i] + arr[i]
            elif len(dimensions) == 2:
                m, n = dimensions[0], dimensions[1]
                self.prefix = [[0] * (n + 1) for _ in range(m + 1)]
                
                for i in range(m):
                    for j in range(n):
                        self.prefix[i + 1][j + 1] = self.prefix[i][j + 1] + self.prefix[i + 1][j] - self.prefix[i][j] + arr[i][j]
            elif len(dimensions) == 3:
                l, m, n = dimensions[0], dimensions[1], dimensions[2]
                self.prefix = [[[0] * (n + 1) for _ in range(m + 1)] for _ in range(l + 1)]
                
                for i in range(l):
                    for j in range(m):
                        for k in range(n):
                            self.prefix[i + 1][j + 1][k + 1] = self.prefix[i][j + 1][k + 1] + self.prefix[i + 1][j][k + 1] + self.prefix[i + 1][j + 1][k] - self.prefix[i][j][k + 1] - self.prefix[i][j + 1][k] - self.prefix[i + 1][j][k] + self.prefix[i][j][k] + arr[i][j][k]
        
        def range_sum_query(self, start, end):
            """多维区间和查询"""
            if len(start) == 1:
                return self.prefix[end[0] + 1] - self.prefix[start[0]]
            elif len(start) == 2:
                return self.prefix[end[0] + 1][end[1] + 1] - self.prefix[start[0]][end[1] + 1] - self.prefix[end[0] + 1][start[1]] + self.prefix[start[0]][start[1]]
            elif len(start) == 3:
                return self.prefix[end[0] + 1][end[1] + 1][end[2] + 1] - self.prefix[start[0]][end[1] + 1][end[2] + 1] - self.prefix[end[0] + 1][start[1]][end[2] + 1] - self.prefix[end[0] + 1][end[1] + 1][start[2]] + self.prefix[start[0]][start[1]][end[2] + 1] + self.prefix[start[0]][end[1] + 1][start[2]] + self.prefix[end[0] + 1][start[1]][start[2]] - self.prefix[start[0]][start[1]][start[2]]
        
        def range_sum_count(self, arr, target, dimensions):
            """多维区间和计数"""
            count = 0
            sum_count = {0: 1}
            current_sum = 0
            
            if len(dimensions) == 1:
                for i in range(dimensions[0]):
                    current_sum += arr[i]
                    
                    if current_sum - target in sum_count:
                        count += sum_count[current_sum - target]
                    
                    sum_count[current_sum] = sum_count.get(current_sum, 0) + 1
            elif len(dimensions) == 2:
                for i in range(dimensions[0]):
                    for j in range(dimensions[1]):
                        current_sum += arr[i][j]
                        
                        if current_sum - target in sum_count:
                            count += sum_count[current_sum - target]
                        
                        sum_count[current_sum] = sum_count.get(current_sum, 0) + 1
            elif len(dimensions) == 3:
                for i in range(dimensions[0]):
                    for j in range(dimensions[1]):
                        for k in range(dimensions[2]):
                            current_sum += arr[i][j][k]
                            
                            if current_sum - target in sum_count:
                                count += sum_count[current_sum - target]
                            
                            sum_count[current_sum] = sum_count.get(current_sum, 0) + 1
            
            return count
        
        def range_sum_closest(self, arr, target, dimensions):
            """最接近目标的多维区间和"""
            min_diff = float('inf')
            result = 0
            
            if len(dimensions) == 1:
                for i in range(dimensions[0]):
                    for j in range(i, dimensions[0]):
                        current_sum = self.range_sum_query([i], [j])
                        diff = abs(current_sum - target)
                        
                        if diff < min_diff:
                            min_diff = diff
                            result = current_sum
            elif len(dimensions) == 2:
                for i in range(dimensions[0]):
                    for j in range(dimensions[1]):
                        for k in range(i, dimensions[0]):
                            for l in range(j, dimensions[1]):
                                current_sum = self.range_sum_query([i, j], [k, l])
                                diff = abs(current_sum - target)
                                
                                if diff < min_diff:
                                    min_diff = diff
                                    result = current_sum
            elif len(dimensions) == 3:
                for i in range(dimensions[0]):
                    for j in range(dimensions[1]):
                        for k in range(dimensions[2]):
                            for l in range(i, dimensions[0]):
                                for m in range(j, dimensions[1]):
                                    for n in range(k, dimensions[2]):
                                        current_sum = self.range_sum_query([i, j, k], [l, m, n])
                                        diff = abs(current_sum - target)
                                        
                                        if diff < min_diff:
                                            min_diff = diff
                                            result = current_sum
            
            return result
        
        def range_sum_maximum(self, arr, dimensions):
            """最大多维区间和"""
            max_sum = arr[0] if len(dimensions) == 1 else arr[0][0] if len(dimensions) == 2 else arr[0][0][0]
            
            if len(dimensions) == 1:
                for i in range(dimensions[0]):
                    for j in range(i, dimensions[0]):
                        current_sum = self.range_sum_query([i], [j])
                        max_sum = max(max_sum, current_sum)
            elif len(dimensions) == 2:
                for i in range(dimensions[0]):
                    for j in range(dimensions[1]):
                        for k in range(i, dimensions[0]):
                            for l in range(j, dimensions[1]):
                                current_sum = self.range_sum_query([i, j], [k, l])
                                max_sum = max(max_sum, current_sum)
            elif len(dimensions) == 3:
                for i in range(dimensions[0]):
                    for j in range(dimensions[1]):
                        for k in range(dimensions[2]):
                            for l in range(i, dimensions[0]):
                                for m in range(j, dimensions[1]):
                                    for n in range(k, dimensions[2]):
                                        current_sum = self.range_sum_query([i, j, k], [l, m, n])
                                        max_sum = max(max_sum, current_sum)
            
            return max_sum
        
        def range_sum_minimum(self, arr, dimensions):
            """最小多维区间和"""
            min_sum = arr[0] if len(dimensions) == 1 else arr[0][0] if len(dimensions) == 2 else arr[0][0][0]
            
            if len(dimensions) == 1:
                for i in range(dimensions[0]):
                    for j in range(i, dimensions[0]):
                        current_sum = self.range_sum_query([i], [j])
                        min_sum = min(min_sum, current_sum)
            elif len(dimensions) == 2:
                for i in range(dimensions[0]):
                    for j in range(dimensions[1]):
                        for k in range(i, dimensions[0]):
                            for l in range(j, dimensions[1]):
                                current_sum = self.range_sum_query([i, j], [k, l])
