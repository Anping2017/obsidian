## 🎯 核心概念

**位运算技巧**是一种高效的算法优化技术，通过直接操作二进制位来解决问题。位运算技巧通常能将O(n)的操作优化到O(1)，是算法优化的重要方法。

def bit_manipulation_algorithms():
    """位运算算法构建"""
    
    class BitManipulation:
        """位运算类"""
        def __init__(self):
            self.results = []
        
        def is_power_of_two(self, n):
            """检查是否为2的幂"""
            return n > 0 and n & (n - 1) == 0
        
        def get_lowest_bit(self, n):
            """获取最低位的1"""
            return n & (-n)
        
        def remove_lowest_bit(self, n):
            """移除最低位的1"""
            return n & (n - 1)
        
        def set_bit(self, n, i):
            """设置第i位为1"""
            return n | (1 << i)
        
        def clear_bit(self, n, i):
            """清除第i位"""
            return n & ~(1 << i)
        
        def toggle_bit(self, n, i):
            """切换第i位"""
            return n ^ (1 << i)
        
        def get_bit(self, n, i):
            """获取第i位"""
            return (n >> i) & 1
        
        def count_bits(self, n):
            """计算1的个数"""
            count = 0
            while n:
                count += 1
                n &= n - 1
            return count
        
        def reverse_bits(self, n):
            """反转位"""
            reversed_n = 0
            for _ in range(32):
                reversed_n = (reversed_n << 1) | (n & 1)
                n >>= 1
            return reversed_n
        
        def single_number(self, nums):
            """只出现一次的数字"""
            result = 0
            for num in nums:
                result ^= num
            return result
        
        def missing_number(self, nums):
            """缺失的数字"""
            n = len(nums)
            expected_sum = n * (n + 1) // 2
            actual_sum = sum(nums)
            return expected_sum - actual_sum
        
        def hamming_distance(self, x, y):
            """汉明距离"""
            xor = x ^ y
            distance = 0
            while xor:
                distance += 1
                xor &= xor - 1
            return distance
        
        def total_hamming_distance(self, nums):
            """总汉明距离"""
            total = 0
            for i in range(32):
                count = 0
                for num in nums:
                    count += (num >> i) & 1
                total += count * (len(nums) - count)
            return total
        
        def find_complement(self, num):
            """数字的补数"""
            mask = 1
            while mask < num:
                mask = (mask << 1) + 1
            return mask ^ num
        
        def is_power_of_four(self, n):
            """检查是否为4的幂"""
            return n > 0 and n & (n - 1) == 0 and n & 0xAAAAAAAA == 0
        
        def is_power_of_three(self, n):
            """检查是否为3的幂"""
            if n <= 0:
                return False
            while n % 3 == 0:
                n //= 3
            return n == 1
        
        def range_bitwise_and(self, m, n):
            """区间按位与"""
            shift = 0
            while m < n:
                m >>= 1
                n >>= 1
                shift += 1
            return m << shift
        
        def get_sum(self, a, b):
            """不使用加号的加法"""
            while b:
                carry = a & b
                a = a ^ b
                b = carry << 1
            return a
        
        def get_difference(self, a, b):
            """不使用减号的减法"""
            while b:
                borrow = (~a) & b
                a = a ^ b
                b = borrow << 1
            return a
        
        def get_product(self, a, b):
            """不使用乘号的乘法"""
            result = 0
            while b:
                if b & 1:
                    result += a
                a <<= 1
                b >>= 1
            return result
        
        def get_quotient(self, dividend, divisor):
            """不使用除号的除法"""
            if dividend == 0:
                return 0
            if divisor == 1:
                return dividend
            if divisor == -1:
                return -dividend
            
            sign = -1 if (dividend < 0) ^ (divisor < 0) else 1
            dividend = abs(dividend)
            divisor = abs(divisor)
            
            result = 0
            while dividend >= divisor:
                temp = divisor
                multiple = 1
                while dividend >= (temp << 1):
                    temp <<= 1
                    multiple <<= 1
                dividend -= temp
                result += multiple
            
            return sign * result
    
    return BitManipulation

def state_compression():
    """状态压缩"""
    
    class StateCompression:
        """状态压缩类"""
        def __init__(self):
            self.results = []
        
        def subset_sum(self, nums, target):
            """子集和问题"""
            n = len(nums)
            for mask in range(1 << n):
                current_sum = 0
                for i in range(n):
                    if mask & (1 << i):
                        current_sum += nums[i]
                if current_sum == target:
                    return True
            return False
        
        def subset_count(self, nums, target):
            """子集计数"""
            n = len(nums)
            count = 0
            for mask in range(1 << n):
                current_sum = 0
                for i in range(n):
                    if mask & (1 << i):
                        current_sum += nums[i]
                if current_sum == target:
                    count += 1
            return count
        
        def subset_max_sum(self, nums):
            """子集最大和"""
            n = len(nums)
            max_sum = float('-inf')
            for mask in range(1 << n):
                current_sum = 0
                for i in range(n):
                    if mask & (1 << i):
                        current_sum += nums[i]
                max_sum = max(max_sum, current_sum)
            return max_sum
        
        def subset_min_sum(self, nums):
            """子集最小和"""
            n = len(nums)
            min_sum = float('inf')
            for mask in range(1 << n):
                current_sum = 0
                for i in range(n):
                    if mask & (1 << i):
                        current_sum += nums[i]
                min_sum = min(min_sum, current_sum)
            return min_sum
        
        def subset_product(self, nums):
            """子集乘积"""
            n = len(nums)
            max_product = float('-inf')
            for mask in range(1 << n):
                current_product = 1
                for i in range(n):
                    if mask & (1 << i):
                        current_product *= nums[i]
                max_product = max(max_product, current_product)
            return max_product
        
        def subset_xor(self, nums):
            """子集异或"""
            n = len(nums)
            max_xor = 0
            for mask in range(1 << n):
                current_xor = 0
                for i in range(n):
                    if mask & (1 << i):
                        current_xor ^= nums[i]
                max_xor = max(max_xor, current_xor)
            return max_xor
        
        def subset_gcd(self, nums):
            """子集最大公约数"""
            n = len(nums)
            max_gcd = 0
            for mask in range(1 << n):
                current_gcd = 0
                for i in range(n):
                    if mask & (1 << i):
                        current_gcd = self.gcd(current_gcd, nums[i])
                max_gcd = max(max_gcd, current_gcd)
            return max_gcd
        
        def gcd(self, a, b):
            """最大公约数"""
            while b:
                a, b = b, a % b
            return a
        
        def subset_lcm(self, nums):

            """最小公倍数"""
            return a * b // self.gcd(a, b)

def bit_manipulation_optimization():
    """位运算优化"""
    
    class BitManipulationOptimization:
        """位运算优化类"""
        def __init__(self):
            self.results = []
        
        def fast_power(self, base, exp):
            """快速幂"""
            result = 1
            while exp > 0:
                if exp & 1:
                    result *= base
                base *= base
                exp >>= 1
            return result
        
        def fast_multiply(self, a, b):
            """快速乘法"""
            result = 0
            while b:
                if b & 1:
                    result += a
                a <<= 1
                b >>= 1
            return result
        
        def fast_divide(self, dividend, divisor):
            """快速除法"""
            if dividend == 0:
                return 0
            if divisor == 1:
                return dividend
            if divisor == -1:
                return -dividend
            
            sign = -1 if (dividend < 0) ^ (divisor < 0) else 1
            dividend = abs(dividend)
            divisor = abs(divisor)
            
            result = 0
            while dividend >= divisor:
                temp = divisor
                multiple = 1
                while dividend >= (temp << 1):
                    temp <<= 1
                    multiple <<= 1
                dividend -= temp
                result += multiple
            
            return sign * result
        
        def fast_mod(self, a, b):
            """快速取模"""
            return a & (b - 1)
        
        def fast_abs(self, n):
            """快速绝对值"""
            mask = n >> 31
            return (n + mask) ^ mask
        
        def fast_max(self, a, b):
            """快速最大值"""
            return a ^ ((a ^ b) & -(a < b))
        
        def fast_min(self, a, b):
            """快速最小值"""
            return b ^ ((a ^ b) & -(a < b))
        
        def fast_swap(self, a, b):
            """快速交换"""
            a ^= b
            b ^= a
            a ^= b
            return a, b
        
        def fast_negate(self, n):
            """快速取反"""
            return ~n + 1
        
        def fast_abs_diff(self, a, b):
            """快速绝对差"""
            return (a ^ b) - (a < b)
        
        def fast_sign(self, n):
            """快速符号"""
            return (n > 0) - (n < 0)
        
        def fast_parity(self, n):
            """快速奇偶性"""
            n ^= n >> 16
            n ^= n >> 8
            n ^= n >> 4
            n ^= n >> 2
            n ^= n >> 1
            return n & 1
        
        def fast_log2(self, n):
            """快速对数"""
            if n == 0:
                return -1
            result = 0
            if n & 0xFFFF0000:
                result += 16
                n >>= 16
            if n & 0xFF00:
                result += 8
                n >>= 8
            if n & 0xF0:
                result += 4
                n >>= 4
            if n & 0xC:
                result += 2
                n >>= 2
            if n & 0x2:
                result += 1
            return result
        
        def fast_sqrt(self, n):
            """快速平方根"""
            if n < 2:
                return n
            
            result = 0
            bit = 1 << 15
            while bit:
                if (result + bit) * (result + bit) <= n:
                    result += bit
                bit >>= 1
            return result

def bit_manipulation_applications():

            result = []

        def hamming_weight(self, n):
            """汉明重量"""
            count = 0
            while n:

                n &= n - 1
            return count
        
        def reverse_bits(self, n):
            """反转位"""
            result = 0
            for _ in range(32):
                result = (result << 1) | (n & 1)
                n >>= 1
            return result
        
        def number_of_1_bits(self, n):
            """1的个数"""
            count = 0
            while n:
                count += n & 1
                n >>= 1
            return count
        
        def counting_bits(self, n):
            """计数位"""
            result = [0] * (n + 1)
            for i in range(1, n + 1):
                result[i] = result[i & (i - 1)] + 1
            return result
        
        def single_number_ii(self, nums):
            """只出现一次的数字II"""
            ones = twos = 0
            for num in nums:
                ones = (ones ^ num) & ~twos
                twos = (twos ^ num) & ~ones
            return ones
        
        def single_number_iii(self, nums):
            """只出现一次的数字III"""
            xor = 0
            for num in nums:
                xor ^= num
            
            diff = xor & -xor
            
            x = 0
            for num in nums:
                if num & diff:
                    x ^= num
            
            return [x, xor ^ x]
        
        def maximum_xor_subarray(self, nums):
            """最大异或子数组"""
            max_xor = 0
            for i in range(len(nums)):
                current_xor = 0
                for j in range(i, len(nums)):
                    current_xor ^= nums[j]
                    max_xor = max(max_xor, current_xor)
            return max_xor
        
        def maximum_xor_two_numbers(self, nums):
            """最大异或两个数"""
            max_xor = 0
            for i in range(len(nums)):
                for j in range(i + 1, len(nums)):
                    max_xor = max(max_xor, nums[i] ^ nums[j])
            return max_xor
        
        def find_maximum_xor(self, nums):
            """找到最大异或"""
            max_xor = 0
            mask = 0
            
            for i in range(31, -1, -1):
                mask |= (1 << i)
                prefixes = set()
                for num in nums:
                    prefixes.add(num & mask)
                
                temp = max_xor | (1 << i)
                for prefix in prefixes:
                    if temp ^ prefix in prefixes:
                        max_xor = temp
                        break
            
            return max_xor
        
        def find_duplicate(self, nums):
            """找到重复数字"""
            slow = fast = nums[0]
            
            while True:
                slow = nums[slow]
                fast = nums[nums[fast]]
                if slow == fast:
                    break
            
            slow = nums[0]
            while slow != fast:
                slow = nums[slow]
                fast = nums[fast]
            
            return slow
        
        def find_missing_positive(self, nums):
            """找到缺失的正数"""
            n = len(nums)
            
            for i in range(n):
                while 1 <= nums[i] <= n and nums[nums[i] - 1] != nums[i]:
                    nums[nums[i] - 1], nums[i] = nums[i], nums[nums[i] - 1]
            
            for i in range(n):
                if nums[i] != i + 1:
                    return i + 1
            
            return n + 1
        
        def find_first_missing_positive(self, nums):
            """找到第一个缺失的正数"""
            n = len(nums)
            
            for i in range(n):
                if nums[i] <= 0 or nums[i] > n:
                    nums[i] = n + 1
            
            for i in range(n):
                num = abs(nums[i])
                if num <= n:
                    nums[num - 1] = -abs(nums[num - 1])
            
            for i in range(n):
                if nums[i] > 0:
                    return i + 1
            
            return n + 1
        
        def find_all_duplicates(self, nums):
            """找到所有重复数字"""
            result = []
            for num in nums:
                index = abs(num) - 1
                if nums[index] < 0:
                    result.append(abs(num))
                else:
                    nums[index] = -nums[index]
            return result

                index = abs(num) - 1
                if nums[index] > 0:

                if slow == fast:
                    break
            
            slow = nums[0]
            while slow != fast:
                slow = nums[slow]
                fast = nums[fast]
            
            return slow
        
        def find_duplicate_numbers(self, nums):
            """找到重复数字"""

            for num in nums:
                index = abs(num) - 1
                if nums[index] < 0:
                    result.append(abs(num))
                else:
                    nums[index] = -nums[index]
            return result
        
        def find_missing_number(self, nums):
            """找到缺失数字"""
            n = len(nums)
            expected_sum = n * (n + 1) // 2
            actual_sum = sum(nums)
            return expected_sum - actual_sum
