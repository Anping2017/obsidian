def optimization_basics():
    """优化基础"""
    
    def two_pointers(arr, target):
        """双指针技巧"""
        left, right = 0, len(arr) - 1
        
        while left < right:
            current_sum = arr[left] + arr[right]
            if current_sum == target:
                return [left, right]
            elif current_sum < target:
                left += 1
            else:
                right -= 1
        
        return None
    
    def sliding_window(arr, k):
        """滑动窗口技巧"""
        if len(arr) < k:
            return []
        
        window_sum = sum(arr[:k])
        max_sum = window_sum
        
        for i in range(k, len(arr)):
            window_sum = window_sum - arr[i - k] + arr[i]
            max_sum = max(max_sum, window_sum)
        
        return max_sum
    
    def prefix_sum(arr):
        """前缀和技巧"""
        prefix = [0] * (len(arr) + 1)
        
        for i in range(len(arr)):
            prefix[i + 1] = prefix[i] + arr[i]
        
        return prefix
    
    def bit_manipulation(n):
        """位运算技巧"""
        # 检查是否为2的幂
        is_power_of_two = n & (n - 1) == 0
        
        # 获取最低位的1
        lowest_bit = n & (-n)
        
        # 移除最低位的1
        remove_lowest_bit = n & (n - 1)
        
        # 设置第i位为1
        def set_bit(n, i):
            return n | (1 << i)
        
        # 清除第i位
        def clear_bit(n, i):
            return n & ~(1 << i)
        
        # 切换第i位
        def toggle_bit(n, i):
            return n ^ (1 << i)
        
        return {
            "is_power_of_two": is_power_of_two,
            "lowest_bit": lowest_bit,
            "remove_lowest_bit": remove_lowest_bit,
            "set_bit": set_bit,
            "clear_bit": clear_bit,
            "toggle_bit": toggle_bit
        }
    
    def memoization(func):
        """记忆化技巧"""
        cache = {}
        
        def wrapper(*args):
            if args in cache:
                return cache[args]
            
            result = func(*args)
            cache[args] = result
            return result
        
        return wrapper
    
    def space_optimization(arr):
        """空间优化技巧"""
        # 原地修改
        for i in range(len(arr)):
            arr[i] *= 2
        
        # 滚动数组
        prev, curr = 0, 1
        for i in range(2, len(arr)):
            next_val = prev + curr
            prev, curr = curr, next_val
        
        return curr
    
    return two_pointers, sliding_window, prefix_sum, bit_manipulation, memoization, space_optimization

def optimization_algorithms():
    """优化算法构建"""
    
    class OptimizationTechniques:
        """优化技巧类"""
        def __init__(self):
            self.cache = {}
        
        def two_pointers_sum(self, arr, target):
            """双指针求和"""
            left, right = 0, len(arr) - 1
            
            while left < right:
                current_sum = arr[left] + arr[right]
                if current_sum == target:
                    return [left, right]
                elif current_sum < target:
                    left += 1
                else:
                    right -= 1
            
            return None
        
        def sliding_window_max(self, arr, k):
            """滑动窗口最大值"""
            if len(arr) < k:
                return []
            
            window_sum = sum(arr[:k])
            max_sum = window_sum
            
            for i in range(k, len(arr)):
                window_sum = window_sum - arr[i - k] + arr[i]
                max_sum = max(max_sum, window_sum)
            
            return max_sum
        
        def prefix_sum_range(self, arr, left, right):
            """前缀和区间查询"""
            prefix = [0] * (len(arr) + 1)
            
            for i in range(len(arr)):
                prefix[i + 1] = prefix[i] + arr[i]
            
            return prefix[right + 1] - prefix[left]
        
        def bit_operations(self, n):
            """位运算操作"""
            # 计算1的个数
            count = 0
            while n:
                count += 1
                n &= n - 1
            
            # 反转位
            reversed_n = 0
            for _ in range(32):
                reversed_n = (reversed_n << 1) | (n & 1)
                n >>= 1
            
            return count, reversed_n
        
        def memoized_fibonacci(self, n):
            """记忆化斐波那契"""
            if n in self.cache:
                return self.cache[n]
            
            if n <= 1:
                return n
            
            result = self.memoized_fibonacci(n - 1) + self.memoized_fibonacci(n - 2)
            self.cache[n] = result
            return result
        
        def space_optimized_fibonacci(self, n):
            """空间优化斐波那契"""
            if n <= 1:
                return n
            
            prev, curr = 0, 1
            for i in range(2, n + 1):
                prev, curr = curr, prev + curr
            
            return curr
        
        def in_place_reverse(self, arr):
            """原地反转"""
            left, right = 0, len(arr) - 1
            
            while left < right:
                arr[left], arr[right] = arr[right], arr[left]
                left += 1
                right -= 1
            
            return arr
        
        def rolling_array_dp(self, n):
            """滚动数组动态规划"""
            if n <= 1:
                return n
            
            prev, curr = 0, 1
            for i in range(2, n + 1):
                prev, curr = curr, prev + curr
            
            return curr
        
        def binary_search_optimization(self, arr, target):
            """二分查找优化"""
            left, right = 0, len(arr) - 1
            
            while left <= right:
                mid = left + (right - left) // 2  # 避免溢出
                
                if arr[mid] == target:
                    return mid
                elif arr[mid] < target:
                    left = mid + 1
                else:
                    right = mid - 1
            
            return -1
        
        def fast_power(self, base, exp):
            """快速幂"""
            result = 1
            
            while exp > 0:
                if exp % 2 == 1:
                    result *= base
                base *= base
                exp //= 2
            
            return result
        
        def matrix_exponentiation(self, matrix, n):
            """矩阵快速幂"""
            if n == 1:
                return matrix
            
            if n % 2 == 0:
                half = self.matrix_exponentiation(matrix, n // 2)
                return self.matrix_multiply(half, half)
            else:
                return self.matrix_multiply(matrix, self.matrix_exponentiation(matrix, n - 1))
        
        def matrix_multiply(self, A, B):
            """矩阵乘法"""
            n = len(A)
            result = [[0] * n for _ in range(n)]
            
            for i in range(n):
                for j in range(n):
                    for k in range(n):
                        result[i][j] += A[i][k] * B[k][j]
            
            return result
    
    return OptimizationTechniques

def performance_optimization():
    """性能优化"""
    
    class PerformanceOptimizer:
        """性能优化器"""
        def __init__(self):
            self.cache = {}
            self.metrics = {}
        
        def optimize_time_complexity(self, func):
            """时间复杂度优化"""
            def wrapper(*args, **kwargs):
                start_time = time.time()
                result = func(*args, **kwargs)
                end_time = time.time()
                
                self.metrics[func.__name__] = {
                    "execution_time": end_time - start_time,
                    "calls": self.metrics.get(func.__name__, {}).get("calls", 0) + 1
                }
                
                return result
            
            return wrapper
        
        def optimize_space_complexity(self, func):
            """空间复杂度优化"""
            def wrapper(*args, **kwargs):
                import psutil
                import os
                
                process = psutil.Process(os.getpid())
                start_memory = process.memory_info().rss
                
                result = func(*args, **kwargs)
                
                end_memory = process.memory_info().rss
                memory_used = end_memory - start_memory
                
                self.metrics[func.__name__] = {
                    "memory_used": memory_used,
                    "calls": self.metrics.get(func.__name__, {}).get("calls", 0) + 1
                }
                
                return result
            
            return wrapper
        
        def cache_optimization(self, func):
            """缓存优化"""
            def wrapper(*args, **kwargs):
                cache_key = str(args) + str(kwargs)
                
                if cache_key in self.cache:
                    return self.cache[cache_key]
                
                result = func(*args, **kwargs)
                self.cache[cache_key] = result
                
                return result
            
            return wrapper
        
        def parallel_optimization(self, func):
            """并行优化"""
            def wrapper(*args, **kwargs):
                import concurrent.futures
                
                with concurrent.futures.ThreadPoolExecutor() as executor:
                    future = executor.submit(func, *args, **kwargs)
                    result = future.result()
                
                return result
            
            return wrapper
        
        def get_metrics(self):
            """获取性能指标"""
            return self.metrics
    
    class AlgorithmProfiler:
        """算法性能分析器"""
        def __init__(self):
            self.profiles = {}
        
        def profile_algorithm(self, func, *args, **kwargs):
            """分析算法性能"""
            import time
            import psutil
            import os
            
            process = psutil.Process(os.getpid())
            
            # 时间分析
            start_time = time.time()
            start_memory = process.memory_info().rss
            
            result = func(*args, **kwargs)
            
            end_time = time.time()
            end_memory = process.memory_info().rss
            
            # 性能指标
            execution_time = end_time - start_time
            memory_used = end_memory - start_memory
            
            self.profiles[func.__name__] = {
                "execution_time": execution_time,
                "memory_used": memory_used,
                "result": result
            }
            
            return result
        
        def compare_algorithms(self, algorithms, *args, **kwargs):
            """比较算法性能"""
            results = {}
            
            for name, func in algorithms.items():
                result = self.profile_algorithm(func, *args, **kwargs)
                results[name] = self.profiles[func.__name__]
            
            return results
    
    return PerformanceOptimizer, AlgorithmProfiler
