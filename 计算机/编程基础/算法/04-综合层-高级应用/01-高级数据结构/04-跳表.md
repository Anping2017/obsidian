# 跳表

## 🎯 核心概念

**跳表(Skip List)**是一种随机化的数据结构，可以看作是多个有序链表的组合。跳表通过概率性地在多个层级上建立索引，实现O(log n)的查找、插入和删除操作，在Redis、LevelDB等系统中被广泛使用。

### 2. 跳表构建
```python
def skip_list_construction():
    """跳表构建"""
    
    import random
    
    class SkipListNode:
        """跳表节点"""
        def __init__(self, val, level):
            self.val = val
            self.forward = [None] * (level + 1)
            self.level = level
    
    class SkipList:
        """跳表"""
        def __init__(self, max_level=16, probability=0.5):
            self.max_level = max_level
            self.probability = probability
            self.level = 0
            self.header = SkipListNode(-1, max_level)
        
        def random_level(self):
            """随机生成层级"""
            level = 0
            while random.random() < self.probability and level < self.max_level:
                level += 1
            return level
        
        def search(self, val):
            """搜索节点"""
            current = self.header
            
            # 从最高层开始搜索
            for i in range(self.level, -1, -1):
                while current.forward[i] and current.forward[i].val < val:
                    current = current.forward[i]
            
            # 移动到下一层
            current = current.forward[0]
            
            if current and current.val == val:
                return current
            return None
        
        def insert(self, val):
            """插入节点"""
            update = [None] * (self.max_level + 1)
            current = self.header
            
            # 找到插入位置
            for i in range(self.level, -1, -1):
                while current.forward[i] and current.forward[i].val < val:
                    current = current.forward[i]
                update[i] = current
            
            current = current.forward[0]
            
            # 如果节点已存在，更新值
            if current and current.val == val:
                current.val = val
                return
            
            # 生成新节点的层级
            new_level = self.random_level()
            
            # 如果新层级大于当前层级，更新header
            if new_level > self.level:
                for i in range(self.level + 1, new_level + 1):
                    update[i] = self.header
                self.level = new_level
            
            # 创建新节点
            new_node = SkipListNode(val, new_level)
            
            # 更新指针
            for i in range(new_level + 1):
                new_node.forward[i] = update[i].forward[i]
                update[i].forward[i] = new_node
        
        def delete(self, val):
            """删除节点"""
            update = [None] * (self.max_level + 1)
            current = self.header
            
            # 找到删除位置
            for i in range(self.level, -1, -1):
                while current.forward[i] and current.forward[i].val < val:
                    current = current.forward[i]
                update[i] = current
            
            current = current.forward[0]
            
            # 如果节点存在，删除它
            if current and current.val == val:
                for i in range(self.level + 1):
                    if update[i].forward[i] != current:
                        break
                    update[i].forward[i] = current.forward[i]
                
                # 更新层级
                while self.level > 0 and self.header.forward[self.level] is None:
                    self.level -= 1
                
                return True
            return False
        
        def display(self):
            """显示跳表"""
            for i in range(self.level, -1, -1):
                current = self.header.forward[i]
                print(f"Level {i}: ", end="")
                while current:
                    print(f"{current.val} -> ", end="")
                    current = current.forward[i]
                print("None")
    
    return SkipList
```

```python
    def top_k_elements(skip_list, k):
        """Top K元素"""
        result = []
        current = skip_list.header.forward[0]
        
        while current and len(result) < k:
            for _ in range(current.count):
                if len(result) >= k:
                    break
                result.append(current.val)
            current = current.forward[0]
        
        return result
    
    def rank_query(skip_list, val):
        """排名查询"""
        rank = 0
        current = skip_list.header
        
        for i in range(skip_list.level, -1, -1):
            while current.forward[i] and current.forward[i].val < val:
                rank += current.forward[i].count
                current = current.forward[i]
        
        return rank
    
    def percentile_query(skip_list, percentile):
        """百分位数查询"""
        if percentile < 0 or percentile > 100:
            return None
        
        target_rank = int(skip_list.size * percentile / 100)
        current = skip_list.header.forward[0]
        current_rank = 0
        
        while current:
            if current_rank + current.count > target_rank:
                return current.val
            current_rank += current.count
            current = current.forward[0]
        
        return None
    
    return sorted_set_operations, top_k_elements, rank_query, percentile_query
```

```python
def range_query_applications():
    """范围查询应用"""
    
    def time_series_query(skip_list, start_time, end_time):
        """时间序列查询"""
        result = []
        current = skip_list.header
        
        for i in range(skip_list.level, -1, -1):
            while current.forward[i] and current.forward[i].val < start_time:
                current = current.forward[i]
        
        current = current.forward[0]
        
        while current and current.val <= end_time:
            for _ in range(current.count):
                result.append(current.val)
            current = current.forward[0]
        
        return result
    
    def interval_intersection(skip_list, intervals):
        """区间交集"""
        result = []
        
        for start, end in intervals:
            intersection = skip_list.range_query(start, end)
            if intersection:
                result.append((start, end, intersection))
        
        return result
    
    def nearest_neighbor_query(skip_list, val, k):
        """最近邻查询"""
        result = []
        current = skip_list.header
        
        # 找到插入位置
        for i in range(skip_list.level, -1, -1):
            while current.forward[i] and current.forward[i].val < val:
                current = current.forward[i]
        
        # 向前和向后搜索
        left = current
        right = current.forward[0]
        
        while len(result) < k and (left != skip_list.header or right):
            if left != skip_list.header and (not right or abs(left.val - val) <= abs(right.val - val)):
                for _ in range(left.count):
                    if len(result) >= k:
                        break
                    result.append(left.val)
                left = left.forward[0]
            elif right:
                for _ in range(right.count):
                    if len(result) >= k:
                        break
                    result.append(right.val)
                right = right.forward[0]
        
        return result
    
    def sliding_window_query(skip_list, window_size):
        """滑动窗口查询"""
        result = []
        current = skip_list.header.forward[0]
        
        while current:
            window_start = current.val
            window_end = window_start + window_size
            
            window_data = skip_list.range_query(window_start, window_end)
            if window_data:
                result.append((window_start, window_end, window_data))
            
            current = current.forward[0]
        
        return result
    
    return time_series_query, interval_intersection, nearest_neighbor_query, sliding_window_query
```

```python
def concurrent_applications():

        def __init__(self, max_level=16):

            self.header = AdvancedSkipListNode(-1, max_level)
            self.lock = threading.RLock()
            self.size = 0
        
        def search(self, val):
            """搜索节点"""
            with self.lock:
                current = self.header
                
                for i in range(self.level, -1, -1):
                    while current.forward[i] and current.forward[i].val < val:
                        current = current.forward[i]
                
                current = current.forward[0]
                
                if current and current.val == val:
                    return current
                return None
        
        def insert(self, val):
            """插入节点"""
            with self.lock:
                update = [None] * (self.max_level + 1)
                current = self.header
                
                for i in range(self.level, -1, -1):
                    while current.forward[i] and current.forward[i].val < val:
                        current = current.forward[i]
                    update[i] = current
                
                current = current.forward[0]
                
                if current and current.val == val:
                    current.count += 1

                if new_level > self.level:
                    for i in range(self.level + 1, new_level + 1):
                        update[i] = self.header
                    self.level = new_level

                for i in range(new_level + 1):
                    new_node.forward[i] = update[i].forward[i]

                self.size += 1
```
