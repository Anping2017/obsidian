# 树状数组

## 🎯 核心概念

**树状数组(Binary Indexed Tree, BIT)**是一种支持单点更新和区间查询的数据结构。树状数组利用二进制表示的特性，通过巧妙的索引设计实现高效的区间操作，在动态数组、逆序对统计、区间和查询等领域有重要应用。

### 2. 树状数组构建
```python
def bit_construction():
    """树状数组构建"""
    
    class BinaryIndexedTree:
        """树状数组"""
        def __init__(self, arr):
            self.n = len(arr)
            self.tree = [0] * (self.n + 1)
            self.build(arr)
        
        def build(self, arr):
            """构建树状数组"""
            for i in range(self.n):
                self.update(i + 1, arr[i])
        
        def lowbit(self, x):
            """计算lowbit"""
            return x & (-x)
        
        def update(self, idx, val):
            """单点更新"""
            while idx <= self.n:
                self.tree[idx] += val
                idx += self.lowbit(idx)
        
        def query(self, idx):
            """前缀查询"""
            result = 0
            while idx > 0:
                result += self.tree[idx]
                idx -= self.lowbit(idx)
            return result
        
        def range_query(self, l, r):
            """区间查询"""
            return self.query(r) - self.query(l - 1)
    
    return BinaryIndexedTree

def bit_examples():

        return result
    
    def bit_example_3():
        """示例3: 逆序对统计"""
        arr = [5, 2, 6, 1, 3]
        bit = BinaryIndexedTree([0] * len(arr))
        
        # 离散化
        sorted_arr = sorted(set(arr))
        mapping = {val: i + 1 for i, val in enumerate(sorted_arr)}
        
        # 统计逆序对
        inversions = 0
        for val in reversed(arr):
            idx = mapping[val]
            inversions += bit.query(idx - 1)
            bit.update(idx, 1)
        
        print(f"数组: {arr}")
        print(f"逆序对数量: {inversions}")
        
        return inversions
    
    return bit_example_1, bit_example_2, bit_example_3
```

### 2. 逆序对统计
```python
def inversion_count_applications():
    """逆序对统计应用"""
    
    def count_inversions(arr):
        """统计逆序对"""
        n = len(arr)
        
        # 离散化
        sorted_arr = sorted(set(arr))
        mapping = {val: i + 1 for i, val in enumerate(sorted_arr)}
        
        # 树状数组
        bit = BinaryIndexedTree([0] * len(sorted_arr))
        
        # 统计逆序对
        inversions = 0
        for val in reversed(arr):
            idx = mapping[val]
            inversions += bit.query(idx - 1)
            bit.update(idx, 1)
        
        return inversions
    
    def count_inversions_2d(points):
        """二维逆序对统计"""
        n = len(points)
        
        # 按x坐标排序
        points.sort()
        
        # 离散化y坐标
        y_coords = [point[1] for point in points]
        sorted_y = sorted(set(y_coords))
        mapping = {val: i + 1 for i, val in enumerate(sorted_y)}
        
        # 树状数组
        bit = BinaryIndexedTree([0] * len(sorted_y))
        
        # 统计逆序对
        inversions = 0
        for _, y in points:
            idx = mapping[y]
            inversions += bit.query(idx - 1)
            bit.update(idx, 1)
        
        return inversions
    
    def count_inversions_with_updates(arr, updates):
        """带更新的逆序对统计"""
        n = len(arr)
        
        # 离散化
        all_values = set(arr)
        for _, val in updates:
            all_values.add(val)
        
        sorted_values = sorted(all_values)
        mapping = {val: i + 1 for i, val in enumerate(sorted_values)}
        
        # 树状数组
        bit = BinaryIndexedTree([0] * len(sorted_values))
        
        # 初始逆序对
        inversions = 0
        for val in reversed(arr):
            idx = mapping[val]
            inversions += bit.query(idx - 1)
            bit.update(idx, 1)
        
        results = [inversions]
        
        # 处理更新
        for pos, new_val in updates:
            old_val = arr[pos]
            old_idx = mapping[old_val]
            new_idx = mapping[new_val]
            
            # 移除旧值的影响
            bit.update(old_idx, -1)
            inversions -= bit.query(old_idx - 1)
            
            # 添加新值的影响
            bit.update(new_idx, 1)
            inversions += bit.query(new_idx - 1)
            
            arr[pos] = new_val
            results.append(inversions)
        
        return results
    
    return count_inversions, count_inversions_2d, count_inversions_with_updates
```

### 3. 动态数组
```python
def dynamic_array_applications():
    """动态数组应用"""
    
    def dynamic_sum_array(operations):
        """动态和数组"""
        bit = BinaryIndexedTree([0] * 1000)  # 假设最大长度为1000
        results = []
        
        for op in operations:
            if op[0] == 'update':
                pos, val = op[1], op[2]
                bit.update(pos, val)
            elif op[0] == 'query':
                l, r = op[1], op[2]
                result = bit.range_query(l, r)
                results.append(result)
        
        return results
    
    def dynamic_rank_array(arr, queries):
        """动态排名数组"""
        n = len(arr)
        
        # 离散化
        sorted_arr = sorted(set(arr))
        mapping = {val: i + 1 for i, val in enumerate(sorted_arr)}
        
        # 树状数组
        bit = BinaryIndexedTree([0] * len(sorted_arr))
        
        # 初始化
        for val in arr:
            idx = mapping[val]
            bit.update(idx, 1)
        
        results = []
        for query in queries:
            if query[0] == 'rank':
                val = query[1]
                idx = mapping[val]
                rank = bit.query(idx - 1) + 1
                results.append(rank)
            elif query[0] == 'update':
                pos, new_val = query[1], query[2]
                old_val = arr[pos]
                old_idx = mapping[old_val]
                new_idx = mapping[new_val]
                
                bit.update(old_idx, -1)
                bit.update(new_idx, 1)
                arr[pos] = new_val
        
        return results
    
    def dynamic_median_array(arr, queries):
        """动态中位数数组"""
        n = len(arr)
        
        # 离散化
        sorted_arr = sorted(set(arr))
        mapping = {val: i + 1 for i, val in enumerate(sorted_arr)}
        reverse_mapping = {i + 1: val for i, val in enumerate(sorted_arr)}
        
        # 树状数组
        bit = BinaryIndexedTree([0] * len(sorted_arr))
        
        # 初始化
        for val in arr:
            idx = mapping[val]
            bit.update(idx, 1)
        
        results = []
        for query in queries:
            if query[0] == 'median':
                total = bit.query(len(sorted_arr))
                if total % 2 == 1:
                    target = (total + 1) // 2
                else:
                    target = total // 2
                
                # 二分查找中位数
                left, right = 1, len(sorted_arr)
                while left < right:
                    mid = (left + right) // 2
                    if bit.query(mid) >= target:
                        right = mid
                    else:
                        left = mid + 1
                
                median = reverse_mapping[left]
                results.append(median)
            elif query[0] == 'update':
                pos, new_val = query[1], query[2]
                old_val = arr[pos]
                old_idx = mapping[old_val]
                new_idx = mapping[new_val]
                
                bit.update(old_idx, -1)
                bit.update(new_idx, 1)
                arr[pos] = new_val
        
        return results
    
    return dynamic_sum_array, dynamic_rank_array, dynamic_median_array
```
