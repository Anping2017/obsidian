# 位运算技巧

## 概述

位运算技巧（Bit Manipulation Techniques）是算法优化中的重要方法，通过直接操作二进制位来解决问题。位运算技巧可以显著提高算法效率，减少内存使用，是算法竞赛和系统编程中的重要技能。

## 核心概念

### 基本定义
- **位运算**：直接操作二进制位的运算
- **位掩码**：用于选择特定位的模式
- **位操作**：设置、清除、翻转、检查位
- **位计数**：计算二进制中1的个数

### 关键特性
- **高效性**：位运算速度极快
- **空间效率**：可以用一个整数表示多个状态
- **简洁性**：代码实现简洁
- **通用性**：适用于多种问题类型

from typing import List, Tuple, Any, Callable, Optional
import time

class BitManipulationTechniques:
    """位运算技巧算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.data = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, data: List[int], *args) -> Any:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(data, *args)
    
    def set_data(self, data: List[int]) -> None:
        """设置数据"""
        self.data = data

# 基本位操作
def set_bit(num: int, pos: int) -> int:
    """设置指定位为1"""
    return num | (1 << pos)

def clear_bit(num: int, pos: int) -> int:
    """清除指定位（设为0）"""
    return num & ~(1 << pos)

def toggle_bit(num: int, pos: int) -> int:
    """翻转指定位"""
    return num ^ (1 << pos)

def get_bit(num: int, pos: int) -> int:
    """获取指定位的值"""
    return (num >> pos) & 1

def count_set_bits(num: int) -> int:
    """计算二进制中1的个数"""
    count = 0
    while num:
        count += num & 1
        num >>= 1
    return count

def count_set_bits_brian_kernighan(num: int) -> int:
    """使用Brian Kernighan算法计算1的个数"""
    count = 0
    while num:
        num &= num - 1  # 清除最右边的1
        count += 1
    return count

# 位掩码操作
def create_mask(start: int, end: int) -> int:
    """创建位掩码"""
    if start > end:
        return 0
    return (1 << (end - start + 1)) - 1 << start

def extract_bits(num: int, start: int, end: int) -> int:
    """提取指定位范围"""
    mask = create_mask(start, end)
    return (num & mask) >> start

def set_bits(num: int, start: int, end: int, value: int) -> int:
    """设置指定位范围"""
    mask = create_mask(start, end)
    return (num & ~mask) | ((value << start) & mask)

# 位运算技巧
def is_power_of_two(num: int) -> bool:
    """判断是否为2的幂"""
    return num > 0 and (num & (num - 1)) == 0

def next_power_of_two(num: int) -> int:
    """找到大于等于num的最小2的幂"""
    if num <= 0:
        return 1
    num -= 1
    num |= num >> 1
    num |= num >> 2
    num |= num >> 4
    num |= num >> 8
    num |= num >> 16
    return num + 1

def reverse_bits(num: int, bits: int = 32) -> int:
    """反转指定位数"""
    result = 0
    for i in range(bits):
        result = (result << 1) | (num & 1)
        num >>= 1
    return result

def swap_bits(num: int, i: int, j: int) -> int:
    """交换两个位"""
    if ((num >> i) & 1) != ((num >> j) & 1):
        num ^= (1 << i) | (1 << j)
    return num

# 子集生成（位掩码）
def generate_subsets(nums: List[int]) -> List[List[int]]:
    """使用位掩码生成所有子集"""
    n = len(nums)
    subsets = []
    
    for mask in range(1 << n):
        subset = []
        for i in range(n):
            if mask & (1 << i):
                subset.append(nums[i])
        subsets.append(subset)
    
    return subsets

# 组合生成（位掩码）
def generate_combinations(nums: List[int], k: int) -> List[List[int]]:
    """使用位掩码生成k个元素的组合"""
    n = len(nums)
    combinations = []
    
    for mask in range(1 << n):
        if count_set_bits(mask) == k:
            combination = []
            for i in range(n):
                if mask & (1 << i):
                    combination.append(nums[i])
            combinations.append(combination)
    
    return combinations

# 全排列生成（位掩码）
def generate_permutations(nums: List[int]) -> List[List[int]]:
    """使用位掩码生成全排列"""
    n = len(nums)
    permutations = []
    
    def backtrack(perm, used):
        if len(perm) == n:
            permutations.append(perm[:])
            return
        
        for i in range(n):
            if not (used & (1 << i)):
                perm.append(nums[i])
                backtrack(perm, used | (1 << i))
                perm.pop()
    
    backtrack([], 0)
    return permutations

# 位运算优化
def fast_multiply_by_power_of_two(num: int, power: int) -> int:
    """快速乘以2的幂"""
    return num << power

def fast_divide_by_power_of_two(num: int, power: int) -> int:
    """快速除以2的幂"""
    return num >> power

def fast_modulo_by_power_of_two(num: int, power: int) -> int:
    """快速对2的幂取模"""
    return num & ((1 << power) - 1)

def fast_ceiling_divide(num: int, divisor: int) -> int:
    """快速向上取整除法"""
    return (num + divisor - 1) // divisor

def fast_floor_divide(num: int, divisor: int) -> int:
    """快速向下取整除法"""
    return num // divisor

# 位运算技巧应用
def single_number(nums: List[int]) -> int:
    """找出只出现一次的数字"""
    result = 0
    for num in nums:
        result ^= num
    return result

def single_number_ii(nums: List[int]) -> int:
    """找出只出现一次的数字（其他数字出现3次）"""
    ones = twos = 0
    for num in nums:
        twos |= ones & num
        ones ^= num
        threes = ones & twos
        ones &= ~threes
        twos &= ~threes
    return ones

def missing_number(nums: List[int]) -> int:
    """找出缺失的数字"""
    n = len(nums)
    expected_sum = n * (n + 1) // 2
    actual_sum = sum(nums)
    return expected_sum - actual_sum

def missing_number_xor(nums: List[int]) -> int:
    """使用异或找出缺失的数字"""
    n = len(nums)
    result = n
    for i in range(n):
        result ^= i ^ nums[i]
    return result
