# 基础模板

## 概述

基础模板是算法实现中的重要工具，提供了常用算法的标准实现模板。通过掌握这些基础模板，可以快速实现各种算法，提高编程效率和代码质量。

```python
# 示例：两数之和
def two_sum(nums, target):
    """两数之和"""
    nums.sort()
    left, right = 0, len(nums) - 1
    
    while left < right:
        current_sum = nums[left] + nums[right]
        if current_sum == target:
            return [left, right]
        elif current_sum < target:
            left += 1
        else:
            right -= 1
    
    return []
```

```python
def has_cycle(head):
    """链表中环的检测"""
    if not head or not head.next:
        return False
    
    slow = fast = head
    
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        
        if slow == fast:
            return True
    
    return False
```

### 2. 滑动窗口模板

#### 固定窗口模板
```python
def fixed_window_template(nums, k):
    """固定窗口模板"""
    window_sum = sum(nums[:k])
    max_sum = window_sum
    
    for i in range(k, len(nums)):
        window_sum = window_sum - nums[i - k] + nums[i]
        max_sum = max(max_sum, window_sum)
    
    return max_sum

# 示例：滑动窗口最大值
def max_sliding_window(nums, k):
    """滑动窗口最大值"""
    from collections import deque
    
    dq = deque()
    result = []
    
    for i in range(len(nums)):
        # 移除窗口左边超出范围的元素
        while dq and dq[0] <= i - k:
            dq.popleft()
        
        # 移除窗口右边小于当前元素的元素
        while dq and nums[dq[-1]] < nums[i]:
            dq.pop()
        
        dq.append(i)
        
        # 当窗口大小达到k时，记录最大值
        if i >= k - 1:
            result.append(nums[dq[0]])
    
    return result
```

```python
def variable_window_template(s):

    left = 0

# 示例：无重复字符的最长子串
def length_of_longest_substring(s):
    """无重复字符的最长子串"""
    char_set = set()
    left = 0
    max_length = 0
    
    for right in range(len(s)):
        while s[right] in char_set:
            char_set.remove(s[left])
            left += 1
        
        char_set.add(s[right])
        max_length = max(max_length, right - left + 1)
    
    return max_length
```

```python
    n = len(nums)
    dp = [0] * n

def length_of_lis(nums):
    """最长递增子序列"""
    if not nums:
        return 0
    
    n = len(nums)
    dp = [1] * n
    
    for i in range(n):
        for j in range(i):
```

#### 二维DP模板
```python
def two_dimensional_dp_template(text1, text2):
    """二维DP模板"""
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if condition:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    
    return dp[m][n]

# 示例：最长公共子序列
def longest_common_subsequence(text1, text2):
    """最长公共子序列"""
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    
    return dp[m][n]
```

```python
# 示例：全排列
def permute(nums):
    """全排列"""
    result = []
    
    def backtrack(path):
        if len(path) == len(nums):
            result.append(path[:])
            return
        
        for num in nums:
            if num not in path:
                path.append(num)
                backtrack(path)
                path.pop()
    
    backtrack([])
    return result
```

```python
def optimized_backtrack_template(nums):

        if is_valid_solution(path):
            result.append(path[:])

def combine(n, k):
    """组合"""
    result = []
    
    def backtrack(start, path):
        if len(path) == k:
            result.append(path[:])
            return
        
        for i in range(start, n + 1):
            path.append(i)
            backtrack(i + 1, path)
            path.pop()
    
    backtrack(1, [])
    return result
```

```python
def merge_sort(nums):
    """归并排序"""
    if len(nums) <= 1:
        return nums
    
    mid = len(nums) // 2
    left = merge_sort(nums[:mid])
    right = merge_sort(nums[mid:])
    
    return merge(left, right)

def merge(left, right):
    """合并两个有序数组"""
    result = []
    i = j = 0
    
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    
    result.extend(left[i:])
    result.extend(right[j:])
    return result
```

```python
def activity_selection(activities):
    """活动选择问题"""
    activities.sort(key=lambda x: x[1])  # 按结束时间排序
    
    result = [activities[0]]
    last_end_time = activities[0][1]
    
    for start, end in activities[1:]:
        if start >= last_end_time:
            result.append((start, end))
            last_end_time = end
    
    return result
```

```python
def dfs_template(graph, start, visited):
    """DFS模板"""

    for neighbor in graph[start]:
        if neighbor not in visited:

# 示例：岛屿数量
def num_islands(grid):
    """岛屿数量"""
    if not grid or not grid[0]:
        return 0
    
    m, n = len(grid), len(grid[0])
    count = 0
    
    def dfs(i, j):
        if i < 0 or i >= m or j < 0 or j >= n or grid[i][j] != '1':
            return
        
        grid[i][j] = '0'
        dfs(i + 1, j)
        dfs(i - 1, j)
        dfs(i, j + 1)
        dfs(i, j - 1)
    
    for i in range(m):
        for j in range(n):
            if grid[i][j] == '1':
                dfs(i, j)
                count += 1
    
    return count
```

#### BFS模板
```python
def bfs_template(graph, start):
    """BFS模板"""
    from collections import deque
    
    queue = deque([start])
    visited = {start}
    
    while queue:
        node = queue.popleft()
        
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

# 示例：最短路径
def shortest_path(graph, start, end):
    """最短路径"""
    from collections import deque
    
    queue = deque([(start, 0)])
    visited = {start}
    
    while queue:
        node, distance = queue.popleft()
        
        if node == end:
            return distance
        
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, distance + 1))
    
    return -1
```
