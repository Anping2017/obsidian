# 图论模板

```python
from collections import defaultdict

class Graph:
    """邻接表图模板"""
    
    def __init__(self):
        self.graph = defaultdict(list)
        self.vertices = set()
    
    def add_edge(self, u, v, weight=1):
        """添加边"""
        self.graph[u].append((v, weight))
        self.vertices.add(u)
        self.vertices.add(v)
    
    def add_vertex(self, vertex):
        """添加顶点"""
        self.vertices.add(vertex)
    
    def get_neighbors(self, vertex):
        """获取邻居"""
        return self.graph[vertex]
    
    def get_vertices(self):
        """获取所有顶点"""
        return list(self.vertices)
    
    def get_edges(self):
        """获取所有边"""
        edges = []
        for u in self.graph:
            for v, weight in self.graph[u]:
                edges.append((u, v, weight))
        return edges
    
    def __str__(self):
        result = []
        for u in self.graph:
            for v, weight in self.graph[u]:
                result.append(f"{u} -> {v} (weight: {weight})")
        return "\n".join(result)

# 使用示例
g = Graph()
g.add_edge(0, 1, 5)
g.add_edge(0, 2, 3)
g.add_edge(1, 2, 2)
g.add_edge(2, 3, 7)
print("图结构:")
print(g)
```

```python
class AdjacencyMatrix:
    """邻接矩阵图模板"""
    
    def __init__(self, num_vertices):
        self.num_vertices = num_vertices
        self.matrix = [[0] * num_vertices for _ in range(num_vertices)]
    
    def add_edge(self, u, v, weight=1):
        """添加边"""
        self.matrix[u][v] = weight
        self.matrix[v][u] = weight  # 无向图
    
    def remove_edge(self, u, v):
        """删除边"""
        self.matrix[u][v] = 0
        self.matrix[v][u] = 0
    
    def has_edge(self, u, v):
        """判断是否有边"""
        return self.matrix[u][v] != 0
    
    def get_weight(self, u, v):
        """获取边的权重"""
        return self.matrix[u][v]
    
    def get_neighbors(self, vertex):
        """获取邻居"""
        neighbors = []
        for i in range(self.num_vertices):
            if self.matrix[vertex][i] != 0:
                neighbors.append((i, self.matrix[vertex][i]))
        return neighbors
    
    def __str__(self):
        result = []
        for i in range(self.num_vertices):
            row = []
            for j in range(self.num_vertices):
                row.append(str(self.matrix[i][j]))
            result.append(" ".join(row))
        return "\n".join(result)

# 使用示例
g = AdjacencyMatrix(4)
g.add_edge(0, 1, 5)
g.add_edge(0, 2, 3)
g.add_edge(1, 2, 2)
g.add_edge(2, 3, 7)
print("邻接矩阵:")
print(g)
```

```python
def dfs_recursive(graph, start, visited=None):
    """DFS递归模板"""
    if visited is None:
        visited = set()
    
    visited.add(start)
    print(start, end=" ")
    
    for neighbor, _ in graph.get_neighbors(start):
        if neighbor not in visited:
            dfs_recursive(graph, neighbor, visited)
    
    return visited

def dfs_iterative(graph, start):
    """DFS迭代模板"""
    visited = set()
    stack = [start]
    
    while stack:
        vertex = stack.pop()
        if vertex not in visited:
            visited.add(vertex)
            print(vertex, end=" ")
            
            for neighbor, _ in graph.get_neighbors(vertex):
                if neighbor not in visited:
                    stack.append(neighbor)
    
    return visited

def dfs_path(graph, start, end):
    """DFS找路径"""
    def dfs_helper(vertex, path):
        if vertex == end:
            return path
        
        for neighbor, _ in graph.get_neighbors(vertex):
            if neighbor not in path:
                result = dfs_helper(neighbor, path + [neighbor])
                if result:
                    return result
        
        return None
    
    return dfs_helper(start, [start])

# 使用示例

g.add_edge(0, 2)
g.add_edge(1, 3)
g.add_edge(2, 3)
g.add_edge(3, 4)
```

#### BFS模板
```python
from collections import deque

def bfs(graph, start):
    """BFS模板"""
    visited = set()
    queue = deque([start])
    visited.add(start)
    
    while queue:
        vertex = queue.popleft()
        print(vertex, end=" ")
        
        for neighbor, _ in graph.get_neighbors(vertex):
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    
    return visited

def bfs_shortest_path(graph, start, end):
    """BFS最短路径"""
    if start == end:
        return [start]
    
    visited = set()
    queue = deque([(start, [start])])
    visited.add(start)
    
    while queue:
        vertex, path = queue.popleft()
        
        for neighbor, _ in graph.get_neighbors(vertex):
            if neighbor == end:
                return path + [neighbor]
            
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, path + [neighbor]))
    
    return None

def bfs_level_order(graph, start):
    """BFS层序遍历"""
    visited = set()
    queue = deque([(start, 0)])
    visited.add(start)
    levels = {}
    
    while queue:
        vertex, level = queue.popleft()
        
        if level not in levels:
            levels[level] = []
        levels[level].append(vertex)
        
        for neighbor, _ in graph.get_neighbors(vertex):
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, level + 1))
    
    return levels

# 使用示例
g = Graph()
g.add_edge(0, 1)
g.add_edge(0, 2)
g.add_edge(1, 3)
g.add_edge(2, 3)
g.add_edge(3, 4)

print("BFS遍历:")
bfs(g, 0)
print()

print("BFS最短路径:")
path = bfs_shortest_path(g, 0, 4)
print(f"从0到4的最短路径: {path}")

print("BFS层序遍历:")
levels = bfs_level_order(g, 0)
for level, vertices in levels.items():
    print(f"第{level}层: {vertices}")
```

### 3. 最短路径模板

#### Dijkstra算法模板
```python
import heapq

def dijkstra(graph, start):
    """Dijkstra算法"""
    distances = {vertex: float('inf') for vertex in graph.get_vertices()}
    distances[start] = 0
    pq = [(0, start)]
    visited = set()
    
    while pq:
        dist, u = heapq.heappop(pq)
        
        if u in visited:
            continue
        
        visited.add(u)
        
        for v, weight in graph.get_neighbors(u):
            if v not in visited:
                new_dist = dist + weight
                if new_dist < distances[v]:
                    distances[v] = new_dist
                    heapq.heappush(pq, (new_dist, v))
    
    return distances

def dijkstra_path(graph, start, end):
    """Dijkstra算法找路径"""
    distances = {vertex: float('inf') for vertex in graph.get_vertices()}
    distances[start] = 0
    pq = [(0, start)]
    visited = set()
    parent = {}
    
    while pq:
        dist, u = heapq.heappop(pq)
        
        if u in visited:
            continue
        
        visited.add(u)
        
        if u == end:
            break
        
        for v, weight in graph.get_neighbors(u):
            if v not in visited:
                new_dist = dist + weight
                if new_dist < distances[v]:
                    distances[v] = new_dist
                    parent[v] = u
                    heapq.heappush(pq, (new_dist, v))
    
    # 重构路径
    if end not in parent and start != end:
        return None
    
    path = []
    current = end
    while current is not None:
        path.append(current)
        current = parent.get(current)
    
    return path[::-1]

# 使用示例
g = Graph()
g.add_edge(0, 1, 4)
g.add_edge(0, 2, 2)
g.add_edge(1, 2, 1)
g.add_edge(1, 3, 5)
g.add_edge(2, 3, 8)
g.add_edge(2, 4, 10)
g.add_edge(3, 4, 2)

print("Dijkstra最短距离:")
distances = dijkstra(g, 0)
for vertex, dist in distances.items():
    print(f"到顶点{vertex}的距离: {dist}")

print("Dijkstra最短路径:")
path = dijkstra_path(g, 0, 4)
print(f"从0到4的最短路径: {path}")
```

#### Floyd-Warshall算法模板
```python
def floyd_warshall(graph):
    """Floyd-Warshall算法"""
    vertices = graph.get_vertices()
    n = len(vertices)
    
    # 初始化距离矩阵
    dist = [[float('inf')] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    # 添加边的权重
    for u, v, weight in graph.get_edges():
        i, j = vertices.index(u), vertices.index(v)
        dist[i][j] = weight
        dist[j][i] = weight  # 无向图
    
    # Floyd-Warshall算法
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    return dist, vertices

# 使用示例
g = Graph()
g.add_edge(0, 1, 4)
g.add_edge(0, 2, 2)
g.add_edge(1, 2, 1)
g.add_edge(1, 3, 5)
g.add_edge(2, 3, 8)
g.add_edge(2, 4, 10)
g.add_edge(3, 4, 2)

print("Floyd-Warshall最短距离矩阵:")
dist, vertices = floyd_warshall(g)
for i in range(len(vertices)):
    for j in range(len(vertices)):
        print(f"从{vertices[i]}到{vertices[j]}: {dist[i][j]}")
```

### 4. 最小生成树模板

#### Kruskal算法模板
```python
class UnionFind:
    """并查集"""
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        
        if self.rank[px] < self.rank[py]:
            self.parent[px] = py
        elif self.rank[px] > self.rank[py]:
            self.parent[py] = px
        else:
            self.parent[py] = px
            self.rank[px] += 1
        
        return True

def kruskal(graph):
    """Kruskal算法"""
    vertices = graph.get_vertices()
    n = len(vertices)
    uf = UnionFind(n)
    
    # 获取所有边并按权重排序
    edges = graph.get_edges()
    edges.sort(key=lambda x: x[2])
    
    mst = []
    total_weight = 0
    
    for u, v, weight in edges:
        i, j = vertices.index(u), vertices.index(v)
        if uf.union(i, j):
            mst.append((u, v, weight))
            total_weight += weight
            if len(mst) == n - 1:
                break
    
    return mst, total_weight

# 使用示例
g = Graph()
g.add_edge(0, 1, 4)
g.add_edge(0, 2, 2)
g.add_edge(1, 2, 1)
g.add_edge(1, 3, 5)
g.add_edge(2, 3, 8)
g.add_edge(2, 4, 10)
g.add_edge(3, 4, 2)

print("Kruskal最小生成树:")
mst, weight = kruskal(g)
print(f"总权重: {weight}")
for u, v, w in mst:
    print(f"{u} - {v}: {w}")
```

#### Prim算法模板
```python
def prim(graph, start):
    """Prim算法"""
    vertices = graph.get_vertices()
    n = len(vertices)
    visited = set()
    mst = []
    total_weight = 0
    
    # 优先队列存储(权重, 起点, 终点)
    pq = []
    visited.add(start)
    
    # 添加起始顶点的所有边
    for neighbor, weight in graph.get_neighbors(start):
        heapq.heappush(pq, (weight, start, neighbor))
    
    while pq and len(visited) < n:
        weight, u, v = heapq.heappop(pq)
        
        if v in visited:
            continue
        
        visited.add(v)
        mst.append((u, v, weight))
        total_weight += weight
        
        # 添加新顶点的所有边
        for neighbor, w in graph.get_neighbors(v):
            if neighbor not in visited:
                heapq.heappush(pq, (w, v, neighbor))
    
    return mst, total_weight

# 使用示例
g = Graph()
g.add_edge(0, 1, 4)
g.add_edge(0, 2, 2)
g.add_edge(1, 2, 1)
g.add_edge(1, 3, 5)
g.add_edge(2, 3, 8)
g.add_edge(2, 4, 10)
g.add_edge(3, 4, 2)

print(f"总权重: {weight}")

    print(f"{u} - {v}: {w}")
```

### 5. 拓扑排序模板

#### 拓扑排序模板
```python
def topological_sort(graph):
    """拓扑排序"""
    vertices = graph.get_vertices()
    in_degree = {vertex: 0 for vertex in vertices}
    
    # 计算入度
    for u in vertices:
        for v, _ in graph.get_neighbors(u):
            in_degree[v] += 1
    
    # 找到入度为0的顶点
    queue = deque([v for v in vertices if in_degree[v] == 0])
    result = []
    
    while queue:
        u = queue.popleft()
        result.append(u)
        
        for v, _ in graph.get_neighbors(u):
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)
    
    # 检查是否有环
    if len(result) != len(vertices):
        return None  # 有环
    
    return result

def topological_sort_dfs(graph):
    """DFS拓扑排序"""
    vertices = graph.get_vertices()
    visited = set()
    temp_visited = set()
    result = []
    
    def dfs(vertex):
        if vertex in temp_visited:
            return False  # 有环
        
        if vertex in visited:
            return True
        
        temp_visited.add(vertex)
        
        for neighbor, _ in graph.get_neighbors(vertex):
            if not dfs(neighbor):
                return False
        
        temp_visited.remove(vertex)
        visited.add(vertex)
        result.append(vertex)
        return True
    
    for vertex in vertices:
        if vertex not in visited:
            if not dfs(vertex):
                return None  # 有环
    
    return result[::-1]

# 使用示例
g = Graph()
g.add_edge(0, 1)
g.add_edge(0, 2)
g.add_edge(1, 3)
g.add_edge(2, 3)
g.add_edge(3, 4)

print("拓扑排序:")
result = topological_sort(g)
if result:
    print(f"拓扑排序结果: {result}")
else:
    print("图中存在环")

print("DFS拓扑排序:")
result = topological_sort_dfs(g)
if result:
    print(f"DFS拓扑排序结果: {result}")
else:
    print("图中存在环")
```
