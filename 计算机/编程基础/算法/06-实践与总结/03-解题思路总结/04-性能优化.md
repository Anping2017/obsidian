# 性能优化

## 概述

性能优化是算法实现中的重要技能，通过系统性的优化方法可以显著提高算法的执行效率。掌握有效的性能优化技巧可以提高程序运行速度，减少资源消耗，提升用户体验。

## 核心概念

### 基本定义
- **性能优化**：提高算法执行效率的方法
- **时间复杂度**：算法执行时间随输入规模的变化
- **空间复杂度**：算法所需内存随输入规模的变化
- **优化策略**：系统性的优化方法

### 关键特性
- **系统性**：提供完整的优化流程
- **高效性**：显著提高性能
- **平衡性**：平衡时间和空间复杂度
- **可测量性**：可以量化的优化效果

def time_complexity_optimization():
    """时间复杂度优化示例"""
    
    # 原始算法：O(n^2)
    def find_duplicate_original(nums):
        for i in range(len(nums)):
            for j in range(i + 1, len(nums)):
                if nums[i] == nums[j]:
                    return nums[i]
        return -1
    
    # 优化算法：O(n)
    def find_duplicate_optimized(nums):
        seen = set()
        for num in nums:
            if num in seen:
                return num
            seen.add(num)
        return -1
    
    # 测试
    nums = [1, 3, 4, 2, 2]
    print(f"原始算法结果: {find_duplicate_original(nums)}")
    print(f"优化算法结果: {find_duplicate_optimized(nums)}")

def space_complexity_optimization():
    """空间复杂度优化示例"""
    
    # 原始算法：O(n) 额外空间
    def remove_duplicates_original(nums):
        unique_nums = []
        for num in nums:
            if num not in unique_nums:
                unique_nums.append(num)
        return unique_nums
    
    # 优化算法：O(1) 额外空间
    def remove_duplicates_optimized(nums):
        if not nums:
            return 0
        
        write_index = 1
        for read_index in range(1, len(nums)):
            if nums[read_index] != nums[read_index - 1]:
                nums[write_index] = nums[read_index]
                write_index += 1
        
        return write_index
    
    # 测试
    nums = [1, 1, 2, 2, 3, 3, 4]
    print(f"原始数组: {nums}")
    length = remove_duplicates_optimized(nums)
    print(f"去重后长度: {length}")
    print(f"去重后数组: {nums[:length]}")

def algorithm_selection_optimization():
    """算法选择优化示例"""
    
    # 根据数据规模选择算法
    def sort_optimized(arr):
        n = len(arr)
        
        if n <= 10:
            # 小数据量：使用插入排序
            return insertion_sort(arr)
        elif n <= 1000:
            # 中等数据量：使用快速排序
            return quick_sort(arr)
        else:
            # 大数据量：使用归并排序
            return merge_sort(arr)
    
    def insertion_sort(arr):
        for i in range(1, len(arr)):
            key = arr[i]
            j = i - 1
            while j >= 0 and arr[j] > key:
                arr[j + 1] = arr[j]
                j -= 1
            arr[j + 1] = key
        return arr
    
    def quick_sort(arr):
        if len(arr) <= 1:
            return arr
        
        pivot = arr[len(arr) // 2]
        left = [x for x in arr if x < pivot]
        middle = [x for x in arr if x == pivot]
        right = [x for x in arr if x > pivot]
        
        return quick_sort(left) + middle + quick_sort(right)
    
    def merge_sort(arr):
        if len(arr) <= 1:
            return arr
        
        mid = len(arr) // 2
        left = merge_sort(arr[:mid])
        right = merge_sort(arr[mid:])
        
        return merge(left, right)
    
    def merge(left, right):
        result = []
        i = j = 0
        
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        
        result.extend(left[i:])
        result.extend(right[j:])
        return result
    
    # 测试
    test_cases = [
        [3, 1, 4, 1, 5],  # 小数据量
        list(range(100, 0, -1)),  # 中等数据量
        list(range(1000, 0, -1))   # 大数据量
    ]
    
    for i, arr in enumerate(test_cases):
        print(f"测试用例 {i + 1}: 长度 {len(arr)}")
        result = sort_optimized(arr.copy())
        print(f"排序结果: {result[:10]}..." if len(result) > 10 else f"排序结果: {result}")

# 使用示例

def data_structure_optimization():
    """数据结构选择优化示例"""
    
    # 根据操作频率选择数据结构
    class OptimizedDataStructure:
        def __init__(self):
            self.data = []
            self.index_map = {}  # 用于快速查找
        
        def insert(self, value):
            """插入操作：O(1)"""
            self.data.append(value)
            self.index_map[value] = len(self.data) - 1
        
        def find(self, value):
            """查找操作：O(1)"""
            return self.index_map.get(value, -1)
        
        def delete(self, value):
            """删除操作：O(1)"""
            if value in self.index_map:
                index = self.index_map[value]
                # 将最后一个元素移到删除位置
                last_value = self.data[-1]
                self.data[index] = last_value
                self.index_map[last_value] = index
                
                # 删除最后一个元素
                self.data.pop()
                del self.index_map[value]
                return True
            return False
    
    # 测试
    ds = OptimizedDataStructure()
    
    # 插入操作
    for i in range(5):
        ds.insert(i)
    
    print(f"插入后数据: {ds.data}")
    
    # 查找操作
    print(f"查找3: {ds.find(3)}")
    print(f"查找10: {ds.find(10)}")
    
    # 删除操作
    ds.delete(2)
    print(f"删除2后数据: {ds.data}")

def cache_optimization():
    """缓存优化示例"""
    
    # 使用装饰器实现缓存
    def memoize(func):
        cache = {}
        
        def wrapper(*args):
            if args in cache:
                return cache[args]
            result = func(*args)
            cache[args] = result
            return result
        
        return wrapper
    
    # 原始斐波那契数列（无缓存）
    def fibonacci_original(n):
        if n <= 1:
            return n
        return fibonacci_original(n - 1) + fibonacci_original(n - 2)
    
    # 优化斐波那契数列（有缓存）
    @memoize
    def fibonacci_cached(n):
        if n <= 1:
            return n
        return fibonacci_cached(n - 1) + fibonacci_cached(n - 2)
    
    # 测试性能
    import time
    
    n = 35
    
    # 测试原始版本
    start_time = time.time()
    result_original = fibonacci_original(n)
    original_time = time.time() - start_time
    
    # 测试缓存版本
    start_time = time.time()
    result_cached = fibonacci_cached(n)
    cached_time = time.time() - start_time
    
    print(f"原始版本: {result_original}, 时间: {original_time:.6f}秒")
    print(f"缓存版本: {result_cached}, 时间: {cached_time:.6f}秒")
    print(f"性能提升: {original_time / cached_time:.2f}倍")

def loop_optimization():
    """循环优化示例"""
    
    # 原始代码：嵌套循环
    def find_common_elements_original(arr1, arr2):
        common = []
        for i in range(len(arr1)):
            for j in range(len(arr2)):
                if arr1[i] == arr2[j]:
                    common.append(arr1[i])
        return common
    
    # 优化代码：使用集合
    def find_common_elements_optimized(arr1, arr2):

def memory_optimization():
    """内存优化示例"""
    
    # 原始代码：创建多个列表
    def process_data_original(data):
        result1 = []
        result2 = []
        result3 = []
        
        for item in data:
            if item > 0:
                result1.append(item)
            elif item < 0:
                result2.append(item)
            else:
                result3.append(item)
        
        return result1, result2, result3
    
    # 优化代码：使用生成器
    def process_data_optimized(data):
        def positive_generator():
            for item in data:
                if item > 0:
                    yield item
        
        def negative_generator():
            for item in data:
                if item < 0:
                    yield item
        
        def zero_generator():
            for item in data:
                if item == 0:
                    yield item
        
        return positive_generator(), negative_generator(), zero_generator()
    
    # 测试
    data = [1, -2, 0, 3, -4, 5, 0]
    
    print("原始方法:")
    result1, result2, result3 = process_data_original(data)
    print(f"正数: {list(result1)}")
    print(f"负数: {list(result2)}")
    print(f"零: {list(result3)}")
    
    print("\n优化方法:")
    pos_gen, neg_gen, zero_gen = process_data_optimized(data)
    print(f"正数: {list(pos_gen)}")
    print(f"负数: {list(neg_gen)}")
    print(f"零: {list(zero_gen)}")

import concurrent.futures
import multiprocessing

def parallel_optimization():
    """并行优化示例"""
    
    def cpu_intensive_task(n):
        """CPU密集型任务"""
        result = 0
        for i in range(n):
            result += i ** 2
        return result
    
    def parallel_processing(data):
        """并行处理"""
        with concurrent.futures.ProcessPoolExecutor() as executor:
            results = list(executor.map(cpu_intensive_task, data))
        return results
    
    def sequential_processing(data):
        """顺序处理"""
        results = []
        for item in data:
            results.append(cpu_intensive_task(item))
        return results
    
    # 测试
    data = [100000, 200000, 300000, 400000]
    
    import time
    
    # 顺序处理
    start_time = time.time()
    sequential_results = sequential_processing(data)
    sequential_time = time.time() - start_time
    
    # 并行处理
    start_time = time.time()
    parallel_results = parallel_processing(data)
    parallel_time = time.time() - start_time
    
    print(f"顺序处理时间: {sequential_time:.6f}秒")
    print(f"并行处理时间: {parallel_time:.6f}秒")
    print(f"性能提升: {sequential_time / parallel_time:.2f}倍")

def compilation_optimization():
    """编译优化示例"""
    
    # 使用Numba进行JIT编译
    try:
        from numba import jit
        
        @jit(nopython=True)
        def fast_fibonacci(n):
            if n <= 1:
                return n
            a, b = 0, 1
            for _ in range(2, n + 1):
                a, b = b, a + b
            return b
        
        def slow_fibonacci(n):
            if n <= 1:
                return n
