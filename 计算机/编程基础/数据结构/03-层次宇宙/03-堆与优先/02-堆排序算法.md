# 堆排序算法

## 📖 核心概念

**堆排序算法**是一种基于堆数据结构的比较排序算法。它利用堆的性质来实现排序，具有O(n log n)的时间复杂度和O(1)的空间复杂度。

### 🏗️ 堆排序原理

```mermaid
graph TD
    A[堆排序] --> B[建堆过程]
    A --> C[排序过程]
    A --> D[堆维护]
    
    B --> B1[自底向上建堆]
    B --> B2[O(n)时间复杂度]
    B --> B3[堆化操作]
    
    C --> C1[交换根节点]
    C --> C2[减少堆大小]
    C --> C3[重新堆化]
    
    D --> D1[下沉操作]
    D --> D2[保持堆性质]
    D --> D3[递归调整]
```

## 🔧 堆排序实现

### 基础堆排序

```cpp
class HeapSort {
private:
    // 下沉操作（堆化）
    void heapify(vector<int>& arr, int n, int i) {
        int largest = i;        // 假设根节点最大
        int left = 2 * i + 1;   // 左子节点
        int right = 2 * i + 2;  // 右子节点
        
        // 如果左子节点比根节点大
        if (left < n && arr[left] > arr[largest]) {
            largest = left;
        }
        
        // 如果右子节点比当前最大值大
        if (right < n && arr[right] > arr[largest]) {
            largest = right;
        }
        
        // 如果最大值不是根节点
        if (largest != i) {
            swap(arr[i], arr[largest]);
            
            // 递归堆化受影响的子树
            heapify(arr, n, largest);
        }
    }
    
    // 建堆过程
    void buildHeap(vector<int>& arr) {
        int n = arr.size();
        
        // 从最后一个非叶子节点开始，向上堆化
        for (int i = n / 2 - 1; i >= 0; i--) {
            heapify(arr, n, i);
        }
    }
    
public:
    // 堆排序主函数
    void heapSort(vector<int>& arr) {
        int n = arr.size();
        
        // 第一步：建堆
        buildHeap(arr);
        
        // 第二步：逐个提取元素
        for (int i = n - 1; i > 0; i--) {
            // 将当前最大值（根节点）移到末尾
            swap(arr[0], arr[i]);
            
            // 在减少的堆上调用heapify
            heapify(arr, i, 0);
        }
    }
    
    // 打印数组
    void printArray(const vector<int>& arr) {
        for (int i = 0; i < arr.size(); ++i) {
            cout << arr[i] << " ";
        }
        cout << endl;
    }
};
```

### 优化版本

```cpp
class OptimizedHeapSort {
private:
    // 优化的堆化操作（迭代版本）
    void heapifyIterative(vector<int>& arr, int n, int i) {
        while (true) {
            int largest = i;
            int left = 2 * i + 1;
            int right = 2 * i + 2;
            
            if (left < n && arr[left] > arr[largest]) {
                largest = left;
            }
            
            if (right < n && arr[right] > arr[largest]) {
                largest = right;
            }
            
            if (largest == i) {
                break;  // 堆性质已满足
            }
            
            swap(arr[i], arr[largest]);
            i = largest;
        }
    }
    
    // 自底向上建堆（更高效）
    void buildHeapBottomUp(vector<int>& arr) {
        int n = arr.size();
        
        // 从最后一个非叶子节点开始
        for (int i = (n - 2) / 2; i >= 0; i--) {
            heapifyIterative(arr, n, i);
        }
    }
    
public:
    // 优化的堆排序
    void heapSort(vector<int>& arr) {
        int n = arr.size();
        
        // 建堆
        buildHeapBottomUp(arr);
        
        // 排序
        for (int i = n - 1; i > 0; i--) {
            swap(arr[0], arr[i]);
            heapifyIterative(arr, i, 0);
        }
    }
    
    // 部分排序（只排序前k个元素）
    void partialHeapSort(vector<int>& arr, int k) {
        int n = arr.size();
        
        // 建堆
        buildHeapBottomUp(arr);
        
        // 只提取前k个最大元素
        for (int i = n - 1; i >= n - k && i > 0; i--) {
            swap(arr[0], arr[i]);
            heapifyIterative(arr, i, 0);
        }
    }
};
```

## 📊 性能分析

### 时间复杂度分析

```cpp
class HeapSortAnalysis {
public:
    // 建堆时间复杂度分析
    void analyzeBuildHeapTime() {
        // 建堆的时间复杂度是O(n)，不是O(n log n)
        // 证明：
        // 对于高度为h的完全二叉树，有n = 2^(h+1) - 1个节点
        // 在高度i的层上，最多有2^i个节点
        // 每个节点最多需要下沉h-i层
        // 总时间 = Σ(i=0 to h) 2^i * (h-i)
        // 通过数学推导可以证明这是O(n)
    }
    
    // 堆排序总时间复杂度
    void analyzeTotalTime() {
        // 建堆：O(n)
        // 排序：n-1次提取，每次O(log n)
        // 总时间：O(n) + O(n log n) = O(n log n)
    }
    
    // 空间复杂度分析
    void analyzeSpaceComplexity() {
        // 原地排序：O(1)额外空间
        // 递归栈：O(log n)（如果使用递归版本）
        // 迭代版本：O(1)
    }
};
```

## 🎯 应用场景

### 1. Top K问题

```cpp
class TopKProblems {
public:
    // 找出数组中第k大的元素
    int findKthLargest(vector<int>& nums, int k) {
        int n = nums.size();
        
        // 建堆
        for (int i = n / 2 - 1; i >= 0; i--) {
            heapify(nums, n, i);
        }
        
        // 提取前k个最大元素
        for (int i = n - 1; i >= n - k; i--) {
            swap(nums[0], nums[i]);
            heapify(nums, i, 0);
        }
        
        return nums[n - k];
    }
    
    // 找出数组中前k个最大元素
    vector<int> findTopK(vector<int>& nums, int k) {
        vector<int> result;
        int n = nums.size();
        
        // 建堆
        for (int i = n / 2 - 1; i >= 0; i--) {
            heapify(nums, n, i);
        }
        
        // 提取前k个最大元素
        for (int i = n - 1; i >= n - k; i--) {
            result.push_back(nums[0]);
            swap(nums[0], nums[i]);
            heapify(nums, i, 0);
        }
        
        return result;
    }
    
private:
    void heapify(vector<int>& arr, int n, int i) {
        int largest = i;
        int left = 2 * i + 1;
        int right = 2 * i + 2;
        
        if (left < n && arr[left] > arr[largest]) {
            largest = left;
        }
        
        if (right < n && arr[right] > arr[largest]) {
            largest = right;
        }
        
        if (largest != i) {
            swap(arr[i], arr[largest]);
            heapify(arr, n, largest);
        }
    }
};
```

### 2. 优先队列实现

```cpp
class PriorityQueue {
private:
    vector<int> heap;
    
    void heapifyUp(int index) {
        while (index > 0) {
            int parent = (index - 1) / 2;
            if (heap[index] <= heap[parent]) {
                break;
            }
            swap(heap[index], heap[parent]);
            index = parent;
        }
    }
    
    void heapifyDown(int index) {
        int n = heap.size();
        while (true) {
            int largest = index;
            int left = 2 * index + 1;
            int right = 2 * index + 2;
            
            if (left < n && heap[left] > heap[largest]) {
                largest = left;
            }
            
            if (right < n && heap[right] > heap[largest]) {
                largest = right;
            }
            
            if (largest == index) {
                break;
            }
            
            swap(heap[index], heap[largest]);
            index = largest;
        }
    }
    
public:
    // 插入元素
    void push(int value) {
        heap.push_back(value);
        heapifyUp(heap.size() - 1);
    }
    
    // 删除最大元素
    void pop() {
        if (heap.empty()) return;
        
        heap[0] = heap.back();
        heap.pop_back();
        
        if (!heap.empty()) {
            heapifyDown(0);
        }
    }
    
    // 获取最大元素
    int top() {
        if (heap.empty()) {
            throw runtime_error("Priority queue is empty");
        }
        return heap[0];
    }
    
    // 检查是否为空
    bool empty() {
        return heap.empty();
    }
    
    // 获取大小
    int size() {
        return heap.size();
    }
};
```

## 🔧 优化技巧

### 1. 减少比较次数

```cpp
class OptimizedHeapSort {
private:
    // 使用位运算优化除法
    void heapifyOptimized(vector<int>& arr, int n, int i) {
        int largest = i;
        int left = (i << 1) + 1;   // 等价于 2*i + 1
        int right = (i << 1) + 2;  // 等价于 2*i + 2
        
        if (left < n && arr[left] > arr[largest]) {
            largest = left;
        }
        
        if (right < n && arr[right] > arr[largest]) {
            largest = right;
        }
        
        if (largest != i) {
            swap(arr[i], arr[largest]);
            heapifyOptimized(arr, n, largest);
        }
    }
    
public:
    // 优化的堆排序
    void heapSortOptimized(vector<int>& arr) {
        int n = arr.size();
        
        // 建堆
        for (int i = (n >> 1) - 1; i >= 0; i--) {  // 等价于 n/2 - 1
            heapifyOptimized(arr, n, i);
        }
        
        // 排序
        for (int i = n - 1; i > 0; i--) {
            swap(arr[0], arr[i]);
            heapifyOptimized(arr, i, 0);
        }
    }
};
```

        }
    }
};

## 💡 堆排序要点

1. **时间复杂度**：O(n log n)，稳定
2. **空间复杂度**：O(1)，原地排序
3. **稳定性**：不稳定排序
4. **应用**：Top K问题、优先队列
