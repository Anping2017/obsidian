# 跳表详解

## 📖 核心概念

**跳表（Skip List）**是一种概率型数据结构，通过多层链表实现O(log n)的查找、插入和删除操作。它是对传统链表的改进，通过随机化技术保持平衡。

### 🏗️ 跳表结构

```mermaid
graph TD
    A[跳表结构] --> B[多层链表]
    A --> C[随机化层数]
    A --> D[跳跃指针]
    A --> E[性能特性]
    
    B --> B1[第0层：完整链表]
    B --> B2[第1层：部分节点]
    B --> B3[第2层：更少节点]
    
    C --> C1[概率决定层数]
    C --> C2[期望层数log n]
    
    D --> D1[向右指针]
    D --> D2[向下指针]
    
    E --> E1[O(log n)查找]
    E --> E2[O(log n)插入]
    E --> E3[O(log n)删除]
```

## 🔧 跳表实现

### 基础跳表

```cpp
class SkipList {
private:
    struct Node {
        int key;
        int value;
        vector<Node*> next;
        int level;
        
        Node(int k, int v, int l) : key(k), value(v), level(l) {
            next.resize(l + 1, nullptr);
        }
    };
    
    Node* head;
    int maxLevel;
    int currentLevel;
    double probability;
    
    // 生成随机层数
    int generateRandomLevel() {
        int level = 0;
        while (level < maxLevel && ((double)rand() / RAND_MAX) < probability) {
            level++;
        }
        return level;
    }
    
public:
    SkipList(int maxL = 16, double p = 0.5) : maxLevel(maxL), probability(p) {
        head = new Node(INT_MIN, 0, maxLevel);
        currentLevel = 0;
    }
    
    // 查找节点
    Node* search(int key) {
        Node* current = head;
        
        for (int i = currentLevel; i >= 0; i--) {
            while (current->next[i] && current->next[i]->key < key) {
                current = current->next[i];
            }
        }
        
        current = current->next[0];
        return (current && current->key == key) ? current : nullptr;
    }
    
    // 插入节点
    void insert(int key, int value) {
        vector<Node*> update(maxLevel + 1);
        Node* current = head;
        
        // 找到插入位置
        for (int i = currentLevel; i >= 0; i--) {
            while (current->next[i] && current->next[i]->key < key) {
                current = current->next[i];
            }
            update[i] = current;
        }
        
        current = current->next[0];
        
        // 如果键已存在，更新值
        if (current && current->key == key) {
            current->value = value;
            return;
        }
        
        // 生成新节点的层数
        int newLevel = generateRandomLevel();
        
        // 如果新层数超过当前层数，更新头节点
        if (newLevel > currentLevel) {
            for (int i = currentLevel + 1; i <= newLevel; i++) {
                update[i] = head;
            }
            currentLevel = newLevel;
        }
        
        // 创建新节点
        Node* newNode = new Node(key, value, newLevel);
        
        // 更新指针
        for (int i = 0; i <= newLevel; i++) {
            newNode->next[i] = update[i]->next[i];
            update[i]->next[i] = newNode;
        }
    }
    
    // 删除节点
    bool remove(int key) {
        vector<Node*> update(maxLevel + 1);
        Node* current = head;
        
        // 找到删除位置
        for (int i = currentLevel; i >= 0; i--) {
            while (current->next[i] && current->next[i]->key < key) {
                current = current->next[i];
            }
            update[i] = current;
        }
        
        current = current->next[0];
        
        // 如果节点不存在
        if (!current || current->key != key) {
            return false;
        }
        
        // 更新指针
        for (int i = 0; i <= currentLevel; i++) {
            if (update[i]->next[i] != current) {
                break;
            }
            update[i]->next[i] = current->next[i];
        }
        
        // 删除节点
        delete current;
        
        // 更新当前层数
        while (currentLevel > 0 && head->next[currentLevel] == nullptr) {
            currentLevel--;
        }
        
        return true;
    }
    
    // 显示跳表结构
    void display() {
        cout << "Skip List Structure:" << endl;
        cout << "==================" << endl;
        
        for (int i = currentLevel; i >= 0; i--) {
            cout << "Level " << i << ": ";
            Node* current = head->next[i];
            while (current) {
                cout << current->key << "(" << current->value << ") ";
                current = current->next[i];
            }
            cout << endl;
        }
    }
    
    // 显示统计信息
    void displayStats() {
        cout << "Skip List Statistics:" << endl;
        cout << "===================" << endl;
        
        int nodeCount = 0;
        int totalLevels = 0;
        
        Node* current = head->next[0];
        while (current) {
            nodeCount++;
            totalLevels += current->level + 1;
            current = current->next[0];
        }
        
        cout << "Number of nodes: " << nodeCount << endl;
        cout << "Current level: " << currentLevel << endl;
        cout << "Max level: " << maxLevel << endl;
        cout << "Average level: " << (nodeCount > 0 ? (double)totalLevels / nodeCount : 0) << endl;
        cout << "Probability: " << probability << endl;
    }
    
    // 获取所有键值对
    vector<pair<int, int>> getAll() {
        vector<pair<int, int>> result;
        Node* current = head->next[0];
        
        while (current) {
            result.push_back({current->key, current->value});
            current = current->next[0];
        }
        
        return result;
    }
};
```

### 并发跳表

```cpp
class ConcurrentSkipList {
private:
    struct Node {
        int key;
        int value;
        vector<Node*> next;
        int level;
        mutex nodeMutex;
        bool marked;
        bool fullyLinked;
        
        Node(int k, int v, int l) : key(k), value(v), level(l), marked(false), fullyLinked(false) {
            next.resize(l + 1, nullptr);
        }
    };
    
    Node* head;
    int maxLevel;
    int currentLevel;
    double probability;
    mutex levelMutex;
    
    // 生成随机层数
    int generateRandomLevel() {
        int level = 0;
        while (level < maxLevel && ((double)rand() / RAND_MAX) < probability) {
            level++;
        }
        return level;
    }
    
    // 查找节点（无锁）
    pair<Node*, vector<Node*>> findNode(int key) {
        vector<Node*> preds(maxLevel + 1);
        vector<Node*> succs(maxLevel + 1);
        
        while (true) {
            Node* pred = head;
            bool found = true;
            
            for (int i = currentLevel; i >= 0; i--) {
                Node* curr = pred->next[i];
                while (curr && curr->key < key) {
                    pred = curr;
                    curr = pred->next[i];
                }
                preds[i] = pred;
                succs[i] = curr;
                
                if (curr && curr->key == key) {
                    found = false;
                }
            }
            
            if (found) {
                return {nullptr, preds};
            }
            
            Node* node = succs[0];
            if (node && node->key == key && node->fullyLinked && !node->marked) {
                return {node, preds};
            }
        }
    }
    
public:
    ConcurrentSkipList(int maxL = 16, double p = 0.5) : maxLevel(maxL), probability(p) {
        head = new Node(INT_MIN, 0, maxLevel);
        currentLevel = 0;
    }
    
    // 查找节点
    Node* search(int key) {
        auto result = findNode(key);
        return result.first;
    }
    
    // 插入节点
    bool insert(int key, int value) {
        int newLevel = generateRandomLevel();
        
        while (true) {
            auto result = findNode(key);
            Node* node = result.first;
            vector<Node*> preds = result.second;
            
            if (node) {
                // 节点已存在，更新值
                node->value = value;
                return false;
            }
            
            // 创建新节点
            Node* newNode = new Node(key, value, newLevel);
            
            // 更新指针
            for (int i = 0; i <= newLevel; i++) {
                newNode->next[i] = preds[i]->next[i];
            }
            
            // 原子更新
            for (int i = 0; i <= newLevel; i++) {
                preds[i]->next[i] = newNode;
            }
            
            newNode->fullyLinked = true;
            return true;
        }
    }
    
    // 删除节点
    bool remove(int key) {
        while (true) {
            auto result = findNode(key);
            Node* node = result.first;
            vector<Node*> preds = result.second;
            
            if (!node) {
                return false;
            }
            
            // 标记节点为删除
            node->marked = true;
            
            // 更新指针
            for (int i = 0; i <= node->level; i++) {
                preds[i]->next[i] = node->next[i];
            }
            
            delete node;
            return true;
        }
    }
    
    // 显示并发跳表结构
    void display() {
        cout << "Concurrent Skip List Structure:" << endl;
        cout << "=============================" << endl;
        
        for (int i = currentLevel; i >= 0; i--) {
            cout << "Level " << i << ": ";
            Node* current = head->next[i];
            while (current) {
                cout << current->key << "(" << current->value << ") ";
                current = current->next[i];
            }
            cout << endl;
        }
    }
    
    // 显示统计信息
    void displayStats() {
        cout << "Concurrent Skip List Statistics:" << endl;
        cout << "==============================" << endl;
        
        int nodeCount = 0;
        int totalLevels = 0;
        
        Node* current = head->next[0];
        while (current) {
            nodeCount++;
            totalLevels += current->level + 1;
            current = current->next[0];
        }
        
        cout << "Number of nodes: " << nodeCount << endl;
        cout << "Current level: " << currentLevel << endl;
        cout << "Max level: " << maxLevel << endl;
        cout << "Average level: " << (nodeCount > 0 ? (double)totalLevels / nodeCount : 0) << endl;
        cout << "Probability: " << probability << endl;
    }
};
```

class SkipListUtils {
public:
    // 性能测试
    static void performanceTest(SkipList& skipList, int n) {
        cout << "Skip List Performance Test:" << endl;
        cout << "=========================" << endl;
        
        // 插入测试
        auto start = chrono::high_resolution_clock::now();
        for (int i = 0; i < n; i++) {
            skipList.insert(i, i * 2);
        }
        auto end = chrono::high_resolution_clock::now();
        auto insertTime = chrono::duration_cast<chrono::microseconds>(end - start);
        
        // 查找测试
        start = chrono::high_resolution_clock::now();
        for (int i = 0; i < n; i++) {
            skipList.search(i);
        }
        end = chrono::high_resolution_clock::now();
        auto searchTime = chrono::duration_cast<chrono::microseconds>(end - start);
        
        // 删除测试
        start = chrono::high_resolution_clock::now();
        for (int i = 0; i < n; i++) {
            skipList.remove(i);
        }
        end = chrono::high_resolution_clock::now();
        auto deleteTime = chrono::duration_cast<chrono::microseconds>(end - start);
        
        cout << "Insert time: " << insertTime.count() << " microseconds" << endl;
        cout << "Search time: " << searchTime.count() << " microseconds" << endl;
        cout << "Delete time: " << deleteTime.count() << " microseconds" << endl;
        cout << "Average insert time: " << (double)insertTime.count() / n << " microseconds" << endl;
        cout << "Average search time: " << (double)searchTime.count() / n << " microseconds" << endl;
        cout << "Average delete time: " << (double)deleteTime.count() / n << " microseconds" << endl;
    }
    
    // 内存使用分析
    static void memoryAnalysis(SkipList& skipList) {
        cout << "Skip List Memory Analysis:" << endl;
        cout << "========================" << endl;
        
        int nodeCount = 0;
        int totalLevels = 0;
        
        SkipList::Node* current = skipList.head->next[0];
        while (current) {
            nodeCount++;
            totalLevels += current->level + 1;
            current = current->next[0];
        }
        
        cout << "Number of nodes: " << nodeCount << endl;
        cout << "Total levels: " << totalLevels << endl;
        cout << "Average level: " << (nodeCount > 0 ? (double)totalLevels / nodeCount : 0) << endl;
        cout << "Memory per node: " << sizeof(SkipList::Node) << " bytes" << endl;
        cout << "Total memory: " << nodeCount * sizeof(SkipList::Node) << " bytes" << endl;
    }
    
    // 概率分析
    static void probabilityAnalysis(double p) {
        cout << "Skip List Probability Analysis:" << endl;
        cout << "============================" << endl;
        
        cout << "Probability: " << p << endl;
        cout << "Expected level: " << (1.0 / (1.0 - p)) << endl;
        cout << "Level 0 probability: " << (1.0 - p) << endl;
        cout << "Level 1 probability: " << p * (1.0 - p) << endl;
        cout << "Level 2 probability: " << p * p * (1.0 - p) << endl;
        cout << "Level 3 probability: " << p * p * p * (1.0 - p) << endl;
    }
    
    // 显示跳表可视化
    static void visualizeSkipList(SkipList& skipList) {
        cout << "Skip List Visualization:" << endl;
        cout << "=======================" << endl;
        
        vector<pair<int, int>> elements = skipList.getAll();
        
        cout << "Elements: ";
        for (auto& pair : elements) {
            cout << pair.first << " ";
        }
        cout << endl;
        
        skipList.display();
    }
    
    // 比较不同概率
    static void compareProbabilities() {
        cout << "Skip List Probability Comparison:" << endl;
        cout << "==============================" << endl;
        
        vector<double> probabilities = {0.25, 0.5, 0.75};
        
        for (double p : probabilities) {
            cout << "Probability " << p << ":" << endl;
            probabilityAnalysis(p);
            cout << endl;
        }
    }
};

## 💡 跳表要点

1. **概率型结构**: 通过随机化保持平衡
2. **多层链表**: 从高层快速定位，从低层精确查找
3. **O(log n)性能**: 查找、插入、删除都是对数时间
4. **并发友好**: 支持无锁并发访问
