# 最长公共子序列

## 📖 核心概念

**最长公共子序列（LCS）**是动态规划中的经典问题，用于找到两个序列中最长的公共子序列。子序列不要求连续，但必须保持原有顺序。

### 🏗️ LCS问题分类

```mermaid

    B --> B1[二维DP]
    B --> B2[O(m*n)空间]

    C --> C2[O(min(m,n))空间]
    
    D --> D1[三个序列]
    D --> D2[多个序列]
    
    E --> E1[最长公共子串]
    E --> E2[编辑距离]
```

## 🔧 LCS实现

### 基础LCS

```cpp
class LCSBasic {
public:
    // 基础LCS算法
    int longestCommonSubsequence(string text1, string text2) {
        int m = text1.length();
        int n = text2.length();
        
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
        
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (text1[i - 1] == text2[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                } else {
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
                }
            }
        }
        
        return dp[m][n];
    }
    
    // 获取LCS序列
    string getLCS(string text1, string text2) {
        int m = text1.length();
        int n = text2.length();
        
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
        
        // 构建DP表
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (text1[i - 1] == text2[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                } else {
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
                }
            }
        }
        
        // 回溯构建LCS
        string result = "";
        int i = m, j = n;
        
        while (i > 0 && j > 0) {
            if (text1[i - 1] == text2[j - 1]) {
                result = text1[i - 1] + result;
                i--;
                j--;
            } else if (dp[i - 1][j] > dp[i][j - 1]) {
                i--;
            } else {
                j--;
            }
        }
        
        return result;
    }
    
    // 显示LCS过程
    void displayLCS(string text1, string text2) {
        cout << "Longest Common Subsequence:" << endl;
        cout << "=========================" << endl;
        
        cout << "Text 1: " << text1 << endl;
        cout << "Text 2: " << text2 << endl;
        
        int length = longestCommonSubsequence(text1, text2);
        string sequence = getLCS(text1, text2);
        
        cout << "LCS Length: " << length << endl;
        cout << "LCS Sequence: " << sequence << endl;
    }
    
    // 显示DP表
    void displayDPTable(string text1, string text2) {
        cout << "DP Table:" << endl;
        cout << "========" << endl;
        
        int m = text1.length();
        int n = text2.length();
        
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
        
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (text1[i - 1] == text2[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                } else {
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
                }
            }
        }
        
        // 打印表头
        cout << "   ";
        for (int j = 0; j <= n; ++j) {
            if (j == 0) cout << "  ";
            else cout << text2[j - 1] << " ";
        }
        cout << endl;
        
        // 打印DP表
        for (int i = 0; i <= m; ++i) {
            if (i == 0) cout << "  ";
            else cout << text1[i - 1] << " ";
            
            for (int j = 0; j <= n; ++j) {
                cout << dp[i][j] << " ";
            }
            cout << endl;
        }
    }
};
```

### 空间优化LCS

```cpp
class LCSOptimized {
public:
    // 空间优化LCS
    int longestCommonSubsequenceOptimized(string text1, string text2) {
        int m = text1.length();
        int n = text2.length();
        
        if (m < n) {
            swap(text1, text2);
            swap(m, n);
        }
        
        vector<int> prev(n + 1, 0);
        vector<int> curr(n + 1, 0);
        
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (text1[i - 1] == text2[j - 1]) {
                    curr[j] = prev[j - 1] + 1;
                } else {
                    curr[j] = max(prev[j], curr[j - 1]);
                }
            }
            prev = curr;
        }
        
        return curr[n];
    }
    
    // 获取LCS序列（空间优化版本）
    string getLCSOptimized(string text1, string text2) {
        int m = text1.length();
        int n = text2.length();
        
        // 先计算长度
        int length = longestCommonSubsequenceOptimized(text1, text2);
        
        // 使用滚动数组重建LCS
        vector<vector<int>> dp(2, vector<int>(n + 1, 0));
        
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (text1[i - 1] == text2[j - 1]) {
                    dp[i % 2][j] = dp[(i - 1) % 2][j - 1] + 1;
                } else {
                    dp[i % 2][j] = max(dp[(i - 1) % 2][j], dp[i % 2][j - 1]);
                }
            }
        }
        
        // 回溯构建LCS
        string result = "";
        int i = m, j = n;
        
        while (i > 0 && j > 0) {
            if (text1[i - 1] == text2[j - 1]) {
                result = text1[i - 1] + result;
                i--;
                j--;
            } else if (dp[i % 2][j - 1] > dp[(i - 1) % 2][j]) {
                j--;
            } else {
                i--;
            }
        }
        
        return result;
    }
    
    // 显示空间优化LCS
    void displayOptimizedLCS(string text1, string text2) {
        cout << "Space Optimized LCS:" << endl;
        cout << "==================" << endl;
        
        cout << "Text 1: " << text1 << endl;
        cout << "Text 2: " << text2 << endl;
        
        int length = longestCommonSubsequenceOptimized(text1, text2);
        string sequence = getLCSOptimized(text1, text2);
        
        cout << "LCS Length: " << length << endl;
        cout << "LCS Sequence: " << sequence << endl;
        cout << "Space Complexity: O(min(m,n)) instead of O(m*n)" << endl;
    }
};
```

### 多序列LCS

```cpp
class LCSMultiple {
public:
    // 三个序列的LCS
    int longestCommonSubsequence3(string text1, string text2, string text3) {
        int m = text1.length();
        int n = text2.length();
        int p = text3.length();
        
        vector<vector<vector<int>>> dp(m + 1, vector<vector<int>>(n + 1, vector<int>(p + 1, 0)));
        
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                for (int k = 1; k <= p; ++k) {
                    if (text1[i - 1] == text2[j - 1] && text2[j - 1] == text3[k - 1]) {
                        dp[i][j][k] = dp[i - 1][j - 1][k - 1] + 1;
                    } else {
                        dp[i][j][k] = max({dp[i - 1][j][k], dp[i][j - 1][k], dp[i][j][k - 1]});
                    }
                }
            }
        }
        
        return dp[m][n][p];
    }
    
    // 多个序列的LCS
    int longestCommonSubsequenceMultiple(vector<string>& texts) {
        if (texts.empty()) return 0;
        if (texts.size() == 1) return texts[0].length();
        if (texts.size() == 2) {
            LCSBasic lcs;
            return lcs.longestCommonSubsequence(texts[0], texts[1]);
        }
        
        // 对于多个序列，使用递归方法
        return lcsMultipleRecursive(texts, 0, texts.size() - 1);
    }
    
private:
    int lcsMultipleRecursive(vector<string>& texts, int left, int right) {
        if (left == right) return texts[left].length();
        
        int mid = (left + right) / 2;
        int leftLCS = lcsMultipleRecursive(texts, left, mid);
        int rightLCS = lcsMultipleRecursive(texts, mid + 1, right);
        
        // 合并两个LCS
        LCSBasic lcs;
        return lcs.longestCommonSubsequence(texts[left], texts[right]);
    }
    
public:
    // 显示多序列LCS
    void displayMultipleLCS(vector<string>& texts) {
        cout << "Multiple Sequence LCS:" << endl;
        cout << "====================" << endl;
        
        for (int i = 0; i < texts.size(); ++i) {
            cout << "Text " << (i + 1) << ": " << texts[i] << endl;
        }
        
        int result = longestCommonSubsequenceMultiple(texts);
        cout << "LCS Length: " << result << endl;
    }
    
    void display3SequenceLCS(string text1, string text2, string text3) {
        cout << "Three Sequence LCS:" << endl;
        cout << "=================" << endl;
        
        cout << "Text 1: " << text1 << endl;
        cout << "Text 2: " << text2 << endl;
        cout << "Text 3: " << text3 << endl;
        
        int result = longestCommonSubsequence3(text1, text2, text3);
        cout << "LCS Length: " << result << endl;
    }
};
```

### LCS变种

```cpp
class LCSVariants {
public:
    // 最长公共子串（连续）
    int longestCommonSubstring(string text1, string text2) {
        int m = text1.length();
        int n = text2.length();
        
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
        int maxLength = 0;
        
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (text1[i - 1] == text2[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                    maxLength = max(maxLength, dp[i][j]);
                } else {
                    dp[i][j] = 0;
                }
            }
        }
        
        return maxLength;
    }
    
    // 获取最长公共子串
    string getLongestCommonSubstring(string text1, string text2) {
        int m = text1.length();
        int n = text2.length();
        
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
        int maxLength = 0;
        int endIndex = 0;
        
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (text1[i - 1] == text2[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                    if (dp[i][j] > maxLength) {
                        maxLength = dp[i][j];
                        endIndex = i - 1;
                    }
                } else {
                    dp[i][j] = 0;
                }
            }
        }
        
        return text1.substr(endIndex - maxLength + 1, maxLength);
    }
    
    // 编辑距离
    int editDistance(string word1, string word2) {
        int m = word1.length();
        int n = word2.length();
        
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
        
        for (int i = 0; i <= m; ++i) {
            dp[i][0] = i;
        }
        
        for (int j = 0; j <= n; ++j) {
            dp[0][j] = j;
        }
        
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (word1[i - 1] == word2[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1];
                } else {
                    dp[i][j] = 1 + min({dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1]});
                }
            }
        }
        
        return dp[m][n];
    }
    
    // 显示LCS变种
    void displayLongestCommonSubstring(string text1, string text2) {
        cout << "Longest Common Substring:" << endl;
        cout << "=======================" << endl;
        
        cout << "Text 1: " << text1 << endl;
        cout << "Text 2: " << text2 << endl;
        
        int length = longestCommonSubstring(text1, text2);
        string substring = getLongestCommonSubstring(text1, text2);
        
        cout << "Length: " << length << endl;
        cout << "Substring: " << substring << endl;
    }
    
    void displayEditDistance(string word1, string word2) {
        cout << "Edit Distance:" << endl;
        cout << "=============" << endl;
        
        cout << "Word 1: " << word1 << endl;
        cout << "Word 2: " << word2 << endl;
        
        int distance = editDistance(word1, word2);
        cout << "Edit distance: " << distance << endl;
    }
};
```

## 💡 LCS要点

1. **状态定义**: dp[i][j]表示text1[0:i]和text2[0:j]的LCS长度
2. **状态转移**: 字符相等时+1，否则取最大值
3. **空间优化**: 使用滚动数组减少空间复杂度
4. **回溯重建**: 从DP表反向构建LCS序列
