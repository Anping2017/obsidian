# 搜索技术

## 📖 核心概念

**搜索技术**是算法设计中的重要分支，包括线性搜索、二分搜索、深度优先搜索、广度优先搜索等多种搜索策略。

### 基本线性搜索

```cpp
class LinearSearch {
public:
    // 基本线性搜索
    static int search(const vector<int>& arr, int target) {
        for (int i = 0; i < arr.size(); ++i) {
            if (arr[i] == target) {
                return i;
            }
        }
        return -1;
    }
    
    // 查找所有匹配项
    static vector<int> searchAll(const vector<int>& arr, int target) {
        vector<int> indices;
        for (int i = 0; i < arr.size(); ++i) {
            if (arr[i] == target) {
                indices.push_back(i);
            }
        }
        return indices;
    }
    
    // 查找第一个匹配项
    static int findFirst(const vector<int>& arr, int target) {
        for (int i = 0; i < arr.size(); ++i) {
            if (arr[i] == target) {
                return i;
            }
        }
        return -1;
    }
    
    // 查找最后一个匹配项
    static int findLast(const vector<int>& arr, int target) {
        for (int i = arr.size() - 1; i >= 0; --i) {
            if (arr[i] == target) {
                return i;
            }
        }
        return -1;
    }
    
    // 查找最大值
    static int findMax(const vector<int>& arr) {
        if (arr.empty()) return -1;
        
        int maxVal = arr[0];
        for (int i = 1; i < arr.size(); ++i) {
            if (arr[i] > maxVal) {
                maxVal = arr[i];
            }
        }
        return maxVal;
    }
    
    // 查找最小值
    static int findMin(const vector<int>& arr) {
        if (arr.empty()) return -1;
        
        int minVal = arr[0];
        for (int i = 1; i < arr.size(); ++i) {
            if (arr[i] < minVal) {
                minVal = arr[i];
            }
        }
        return minVal;
    }
};
```

### 优化线性搜索

```cpp
class OptimizedLinearSearch {
public:
    // 哨兵搜索
    static int sentinelSearch(vector<int>& arr, int target) {
        int n = arr.size();
        if (n == 0) return -1;
        
        int last = arr[n - 1];
        arr[n - 1] = target;
        
        int i = 0;
        while (arr[i] != target) {
            i++;
        }
        
        arr[n - 1] = last;
        
        if (i < n - 1 || arr[n - 1] == target) {
            return i;
        }
        return -1;
    }
    
    // 移动到头搜索
    static int moveToFrontSearch(vector<int>& arr, int target) {
        for (int i = 0; i < arr.size(); ++i) {
            if (arr[i] == target) {
                if (i != 0) {
                    swap(arr[0], arr[i]);
                }
                return i;
            }
        }
        return -1;
    }
    
    // 转置搜索
    static int transposeSearch(vector<int>& arr, int target) {
        for (int i = 0; i < arr.size(); ++i) {
            if (arr[i] == target) {
                if (i != 0) {
                    swap(arr[i], arr[i - 1]);
                }
                return i;
            }
        }
        return -1;
    }
};
```

## 🎯 二分搜索

### 基本二分搜索

```cpp
class BinarySearch {
public:
    // 基本二分搜索
    static int search(const vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] == target) {
                return mid;
            } else if (arr[mid] < target) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        
        return -1;
    }
    
    // 递归二分搜索
    static int searchRecursive(const vector<int>& arr, int target, int left, int right) {
        if (left > right) return -1;
        
        int mid = left + (right - left) / 2;
        
        if (arr[mid] == target) {
            return mid;
        } else if (arr[mid] < target) {
            return searchRecursive(arr, target, mid + 1, right);
        } else {
            return searchRecursive(arr, target, left, mid - 1);
        }
    }
    
    // 查找第一个位置
    static int findFirst(const vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        int result = -1;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] == target) {
                result = mid;
                right = mid - 1;
            } else if (arr[mid] < target) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        
        return result;
    }
    
    // 查找最后一个位置
    static int findLast(const vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        int result = -1;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] == target) {
                result = mid;
                left = mid + 1;
            } else if (arr[mid] < target) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        
        return result;
    }
    
    // 查找插入位置
    static int findInsertPosition(const vector<int>& arr, int target) {
        int left = 0, right = arr.size();
        
        while (left < right) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] < target) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }
        
        return left;
    }
};
```

### 高级二分搜索

```cpp
class AdvancedBinarySearch {
public:
    // 在旋转数组中搜索
    static int searchInRotatedArray(const vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] == target) {
                return mid;
            }
            
            if (arr[left] <= arr[mid]) {
                if (target >= arr[left] && target < arr[mid]) {
                    right = mid - 1;
                } else {
                    left = mid + 1;
                }
            } else {
                if (target > arr[mid] && target <= arr[right]) {
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
        }
        
        return -1;
    }
    
    // 查找峰值元素
    static int findPeakElement(const vector<int>& arr) {
        int left = 0, right = arr.size() - 1;
        
        while (left < right) {
            int mid = left + (right - left) / 2;
            
            if (arr[mid] > arr[mid + 1]) {
                right = mid;
            } else {
                left = mid + 1;
            }
        }
        
        return left;
    }
    
    // 查找平方根
    static int sqrt(int x) {
        if (x < 2) return x;
        
        int left = 2, right = x / 2;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;
            long square = (long)mid * mid;
            
            if (square == x) {
                return mid;
            } else if (square < x) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        
        return right;
    }
};
```

## 📊 深度优先搜索

### DFS实现

```cpp
class DepthFirstSearch {
public:
    // 图的DFS
    static vector<int> dfs(vector<vector<int>>& graph, int start) {
        vector<int> result;
        vector<bool> visited(graph.size(), false);
        
        dfsHelper(graph, start, visited, result);
        return result;
    }
    
    static void dfsHelper(vector<vector<int>>& graph, int vertex, vector<bool>& visited, vector<int>& result) {
        visited[vertex] = true;
        result.push_back(vertex);
        
        for (int neighbor : graph[vertex]) {
            if (!visited[neighbor]) {
                dfsHelper(graph, neighbor, visited, result);
            }
        }
    }
    
    // 迭代DFS
    static vector<int> dfsIterative(vector<vector<int>>& graph, int start) {
        vector<int> result;
        vector<bool> visited(graph.size(), false);
        stack<int> stk;
        
        stk.push(start);
        
        while (!stk.empty()) {
            int vertex = stk.top();
            stk.pop();
            
            if (!visited[vertex]) {
                visited[vertex] = true;
                result.push_back(vertex);
                
                for (int neighbor : graph[vertex]) {
                    if (!visited[neighbor]) {
                        stk.push(neighbor);
                    }
                }
            }
        }
        
        return result;
    }
    
    // 路径查找
    static bool hasPath(vector<vector<int>>& graph, int from, int to) {
        vector<bool> visited(graph.size(), false);
        return hasPathHelper(graph, from, to, visited);
    }
    
    static bool hasPathHelper(vector<vector<int>>& graph, int from, int to, vector<bool>& visited) {
        if (from == to) return true;
        
        visited[from] = true;
        
        for (int neighbor : graph[from]) {
            if (!visited[neighbor] && hasPathHelper(graph, neighbor, to, visited)) {
                return true;
            }
        }
        
        return false;
    }
    
    // 连通分量
    static int countConnectedComponents(vector<vector<int>>& graph) {
        vector<bool> visited(graph.size(), false);
        int count = 0;
        
        for (int i = 0; i < graph.size(); ++i) {
            if (!visited[i]) {
                dfsHelper(graph, i, visited, vector<int>());
                count++;
            }
        }
        
        return count;
    }
};
```

## 🎮 广度优先搜索

### BFS实现

```cpp
class BreadthFirstSearch {
public:
    // 图的BFS
    static vector<int> bfs(vector<vector<int>>& graph, int start) {
        vector<int> result;
        vector<bool> visited(graph.size(), false);
        queue<int> q;
        
        q.push(start);
        visited[start] = true;
        
        while (!q.empty()) {
            int vertex = q.front();
            q.pop();
            result.push_back(vertex);
            
            for (int neighbor : graph[vertex]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    q.push(neighbor);
                }
            }
        }
        
        return result;
    }
    
    // 最短路径
    static int shortestPath(vector<vector<int>>& graph, int from, int to) {
        vector<bool> visited(graph.size(), false);
        vector<int> distance(graph.size(), -1);
        queue<int> q;
        
        q.push(from);
        visited[from] = true;
        distance[from] = 0;
        
        while (!q.empty()) {
            int vertex = q.front();
            q.pop();
            
            if (vertex == to) {
                return distance[vertex];
            }
            
            for (int neighbor : graph[vertex]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    distance[neighbor] = distance[vertex] + 1;
                    q.push(neighbor);
                }
            }
        }
        
        return -1;
    }
    
    // 层序遍历
    static vector<vector<int>> levelOrder(vector<vector<int>>& graph, int start) {
        vector<vector<int>> result;
        vector<bool> visited(graph.size(), false);
        queue<int> q;
        
        q.push(start);
        visited[start] = true;
        
        while (!q.empty()) {
            int levelSize = q.size();
            vector<int> level;
            
            for (int i = 0; i < levelSize; ++i) {
                int vertex = q.front();
                q.pop();
                level.push_back(vertex);
                
                for (int neighbor : graph[vertex]) {
                    if (!visited[neighbor]) {
                        visited[neighbor] = true;
                        q.push(neighbor);
                    }
                }
            }
            
            result.push_back(level);
        }
        
        return result;
    }
};
```

## 💡 搜索技术要点

1. **线性搜索**：适用于无序数据，O(n)复杂度
2. **二分搜索**：适用于有序数据，O(log n)复杂度
3. **深度优先搜索**：递归或栈实现，适合路径查找
4. **广度优先搜索**：队列实现，适合最短路径
