# 其他排序算法

## 📖 核心概念

**其他排序算法**包括一些特殊的排序方法，如计数排序、基数排序、桶排序等。这些算法在某些特定条件下具有优秀的性能表现。

### 🏗️ 其他排序算法分类

```mermaid

    A[其他排序算法] --> B[计数排序]
    A --> C[基数排序]
    A --> D[桶排序]
    A --> E[希尔排序]
    
    B --> B1[O(n+k)]
    B --> B2[非比较排序]
    
    C --> C1[O(d(n+k))]
    C --> C2[按位排序]

    D --> D2[分桶排序]
    
    E --> E1[O(n^1.3)]
    E --> E2[插入排序改进]
```

### 计数排序

```cpp
class CountingSort {
public:
    // 计数排序
    void countingSort(vector<int>& arr) {
        if (arr.empty()) return;
        
        int maxVal = *max_element(arr.begin(), arr.end());
        int minVal = *min_element(arr.begin(), arr.end());
        int range = maxVal - minVal + 1;
        
        vector<int> count(range, 0);
        vector<int> output(arr.size());
        
        // 计数
        for (int num : arr) {
            count[num - minVal]++;
        }
        
        // 累加计数
        for (int i = 1; i < range; ++i) {
            count[i] += count[i - 1];
        }
        
        // 构建输出数组
        for (int i = arr.size() - 1; i >= 0; --i) {
            output[count[arr[i] - minVal] - 1] = arr[i];
            count[arr[i] - minVal]--;
        }
        
        // 复制回原数组
        for (int i = 0; i < arr.size(); ++i) {
            arr[i] = output[i];
        }
    }
    
    // 计数排序（稳定版本）
    void countingSortStable(vector<int>& arr) {
        if (arr.empty()) return;
        
        int maxVal = *max_element(arr.begin(), arr.end());
        int minVal = *min_element(arr.begin(), arr.end());
        int range = maxVal - minVal + 1;
        
        vector<int> count(range, 0);
        
        // 计数
        for (int num : arr) {
            count[num - minVal]++;
        }
        
        // 重建数组
        int index = 0;
        for (int i = 0; i < range; ++i) {
            while (count[i] > 0) {
                arr[index++] = i + minVal;
                count[i]--;
            }
        }
    }
    
    // 计数排序（带索引）
    vector<int> countingSortWithIndex(vector<int>& arr) {
        if (arr.empty()) return {};
        
        int maxVal = *max_element(arr.begin(), arr.end());
        int minVal = *min_element(arr.begin(), arr.end());
        int range = maxVal - minVal + 1;
        
        vector<int> count(range, 0);
        vector<int> output(arr.size());
        vector<int> index(arr.size());
        
        // 初始化索引
        for (int i = 0; i < arr.size(); ++i) {
            index[i] = i;
        }
        
        // 计数
        for (int num : arr) {
            count[num - minVal]++;
        }
        
        // 累加计数
        for (int i = 1; i < range; ++i) {
            count[i] += count[i - 1];
        }
        
        // 构建输出数组
        for (int i = arr.size() - 1; i >= 0; --i) {
            output[count[arr[i] - minVal] - 1] = arr[i];
            count[arr[i] - minVal]--;
        }
        
        return output;
    }
    
    // 显示计数排序过程
    void displayCountingSort(vector<int>& arr) {
        cout << "Counting Sort Process:" << endl;
        cout << "====================" << endl;
        
        cout << "Original array: ";
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
        
        int maxVal = *max_element(arr.begin(), arr.end());
        int minVal = *min_element(arr.begin(), arr.end());
        int range = maxVal - minVal + 1;
        
        vector<int> count(range, 0);
        
        // 计数
        for (int num : arr) {

        }
        
        cout << "Count array: ";
```

### 基数排序

```cpp
class RadixSort {
public:
    // 基数排序
    void radixSort(vector<int>& arr) {
        if (arr.empty()) return;
        
        int maxVal = *max_element(arr.begin(), arr.end());
        
        // 按位排序
        for (int exp = 1; maxVal / exp > 0; exp *= 10) {
            countingSortByDigit(arr, exp);
        }
    }
    
    // 按位计数排序
    void countingSortByDigit(vector<int>& arr, int exp) {
        vector<int> output(arr.size());
        vector<int> count(10, 0);
        
        // 计数
        for (int num : arr) {
            count[(num / exp) % 10]++;
        }
        
        // 累加计数
        for (int i = 1; i < 10; ++i) {
            count[i] += count[i - 1];
        }
        
        // 构建输出数组
        for (int i = arr.size() - 1; i >= 0; --i) {
            output[count[(arr[i] / exp) % 10] - 1] = arr[i];
            count[(arr[i] / exp) % 10]--;
        }
        
        // 复制回原数组
        for (int i = 0; i < arr.size(); ++i) {
            arr[i] = output[i];
        }
    }
    
    // 基数排序（字符串版本）
    void radixSortStrings(vector<string>& arr) {
        if (arr.empty()) return;
        
        int maxLen = 0;
        for (const string& str : arr) {
            maxLen = max(maxLen, (int)str.length());
        }
        
        // 按字符排序
        for (int pos = maxLen - 1; pos >= 0; --pos) {
            countingSortByChar(arr, pos);
        }
    }
    
    // 按字符计数排序
    void countingSortByChar(vector<string>& arr, int pos) {
        vector<string> output(arr.size());
        vector<int> count(256, 0);
        
        // 计数
        for (const string& str : arr) {
            char c = (pos < str.length()) ? str[pos] : 0;
            count[c]++;
        }
        
        // 累加计数
        for (int i = 1; i < 256; ++i) {
            count[i] += count[i - 1];
        }
        
        // 构建输出数组
        for (int i = arr.size() - 1; i >= 0; --i) {
            char c = (pos < arr[i].length()) ? arr[i][pos] : 0;
            output[count[c] - 1] = arr[i];
            count[c]--;
        }
        
        // 复制回原数组
        for (int i = 0; i < arr.size(); ++i) {
            arr[i] = output[i];
        }
    }
    
    // 显示基数排序过程
    void displayRadixSort(vector<int>& arr) {
        cout << "Radix Sort Process:" << endl;
        cout << "==================" << endl;
        
        cout << "Original array: ";
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
        
        int maxVal = *max_element(arr.begin(), arr.end());
        int exp = 1;
        
        while (maxVal / exp > 0) {
            cout << "Sorting by digit " << exp << ":" << endl;
            
            countingSortByDigit(arr, exp);
            
            cout << "After sorting: ";
            for (int num : arr) {
                cout << num << " ";
            }
            cout << endl;
            
            exp *= 10;
        }
        
        cout << "Final sorted array: ";
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
    }
};
```

### 桶排序

```cpp
class BucketSort {
public:
    // 桶排序
    void bucketSort(vector<int>& arr) {
        if (arr.empty()) return;
        
        int maxVal = *max_element(arr.begin(), arr.end());
        int minVal = *min_element(arr.begin(), arr.end());
        int range = maxVal - minVal + 1;
        
        int bucketSize = range / arr.size() + 1;
        vector<vector<int>> buckets(bucketSize);
        
        // 分配到桶
        for (int num : arr) {
            int bucketIndex = (num - minVal) / bucketSize;
            buckets[bucketIndex].push_back(num);
        }
        
        // 对每个桶排序
        for (auto& bucket : buckets) {
            sort(bucket.begin(), bucket.end());
        }
        
        // 合并桶
        int index = 0;
        for (const auto& bucket : buckets) {
            for (int num : bucket) {
                arr[index++] = num;
            }
        }
    }
    
    // 桶排序（浮点数版本）
    void bucketSortFloat(vector<double>& arr) {
        if (arr.empty()) return;
        
        int n = arr.size();
        vector<vector<double>> buckets(n);
        
        // 分配到桶
        for (double num : arr) {
            int bucketIndex = n * num;
            buckets[bucketIndex].push_back(num);
        }
        
        // 对每个桶排序
        for (auto& bucket : buckets) {
            sort(bucket.begin(), bucket.end());
        }
        
        // 合并桶
        int index = 0;
        for (const auto& bucket : buckets) {
            for (double num : bucket) {
                arr[index++] = num;
            }
        }
    }
    
    // 桶排序（字符串版本）
    void bucketSortStrings(vector<string>& arr) {
        if (arr.empty()) return;
        
        int maxLen = 0;
        for (const string& str : arr) {
            maxLen = max(maxLen, (int)str.length());
        }
        
        vector<vector<string>> buckets(maxLen + 1);
        
        // 按长度分配到桶
        for (const string& str : arr) {
            buckets[str.length()].push_back(str);
        }
        
        // 对每个桶排序
        for (auto& bucket : buckets) {
            sort(bucket.begin(), bucket.end());
        }
        
        // 合并桶
        int index = 0;
        for (const auto& bucket : buckets) {
            for (const string& str : bucket) {
                arr[index++] = str;
            }
        }
    }
    
    // 显示桶排序过程
    void displayBucketSort(vector<int>& arr) {
        cout << "Bucket Sort Process:" << endl;
        cout << "===================" << endl;
        
        cout << "Original array: ";
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
        
        int maxVal = *max_element(arr.begin(), arr.end());
        int minVal = *min_element(arr.begin(), arr.end());
        int range = maxVal - minVal + 1;
        
        int bucketSize = range / arr.size() + 1;
        vector<vector<int>> buckets(bucketSize);
        
        // 分配到桶
        for (int num : arr) {
            int bucketIndex = (num - minVal) / bucketSize;
            buckets[bucketIndex].push_back(num);
        }
        
        cout << "Buckets:" << endl;
        for (int i = 0; i < buckets.size(); ++i) {
            if (!buckets[i].empty()) {
                cout << "Bucket " << i << ": ";
                for (int num : buckets[i]) {
                    cout << num << " ";
                }
                cout << endl;
            }
        }
        
        bucketSort(arr);
        
        cout << "Sorted array: ";
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
    }
};
```

### 希尔排序

```cpp
class ShellSort {
public:
    // 希尔排序
    void shellSort(vector<int>& arr) {
        int n = arr.size();
        
        // 使用Knuth序列
        int gap = 1;
        while (gap < n / 3) {
            gap = gap * 3 + 1;
        }
        
        while (gap > 0) {
            for (int i = gap; i < n; ++i) {
                int temp = arr[i];
                int j = i;
                
                while (j >= gap && arr[j - gap] > temp) {
                    arr[j] = arr[j - gap];
                    j -= gap;
                }
                
                arr[j] = temp;
            }
            
            gap = (gap - 1) / 3;
        }
    }
    
    // 希尔排序（不同增量序列）
    void shellSortWithGaps(vector<int>& arr) {
        int n = arr.size();
        vector<int> gaps = {n / 2, n / 4, n / 8, 1};
        
        for (int gap : gaps) {
            for (int i = gap; i < n; ++i) {
                int temp = arr[i];
                int j = i;
                
                while (j >= gap && arr[j - gap] > temp) {
                    arr[j] = arr[j - gap];
                    j -= gap;
                }
                
                arr[j] = temp;
            }
        }
    }
    
    // 希尔排序（Sedgewick序列）
    void shellSortSedgewick(vector<int>& arr) {
        int n = arr.size();
        vector<int> gaps = {1, 5, 19, 41, 109, 209, 505, 929, 2161, 3905, 8929, 16001, 36289, 64769, 146305, 260609};
        
        int gapIndex = 0;
        while (gaps[gapIndex] < n) {
            gapIndex++;
        }
        gapIndex--;
        
        while (gapIndex >= 0) {
            int gap = gaps[gapIndex];
            
            for (int i = gap; i < n; ++i) {
                int temp = arr[i];
                int j = i;
                
                while (j >= gap && arr[j - gap] > temp) {
                    arr[j] = arr[j - gap];
                    j -= gap;
                }
                
                arr[j] = temp;
            }
            
            gapIndex--;
        }
    }
    
    // 显示希尔排序过程
    void displayShellSort(vector<int>& arr) {
        cout << "Shell Sort Process:" << endl;
        cout << "==================" << endl;
        
        cout << "Original array: ";
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
        
        int n = arr.size();
        int gap = 1;
        
        while (gap < n / 3) {
            gap = gap * 3 + 1;
        }
        
        while (gap > 0) {
            cout << "Gap: " << gap << endl;
            
            for (int i = gap; i < n; ++i) {
                int temp = arr[i];
                int j = i;
                
                while (j >= gap && arr[j - gap] > temp) {
                    arr[j] = arr[j - gap];
                    j -= gap;
                }
                
                arr[j] = temp;
            }
            
            cout << "After gap " << gap << ": ";
            for (int num : arr) {
                cout << num << " ";
            }
            cout << endl;
            
            gap = (gap - 1) / 3;
        }
        
        cout << "Final sorted array: ";
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
    }
};
```

## 💡 其他排序算法要点

1. **计数排序**: 适合小范围整数，时间复杂度O(n+k)
2. **基数排序**: 按位排序，适合整数和字符串
3. **桶排序**: 适合均匀分布数据，时间复杂度O(n+k)
4. **希尔排序**: 插入排序的改进，时间复杂度O(n^1.3)
