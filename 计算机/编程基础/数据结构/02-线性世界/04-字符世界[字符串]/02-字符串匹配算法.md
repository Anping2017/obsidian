# 字符串匹配算法

## 📖 核心概念

**字符串匹配算法**是计算机科学中的重要算法，用于在一个文本串中查找一个模式串的所有出现位置。不同的算法有不同的时间复杂度和适用场景。

### 🏗️ 字符串匹配算法分类

```mermaid

    D --> D2[跳跃距离优化]
    
    E --> E1[哈希值比较]
    E --> E2[滚动哈希]
```

## 🔧 字符串匹配算法实现

### 朴素字符串匹配

```cpp
class NaiveStringMatching {
public:
    vector<int> search(string text, string pattern) {
        vector<int> positions;
        int n = text.length();
        int m = pattern.length();
        
        for (int i = 0; i <= n - m; ++i) {
            int j = 0;
            while (j < m && text[i + j] == pattern[j]) {
                j++;
            }
            
            if (j == m) {
                positions.push_back(i);
            }
        }
        
        return positions;
    }
    
    void displaySearch(string text, string pattern) {
        vector<int> positions = search(text, pattern);
        
        cout << "Text: " << text << endl;
        cout << "Pattern: " << pattern << endl;
        cout << "Found at positions: ";
        
        for (int pos : positions) {
            cout << pos << " ";
        }
        
        if (positions.empty()) {
            cout << "Not found";
        }
        cout << endl;
    }
};
```

### KMP算法

```cpp
class KMPAlgorithm {
private:
    vector<int> computeFailureFunction(string pattern) {
        int m = pattern.length();
        vector<int> failure(m, 0);
        
        int j = 0;
        for (int i = 1; i < m; ++i) {
            while (j > 0 && pattern[i] != pattern[j]) {
                j = failure[j - 1];
            }
            
            if (pattern[i] == pattern[j]) {
                j++;
            }
            
            failure[i] = j;
        }
        
        return failure;
    }
    
public:
    vector<int> search(string text, string pattern) {
        vector<int> positions;
        int n = text.length();
        int m = pattern.length();
        
        if (m == 0) return positions;
        
        vector<int> failure = computeFailureFunction(pattern);
        
        int j = 0;
        for (int i = 0; i < n; ++i) {
            while (j > 0 && text[i] != pattern[j]) {
                j = failure[j - 1];
            }
            
            if (text[i] == pattern[j]) {
                j++;
            }
            
            if (j == m) {
                positions.push_back(i - m + 1);
                j = failure[j - 1];
            }
        }
        
        return positions;
    }
    
    void displayFailureFunction(string pattern) {
        vector<int> failure = computeFailureFunction(pattern);
        
        cout << "Pattern: " << pattern << endl;
        cout << "Failure function: ";
        for (int i = 0; i < failure.size(); ++i) {
            cout << failure[i] << " ";
        }
        cout << endl;
    }
    
    void displaySearch(string text, string pattern) {
        vector<int> positions = search(text, pattern);
        
        cout << "Text: " << text << endl;
        cout << "Pattern: " << pattern << endl;
        cout << "Found at positions: ";
        
        for (int pos : positions) {
            cout << pos << " ";
        }
        
        if (positions.empty()) {
            cout << "Not found";
        }
        cout << endl;
    }
};
```

### Boyer-Moore算法

```cpp
class BoyerMooreAlgorithm {
private:
    map<char, int> badCharacterTable;
    vector<int> goodSuffixTable;
    
    void buildBadCharacterTable(string pattern) {
        int m = pattern.length();
        badCharacterTable.clear();
        
        for (int i = 0; i < m; ++i) {
            badCharacterTable[pattern[i]] = i;
        }
    }
    
    void buildGoodSuffixTable(string pattern) {
        int m = pattern.length();
        goodSuffixTable.resize(m, 0);
        
        vector<int> border(m, 0);
        
        // 计算border数组
        int i = 1, j = 0;
        while (i < m) {
            if (pattern[i] == pattern[j]) {
                border[i] = j + 1;
                i++;
                j++;
            } else if (j > 0) {
                j = border[j - 1];
            } else {
                border[i] = 0;
                i++;
            }
        }
        
        // 计算good suffix table
        for (int i = 0; i < m; ++i) {
            goodSuffixTable[i] = m - border[i];
        }
    }
    
public:
    vector<int> search(string text, string pattern) {
        vector<int> positions;
        int n = text.length();
        int m = pattern.length();
        
        if (m == 0) return positions;
        
        buildBadCharacterTable(pattern);
        buildGoodSuffixTable(pattern);
        
        int i = 0;
        while (i <= n - m) {
            int j = m - 1;
            
            // 从右到左匹配
            while (j >= 0 && text[i + j] == pattern[j]) {
                j--;
            }
            
            if (j < 0) {
                positions.push_back(i);
                i += goodSuffixTable[0];
            } else {
                int badCharShift = j - badCharacterTable[text[i + j]];
                int goodSuffixShift = goodSuffixTable[j];
                i += max(1, max(badCharShift, goodSuffixShift));
            }
        }
        
        return positions;
    }
    
    void displaySearch(string text, string pattern) {
        vector<int> positions = search(text, pattern);
        
        cout << "Text: " << text << endl;
        cout << "Pattern: " << pattern << endl;
        cout << "Found at positions: ";
        
        for (int pos : positions) {
            cout << pos << " ";
        }
        
        if (positions.empty()) {
            cout << "Not found";
        }
        cout << endl;
    }
};
```

### Rabin-Karp算法

```cpp
class RabinKarpAlgorithm {
private:
    int base = 256;
    int mod = 1000000007;
    
    long long computeHash(string str, int length) {
        long long hash = 0;
        for (int i = 0; i < length; ++i) {
            hash = (hash * base + str[i]) % mod;
        }
        return hash;
    }
    
    long long updateHash(long long hash, char oldChar, char newChar, int patternLength) {
        hash = (hash - oldChar * pow(base, patternLength - 1)) % mod;
        hash = (hash * base + newChar) % mod;
        return hash < 0 ? hash + mod : hash;
    }
    
public:
    vector<int> search(string text, string pattern) {
        vector<int> positions;
        int n = text.length();
        int m = pattern.length();
        
        if (m == 0 || m > n) return positions;
        
        long long patternHash = computeHash(pattern, m);
        long long textHash = computeHash(text, m);
        
        for (int i = 0; i <= n - m; ++i) {
            if (textHash == patternHash) {
                // 验证是否真正匹配
                bool match = true;
                for (int j = 0; j < m; ++j) {
                    if (text[i + j] != pattern[j]) {
                        match = false;
                        break;
                    }
                }
                
                if (match) {
                    positions.push_back(i);
                }
            }
            
            if (i < n - m) {
                textHash = updateHash(textHash, text[i], text[i + m], m);
            }
        }
        
        return positions;
    }
    
    void displaySearch(string text, string pattern) {
        vector<int> positions = search(text, pattern);
        
        cout << "Text: " << text << endl;
        cout << "Pattern: " << pattern << endl;
        cout << "Found at positions: ";
        
        for (int pos : positions) {
            cout << pos << " ";
        }
        
        if (positions.empty()) {
            cout << "Not found";
        }
        cout << endl;
    }
};
```

### 文本编辑器搜索

```cpp
class TextEditorSearch {
private:
    string text;
    vector<string> lines;
    
public:
    TextEditorSearch(string content) : text(content) {
        // 将文本分割为行
        stringstream ss(content);
        string line;
        while (getline(ss, line)) {
            lines.push_back(line);
        }
    }
    
    vector<pair<int, int>> searchInText(string pattern) {
        vector<pair<int, int>> results; // (line, column)
        
        for (int i = 0; i < lines.size(); ++i) {
            KMPAlgorithm kmp;
            vector<int> positions = kmp.search(lines[i], pattern);
            
            for (int pos : positions) {
                results.push_back({i + 1, pos + 1}); // 1-based indexing
            }
        }
        
        return results;
    }
    
    void displaySearchResults(string pattern) {
        vector<pair<int, int>> results = searchInText(pattern);
        
        cout << "Search results for \"" << pattern << "\":" << endl;
        
        if (results.empty()) {
            cout << "No matches found." << endl;
            return;
        }
        
        for (auto result : results) {
            int line = result.first;
            int column = result.second;
            
            cout << "Line " << line << ", Column " << column << ": ";
            cout << lines[line - 1] << endl;
        }
    }
    
    void highlightMatches(string pattern) {
        vector<pair<int, int>> results = searchInText(pattern);
        
        cout << "Text with highlighted matches:" << endl;
        
        for (int i = 0; i < lines.size(); ++i) {
            cout << "Line " << (i + 1) << ": ";
            
            string line = lines[i];
            vector<bool> highlighted(line.length(), false);
            
            // 标记匹配位置
            for (auto result : results) {
                if (result.first == i + 1) {
                    int start = result.second - 1;
                    for (int j = 0; j < pattern.length(); ++j) {
                        if (start + j < line.length()) {
                            highlighted[start + j] = true;
                        }
                    }
                }
            }
            
            // 输出带高亮的行
            for (int j = 0; j < line.length(); ++j) {
                if (highlighted[j]) {
                    cout << "[" << line[j] << "]";
                } else {
                    cout << line[j];
                }
            }
            cout << endl;
        }
    }
};
```

### 模式匹配统计

```cpp
class PatternMatchingStats {
private:
    string text;
    
public:
    PatternMatchingStats(string content) : text(content) {}
    
    map<string, int> countPatterns(vector<string> patterns) {
        map<string, int> counts;
        
        for (string pattern : patterns) {
            KMPAlgorithm kmp;
            vector<int> positions = kmp.search(text, pattern);
            counts[pattern] = positions.size();
        }
        
        return counts;
    }
    
    void displayPatternStats(vector<string> patterns) {
        map<string, int> counts = countPatterns(patterns);
        
        cout << "Pattern matching statistics:" << endl;
        cout << "===========================" << endl;
        
        for (auto pair : counts) {
            cout << "\"" << pair.first << "\": " << pair.second << " occurrences" << endl;
        }
    }
    
    vector<string> findMostFrequentPatterns(int minLength, int maxLength) {
        map<string, int> patternCounts;
        
        for (int len = minLength; len <= maxLength; ++len) {
            for (int i = 0; i <= text.length() - len; ++i) {
                string pattern = text.substr(i, len);
                patternCounts[pattern]++;
            }
        }
        
        vector<pair<string, int>> sortedPatterns;
        for (auto pair : patternCounts) {
            sortedPatterns.push_back(pair);
        }
        
        sort(sortedPatterns.begin(), sortedPatterns.end(), 
             [](const pair<string, int>& a, const pair<string, int>& b) {
                 return a.second > b.second;
             });
        
        vector<string> result;
        for (int i = 0; i < min(10, (int)sortedPatterns.size()); ++i) {
            result.push_back(sortedPatterns[i].first);
        }
        
        return result;
    }
    
    void displayMostFrequentPatterns(int minLength, int maxLength) {
        vector<string> patterns = findMostFrequentPatterns(minLength, maxLength);
        
        cout << "Most frequent patterns:" << endl;
        cout << "======================" << endl;
        
        for (int i = 0; i < patterns.size(); ++i) {
            KMPAlgorithm kmp;
            vector<int> positions = kmp.search(text, patterns[i]);
            cout << (i + 1) << ". \"" << patterns[i] << "\": " << positions.size() << " occurrences" << endl;
        }
    }
};
```

class StringMatchingAnalysis {
public:

        cout << "String Matching Algorithm Time Complexity:" << endl;

        cout << "   - Space Complexity: O(1)" << endl;
        cout << "   - Best Case: O(n)" << endl;
        cout << "   - Worst Case: O(nm)" << endl;
        
        cout << "2. KMP Algorithm:" << endl;

        cout << "   - Preprocessing: O(m)" << endl;
        cout << "   - Searching: O(n)" << endl;
        
        cout << "3. Boyer-Moore Algorithm:" << endl;
        cout << "   - Time Complexity: O(n + m)" << endl;
        cout << "   - Space Complexity: O(m)" << endl;
        cout << "   - Best Case: O(n/m)" << endl;
        cout << "   - Worst Case: O(nm)" << endl;
        
        cout << "4. Rabin-Karp Algorithm:" << endl;
        cout << "   - Time Complexity: O(n + m)" << endl;
        cout << "   - Space Complexity: O(1)" << endl;
        cout << "   - Average Case: O(n + m)" << endl;
        cout << "   - Worst Case: O(nm)" << endl;
    }
    
    static void analyzeSpaceComplexity() {
        cout << "String Matching Algorithm Space Complexity:" << endl;
        cout << "==========================================" << endl;
        
        cout << "1. Naive Algorithm:" << endl;
        cout << "   - Space Complexity: O(1)" << endl;
        cout << "   - No additional space required" << endl;
        
        cout << "2. KMP Algorithm:" << endl;
        cout << "   - Space Complexity: O(m)" << endl;
        cout << "   - Failure function array" << endl;
        
        cout << "3. Boyer-Moore Algorithm:" << endl;
        cout << "   - Space Complexity: O(m)" << endl;
        cout << "   - Bad character table" << endl;
        cout << "   - Good suffix table" << endl;
        
        cout << "4. Rabin-Karp Algorithm:" << endl;
        cout << "   - Space Complexity: O(1)" << endl;
        cout << "   - Constant space for hash values" << endl;
    }
};
