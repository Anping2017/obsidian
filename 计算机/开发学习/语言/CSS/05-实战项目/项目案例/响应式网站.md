## JavaScript交互

### 1. 导航交互
```javascript
// js/navigation.js
class Navigation {
    constructor() {
        this.menuToggle = document.querySelector('.header__menu-toggle');
        this.nav = document.querySelector('.header__nav');
        this.navLinks = document.querySelectorAll('.header__nav-link');
        this.init();
    }
    
    init() {
        this.menuToggle?.addEventListener('click', () => this.toggleMenu());
        this.navLinks.forEach(link => {
            link.addEventListener('click', () => this.closeMenu());
        });
        
        // 点击外部关闭菜单
        document.addEventListener('click', (e) => {
            if (!this.nav.contains(e.target) && !this.menuToggle.contains(e.target)) {
                this.closeMenu();
            }
        });
    }
    
    toggleMenu() {
        this.nav.classList.toggle('header__nav--open');
        this.menuToggle.classList.toggle('active');
    }
    
    closeMenu() {
        this.nav.classList.remove('header__nav--open');
        this.menuToggle.classList.remove('active');
    }
}

// 初始化导航
document.addEventListener('DOMContentLoaded', () => {
    new Navigation();
});
```

### 2. 响应式处理
```javascript
// js/responsive.js
class ResponsiveHandler {
    constructor() {
        this.breakpoints = {
            sm: 576,
            md: 768,
            lg: 992,
            xl: 1200
        };
        this.currentBreakpoint = this.getCurrentBreakpoint();
        this.init();
    }
    
    init() {
        window.addEventListener('resize', () => this.handleResize());
        this.handleResize();
    }
    
    getCurrentBreakpoint() {
        const width = window.innerWidth;
        if (width >= this.breakpoints.xl) return 'xl';
        if (width >= this.breakpoints.lg) return 'lg';
        if (width >= this.breakpoints.md) return 'md';
        if (width >= this.breakpoints.sm) return 'sm';
        return 'xs';
    }
    
    handleResize() {
        const newBreakpoint = this.getCurrentBreakpoint();
        if (newBreakpoint !== this.currentBreakpoint) {
            this.currentBreakpoint = newBreakpoint;
            this.onBreakpointChange(newBreakpoint);
        }
    }
    
    onBreakpointChange(breakpoint) {
        // 处理断点变化
        console.log(`Breakpoint changed to: ${breakpoint}`);
        
        // 关闭移动端菜单
        if (breakpoint !== 'xs' && breakpoint !== 'sm') {
            const nav = document.querySelector('.header__nav');
            nav?.classList.remove('header__nav--open');
        }
    }
}

// 初始化响应式处理
document.addEventListener('DOMContentLoaded', () => {
    new ResponsiveHandler();
});
```
