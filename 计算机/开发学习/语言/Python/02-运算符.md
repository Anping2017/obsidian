## 算术运算符

用于基本的数学运算：

```python
a = 10
b = 3

# 加法
result = a + b  # 13
print(f"{a} + {b} = {result}")

# 减法
result = a - b  # 7
print(f"{a} - {b} = {result}")

# 乘法
result = a * b  # 30
print(f"{a} * {b} = {result}")

# 除法（返回浮点数）
result = a / b  # 3.3333333333333335
print(f"{a} / {b} = {result}")

# 整除（返回整数）
result = a // b  # 3
print(f"{a} // {b} = {result}")

# 取余
result = a % b  # 1
print(f"{a} % {b} = {result}")

# 幂运算
result = a ** b  # 1000
print(f"{a} ** {b} = {result}")
```

### 字符串和列表的算术运算

```python
# 字符串拼接
str1 = "Hello"
str2 = "World"
result = str1 + " " + str2  # "Hello World"

# 字符串重复
result = str1 * 3  # "HelloHelloHello"

# 列表拼接
list1 = [1, 2, 3]
list2 = [4, 5, 6]
result = list1 + list2  # [1, 2, 3, 4, 5, 6]

# 列表重复
result = list1 * 2  # [1, 2, 3, 1, 2, 3]
```

## 比较运算符

用于比较两个值的大小关系：

```python
a = 10
b = 5

# 等于
result = a == b  # False
print(f"{a} == {b}: {result}")

# 不等于
result = a != b  # True
print(f"{a} != {b}: {result}")

# 大于
result = a > b  # True
print(f"{a} > {b}: {result}")

# 小于
result = a < b  # False
print(f"{a} < {b}: {result}")

# 大于等于
result = a >= b  # True
print(f"{a} >= {b}: {result}")

# 小于等于
result = a <= b  # False
print(f"{a} <= {b}: {result}")

# 链式比较
result = 5 <= a <= 15  # True
print(f"5 <= {a} <= 15: {result}")
```

### 字符串比较

```python
str1 = "apple"
str2 = "banana"

# 按字典序比较
result = str1 < str2  # True
print(f'"{str1}" < "{str2}": {result}')

# 大小写敏感
result = "Apple" == "apple"  # False
print(f'"Apple" == "apple": {result}')
```

## 逻辑运算符

用于组合多个条件：

```python
a = True
b = False

# 逻辑与 (and)
result = a and b  # False
print(f"{a} and {b}: {result}")

# 逻辑或 (or)
result = a or b  # True
print(f"{a} or {b}: {result}")

# 逻辑非 (not)
result = not a  # False
print(f"not {a}: {result}")

# 复杂逻辑表达式
age = 25
score = 85
result = age >= 18 and score >= 80  # True
print(f"age >= 18 and score >= 80: {result}")
```

### 短路求值

```python
# and 运算符：如果第一个为False，不会计算第二个
def func1():
    print("func1被调用")
    return False

def func2():
    print("func2被调用")
    return True

result = func1() and func2()  # 只输出"func1被调用"

# or 运算符：如果第一个为True，不会计算第二个
result = func2() or func1()  # 只输出"func2被调用"
```

## 赋值运算符

用于给变量赋值：

```python
x = 10

# 基本赋值
x = 5

# 加法赋值
x += 3  # 等价于 x = x + 3
print(f"x += 3: {x}")  # 8

# 减法赋值
x -= 2  # 等价于 x = x - 2
print(f"x -= 2: {x}")  # 6

# 乘法赋值
x *= 2  # 等价于 x = x * 2
print(f"x *= 2: {x}")  # 12

# 除法赋值
x /= 3  # 等价于 x = x / 3
print(f"x /= 3: {x}")  # 4.0

# 整除赋值
x //= 2  # 等价于 x = x // 2
print(f"x //= 2: {x}")  # 2.0

# 取余赋值
x %= 3  # 等价于 x = x % 3
print(f"x %= 3: {x}")  # 2.0

# 幂赋值
x **= 3  # 等价于 x = x ** 3
print(f"x **= 3: {x}")  # 8.0
```

## 三元运算符

Python中的三元运算符语法：`值1 if 条件 else 值2`

```python
age = 20

# 基本用法
status = "成年人" if age >= 18 else "未成年人"
print(f"年龄{age}岁，{status}")

# 嵌套使用
score = 85
grade = "优秀" if score >= 90 else "良好" if score >= 80 else "及格" if score >= 60 else "不及格"
print(f"分数{score}，等级：{grade}")

# 与函数结合
def get_max(a, b):
    return a if a > b else b

result = get_max(10, 20)
print(f"最大值：{result}")
```

## 位运算符

用于对二进制位进行操作：

```python
a = 12  # 二进制：1100
b = 10  # 二进制：1010

# 按位与 (&)
result = a & b  # 8 (二进制：1000)
print(f"{a} & {b} = {result}")

# 按位或 (|)
result = a | b  # 14 (二进制：1110)
print(f"{a} | {b} = {result}")

# 按位异或 (^)
result = a ^ b  # 6 (二进制：0110)
print(f"{a} ^ {b} = {result}")

# 按位取反 (~)
result = ~a  # -13
print(f"~{a} = {result}")

# 左移 (<<)
result = a << 2  # 48 (二进制：110000)
print(f"{a} << 2 = {result}")

# 右移 (>>)
result = a >> 2  # 3 (二进制：11)
print(f"{a} >> 2 = {result}")
```

### 位运算应用

```python
# 判断奇偶性
def is_even(n):
    return (n & 1) == 0

print(f"8是偶数：{is_even(8)}")
print(f"7是偶数：{is_even(7)}")

# 快速计算2的幂
def power_of_2(n):
    return 1 << n

print(f"2^3 = {power_of_2(3)}")
print(f"2^5 = {power_of_2(5)}")
```

## 运算符优先级

Python运算符的优先级（从高到低）：

```python
# 1. 括号 ()
result = (2 + 3) * 4  # 20

# 2. 幂运算 **
result = 2 ** 3 ** 2  # 512 (从右到左)

# 3. 一元运算符 +x, -x, ~x
result = -2 ** 2  # -4

# 4. 乘除取余 * / // %
result = 2 + 3 * 4  # 14

# 5. 加减 + -
result = 2 * 3 + 4  # 10

# 6. 位移 << >>
result = 2 + 3 << 1  # 10

# 7. 按位与 &
result = 2 + 3 & 4  # 4

# 8. 按位异或 ^
result = 2 + 3 ^ 4  # 5

# 9. 按位或 |
result = 2 + 3 | 4  # 7

# 10. 比较运算符 == != < > <= >=
result = 2 + 3 == 5  # True

# 11. 身份运算符 is is not
result = 2 + 3 is 5  # True

# 12. 成员运算符 in not in
result = 2 in [1, 2, 3]  # True

# 13. 逻辑非 not
result = not 2 + 3 == 5  # False

# 14. 逻辑与 and
result = 2 + 3 == 5 and 4 > 3  # True

# 15. 逻辑或 or
result = 2 + 3 == 5 or 4 < 3  # True

# 16. 条件表达式 if else
result = 2 + 3 if True else 4  # 5

# 17. 赋值运算符 = += -= *= /= //= %= **= &= |= ^= >>= <<=
x = 2 + 3  # 5
```

## 示例代码

```python
# 计算器示例
def calculator():
    print("简单计算器")
    print("支持 +, -, *, /, //, %, ** 运算")
    
    try:
        num1 = float(input("请输入第一个数字: "))
        operator = input("请输入运算符: ")
        num2 = float(input("请输入第二个数字: "))
        
        if operator == '+':
            result = num1 + num2
        elif operator == '-':
            result = num1 - num2
        elif operator == '*':
            result = num1 * num2
        elif operator == '/':
            if num2 == 0:
                print("错误：除数不能为零")
                return
            result = num1 / num2
        elif operator == '//':
            if num2 == 0:
                print("错误：除数不能为零")
                return
            result = num1 // num2
        elif operator == '%':
            if num2 == 0:
                print("错误：除数不能为零")
                return
            result = num1 % num2
        elif operator == '**':
            result = num1 ** num2
        else:
            print("不支持的运算符")
            return
        
        print(f"{num1} {operator} {num2} = {result}")
        
    except ValueError:
        print("错误：请输入有效的数字")

# 运行计算器
# calculator()
```
