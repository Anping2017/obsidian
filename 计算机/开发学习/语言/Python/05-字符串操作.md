## 字符串基础

### 字符串创建

```python
# 单引号字符串
str1 = 'Hello World'

# 双引号字符串
str2 = "Hello World"

# 三引号字符串（多行）
str3 = """
这是一个
多行字符串
"""

# 原始字符串（r前缀）
str4 = r"C:\Users\Name\Documents"

# 转义字符
str5 = "He said \"Hello\""
str6 = "Line 1\nLine 2"
str7 = "Column1\tColumn2"
```

### 字符串特性

```python
# 字符串是不可变的
text = "Hello"
# text[0] = 'h'  # 错误！字符串不可变

# 字符串可以索引
text = "Python"
print(text[0])    # P
print(text[-1])   # n
print(text[2:5])  # tho
```

## 字符串拼接

### 使用 + 操作符

```python
# 基本拼接
first_name = "张"
last_name = "三"
full_name = first_name + last_name
print(full_name)  # 张三

# 多个字符串拼接
greeting = "Hello" + " " + "World"
print(greeting)  # Hello World
```

### 使用 join() 方法

```python
# 列表拼接
words = ["Hello", "World", "Python"]
sentence = " ".join(words)
print(sentence)  # Hello World Python

# 使用不同分隔符
numbers = ["1", "2", "3", "4", "5"]
result = "-".join(numbers)
print(result)  # 1-2-3-4-5

# 空字符串拼接
chars = ["P", "y", "t", "h", "o", "n"]
word = "".join(chars)
print(word)  # Python
```

### 使用 * 操作符

```python
# 字符串重复
text = "Hello"
repeated = text * 3
print(repeated)  # HelloHelloHello

# 与拼接结合
pattern = "-" * 10
print(pattern)  # ----------
```

## 字符串切片

### 基本切片

```python
text = "Python Programming"

# 基本切片 [start:end]
print(text[0:6])    # Python
print(text[7:18])   # Programming

# 省略开始索引
print(text[:6])     # Python
print(text[7:])     # Programming

# 省略结束索引
print(text[0:6])    # Python
print(text[7:])     # Programming

# 负数索引
print(text[-11:])   # Programming
print(text[:-11])   # Python
```

### 步长切片

```python
text = "Python Programming"

# 使用步长
print(text[::2])    # Pto rgamn
print(text[1::2])   # yhnPormig

# 反向切片
print(text[::-1])   # gnimmargorP nohtyP
print(text[5:0:-1]) # nohty
```

### 切片应用

```python
# 提取文件名和扩展名
filename = "document.pdf"
name = filename[:-4]  # document
extension = filename[-3:]  # pdf

# 提取日期部分
date = "2024-01-15"
year = date[:4]    # 2024
month = date[5:7]  # 01
day = date[8:]     # 15
```

## 字符串查找

### find() 和 index() 方法

```python
text = "Python Programming"

# find() 方法
pos = text.find("Python")
print(f"Python位置：{pos}")  # 0

pos = text.find("Java")
print(f"Java位置：{pos}")  # -1 (未找到)

# index() 方法
pos = text.index("Programming")
print(f"Programming位置：{pos}")  # 7

# pos = text.index("Java")  # 抛出异常

# 指定搜索范围
pos = text.find("o", 5)  # 从位置5开始搜索
print(f"从位置5开始搜索'o'：{pos}")  # 9
```

### startswith() 和 endswith() 方法

```python
filename = "document.pdf"

# 检查开头
if filename.startswith("doc"):
    print("文件名以'doc'开头")

# 检查结尾
if filename.endswith(".pdf"):
    print("这是一个PDF文件")

# 检查多个可能
if filename.endswith((".pdf", ".doc", ".txt")):
    print("这是一个文档文件")
```

### count() 方法

```python
text = "Python Programming"

# 统计字符出现次数
count = text.count("o")
print(f"'o'出现次数：{count}")  # 2

# 统计子字符串出现次数
count = text.count("ing")
print(f"'ing'出现次数：{count}")  # 1
```

## 字符串替换

### replace() 方法

```python
text = "Hello World"

# 基本替换
new_text = text.replace("World", "Python")
print(new_text)  # Hello Python

# 替换多个字符
text = "aabbcc"
new_text = text.replace("a", "x")
print(new_text)  # xxbbcc

# 限制替换次数
text = "aabbcc"
new_text = text.replace("a", "x", 1)
print(new_text)  # xabbcc
```

### 正则表达式替换

```python
import re

text = "电话：138-1234-5678"

# 替换电话号码格式
new_text = re.sub(r'\d{3}-\d{4}-\d{4}', '***-****-****', text)
print(new_text)  # 电话：***-****-****
```

## 字符串格式化

### f-string格式化（推荐）

```python
name = "张三"
age = 25
score = 85.5

# 基本格式化
message = f"姓名：{name}，年龄：{age}，分数：{score}"
print(message)

# 格式化数字
pi = 3.14159
print(f"π的值：{pi:.2f}")  # 保留2位小数
print(f"π的值：{pi:.4f}")  # 保留4位小数

# 格式化宽度
print(f"姓名：{name:>10}")  # 右对齐，宽度10
print(f"年龄：{age:<10}")   # 左对齐，宽度10
print(f"分数：{score:^10}") # 居中对齐，宽度10

# 格式化日期
from datetime import datetime
now = datetime.now()
print(f"当前时间：{now:%Y-%m-%d %H:%M:%S}")
```

### format() 方法

```python
name = "张三"
age = 25
score = 85.5

# 位置参数
message = "姓名：{}，年龄：{}，分数：{}".format(name, age, score)
print(message)

# 关键字参数
message = "姓名：{name}，年龄：{age}，分数：{score}".format(
    name=name, age=age, score=score
)
print(message)

# 混合使用
message = "姓名：{0}，年龄：{age}，分数：{1}".format(name, score, age=age)
print(message)

# 格式化选项
pi = 3.14159
print("π的值：{:.2f}".format(pi))  # 保留2位小数
print("π的值：{:.4f}".format(pi))  # 保留4位小数
```

### % 格式化（传统方法）

```python
name = "张三"
age = 25
score = 85.5

# 基本格式化
message = "姓名：%s，年龄：%d，分数：%.2f" % (name, age, score)
print(message)

# 格式化选项
pi = 3.14159
print("π的值：%.2f" % pi)  # 保留2位小数
print("π的值：%.4f" % pi)  # 保留4位小数
```

## 字符串方法

### 大小写转换

```python
text = "Hello World"

# 转换为大写
upper_text = text.upper()
print(upper_text)  # HELLO WORLD

# 转换为小写
lower_text = text.lower()

# 首字母大写
title_text = text.title()
print(title_text)  # Hello World

# 首字母小写
capitalize_text = text.capitalize()
print(capitalize_text)  # Hello world

# 大小写交换
swap_text = text.swapcase()
print(swap_text)  # hELLO wORLD
```

### 去除空白字符

```python
text = "  Hello World  "

# 去除两端空白
stripped = text.strip()
print(f"'{stripped}'")  # 'Hello World'

# 去除左端空白
left_stripped = text.lstrip()
print(f"'{left_stripped}'")  # 'Hello World  '

# 去除右端空白
right_stripped = text.rstrip()
print(f"'{right_stripped}'")  # '  Hello World'

# 去除指定字符
text = "***Hello World***"
stripped = text.strip("*")
print(stripped)  # Hello World
```

### 字符串分割

```python
text = "apple,banana,orange"

# 基本分割
fruits = text.split(",")
print(fruits)  # ['apple', 'banana', 'orange']

# 限制分割次数
fruits = text.split(",", 1)
print(fruits)  # ['apple', 'banana,orange']

# 分割行
multiline = "line1\nline2\nline3"
lines = multiline.splitlines()
print(lines)  # ['line1', 'line2', 'line3']

# 分割空白字符
text = "  apple   banana   orange  "
fruits = text.split()
print(fruits)  # ['apple', 'banana', 'orange']
```

### 字符串连接

```python
# 使用join()连接
fruits = ["apple", "banana", "orange"]
text = ", ".join(fruits)
print(text)  # apple, banana, orange

# 使用不同分隔符
text = " | ".join(fruits)
print(text)  # apple | banana | orange
```

## 示例代码

### 文本处理工具

```python
def text_processor():
    """文本处理工具"""
    
    def clean_text(text):
        """清理文本"""
        # 去除多余空白
        cleaned = " ".join(text.split())
        return cleaned
    
    def count_words(text):
        """统计单词数"""
        words = text.split()
        return len(words)
    
    def count_characters(text):
        """统计字符数"""
        return len(text)
    
    def reverse_text(text):
        """反转文本"""
        return text[::-1]
    
    def capitalize_sentences(text):
        """句子首字母大写"""
        sentences = text.split(". ")
        capitalized = [s.capitalize() for s in sentences]
        return ". ".join(capitalized)
    
    def extract_numbers(text):
        """提取数字"""
        import re
        numbers = re.findall(r'\d+', text)
        return [int(num) for num in numbers]
    
    def word_frequency(text):
        """词频统计"""
        words = text.lower().split()
        frequency = {}
        for word in words:
            # 去除标点符号
            word = word.strip(".,!?;:")
            if word:
                frequency[word] = frequency.get(word, 0) + 1
        return frequency
    
    # 主程序
    while True:
        print("\n=== 文本处理工具 ===")
        print("1. 清理文本")
        print("2. 统计单词数")
        print("3. 统计字符数")
        print("4. 反转文本")
        print("5. 句子首字母大写")
        print("6. 提取数字")
        print("7. 词频统计")
        print("8. 退出")
        
        choice = input("请选择操作（1-8）：")
        
        if choice == '8':
            print("感谢使用！")
            break
        
        text = input("请输入文本：")
        
        if choice == '1':
            result = clean_text(text)
            print(f"清理后的文本：{result}")
            
        elif choice == '2':
            count = count_words(text)
            print(f"单词数：{count}")
            
        elif choice == '3':
            count = count_characters(text)
            print(f"字符数：{count}")
            
        elif choice == '4':
            result = reverse_text(text)
            print(f"反转后的文本：{result}")
            
        elif choice == '5':
            result = capitalize_sentences(text)
            print(f"首字母大写：{result}")
            
        elif choice == '6':
            numbers = extract_numbers(text)
            print(f"提取的数字：{numbers}")
            
        elif choice == '7':
            frequency = word_frequency(text)
            print("词频统计：")
            for word, count in sorted(frequency.items()):
                print(f"  {word}: {count}")

# 运行文本处理工具
# text_processor()
```

### 密码生成器

```python
import random
import string

def password_generator():
    """密码生成器"""
    
    def generate_password(length=12, include_symbols=True):
        """生成密码"""
        characters = string.ascii_letters + string.digits
        if include_symbols:
            characters += "!@#$%^&*()_+-=[]{}|;:,.<>?"
        
        password = ''.join(random.choice(characters) for _ in range(length))
        return password
    
    def check_password_strength(password):
        """检查密码强度"""
        score = 0
        feedback = []
        
        # 长度检查
        if len(password) >= 8:
            score += 1
        else:
            feedback.append("密码长度至少8位")
        
        # 包含小写字母
        if any(c.islower() for c in password):
            score += 1
        else:
            feedback.append("包含小写字母")
        
        # 包含大写字母
        if any(c.isupper() for c in password):
            score += 1
        else:
            feedback.append("包含大写字母")
        
        # 包含数字
        if any(c.isdigit() for c in password):
            score += 1
        else:
            feedback.append("包含数字")
        
        # 包含特殊字符
        if any(c in "!@#$%^&*()_+-=[]{}|;:,.<>?" for c in password):
            score += 1
        else:
            feedback.append("包含特殊字符")
        
        # 强度评级
        if score <= 2:
            strength = "弱"
        elif score <= 4:
            strength = "中等"
        else:
            strength = "强"
        
        return strength, feedback
    
    # 主程序
    while True:
        print("\n=== 密码生成器 ===")
        print("1. 生成密码")
        print("2. 检查密码强度")
        print("3. 退出")
        
        choice = input("请选择操作（1-3）：")
        
        if choice == '1':
            try:
                length = int(input("密码长度（默认12）：") or "12")
                include_symbols = input("包含特殊字符？(y/n，默认y)：").lower() != 'n'
                
                password = generate_password(length, include_symbols)
                print(f"生成的密码：{password}")
                
                # 检查强度
                strength, feedback = check_password_strength(password)
                print(f"密码强度：{strength}")
                
            except ValueError:
                print("请输入有效的数字")
                
        elif choice == '2':
            password = input("请输入要检查的密码：")
            strength, feedback = check_password_strength(password)
            print(f"密码强度：{strength}")
            if feedback:
                print("建议改进：")
                for item in feedback:
                    print(f"  - {item}")
                    
        elif choice == '3':
            print("感谢使用！")
            break
            
        else:
            print("无效选择，请重新输入")

# 运行密码生成器
# password_generator()
```
