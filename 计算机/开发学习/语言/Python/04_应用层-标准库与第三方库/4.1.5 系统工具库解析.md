import os
import sys
import platform
import subprocess
import shutil
import tempfile
from pathlib import Path
from typing import List, Dict, Any, Optional
import psutil

class SystemToolsMastery:
    """系统工具精通技术"""
    
    @staticmethod
    def os_operations():
        """操作系统操作"""
        
        # ✅ 环境变量和路径
        def environment_handling():
            """环境变量处理"""
            
            # 获取环境变量
            python_path = os.environ.get('PYTHON_PATH', '未设置')
            path_env = os.environ.get('PATH', '').split(os.pathsep)[:3]  # 前3个路径
            
            print(f"Python路径: {python_path}")
            print(f"PATH前3项: {path_env}")
            
            # 设置环境变量（仅在当前进程中）
            os.environ['CUSTOM_VAR'] = 'test_value'
            print(f"自定义变量: {os.environ.get('CUSTOM_VAR')}")
            
            # 当前工作目录
            current_dir = os.getcwd()
            print(f"当前目录: {current_dir}")
            
            # 切换目录
            os.chdir('..')  # 上级目录
            parent_dir = os.getcwd()
            os.chdir(current_dir)  # 恢复
            
            print(f"上级目录: {parent_dir}")
        
        # ✅ 文件和目录操作
        def file_directory_operations():
            """文件和目录操作"""
            
            # 创建测试目录结构
            test_dir = 'test_system_operations'
            if os.path.exists(test_dir):
                shutil.rmtree(test_dir)
            
            os.makedirs(test_dir, exist_ok=True)
            
            # 创建子目录
            sub_dirs = ['logs', 'data', 'temp']
            for sub_dir in sub_dirs:
                os.makedirs(os.path.join(test_dir, sub_dir), exist_ok=True)
            
            # 创建文件
            test_file = os.path.join(test_dir, 'test.txt')
            with open(test_file, 'w') as f:
                f.write("系统操作测试文件")
            
            # 文件信息
            file_stat = os.stat(test_file)
            file_size = file_stat.st_size
            file_mtime = file_stat.st_mtime
            
            print(f"文件大小: {file_size} 字节")
            print(f"修改时间: {file_stat.st_mtime}")
            
            # 遍历目录
            print(f"\n目录结构 {test_dir}:")
            for root, dirs, files in os.walk(test_dir):
                level = root.replace(test_dir, '').count(os.sep)
                indent = ' ' * 2 * level
                print(f"{indent}{os.path.basename(root)}/")
                
                sub_indent = ' ' * 2 * (level + 1)
                for file in files:
                    print(f"{sub_indent}{file}")
            
            # 清理
            shutil.rmtree(test_dir)
        
        environment_handling()
        file_directory_operations()
    
    @staticmethod
    def subprocess_execution():
        """子进程管理"""
        
        # ✅ 基本命令执行
        def basic_command_execution():
            """基本命令执行"""
            
            # 执行简单命令
            result = subprocess.run(['python', '--version'], 
                                  capture_output=True, text=True)
            print(f"Python版本: {result.stdout.strip()}")
            
            # 执行带参数的命令
            try:
                result = subprocess.run(['dir'], 
                                      capture_output=True, text=True, 
                                      shell=True, check=True)
                lines = result.stdout.strip().split('\n')[:5]  # 前5行
                print("目录列表前5行:")
                for line in lines:
                    print(f"  {line}")
            except subprocess.CalledProcessError as e:
                print(f"命令执行失败: {e}")
        
        # ✅ 高级进程管理
        def advanced_process_management():
            """高级进程管理"""
            
            # 后台进程
            def run_background_process():
                """运行后台进程"""
                # Windows示例
                if platform.system() == 'Windows':
                    cmd = ['ping', '127.0.0.1', '-n', '3']
                else:
                    cmd = ['ping', '-c', '3', '127.0.0.1']
                
                process = subprocess.Popen(cmd, stdout=subprocess.PIPE, 
                                         stderr=subprocess.PIPE, text=True)
                
                # 等待进程完成，设置超时
                try:
                    stdout, stderr = process.communicate(timeout=10)
                    print(f"进程退出码: {process.returncode}")
                    print(f"输出前100字符: {stdout[:100]}...")
                except subprocess.TimeoutExpired:
                    process.kill()
                    print("进程超时，已终止")
            
            run_background_process()
        
        basic_command_execution()
        advanced_process_management()
    
    @staticmethod
    def psutil_system_monitoring():
        """系统监控技术"""
        
        # ✅ CPU监控
        def cpu_monitoring():
            """CPU监控"""
            try:
                # CPU核心数
                cpu_count = psutil.cpu_count()
                print(f"CPU核心数: {cpu_count}")
                
                # CPU使用率
                cpu_percent = psutil.cpu_percent(interval=1)
                print(f"CPU使用率: {cpu_percent}%")
                
                # CPU详细信息
                cpu_times = psutil.cpu_times()
                print(f"CPU时间分布:")
                print(f"  用户态: {cpu_times.user:.2f}s")
                print(f"  核心态: {cpu_times.system:.2f}s")
                print(f"  空闲时间: {cpu_times.idle:.2f}s")
                
            except Exception as e:
                print(f"CPU监控需要安装psutil: pip install psutil")
        
        # ✅ 内存监控
        def memory_monitoring():
            """内存监控"""
            try:
                # 虚拟内存
                virtual_memory = psutil.virtual_memory()
                print(f"内存使用:")
                print(f"  总内存: {virtual_memory.total / (1024**3):.1f} GB")
                print(f"  可用内存: {virtual_memory.available / (1024**3):.1f} GB")
                print(f"  使用率: {virtual_memory.percent}%")
                
                # 交换内存
                swap_memory = psutil.swap_memory()
                print(f"交换内存:")
                print(f"  总量: {swap_memory.total / (1024**3):.1f} GB")
                print(f"  使用量: {swap_memory.used / (1024**3):.1f} GB")
                
            except Exception as e:
                print(f"内存监控失败: {e}")
        
        # ✅ 磁盘监控
        def disk_monitoring():
            """磁盘监控"""
            try:
                # 磁盘使用
                disk_usage = psutil.disk_usage('/')
                print(f"磁盘使用:")
                print(f"  总量: {disk_usage.total / (1024**3):.1f} GB")
                print(f"  已用: {disk_usage.used / (1024**3):.1f} GB")
                print(f"  可用: {disk_usage.free / (1024**3):.1f} GB")
                print(f"  使用率: {disk_usage.used/disk_usage.total*100:.1f}%")
                
                # 磁盘I/O
                disk_io = psutil.disk_io_counters()
                print(f"磁盘I/O:")
                print(f"  读次数: {disk_io.read_count}")
                print(f"  写次数: {disk_io.write_count}")
                print(f"  读字节数: {disk_io.read_bytes / (1024**2):.1f} MB")
                print(f"  写字节数: {disk_io.write_bytes / (1024**2):.1f} MB")
                
            except Exception as e:
                print(f"磁盘监控失败: {e}")
        
        cpu_monitoring()
        memory_monitoring()
        disk_monitoring()
    
    @staticmethod
    def network_monitoring():
        """网络监控"""
        
        try:
            # ✅ 网络连接
            def network_connections():
                """网络连接监控"""
                
                # 获取网络连接
                connections = psutil.net_connections(kind='inet')
                
                # 统计连接状态

                for state, count in connection_stats.items():
                    print(f"  {state}: {count}")
            
            # ✅ 网络I/O
            def network_io():
                """网络I/O统计"""
                
                net_io = psutil.net_io_counters()
                print(f"网络I/O:")

                print(f"  发送字节数: {net_io.bytes_sent / (1024**2):.1f} MB")
                print(f"  接收包数: {net_io.packets_recv}")
                print(f"  发送包数: {net_io.packets_sent}")
            
            network_connections()
            network_io()
            
        except ImportError:
            print("网络监控需要安装psutil: pip install psutil")
    
    @staticmethod
    def process_management():
        """进程管理"""
        
        try:
            # ✅ 进程列表
            def list_processes():
                """列出进程"""
                
                processes = []
                for proc in psutil.process_iter(['pid', 'name', 'cpu_percent']):
                    try:
                        proc_info = proc.info
                        processes.append(proc_info)
                    except (psutil.NoSuchProcess, psutil.AccessDenied):
                        pass
                
                # 按CPU使用率排序
                processes.sort(key=lambda x: x['cpu_percent'] or 0, reverse=True)
                
                print("CPU使用率最高的进程:")
                for proc in processes[:5]:
                    name = proc['name']
                    pid = proc['pid']
                    cpu = proc['cpu_percent'] or 0
                    print(f"  {name} (PID: {pid}) - {cpu}%")
            
            # ✅ 进程详细信息
            def process_details():
                """进程详细信息"""
                
                # 查找Python进程
                python_processes = []
                for proc in psutil.process_iter(['pid', 'name', 'cmdline']):
                    try:
                        if proc.info['name'] and 'python' in proc.info['name'].lower():
                            python_processes.append(proc)
                    except (psutil.NoSuchProcess, psutil.AccessDenied):
                        pass
                
                if python_processes:
                    proc = python_processes[0]
                    print(f"Python进程详情:")
                    print(f"  PID: {proc.pid}")
                    print(f"  名称: {proc.name()}")
                    print(f"  命令行: {' '.join(proc.cmdline())}")
                    print(f"  内存使用: {proc.memory_info().rss / (1024**2):.1f} MB")
                    print(f"  CPU使用率: {proc.cpu_percent()}%")
                    print(f"  状态: {proc.status()}")
            
            list_processes()
            process_details()
            
        except ImportError:
            print("进程管理需要安装psutil: pip install psutil")

```python
class SystemMonitor:
    """系统监控仪表盘"""
    
    def __init__(self):
        self.metrics = {}
    
    def collect_system_metrics(self):
        """收集系统指标"""
        metrics = {
            'timestamp': datetime.now().isoformat(),
            'cpu_percent': psutil.cpu_percent(interval=1),
            'memory_usage': psutil.virtual_memory().percent,
            'disk_usage': psutil.disk_usage('/').percent if os.name != 'nt' else psutil.disk_usage('C:\\').percent,
            'active_processes': len(psutil.pids())
        }
        
        self.metrics = metrics
        return metrics
    
    def generate_status_report(self):
        """生成状态报告"""
        if not self.metrics:
            self.collect_system_metrics()
        
        status = "正常"
        if self.metrics['cpu_percent'] > 80:
            status = "CPU高负载"
        elif self.metrics['memory_usage'] > 90:
            status = "内存不足"
        
        return {
            'status': status,
            'metrics': self.metrics
        }
```

```python
# ✅ 自动化清理工具
class SystemCacheCleaner:
    """系统缓存清理工具"""
    
    def __init__(self):
        self.cleanup_rules = {
            'temp_files': {'max_age_days': 7, 'max_size_mb': 100},
            'logs': {'max_age_days': 30, 'max_size_mb': 500},
            'cache': {'max_age_days': 14, 'max_size_mb': 200}
        }
    
    def cleanup_directory(self, directory: str, file_type: str):
        """清理目录"""
        if file_type not in self.cleanup_rules:
            return
        
        rules = self.cleanup_rules[file_type]
        now = time.time()
        max_age_seconds = rules['max_age_days'] * 24 * 3600
        
        total_size = 0
        files_to_delete = []
        
        for root, dirs, files in os.walk(directory):
            for file in files:
                filepath = os.path.join(root, file)
                
                try:
                    stat = os.stat(filepath)
                    file_age = now - stat.st_mtime
                    file_size = stat.st_size
                    
                    if file_age > max_age_seconds:
                        total_size += file_size
                        files_to_delete.append(filepath)
                        
                except OSError:
                    continue
        
        return {
            'files_count': len(files_to_delete),
            'total_size_mb': total_size / (1024**2),
            'files': files_to_delete
        }
```
