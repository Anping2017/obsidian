## 列表 (List)

列表是有序的、可变的序列，可以存储不同类型的元素。

### 列表创建

```python
# 空列表
empty_list = []
empty_list = list()

# 基本列表
numbers = [1, 2, 3, 4, 5]
fruits = ["苹果", "香蕉", "橙子"]
mixed = [1, "hello", 3.14, True]

# 使用list()构造函数
numbers = list(range(1, 6))  # [1, 2, 3, 4, 5]
chars = list("Python")  # ['P', 'y', 't', 'h', 'o', 'n']

# 列表重复
repeated = [0] * 5  # [0, 0, 0, 0, 0]
```

### 列表操作

```python
fruits = ["苹果", "香蕉", "橙子"]

# 访问元素
print(fruits[0])    # 苹果
print(fruits[-1])   # 橙子

# 切片操作
print(fruits[1:3])  # ['香蕉', '橙子']
print(fruits[:2])   # ['苹果', '香蕉']
print(fruits[1:])   # ['香蕉', '橙子']

# 修改元素
fruits[0] = "葡萄"
print(fruits)  # ['葡萄', '香蕉', '橙子']

# 添加元素
fruits.append("草莓")  # 末尾添加
fruits.insert(1, "桃子")  # 指定位置插入
print(fruits)  # ['葡萄', '桃子', '香蕉', '橙子', '草莓']

# 删除元素
fruits.remove("香蕉")  # 删除指定值
del fruits[0]  # 删除指定索引
popped = fruits.pop()  # 删除并返回最后一个元素
print(f"删除的元素：{popped}")
print(fruits)  # ['桃子', '橙子']
```

### 列表方法

```python
numbers = [3, 1, 4, 1, 5, 9, 2, 6]

# 添加元素
numbers.append(7)  # 末尾添加
numbers.extend([8, 9])  # 扩展列表
numbers.insert(0, 0)  # 指定位置插入

# 删除元素
numbers.remove(1)  # 删除第一个匹配值
popped = numbers.pop()  # 删除并返回最后一个
popped = numbers.pop(0)  # 删除并返回指定位置

# 查找元素
index = numbers.index(4)  # 查找元素索引
count = numbers.count(1)  # 统计元素出现次数

# 排序
numbers.sort()  # 原地排序
numbers.sort(reverse=True)  # 降序排序
sorted_numbers = sorted(numbers)  # 返回新列表

# 反转
numbers.reverse()  # 原地反转
reversed_numbers = numbers[::-1]  # 返回新列表

# 清空
numbers.clear()  # 清空列表
```

### 列表推导式

```python
# 基本列表推导式
squares = [x**2 for x in range(10)]
print(squares)  # [0, 1, 4, 9, 16, 25, 36, 49, 64, 81]

# 带条件的列表推导式
even_squares = [x**2 for x in range(10) if x % 2 == 0]
print(even_squares)  # [0, 4, 16, 36, 64]

# 嵌套列表推导式
matrix = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]
flattened = [num for row in matrix for num in row]
print(flattened)  # [1, 2, 3, 4, 5, 6, 7, 8, 9]

# 字符串处理
words = ["hello", "world", "python"]
upper_words = [word.upper() for word in words]
print(upper_words)  # ['HELLO', 'WORLD', 'PYTHON']
```

## 元组 (Tuple)

元组是有序的、不可变的序列，通常用于存储相关的数据。

### 元组创建

```python
# 空元组
empty_tuple = ()
empty_tuple = tuple()

# 基本元组
coordinates = (10, 20)
person = ("张三", 25, "北京")
colors = ("红色", "绿色", "蓝色")

# 单元素元组（注意逗号）
single = (42,)  # 单元素元组
not_tuple = (42)  # 这不是元组，是整数

# 使用tuple()构造函数
numbers = tuple(range(1, 6))  # (1, 2, 3, 4, 5)
chars = tuple("Python")  # ('P', 'y', 't', 'h', 'o', 'n')
```

### 元组操作

```python
person = ("张三", 25, "北京")

# 访问元素
name = person[0]  # 张三
age = person[1]   # 25
city = person[2]  # 北京

# 切片操作
info = person[1:]  # (25, "北京")

# 解包
name, age, city = person
print(f"姓名：{name}，年龄：{age}，城市：{city}")

# 部分解包
name, *rest = person
print(f"姓名：{name}，其他：{rest}")

# 元组拼接
tuple1 = (1, 2, 3)
tuple2 = (4, 5, 6)
combined = tuple1 + tuple2
print(combined)  # (1, 2, 3, 4, 5, 6)

# 元组重复
repeated = tuple1 * 3
print(repeated)  # (1, 2, 3, 1, 2, 3)
```

### 元组方法

```python
numbers = (1, 2, 3, 2, 4, 2)

# 查找元素
index = numbers.index(3)  # 查找元素索引
count = numbers.count(2)  # 统计元素出现次数

# 注意：元组没有修改方法，因为元组是不可变的
```

## 集合 (Set)

集合是无序的、不重复的元素集合，支持数学集合运算。

### 集合创建

```python
# 空集合
empty_set = set()  # 注意：不能用 {} 创建空集合

# 基本集合
fruits = {"苹果", "香蕉", "橙子"}
numbers = {1, 2, 3, 4, 5}

# 使用set()构造函数
chars = set("hello")  # {'h', 'e', 'l', 'o'}
numbers = set(range(1, 6))  # {1, 2, 3, 4, 5}

# 从列表创建集合（去重）
numbers = [1, 2, 2, 3, 3, 4]
unique_numbers = set(numbers)  # {1, 2, 3, 4}
```

### 集合操作

```python
set1 = {1, 2, 3, 4}
set2 = {3, 4, 5, 6}

# 并集
union = set1 | set2  # {1, 2, 3, 4, 5, 6}
union = set1.union(set2)

# 交集
intersection = set1 & set2  # {3, 4}
intersection = set1.intersection(set2)

# 差集
difference = set1 - set2  # {1, 2}
difference = set1.difference(set2)

# 对称差集
symmetric_diff = set1 ^ set2  # {1, 2, 5, 6}
symmetric_diff = set1.symmetric_difference(set2)

# 子集和超集
set3 = {1, 2}
is_subset = set3.issubset(set1)  # True
is_superset = set1.issuperset(set3)  # True
```

### 集合方法

```python
fruits = {"苹果", "香蕉"}

# 添加元素
fruits.add("橙子")  # 添加单个元素
fruits.update(["葡萄", "草莓"])  # 添加多个元素

# 删除元素
fruits.remove("苹果")  # 删除元素（不存在会报错）
fruits.discard("香蕉")  # 删除元素（不存在不报错）
popped = fruits.pop()  # 删除并返回任意元素

# 清空
fruits.clear()  # 清空集合

# 检查元素
fruits = {"苹果", "香蕉", "橙子"}
has_apple = "苹果" in fruits  # True
has_grape = "葡萄" in fruits  # False
```

## 字典 (Dict)

字典是键值对的集合，通过键来访问值。

### 字典创建

```python
# 空字典
empty_dict = {}
empty_dict = dict()

# 基本字典
person = {"name": "张三", "age": 25, "city": "北京"}
scores = {"数学": 90, "英语": 85, "物理": 88}

# 使用dict()构造函数
person = dict(name="张三", age=25, city="北京")
person = dict([("name", "张三"), ("age", 25), ("city", "北京")])

# 使用zip()创建字典
keys = ["name", "age", "city"]
values = ["张三", 25, "北京"]
person = dict(zip(keys, values))
```

### 字典操作

```python
person = {"name": "张三", "age": 25, "city": "北京"}

# 访问元素
name = person["name"]  # 张三
age = person.get("age")  # 25
height = person.get("height", 170)  # 默认值170

# 修改元素
person["age"] = 26  # 修改现有键
person["height"] = 175  # 添加新键

# 删除元素
del person["city"]  # 删除键值对
popped = person.pop("age")  # 删除并返回值
popped = person.pop("height", 170)  # 删除并返回默认值

# 检查键
has_name = "name" in person  # True
has_height = "height" in person  # False
```

### 字典方法

```python
person = {"name": "张三", "age": 25, "city": "北京"}

# 获取所有键
keys = person.keys()  # dict_keys(['name', 'age', 'city'])
keys_list = list(person.keys())

# 获取所有值
values = person.values()  # dict_values(['张三', 25, '北京'])
values_list = list(person.values())

# 获取所有键值对
items = person.items()  # dict_items([('name', '张三'), ('age', 25), ('city', '北京')])
items_list = list(person.items())

# 更新字典
person.update({"height": 175, "weight": 70})
person.update([("phone", "13800138000"), ("email", "zhangsan@example.com")])

# 清空字典
person.clear()

# 复制字典
person = {"name": "张三", "age": 25}
person_copy = person.copy()
person_copy = dict(person)
```

### 字典推导式

```python
# 基本字典推导式
squares = {x: x**2 for x in range(5)}
print(squares)  # {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

# 带条件的字典推导式
even_squares = {x: x**2 for x in range(10) if x % 2 == 0}
print(even_squares)  # {0: 0, 2: 4, 4: 16, 6: 36, 8: 64}

# 从列表创建字典
fruits = ["苹果", "香蕉", "橙子"]
fruit_dict = {fruit: len(fruit) for fruit in fruits}
print(fruit_dict)  # {'苹果': 2, '香蕉': 2, '橙子': 2}

# 字典转换
original = {"a": 1, "b": 2, "c": 3}
inverted = {v: k for k, v in original.items()}
print(inverted)  # {1: 'a', 2: 'b', 3: 'c'}
```

## 数据结构比较

| 特性 | 列表 | 元组 | 集合 | 字典 |
|------|------|------|------|------|
| 有序性 | ✅ | ✅ | ❌ | ✅ (Python 3.7+) |
| 可变性 | ✅ | ❌ | ✅ | ✅ |
| 重复元素 | ✅ | ✅ | ❌ | ❌ (键) |
| 索引访问 | ✅ | ✅ | ❌ | ✅ (键) |
| 主要用途 | 存储序列数据 | 存储不可变数据 | 去重、集合运算 | 键值对映射 |

### 学生成绩管理系统

```python
def student_grade_system():
    """学生成绩管理系统"""
    students = {}  # 字典存储学生信息
    
    def add_student(name, grades):
        """添加学生"""
        students[name] = grades
        print(f"学生{name}已添加")
    
    def update_grade(name, subject, grade):
        """更新成绩"""
        if name in students:
            students[name][subject] = grade
            print(f"{name}的{subject}成绩已更新为{grade}")
        else:
            print(f"未找到学生{name}")
    
    def get_average(name):
        """计算平均分"""
        if name in students:
            grades = students[name].values()
            average = sum(grades) / len(grades)
            return average
        return None
    
    def get_rankings():
        """获取排名"""
        averages = {}
        for name, grades in students.items():
            averages[name] = get_average(name)
        
        # 按平均分排序
        sorted_students = sorted(averages.items(), key=lambda x: x[1], reverse=True)
        return sorted_students
    
    def get_subject_statistics(subject):
        """获取科目统计"""
        subject_grades = []
        for grades in students.values():
            if subject in grades:
                subject_grades.append(grades[subject])
        
        if subject_grades:
            return {
                "平均分": sum(subject_grades) / len(subject_grades),
                "最高分": max(subject_grades),
                "最低分": min(subject_grades),
                "人数": len(subject_grades)
            }
        return None
    
    # 主程序
    while True:
        print("\n=== 学生成绩管理系统 ===")
        print("1. 添加学生")
        print("2. 更新成绩")
        print("3. 查看学生信息")
        print("4. 计算平均分")
        print("5. 查看排名")
        print("6. 科目统计")
        print("7. 退出")
        
        choice = input("请选择操作（1-7）：")
        
        if choice == '1':
            name = input("学生姓名：")
            print("请输入各科成绩（输入'done'结束）：")
            grades = {}
            while True:
                subject = input("科目：")
                if subject == 'done':
                    break
                try:
                    grade = float(input("成绩："))
                    grades[subject] = grade
                except ValueError:
                    print("请输入有效的成绩")
            add_student(name, grades)
            
        elif choice == '2':
            name = input("学生姓名：")
            subject = input("科目：")
            try:
                grade = float(input("新成绩："))
                update_grade(name, subject, grade)
            except ValueError:
                print("请输入有效的成绩")
                
        elif choice == '3':
            name = input("学生姓名：")
            if name in students:
                print(f"\n{name}的成绩：")
                for subject, grade in students[name].items():
                    print(f"  {subject}: {grade}")
                average = get_average(name)
                print(f"平均分: {average:.2f}")
            else:
                print("未找到该学生")
                
        elif choice == '4':
            name = input("学生姓名：")
            average = get_average(name)
            if average is not None:
                print(f"{name}的平均分: {average:.2f}")
            else:
                print("未找到该学生")
                
        elif choice == '5':
            rankings = get_rankings()
            print("\n=== 学生排名 ===")
            for i, (name, average) in enumerate(rankings, 1):
                print(f"{i}. {name}: {average:.2f}")
                
        elif choice == '6':
            subject = input("科目名称：")
            stats = get_subject_statistics(subject)
            if stats:
                print(f"\n{subject}统计：")
                for key, value in stats.items():
                    print(f"  {key}: {value}")
            else:
                print("该科目没有成绩记录")
                
        elif choice == '7':
            print("感谢使用！")
            break
            
        else:
            print("无效选择，请重新输入")

# 运行学生成绩管理系统
# student_grade_system()
```

### 数据分析工具

```python
def data_analysis_tool():
    """数据分析工具"""
    
    def analyze_numbers(numbers):
        """分析数字列表"""
        if not numbers:
            return None
        
        analysis = {
            "总数": len(numbers),
            "总和": sum(numbers),
            "平均值": sum(numbers) / len(numbers),
            "最大值": max(numbers),
            "最小值": min(numbers),
            "中位数": sorted(numbers)[len(numbers) // 2] if len(numbers) % 2 == 1 
                     else (sorted(numbers)[len(numbers) // 2 - 1] + sorted(numbers)[len(numbers) // 2]) / 2
        }
        return analysis
    
    def find_duplicates(data):
        """查找重复元素"""
        seen = set()
        duplicates = set()
        for item in data:
            if item in seen:
                duplicates.add(item)
            else:
                seen.add(item)
        return list(duplicates)
    
    def group_by_length(words):
        """按长度分组单词"""
        groups = {}
        for word in words:
            length = len(word)
            if length not in groups:
                groups[length] = []
            groups[length].append(word)
        return groups
    
    def frequency_analysis(text):
        """词频分析"""
        words = text.lower().split()
        frequency = {}
        for word in words:
            # 去除标点符号
            word = word.strip(".,!?;:")
            if word:
                frequency[word] = frequency.get(word, 0) + 1
        
        # 按频率排序
        sorted_frequency = sorted(frequency.items(), key=lambda x: x[1], reverse=True)
        return sorted_frequency
    
    def correlation_analysis(x_data, y_data):
        """相关性分析"""
        if len(x_data) != len(y_data):
            return None
        
        n = len(x_data)
        sum_x = sum(x_data)
        sum_y = sum(y_data)
        sum_xy = sum(x * y for x, y in zip(x_data, y_data))
        sum_x2 = sum(x * x for x in x_data)
        sum_y2 = sum(y * y for y in y_data)
        
        # 计算相关系数
        numerator = n * sum_xy - sum_x * sum_y
        denominator = ((n * sum_x2 - sum_x**2) * (n * sum_y2 - sum_y**2))**0.5
        
        if denominator == 0:
            return 0
        
        correlation = numerator / denominator
        return correlation
    
    # 主程序
    while True:
        print("\n=== 数据分析工具 ===")
        print("1. 数字分析")
        print("2. 查找重复元素")
        print("3. 按长度分组")
        print("4. 词频分析")
        print("5. 相关性分析")
        print("6. 退出")
        
        choice = input("请选择操作（1-6）：")
        
        if choice == '1':
            numbers_input = input("请输入数字（用空格分隔）：")
            try:
                numbers = [float(x) for x in numbers_input.split()]
                analysis = analyze_numbers(numbers)
                if analysis:
                    print("\n=== 数字分析结果 ===")
                    for key, value in analysis.items():
                        print(f"{key}: {value}")
                else:
                    print("没有输入数字")
            except ValueError:
                print("请输入有效的数字")
                
        elif choice == '2':
            data_input = input("请输入数据（用空格分隔）：")
            data = data_input.split()
            duplicates = find_duplicates(data)
            if duplicates:
                print(f"重复元素: {duplicates}")
            else:
                print("没有重复元素")
                
        elif choice == '3':
            words_input = input("请输入单词（用空格分隔）：")
            words = words_input.split()
            groups = group_by_length(words)
            print("\n=== 按长度分组 ===")
            for length, word_list in sorted(groups.items()):
                print(f"长度{length}: {word_list}")
                
        elif choice == '4':
            text = input("请输入文本：")
            frequency = frequency_analysis(text)
            print("\n=== 词频分析 ===")
            for word, count in frequency[:10]:  # 显示前10个
                print(f"{word}: {count}")
                
        elif choice == '5':
            x_input = input("请输入X数据（用空格分隔）：")
            y_input = input("请输入Y数据（用空格分隔）：")
            try:
                x_data = [float(x) for x in x_input.split()]
                y_data = [float(y) for y in y_input.split()]
                correlation = correlation_analysis(x_data, y_data)
                if correlation is not None:
                    print(f"相关系数: {correlation:.4f}")
                    if abs(correlation) > 0.7:
                        strength = "强"
                    elif abs(correlation) > 0.3:
                        strength = "中等"
                    else:
                        strength = "弱"
                    print(f"相关性强度: {strength}")
                else:
                    print("数据长度不匹配")
            except ValueError:
                print("请输入有效的数字")
                
        elif choice == '6':
            print("感谢使用！")
            break
            
        else:
            print("无效选择，请重新输入")

# 运行数据分析工具
# data_analysis_tool()
```
