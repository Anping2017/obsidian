```python
# ✅ 测试金字塔框架
class TestPyramid:
    """测试金字塔框架"""
    
    @staticmethod
    def unit_tests():
        """单元测试层"""
        
        def calculator_unit_tests():
            """计算器单元测试"""
            
            class Calculator:
                def add(self, a, b):
                    return a + b
                
                @staticmethod
                def multiply(a, b):
                    return a * b
                
                @staticmethod
                def divide(a, b):
                    if b == 0:
                        raise ValueError("Division by zero")
                    return a / b
            
            calc = Calculator()
            
            assert calc.add(2, 3) == 5
            assert Calculator.multiply(4, 5) == 20
            assert Calculator.divide(10, 2) == 5
            
            with pytest.raises(ValueError):
                Calculator.divide(10, 0)
```
