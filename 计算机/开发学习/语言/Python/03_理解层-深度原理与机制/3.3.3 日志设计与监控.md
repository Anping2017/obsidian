import logging
import json
from typing import Dict, Any, Optional, Union
from datetime import datetime
from contextlib import contextmanager
import sys
import traceback

class StructuredLoggingSystem:
    """结构化日志系统"""
    
    def __init__(self):
        self.loggers = {}
        self.setup_basic_logging()
    
    def setup_basic_logging(self):
        """基础日志配置"""
        
        # 日志格式定义
        LOG_FORMAT = '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        
        # 基础配置
        logging.basicConfig(
            level=logging.INFO,
            format=LOG_FORMAT,
            handlers=[
                logging.StreamHandler(sys.stdout),
                logging.FileHandler('app.log', encoding='utf-8')
            ]
        )
        
        self.root_logger = logging.getLogger()
    
    def create_service_logger(self, service_name: str, log_file: str = None):
        """创建服务专用日志器"""
        
        if service_name in self.loggers:
            return self.loggers[service_name]
        
        # 创建专用日志器
        logger = logging.getLogger(service_name)
        logger.setLevel(logging.DEBUG)
        
        # 防止重复处理
        if not logger.handlers:
            # 文件处理器
            if log_file:
                file_handler = logging.FileHandler(log_file, encoding='utf-8')
                file_handler.setFormatter(logging.Formatter(LOG_FORMAT))
                logger.addHandler(file_handler)
            
            # 控制台处理器
            console_handler = logging.StreamHandler(sys.stdout)
            console_handler.setFormatter(logging.Formatter(LOG_FORMAT))
            logger.addHandler(console_handler)
        
        self.loggers[service_name] = logger
        return logger
    
    @staticmethod
    def structured_log_example():
        """结构化日志示例"""
        
        # ✅ 业务日志结构
        class BusinessLogger:
            """业务日志记录器"""
            
            def __init__(self, service_name: str):
                self.logger = logging.getLogger(service_name)
                self.context = {}
            
            def set_context(self, **context):
                """设置上下文信息"""
                self.context.update(context)
            
            def log_business_event(self, event_type: str, **event_data):
                """记录业务事件"""
                log_entry = {
                    'timestamp': datetime.now().isoformat(),
                    'event_type': event_type,
                    'service': self.logger.name,
                    'level': 'INFO',
                    'data': event_data,
                    'context': self.context.copy()
                }
                
                self.logger.info(json.dumps(log_entry, ensure_ascii=False))
            
            def log_user_action(self, user_id: str, action: str, **action_data):
                """记录用户行为"""
                self.log_business_event(
                    'user_action',
                    user_id=user_id,
                    action=action,
                    **action_data
                )
            
            def log_transaction(self, transaction_id: str, status: str, 
                              amount: float = None, **transaction_data):
                """记录交易信息"""
                self.log_business_event(
                    'transaction',
                    transaction_id=transaction_id,
                    status=status,
                    amount=amount,
                    **transaction_data
                )
        
        # 使用示例
        business_logger = BusinessLogger('payment_service')
        business_logger.set_context(request_id='req_123', session_id='sess_456')
        
        business_logger.log_user_action(
            user_id='user_789',
            action='purchase',
            product_id='prod_001',
            quantity=2
        )
        
        business_logger.log_transaction(
            transaction_id='txn_001',
            status='completed',
            amount=99.98,
            payment_method='credit_card'
        )

class PerformanceMonitoring:
    """性能监控系统"""
    
    def __init__(self):
        self.metrics_logger = logging.getLogger('performance')
        self.metrics_logger.setLevel(logging.INFO)
        
        # 性能指标处理器
        handler = logging.FileHandler('performance.log', encoding='utf-8')
        handler.setFormatter(logging.Formatter('%(message)s'))
        self.metrics_logger.addHandler(handler)
    
    def log_execution_time(self, function_name: str, execution_time: float, 
                          **metadata):
        """记录执行时间"""
        metric_data = {
            'timestamp': datetime.now().isoformat(),
            'metric_type': 'execution_time',
            'function_name': function_name,
            'value': execution_time,
            'unit': 'seconds',
            'metadata': metadata
        }
        
        self.metrics_logger.info(json.dumps(metric_data, ensure_ascii=False))
    
    def log_memory_usage(self, current_memory: float, peak_memory: float = None,
                        **metadata):
        """记录内存使用"""
        metric_data = {
            'timestamp': datetime.now().isoformat(),
            'metric_type': 'memory_usage',
            'current_memory': current_memory,
            'peak_memory': peak_memory,
            'unit': 'bytes',
            'metadata': metadata
        }
        
        self.metrics_logger.info(json.dumps(metric_data, ensure_ascii=False))
    
    def log_api_metrics(self, endpoint: str, method: str, status_code: int,
                       response_time: float, request_size: int = None,
                       response_size: int = None):
        """记录API指标"""
        metric_data = {
            'timestamp': datetime.now().isoformat(),
            'metric_type': 'api_request',
            'endpoint': endpoint,
            'method': method,
            'status_code': status_code,
            'response_time': response_time,
            'request_size': request_size,
            'response_size': response_size
        }
        
        self.metrics_logger.info(json.dumps(metric_data, ensure_ascii=False))

# ✅ 错误监控和追踪
class ErrorMonitoring:
    """错误监控系统"""
    
    def __init__(self):
        self.error_logger = logging.getLogger('error_monitoring')
        self.error_logger.setLevel(logging.ERROR)
        
        # 错误处理器
        handler = logging.FileHandler('errors.log', encoding='utf-8')
        handler.setFormatter(logging.Formatter('%(message)s'))
        self.error_logger.addHandler(handler)
    
    def log_error_with_context(self, error: Exception, **context):
        """记录错误及其上下文"""
        error_data = {
            'timestamp': datetime.now().isoformat(),
            'error_type': type(error).__name__,
            'error_message': str(error),
            'context': context,
            'stack_trace': traceback.format_exc()
        }
        
        self.error_logger.error(json.dumps(error_data, ensure_ascii=False))
    
    def log_security_event(self, event_type: str, user_id: str = None,
                          ip_address: str = None, **event_data):
        """记录安全事件"""
        security_data = {
            'timestamp': datetime.now().isoformat(),
            'event_type': event_type,
            'user_id': user_id,
            'ip_address': ip_address,
            'severity': 'HIGH',
            'event_data': event_data
        }
        
        self.error_logger.error(json.dumps(security_data, ensure_ascii=False))

class MonitoringDashboard:
    """监控仪表盘"""
    
    def __init__(self):
        self.system_metrics = {}
        self.business_metrics = {}
        self.health_checks = {}
    
    def collect_system_metrics(self):
        """收集系统指标"""
        import psutil
        
        metrics = {
            'timestamp': datetime.now().isoformat(),
            'cpu_percent': psutil.cpu_percent(interval=1),
            'memory_usage': dict(psutil.virtual_memory()._asdict()),
            'disk_usage': dict(psutil.disk_usage('/')._asdict()),
            'network_io': dict(psutil.net_io_counters()._asdict())
        }
        
        self.system_metrics = metrics
        return metrics
    
    def collect_business_metrics(self, business_data: dict):
        """收集业务指标"""
        metrics = {
            'timestamp': datetime.now().isoformat(),
            'active_users': business_data.get('active_users', 0),
            'transactions_per_minute': business_data.get('transactions_per_minute', 0),
            'conversion_rate': business_data.get('conversion_rate', 0.0),
            'api_call_count': business_data.get('api_call_count', 0)
        }
        
        self.business_metrics = metrics
        return metrics
    
    def health_check(self, service_name: str, check_results: dict):
        """健康检查"""
        health_status = {
            'timestamp': datetime.now().isoformat(),
            'service': service_name,
            'status': 'healthy' if sum(check_results.values()) == len(check_results) else 'unhealthy',
            'checks': check_results,
            'overall_score': sum(check_results.values()) / len(check_results) if check_results else 0
        }
        
        self.health_checks[service_name] = health_status
        return health_status
    
    def generate_dashboard_data(self):
        """生成仪表盘数据"""
        dashboard_data = {
            'timestamp': datetime.now().isoformat(),
            'system_metrics': self.system_metrics,
            'business_metrics': self.business_metrics,
            'health_checks': self.health_checks,
            'status': 'operational'
        }
        
        return dashboard_data

# ✅ 上下文管理器日志
@contextmanager
def log_execution_context(logger: logging.Logger, operation_name: str, **context):
    """执行上下文日志管理器"""
    start_time = datetime.now()
    logger.info(f"开始执行: {operation_name}", extra={'context': context})
    
    try:
        yield
        logger.info(f"成功完成: {operation_name}")
    except Exception as e:
        logger.error(f"执行失败: {operation_name}", exc_info=True)
        raise
    finally:
        duration = (datetime.now() - start_time).total_seconds()
        logger.info(f"执行耗时: {duration:.3f}秒")

# 使用示例
logger = logging.getLogger('service')
with log_execution_context(logger, 'data_validation', user_id='123'):
    # 执行业务逻辑
    pass

# ✅ 智能日志采样
class SmartLogSampler:
    """智能日志采样器"""
    
    def __init__(self, sample_rate: float = 0.1):
        self.sample_rate = sample_rate
        self.importance_categories = {
            'critical': 1.0,    # 关键日志总是记录
            'error': 0.5,       # 错误日志50%采样
            'warning': 0.1,     # 警告日志10%采样
            'info':  0.05,       # 信息日志5%采样
            'debug': 0.001      # 调试日志0.1%采样
        }
    
    def should_log(self, level: str, extra_data: dict = None) -> bool:
        """判断是否应该记录日志"""
        if level.lower() not in self.importance_categories:
            return True
        
        importance_rate = self.importance_categories[level.lower()]
        
        # 根据业务重要性调整采样率
        if extra_data and 'business_critical' in extra_data:
            importance_rate *= 2
        
        return random.random() < min(importance_rate, 1.0)
```

# ✅ 分布式追踪
class DistributedTracing:
    """分布式追踪系统"""
    
    def __init__(self):
        self.trace_id = None
        self.span_id = None
        self.parent_span_id = None
    
    def start_trace(self, trace_id: str = None):
        """开始追踪"""
        import uuid
        self.trace_id = trace_id or str(uuid.uuid4())
        self.span_id = str(uuid.uuid4())
        self.parent_span_id = None
        
        logging.getLogger('tracing').info(
            f"开始追踪: {self.trace_id}",
            extra={
                'trace_id': self.trace_id,
                'span_id': self.span_id
            }
        )
    
    def create_child_span(self, operation_name: str):
        """创建子Span"""
        import uuid
        child_id = str(uuid.uuid4())
        
        logging.getLogger('tracing').info(
            f"创建子Span: {child_id} 用于操作: {operation_name}",
            extra={
                'trace_id': self.trace_id,
                'parent_span_id': self.span_id,
                'span_id': child_id,
                'operation': operation_name
            }
        )
        
        return child_id
```
