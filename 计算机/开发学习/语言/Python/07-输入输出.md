## 输出操作

### print函数

```python
# 基本输出
print("Hello, World!")
print(42)
print(3.14)

# 输出多个值
name = "张三"
age = 25
print("姓名：", name, "年龄：", age)

# 使用分隔符
print("apple", "banana", "orange", sep=", ")  # apple, banana, orange
print("apple", "banana", "orange", sep="-")   # apple-banana-orange

# 控制结束符
print("Hello", end=" ")
print("World")  # Hello World

# 输出到文件
with open("output.txt", "w") as f:
    print("Hello, File!", file=f)
```

### 格式化输出

```python
name = "张三"
age = 25
score = 85.5

# 使用f-string（推荐）
print(f"姓名：{name}，年龄：{age}，分数：{score}")

# 格式化数字
pi = 3.14159
print(f"π的值：{pi:.2f}")  # 保留2位小数
print(f"π的值：{pi:.4f}")  # 保留4位小数

# 格式化宽度
print(f"姓名：{name:>10}")  # 右对齐，宽度10
print(f"年龄：{age:<10}")   # 左对齐，宽度10
print(f"分数：{score:^10}") # 居中对齐，宽度10

# 使用format()方法
print("姓名：{}，年龄：{}，分数：{}".format(name, age, score))
print("姓名：{name}，年龄：{age}，分数：{score}".format(name=name, age=age, score=score))

# 使用%格式化（传统方法）
print("姓名：%s，年龄：%d，分数：%.2f" % (name, age, score))
```

### 输出重定向

```python
import sys

# 重定向到文件
with open("output.txt", "w") as f:
    print("这行文字会写入文件", file=f)

# 重定向到变量
from io import StringIO
output = StringIO()
print("这行文字会存储在变量中", file=output)
content = output.getvalue()
print(f"捕获的内容：{content}")

# 临时重定向
import sys
original_stdout = sys.stdout
with open("output.txt", "w") as f:
    sys.stdout = f
    print("这行文字会写入文件")
sys.stdout = original_stdout
```

### 输入验证

```python
def get_valid_input(prompt, input_type=str, min_val=None, max_val=None):
    """获取有效输入"""
    while True:
        try:
            value = input(prompt)
            if input_type == int:
                value = int(value)
            elif input_type == float:
                value = float(value)
            
            if min_val is not None and value < min_val:
                print(f"值不能小于{min_val}")
                continue
            if max_val is not None and value > max_val:
                print(f"值不能大于{max_val}")
                continue
            
            return value
        except ValueError:
            print("请输入有效的值")

# 使用输入验证
age = get_valid_input("请输入年龄：", int, 0, 120)
score = get_valid_input("请输入分数：", float, 0, 100)
name = get_valid_input("请输入姓名：", str)
```

### 多行输入

```python
def get_multiline_input():
    """获取多行输入"""
    print("请输入多行文本（输入'END'结束）：")
    lines = []
    while True:
        line = input()
        if line == "END":
            break
        lines.append(line)
    return "\n".join(lines)

# 使用多行输入
text = get_multiline_input()
print("您输入的内容：")
print(text)
```

### 文件读取

```python
# 读取整个文件
with open("example.txt", "r", encoding="utf-8") as file:
    content = file.read()
    print(content)

# 逐行读取
with open("example.txt", "r", encoding="utf-8") as file:
    for line in file:
        print(line.strip())  # 去除换行符

# 读取所有行
with open("example.txt", "r", encoding="utf-8") as file:
    lines = file.readlines()
    for line in lines:
        print(line.strip())

# 读取指定字节数
with open("example.txt", "r", encoding="utf-8") as file:
    chunk = file.read(100)  # 读取前100个字符
    print(chunk)
```

### 上下文管理器

```python
# 自定义上下文管理器
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None
    
    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file:
            self.file.close()

# 使用自定义上下文管理器
with FileManager("example.txt", "r") as file:
    content = file.read()
    print(content)
```

## 异常处理

```python
# 基本异常处理
try:
    with open("nonexistent.txt", "r") as file:
        content = file.read()
except FileNotFoundError:
    print("文件不存在")
except PermissionError:
    print("没有权限访问文件")
except Exception as e:
    print(f"发生错误：{e}")

# 输入异常处理
def safe_input(prompt, input_type=str):
    """安全的输入函数"""
    while True:
        try:
            value = input(prompt)
            if input_type == int:
                return int(value)
            elif input_type == float:
                return float(value)
            else:
                return value
        except ValueError:
            print("输入格式错误，请重新输入")
        except KeyboardInterrupt:
            print("\n程序被用户中断")
            return None

# 使用安全输入
age = safe_input("请输入年龄：", int)
if age is not None:
    print(f"您的年龄是：{age}")
```

## 示例代码

### 文件管理器

```python
def file_manager():
    """简单文件管理器"""
    
    def create_file(filename, content=""):
        """创建文件"""
        try:
            with open(filename, "w", encoding="utf-8") as file:
                file.write(content)
            print(f"文件 {filename} 创建成功")
        except Exception as e:
            print(f"创建文件失败：{e}")
    
    def read_file(filename):
        """读取文件"""
        try:
            with open(filename, "r", encoding="utf-8") as file:
                content = file.read()
            print(f"文件 {filename} 内容：")
            print(content)
        except FileNotFoundError:
            print(f"文件 {filename} 不存在")
        except Exception as e:
            print(f"读取文件失败：{e}")
    
    def append_file(filename, content):
        """追加内容到文件"""
        try:
            with open(filename, "a", encoding="utf-8") as file:
                file.write(content + "\n")
            print(f"内容已追加到 {filename}")
        except Exception as e:
            print(f"追加内容失败：{e}")
    
    def copy_file(source, destination):
        """复制文件"""
        try:
            with open(source, "r", encoding="utf-8") as src:
                content = src.read()
            with open(destination, "w", encoding="utf-8") as dst:
                dst.write(content)
            print(f"文件 {source} 已复制到 {destination}")
        except FileNotFoundError:
            print(f"源文件 {source} 不存在")
        except Exception as e:
            print(f"复制文件失败：{e}")
    
    def get_file_info(filename):
        """获取文件信息"""
        try:
            import os
            if os.path.exists(filename):
                size = os.path.getsize(filename)
                print(f"文件 {filename} 信息：")
                print(f"  大小：{size} 字节")
                print(f"  路径：{os.path.abspath(filename)}")
            else:
                print(f"文件 {filename} 不存在")
        except Exception as e:
            print(f"获取文件信息失败：{e}")
    
    # 主程序
    while True:
        print("\n=== 文件管理器 ===")
        print("1. 创建文件")
        print("2. 读取文件")
        print("3. 追加内容")
        print("4. 复制文件")
        print("5. 文件信息")
        print("6. 退出")
        
        choice = input("请选择操作（1-6）：")
        
        if choice == '1':
            filename = input("文件名：")
            content = input("文件内容（可选）：")
            create_file(filename, content)
            
        elif choice == '2':
            filename = input("文件名：")
            read_file(filename)
            
        elif choice == '3':
            filename = input("文件名：")
            content = input("要追加的内容：")
            append_file(filename, content)
            
        elif choice == '4':
            source = input("源文件名：")
            destination = input("目标文件名：")
            copy_file(source, destination)
            
        elif choice == '5':
            filename = input("文件名：")
            get_file_info(filename)
            
        elif choice == '6':
            print("感谢使用！")
            break
            
        else:
            print("无效选择，请重新输入")

# 运行文件管理器

```

### 日志记录器

```python
import datetime
import os

def logger():
    """简单日志记录器"""
    
    def log_message(message, level="INFO"):
        """记录日志消息"""
        timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        log_entry = f"[{timestamp}] {level}: {message}\n"
        
        # 确保日志目录存在
        log_dir = "logs"
        if not os.path.exists(log_dir):
            os.makedirs(log_dir)
        
        # 写入日志文件
        log_file = os.path.join(log_dir, "app.log")
        try:
            with open(log_file, "a", encoding="utf-8") as file:
                file.write(log_entry)
            print(f"日志已记录：{message}")
        except Exception as e:
            print(f"记录日志失败：{e}")
    
    def read_logs():
        """读取日志"""
        log_file = "logs/app.log"
        try:
            if os.path.exists(log_file):
                with open(log_file, "r", encoding="utf-8") as file:
                    content = file.read()
                print("=== 日志内容 ===")
                print(content)
            else:
                print("日志文件不存在")
        except Exception as e:
            print(f"读取日志失败：{e}")
    
    def clear_logs():
        """清空日志"""
        log_file = "logs/app.log"
        try:
            if os.path.exists(log_file):
                with open(log_file, "w", encoding="utf-8") as file:
                    file.write("")
                print("日志已清空")
            else:
                print("日志文件不存在")
        except Exception as e:
            print(f"清空日志失败：{e}")
    
    # 主程序
    while True:
        print("\n=== 日志记录器 ===")
        print("1. 记录信息日志")
        print("2. 记录警告日志")
        print("3. 记录错误日志")
        print("4. 查看日志")
        print("5. 清空日志")
        print("6. 退出")
        
        choice = input("请选择操作（1-6）：")
        
        if choice == '1':
            message = input("信息消息：")
            log_message(message, "INFO")
            
        elif choice == '2':
            message = input("警告消息：")
            log_message(message, "WARNING")
            
        elif choice == '3':
            message = input("错误消息：")
            log_message(message, "ERROR")
            
        elif choice == '4':
            read_logs()
            
        elif choice == '5':
            confirm = input("确定要清空日志吗？(y/n)：")
            if confirm.lower() == 'y':
                clear_logs()
                
        elif choice == '6':
            print("感谢使用！")
            break
            
        else:
            print("无效选择，请重新输入")

# 运行日志记录器

```

### 数据备份工具

```python
import shutil
import os
import datetime

def backup_tool():
    """数据备份工具"""
    
    def backup_file(source, backup_dir="backups"):
        """备份单个文件"""
        try:
            # 创建备份目录
            if not os.path.exists(backup_dir):
                os.makedirs(backup_dir)
            
            # 生成备份文件名
            timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
            filename = os.path.basename(source)
            name, ext = os.path.splitext(filename)
            backup_filename = f"{name}_{timestamp}{ext}"
            backup_path = os.path.join(backup_dir, backup_filename)
            
            # 复制文件
            shutil.copy2(source, backup_path)
            print(f"文件 {source} 已备份到 {backup_path}")
            
        except FileNotFoundError:
            print(f"源文件 {source} 不存在")
        except Exception as e:
            print(f"备份失败：{e}")
    
    def backup_directory(source, backup_dir="backups"):
        """备份整个目录"""
        try:
            # 创建备份目录
            if not os.path.exists(backup_dir):
                os.makedirs(backup_dir)
            
            # 生成备份目录名
            timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
            dirname = os.path.basename(source)
            backup_dirname = f"{dirname}_{timestamp}"
            backup_path = os.path.join(backup_dir, backup_dirname)
            
            # 复制目录
            shutil.copytree(source, backup_path)
            print(f"目录 {source} 已备份到 {backup_path}")
            
        except FileNotFoundError:
            print(f"源目录 {source} 不存在")
        except Exception as e:
            print(f"备份失败：{e}")
    
    def list_backups(backup_dir="backups"):
        """列出备份文件"""
        try:
            if os.path.exists(backup_dir):
                backups = os.listdir(backup_dir)
                if backups:
                    print(f"=== {backup_dir} 中的备份 ===")
                    for backup in sorted(backups):
                        backup_path = os.path.join(backup_dir, backup)
                        size = os.path.getsize(backup_path) if os.path.isfile(backup_path) else "目录"
                        print(f"  {backup} ({size})")
                else:
                    print("没有找到备份文件")
            else:
                print("备份目录不存在")
        except Exception as e:
            print(f"列出备份失败：{e}")
    
    def restore_backup(backup_path, destination):
        """恢复备份"""
        try:
            if os.path.exists(backup_path):
                if os.path.isfile(backup_path):
                    shutil.copy2(backup_path, destination)
                    print(f"文件已从 {backup_path} 恢复到 {destination}")
                elif os.path.isdir(backup_path):
                    if os.path.exists(destination):
                        shutil.rmtree(destination)
                    shutil.copytree(backup_path, destination)
                    print(f"目录已从 {backup_path} 恢复到 {destination}")
            else:
                print(f"备份 {backup_path} 不存在")
        except Exception as e:
            print(f"恢复失败：{e}")
    
    # 主程序
    while True:
        print("\n=== 数据备份工具 ===")
        print("1. 备份文件")
        print("2. 备份目录")
        print("3. 列出备份")
        print("4. 恢复备份")
        print("5. 退出")
        
        choice = input("请选择操作（1-5）：")
        
        if choice == '1':
            source = input("要备份的文件路径：")
            backup_file(source)
            
        elif choice == '2':
            source = input("要备份的目录路径：")
            backup_directory(source)
            
        elif choice == '3':
            list_backups()
            
        elif choice == '4':
            backup_path = input("备份文件/目录路径：")
            destination = input("恢复目标路径：")
            restore_backup(backup_path, destination)
            
        elif choice == '5':
            print("感谢使用！")
            break
            
        else:
            print("无效选择，请重新输入")

```
