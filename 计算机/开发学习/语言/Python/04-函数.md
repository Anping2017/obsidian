## 函数定义

函数是组织好的、可重复使用的代码块：

```python
# 基本函数定义
def greet():
    """简单的问候函数"""
    print("Hello, World!")

# 带参数的函数
def greet_person(name):
    """问候指定的人"""
    print(f"Hello, {name}!")

# 带多个参数的函数
def add_numbers(a, b):
    """计算两个数的和"""
    return a + b

# 带默认参数的函数
def greet_with_title(name, title="先生"):
    """带称谓的问候"""
    print(f"Hello, {title}{name}!")
```

### 函数文档字符串

```python
def calculate_area(length, width):
    """
    计算矩形的面积
    
    参数:
        length (float): 矩形的长度
        width (float): 矩形的宽度
    
    返回:
        float: 矩形的面积
    """
    return length * width

# 查看函数文档
print(calculate_area.__doc__)
```


## 参数

### 位置参数

按位置顺序传递参数：

```python
def introduce(name, age, city):
    """介绍个人信息"""
    print(f"我叫{name}，今年{age}岁，来自{city}")

# 按位置传递参数
introduce("张三", 25, "北京")
introduce("李四", 30, "上海")
```

### 默认参数

为参数设置默认值：

```python
def create_user(name, age, city="未知", status="活跃"):
    """创建用户信息"""
    print(f"用户：{name}")
    print(f"年龄：{age}")
    print(f"城市：{city}")
    print(f"状态：{status}")

# 使用默认参数
create_user("张三", 25)
create_user("李四", 30, "上海")
create_user("王五", 28, "广州", "离线")
```

### 关键字参数

通过参数名传递参数：

```python
def create_user(name, age, city="未知", status="活跃"):
    """创建用户信息"""
    print(f"用户：{name}")
    print(f"年龄：{age}")
    print(f"城市：{city}")
    print(f"状态：{status}")

# 使用关键字参数
create_user(name="张三", age=25)
create_user(name="李四", age=30, city="上海")
create_user(name="王五", age=28, status="离线", city="广州")
```

### 可变参数

#### *args - 可变位置参数

```python
def sum_numbers(*args):
    """计算任意数量数字的和"""
    total = 0
    for num in args:
        total += num
    return total

# 调用可变参数函数
result1 = sum_numbers(1, 2, 3)
result2 = sum_numbers(1, 2, 3, 4, 5)
result3 = sum_numbers()

print(f"结果1：{result1}")  # 6
print(f"结果2：{result2}")  # 15
print(f"结果3：{result3}")  # 0
```

#### **kwargs - 可变关键字参数

```python
def create_profile(**kwargs):
    """创建用户档案"""
    print("用户档案：")
    for key, value in kwargs.items():
        print(f"  {key}: {value}")

# 调用可变关键字参数函数
create_profile(name="张三", age=25, city="北京")
create_profile(name="李四", age=30, city="上海", hobby="编程")
```

#### 组合使用

```python
def complex_function(required_arg, *args, default_arg="默认值", **kwargs):
    """复杂参数函数"""
    print(f"必需参数：{required_arg}")
    print(f"位置参数：{args}")
    print(f"默认参数：{default_arg}")
    print(f"关键字参数：{kwargs}")

# 调用复杂函数
complex_function("必需", 1, 2, 3, default_arg="新值", key1="值1", key2="值2")
```

### 参数解包

```python
def multiply(a, b, c):
    """计算三个数的乘积"""
    return a * b * c

# 使用列表解包
numbers = [2, 3, 4]
result = multiply(*numbers)
print(f"结果：{result}")  # 24

# 使用字典解包
params = {"a": 2, "b": 3, "c": 4}
result = multiply(**params)
print(f"结果：{result}")  # 24
```

### 多个返回值

```python
def get_stats(numbers):
    """计算统计信息"""
    if not numbers:
        return 0, 0, 0
    
    total = sum(numbers)
    average = total / len(numbers)
    maximum = max(numbers)
    
    return total, average, maximum

# 接收多个返回值
numbers = [1, 2, 3, 4, 5]
total, avg, max_val = get_stats(numbers)
print(f"总和：{total}")
print(f"平均值：{avg}")
print(f"最大值：{max_val}")
```

### 返回函数

```python
def create_multiplier(factor):
    """创建乘法器函数"""
    def multiplier(x):
        return x * factor
    return multiplier

# 创建不同的乘法器
double = create_multiplier(2)
triple = create_multiplier(3)

print(f"2的2倍：{double(2)}")  # 4
print(f"2的3倍：{triple(2)}")  # 6
```

## 作用域

### 局部变量和全局变量

```python
# 全局变量
global_var = "我是全局变量"

def test_scope():
    # 局部变量
    local_var = "我是局部变量"
    print(f"函数内访问全局变量：{global_var}")
    print(f"函数内访问局部变量：{local_var}")

def modify_global():
    global global_var
    global_var = "全局变量被修改了"
    print(f"函数内修改全局变量：{global_var}")

# 测试作用域
test_scope()
print(f"函数外访问全局变量：{global_var}")

modify_global()
print(f"函数外访问修改后的全局变量：{global_var}")
```

### 嵌套函数

```python
def outer_function(x):
    """外部函数"""
    def inner_function(y):
        """内部函数"""
        return x + y
    return inner_function

# 创建闭包
closure = outer_function(10)
result = closure(5)
print(f"闭包结果：{result}")  # 15
```

## lambda表达式

lambda表达式用于创建匿名函数：

```python
# 基本lambda表达式
square = lambda x: x * x
print(f"5的平方：{square(5)}")

# 多参数lambda
add = lambda x, y: x + y
print(f"3 + 4 = {add(3, 4)}")

# 带默认参数的lambda
greet = lambda name, title="先生": f"Hello, {title}{name}!"
print(greet("张三"))
print(greet("李四", "女士"))
```

### lambda与内置函数结合

```python
# 与map()结合
numbers = [1, 2, 3, 4, 5]
squares = list(map(lambda x: x * x, numbers))
print(f"平方数：{squares}")

# 与filter()结合
even_numbers = list(filter(lambda x: x % 2 == 0, numbers))
print(f"偶数：{even_numbers}")

# 与sorted()结合
students = [("张三", 85), ("李四", 92), ("王五", 78)]
sorted_students = sorted(students, key=lambda x: x[1], reverse=True)
print(f"按成绩排序：{sorted_students}")
```

## 高阶函数

### 函数作为参数

```python
def apply_operation(x, y, operation):
    """应用指定操作"""
    return operation(x, y)

# 定义操作函数
def add(a, b):
    return a + b

def multiply(a, b):
    return a * b

# 使用高阶函数
result1 = apply_operation(5, 3, add)
result2 = apply_operation(5, 3, multiply)
result3 = apply_operation(5, 3, lambda x, y: x ** y)

print(f"加法：{result1}")  # 8
print(f"乘法：{result2}")  # 15
print(f"幂运算：{result3}")  # 125
```

### 装饰器

```python
def timer(func):
    """计时装饰器"""
    import time
    def wrapper(*args, **kwargs):
        start_time = time.time()
        result = func(*args, **kwargs)
        end_time = time.time()
        print(f"{func.__name__}执行时间：{end_time - start_time:.4f}秒")
        return result
    return wrapper

@timer
def slow_function():
    """慢函数"""
    import time
    time.sleep(1)
    return "完成"

# 使用装饰器
result = slow_function()
print(result)
```

## 示例代码

### 计算器函数

```python
def calculator():
    """简单计算器"""
    def add(a, b):
        return a + b
    
    def subtract(a, b):
        return a - b
    
    def multiply(a, b):
        return a * b
    
    def divide(a, b):
        if b == 0:
            return "错误：除数不能为零"
        return a / b
    
    operations = {
        '+': add,
        '-': subtract,
        '*': multiply,
        '/': divide
    }
    
    while True:
        print("\n=== 计算器 ===")
        print("支持的操作：+, -, *, /")
        print("输入'quit'退出")
        
        try:
            expression = input("请输入表达式（如：5 + 3）：")
            if expression.lower() == 'quit':
                break
            
            parts = expression.split()
            if len(parts) != 3:
                print("格式错误，请使用：数字 操作符 数字")
                continue
            
            a, op, b = float(parts[0]), parts[1], float(parts[2])
            
            if op in operations:
                result = operations[op](a, b)
                print(f"{a} {op} {b} = {result}")
            else:
                print("不支持的操作符")
                
        except ValueError:
            print("请输入有效的数字")
        except Exception as e:
            print(f"发生错误：{e}")

# 运行计算器
# calculator()
```

### 学生管理系统

```python
def student_management_system():
    """学生管理系统"""
    students = []
    
    def add_student(name, age, grade):
        """添加学生"""
        student = {"name": name, "age": age, "grade": grade}
        students.append(student)
        print(f"学生{name}已添加")
    
    def find_student(name):
        """查找学生"""
        for student in students:
            if student["name"] == name:
                return student
        return None
    
    def update_student(name, **kwargs):
        """更新学生信息"""
        student = find_student(name)
        if student:
            for key, value in kwargs.items():
                if key in student:
                    student[key] = value
            print(f"学生{name}信息已更新")
        else:
            print(f"未找到学生{name}")
    
    def delete_student(name):
        """删除学生"""
        global students
        students = [s for s in students if s["name"] != name]
        print(f"学生{name}已删除")
    
    def list_students():
        """列出所有学生"""
        if not students:
            print("没有学生记录")
            return
        
        print("\n=== 学生列表 ===")
        for i, student in enumerate(students, 1):
            print(f"{i}. {student['name']} - 年龄：{student['age']} - 年级：{student['grade']}")
    
    def get_statistics():
        """获取统计信息"""
        if not students:
            print("没有学生记录")
            return
        
        total = len(students)
        ages = [s["age"] for s in students]
        avg_age = sum(ages) / len(ages)
        
        print(f"\n=== 统计信息 ===")
        print(f"总学生数：{total}")
        print(f"平均年龄：{avg_age:.1f}")
        print(f"最小年龄：{min(ages)}")
        print(f"最大年龄：{max(ages)}")
    
    # 主菜单
    while True:
        print("\n=== 学生管理系统 ===")
        print("1. 添加学生")
        print("2. 查找学生")
        print("3. 更新学生")
        print("4. 删除学生")
        print("5. 列出学生")
        print("6. 统计信息")
        print("7. 退出")
        
        choice = input("请选择操作（1-7）：")
        
        if choice == '1':
            name = input("姓名：")
            age = int(input("年龄："))
            grade = input("年级：")
            add_student(name, age, grade)
            
        elif choice == '2':
            name = input("要查找的学生姓名：")
            student = find_student(name)
            if student:
                print(f"找到学生：{student}")
            else:
                print("未找到该学生")
                
        elif choice == '3':
            name = input("要更新的学生姓名：")
            print("输入要更新的字段（留空跳过）：")
            age = input("新年龄：")
            grade = input("新年级：")
            
            updates = {}
            if age:
                updates["age"] = int(age)
            if grade:
                updates["grade"] = grade
                
            update_student(name, **updates)
            
        elif choice == '4':
            name = input("要删除的学生姓名：")
            delete_student(name)
            
        elif choice == '5':
            list_students()
            
        elif choice == '6':
            get_statistics()
            
        elif choice == '7':
            print("感谢使用！")
            break
            
        else:
            print("无效选择，请重新输入")

# 运行学生管理系统
# student_management_system()
```
