# 2.3.2 闭包与词法作用域

## 🎯 核心理念

闭包是Python的"词法绑定艺术"。它捕获外层的自由变量，即使外层函数返回后，内层函数仍能访问这些变量，体现了"词汇决定作用域"的语言特性。

def closure_deep_dive():

    def create_counter():
        """创建计数器 - 闭包基础"""
        count = [0]  # 使用列表避免问题
        
        def counter():
            count[0] += 1
            return count[0]
        
        return counter

    counter1 = create_counter()
    counter2 = create_counter()
    
    print(f"计数器1: {counter1()}, {counter1()}, {counter1()}")

                print(f"访问外部变量: x={x}, y={y}, 内部参数: z={z}")
                return x + y + z
            
            return inner_function
        
        closure_func = outer_function(10)
        result = closure_func(5)
        print(f"闭包结果: {result}")

class AdvancedClosurePatterns:
    """高级闭包模式"""
    
    @staticmethod
    def configuration_manager():
        """配置管理器 - 闭包实现"""
        
        def create_config_manager(default_config):
            """创建配置管理器"""
            config = default_config.copy()
            
            def get_config(key):
                return config.get(key)
            
            def set_config(key, value):
                config[key] = value
                return config
            
            def update_config(updates):
                config.update(updates)
                return config
            
            def reset_config():
                config.clear()
                config.update(default_config)
                return config
            
            # 返回所有方法
            return {
                'get': get_config,
                'set': set_config,
                'update': update_config,
                'reset': reset_config,
                '_config': lambda: config.copy()
            }
        
        # 创建配置管理器
        default_settings = {
            'debug': False,
            'max_connections': 100,
            'timeout': 30,
            'retry_count': 3
        }
        
        config_mgr = create_config_manager(default_settings)
        
        # 测试配置管理
        print(f"初始调试模式: {config_mgr['get']('debug')}")
        config_mgr['set']('debug', True)
        print(f"更新后调试模式: {config_mgr['get']('debug')}")
        print(f"完整配置: {config_mgr['_config']()}")

def closure_trap_solutions():
    """闭包陷阱与解决方案"""
    
    # ❌ 常见陷阱
    def common_trap():
        """常见闭包陷阱"""
        functions = []
        
        for i in range(3):
            def trap_function():
                return i  # 闭包捕获的是变量引用，不是值
            functions.append(trap_function)
        
        print("陷阱结果:")
        for j, func in enumerate(functions):
            print(f"func{j}(): {func()}")
    
    # ✅ 解决方案1: 默认参数
    def solution_1():
        """解决方案1: 默认参数"""
        functions = []
        
        for i in range(3):
            def good_function(i=i):
                return i
            functions.append(good_function)
        
        print("\n解决方案1结果:")
        for j, func in enumerate(functions):
            print(f"func{j}(): {func()}")
    
    # ✅ 解决方案2: 工厂函数
    def solution_2():
        """解决方案2: 工厂函数"""
        functions = []
        
        def function_factory(value):
            def captured_function():
                return value
            return captured_function
        
        for i in range(3):
            functions.append(function_factory(i))
        
        print("\n解决方案2结果:")
        for j, func in enumerate(functions):
            print(f"func{j}(): {func()}")
    
    common_trap()
    solution_1()
    solution_2()

                x = "shadowed"  # 遮蔽外部变量
                print(f"inner2 看到: {x}")

                nonlocal x  # 修改外部变量
                x = "modified"
                print(f"inner3 修改后: {x}")
            
            print(f"初始 x: {x}")
            inner2()
            print(f"inner2 调用后的 x: {x}")
            inner3()
            print(f"inner3 调用后的 x: {x}")
        
        shadowing_demo()

def practical_closure_applications():
    """闭包实际应用案例"""
    
    # ✅ 事件处理器
    def create_event_handler(event_type):
        """创建事件处理器"""
        handlers = {}
        
        def register_handler(callback):
            handlers[event_type] = handlers.get(event_type, []) + [callback]
            return len(handlers[event_type]) - 1
        
        def trigger_event(data):
            if event_type in handlers:
                for handler in handlers[event_type]:
                    handler(data)
        
        return register_handler, trigger_event
    
    # 测试事件处理器
    on_login, fire_login = create_event_handler('login')
    
    # 注册处理函数
    on_login(lambda data: print(f"用户 {data['user']} 登录"))
    on_login(lambda data: print(f"登录时间: {data['time']}"))
    fire_login({'user': 'Alice', 'time': '2024/01/01 10:00:00'})
    
    # ✅ 中间件链
    def create_middleware_chain():
        """创建中间件链"""
        middlewares = []
        
        def use(middleware):
            middlewares.append(middleware)
        
        def execute(data):
            processed_data = data
            for middleware in middlewares:
                processed_data = middleware(processed_data)
                if processed_data is None:
                    break
            return processed_data
        
        return use, execute
    
    # 测试中间件链
    add_middleware, execute_pipeline = create_middleware_chain()
    
    # 添加中间件
    add_middleware(lambda data: data.upper())
    add_middleware(lambda data: f"***{data}***")
    add_middleware(lambda data: data.replace("***", ""))
    
    result = execute_pipeline("hello world")
    print(f"中间件处理结果: {result}")
