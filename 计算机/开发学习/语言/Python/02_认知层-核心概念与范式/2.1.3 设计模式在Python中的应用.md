## 🎯 核心理念

设计模式是"编程智慧的精髓"。Python的动态特性和强大语法让设计模式的实现变得简洁优雅，体现了"简单胜过复杂"的哲学。

### 🏗️ 创建型模式

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Union, Dict, Any

class CreationalPatterns:
    """创建型设计模式"""
    
    @staticmethod
    def singleton_pattern():
        """单例模式 - Python实现"""
        
        # ✅ 装饰器方式
        def singleton(cls):
            instances = {}
            def getinstance(*args, **kwargs):
                if cls not in instances:
                    instances[cls] = cls(*args, **kwargs)
                return instances[cls]
            return getinstance
        
        @singleton
        class Logger:
            def __init__(self):
                self.logs = []
            
            def log(self, message):
                self.logs.append(message)
                print(f"LOG: {message}")
        
        # ✅ __new__ 方式
        class DatabaseConnection:
            _instance = None
            
            def __new__(cls):
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
                return cls._instance
            
            def __init__(self):
                if not hasattr(self, 'connection'):
                    self.connection = "Connected to DB"
        
        # 验证单例行为
        logger1 = Logger()
        logger2 = Logger()
        print(f"Logger singleton: {logger1 is logger2}")
        
        db1 = DatabaseConnection()
        db2 = DatabaseConnection()
        print(f"DB singleton: {db1 is db2}")
    
    @staticmethod
    def factory_pattern():
        """工厂模式 - Python实现"""
        
        # ✅ 抽象产品
        class Shape(ABC):
            @abstractmethod
            def draw(self):
                pass
        
        @abstractmethod
            def area(self):
                pass
        
        # ✅ 具体产品
        class Circle(Shape):
            def __init__(self, radius):
                self.radius = radius
            
            def draw(self):
                return f"Drawing circle with radius {self.radius}"
            
            def area(self):
                return 3.14159 * self.radius ** 2
        
        class Rectangle(Shape):
            def __init__(self, width, height):
                self.width = width
                self.height = height
            
            def draw(self):
                return f"Drawing rectangle {self.width}x{self.height}"
            
            def area(self):
                return self.width * self.height
        
        # ✅ 工厂类
        class ShapeFactory:
            @staticmethod
            def create_shape(shape_type: str, **kwargs):
                shapes = {
                    'circle': Circle,
                    'rectangle': Rectangle
                }
                
                shape_class = shapes.get(shape_type.lower())
                if shape_class:
                    return shape_class(**kwargs)
                else:
                    raise ValueError(f"Unknown shape type: {shape_type}")
        
        # ✅ 工厂注册模式
        class ShapeFactoryRegistry:
            """注册式工厂"""
            _shapes = {}
            
            @classmethod
            def register(cls, name: str):
                def decorator(shape_class):
                    cls._shapes[name] = shape_class
                    return shape_class
                return decorator
            
            @classmethod
            def create(cls, name: str, **kwargs):
                shape_class = cls._shapes.get(name)
                if shape_class:
                    return shape_class(**kwargs)
                raise ValueError(f"Unknown shape: {name}")
        
        # 注册新形状
        @ShapeFactoryRegistry.register('triangle')
        class Triangle(Shape):
            def __init__(self, base, height):
                self.base = base
                self.height = height
            
            def draw(self):
                return f"Drawing triangle {self.base}x{self.height}"
            
            def area(self):
                return 0.5 * self.base * self.height
        
        # 测试工厂模式
        circle = ShapeFactory.create_shape('circle', radius=5)
        rectangle = ShapeFactory.create_shape('rectangle', width=10, height=20)
        triangle = ShapeFactoryRegistry.create('triangle', base=10, height=15)
        
        print(circle.draw())
        print(rectangle.draw())
        print(triangle.draw())
    
    @staticmethod
    def builder_pattern():
        """建造者模式 - Python实现"""
        
        @dataclass
        class Computer:
            cpu: str = ""
            memory: str = ""
            storage: str = ""
            graphics: str = ""
            display: str = ""
        
        class ComputerBuilder:
            """计算机建造者"""
            def __init__(self):
                self.computer = Computer()
            
            def set_cpu(self, cpu: str):
                self.computer.cpu = cpu
                return self
            
            def set_memory(self, memory: str):
                self.computer.memory = memory
                return self
            
            def set_storage(self, storage: str):
                self.computer.storage = storage
                return self
            
            def set_graphics(self, graphics: str):
                self.computer.graphics = graphics
                return self
            
            def set_display(self, display: str):
                self.computer.display = display
                return self
            
            def build(self):
                return self.computer
        
        class GamingComputerBuilder(ComputerBuilder):
            """游戏电脑建造者"""
            def __init__(self):
                super().__init__()
                self.set_gaming_specs()
            
            def set_gaming_specs(self):
                self.set_cpu("Intel i7")
                self.set_memory("32GB DDR4")
                self.set_graphics("RTX 4090")
                return self
        
        # ✅ 指挥者模式
        class ComputerDirector:
            """计算机建造指挥者"""
            @staticmethod
            def build_gaming_computer():
                return (ComputerBuilder()
                        .set_cpu("Intel i7-12700K")
                        .set_memory("32GB DDR4")
                        .set_storage("1TB NVMe SSD")
                        .set_graphics("RTX 4090")
                        .set_display("27\" 4K 144Hz")
                        .build())
            
            @staticmethod
            def build_office_computer():
                return (ComputerBuilder()
                        .set_cpu("Intel i5-12400")
                        .set_memory("16GB DDR4")
                        .set_storage("512GB SSD")
                        .set_graphics("Integrated")
                        .set_display("24\" 1080p")
                        .build())
        
        # 测试建造者模式
        gaming_pc = ComputerDirector.build_gaming_computer()
        office_pc = ComputerDirector.build_office_computer()
        
        print(f"Gaming PC: {gaming_pc}")
        print(f"Office PC: {office_pc}")
```

```python
class StructuralPatterns:
    """结构型设计模式"""
    
    @staticmethod
    def adapter_pattern():
        """适配器模式 - Python实现"""
        
        # ✅ 被适配的类
        class LegacyJSON:
            def json_encode(self, data):
                import json
                return f"LEGACY:{json.dumps(data)}"
        
        # ✅ 目标接口
        class ModernJSON:
            def encode(self, data):
                return {"success": True, "data": data, "format": "modern"}
        
        # ✅ 适配器
        class JSONAdapter:
            def __init__(self, legacy_json):
                self.legacy_json = legacy_json
            
            def encode(self, data):
                # 适配现代接口到遗留系统
                legacy_result = self.legacy_json.json_encode(data)
                return {"success": True, "data": legacy_result, "format": "adapted"}
        
        # 测试适配器
        legacy = LegacyJSON()
        adapter = JSONAdapter(legacy)
        modern = ModernJSON()
        
        test_data = {"name": "Python", "version": "3.11"}
        
        print(f"Legacy result: {legacy.json_encode(test_data)}")
        print(f"Modern result: {modern.encode(test_data)}")
        print(f"Adapted result: {adapter.encode(test_data)}")
    
    @staticmethod
    def decorator_pattern():
        """装饰器模式 - Python实现"""
        
        # ✅ 基础组件
        class Coffee:
            def cost(self):
                return 5.0
            
            def description(self):
                return "Coffee"
        
        # ✅ 装饰器基类
        class CoffeeDecorator(Coffee):
            def __init__(self, coffee: Coffee):
                self.coffee = coffee
            
            def cost(self):
                return self.coffee.cost()
            
            def description(self):
                return self.coffee.description()
        
        # ✅ 具体装饰器
        class MilkDecorator(CoffeeDecorator):
            def cost(self):
                return self.coffee.cost() + 1.0
            
            def description(self):
                return f"{self.coffee.description()}, Milk"
        
        class SugarDecorator(CoffeeDecorator):
            def cost(self):
                return self.coffee.cost() + 0.5
            
            def description(self):
                return f"{self.coffee.description()}, Sugar"
        
        class VanillaDecorator(CoffeeDecorator):
            def cost(self):
                return self.coffee.cost() + 2.0
            
            def description(self):
                return f"{self.coffee.description()}, Vanilla"
        
        # 测试装饰器模式
        coffee = Coffee()
        print(f"Basic coffee: {coffee.description()}, Cost: ${coffee.cost()}")
        
        fancy_coffee = VanillaDecorator(
            MilkDecorator(
                SugarDecorator(coffee)
            )
        )
        
        print(f"Fancy coffee: {fancy_coffee.description()}, Cost: ${fancy_coffee.cost()}")
    
    @staticmethod
    def facade_pattern():
        """外观模式 - Python实现"""
        
        # ✅ 子系统类
        class CPU:
            def start(self):
                return "CPU started"
            
            def execute(self, program):
                return f"CPU executing: {program}"
        
        class Memory:
            def load(self, program):
                return f"Memory loaded: {program}"
            
            def allocate(self, size):
                return f"Memory allocated: {size}MB"
        
        class HardDrive:
            def read(self, sector):
                return f"HardDrive reading sector: {sector}"
        
        # ✅ 外观类
        class ComputerFacade:
            """计算机外观"""
            def __init__(self):
                self.cpu = CPU()
                self.memory = Memory()
                self.hard_drive = HardDrive()
            
            def start_computer(self, program="boot_system"):
                """启动计算机的简化接口"""
                results = []
                results.append(self.cpu.start())
                results.append(self.memory.allocate(1024))
                results.append(self.hard_drive.read(0))
                results.append(self.memory.load(program))
                results.append(self.cpu.execute(program))
                
                return "\n".join(results)
        
        # 测试外观模式
        computer = ComputerFacade()
        print("Starting computer:")
        print(computer.start_computer())
    
    @staticmethod
    def proxy_pattern():
        """代理模式 - Python实现"""
        
        # ✅ 主题接口
        class FileStorage(ABC):
            @abstractmethod
            def read(self, filename):
                pass
            
            @abstractmethod
            def write(self, filename, content):
                pass
        
        # ✅ 真实主题
        class ActualFileStorage(FileStorage):
            def read(self, filename):
                print(f"Reading file: {filename}")
                return f"Content of {filename}"
            
            def write(self, filename, content):
                print(f"Write {len(content)} characters to {filename}")
        
        # ✅ 代理
        class CachedFileStorage(FileStorage):
            """缓存代理"""
            def __init__(self):
                self.storage = ActualFileStorage()
                self.cache = {}
            
            def read(self, filename):
                if filename in self.cache:
                    print(f"Cache hit for {filename}")
                    return self.cache[filename]
                
                content = self.storage.read(filename)
                self.cache[filename] = content
                return content
            
            def write(self, filename, content):
                self.storage.write(filename, content)
                if filename in self.cache:
                    del self.cache[filename]
        
        # ✅ 虚拟代理
        class LazyFileStorage(FileStorage):
            """虚拟代理 - 延迟加载"""
            def __init__(self):
                self._storage = None
            
            @property
            def storage(self):
                if self._storage is None:
                    print("Creating actual storage instance...")
                    self._storage = ActualFileStorage()
                return self._storage
            
            def read(self, filename):
                return self.storage.read(filename)
            
            def write(self, filename, content):
                return self.storage.write(filename, content)
        
        # 测试代理模式
        cached = CachedFileStorage()
        print(cached.read("test.txt"))  # 从磁盘读取
        print(cached.read("test.txt"))  # 从缓存读取
        
        lazy = LazyFileStorage()
        print(lazy.read("lazy.txt"))   # 延迟创建实例
```

### 🎯 行为型模式

```python
class BehavioralPatterns:
    """行为型设计模式"""
    
    @staticmethod
    def observer_pattern():
        """观察者模式 - Python实现"""
        
        # ✅ 主题接口
        class Observable(ABC):
            @abstractmethod
            def subscribe(self, observer):
                pass
            
            @abstractmethod
            def unsubscribe(self, observer):
                pass
            
            @abstractmethod
            def notify(self, message):
                pass
        
        # ✅ 观察者接口
        class Observer(ABC):
            @abstractmethod
            def update(self, message):
                pass
        
        # ✅ 具体主题
        class NewsAgency(Observable):
            def __init__(self):
                self.observers = []
                self.news = []
            
            def subscribe(self, observer):
                self.observers.append(observer)
            
            def unsubscribe(self, observer):
                if observer in self.observers:
                    self.observers.remove(observer)
            
            def notify(self, news_item):
                for observer in self.observers:
                    observer.update(news_item)
                self.news.append(news_item)
            
            def publish_news(self, headline, content):
                news_item = f"{headline}: {content}"
                self.notify(news_item)
        
        # ✅ 具体观察者
        class NewspaperSubscriber(Observer):
            def __init__(self, name):
                self.name = name
                self.received_news = []
            
            def update(self, news_item):
                self.received_news.append(news_item)
                print(f"{self.name} received: {news_item}")
        
        class TVStationSubscriber(Observer):
            def __init__(self, station_name):
                self.station_name = station_name
                self.broadcasts = []
            
            def update(self, news_item):
                self.broadcasts.append(news_item)
                print(f"📺 {self.station_name} broadcasting: {news_item}")
        
        # 测试观察者模式
        agency = NewsAgency()
        
        newspaper = NewspaperSubscriber("Daily News")
        tv_station = TVStationSubscriber("CNN")
        
        agency.subscribe(newspaper)
        agency.subscribe(tv_station)
        
        agency.publish_news("Breaking", "Python 3.12 released!")
        agency.publish_news("Tech News", "New AI model breaks records")
        
        agency.unsubscribe(newspaper)
        agency.publish_news("Sports", "Olympics starts tomorrow")
    
    @staticmethod
    def strategy_pattern():
        """策略模式 - Python实现"""
        
        # ✅ 策略接口
        class PaymentStrategy(ABC):
            @abstractmethod
            def pay(self, amount):
                pass
        
        # ✅ 具体策略
        class CreditCardPayment(PaymentStrategy):
            def __init__(self, card_number):
                self.card_number = card_number
            
            def pay(self, amount):
                return f"Paid ${amount} using Credit Card ending in {self.card_number[-4:]}"
        
        class PayPalPayment(PaymentStrategy):
            def __init__(self, email):
                self.email = email
            
            def pay(self, amount):
                return f"Paid ${amount} using PayPal account {self.email}"
        
        class BitcoinPayment(PaymentStrategy):
            def __init__(self, wallet_address):
                self.wallet_address = wallet_address
            
            def pay(self, amount):
                return f"Paid ${amount} using Bitcoin wallet {self.wallet_address[:8]}..."
        
        # ✅ 上下文类
        class PaymentProcessor:
            def __init__(self):
                self.status = PaymentStrategy(None)
            
            def set_payment_strategy(self, strategy: PaymentStrategy):
                self.strategy = strategy
            
            def process_payment(self, amount):
                if not self.strategy:
                    raise ValueError("No payment strategy set")
                return self.strategy.pay(amount)
        
        # ✅ 工厂式策略选择
        def create_payment_strategy(payment_type, **kwargs):
            strategies = {
                'credit_card': CreditCardPayment,
                'paypal': PayPalPayment,
                'bitcoin': BitcoinPayment
            }
            
            strategy_class = strategies.get(payment_type)
            if strategy_class:
                return strategy_class(**kwargs)
            raise ValueError(f"Unknown payment type: {payment_type}")
        
        # 测试策略模式
        processor = PaymentProcessor()
        
        # 信用卡支付
        cc_strategy = create_payment_strategy('credit_card', card_number='1234567890123456')
        processor.set_payment_strategy(cc_strategy)
        print(processor.process_payment(100.0))
        
        # PayPal支付
        pp_strategy = create_payment_strategy('paypal', email='user@example.com')
        processor.set_payment_strategy(pp_strategy)
        print(processor.process_payment(50.0))
        
        # Bitcoin支付
        btc_strategy = create_payment_strategy('bitcoin', wallet_address='1BvBMSEYstWetqTFn5Au4m4GFg7xJaNVN2')
        processor.set_payment_strategy(btc_strategy)
        print(processor.process_payment(25.0))
    
    @staticmethod
    def command_pattern():
        """命令模式 - Python实现"""
        
        # ✅ 命令接口
        class Command(ABC):
            @abstractmethod
            def execute(self):
                pass
            
            @abstractmethod
            def undo(self):
                pass
        
        # ✅ 接收者
        class LightBulb:
            def __init__(self):
                self.is_on = False
            
            def turn_on(self):
                self.is_on = True
                return "Light is ON"
            
            def turn_off(self):
                self.is_on = False
                return "Light is OFF"
        
        class Fan:
            def __init__(self):
                self.speed = 0
            
            def increase_speed(self):
                self.speed = min(self.speed + 1, 3)
                return f"Fan speed: {self.speed}"
            
            def decrease_speed(self):
                self.speed = max(self.speed - 1, 0)
                return f"Fan speed: {self.speed}"
        
        # ✅ 具体命令
        class LightOnCommand(Command):
            def __init__(self, light):
                self.light = light
            
            def execute(self):
                return self.light.turn_on()
            
            def undo(self):
                return self.light.turn_off()
        
        class FanSpeedIncreaseCommand(Command):
            def __init__(self, fan):
                self.fan = fan
            
            def execute(self):
                return self.fan.increase_speed()
            
            def undo(self):
                return self.fan.decrease_speed()


# ✅ 调用者/遥控器
class RemoteControl:
    """智能遥控器"""
    def __init__(self):
        self.commands = {}
        self.current_command = None
    
    def set_command(self, slot, command: Command):
        self.commands[slot] = command
    
    def press_button(self, slot):
        if slot in self.commands:
            command = self.commands[slot]
            result = command.execute()
            self.current_command = command
            return result
    
    def press_undo(self):
        if self.current_command:
            return self.current_command.undo()

# 测试命令模式
light = LightBulb()
fan = Fan()

light_on_cmd = LightOnCommand(light)
fan_speed_cmd = FanSpeedIncreaseCommand(fan)

remote = RemoteControl()
remote.set_command(1, light_on_cmd)
remote.set_command(2, fan_speed_cmd)

print(remote.press_button(1))  # 开灯
print(remote.press_button(2))  # 增加风扇速度
print(remote.press_undo())      # 撤销风扇操作
print(remote.press_undo())      # 撤销开灯操作
```
