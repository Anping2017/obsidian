# 2.4 设计模式精要

## 🎯 核心理念

设计模式是软件开发的经验结晶，是解决常见问题的优雅方案。掌握设计模式能让代码更加灵活、可维护、可扩展，是实现高质量软件架构的基础。

## 🚀 设计模式体系

### 创建型模式

```python
from abc import ABC, abstractmethod
from typing import List, Dict, Any, Optional
import threading

class DesignPatternMastery:
    """设计模式精通技术"""
    
    @staticmethod
    def creational_patterns():
        """创建型模式"""
        
        # ✅ 单例模式
        class Singleton:
            """线程安全的单例模式"""
            _instance = None
            _lock = threading.Lock()
            
            def __new__(cls):
                if cls._instance is None:
                    with cls._lock:
                        if cls._instance is None:
                            cls._instance = super(Singleton, cls).__new__(cls)
                return cls._instance
            
            def __init__(self):
                if not hasattr(self, 'initialized'):
                    self.initialized = True
                    self._data = {}
            
            def get_data(self, key):
                return self._data.get(key)
            
            def set_data(self, key, value):
                self._data[key] = value
        
        # ✅ 工厂模式
        class ShapeFactory:
            """工厂模式"""
            
            @staticmethod
            def create_shape(shape_type: str):
                if shape_type == "circle":
                    return Circle()
                elif shape_type == "rectangle":
                    return Rectangle()
                elif shape_type == "triangle":
                    return Triangle()
                else:
                    raise ValueError(f"未知形状类型: {shape_type}")
        
        class Circle:
            def draw(self):
                return "绘制圆形"
        
        class Rectangle:
            def draw(self):
                return "绘制矩形"
        
        class Triangle:
            def draw(self):
                return "绘制三角形"
        
        # ✅ 抽象工厂模式
        class AbstractFactory(ABC):
            """抽象工厂"""
            
            @abstractmethod
            def create_button(self):
                pass
            
            @abstractmethod
            def create_checkbox(self):
                pass
        
        class WindowsFactory(AbstractFactory):
            """Windows工厂"""
            
            def create_button(self):
                return WindowsButton()
            
            def create_checkbox(self):
                return WindowsCheckbox()
        
        class MacFactory(AbstractFactory):
            """Mac工厂"""
            
            def create_button(self):
                return MacButton()
            
            def create_checkbox(self):
                return MacCheckbox()
        
        class WindowsButton:
            def render(self):
                return "Windows按钮"
        
        class MacButton:
            def render(self):
                return "Mac按钮"
        
        class WindowsCheckbox:
            def render(self):
                return "Windows复选框"
        
        class MacCheckbox:
            def render(self):
                return "Mac复选框"
        
        # 测试创建型模式
        print("=== 创建型模式测试 ===")
        
        # 单例测试
        s1 = Singleton()
        s2 = Singleton()
        print(f"单例模式: {s1 is s2}")
        
        # 工厂模式测试
        circle = ShapeFactory.create_shape("circle")
        print(f"工厂模式: {circle.draw()}")
        
        # 抽象工厂测试
        win_factory = WindowsFactory()
        win_button = win_factory.create_button()
        print(f"抽象工厂: {win_button.render()}")
        
        return Singleton, ShapeFactory, AbstractFactory
    
    @staticmethod
    def structural_patterns():
        """结构型模式"""
        
        # ✅ 适配器模式
        from abc import ABC, abstractmethod
        
        class Target(ABC):
            """目标接口"""
            
            @abstractmethod
            def request(self):
                pass
        
        class Adaptee:
            """被适配类"""
            
            def specific_request(self):
                return "适配器特定请求"
        
        class Adapter(Target):
            """适配器"""
            
            def __init__(self, adaptee: Adaptee):
                self.adaptee = adaptee
            
            def request(self):
                return f"适配器: {self.adaptee.specific_request()}"
        
        # ✅ 装饰器模式
        from abc import ABC, abstractmethod
        
        class Component(ABC):
            """组件抽象类"""
            
            @abstractmethod
            def operation(self):
                pass
        
        class ConcreteComponent(Component):
            """具体组件"""
            
            def operation(self):
                return "具体组件操作"
        
        class Decorator(Component):
            """装饰器基类"""
            
            def __init__(self, component: Component):
                self._component = component
            
            def operation(self):
                return self._component.operation()
        
        class ConcreteDecoratorA(Decorator):
            """具体装饰器A"""
            
            def operation(self):
                return f"装饰器A({self._component.operation()})"
        
        class ConcreteDecoratorB(Decorator):
            """具体装饰器B"""
            
            def operation(self):
                return f"装饰器B({self._component.operation()})"
        
        # ✅ 外观模式
        class SubSystemA:
            def operation_a(self):
                return "子系统A操作"
        
        class SubSystemB:
            def operation_b(self):
                return "子系统B操作"
        
        class Facade:
            """外观类"""
            
            def __init__(self):
                self.subsystem_a = SubSystemA()
                self.subsystem_b = SubSystemB()
            
            def operation(self):
                result_a = self.subsystem_a.operation_a()
                result_b = self.subsystem_b.operation_b()
                return f"外观: {result_a} + {result_b}"
        
        # 测试结构型模式
        print("\n=== 结构型模式测试 ===")
        
        # 装饰器适配器测试
        adaptee = Adaptee()
        adapter = Adapter(adaptee)
        print(f"适配器模式: {adapter.request()}")
        
        # 装饰器测试
        component = ConcreteComponent()
        decorated = ConcreteDecoratorA(ConcreteDecoratorB(component))
        print(f"装饰器模式: {decorated.operation()}")
        
        # 外观测试
        facade = Facade()
        print(f"外观模式: {facade.operation()}")
        
        return Adapter, Decorator, Facade
    
    @staticmethod
    def behavioral_patterns():
        """行为型模式"""
        
        # ✅ 观察者模式
        from abc import ABC, abstractmethod
        
        class Observer(ABC):
            """观察者接口"""
            
            @abstractmethod
            def update(self, subject):
                pass
        
        class Subject:
            """主题类"""
            
            def __init__(self):
                self._observers = []
                self._state = None
            
            def attach(self, observer: Observer):
                self._observers.append(observer)
            
            def detach(self, observer: Observer):
                self._observers.remove(observer)
            
            def notify(self):
                for observer in self._observers:
                    observer.update(self)
            
            @property
            def get_state(self):
                return self._state
            
            def set_state(self, state):
                self._state = state
                self.notify()
        
        class ConcreteObserver(Observer):
            """具体观察者"""
            
            def __init__(self, name):
                self.name = name
            
            def update(self, subject):
                print(f"{self.name} 收到更新: {subject.get_state}")
        
        # ✅ 策略模式
        from abc import ABC, abstractmethod
        
        class Strategy(ABC):
            """策略接口"""
            
            @abstractmethod
            def execute(self, data):
                pass
        
        class QuickSort(Strategy):
            """快速排序策略"""
            
            def execute(self, data):
                return sorted(data)
        
        class BubbleSort(Strategy):
            """冒泡排序策略"""
            
            def execute(self, data):
                return sorted(data, reverse=True)[::-1]
        
        class Context:
            """上下文类"""
            
            def __init__(self, strategy: Strategy):
                self._strategy = strategy
            
            def set_strategy(self, strategy: Strategy):
                self._strategy = strategy
            
            def sort_data(self, data):
                return self._strategy.execute(data)
        
        # ✅ 命令模式
        from abc import ABC, abstractmethod
        
        class Command(ABC):
            """命令接口"""
            
            @abstractmethod
            def execute(self):
                pass
        
        class Receiver:
            """接收者"""
            
            def action(self, message):
                return f"执行操作: {message}"
        
        class ConcreteCommand(Command):
            """具体命令"""
            
            def __init__(self, receiver: Receiver, message: str):
                self._receiver = receiver
                self._message = message
            
            def execute(self):
                return self._receiver.action(self._message)
        
        class Invoker:
            """调用者"""
            
            def __init__(self):
                self._command = None
            
            def set_command(self, command: Command):
                self._command = command
            
            def execute_command(self):
                if self._command:
                    return self._command.execute()
        
        # ✅ 责任链模式
        from abc import ABC, abstractmethod
        
        class Handler(ABC):
            """处理器抽象类"""
            
            def __init__(self):
                self._next_handler = None
            
            def set_next(self, handler):
                self._next_handler = handler
            
            @abstractmethod
            def handle(self, request):
                pass
        
        @abstractmethod
        class ConcreteHandler1(Handler):
            """具体处理器1"""
            
            def handle(self, request):
                if request <= 10:
                    return f"处理器1处理请求: {request}"
                elif self._next_handler:
                    return self._next_handler.handle(request)
                else:
                    return f"无法处理请求: {request}"
        
        class ConcreteHandler2(Handler):
            """具体处理器2"""
            
            def handle(self, request):
                if request <= 100:
                    return f"处理器2处理请求: {request}"
                elif self._next_handler:
                    return self._next_handler.handle(request)
                else:
                    return f"无法处理请求: {request}"
        
        # 测试行为型模式
        print("\n=== 行为型模式测试 ===")
        
        # 观察者测试
        subject = Subject()
        observer1 = ConcreteObserver("观察者1")
        observer2 = ConcreteObserver("观察者2")
        
        subject.attach(observer1)
        subject.attach(observer2)
        subject.set_state("状态更新")
        
        # 策略测试
        data = [3, 1, 4, 1, 5]
        context = Context(QuickSort())
        print(f"策略模式-快速快速排序: {context.sort_data(data)}")
        
        context.set_strategy(BubbleSort())
        print(f"策略模式-冒泡排序: {context.sort_data(data)}")
        
        # 命令测试
        receiver = Receiver()
        command = ConcreteCommand(receiver, "保存文件")
        invoker = Invoker()
        
        invoker.set_command(command)
        print(f"命令模式: {invoker.execute_command()}")
        
        # 责任链测试
        handler1 = ConcreteHandler1()
        handler2 = ConcreteHandler2()
        
        handler1.set_next(handler2)
        
        print(f"责任链模式(5): {handler1.handle(5)}")
        print(f"责任链模式(50): {handler1.handle(50)}")
        print(f"责任链模式(500): {handler1.handle(500)}")
        
        return Subject, Strategy, Command, Handler

class AdvancedPatterns:
    """高级设计模式"""
    
    @staticmethod
    def dependency_injection():
        """依赖注入模式"""
        
        from abc import ABC, abstractmethod
        
        class DatabaseInterface(ABC):
            """数据库接口"""
            
            @abstractmethod
            def save(self, data):
                pass
        
        class MySQLDatabase(DatabaseInterface):
            """MySQL数据库实现"""
            
            def save(self, data):
                return f"MySQL保存: {data}"
        
        class PostgreSQLDatabase(DatabaseInterface):
            """PostgreSQL数据库实现"""
            
            def save(self, data):
                return f"PostgreSQL保存: {data}"
        
        @staticmethod
        def create_user_service(db_type):
            """工厂函数"""
            if db_type == "mysql":
                db = MySQLDatabase()
            elif db_type == "postgresql":
                db = PostgreSQLDatabase()
            else:
                raise ValueError("不支持的数据库类型")
            
            class UserService:
                """用户服务"""
                
                def __init__(self, database: DatabaseInterface):
                    self._database = database
                
                def create_user(self, user_data):
                    return self._database.save(user_data)
            
            return UserService(db)
        
        def test_dependency_injection():
            """测试依赖注入"""
            
            # MySQL依赖注入
            user_service_mysql = create_user_service("mysql")
            result1 = user_service_mysql.create_user({"name": "张三", "age": 25})
            
            # PostgreSQL依赖注入
            user_service_postgres = create_user_service("postgresql")
            result2 = user_service_postgres.create_user({"name": "李四", "age": 30})
            
            print(f"依赖注入-MySQL: {result1}")
            print(f"依赖注入-PostgreSQL: {result2}")
        
        test_dependency_injection()
    
    @staticmethod
    def builder_pattern():
        """建造者模式"""
        
        class Product:
            """产品类"""
            
            def __init__(self):
                self.parts = []
            
            def add_part(self, part):
                self.parts.append(part)
            
            def __str__(self):
                return f"产品部件: {', '.join(self.parts)}"
        
        from abc import ABC, abstractmethod
        
        @abstractmethod
        class Builder:
            """建造者抽象类"""
            
            def __init__(self):
                self.product = Product()
            
            @abstractmethod
            def build_part_a(self):
                pass
            
            @abstractmethod
            def build_part_b(self):
                pass
            
            @abstractmethod
            def get_product(self):
                pass
        
        @abstractmethod
        class ConcreteBuilder(Builder):
            """具体建造者"""
            
            def build_part_a(self):
                self.product.add_part("部件A")
            
            def build_part_b(self):
                self.product.add_part("部件B")
            
            def get_product(self):
                return self.product
        
        class Director:
            """指挥者"""
            
            def __init__(self, builder: Builder):
                self._builder = builder
            
            def construct(self):
                self._builder.build_part_a()
                self._builder.build_part_b()
                return self._builder.get_product()
        
        def test_builder():
            """测试建造者模式"""
            
            builder = ConcreteBuilder()
            director = Director(builder)
            product = director.construct()
            
            print(f"建造者模式: {product}")
        
        test_builder()

# ✅ 工厂+单例+观察者组合
import threading

class NotificationSystem:
    """通知系统"""
    
    _instance = None
    _lock = threading.Lock()
    
    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
                    cls._instance._observers = []
        return cls._instance
    
    def attach(self, observer):
        """添加观察者"""
        self._observers.append(observer)
    
    def notify(self, message):
        """通知所有观察者"""
        for observer in self._observers:
            observer.update(message)

class EmailNotifier:
    def update(self, message):
        print(f"邮件通知: {message}")

class SMSNotifier:
    def update(self, message):
        print(f"短信通知: {message}")

class NotificationFactory:
    """通知工厂"""
    
    @staticmethod
    def create_notifier(notifier_type):
        if notifier_type == "email":
            return EmailNotifier()
        elif notifier_type == "sms":
            return SMSNotifier()
        else:
            raise ValueError("不支持的观察者类型")

# ✅ 状态机+策略模式
@abstractmethod
class State:
    """状态接口"""
    
    @abstractmethod
    def handle(self, context, event):
        pass

class IdleState(State):
    """空闲状态"""
    
    def handle(self, context, event):
        if event == "start":
            return WorkingState()
        return self

class WorkingState(State):
    """工作状态"""
    
    def handle(self, context, event):
        if event == "stop":
            return IdleState()
        elif event == "pause":
            return PausedState()
        return self

class PausedState(State):
    """暂停状态"""
    
    def handle(self, context, event):
        if event == "resume":
            return WorkingState()
        elif event == "stop":
            return IdleState()
        return self

class StateMachine:
    """状态机"""
    
    def __init__(self, initial_state):
        self.current_state = initial_state
    
    def transition(self, event):
        """状态转换"""
        self.current_state = self.current_state.handle(self, event)
```
