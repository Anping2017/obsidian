## 🎯 核心理念

装饰器是Python的"语法糖衣魔法"。它不仅仅是一种设计模式，更是Python语言的优雅特性，体现了"代码既优雅又强大"的哲学。

### 🧠 装饰器基础与原理

```python
def basic_decorator_concept():
    """装饰器基础概念"""
    
    # ✅ 最简单的装饰器
    def simple_decorator(func):
        def wrapper(*args, **kwargs):
            print(f"Before calling {func.__name__}")
            result = func(*args, **kwargs)
            print(f"After calling {func.__name__}")
            return result
        return wrapper
    
    @simple_decorator
    def greet(name):
        return f"Hello, {name}!"
    
    # 等价于: greet = simple_decorator(greet)
    print(greet("Python"))

# ✅ 带参数的装饰器
class AdvancedDecoratorPatterns:
    """高级装饰器模式"""
    
    @staticmethod
    def parameterized_decorator():
        """参数化装饰器"""
        
        def repeat(times):
            def decorator(func):
                def wrapper(*args, **kwargs):
                    results = []
                    for i in range(times):
                        result = func(*args, **kwargs)
                        results.append(result)
                    return results
                return wrapper
            return decorator
        
        @repeat(3)
        def print_message(msg):
            print(f"Message: {msg}")
            return f"Processed: {msg}"
        
        results = print_message("Hello World")
        print(f"Results: {results}")
    
    @staticmethod
    def class_based_decorator():
        """基于类的装饰器"""
        
        @dataclass
        class CacheDecorator:
            """缓存装饰器类"""
            maxsize: int = 128
            
            def __init__(self, func):
                self.func = func
                self.cache = {}
                functools.update_wrapper(self, func)
            
            def __call__(self, *args, **kwargs):
                # 创建缓存键
                cache_key = str(args) + str(sorted(kwargs.items()))
                
                if cache_key in self.cache:
                    print(f"Cache hit for {self.func.__name__}")
                    return self.cache[cache_key]
                
                print(f"Cache miss for {self.func.__name__}")
                result = self.func(*args, **kwargs)
                
                # 简单的LRU缓存
                if len(self.cache) >= self.maxsize:
                    oldest_key = next(iter(self.cache))
                    del self.cache[oldest_key]
                
                self.cache[cache_key] = result
                return result
        
        @CacheDecorator
        def expensive_calculation(n):
            """模拟耗时计算"""
            import time
            time.sleep(0.1)
            return n * n
        
        # 测试缓存效果
        print(expensive_calculation(5))  # 缓存未命中
        print(expensive_calculation(5))  # 缓存命中
        print(expensive_calculation(10))  # 缓存未命中
        print(expensive_calculation(5))  # 缓存命中
    
    @staticmethod
    def property_decorator():
        """属性装饰器"""
        
        class Temperature:
            def __init__(self):
                self._celsius = 0
            
            @property
            def celsius(self):
                """获取摄氏度"""
                return self._celsius
            
            @celsius.setter
            def celsius(self, value):
                """设置摄氏度"""
                if value < -273.15:
                    raise ValueError("Temperature cannot be below absolute zero")
                self._celsius = value
            
            @property
            def fahrenheit(self):
                """华氏度属性"""
                return self._celsius * 9/5 + 32
            
            @fahrenheit.setter
            def fahrenheit(self, value):
                """设置华氏度"""
                self.celsius = (value - 32) * 5/9
        
        # 测试属性装饰器
        temp = Temperature()
        temp.celsius = 25
        print(f"Celsius: {temp.celsius}, Fahrenheit: {temp.fahrenheit:.2f}")
        
        temp.fahrenheit = 100
        print(f"Celsius: {temp.celsius:.2f}, Fahrenheit: {temp.fahrenheit:.2f}")
    
    @staticmethod
    def method_decorator():
        """方法装饰器"""
        
        class AuthRequired:
            """需要认证的装饰器"""
            @staticmethod
            def require_auth(func):
                def wrapper(self, *args, **kwargs):
                    if not hasattr(self, 'is_authenticated') or not self.is_authenticated:
                        raise PermissionError("Authentication required")
                    return func(self, *args, **kwargs)
                return wrapper
        
        class UserService:
            def __init__(self):
                self.is_authenticated = False
            
            def login(self, password):
                if password == "secret123":
                    self.is_authenticated = True
                    return True
                return False
            
            @AuthRequired.require_auth
            def get_profile(self):
                return {"username": "user", "email": "user@example.com"}
            
            @AuthRequired.require_auth
            def update_profile(self, profile_data):
                return f"Profile updated with: {profile_data}"
        
        # 测试方法装饰器
        service = UserService()
        
        # 未登录时访问受保护的方法
        try:
            service.get_profile()
        except PermissionError as e:
            print(f"Permission denied: {e}")
        
        # 登录后访问
        service.login("secret123")
        print(service.get_profile())
        print(service.update_profile({"nickname": "PythonDev"}))

# ✅ 装饰器工厂模式
class DecoratorFactory:
    """装饰器工厂"""
    
    @staticmethod
    def timing_decorator(unit="ms"):
        """计时装饰器工厂"""
        def decorator(func):
            def wrapper(*args, **kwargs):
                start = time.perf_counter()
                result = func(*args, **kwargs)
                end = time.perf_counter()
                
                if unit == "ms":
                    duration = (end - start) * 1000
                    print(f"{func.__name__} took {duration:.2f} ms")
                elif unit == "s":
                    duration = end - start
                    print(f"{func.__name__} took {duration:.4f} seconds")
                
                return result
            return wrapper
        return decorator
    
    @staticmethod
    def retry_decorator(max_attempts=3, delay=1):
        """重试装饰器工厂"""
        def decorator(func):
            def wrapper(*args, **kwargs):
                for attempt in range(max_attempts):
                    try:
                        return func(*args, **kwargs)
                    except except as e:
                        if attempt == max_attempts - 1:
                            raise e
                        print(f"Attempt {attempt + 1} failed: {e}")
                        time.sleep(delay)
            return wrapper
        return decorator
    
    @staticmethod
    def validation_decorator(validator_func):
        """验证装饰器工厂"""
        def decorator(func):
            def wrapper(*args, **kwargs):
                # 验证参数
                if not validator_func(args, kwargs):
                    raise ValueError("Validation failed")
                return func(*args, **kwargs)
            return wrapper
        return decorator


# ✅ 现代Python装饰器使用
@DecoratorFactory.timing_decorator("ms")
def fast_calculation():
    return sum(range(100000))

@DecoratorFactory.retry_decorator(max_attempts=3, delay=1)
def unreliable_function():
    import random
    if random.random() < 0.7:  # 70% 失败率
        raise ConnectionError("Network error")
    return "Success!"

def validate_positive_numbers(args, kwargs):
    """验证所有数字参数为正数"""
    for arg in args:
        if isinstance(arg, (int, float)) and arg <= 0:
            return False
    return True

@DecoratorFactory.validation_decorator(validate_positive_numbers)
def calculate_area(length, width):
    return length * width

# 测试装饰器工厂
fast_calculation()
calculate_area(5, 10)

try:
    calculate_area(-1, 10)
except ValueError as e:
    print(f"Validation error: {e}")
