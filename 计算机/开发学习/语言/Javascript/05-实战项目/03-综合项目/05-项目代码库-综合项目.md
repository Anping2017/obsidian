```javascript
// src/components/ProductList.vue
<template>
  <div class="product-list">
    <div class="filters">
      <el-select v-model="filters.category" placeholder="选择分类" @change="handleFilterChange">
        <el-option
          v-for="category in categories"
          :key="category.id"
          :label="category.name"
          :value="category.id"
        />
      </el-select>
      
      <el-select v-model="filters.brand" placeholder="选择品牌" @change="handleFilterChange">
        <el-option
          v-for="brand in brands"
          :key="brand.id"
          :label="brand.name"
          :value="brand.id"
        />
      </el-select>
      
      <el-slider
        v-model="filters.priceRange"
        range
        :min="0"
        :max="10000"
        @change="handleFilterChange"
      />
    </div>
    
    <div class="product-grid">
      <ProductCard
        v-for="product in products"
        :key="product.id"
        :product="product"
        @add-to-cart="handleAddToCart"
        @view-detail="handleViewDetail"
      />
    </div>
    
    <el-pagination
      v-model:current-page="currentPage"
      :page-size="pageSize"
      :total="total"
      @current-change="handlePageChange"
    />
  </div>
</template>

<script>
import { ref, reactive, onMounted, watch } from 'vue'
import { useStore } from 'vuex'
import ProductCard from './ProductCard.vue'
import ProductService from '../services/ProductService.js'

export default {
  name: 'ProductList',
  components: {
    ProductCard
  },
  setup() {
    const store = useStore()
    const productService = new ProductService()
    
    const products = ref([])
    const categories = ref([])
    const brands = ref([])
    const currentPage = ref(1)
    const pageSize = ref(20)
    const total = ref(0)
    const loading = ref(false)
    
    const filters = reactive({
      category: '',
      brand: '',
      priceRange: [0, 10000],
      sort: 'default'
    })
    
    // 加载产品列表
    const loadProducts = async () => {
      loading.value = true
      try {
        const params = {
          page: currentPage.value,
          limit: pageSize.value,
          ...filters
        }
        
        const response = await productService.getProducts(params)
        products.value = response.products
        total.value = response.total
      } catch (error) {
        console.error('加载产品失败:', error)
        ElMessage.error('加载产品失败')
      } finally {
        loading.value = false
      }
    }
    
    // 加载分类和品牌
    const loadFilters = async () => {
      try {
        const [categoriesRes, brandsRes] = await Promise.all([
          productService.getCategories(),
          productService.getBrands()
        ])
        
        categories.value = categoriesRes
        brands.value = brandsRes
      } catch (error) {
        console.error('加载筛选条件失败:', error)
      }
    }
    
    // 处理筛选变化
    const handleFilterChange = () => {
      currentPage.value = 1
      loadProducts()
    }
    
    // 处理分页变化
    const handlePageChange = (page) => {
      currentPage.value = page
      loadProducts()
    }
    
    // 处理添加到购物车
    const handleAddToCart = async (product) => {
      try {
        await store.dispatch('cart/addItem', {
          productId: product.id,
          quantity: 1
        })
        ElMessage.success('已添加到购物车')
      } catch (error) {
        console.error('添加到购物车失败:', error)
        ElMessage.error('添加到购物车失败')
      }
    }
    
    // 处理查看详情
    const handleViewDetail = (product) => {
      router.push(`/products/${product.id}`)
    }
    
    // 监听筛选条件变化
    watch(filters, () => {
      handleFilterChange()
    }, { deep: true })
    
    onMounted(() => {
      loadProducts()
      loadFilters()
    })
    
    return {
      products,
      categories,
      brands,
      currentPage,
      pageSize,
      total,
      loading,
      filters,
      handleFilterChange,
      handlePageChange,
      handleAddToCart,
      handleViewDetail
    }
  }
}
</script>
```

### 后端API实现
```javascript
// backend/src/controllers/ProductController.js
const Product = require('../models/Product')
const Category = require('../models/Category')
const Brand = require('../models/Brand')

class ProductController {
  // 获取产品列表
  async getProducts(req, res) {
    try {
      const {
        page = 1,
        limit = 20,
        category,
        brand,
        priceRange,
        sort = 'default',
        keyword
      } = req.query
      
      // 构建查询条件
      const query = {}
      
      if (category) {
        query.category = category
      }
      
      if (brand) {
        query.brand = brand
      }
      
      if (priceRange) {
        const [minPrice, maxPrice] = priceRange.split(',')
        query.price = {
          $gte: parseFloat(minPrice),
          $lte: parseFloat(maxPrice)
        }
      }
      
      if (keyword) {
        query.$or = [
          { name: { $regex: keyword, $options: 'i' } },
          { description: { $regex: keyword, $options: 'i' } }
        ]
      }
      
      // 构建排序条件
      let sortOption = {}
      switch (sort) {
        case 'price_asc':
          sortOption = { price: 1 }
          break
        case 'price_desc':
          sortOption = { price: -1 }
          break
        case 'sales_desc':
          sortOption = { sales: -1 }
          break
        case 'rating_desc':
          sortOption = { rating: -1 }
          break
        case 'created_at_desc':
          sortOption = { createdAt: -1 }
          break
        default:
          sortOption = { createdAt: -1 }
      }
      
      // 执行查询
      const skip = (page - 1) * limit
      const [products, total] = await Promise.all([
        Product.find(query)
          .populate('category', 'name')
          .populate('brand', 'name')
          .sort(sortOption)
          .skip(skip)
          .limit(parseInt(limit)),
        Product.countDocuments(query)
      ])
      
      res.json({
        success: true,
        data: {
          products,
          total,
          page: parseInt(page),
          limit: parseInt(limit),
          pages: Math.ceil(total / limit)
        }
      })
    } catch (error) {
      console.error('获取产品列表失败:', error)
      res.status(500).json({
        success: false,
        message: '获取产品列表失败'
      })
    }
  }
  
  // 获取产品详情
  async getProduct(req, res) {
    try {
      const { id } = req.params
      
      const product = await Product.findById(id)
        .populate('category', 'name')
        .populate('brand', 'name')
        .populate('reviews.user', 'name avatar')
      
      if (!product) {
        return res.status(404).json({
          success: false,
          message: '产品不存在'
        })
      }
      
      // 增加浏览量
      product.views += 1
      await product.save()
      
      res.json({
        success: true,
        data: product
      })
    } catch (error) {
      console.error('获取产品详情失败:', error)
      res.status(500).json({
        success: false,
        message: '获取产品详情失败'
      })
    }
  }
  
  // 创建产品
  async createProduct(req, res) {
    try {
      const productData = req.body
      
      const product = new Product(productData)
      await product.save()
      
      res.status(201).json({
        success: true,
        data: product
      })
    } catch (error) {
      console.error('创建产品失败:', error)
      res.status(500).json({
        success: false,
        message: '创建产品失败'
      })
    }
  }
  
  // 更新产品
  async updateProduct(req, res) {
    try {
      const { id } = req.params
      const updateData = req.body
      
      const product = await Product.findByIdAndUpdate(
        id,
        updateData,
        { new: true, runValidators: true }
      )
      
      if (!product) {
        return res.status(404).json({
          success: false,
          message: '产品不存在'
        })
      }
      
      res.json({
        success: true,
        data: product
      })
    } catch (error) {
      console.error('更新产品失败:', error)
      res.status(500).json({
        success: false,
        message: '更新产品失败'
      })
    }
  }
  
  // 删除产品
  async deleteProduct(req, res) {
    try {
      const { id } = req.params
      
      const product = await Product.findByIdAndDelete(id)
      
      if (!product) {
        return res.status(404).json({
          success: false,
          message: '产品不存在'
        })
      }
      
      res.json({
        success: true,
        message: '产品删除成功'
      })
    } catch (error) {
      console.error('删除产品失败:', error)
      res.status(500).json({
        success: false,
        message: '删除产品失败'
      })
    }
  }
  
  // 获取分类列表
  async getCategories(req, res) {
    try {
      const categories = await Category.find().sort({ name: 1 })
      
      res.json({
        success: true,
        data: categories
      })
    } catch (error) {
      console.error('获取分类列表失败:', error)
      res.status(500).json({
        success: false,
        message: '获取分类列表失败'
      })
    }
  }
  
  // 获取品牌列表
  async getBrands(req, res) {
    try {
      const brands = await Brand.find().sort({ name: 1 })
      
      res.json({
        success: true,
        data: brands
      })
    } catch (error) {
      console.error('获取品牌列表失败:', error)
      res.status(500).json({
        success: false,
        message: '获取品牌列表失败'
      })
    }
  }
}

module.exports = new ProductController()
```

### 前端编辑器实现
```javascript
// src/components/ArticleEditor.vue
<template>
  <div class="article-editor">
    <div class="editor-header">
      <div class="editor-actions">
        <el-button @click="saveDraft" :loading="saving">保存草稿</el-button>
        <el-button type="primary" @click="publish" :loading="publishing">发布</el-button>
        <el-button @click="preview">预览</el-button>
      </div>
      <div class="editor-status">
        <span class="status-text">{{ status }}</span>
        <span class="word-count">{{ wordCount }} 字</span>
      </div>
    </div>
    
    <div class="editor-form">
      <el-form :model="articleForm" label-width="80px">
        <el-form-item label="文章标题">
          <el-input
            v-model="articleForm.title"
            placeholder="输入文章标题"
            @input="handleContentChange"
          />
        </el-form-item>
        
        <el-form-item label="文章摘要">
          <el-input
            v-model="articleForm.excerpt"
            type="textarea"
            :rows="3"
            placeholder="输入文章摘要（可选）"
            @input="handleContentChange"
          />
        </el-form-item>
        
        <el-form-item label="分类">
          <el-select
            v-model="articleForm.categories"
            multiple
            placeholder="选择分类"
            @change="handleContentChange"
          >
            <el-option
              v-for="category in categories"
              :key="category.id"
              :label="category.name"
              :value="category.id"
            />
          </el-select>
        </el-form-item>
        
        <el-form-item label="标签">
          <el-input
            v-model="articleForm.tags"
            placeholder="输入标签，用逗号分隔"
            @input="handleContentChange"
          />
        </el-form-item>
        
        <el-form-item label="特色图片">
          <el-upload
            class="image-uploader"
            action="/api/upload/image"
            :show-file-list="false"
            :on-success="handleImageSuccess"
            :before-upload="beforeImageUpload"
          >
            <img v-if="articleForm.featuredImage" :src="articleForm.featuredImage" class="uploaded-image" />
            <el-icon v-else class="image-uploader-icon"><Plus /></el-icon>
          </el-upload>
        </el-form-item>
      </el-form>
    </div>
    
    <div class="editor-content">
      <div ref="editor" class="quill-editor"></div>
    </div>
  </div>
</template>

<script>
import { ref, reactive, onMounted, onUnmounted, nextTick } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { ElMessage } from 'element-plus'
import { Plus } from '@element-plus/icons-vue'
import Quill from 'quill'
import 'quill/dist/quill.snow.css'
import ArticleService from '../services/ArticleService.js'
import CategoryService from '../services/CategoryService.js'

export default {
  name: 'ArticleEditor',
  components: {
    Plus
  },
  setup() {
    const route = useRoute()
    const router = useRouter()
    const articleService = new ArticleService()
    const categoryService = new CategoryService()
    
    const editor = ref(null)
    const quillEditor = ref(null)
    const saving = ref(false)
    const publishing = ref(false)
    const status = ref('已保存')
    const wordCount = ref(0)
    const categories = ref([])
    
    const articleForm = reactive({
      title: '',
      excerpt: '',
      content: '',
      categories: [],
      tags: '',
      featuredImage: ''
    })
    
    // 初始化编辑器
    const initEditor = () => {
      quillEditor.value = new Quill(editor.value, {
        theme: 'snow',
        modules: {
          toolbar: [
            [{ 'header': [1, 2, 3, 4, 5, 6, false] }],
            ['bold', 'italic', 'underline', 'strike'],
            [{ 'color': [] }, { 'background': [] }],
            [{ 'list': 'ordered'}, { 'list': 'bullet' }],
            [{ 'indent': '-1'}, { 'indent': '+1' }],
            ['blockquote', 'code-block'],
            ['link', 'image', 'video'],
            ['clean']
          ]
        },
        placeholder: '开始写作...'
      })
      
      // 监听内容变化
      quillEditor.value.on('text-change', () => {
        articleForm.content = quillEditor.value.getContents()
        updateWordCount()
        handleContentChange()
      })
    }
    
    // 更新字数统计
    const updateWordCount = () => {
      const text = quillEditor.value.getText()
      wordCount.value = text.length
    }
    
    // 处理内容变化
    const handleContentChange = () => {
      status.value = '未保存'
    }
    
    // 保存草稿
    const saveDraft = async () => {
      saving.value = true
      try {
        const articleData = {
          ...articleForm,
          tags: articleForm.tags.split(',').map(tag => tag.trim()).filter(tag => tag),
          status: 'draft'
        }
        
        if (route.params.id) {
          await articleService.updateArticle(route.params.id, articleData)
        } else {
          await articleService.createArticle(articleData)
        }
        
        status.value = '已保存'
        ElMessage.success('草稿保存成功')
      } catch (error) {
        console.error('保存草稿失败:', error)
        ElMessage.error('保存草稿失败')
      } finally {
        saving.value = false
      }
    }
    
    // 发布文章
    const publish = async () => {
      publishing.value = true
      try {
        const articleData = {
          ...articleForm,
          tags: articleForm.tags.split(',').map(tag => tag.trim()).filter(tag => tag),
          status: 'published'
        }
        
        if (route.params.id) {
          await articleService.updateArticle(route.params.id, articleData)
        } else {
          await articleService.createArticle(articleData)
        }
        
        status.value = '已发布'
        ElMessage.success('文章发布成功')
        router.push('/articles')
      } catch (error) {
        console.error('发布文章失败:', error)
        ElMessage.error('发布文章失败')
      } finally {
        publishing.value = false
      }
    }
    
    // 预览文章
    const preview = () => {
      const previewData = {
        title: articleForm.title,
        content: quillEditor.value.root.innerHTML,
        excerpt: articleForm.excerpt
      }
      
      sessionStorage.setItem('previewData', JSON.stringify(previewData))
      window.open('/preview', '_blank')
    }
    
    // 处理图片上传成功
    const handleImageSuccess = (response) => {
      articleForm.featuredImage = response.url
      handleContentChange()
    }
    
    // 图片上传前验证
    const beforeImageUpload = (file) => {
      const isImage = file.type.startsWith('image/')
      const isLt2M = file.size / 1024 / 1024 < 2
      
      if (!isImage) {
        ElMessage.error('只能上传图片文件!')
        return false
      }
      if (!isLt2M) {
        ElMessage.error('图片大小不能超过 2MB!')
        return false
      }
      return true
    }
    
    // 加载文章数据
    const loadArticle = async () => {
      if (route.params.id) {
        try {
          const article = await articleService.getArticle(route.params.id)
          
          articleForm.title = article.title
          articleForm.excerpt = article.excerpt
          articleForm.categories = article.categories
          articleForm.tags = article.tags.join(', ')
          articleForm.featuredImage = article.featuredImage
          
          if (quillEditor.value) {
            quillEditor.value.setContents(article.content)
          }
          
          status.value = '已加载'
        } catch (error) {
          console.error('加载文章失败:', error)
          ElMessage.error('加载文章失败')
        }
      }
    }
    
    // 加载分类数据
    const loadCategories = async () => {
      try {
        categories.value = await categoryService.getCategories()
      } catch (error) {
        console.error('加载分类失败:', error)
      }
    }
    
    // 自动保存
    const startAutoSave = () => {
      setInterval(() => {
        if (status.value === '未保存') {
          saveDraft()
        }
      }, 30000) // 30秒自动保存
    }
    
    onMounted(async () => {
      await nextTick()
      initEditor()
      await loadCategories()
      await loadArticle()
      startAutoSave()
    })
    
    onUnmounted(() => {
      if (quillEditor.value) {
        quillEditor.value = null
      }
    })
    
    return {
      editor,
      saving,
      publishing,
      status,
      wordCount,
      categories,
      articleForm,
      saveDraft,
      publish,
      preview,
      handleImageSuccess,
      beforeImageUpload,
      handleContentChange
    }
  }
}
</script>
```

### 视频播放器组件
```javascript
// src/components/VideoPlayer.vue
<template>
  <div class="video-player">
    <div class="player-container">
      <video
        ref="videoElement"
        class="video-element"
        :poster="video.poster"
        @loadedmetadata="onLoadedMetadata"
        @timeupdate="onTimeUpdate"
        @ended="onEnded"
        @play="onPlay"
        @pause="onPause"
      >
        <source :src="video.url" type="video/mp4" />
        您的浏览器不支持视频播放
      </video>
      
      <div class="player-controls" v-show="showControls">
        <div class="progress-bar" @click="seekTo">
          <div class="progress" :style="{ width: progressPercent + '%' }"></div>
          <div class="buffer" :style="{ width: bufferPercent + '%' }"></div>
        </div>
        
        <div class="control-buttons">
          <button @click="togglePlay" class="play-btn">
            <i :class="isPlaying ? 'icon-pause' : 'icon-play'"></i>
          </button>
          
          <div class="time-display">
            {{ formatTime(currentTime) }} / {{ formatTime(duration) }}
          </div>
          
          <div class="volume-control">
            <button @click="toggleMute" class="mute-btn">
              <i :class="isMuted ? 'icon-mute' : 'icon-volume'"></i>
            </button>
            <input
              type="range"
              v-model="volume"
              min="0"
              max="1"
              step="0.1"
              @input="onVolumeChange"
              class="volume-slider"
            />
          </div>
          
          <div class="playback-speed">
            <select v-model="playbackRate" @change="onPlaybackRateChange">
              <option value="0.5">0.5x</option>
              <option value="0.75">0.75x</option>
              <option value="1">1x</option>
              <option value="1.25">1.25x</option>
              <option value="1.5">1.5x</option>
              <option value="2">2x</option>
            </select>
          </div>
          
          <button @click="toggleFullscreen" class="fullscreen-btn">
            <i class="icon-fullscreen"></i>
          </button>
        </div>
      </div>
    </div>
    
    <div class="video-info">
      <h2 class="video-title">{{ video.title }}</h2>
      <div class="video-meta">
        <span class="video-duration">{{ formatTime(duration) }}</span>
        <span class="video-views">{{ video.views }} 次观看</span>
        <span class="video-date">{{ formatDate(video.createdAt) }}</span>
      </div>
    </div>
  </div>
</template>

<script>
import { ref, reactive, onMounted, onUnmounted, watch } from 'vue'
import { useRoute } from 'vue-router'
import VideoService from '../services/VideoService.js'

export default {
  name: 'VideoPlayer',
  props: {
    video: {
      type: Object,
      required: true
    }
  },
  setup(props) {
    const route = useRoute()
    const videoService = new VideoService()
    
    const videoElement = ref(null)
    const showControls = ref(true)
    const isPlaying = ref(false)
    const isMuted = ref(false)
    const currentTime = ref(0)
    const duration = ref(0)
    const volume = ref(1)
    const playbackRate = ref(1)
    const progressPercent = ref(0)
    const bufferPercent = ref(0)
    
    let controlsTimer = null
    let progressTimer = null
    
    // 视频加载完成
    const onLoadedMetadata = () => {
      duration.value = videoElement.value.duration
      loadProgress()
    }
    
    // 时间更新
    const onTimeUpdate = () => {
      currentTime.value = videoElement.value.currentTime
      progressPercent.value = (currentTime.value / duration.value) * 100
      
      // 更新学习进度
      updateProgress()
    }
    
    // 视频播放
    const onPlay = () => {
      isPlaying.value = true
      startProgressTimer()
    }
    
    // 视频暂停
    const onPause = () => {
      isPlaying.value = false
      stopProgressTimer()
    }
    
    // 视频结束
    const onEnded = () => {
      isPlaying.value = false
      currentTime.value = duration.value
      progressPercent.value = 100
      
      // 标记为已完成
      markAsCompleted()
    }
    
    // 切换播放/暂停
    const togglePlay = () => {
      if (isPlaying.value) {
        videoElement.value.pause()
      } else {
        videoElement.value.play()
      }
    }
    
    // 跳转到指定时间
    const seekTo = (event) => {
      const rect = event.currentTarget.getBoundingClientRect()
      const clickX = event.clientX - rect.left
      const percent = clickX / rect.width
      const time = percent * duration.value
      
      videoElement.value.currentTime = time
    }
    
    // 切换静音
    const toggleMute = () => {
      isMuted.value = !isMuted.value
      videoElement.value.muted = isMuted.value
    }
    
    // 音量变化
    const onVolumeChange = () => {
      videoElement.value.volume = volume.value
      isMuted.value = volume.value === 0
    }
    
    // 播放速度变化
    const onPlaybackRateChange = () => {
      videoElement.value.playbackRate = playbackRate.value
    }
    
    // 切换全屏
    const toggleFullscreen = () => {
      if (!document.fullscreenElement) {
        videoElement.value.requestFullscreen()
      } else {
        document.exitFullscreen()
      }
    }
    
    // 格式化时间
    const formatTime = (time) => {
      if (!time || isNaN(time)) return '00:00'
      
      const hours = Math.floor(time / 3600)
      const minutes = Math.floor((time % 3600) / 60)
      const seconds = Math.floor(time % 60)
      
      if (hours > 0) {
        return `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`
      } else {
        return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`
      }
    }
    
    // 格式化日期
    const formatDate = (date) => {
      return new Date(date).toLocaleDateString('zh-CN')
    }
    
    // 加载进度
    const loadProgress = async () => {
      try {
        const progress = await videoService.getProgress(props.video.id)
        if (progress) {
          currentTime.value = progress.currentTime
          videoElement.value.currentTime = progress.currentTime
        }
      } catch (error) {
        console.error('加载进度失败:', error)
      }
    }
    
    // 更新进度
    const updateProgress = async () => {
      try {
        await videoService.updateProgress(props.video.id, {
          currentTime: currentTime.value,
          duration: duration.value
        })
      } catch (error) {
        console.error('更新进度失败:', error)
      }
    }
    
    // 标记为已完成
    const markAsCompleted = async () => {
      try {
        await videoService.markAsCompleted(props.video.id)
      } catch (error) {
        console.error('标记完成失败:', error)
      }
    }
    
    // 开始进度计时器
    const startProgressTimer = () => {
      progressTimer = setInterval(() => {
        updateProgress()
      }, 5000) // 每5秒更新一次进度
    }
    
    // 停止进度计时器
    const stopProgressTimer = () => {
      if (progressTimer) {
        clearInterval(progressTimer)
        progressTimer = null
      }
    }
    
    // 显示/隐藏控制条
    const showControlsBar = () => {
      showControls.value = true
      
      if (controlsTimer) {
        clearTimeout(controlsTimer)
      }
      
      controlsTimer = setTimeout(() => {
        if (isPlaying.value) {
          showControls.value = false
        }
      }, 3000)
    }
    
    // 鼠标移动事件
    const handleMouseMove = () => {
      showControlsBar()
    }
    
    // 键盘事件
    const handleKeyDown = (event) => {
      switch (event.code) {
        case 'Space':
          event.preventDefault()
          togglePlay()
          break
        case 'ArrowLeft':
          event.preventDefault()
          videoElement.value.currentTime -= 10
          break
        case 'ArrowRight':
          event.preventDefault()
          videoElement.value.currentTime += 10
          break
        case 'KeyM':
          event.preventDefault()
          toggleMute()
          break
        case 'KeyF':
          event.preventDefault()
          toggleFullscreen()
          break
      }
    }
    
    onMounted(() => {
      // 添加事件监听器
      document.addEventListener('mousemove', handleMouseMove)
      document.addEventListener('keydown', handleKeyDown)
      
      // 加载进度
      loadProgress()
    })
    
    onUnmounted(() => {
      // 移除事件监听器
      document.removeEventListener('mousemove', handleMouseMove)
      document.removeEventListener('keydown', handleKeyDown)
      
      // 清理计时器
      if (controlsTimer) {
        clearTimeout(controlsTimer)
      }
      stopProgressTimer()
    })
    
    return {
      videoElement,
      showControls,
      isPlaying,
      isMuted,
      currentTime,
      duration,
      volume,
      playbackRate,
      progressPercent,
      bufferPercent,
      onLoadedMetadata,
      onTimeUpdate,
      onPlay,
      onPause,
      onEnded,
      togglePlay,
      seekTo,
      toggleMute,
      onVolumeChange,
      onPlaybackRateChange,
      toggleFullscreen,
      formatTime,
      formatDate
    }
  }
}
</script>
```
