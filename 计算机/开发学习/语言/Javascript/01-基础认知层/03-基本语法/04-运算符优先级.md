## 算术运算符

### 基本算术运算
```javascript
// 乘除取余优先级高于加减
console.log(2 + 3 * 4); // 14 (不是20)
console.log(2 + 3 * 4); // 等价于 2 + (3 * 4)

// 使用括号改变优先级
console.log((2 + 3) * 4); // 20

// 一元运算符优先级
console.log(-2 + 3); // 1 (等价于 (-2) + 3)
console.log(-(2 + 3)); // -5
```

### 递增递减运算符
```javascript
// 前置和后置的区别
let x = 5;
console.log(++x); // 6 (先递增，后使用)
console.log(x++); // 6 (先使用，后递增)
console.log(x); // 7

// 优先级示例
let y = 5;
console.log(y++ * 2); // 10 (等价于 (y++) * 2)
console.log(y); // 6

let z = 5;
console.log(++z * 2); // 12 (等价于 (++z) * 2)
console.log(z); // 6
```

## 比较运算符

### 比较运算优先级
```javascript
// 比较运算符优先级低于算术运算符
console.log(2 + 3 > 4); // true (等价于 (2 + 3) > 4)
console.log(2 + 3 > 4 + 1); // false (等价于 (2 + 3) > (4 + 1))

// 多个比较运算符
console.log(1 < 2 < 3); // true (等价于 (1 < 2) < 3，即 true < 3)
console.log(3 > 2 > 1); // false (等价于 (3 > 2) > 1，即 true > 1)
```

### 相等运算符
```javascript
// 相等运算符优先级低于比较运算符
console.log(2 + 3 == 5); // true (等价于 (2 + 3) == 5)
console.log(2 + 3 != 4); // true (等价于 (2 + 3) != 4)

// 严格相等
console.log(2 + 3 === 5); // true
console.log(2 + 3 !== 4); // true
```

## 逻辑运算符

### 逻辑与和逻辑或
```javascript
// 逻辑与优先级高于逻辑或
console.log(true || false && false); // true (等价于 true || (false && false))
console.log(false && true || true); // true (等价于 (false && true) || true)

// 短路求值
let x = 0;
console.log(x && x.toString()); // 0 (不会调用toString)
console.log(x || 'default'); // 'default'

// 实际应用
function getUserName(user) {
    return user && user.name || 'Anonymous';
}
```

### 条件运算符
```javascript
// 条件运算符优先级低于逻辑运算符
console.log(true ? 'yes' : 'no'); // 'yes'
console.log(false ? 'yes' : 'no'); // 'no'

// 嵌套条件运算符
let score = 85;
let grade = score >= 90 ? 'A' : 
            score >= 80 ? 'B' : 
            score >= 70 ? 'C' : 'F';
console.log(grade); // 'B'
```

## 赋值运算符

### 赋值运算优先级
```javascript
// 赋值运算符优先级最低
let x = 2 + 3 * 4; // 14 (等价于 x = (2 + 3 * 4))
let y = (2 + 3) * 4; // 20

// 复合赋值运算符
let z = 10;
z += 2 * 3; // 16 (等价于 z = z + (2 * 3))
console.log(z); // 16

// 链式赋值
let a, b, c;
a = b = c = 5; // 等价于 a = (b = (c = 5))
console.log(a, b, c); // 5 5 5
```

## 位运算符

### 位运算优先级
```javascript
// 位运算符优先级
console.log(5 & 3 | 1); // 1 (等价于 (5 & 3) | 1)
console.log(5 & (3 | 1)); // 5 (使用括号改变优先级)

// 位移运算符
console.log(8 >> 2); // 2 (右移2位)
console.log(8 << 2); // 32 (左移2位)
console.log(-8 >>> 2); // 1073741822 (无符号右移)
```

## 实际应用示例

### 复杂表达式解析
```javascript
// 复杂表达式
let result = a + b * c - d / e > f && g || h ? i : j;

// 等价于
let result = (((a + (b * c)) - (d / e)) > f) && g || h ? i : j;

// 更清晰的写法
let temp1 = b * c;
let temp2 = d / e;
let temp3 = a + temp1 - temp2;
let condition = temp3 > f && g || h;
let result = condition ? i : j;
```

### 函数调用优先级
```javascript
// 函数调用优先级
function getValue() {
    return 42;
}

function multiply(x, y) {
    return x * y;
}

// 函数调用优先级最高
console.log(multiply(2, 3) + 1); // 7 (等价于 (multiply(2, 3)) + 1)
console.log(getValue() * 2); // 84 (等价于 (getValue()) * 2)
```

### 对象属性访问
```javascript
// 对象属性访问优先级
const obj = {
    value: 10,
    getValue() {
        return this.value;
    }
};

// 属性访问优先级最高
console.log(obj.value + 5); // 15 (等价于 (obj.value) + 5)
console.log(obj.getValue() * 2); // 20 (等价于 (obj.getValue()) * 2)
```

## 最佳实践

### 使用括号提高可读性
```javascript
// 不清晰的表达式
let result = a + b * c - d / e > f && g || h ? i : j;

// 使用括号提高可读性
let result = ((a + (b * c)) - (d / e)) > f && g || h ? i : j;

// 或者分解为多个步骤
let step1 = b * c;
let step2 = d / e;
let step3 = a + step1 - step2;
let condition = step3 > f && g || h;
let result = condition ? i : j;
```

### 避免复杂的表达式
```javascript
// 避免复杂的表达式
function calculateComplex(a, b, c, d, e, f, g, h, i, j) {
    // 分解为多个步骤
    const multiplication = b * c;
    const division = d / e;
    const addition = a + multiplication;
    const subtraction = addition - division;
    const comparison = subtraction > f;
    const logicalAnd = comparison && g;
    const logicalOr = logicalAnd || h;
    
    return logicalOr ? i : j;
}
```
