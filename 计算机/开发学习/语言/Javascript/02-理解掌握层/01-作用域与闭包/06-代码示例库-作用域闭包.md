# 代码示例库-作用域闭包

## 基础作用域示例

### 1. 变量作用域
```javascript
// 全局作用域
var globalVar = 'global';

function outerFunction() {
    // 函数作用域
    var outerVar = 'outer';
    
    function innerFunction() {
        // 内层函数作用域
        var innerVar = 'inner';
        
        // 可以访问所有外层变量
        console.log(globalVar);  // 'global'
        console.log(outerVar);   // 'outer'
        console.log(innerVar);   // 'inner'
    }
    
    innerFunction();
}

outerFunction();
```

### 2. 块级作用域
```javascript
// let和const的块级作用域

blockScopeExample();
```

## 闭包基础示例

### 1. 基本闭包
```javascript
function createClosure() {
    var outerVariable = 'I am outer';
    
    function innerFunction() {
        console.log(outerVariable);
    }
    
    return innerFunction;
}

var closure = createClosure();
closure(); // 输出: 'I am outer'
```

### 2. 闭包保持状态
```javascript
function createCounter() {
    var count = 0;
    
    return function() {
        count++;
        return count;
    };
}

var counter1 = createCounter();
var counter2 = createCounter();

console.log(counter1()); // 1
console.log(counter1()); // 2
console.log(counter2()); // 1 (独立的闭包)
console.log(counter1()); // 3
```

```javascript
function createMultiplier(factor) {
    return function(number) {
        return number * factor;
    };
}

var double = createMultiplier(2);
var triple = createMultiplier(3);

console.log(double(5)); // 10
console.log(triple(5)); // 15
```

## 高级闭包示例

### 1. 模块模式
```javascript
var Calculator = (function() {
    // 私有变量
    var result = 0;
    var history = [];
    
    // 私有函数
    function addToHistory(operation, value) {
        history.push({
            operation: operation,
            value: value,
            result: result,
            timestamp: new Date()
        });
    }
    
    // 公共接口
    return {
        add: function(value) {
            result += value;
            addToHistory('add', value);
            return this;
        },
        
        subtract: function(value) {
            result -= value;
            addToHistory('subtract', value);
            return this;
        },
        
        multiply: function(value) {
            result *= value;
            addToHistory('multiply', value);
            return this;
        },
        
        divide: function(value) {
            if (value === 0) {
                throw new Error('Division by zero');
            }
            result /= value;
            addToHistory('divide', value);
            return this;
        },
        
        getResult: function() {
            return result;
        },
        
        getHistory: function() {
            return history.slice();
        },
        
        reset: function() {
            result = 0;
            history = [];
            return this;
        }
    };
})();

// 使用示例
Calculator.add(5).multiply(2).subtract(3);
console.log(Calculator.getResult()); // 7
console.log(Calculator.getHistory());
```

### 2. 函数工厂
```javascript
function createValidator(type, options = {}) {
    var rules = [];
    var customMessages = options.messages || {};
    
    function addRule(rule, message) {
        rules.push({ rule, message });
    }
    
    return {
        required: function(message) {
            addRule('required', message || customMessages.required || 'This field is required');
            return this;
        },
        
        minLength: function(length, message) {
            addRule('minLength', message || customMessages.minLength || `Minimum length is ${length}`);
            return this;
        },
        
        maxLength: function(length, message) {
            addRule('maxLength', message || customMessages.maxLength || `Maximum length is ${length}`);
            return this;
        },
        
        pattern: function(regex, message) {
            addRule('pattern', message || customMessages.pattern || 'Invalid format');
            return this;
        },
        
        validate: function(value) {
            var errors = [];
            
            for (var i = 0; i < rules.length; i++) {
                var rule = rules[i];
                var isValid = true;
                
                switch (rule.rule) {
                    case 'required':
                        isValid = value !== null && value !== undefined && value !== '';
                        break;
                    case 'minLength':
                        isValid = value.length >= rule.length;
                        break;
                    case 'maxLength':
                        isValid = value.length <= rule.length;
                        break;
                    case 'pattern':
                        isValid = rule.regex.test(value);
                        break;
                }
                
                if (!isValid) {
                    errors.push(rule.message);
                }
            }
            
            return {
                isValid: errors.length === 0,
                errors: errors
            };
        }
    };
}

// 使用示例
var emailValidator = createValidator('email', {
    messages: {
        required: 'Email is required',
        pattern: 'Invalid email format'
    }
}).required().pattern(/^[^\s@]+@[^\s@]+\.[^\s@]+$/);

var passwordValidator = createValidator('password')
    .required('Password is required')
    .minLength(8, 'Password must be at least 8 characters')
    .maxLength(20, 'Password must be no more than 20 characters');

console.log(emailValidator.validate('test@example.com'));
console.log(passwordValidator.validate('123456'));
```

### 3. 缓存和记忆化
```javascript
function createCache(fn, maxSize = 100) {
    var cache = new Map();
    var accessOrder = [];
    
    function get(key) {
        if (cache.has(key)) {
            // 更新访问顺序
            var index = accessOrder.indexOf(key);
            if (index > -1) {
                accessOrder.splice(index, 1);
            }
            accessOrder.push(key);
            return cache.get(key);
        }
        return undefined;
    }
    
    function set(key, value) {
        if (cache.size >= maxSize) {
            // 移除最久未使用的项
            var oldestKey = accessOrder.shift();
            cache.delete(oldestKey);
        }
        
        cache.set(key, value);
        accessOrder.push(key);
    }
    
    return function(...args) {
        var key = JSON.stringify(args);
        var cached = get(key);
        
        if (cached !== undefined) {
            console.log('Cache hit for:', key);
            return cached;
        }
        
        console.log('Cache miss for:', key);
        var result = fn.apply(this, args);
        set(key, result);
        return result;
    };
}

// 使用示例
function expensiveCalculation(n) {
    console.log('Computing for:', n);
    var result = 0;
    for (var i = 0; i < n; i++) {
        result += Math.sqrt(i);
    }
    return result;
}

var cachedCalculation = createCache(expensiveCalculation, 5);

console.log(cachedCalculation(1000)); // 计算
console.log(cachedCalculation(1000)); // 从缓存获取
console.log(cachedCalculation(2000)); // 计算
console.log(cachedCalculation(1000)); // 从缓存获取
```

## 事件处理示例

### 1. 事件监听器管理
```javascript
function createEventListenerManager() {
    var element = document.getElementById('button');
    var data = new Array(100000).fill('data');
    var listeners = [];
    
    function addListener(event, handler) {
        element.addEventListener(event, handler);
        listeners.push({ event, handler });
    }
    
    function removeListener(event, handler) {
        element.removeEventListener(event, handler);
        var index = listeners.findIndex(l => l.event === event && l.handler === handler);
        if (index > -1) {
            listeners.splice(index, 1);
        }
    }
    
    function cleanup() {
        listeners.forEach(({ event, handler }) => {
            element.removeEventListener(event, handler);
        });
        listeners = [];
        data = null;
    }
    
    addListener('click', function() {
        console.log('Clicked', data.length);
    });
    
    return { addListener, removeListener, cleanup };
}
```

function createEventDelegation() {

    var data = new Array(100000).fill('data');

## 状态管理示例

### 1. 简单状态管理器
```javascript
function createStateManager(initialState = {}) {
    var state = { ...initialState };
    var listeners = [];
    var isUpdating = false;
    
    function getState() {
        return { ...state };
    }
    
    function setState(newState) {
        if (isUpdating) {
            console.warn('State update already in progress');
            return;
        }
        
        isUpdating = true;
        
        try {
            var prevState = { ...state };
            state = { ...state, ...newState };
            
            // 通知所有监听器
            listeners.forEach(function(listener) {
                try {
                    listener(state, prevState);
                } catch (error) {
                    console.error('State listener error:', error);
                }
            });
        } finally {
            isUpdating = false;
        }
    }
    
    function subscribe(listener) {
        if (typeof listener !== 'function') {
            throw new Error('Listener must be a function');
        }
        
        listeners.push(listener);
        
        // 返回取消订阅函数
        return function unsubscribe() {
            var index = listeners.indexOf(listener);
            if (index > -1) {
                listeners.splice(index, 1);
            }
        };
    }
    
    return {
        getState: getState,
        setState: setState,
        subscribe: subscribe
    };
}

// 使用示例
var stateManager = createStateManager({ count: 0 });

var unsubscribe = stateManager.subscribe(function(state, prevState) {
    console.log('State changed:', state);
});

stateManager.setState({ count: 1 });
stateManager.setState({ count: 2 });

unsubscribe();
```

### 2. 高级状态管理器
```javascript
class AdvancedStateManager {
    constructor(initialState = {}, options = {}) {
        this.state = { ...initialState };
        this.listeners = new Map();
        this.middlewares = [];
        this.isUpdating = false;
        this.updateQueue = [];
        this.batchUpdates = options.batchUpdates || false;
    }
    
    getState() {
        return { ...this.state };
    }
    
    setState(newState, callback) {
        if (this.isUpdating) {
            if (this.batchUpdates) {
                this.updateQueue.push({ newState, callback });
                return;
            } else {
                console.warn('State update already in progress');
                return;
            }
        }
        
        this.isUpdating = true;
        
        try {
            var prevState = { ...this.state };
            var nextState = { ...this.state, ...newState };
            
            // 应用中间件
            var finalState = this.applyMiddlewares(nextState, prevState);
            
            this.state = finalState;
            
            // 通知监听器
            this.notifyListeners(finalState, prevState);
            
            // 执行回调
            if (callback) {
                callback(finalState, prevState);
            }
        } finally {
            this.isUpdating = false;
            
            // 处理队列中的更新
            if (this.batchUpdates && this.updateQueue.length > 0) {
                var queuedUpdate = this.updateQueue.shift();
                this.setState(queuedUpdate.newState, queuedUpdate.callback);
            }
        }
    }
    
    applyMiddlewares(nextState, prevState) {
        return this.middlewares.reduce(function(state, middleware) {
            return middleware(state, prevState);
        }, nextState);
    }
    
    notifyListeners(nextState, prevState) {
        this.listeners.forEach(function(listener, key) {
            try {
                listener(nextState, prevState);
            } catch (error) {
                console.error(`State listener error for key ${key}:`, error);
            }
        });
    }
    
    subscribe(key, listener) {
        if (typeof listener !== 'function') {
            throw new Error('Listener must be a function');
        }
        
        this.listeners.set(key, listener);
        
        return function unsubscribe() {
            this.listeners.delete(key);
        }.bind(this);
    }
    
    addMiddleware(middleware) {
        if (typeof middleware !== 'function') {
            throw new Error('Middleware must be a function');
        }
        
        this.middlewares.push(middleware);
    }
}
```

## 内存管理示例

### 1. 内存泄漏检测
```javascript
class MemoryLeakDetector {
    constructor() {
        this.baseline = null;
        this.threshold = 10 * 1024 * 1024; // 10MB
        this.interval = 5000; // 5秒
        this.timer = null;
    }
    
    start() {
        this.baseline = this.getCurrentMemory();
        this.timer = setInterval(() => {
            this.checkMemoryLeak();
        }, this.interval);
    }
    
    stop() {
        if (this.timer) {
            clearInterval(this.timer);
            this.timer = null;
        }
    }
    
    getCurrentMemory() {
        return performance.memory ? performance.memory.usedJSHeapSize : 0;
    }
    
    checkMemoryLeak() {
        var current = this.getCurrentMemory();
        var increase = current - this.baseline;
        
        if (increase > this.threshold) {
            console.warn(`潜在内存泄漏检测到: 增加了 ${(increase / 1024 / 1024).toFixed(2)} MB`);
        }
    }
}

// 使用示例
var detector = new MemoryLeakDetector();
detector.start();
```

### 2. 资源管理器
```javascript
class ResourceManager {
    constructor() {
        this.resources = new Set();
        this.cleanupFunctions = new Set();
    }
    
    addResource(resource) {
        this.resources.add(resource);
        return resource;
    }
    
    addCleanupFunction(cleanup) {
        this.cleanupFunctions.add(cleanup);
        return cleanup;
    }
    
    cleanup() {
        this.cleanupFunctions.forEach(cleanup => {
            try {
                cleanup();
            } catch (error) {
                console.error('Cleanup error:', error);
            }
        });
        
        this.cleanupFunctions.clear();
        this.resources.clear();
    }
}

// 使用示例
function createAutoCleanup() {
    var manager = new ResourceManager();
    var data = new Array(100000).fill('data');
    
    manager.addResource(data);
    manager.addCleanupFunction(() => {
        data = null;
    });
    
    // 页面卸载时自动清理
    window.addEventListener('beforeunload', () => {
        manager.cleanup();
    });
    
    return manager;
}
```
