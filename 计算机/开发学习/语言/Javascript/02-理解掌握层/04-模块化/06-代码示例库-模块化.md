### 简单模块定义
```javascript
// 1. 基础模块
// math-utils.js
export const PI = 3.14159;

export function add(a, b) {
    return a + b;
}

export function subtract(a, b) {
    return a - b;
}

export function multiply(a, b) {
    return a * b;
}

export function divide(a, b) {
    if (b === 0) {
        throw new Error('除数不能为零');
    }
    return a / b;
}

// 默认导出
export default class MathUtils {
    static calculateCircleArea(radius) {
        return PI * radius * radius;
    }
    
    static calculateCirclePerimeter(radius) {
        return 2 * PI * radius;
    }
}

// 2. 使用模块
// main.js
import MathUtils, { add, subtract, multiply, divide, PI } from './math-utils.js';

console.log('PI值:', PI);
console.log('加法:', add(5, 3));
console.log('减法:', subtract(5, 3));
console.log('乘法:', multiply(5, 3));
console.log('除法:', divide(6, 2));

const area = MathUtils.calculateCircleArea(5);
console.log('圆面积:', area);
```

```javascript
// 1. 组合多个模块
// geometry.js
import { PI } from './math-utils.js';

export class Circle {
    constructor(radius) {
        this.radius = radius;
    }
    
    getArea() {
        return PI * this.radius * this.radius;
    }
    
    getPerimeter() {
        return 2 * PI * this.radius;
    }
}

export class Rectangle {
    constructor(width, height) {
        this.width = width;
        this.height = height;
    }
    
    getArea() {
        return this.width * this.height;
    }
    
    getPerimeter() {
        return 2 * (this.width + this.height);
    }
}

// 2. 使用组合模块
// app.js
import { Circle, Rectangle } from './geometry.js';

const circle = new Circle(5);
const rectangle = new Rectangle(4, 6);

console.log('圆面积:', circle.getArea());
console.log('矩形面积:', rectangle.getArea());
```

```javascript
// 1. 单例模块
// config.js
class Config {
    constructor() {
        if (Config.instance) {
            return Config.instance;
        }
        
        this.apiUrl = 'https://api.example.com';
        this.timeout = 5000;
        this.retryCount = 3;
        this.debug = false;
        
        Config.instance = this;
    }
    
    setApiUrl(url) {
        this.apiUrl = url;
    }
    
    setTimeout(timeout) {
        this.timeout = timeout;
    }
    
    setDebug(debug) {
        this.debug = debug;
    }
    
    getConfig() {
        return {
            apiUrl: this.apiUrl,
            timeout: this.timeout,
            retryCount: this.retryCount,
            debug: this.debug
        };
    }
}

export default new Config();

// 2. 使用单例模块
// api-client.js
import config from './config.js';

export class ApiClient {
    constructor() {
        this.baseUrl = config.apiUrl;
        this.timeout = config.timeout;
    }
    
    async request(endpoint, options = {}) {
        const url = `${this.baseUrl}${endpoint}`;
        const requestOptions = {
            timeout: this.timeout,
            ...options
        };
        
        if (config.debug) {
            console.log('API请求:', url, requestOptions);
        }
        
        try {
            const response = await fetch(url, requestOptions);
            return await response.json();
        } catch (error) {
            if (config.debug) {
                console.error('API请求失败:', error);
            }
            throw error;
        }
    }
}
```

### 工厂模式模块
```javascript
// 1. 工厂模块
// logger-factory.js
class Logger {
    constructor(level = 'info') {
        this.level = level;
        this.levels = {
            error: 0,
            warn: 1,
            info: 2,
            debug: 3
        };
    }
    
    log(level, message, ...args) {
        if (this.levels[level] <= this.levels[this.level]) {
            console[level](`[${level.toUpperCase()}] ${message}`, ...args);
        }
    }
    
    error(message, ...args) {
        this.log('error', message, ...args);
    }
    
    warn(message, ...args) {
        this.log('warn', message, ...args);
    }
    
    info(message, ...args) {
        this.log('info', message, ...args);
    }
    
    debug(message, ...args) {
        this.log('debug', message, ...args);
    }
}

export class LoggerFactory {
    static createLogger(level = 'info') {
        return new Logger(level);
    }
    
    static createConsoleLogger() {
        return new Logger('info');
    }
    
    static createDebugLogger() {
        return new Logger('debug');
    }
    
    static createErrorLogger() {
        return new Logger('error');
    }
}

// 2. 使用工厂模块
// app.js
import { LoggerFactory } from './logger-factory.js';

const logger = LoggerFactory.createDebugLogger();
logger.info('应用启动');
logger.debug('调试信息');
logger.warn('警告信息');
logger.error('错误信息');
```

### 观察者模式模块
```javascript
// 1. 观察者模块
// event-emitter.js
export class EventEmitter {
    constructor() {
        this.events = new Map();
    }
    
    on(event, listener) {
        if (!this.events.has(event)) {
            this.events.set(event, []);
        }
        this.events.get(event).push(listener);
    }
    
    off(event, listener) {
        if (!this.events.has(event)) {
            return;
        }
        
        const listeners = this.events.get(event);
        const index = listeners.indexOf(listener);
        if (index > -1) {
            listeners.splice(index, 1);
        }
    }
    
    emit(event, ...args) {
        if (!this.events.has(event)) {
            return;
        }
        
        const listeners = this.events.get(event);
        listeners.forEach(listener => {
            try {
                listener(...args);
            } catch (error) {
                console.error(`事件 ${event} 监听器执行错误:`, error);
            }
        });
    }
    
    once(event, listener) {
        const onceWrapper = (...args) => {
            listener(...args);
            this.off(event, onceWrapper);
        };
        this.on(event, onceWrapper);
    }
}

// 2. 使用观察者模块
// user-manager.js
import { EventEmitter } from './event-emitter.js';

export class UserManager extends EventEmitter {
    constructor() {
        super();
        this.users = new Map();
    }
    
    addUser(user) {
        this.users.set(user.id, user);
        this.emit('userAdded', user);
    }
    
    removeUser(userId) {
        const user = this.users.get(userId);
        if (user) {
            this.users.delete(userId);
            this.emit('userRemoved', user);
        }
    }
    
    updateUser(userId, updates) {
        const user = this.users.get(userId);
        if (user) {
            Object.assign(user, updates);
            this.emit('userUpdated', user);
        }
    }
}

// 3. 使用用户管理器
// app.js
import { UserManager } from './user-manager.js';

const userManager = new UserManager();

// 监听事件
userManager.on('userAdded', (user) => {
    console.log('用户已添加:', user.name);
});

userManager.on('userRemoved', (user) => {
    console.log('用户已删除:', user.name);
});

userManager.on('userUpdated', (user) => {
    console.log('用户已更新:', user.name);
});

// 添加用户
userManager.addUser({ id: 1, name: 'John', email: 'john@example.com' });
userManager.updateUser(1, { name: 'John Doe' });
userManager.removeUser(1);
```

### 动态加载
```javascript
// 1. 动态加载模块
// module-loader.js
export class ModuleLoader {
    constructor() {
        this.loadedModules = new Map();
        this.loadingPromises = new Map();
    }
    
    async load(modulePath) {
        if (this.loadedModules.has(modulePath)) {
            return this.loadedModules.get(modulePath);
        }
        
        if (this.loadingPromises.has(modulePath)) {
            return this.loadingPromises.get(modulePath);
        }
        
        const loadingPromise = this.loadModule(modulePath);
        this.loadingPromises.set(modulePath, loadingPromise);
        
        try {
            const module = await loadingPromise;
            this.loadedModules.set(modulePath, module);
            this.loadingPromises.delete(modulePath);
            return module;
        } catch (error) {
            this.loadingPromises.delete(modulePath);
            throw error;
        }
    }
    
    async loadModule(modulePath) {
        const module = await import(modulePath);
        return module;
    }
    
    preload(modulePath) {
        if (!this.loadedModules.has(modulePath) && !this.loadingPromises.has(modulePath)) {
            this.load(modulePath);
        }
    }
}

// 2. 使用动态加载
// app.js
import { ModuleLoader } from './module-loader.js';

const loader = new ModuleLoader();

// 动态加载模块
async function loadUserModule() {
    try {
        const userModule = await loader.load('./user-module.js');
        const user = new userModule.User('John', 'john@example.com');
        console.log('用户:', user);
    } catch (error) {
        console.error('加载用户模块失败:', error);
    }
}

// 预加载模块
loader.preload('./product-module.js');
loader.preload('./order-module.js');

loadUserModule();
```

### 条件加载
```javascript
// 1. 条件加载模块
// conditional-loader.js
export class ConditionalLoader {
    constructor() {
        this.conditions = new Map();
        this.loadedModules = new Map();
    }
    
    addCondition(modulePath, condition) {
        this.conditions.set(modulePath, condition);
    }
    
    async loadIf(modulePath) {
        const condition = this.conditions.get(modulePath);
        if (condition && condition()) {
            if (!this.loadedModules.has(modulePath)) {
                const module = await import(modulePath);
                this.loadedModules.set(modulePath, module);
            }
            return this.loadedModules.get(modulePath);
        }
        return null;
    }
    
    async loadAll() {
        const results = new Map();
        
        for (const [modulePath, condition] of this.conditions) {
            if (condition()) {
                try {
                    const module = await this.loadIf(modulePath);
                    results.set(modulePath, module);
                } catch (error) {
                    console.error(`加载模块 ${modulePath} 失败:`, error);
                }
            }
        }
        
        return results;
    }
}

// 2. 使用条件加载
// app.js
import { ConditionalLoader } from './conditional-loader.js';

const loader = new ConditionalLoader();

// 添加条件
loader.addCondition('./admin-module.js', () => user.role === 'admin');
loader.addCondition('./premium-module.js', () => user.isPremium);
loader.addCondition('./mobile-module.js', () => window.innerWidth < 768);

// 条件加载
const adminModule = await loader.loadIf('./admin-module.js');
const premiumModule = await loader.loadIf('./premium-module.js');
const mobileModule = await loader.loadIf('./mobile-module.js');

// 加载所有符合条件的模块
const allModules = await loader.loadAll();
console.log('已加载模块:', allModules);
```

```javascript
// 1. 测试工具模块
// test-utils.js
export class TestUtils {
    static assert(condition, message) {
        if (!condition) {
            throw new Error(`断言失败: ${message}`);
        }
    }
    
    static assertEquals(actual, expected, message) {
        if (actual !== expected) {
            throw new Error(`断言失败: ${message}. 期望: ${expected}, 实际: ${actual}`);
        }
    }
    
    static assertThrows(fn, expectedError, message) {
        try {
            fn();
            throw new Error(`断言失败: ${message}. 期望抛出错误，但没有抛出`);
        } catch (error) {
            if (expectedError && !(error instanceof expectedError)) {
                throw new Error(`断言失败: ${message}. 期望错误类型: ${expectedError.name}, 实际: ${error.constructor.name}`);
            }
        }
    }
    
    static async assertAsync(fn, message) {
        try {
            await fn();
        } catch (error) {
            throw new Error(`异步断言失败: ${message}. 错误: ${error.message}`);
        }
    }
}

// 2. 测试数学工具模块
// math-utils.test.js

import { add, subtract, multiply, divide } from './math-utils.js';

            this.testAdd();

            this.testDivideByZero();
            
            console.log('所有测试通过!');
        } catch (error) {
            console.error('测试失败:', error.message);
        }
    }
    
    static testAdd() {
        TestUtils.assertEquals(add(2, 3), 5, '加法测试');
        TestUtils.assertEquals(add(-1, 1), 0, '负数加法测试');
        TestUtils.assertEquals(add(0, 0), 0, '零加法测试');
    }
    
    static testSubtract() {
        TestUtils.assertEquals(subtract(5, 3), 2, '减法测试');
        TestUtils.assertEquals(subtract(1, 1), 0, '相等数减法测试');
        TestUtils.assertEquals(subtract(0, 5), -5, '负数结果减法测试');
    }

            () => divide(5, 0),
```
