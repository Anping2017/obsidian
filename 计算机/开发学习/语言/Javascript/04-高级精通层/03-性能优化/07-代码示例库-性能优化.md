// 1. 内存使用监控
class MemoryMonitor {
    constructor() {
        this.initialMemory = this.getMemoryUsage();
        this.peakMemory = this.initialMemory;
        this.memoryHistory = [];
        this.startMonitoring();
    }
    
    getMemoryUsage() {
        if (performance.memory) {
            return {
                used: performance.memory.usedJSHeapSize,
                total: performance.memory.totalJSHeapSize,
                limit: performance.memory.jsHeapSizeLimit
            };
        }
        return null;
    }
    
    startMonitoring() {
        setInterval(() => {
            const current = this.getMemoryUsage();
            if (current) {
                this.memoryHistory.push({
                    timestamp: Date.now(),
                    ...current
                });
                
                if (current.used > this.peakMemory.used) {
                    this.peakMemory = current;
                }
                
                // 保持最近1000个记录
                if (this.memoryHistory.length > 1000) {
                    this.memoryHistory.shift();
                }
            }
        }, 1000);
    }
    
    getMemoryStats() {
        const current = this.getMemoryUsage();
        if (!current) return null;
        
        return {
            current,
            peak: this.peakMemory,
            growth: current.used - this.initialMemory.used,
            usage: (current.used / current.limit) * 100,
            history: this.memoryHistory.slice(-100) // 最近100个记录
        };
    }
    
    checkMemoryLeak() {
        const stats = this.getMemoryStats();
        if (!stats) return false;
        
        // 检查内存使用率是否过高
        if (stats.usage > 80) {
            console.warn('High memory usage detected:', stats.usage + '%');
            return true;
        }
        
        // 检查内存增长趋势
        const recent = stats.history.slice(-10);
        if (recent.length >= 10) {
            const trend = recent[recent.length - 1].used - recent[0].used;
            if (trend > 10 * 1024 * 1024) { // 10MB增长
                console.warn('Potential memory leak detected');
                return true;
            }
        }
        
        return false;
    }
}

// 2. 对象池模式
class ObjectPool {
    constructor(createFn, resetFn, maxSize = 100) {
        this.createFn = createFn;
        this.resetFn = resetFn;
        this.maxSize = maxSize;
        this.pool = [];
        this.active = new Set();
    }
    
    acquire() {
        let obj;
        
        if (this.pool.length > 0) {
            obj = this.pool.pop();
        } else {
            obj = this.createFn();
        }
        
        this.active.add(obj);
        return obj;
    }
    
    release(obj) {
        if (this.active.has(obj)) {
            this.active.delete(obj);
            this.resetFn(obj);
            
            if (this.pool.length < this.maxSize) {
                this.pool.push(obj);
            }
        }
    }
    
    getStats() {
        return {
            poolSize: this.pool.length,
            activeCount: this.active.size,
            totalCreated: this.pool.length + this.active.size
        };
    }
    
    clear() {
        this.pool.length = 0;
        this.active.clear();
    }
}

// 3. 内存泄漏检测
class MemoryLeakDetector {
    constructor() {
        this.snapshots = [];
        this.leakThreshold = 1024 * 1024; // 1MB
        this.growthThreshold = 0.1; // 10%
    }
    
    takeSnapshot(label) {
        const snapshot = {
            label,
            timestamp: Date.now(),
            memory: this.getMemoryUsage(),
            objects: this.countObjects()
        };
        
        this.snapshots.push(snapshot);
        return snapshot;
    }
    
    getMemoryUsage() {
        if (performance.memory) {
            return {
                used: performance.memory.usedJSHeapSize,
                total: performance.memory.totalJSHeapSize,
                limit: performance.memory.jsHeapSizeLimit
            };
        }
        return null;
    }
    
    countObjects() {
        // 简化的对象计数
        return {
            total: 0,
            arrays: 0,
            objects: 0,
            functions: 0
        };
    }
    
    detectLeaks() {
        if (this.snapshots.length < 2) return [];
        
        const leaks = [];
        const latest = this.snapshots[this.snapshots.length - 1];
        const previous = this.snapshots[this.snapshots.length - 2];
        
        const memoryIncrease = latest.memory.used - previous.memory.used;
        const timeIncrease = latest.timestamp - previous.timestamp;
        
        if (memoryIncrease > this.leakThreshold) {
            leaks.push({
                type: 'memory',
                increase: memoryIncrease,
                time: timeIncrease,
                severity: memoryIncrease > 10 * this.leakThreshold ? 'high' : 'medium'
            });
        }
        
        return leaks;
    }
    
    generateReport() {
        const leaks = this.detectLeaks();
        return {
            snapshots: this.snapshots.length,
            leaks: leaks,
            summary: {
                totalLeaks: leaks.length,
                highSeverity: leaks.filter(l => l.severity === 'high').length,
                mediumSeverity: leaks.filter(l => l.severity === 'medium').length
            }
        };
    }
}

// 使用示例

// 定期检测

    leakDetector.takeSnapshot(`Snapshot ${Date.now()}`);
    const report = leakDetector.generateReport();
    if (report.leaks.length > 0) {
        console.log('Memory leaks detected:', report);
    }
}, 10000);

// 1. 智能内存管理
class SmartMemoryManager {
    constructor() {
        this.memoryLimit = 100 * 1024 * 1024; // 100MB
        this.cleanupThreshold = 0.8; // 80%
        this.cleanupInterval = 30000; // 30秒
        this.cache = new Map();
        this.accessCount = new Map();
        this.lastAccess = new Map();
        this.startCleanup();
    }
    
    set(key, value, ttl = 300000) {
        const now = Date.now();
        
        // 检查内存使用
        if (this.getMemoryUsage() > this.memoryLimit * this.cleanupThreshold) {
            this.cleanup();
        }
        
        this.cache.set(key, {
            value,
            ttl,
            createdAt: now
        });
        
        this.accessCount.set(key, 0);
        this.lastAccess.set(key, now);
    }
    
    get(key) {
        const item = this.cache.get(key);
        if (!item) return null;
        
        const now = Date.now();
        
        // 检查TTL
        if (now - item.createdAt > item.ttl) {
            this.delete(key);
            return null;
        }
        
        // 更新访问统计
        this.accessCount.set(key, (this.accessCount.get(key) || 0) + 1);
        this.lastAccess.set(key, now);
        
        return item.value;
    }
    
    delete(key) {
        this.cache.delete(key);
        this.accessCount.delete(key);
        this.lastAccess.delete(key);
    }
    
    getMemoryUsage() {
        if (performance.memory) {
            return performance.memory.usedJSHeapSize;
        }
        return 0;
    }
    
    cleanup() {
        const now = Date.now();
        const toDelete = [];
        
        // 找出需要清理的项
        for (const [key, item] of this.cache) {
            const age = now - item.createdAt;
            const lastAccessTime = this.lastAccess.get(key) || 0;
            const accessCount = this.accessCount.get(key) || 0;
            
            // 清理策略：TTL过期、长时间未访问、访问次数少
            if (age > item.ttl || 
                (now - lastAccessTime > 300000 && accessCount < 2) ||
                (age > 600000 && accessCount === 0)) {
                toDelete.push(key);
            }
        }
        
        // 删除项
        for (const key of toDelete) {
            this.delete(key);
        }
        
        console.log(`Cleaned up ${toDelete.length} cache items`);
    }
    
    startCleanup() {
        setInterval(() => {
            this.cleanup();
        }, this.cleanupInterval);
    }
    
    getStats() {
        return {
            cacheSize: this.cache.size,
            memoryUsage: this.getMemoryUsage(),
            memoryLimit: this.memoryLimit,
            usagePercent: (this.getMemoryUsage() / this.memoryLimit) * 100
        };
    }
}

// 1. 动态导入优化
class DynamicImportOptimizer {
    constructor() {
        this.cache = new Map();
        this.loadingPromises = new Map();
        this.preloadQueue = [];
        this.isPreloading = false;
    }
    
    // 优化的动态导入
    async import(modulePath, options = {}) {
        const {
            preload = false,
            priority = 'normal',
            retryCount = 3,
            timeout = 10000
        } = options;
        
        // 检查缓存
        if (this.cache.has(modulePath)) {
            return this.cache.get(modulePath);
        }
        
        // 检查是否正在加载
        if (this.loadingPromises.has(modulePath)) {
            return this.loadingPromises.get(modulePath);
        }
        
        // 开始加载
        const loadPromise = this.loadWithRetry(modulePath, retryCount, timeout);
        this.loadingPromises.set(modulePath, loadPromise);
        
        try {
            const module = await loadPromise;
            this.cache.set(modulePath, module);
            this.loadingPromises.delete(modulePath);
            return module;
        } catch (error) {
            this.loadingPromises.delete(modulePath);
            throw error;
        }
    }
    
    // 带重试的加载
    async loadWithRetry(modulePath, retryCount, timeout) {
        let lastError;
        
        for (let i = 0; i < retryCount; i++) {
            try {
                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), timeout);
                
                const module = await import(modulePath);
                clearTimeout(timeoutId);
                
                return module;
            } catch (error) {
                lastError = error;
                if (i < retryCount - 1) {
                    await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)));
                }
            }
        }
        
        throw lastError;
    }
    
    // 预加载模块
    preload(modulePath, priority = 'normal') {
        this.preloadQueue.push({ modulePath, priority });
        this.preloadQueue.sort((a, b) => {
            const priorityOrder = { high: 3, normal: 2, low: 1 };
            return priorityOrder[b.priority] - priorityOrder[a.priority];
        });
        
        if (!this.isPreloading) {
            this.processPreloadQueue();
        }
    }
    
    // 处理预加载队列
    async processPreloadQueue() {
        this.isPreloading = true;
        
        while (this.preloadQueue.length > 0) {
            const { modulePath } = this.preloadQueue.shift();
            
            try {
                await this.import(modulePath, { preload: true });
            } catch (error) {
                console.warn(`Preload failed for ${modulePath}:`, error);
            }
        }
        
        this.isPreloading = false;
    }
    
    // 获取统计信息
    getStats() {
        return {
            cachedModules: this.cache.size,
            loadingModules: this.loadingPromises.size,
            preloadQueue: this.preloadQueue.length,
            isPreloading: this.isPreloading
        };
    }
}

// 2. 路由代码分割
class RouteCodeSplitter {
    constructor() {
        this.routes = new Map();
        this.loadedRoutes = new Set();
        this.preloadRoutes = new Set();
    }
    
    // 注册路由
    registerRoute(path, componentPath, options = {}) {
        const {
            preload = false,
            priority = 'normal',
            dependencies = []
        } = options;
        
        this.routes.set(path, {
            componentPath,
            preload,
            priority,
            dependencies,
            loaded: false
        });
        
        if (preload) {
            this.preloadRoutes.add(path);
        }
    }
    
    // 加载路由
    async loadRoute(path) {
        const route = this.routes.get(path);
        if (!route) {
            throw new Error(`Route not found: ${path}`);
        }
        
        if (this.loadedRoutes.has(path)) {
            return route.component;
        }
        
        try {
            // 加载依赖
            for (const dep of route.dependencies) {
                await this.loadRoute(dep);
            }
            
            // 加载组件
            const module = await import(route.componentPath);
            route.component = module.default || module;
            route.loaded = true;
            
            this.loadedRoutes.add(path);
            return route.component;
        } catch (error) {
            console.error(`Failed to load route ${path}:`, error);
            throw error;
        }
    }
    
    // 预加载路由
    async preloadRoute(path) {
        if (this.preloadRoutes.has(path) && !this.loadedRoutes.has(path)) {
            try {
                await this.loadRoute(path);
                console.log(`Preloaded route: ${path}`);
            } catch (error) {
                console.warn(`Failed to preload route ${path}:`, error);
            }
        }
    }
    
    // 预加载所有路由
    async preloadAllRoutes() {
        const preloadPromises = Array.from(this.preloadRoutes).map(path => 
            this.preloadRoute(path)
        );
        
        await Promise.allSettled(preloadPromises);
    }
    
    // 获取路由状态
    getRouteStatus() {
        return {
            totalRoutes: this.routes.size,
            loadedRoutes: this.loadedRoutes.size,
            preloadRoutes: this.preloadRoutes.size,
            routes: Array.from(this.routes.entries()).map(([path, route]) => ({
                path,
                loaded: route.loaded,
                preload: route.preload,
                priority: route.priority
            }))
        };
    }
}

// 1. 多级缓存系统
class MultiLevelCache {
    constructor() {
        this.l1Cache = new Map(); // 内存缓存
        this.l2Cache = new Map(); // 持久化缓存
        this.stats = {
            l1Hits: 0,
            l2Hits: 0,
            misses: 0,
            sets: 0
        };
        this.maxL1Size = 100;
        this.maxL2Size = 1000;
    }
    
    // 获取数据
    async get(key) {
        // 检查L1缓存
        if (this.l1Cache.has(key)) {
            this.stats.l1Hits++;
            return this.l1Cache.get(key);
        }
        
        // 检查L2缓存
        if (this.l2Cache.has(key)) {
            this.stats.l2Hits++;
            const value = this.l2Cache.get(key);
            
            // 提升到L1缓存
            this.setL1(key, value);
            return value;
        }
        
        this.stats.misses++;
        return null;
    }
    
    // 设置数据
    async set(key, value, ttl = 300000) {
        this.stats.sets++;
        
        // 设置到L1缓存
        this.setL1(key, value);
        
        // 设置到L2缓存
        this.setL2(key, value, ttl);
    }
    
    // 设置L1缓存
    setL1(key, value) {
        if (this.l1Cache.size >= this.maxL1Size) {
            // 移除最旧的项
            const firstKey = this.l1Cache.keys().next().value;
            this.l1Cache.delete(firstKey);
        }
        
        this.l1Cache.set(key, value);
    }
    
    // 设置L2缓存
    setL2(key, value, ttl) {
        if (this.l2Cache.size >= this.maxL2Size) {
            // 移除最旧的项
            const firstKey = this.l2Cache.keys().next().value;
            this.l2Cache.delete(firstKey);
        }
        
        this.l2Cache.set(key, {
            value,
            expires: Date.now() + ttl
        });
    }
    
    // 删除数据
    delete(key) {
        this.l1Cache.delete(key);
        this.l2Cache.delete(key);
    }
    
    // 清理过期数据
    cleanup() {
        const now = Date.now();
        
        for (const [key, item] of this.l2Cache) {
            if (item.expires < now) {
                this.l2Cache.delete(key);
            }
        }
    }
    
    // 获取统计信息
    getStats() {
        const total = this.stats.l1Hits + this.stats.l2Hits + this.stats.misses;
        return {
            ...this.stats,
            total,
            hitRate: total > 0 ? ((this.stats.l1Hits + this.stats.l2Hits) / total).toFixed(2) : 0,
            l1HitRate: total > 0 ? (this.stats.l1Hits / total).toFixed(2) : 0,
            l2HitRate: total > 0 ? (this.stats.l2Hits / total).toFixed(2) : 0,
            l1Size: this.l1Cache.size,
            l2Size: this.l2Cache.size
        };
    }
}

class SmartCacheStrategy {
    constructor() {
        this.cache = new Map();
        this.accessPatterns = new Map();
        this.predictionModel = new Map();
        this.stats = {
            hits: 0,
            misses: 0,
            predictions: 0,
            correctPredictions: 0
        };
    }
    
    // 获取数据
    async get(key) {
        const now = Date.now();
        
        // 更新访问模式
        this.updateAccessPattern(key, now);
        
        // 检查缓存
        if (this.cache.has(key)) {
            this.stats.hits++;
            this.updateAccessTime(key, now);
            return this.cache.get(key).value;
        }
        
        this.stats.misses++;
        
        // 预测下一个可能访问的键
        const predictedKeys = this.predictNextAccess(key);
        this.preloadPredictedKeys(predictedKeys);
        
        return null;
    }
    
    // 设置数据
    async set(key, value, ttl = 300000) {
        const now = Date.now();
        
        this.cache.set(key, {
            value,
            createdAt: now,
            expires: now + ttl,
            accessCount: 0,
            lastAccess: now
        });
        
        this.updateAccessPattern(key, now);
    }
    
    // 更新访问模式
    updateAccessPattern(key, timestamp) {
        if (!this.accessPatterns.has(key)) {
            this.accessPatterns.set(key, []);
        }
        
        const pattern = this.accessPatterns.get(key);
        pattern.push(timestamp);
        
        // 保持最近100次访问记录
        if (pattern.length > 100) {
            pattern.shift();
        }
    }
    
    // 更新访问时间
    updateAccessTime(key, timestamp) {
        const item = this.cache.get(key);
        if (item) {
            item.accessCount++;
            item.lastAccess = timestamp;
        }
    }
    
    // 预测下一个访问
    predictNextAccess(currentKey) {
        const predictions = [];
        const currentPattern = this.accessPatterns.get(currentKey) || [];
        
        // 基于访问模式预测
        for (const [key, pattern] of this.accessPatterns) {
            if (key === currentKey) continue;
            
            const similarity = this.calculateSimilarity(currentPattern, pattern);
            if (similarity > 0.7) {
                predictions.push({ key, similarity });
            }
        }
        
        // 按相似度排序
        predictions.sort((a, b) => b.similarity - a.similarity);
        
        return predictions.slice(0, 3).map(p => p.key);
    }
    
    // 计算模式相似度
    calculateSimilarity(pattern1, pattern2) {
        if (pattern1.length === 0 || pattern2.length === 0) return 0;
        
        // 简化的相似度计算
        const avg1 = pattern1.reduce((a, b) => a + b, 0) / pattern1.length;
        const avg2 = pattern2.reduce((a, b) => a + b, 0) / pattern2.length;
        
        const diff = Math.abs(avg1 - avg2);
        return Math.max(0, 1 - diff / (avg1 + avg2));
    }
    
    // 预加载预测的键
    async preloadPredictedKeys(keys) {
        for (const key of keys) {
            if (!this.cache.has(key)) {
                // 这里可以触发预加载逻辑
                console.log(`Preloading predicted key: ${key}`);
            }
        }
    }
    
    // 清理过期数据
    cleanup() {
        const now = Date.now();
        const toDelete = [];
        
        for (const [key, item] of this.cache) {
            if (item.expires < now) {
                toDelete.push(key);
            }
        }
        
        for (const key of toDelete) {
            this.cache.delete(key);
        }
        
        return toDelete.length;
    }
    
    // 获取统计信息
    getStats() {
        const total = this.stats.hits + this.stats.misses;
        return {
            ...this.stats,
            total,
            hitRate: total > 0 ? (this.stats.hits / total).toFixed(2) : 0,
            predictionAccuracy: this.stats.predictions > 0 ? 
                (this.stats.correctPredictions / this.stats.predictions).toFixed(2) : 0,
            cacheSize: this.cache.size,
            accessPatterns: this.accessPatterns.size
        };
    }
}

// 1. 实时性能监控器
class RealTimePerformanceMonitor {
    constructor() {
        this.metrics = new Map();
        this.alerts = [];
        this.thresholds = {
            memoryUsage: 80,
            longTask: 50,
            layoutShift: 0.1,
            slowResource: 1000
        };
        this.init();
    }
    
    init() {
        this.startMemoryMonitoring();
        this.startLongTaskMonitoring();
        this.startLayoutShiftMonitoring();
        this.startResourceMonitoring();
        this.startUserInteractionMonitoring();
    }
    
    // 内存监控
    startMemoryMonitoring() {
        if (performance.memory) {
            setInterval(() => {
                const memory = performance.memory;
                const usage = (memory.usedJSHeapSize / memory.jsHeapSizeLimit) * 100;
                
                this.metrics.set('memory', {
                    used: memory.usedJSHeapSize,
                    total: memory.totalJSHeapSize,
                    limit: memory.jsHeapSizeLimit,
                    usage: usage
                });
                
                if (usage > this.thresholds.memoryUsage) {
                    this.addAlert('high-memory-usage', {
                        usage: usage,
                        used: memory.usedJSHeapSize,
                        limit: memory.jsHeapSizeLimit
                    });
                }
            }, 5000);
        }
    }
    
    // 长任务监控
    startLongTaskMonitoring() {
        if ('PerformanceObserver' in window) {
            const observer = new PerformanceObserver((list) => {
                for (const entry of list.getEntries()) {
                    if (entry.duration > this.thresholds.longTask) {
                        this.addAlert('long-task', {
                            duration: entry.duration,
                            startTime: entry.startTime
                        });
                    }
                }
            });
            
            observer.observe({ entryTypes: ['longtask'] });
        }
    }
    
    // 布局偏移监控
    startLayoutShiftMonitoring() {
        if ('PerformanceObserver' in window) {
            const observer = new PerformanceObserver((list) => {
                for (const entry of list.getEntries()) {
                    if (!entry.hadRecentInput && entry.value > this.thresholds.layoutShift) {
                        this.addAlert('layout-shift', {
                            value: entry.value,
                            startTime: entry.startTime
                        });
                    }
                }
            });
            
            observer.observe({ entryTypes: ['layout-shift'] });
        }
    }
    
    // 资源监控
    startResourceMonitoring() {
        if ('PerformanceObserver' in window) {
            const observer = new PerformanceObserver((list) => {
                for (const entry of list.getEntries()) {
                    if (entry.duration > this.thresholds.slowResource) {
                        this.addAlert('slow-resource', {
                            name: entry.name,
                            duration: entry.duration,
                            size: entry.transferSize
                        });
                    }
                }
            });
            
            observer.observe({ entryTypes: ['resource'] });
        }
    }
    
    // 用户交互监控
    startUserInteractionMonitoring() {
        // 监控点击响应时间
        document.addEventListener('click', (e) => {
            const startTime = performance.now();
            
            requestAnimationFrame(() => {
                const endTime = performance.now();
                const responseTime = endTime - startTime;
                
                if (responseTime > 100) {
                    this.addAlert('slow-click-response', {
                        responseTime: responseTime,
                        target: e.target.tagName
                    });
                }
            });
        });
        
        // 监控滚动性能
        let scrollTimeout;
        window.addEventListener('scroll', () => {
            clearTimeout(scrollTimeout);
            scrollTimeout = setTimeout(() => {
                const scrollTime = performance.now();
                this.metrics.set('scroll', {
                    scrollY: window.scrollY,
                    timestamp: scrollTime
                });
            }, 100);
        });
    }
    
    // 添加警告
    addAlert(type, data) {
        const alert = {
            type,
            data,
            timestamp: Date.now(),
            severity: this.getSeverity(type, data)
        };
        
        this.alerts.push(alert);
        
        // 限制警告数量
        if (this.alerts.length > 100) {
            this.alerts.shift();
        }
        
        // 触发警告事件
        this.dispatchEvent(new CustomEvent('performance-alert', { detail: alert }));
    }
    
    // 获取严重程度
    getSeverity(type, data) {
        const severityMap = {
            'high-memory-usage': 'high',
            'long-task': 'medium',
            'layout-shift': 'medium',
            'slow-resource': 'low',
            'slow-click-response': 'medium'
        };
        
        return severityMap[type] || 'low';
    }
    
    // 获取指标
    getMetrics() {
        return Object.fromEntries(this.metrics);
    }
    
    // 获取警告
    getAlerts(severity = null) {
        if (severity) {
            return this.alerts.filter(alert => alert.severity === severity);
        }
        return this.alerts;
    }
    
    // 生成报告
    generateReport() {
        const metrics = this.getMetrics();
        const alerts = this.getAlerts();
        
        return {
            timestamp: Date.now(),
            metrics,
            alerts,
            summary: {
                totalAlerts: alerts.length,
                highSeverity: alerts.filter(a => a.severity === 'high').length,
                mediumSeverity: alerts.filter(a => a.severity === 'medium').length,
                lowSeverity: alerts.filter(a => a.severity === 'low').length
            }
        };
    }
}

// 2. 性能基准测试
class PerformanceBenchmark {
    constructor() {
        this.benchmarks = new Map();
        this.results = new Map();
    }
    
    // 运行基准测试
    async runBenchmark(name, testFunction, iterations = 100) {
        console.log(`Running benchmark: ${name}`);
        
        const times = [];
        
        for (let i = 0; i < iterations; i++) {
            const startTime = performance.now();
            await testFunction();
            const endTime = performance.now();
            
            times.push(endTime - startTime);
        }
        
        const result = this.calculateStats(times);
        this.results.set(name, result);
        
        console.log(`Benchmark ${name} completed:`, result);
        return result;
    }
    
    // 计算统计信息
    calculateStats(times) {
        const sorted = times.slice().sort((a, b) => a - b);
        const sum = times.reduce((a, b) => a + b, 0);
        
        return {
            iterations: times.length,
            total: sum,
            average: sum / times.length,
            median: sorted[Math.floor(sorted.length / 2)],
            min: Math.min(...times),
            max: Math.max(...times),
            p95: sorted[Math.floor(sorted.length * 0.95)],
            p99: sorted[Math.floor(sorted.length * 0.99)],
            standardDeviation: this.calculateStandardDeviation(times, sum / times.length)
        };
    }
    
    // 计算标准差
    calculateStandardDeviation(times, mean) {
        const variance = times.reduce((sum, time) => sum + Math.pow(time - mean, 2), 0) / times.length;
        return Math.sqrt(variance);
    }
    
    // 比较基准测试
    compareBenchmarks(name1, name2) {
        const result1 = this.results.get(name1);
        const result2 = this.results.get(name2);
        
        if (!result1 || !result2) {
            throw new Error('Benchmark results not found');
        }
        
        const comparison = {
            name1,
            name2,
            averageDiff: result2.average - result1.average,
            averageDiffPercent: ((result2.average - result1.average) / result1.average) * 100,
            medianDiff: result2.median - result1.median,
            medianDiffPercent: ((result2.median - result1.median) / result1.median) * 100,
            faster: result1.average < result2.average ? name1 : name2,
            slower: result1.average > result2.average ? name1 : name2
        };
        
        return comparison;
    }
    
    // 生成报告
    generateReport() {
        const report = {
            timestamp: Date.now(),
            benchmarks: Object.fromEntries(this.results),
            summary: {
                totalBenchmarks: this.results.size,
                fastest: this.getFastestBenchmark(),
                slowest: this.getSlowestBenchmark()
            }
        };
        
        return report;
    }
    
    // 获取最快的基准测试
    getFastestBenchmark() {
        let fastest = null;
        let fastestTime = Infinity;
        
        for (const [name, result] of this.results) {
            if (result.average < fastestTime) {
                fastestTime = result.average;
                fastest = name;
            }
        }
        
        return fastest;
    }
    
    // 获取最慢的基准测试
    getSlowestBenchmark() {
        let slowest = null;
        let slowestTime = 0;
        
        for (const [name, result] of this.results) {
            if (result.average > slowestTime) {
                slowestTime = result.average;
                slowest = name;
            }
        }
        
        return slowest;
    }
}

};

const testFunction2 = async () => {
