# 项目3-推荐系统

## 项目概述

本项目是一个基于深度学习的推荐系统，能够为用户推荐个性化的内容。项目使用协同过滤、内容过滤和深度学习技术，实现了从数据预处理到推荐服务的完整流程，适合学习推荐系统的基本原理和实践方法。

## 项目目标

- **主要目标**：构建一个能够准确推荐个性化内容的推荐系统
- **学习目标**：
  - 掌握推荐系统的基本原理和算法
  - 学习协同过滤和内容过滤技术
  - 理解深度学习在推荐系统中的应用
  - 掌握推荐系统的评估方法和指标
  - 学习推荐系统的工程实践和部署

class RecommendationDataset(Dataset):
    def __init__(self, ratings_data, negative_sampling=False, num_negatives=4):
        self.ratings_data = ratings_data
        self.negative_sampling = negative_sampling
        self.num_negatives = num_negatives
        
        if negative_sampling:
            self._generate_negative_samples()
    
    def _generate_negative_samples(self):
        """
        生成负样本
        """
        # 获取所有用户和物品
        all_users = set(self.ratings_data['user_id'].unique())
        all_items = set(self.ratings_data['item_id'].unique())
        
        # 为每个用户生成负样本
        negative_samples = []
        for user_id in all_users:
            # 获取用户已评分的物品
            user_items = set(self.ratings_data[self.ratings_data['user_id'] == user_id]['item_id'])
            
            # 生成负样本
            negative_items = all_items - user_items
            if len(negative_items) > 0:
                num_negatives = min(self.num_negatives, len(negative_items))
                selected_negatives = random.sample(list(negative_items), num_negatives)
                
                for item_id in selected_negatives:
                    negative_samples.append([user_id, item_id, 0])  # 0表示负样本
        
        # 合并正样本和负样本
        positive_samples = self.ratings_data[['user_id', 'item_id', 'rating']].values
        all_samples = np.vstack([positive_samples, np.array(negative_samples)])
        
        self.samples = pd.DataFrame(all_samples, columns=['user_id', 'item_id', 'rating'])
    
    def __len__(self):
        if self.negative_sampling:
            return len(self.samples)
        return len(self.ratings_data)
    
    def __getitem__(self, idx):
        if self.negative_sampling:
            sample = self.samples.iloc[idx]
        else:
            sample = self.ratings_data.iloc[idx]
        
        return {
            'user_id': torch.tensor(sample['user_id'], dtype=torch.long),
            'item_id': torch.tensor(sample['item_id'], dtype=torch.long),
            'rating': torch.tensor(sample['rating'], dtype=torch.float)
        }

class CollaborativeFiltering:
    def __init__(self, ratings_data):
        self.ratings_data = ratings_data
        self.user_item_matrix = None
        self.user_similarity = None
        self.item_similarity = None
        
    def build_user_item_matrix(self):
        """
        构建用户-物品矩阵
        """
        self.user_item_matrix = self.ratings_data.pivot_table(
            index='user_id', columns='item_id', values='rating', fill_value=0
        )
        return self.user_item_matrix
    
    def calculate_user_similarity(self, method='cosine'):
        """
        计算用户相似度
        """
        if self.user_item_matrix is None:
            self.build_user_item_matrix()
        
        if method == 'cosine':
            self.user_similarity = cosine_similarity(self.user_item_matrix)
        elif method == 'pearson':
            self.user_similarity = self.user_item_matrix.T.corr()
        
        return self.user_similarity
    
    def calculate_item_similarity(self, method='cosine'):
        """
        计算物品相似度
        """
        if self.user_item_matrix is None:
            self.build_user_item_matrix()
        
        if method == 'cosine':
            self.item_similarity = cosine_similarity(self.user_item_matrix.T)
        elif method == 'pearson':
            self.item_similarity = self.user_item_matrix.corr()
        
        return self.item_similarity
    
    def user_based_recommend(self, user_id, n_recommendations=10, n_neighbors=50):
        """
        基于用户的协同过滤推荐
        """
        if self.user_similarity is None:
            self.calculate_user_similarity()
        
        # 获取用户已评分的物品
        user_ratings = self.user_item_matrix.loc[user_id]
        rated_items = user_ratings[user_ratings > 0].index
        
        # 获取相似用户
        user_similarities = self.user_similarity[user_id]
        similar_users = user_similarities.nlargest(n_neighbors + 1).index[1:]  # 排除自己
        
        # 计算推荐分数
        recommendations = {}
        for item_id in self.user_item_matrix.columns:
            if item_id not in rated_items:
                score = 0
                similarity_sum = 0
                
                for similar_user in similar_users:
                    if self.user_item_matrix.loc[similar_user, item_id] > 0:
                        similarity = user_similarities[similar_user]
                        score += similarity * self.user_item_matrix.loc[similar_user, item_id]
                        similarity_sum += abs(similarity)
                
                if similarity_sum > 0:
                    recommendations[item_id] = score / similarity_sum
        
        # 返回推荐结果
        return sorted(recommendations.items(), key=lambda x: x[1], reverse=True)[:n_recommendations]
    
    def item_based_recommend(self, user_id, n_recommendations=10, n_neighbors=50):
        """
        基于物品的协同过滤推荐
        """
        if self.item_similarity is None:
            self.calculate_item_similarity()
        
        # 获取用户已评分的物品
        user_ratings = self.user_item_matrix.loc[user_id]
        rated_items = user_ratings[user_ratings > 0].index
        
        # 计算推荐分数
        recommendations = {}
        for item_id in self.user_item_matrix.columns:
            if item_id not in rated_items:
                score = 0
                similarity_sum = 0
                
                for rated_item in rated_items:
                    similarity = self.item_similarity.loc[item_id, rated_item]
                    score += similarity * user_ratings[rated_item]
                    similarity_sum += abs(similarity)
                
                if similarity_sum > 0:
                    recommendations[item_id] = score / similarity_sum
        
        # 返回推荐结果
        return sorted(recommendations.items(), key=lambda x: x[1], reverse=True)[:n_recommendations]

class MatrixFactorization:
    def __init__(self, num_users, num_items, num_factors=50, learning_rate=0.01, 
                 regularization=0.01):
        self.num_users = num_users
        self.num_items = num_items
        self.num_factors = num_factors
        self.learning_rate = learning_rate
        self.regularization = regularization
        
        # 初始化用户和物品因子矩阵
        self.user_factors = np.random.normal(0, 0.1, (num_users, num_factors))
        self.item_factors = np.random.normal(0, 0.1, (num_items, num_factors))
        
        # 初始化偏置项
        self.user_bias = np.zeros(num_users)
        self.item_bias = np.zeros(num_items)
        self.global_bias = 0.0
    
    def fit(self, ratings_data, epochs=100, verbose=True):
        """
        训练矩阵分解模型
        """
        for epoch in range(epochs):
            total_error = 0
            
            for _, row in ratings_data.iterrows():
                user_id = int(row['user_id'])
                item_id = int(row['item_id'])
                rating = row['rating']
                
                # 预测评分
                predicted = self.predict(user_id, item_id)
                
                # 计算误差
                error = rating - predicted
                total_error += error ** 2
                
                # 更新参数
                self._update_factors(user_id, item_id, error)
            
            # 计算平均误差
            avg_error = total_error / len(ratings_data)
            
            if verbose and epoch % 10 == 0:
                print(f'Epoch {epoch}: Average Error = {avg_error:.4f}')
    
    def _update_factors(self, user_id, item_id, error):
        """
        更新因子矩阵
        """
        # 更新用户因子
        user_factor_old = self.user_factors[user_id].copy()
        self.user_factors[user_id] += self.learning_rate * (
            error * self.item_factors[item_id] - self.regularization * self.user_factors[user_id]
        )
        
        # 更新物品因子
        self.item_factors[item_id] += self.learning_rate * (
            error * user_factor_old - self.regularization * self.item_factors[item_id]
        )
        
        # 更新偏置项
        self.user_bias[user_id] += self.learning_rate * (error - self.regularization * self.user_bias[user_id])
        self.item_bias[item_id] += self.learning_rate * (error - self.regularization * self.item_bias[item_id])
        self.global_bias += self.learning_rate * error
    
    def predict(self, user_id, item_id):
        """
        预测评分
        """
        return (self.global_bias + 
                self.user_bias[user_id] + 
                self.item_bias[item_id] + 
                np.dot(self.user_factors[user_id], self.item_factors[item_id]))
    
    def recommend(self, user_id, n_recommendations=10):
        """
        为用户推荐物品
        """
        # 获取用户已评分的物品
        user_ratings = self.ratings_data[self.ratings_data['user_id'] == user_id]['item_id'].values
        all_items = set(range(self.num_items))
        unrated_items = all_items - set(user_ratings)
        
        # 计算推荐分数
        recommendations = []
        for item_id in unrated_items:
            score = self.predict(user_id, item_id)
            recommendations.append((item_id, score))
        
        # 返回推荐结果
        return sorted(recommendations, key=lambda x: x[1], reverse=True)[:n_recommendations]

class NeuralCollaborativeFiltering(nn.Module):
    def __init__(self, num_users, num_items, embedding_dim=50, hidden_dims=[64, 32], dropout=0.2):
        super(NeuralCollaborativeFiltering, self).__init__()
        
        self.num_users = num_users
        self.num_items = num_items
        self.embedding_dim = embedding_dim
        
        # 用户和物品嵌入层
        self.user_embedding = nn.Embedding(num_users, embedding_dim)
        self.item_embedding = nn.Embedding(num_items, embedding_dim)
        
        # 多层感知机
        layers = []
        input_dim = embedding_dim * 2
        
        for hidden_dim in hidden_dims:
            layers.append(nn.Linear(input_dim, hidden_dim))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(dropout))
            input_dim = hidden_dim
        
        layers.append(nn.Linear(input_dim, 1))
        layers.append(nn.Sigmoid())
        
        self.mlp = nn.Sequential(*layers)
        
        # 初始化权重
        self._init_weights()
    
    def _init_weights(self):
        """
        初始化权重
        """
        nn.init.normal_(self.user_embedding.weight, std=0.01)
        nn.init.normal_(self.item_embedding.weight, std=0.01)
        
        for layer in self.mlp:
            if isinstance(layer, nn.Linear):
                nn.init.normal_(layer.weight, std=0.01)
                nn.init.zeros_(layer.bias)
    
    def forward(self, user_ids, item_ids):
        # 获取嵌入
        user_emb = self.user_embedding(user_ids)
        item_emb = self.item_embedding(item_ids)
        
        # 拼接嵌入
        concat_emb = torch.cat([user_emb, item_emb], dim=1)
        
        # 通过MLP
        output = self.mlp(concat_emb)
        
        return output.squeeze()
    
    def get_user_embeddings(self, user_ids):
        """
        获取用户嵌入
        """
        return self.user_embedding(user_ids)
    
    def get_item_embeddings(self, item_ids):
        """
        获取物品嵌入
        """
        return self.item_embedding(item_ids)

class DeepFM(nn.Module):
    def __init__(self, num_users, num_items, embedding_dim=50, hidden_dims=[64, 32], dropout=0.2):
        super(DeepFM, self).__init__()
        
        self.num_users = num_users
        self.num_items = num_items
        self.embedding_dim = embedding_dim
        
        # 用户和物品嵌入层
        self.user_embedding = nn.Embedding(num_users, embedding_dim)
        self.item_embedding = nn.Embedding(num_items, embedding_dim)
        
        # FM部分
        self.fm_linear = nn.Linear(2, 1)  # 用户和物品的线性部分
        
        # Deep部分
        layers = []
        input_dim = embedding_dim * 2
        
        for hidden_dim in hidden_dims:
            layers.append(nn.Linear(input_dim, hidden_dim))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(dropout))
            input_dim = hidden_dim
        
        layers.append(nn.Linear(input_dim, 1))
        
        self.deep_mlp = nn.Sequential(*layers)
        
        # 初始化权重
        self._init_weights()
    
    def _init_weights(self):
        """
        初始化权重
        """
        nn.init.normal_(self.user_embedding.weight, std=0.01)
        nn.init.normal_(self.item_embedding.weight, std=0.01)
        
        for layer in self.deep_mlp:
            if isinstance(layer, nn.Linear):
                nn.init.normal_(layer.weight, std=0.01)
                nn.init.zeros_(layer.bias)
    
    def forward(self, user_ids, item_ids):
        # 获取嵌入
        user_emb = self.user_embedding(user_ids)
        item_emb = self.item_embedding(item_ids)
        
        # FM部分
        fm_input = torch.stack([user_ids.float(), item_ids.float()], dim=1)
        fm_output = self.fm_linear(fm_input)
        
        # Deep部分
        concat_emb = torch.cat([user_emb, item_emb], dim=1)
        deep_output = self.deep_mlp(concat_emb)
        
        # 组合输出
        output = fm_output + deep_output
        
        return torch.sigmoid(output.squeeze())
    
    def get_user_embeddings(self, user_ids):
        """
        获取用户嵌入
        """
        return self.user_embedding(user_ids)
    
    def get_item_embeddings(self, item_ids):
        """
        获取物品嵌入
        """
        return self.item_embedding(item_ids)

class RecommendationTrainer:
    def __init__(self, model, train_loader, test_loader, device):
        self.model = model
        self.train_loader = train_loader
        self.test_loader = test_loader
        self.device = device
        
        # 移动模型到设备
        self.model.to(device)
        
        # 定义损失函数和优化器
        self.criterion = nn.BCELoss()
        self.optimizer = optim.Adam(model.parameters(), lr=0.001)
        self.scheduler = optim.lr_scheduler.ReduceLROnPlateau(
            self.optimizer, mode='min', factor=0.5, patience=3
        )
        
        # 训练历史
        self.train_losses = []
        self.test_losses = []
        self.train_aucs = []
        self.test_aucs = []
    
    def train_epoch(self):
        """
        训练一个epoch
        """
        self.model.train()
        running_loss = 0.0
        
        for batch_idx, batch in enumerate(self.train_loader):
            user_ids = batch['user_id'].to(self.device)
            item_ids = batch['item_id'].to(self.device)
            ratings = batch['rating'].to(self.device)
            
            # 清零梯度
            self.optimizer.zero_grad()
            
            # 前向传播
            outputs = self.model(user_ids, item_ids)
            loss = self.criterion(outputs, ratings)
            
            # 反向传播
            loss.backward()
            self.optimizer.step()
            
            # 统计
            running_loss += loss.item()
            
            # 打印进度
            if batch_idx % 100 == 0:
                print(f'Batch {batch_idx}/{len(self.train_loader)}, Loss: {loss.item():.4f}')
        
        epoch_loss = running_loss / len(self.train_loader)
        self.train_losses.append(epoch_loss)
        
        return epoch_loss
    
    def test_epoch(self):
        """
        测试一个epoch
        """
        self.model.eval()
        test_loss = 0.0
        
        with torch.no_grad():
            for batch in self.test_loader:
                user_ids = batch['user_id'].to(self.device)
                item_ids = batch['item_id'].to(self.device)
                ratings = batch['rating'].to(self.device)
                
                outputs = self.model(user_ids, item_ids)
                loss = self.criterion(outputs, ratings)
                
                test_loss += loss.item()
        
        test_loss /= len(self.test_loader)
        self.test_losses.append(test_loss)
        
        return test_loss
    
    def train(self, epochs=50):
        """
        训练模型
        """
        print(f"开始训练，共{epochs}个epoch")
        print(f"设备: {self.device}")
        
        best_loss = float('inf')
        
        for epoch in range(epochs):
            print(f'\nEpoch {epoch+1}/{epochs}')
            print('-' * 50)
            
            # 训练
            train_loss = self.train_epoch()
            
            # 测试
            test_loss = self.test_epoch()
            
            # 更新学习率
            self.scheduler.step(test_loss)
            
            # 打印结果
            print(f'Train Loss: {train_loss:.4f}')
            print(f'Test Loss: {test_loss:.4f}')
            print(f'Learning Rate: {self.optimizer.param_groups[0]["lr"]:.6f}')
            
            # 保存最佳模型
            if test_loss < best_loss:
                best_loss = test_loss
                torch.save(self.model.state_dict(), 'best_recommendation_model.pth')
                print(f'新的最佳损失: {best_loss:.4f}')
        
        print(f'\n训练完成！最佳测试损失: {best_loss:.4f}')
        
        return {
            'best_loss': best_loss,

class RecommendationEvaluator:
    def __init__(self, model, test_data, device):
        self.model = model
        self.test_data = test_data
        self.device = device
        
    def evaluate_model(self):
        """
        评估模型性能
        """
        self.model.eval()
        
        # 计算RMSE和MAE
        rmse, mae = self._calculate_rmse_mae()
        
        # 计算NDCG
        ndcg = self._calculate_ndcg()
        
        # 计算Hit Rate
        hit_rate = self._calculate_hit_rate()
        
        # 计算Precision@K和Recall@K
        precision_k, recall_k = self._calculate_precision_recall_k()
        
        return {
            'rmse': rmse,
            'mae': mae,
            'ndcg': ndcg,
            'hit_rate': hit_rate,
            'precision_k': precision_k,
            'recall_k': recall_k
        }
    
    def _calculate_rmse_mae(self):
        """
        计算RMSE和MAE
        """
        predictions = []
        actuals = []
        
        with torch.no_grad():
            for _, row in self.test_data.iterrows():
                user_id = torch.tensor([row['user_id']], dtype=torch.long).to(self.device)
                item_id = torch.tensor([row['item_id']], dtype=torch.long).to(self.device)
                rating = row['rating']
                
                prediction = self.model(user_id, item_id).item()
                predictions.append(prediction)
                actuals.append(rating)
        
        rmse = np.sqrt(mean_squared_error(actuals, predictions))
        mae = mean_absolute_error(actuals, predictions)
        
        return rmse, mae
    
    def _calculate_ndcg(self, k=10):
        """
        计算NDCG@K
        """
        ndcg_scores = []
        
        # 按用户分组
        user_groups = self.test_data.groupby('user_id')
        
        for user_id, user_data in user_groups:
            if len(user_data) < k:
                continue
            
            # 获取用户的所有物品评分
            user_items = user_data['item_id'].values
            user_ratings = user_data['rating'].values
            
            # 预测评分
            predictions = []
            for item_id in user_items:
                user_tensor = torch.tensor([user_id], dtype=torch.long).to(self.device)
                item_tensor = torch.tensor([item_id], dtype=torch.long).to(self.device)
                
                with torch.no_grad():
                    prediction = self.model(user_tensor, item_tensor).item()
                    predictions.append(prediction)
            
            # 计算NDCG
            ndcg = self._ndcg_at_k(user_ratings, predictions, k)
            ndcg_scores.append(ndcg)
        
        return np.mean(ndcg_scores)
    
    def _ndcg_at_k(self, actual, predicted, k):
        """
        计算NDCG@K
        """
        # 按预测分数排序
        sorted_indices = np.argsort(predicted)[::-1]
        
        # 计算DCG
        dcg = 0
        for i in range(min(k, len(actual))):
            dcg += actual[sorted_indices[i]] / np.log2(i + 2)
        
        # 计算IDCG
        idcg = 0
        sorted_actual = np.sort(actual)[::-1]
        for i in range(min(k, len(actual))):
            idcg += sorted_actual[i] / np.log2(i + 2)
        
        return dcg / idcg if idcg > 0 else 0
    
    def _calculate_hit_rate(self, k=10):
        """
        计算Hit Rate@K
        """
        hit_count = 0
        total_users = 0
        
        # 按用户分组
        user_groups = self.test_data.groupby('user_id')
        
        for user_id, user_data in user_groups:
            if len(user_data) < k:
                continue
            
            # 获取用户的高评分物品
            high_rated_items = user_data[user_data['rating'] >= 4]['item_id'].values
            
            if len(high_rated_items) == 0:
                continue
            
            # 预测评分
            predictions = []
            for item_id in user_data['item_id'].values:
                user_tensor = torch.tensor([user_id], dtype=torch.long).to(self.device)
                item_tensor = torch.tensor([item_id], dtype=torch.long).to(self.device)
                
                with torch.no_grad():
                    prediction = self.model(user_tensor, item_tensor).item()
                    predictions.append(prediction)
            
            # 获取Top-K推荐
            top_k_indices = np.argsort(predictions)[::-1][:k]
            top_k_items = user_data['item_id'].iloc[top_k_indices].values
            
            # 检查是否有命中
            if len(set(top_k_items) & set(high_rated_items)) > 0:
                hit_count += 1
            
            total_users += 1
        
        return hit_count / total_users if total_users > 0 else 0
    
    def _calculate_precision_recall_k(self, k=10):
        """
        计算Precision@K和Recall@K
        """
        precision_scores = []
        recall_scores = []
        
        # 按用户分组
        user_groups = self.test_data.groupby('user_id')
        
        for user_id, user_data in user_groups:
            if len(user_data) < k:
                continue
            
            # 获取用户的高评分物品
            high_rated_items = user_data[user_data['rating'] >= 4]['item_id'].values
            
            if len(high_rated_items) == 0:
                continue
            
            # 预测评分
            predictions = []
            for item_id in user_data['item_id'].values:
                user_tensor = torch.tensor([user_id], dtype=torch.long).to(self.device)
                item_tensor = torch.tensor([item_id], dtype=torch.long).to(self.device)
                
                with torch.no_grad():
                    prediction = self.model(user_tensor, item_tensor).item()
                    predictions.append(prediction)
            
            # 获取Top-K推荐
            top_k_indices = np.argsort(predictions)[::-1][:k]
            top_k_items = user_data['item_id'].iloc[top_k_indices].values
            
            # 计算Precision和Recall
            hits = len(set(top_k_items) & set(high_rated_items))
            precision = hits / k
            recall = hits / len(high_rated_items)
            
            precision_scores.append(precision)
            recall_scores.append(recall)
        
        return np.mean(precision_scores), np.mean(recall_scores)
    
    def plot_evaluation_results(self, results):
        """
        绘制评估结果
        """
        fig, axes = plt.subplots(2, 2, figsize=(15, 10))
        
        # RMSE和MAE
        axes[0, 0].bar(['RMSE', 'MAE'], [results['rmse'], results['mae']])
        axes[0, 0].set_title('RMSE and MAE')
        axes[0, 0].set_ylabel('Error')
        
        # NDCG
        axes[0, 1].bar(['NDCG@10'], [results['ndcg']])
        axes[0, 1].set_title('NDCG@10')
        axes[0, 1].set_ylabel('Score')
        axes[0, 1].set_ylim(0, 1)
        
        # Hit Rate
        axes[1, 0].bar(['Hit Rate@10'], [results['hit_rate']])
        axes[1, 0].set_title('Hit Rate@10')

        plt.tight_layout()
