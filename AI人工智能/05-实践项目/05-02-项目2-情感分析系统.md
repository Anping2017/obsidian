# 项目2-情感分析系统

## 项目概述

本项目是一个基于深度学习的文本情感分析系统，能够自动识别文本中的情感倾向（正面、负面、中性）。项目使用自然语言处理技术和深度学习模型，实现了从文本预处理到情感预测的完整流程，适合学习NLP和情感分析的基本概念。

## 项目目标

- **主要目标**：构建一个能够准确识别文本情感倾向的深度学习模型
- **学习目标**：
  - 掌握文本预处理和特征提取技术
  - 学习词向量和文本表示方法
  - 理解RNN、LSTM、Transformer等序列模型
  - 掌握情感分析任务的评估方法
  - 学习模型部署和API服务开发

## 数据集介绍

### 情感分析数据集
- **IMDB电影评论数据集**：50,000条电影评论，正面和负面各25,000条
- **Twitter情感数据集**：包含推文和对应的情感标签
- **Amazon产品评论数据集**：产品评论和星级评分
- **自定义数据集**：根据具体应用场景收集的文本数据

import pandas as pd

class TextPreprocessor:
    def __init__(self, language='english'):
        self.language = language
        self.stop_words = set(stopwords.words(language))
        self.lemmatizer = WordNetLemmatizer()
        
    def clean_text(self, text):
        """
        清理文本
        """
        # 转换为小写
        text = text.lower()
        
        # 移除HTML标签
        text = re.sub(r'<[^>]+>', '', text)
        
        # 移除URL
        text = re.sub(r'http\S+|www\S+|https\S+', '', text)
        
        # 移除邮箱
        text = re.sub(r'\S+@\S+', '', text)
        
        # 移除特殊字符和数字
        text = re.sub(r'[^a-zA-Z\s]', '', text)
        
        # 移除多余空格
        text = re.sub(r'\s+', ' ', text).strip()
        
        return text
    
    def tokenize_text(self, text):
        """
        分词
        """
        return word_tokenize(text)
    
    def remove_stopwords(self, tokens):
        """
        移除停用词
        """
        return [token for token in tokens if token not in self.stop_words]
    
    def lemmatize_tokens(self, tokens):
        """
        词形还原
        """
        return [self.lemmatizer.lemmatize(token) for token in tokens]
    
    def preprocess_text(self, text):
        """
        完整的文本预处理流程
        """
        # 清理文本
        cleaned_text = self.clean_text(text)
        
        # 分词
        tokens = self.tokenize_text(cleaned_text)
        
        # 移除停用词
        tokens = self.remove_stopwords(tokens)
        
        # 词形还原
        tokens = self.lemmatize_tokens(tokens)
        
        return tokens

class SentimentDataset(Dataset):
    def __init__(self, texts, labels, tokenizer, max_length=128):
        self.texts = texts
        self.labels = labels
        self.tokenizer = tokenizer
        self.max_length = max_length
        
    def __len__(self):
        return len(self.texts)
    
    def __getitem__(self, idx):
        text = self.texts[idx]
        label = self.labels[idx]
        
        # 编码文本
        encoding = self.tokenizer(
            text,
            truncation=True,
            padding='max_length',
            max_length=self.max_length,
            return_tensors='pt'
        )
        
        return {
            'input_ids': encoding['input_ids'].flatten(),
            'attention_mask': encoding['attention_mask'].flatten(),
            'labels': torch.tensor(label, dtype=torch.long)
        }

class DataLoader:
    def __init__(self, data_path=None):
        self.data_path = data_path
        self.preprocessor = TextPreprocessor()
        
    def load_imdb_data(self):
        """
        加载IMDB数据集
        """
        try:
            # 尝试从本地加载
            train_data = pd.read_csv('imdb_train.csv')
            test_data = pd.read_csv('imdb_test.csv')
        except:
            # 如果没有本地数据，创建示例数据
            print("未找到本地数据，创建示例数据")
            train_data = self._create_sample_data(1000)
            test_data = self._create_sample_data(200)
        
        return train_data, test_data
    
    def _create_sample_data(self, num_samples):
        """
        创建示例数据
        """
        positive_texts = [
            "This movie is absolutely fantastic! I loved every minute of it.",
            "Great acting and amazing storyline. Highly recommended!",
            "One of the best films I've ever seen. Perfect!",
            "Excellent movie with brilliant performances.",
            "Outstanding! This is a masterpiece of cinema."
        ]
        
        negative_texts = [
            "Terrible movie. Complete waste of time.",
            "Awful acting and boring plot. Don't watch this.",
            "This is the worst film I've ever seen.",
            "Horrible movie with no redeeming qualities.",
            "Disappointing and poorly made. Avoid at all costs."
        ]
        
        texts = []
        labels = []
        
        for _ in range(num_samples // 2):
            texts.append(np.random.choice(positive_texts))
            labels.append(1)  # 正面
            
            texts.append(np.random.choice(negative_texts))
            labels.append(0)  # 负面
        
        return pd.DataFrame({'text': texts, 'label': labels})
    
    def preprocess_dataset(self, data):
        """
        预处理数据集
        """
        processed_texts = []
        
        for text in data['text']:
            processed_tokens = self.preprocessor.preprocess_text(text)
            processed_texts.append(' '.join(processed_tokens))
        
        return processed_texts, data['label'].values
    
    def create_data_loaders(self, train_texts, train_labels, test_texts, test_labels, 
                           tokenizer, batch_size=32, max_length=128):
        """
        创建数据加载器
        """
        train_dataset = SentimentDataset(train_texts, train_labels, tokenizer, max_length)
        test_dataset = SentimentDataset(test_texts, test_labels, tokenizer, max_length)
        
        train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
        test_loader = DataLoader(test_dataset, batch_size=batch_size, shuffle=False)
        
        return train_loader, test_loader

class SentimentRNN(nn.Module):
    def __init__(self, vocab_size, embedding_dim, hidden_dim, output_dim, 
                 num_layers, dropout=0.5):
        super(SentimentRNN, self).__init__()
        
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        self.rnn = nn.RNN(embedding_dim, hidden_dim, num_layers, 
                          batch_first=True, dropout=dropout)
        self.fc = nn.Linear(hidden_dim, output_dim)
        self.dropout = nn.Dropout(dropout)
        
    def forward(self, x):
        # 嵌入层
        embedded = self.embedding(x)
        
        # RNN层
        rnn_out, hidden = self.rnn(embedded)
        
        # 取最后一个时间步的输出
        last_output = rnn_out[:, -1, :]
        
        # 全连接层
        output = self.fc(self.dropout(last_output))
        
        return output

class SentimentLSTM(nn.Module):
    def __init__(self, vocab_size, embedding_dim, hidden_dim, output_dim, 
                 num_layers, dropout=0.5):
        super(SentimentLSTM, self).__init__()
        
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        self.lstm = nn.LSTM(embedding_dim, hidden_dim, num_layers, 
                           batch_first=True, dropout=dropout, bidirectional=True)
        self.fc = nn.Linear(hidden_dim * 2, output_dim)  # 双向LSTM
        self.dropout = nn.Dropout(dropout)
        
    def forward(self, x):
        # 嵌入层
        embedded = self.embedding(x)
        
        # LSTM层
        lstm_out, (hidden, cell) = self.lstm(embedded)
        
        # 取最后一个时间步的输出
        last_output = lstm_out[:, -1, :]
        
        # 全连接层
        output = self.fc(self.dropout(last_output))
        
        return output

class SentimentGRU(nn.Module):
    def __init__(self, vocab_size, embedding_dim, hidden_dim, output_dim, 
                 num_layers, dropout=0.5):
        super(SentimentGRU, self).__init__()
        
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        self.gru = nn.GRU(embedding_dim, hidden_dim, num_layers, 
                         batch_first=True, dropout=dropout, bidirectional=True)
        self.fc = nn.Linear(hidden_dim * 2, output_dim)
        self.dropout = nn.Dropout(dropout)
        
    def forward(self, x):
        # 嵌入层
        embedded = self.embedding(x)
        
        # GRU层
        gru_out, hidden = self.gru(embedded)
        
        # 取最后一个时间步的输出
        last_output = gru_out[:, -1, :]
        
        # 全连接层
        output = self.fc(self.dropout(last_output))
        
        return output

class SentimentCNN(nn.Module):
    def __init__(self, vocab_size, embedding_dim, num_filters, filter_sizes, 
                 output_dim, dropout=0.5):
        super(SentimentCNN, self).__init__()
        
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        
        # 多个卷积层
        self.convs = nn.ModuleList([
            nn.Conv2d(1, num_filters, (filter_size, embedding_dim))
            for filter_size in filter_sizes
        ])
        
        self.fc = nn.Linear(len(filter_sizes) * num_filters, output_dim)
        self.dropout = nn.Dropout(dropout)
        
    def forward(self, x):
        # 嵌入层
        embedded = self.embedding(x).unsqueeze(1)  # 添加通道维度
        
        # 卷积层
        conv_outputs = []
        for conv in self.convs:
            conv_out = F.relu(conv(embedded))
            pooled = F.max_pool2d(conv_out, (conv_out.size(2), 1))
            conv_outputs.append(pooled.squeeze(3))
        
        # 拼接所有卷积输出
        concatenated = torch.cat(conv_outputs, dim=1)
        
        # 全连接层
        output = self.fc(self.dropout(concatenated))
        
        return output

class SentimentTransformer(nn.Module):
    def __init__(self, vocab_size, embedding_dim, num_heads, num_layers, 
                 output_dim, max_length=128, dropout=0.1):
        super(SentimentTransformer, self).__init__()
        
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        self.pos_encoding = self._create_positional_encoding(max_length, embedding_dim)
        
        # Transformer编码器
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=embedding_dim,
            nhead=num_heads,
            dropout=dropout,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers)
        
        self.fc = nn.Linear(embedding_dim, output_dim)
        self.dropout = nn.Dropout(dropout)
        
    def _create_positional_encoding(self, max_length, embedding_dim):
        """
        创建位置编码
        """
        pe = torch.zeros(max_length, embedding_dim)
        position = torch.arange(0, max_length).unsqueeze(1).float()
        
        div_term = torch.exp(torch.arange(0, embedding_dim, 2).float() * 
                           -(np.log(10000.0) / embedding_dim))
        
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        
        return pe.unsqueeze(0)
    
    def forward(self, x):
        seq_len = x.size(1)
        
        # 嵌入层
        embedded = self.embedding(x) * np.sqrt(self.embedding.embedding_dim)
        
        # 添加位置编码
        embedded += self.pos_encoding[:, :seq_len, :].to(x.device)
        
        # Transformer编码器
        transformer_out = self.transformer(embedded)
        
        # 全局平均池化
        pooled = transformer_out.mean(dim=1)
        
        # 全连接层
        output = self.fc(self.dropout(pooled))
        
        return output

class SentimentTrainer:
    def __init__(self, model, train_loader, test_loader, device, num_classes=2):
        self.model = model
        self.train_loader = train_loader
        self.test_loader = test_loader
        self.device = device
        self.num_classes = num_classes
        
        # 移动模型到设备
        self.model.to(device)
        
        # 定义损失函数和优化器
        self.criterion = nn.CrossEntropyLoss()
        self.optimizer = optim.Adam(model.parameters(), lr=0.001)
        self.scheduler = optim.lr_scheduler.ReduceLROnPlateau(
            self.optimizer, mode='min', factor=0.5, patience=3
        )
        
        # 训练历史
        self.train_losses = []
        self.train_accuracies = []
        self.test_losses = []
        self.test_accuracies = []
    
    def train_epoch(self):
        """
        训练一个epoch
        """
        self.model.train()
        running_loss = 0.0
        correct = 0
        total = 0
        
        for batch_idx, batch in enumerate(self.train_loader):
            input_ids = batch['input_ids'].to(self.device)
            attention_mask = batch['attention_mask'].to(self.device)
            labels = batch['labels'].to(self.device)
            
            # 清零梯度
            self.optimizer.zero_grad()
            
            # 前向传播
            outputs = self.model(input_ids)
            loss = self.criterion(outputs, labels)
            
            # 反向传播
            loss.backward()
            self.optimizer.step()
            
            # 统计
            running_loss += loss.item()
            _, predicted = outputs.max(1)
            total += labels.size(0)
            correct += predicted.eq(labels).sum().item()
            
            # 打印进度
            if batch_idx % 100 == 0:
                print(f'Batch {batch_idx}/{len(self.train_loader)}, '
                      f'Loss: {loss.item():.4f}, '
                      f'Acc: {100.*correct/total:.2f}%')
        
        epoch_loss = running_loss / len(self.train_loader)
        epoch_acc = 100. * correct / total
        
        self.train_losses.append(epoch_loss)
        self.train_accuracies.append(epoch_acc)
        
        return epoch_loss, epoch_acc
    
    def test_epoch(self):
        """
        测试一个epoch
        """
        self.model.eval()
        test_loss = 0
        correct = 0
        total = 0
        
        with torch.no_grad():
            for batch in self.test_loader:
                input_ids = batch['input_ids'].to(self.device)
                attention_mask = batch['attention_mask'].to(self.device)
                labels = batch['labels'].to(self.device)
                
                outputs = self.model(input_ids)
                test_loss += self.criterion(outputs, labels).item()
                
                _, predicted = outputs.max(1)
                total += labels.size(0)
                correct += predicted.eq(labels).sum().item()
        
        test_loss /= len(self.test_loader)
        test_acc = 100. * correct / total
        
        self.test_losses.append(test_loss)
        self.test_accuracies.append(test_acc)
        
        return test_loss, test_acc
    
    def train(self, epochs=10):
        """
        训练模型
        """
        print(f"开始训练，共{epochs}个epoch")
        print(f"设备: {self.device}")
        
        best_acc = 0
        
        for epoch in range(epochs):
            print(f'\nEpoch {epoch+1}/{epochs}')
            print('-' * 50)
            
            # 训练
            train_loss, train_acc = self.train_epoch()
            
            # 测试
            test_loss, test_acc = self.test_epoch()
            
            # 更新学习率
            self.scheduler.step(test_loss)
            
            # 打印结果
            print(f'Train Loss: {train_loss:.4f}, Train Acc: {train_acc:.2f}%')
            print(f'Test Loss: {test_loss:.4f}, Test Acc: {test_acc:.2f}%')
            print(f'Learning Rate: {self.optimizer.param_groups[0]["lr"]:.6f}')
            
            # 保存最佳模型
            if test_acc > best_acc:
                best_acc = test_acc
                torch.save(self.model.state_dict(), 'best_sentiment_model.pth')
                print(f'新的最佳准确率: {best_acc:.2f}%')
        
        print(f'\n训练完成！最佳测试准确率: {best_acc:.2f}%')
        
        return {
            'best_acc': best_acc,
            'train_losses': self.train_losses,
            'train_accuracies': self.train_accuracies,
            'test_losses': self.test_losses,
            'test_accuracies': self.test_accuracies
        }
    
    def plot_training_history(self):
        """
        绘制训练历史
        """
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15, 5))
        
        # 损失曲线
        ax1.plot(self.train_losses, label='Train Loss')
        ax1.plot(self.test_losses, label='Test Loss')
        ax1.set_title('Training and Test Loss')
        ax1.set_xlabel('Epoch')
        ax1.set_ylabel('Loss')
        ax1.legend()
        ax1.grid(True)
        
        # 准确率曲线
        ax2.plot(self.train_accuracies, label='Train Accuracy')
        ax2.plot(self.test_accuracies, label='Test Accuracy')
        ax2.set_title('Training and Test Accuracy')
        ax2.set_xlabel('Epoch')
        ax2.set_ylabel('Accuracy (%)')
        ax2.legend()
        ax2.grid(True)
        
        plt.tight_layout()
        plt.show()

class SentimentEvaluator:
    def __init__(self, model, test_loader, device, class_names=['Negative', 'Positive']):
        self.model = model
        self.test_loader = test_loader
        self.device = device
        self.class_names = class_names
        
    def evaluate_model(self):
        """
        评估模型性能
        """
        self.model.eval()
        all_predictions = []
        all_targets = []
        all_probabilities = []
        
        with torch.no_grad():
            for batch in self.test_loader:
                input_ids = batch['input_ids'].to(self.device)
                attention_mask = batch['attention_mask'].to(self.device)
                labels = batch['labels'].to(self.device)
                
                outputs = self.model(input_ids)
                probabilities = F.softmax(outputs, dim=1)
                _, predicted = outputs.max(1)
                
                all_predictions.extend(predicted.cpu().numpy())
                all_targets.extend(labels.cpu().numpy())
                all_probabilities.extend(probabilities.cpu().numpy())
        
        return np.array(all_predictions), np.array(all_targets), np.array(all_probabilities)
    
    def calculate_metrics(self, predictions, targets):
        """
        计算评估指标
        """
        accuracy = accuracy_score(targets, predictions)
        
        # 计算精确率、召回率、F1分数
        precision = precision_score(targets, predictions, average='weighted')
        recall = recall_score(targets, predictions, average='weighted')
        f1 = f1_score(targets, predictions, average='weighted')
        
        # 计算每个类别的指标
        precision_per_class = precision_score(targets, predictions, average=None)
        recall_per_class = recall_score(targets, predictions, average=None)
        f1_per_class = f1_score(targets, predictions, average=None)
        
        return {
            'accuracy': accuracy,
            'precision': precision,
            'recall': recall,
            'f1_score': f1,
            'precision_per_class': precision_per_class,
            'recall_per_class': recall_per_class,
            'f1_per_class': f1_per_class
        }
    
    def plot_confusion_matrix(self, predictions, targets):
        """
        绘制混淆矩阵
        """
        cm = confusion_matrix(targets, predictions)
        
        plt.figure(figsize=(8, 6))
        sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
                    xticklabels=self.class_names,
                    yticklabels=self.class_names)
        plt.title('Confusion Matrix')
        plt.xlabel('Predicted')
        plt.ylabel('Actual')
        plt.show()
        
        return cm
    
    def plot_class_metrics(self, predictions, targets):
        """
        绘制各类别指标
        """
        metrics = self.calculate_metrics(predictions, targets)
        
        fig, axes = plt.subplots(1, 3, figsize=(15, 5))
        
        # 精确率
        axes[0].bar(self.class_names, metrics['precision_per_class'])
        axes[0].set_title('Precision by Class')
        axes[0].set_ylabel('Precision')
        axes[0].set_ylim(0, 1)
        
        # 召回率
        axes[1].bar(self.class_names, metrics['recall_per_class'])
        axes[1].set_title('Recall by Class')
        axes[1].set_ylabel('Recall')
        axes[1].set_ylim(0, 1)
        
        # F1分数
        axes[2].bar(self.class_names, metrics['f1_per_class'])
        axes[2].set_title('F1-Score by Class')
        axes[2].set_ylabel('F1-Score')
        axes[2].set_ylim(0, 1)
        
        plt.tight_layout()
        plt.show()
    
    def analyze_predictions(self, predictions, targets, probabilities, test_texts):
        """
        分析预测结果
        """
        # 找出错误分类的样本
        misclassified_indices = np.where(predictions != targets)[0]
        
        print(f"错误分类样本数量: {len(misclassified_indices)}")
        print(f"错误分类率: {len(misclassified_indices) / len(predictions) * 100:.2f}%")
        
        # 显示一些错误分类的样本
        if len(misclassified_indices) > 0:
            print("\n错误分类样本示例:")
            for i in range(min(5, len(misclassified_indices))):
                idx = misclassified_indices[i]
                print(f"\n样本 {i+1}:")
                print(f"文本: {test_texts[idx]}")
                print(f"真实标签: {self.class_names[targets[idx]]}")
                print(f"预测标签: {self.class_names[predictions[idx]]}")
                print(f"预测概率: {probabilities[idx]}")
        
        # 分析置信度分布
        max_probabilities = np.max(probabilities, axis=1)
        
        plt.figure(figsize=(10, 6))
        plt.hist(max_probabilities, bins=50, alpha=0.7, edgecolor='black')
        plt.title('Distribution of Prediction Confidence')
        plt.xlabel('Maximum Probability')
        plt.ylabel('Frequency')
        plt.show()
        
        return {
            'misclassified_count': len(misclassified_indices),
            'misclassification_rate': len(misclassified_indices) / len(predictions),
            'confidence_distribution': max_probabilities
        }
