```python
            },
            '深度学习': {
                '掌握程度': '75%',
                '核心概念': ['神经网络', 'CNN', 'RNN', 'Transformer'],
                '应用能力': '能够设计和实现深度学习模型'
            },
            '工程实践': {
                '掌握程度': '70%',
                '核心概念': ['软件开发', '系统设计', '部署运维', '项目管理'],
                '应用能力': '能够完成端到端的AI项目开发'
            }
        }
    
    def calculate_overall_score(self):
        """计算总体掌握程度"""
        scores = [float(area['掌握程度'].rstrip('%')) for area in self.knowledge_areas.values()]
        return sum(scores) / len(scores)
    
    def get_improvement_areas(self):
        """获取需要改进的领域"""
        improvement_areas = []
        for area, details in self.knowledge_areas.items():
            if float(details['掌握程度'].rstrip('%')) < 80:
                improvement_areas.append(area)
        return improvement_areas

# 使用示例
knowledge_assessment = KnowledgeAssessment()
overall_score = knowledge_assessment.calculate_overall_score()
improvement_areas = knowledge_assessment.get_improvement_areas()

print(f"总体掌握程度: {overall_score:.1f}%")
print(f"需要改进的领域: {', '.join(improvement_areas)}")
```
