## 🔍 调试基础

### 1. 调试工具

**Python内置调试器：**
```python
import pdb

def debug_function():
    x = 10
    y = 20
    pdb.set_trace()  # 设置断点
    result = x + y
    return result

# 调试命令
# n (next): 执行下一行
# s (step): 进入函数内部
# c (continue): 继续执行
# l (list): 显示当前代码
# p variable: 打印变量值
# pp variable: 美化打印变量
# h (help): 显示帮助
# q (quit): 退出调试器
```

**条件日志：**
```python
import logging

logger = logging.getLogger(__name__)

def process_data(data):
    logger.info(f"开始处理数据，数据量: {len(data)}")
    
    for i, item in enumerate(data):
        if logger.isEnabledFor(logging.DEBUG):
            logger.debug(f"处理第{i}个元素: {item}")
        
        # 处理逻辑
        result = process_item(item)
        
        if result is None:
            logger.warning(f"第{i}个元素处理失败")
        
    logger.info("数据处理完成")
```

### 2. 运行时错误

**异常处理：**
```python
def safe_divide(a, b):
    try:
        result = a / b
        return result
    except ZeroDivisionError:
        print("错误：除数不能为零")
        return None
    except TypeError:
        print("错误：参数类型不正确")
        return None
    except Exception as e:
        print(f"未知错误：{e}")
        return None

# 使用
result = safe_divide(10, 0)
result = safe_divide(10, "2")
```

def calculate_average(numbers):
    assert isinstance(numbers, list), "参数必须是列表"
    assert len(numbers) > 0, "列表不能为空"
    assert all(isinstance(n, (int, float)) for n in numbers), "列表元素必须是数字"

### 2. 远程调试

**远程调试设置：**
```python
import pdb
import socket

class RemoteDebugger:
    def __init__(self, host='localhost', port=4444):
        self.host = host
        self.port = port
        self.socket = None
    
    def start(self):
        """启动远程调试服务器"""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.bind((self.host, self.port))
        self.socket.listen(1)
        print(f"远程调试服务器启动在 {self.host}:{self.port}")
    
    def connect(self):
        """连接到远程调试服务器"""
        conn, addr = self.socket.accept()
        print(f"连接到 {addr}")
        return conn

# 使用远程调试
debugger = RemoteDebugger()
debugger.start()
conn = debugger.connect()
```

### 1. 机器学习调试

**模型调试：**
```python
import numpy as np
import matplotlib.pyplot as plt

def debug_model_training(model, X_train, y_train, X_val, y_val):
    """调试模型训练过程"""
    train_losses = []
    val_losses = []
    
    for epoch in range(100):
        # 训练
        model.fit(X_train, y_train)
        
        # 计算损失
        train_loss = model.evaluate(X_train, y_train)
        val_loss = model.evaluate(X_val, y_val)
        
        train_losses.append(train_loss)
        val_losses.append(val_loss)
        
        # 每10个epoch打印一次
        if epoch % 10 == 0:
            print(f"Epoch {epoch}: Train Loss = {train_loss:.4f}, Val Loss = {val_loss:.4f}")
    
    # 可视化训练过程
    plt.plot(train_losses, label='Training Loss')
    plt.plot(val_losses, label='Validation Loss')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    plt.legend()
    plt.title('Training Progress')
    plt.show()
```

**数据调试：**
```python
def debug_data_pipeline(data):
    """调试数据处理管道"""
    print(f"原始数据形状: {data.shape}")
    print(f"数据类型: {data.dtypes}")
    print(f"缺失值: {data.isnull().sum()}")
    print(f"数据范围: {data.describe()}")
    
    # 检查异常值
    for column in data.columns:
        if data[column].dtype in ['int64', 'float64']:
            Q1 = data[column].quantile(0.25)
            Q3 = data[column].quantile(0.75)
            IQR = Q3 - Q1
            outliers = data[(data[column] < Q1 - 1.5 * IQR) | 
                           (data[column] > Q3 + 1.5 * IQR)]
            if len(outliers) > 0:
                print(f"列 {column} 有 {len(outliers)} 个异常值")
    
    # 可视化数据分布
    data.hist(figsize=(12, 8))
    plt.tight_layout()
    plt.show()
```

### 2. 深度学习调试

**梯度调试：**
```python
import torch
import torch.nn as nn

def debug_gradients(model, loss):
    """调试梯度"""
    # 计算梯度
    loss.backward()
    
    # 检查梯度
    for name, param in model.named_parameters():
        if param.grad is not None:
            grad_norm = param.grad.norm().item()
            print(f"参数 {name}: 梯度范数 = {grad_norm:.6f}")
            
            # 检查梯度爆炸
            if grad_norm > 100:
                print(f"警告: 参数 {name} 梯度爆炸!")
            
            # 检查梯度消失
            if grad_norm < 1e-6:
                print(f"警告: 参数 {name} 梯度消失!")
        else:
            print(f"参数 {name}: 没有梯度")

# 使用示例
model = nn.Linear(10, 1)
x = torch.randn(32, 10)
y = torch.randn(32, 1)
loss = nn.MSELoss()(model(x), y)
debug_gradients(model, loss)
```

**激活值调试：**
```python
def debug_activations(model, x):
    """调试激活值"""
    activations = {}
    
    def hook_fn(name):
        def hook(module, input, output):
            activations[name] = output.detach()
        return hook
    
    # 注册钩子
    hooks = []
    for name, module in model.named_modules():
        if isinstance(module, (nn.ReLU, nn.Sigmoid, nn.Tanh)):
            hook = module.register_forward_hook(hook_fn(name))
            hooks.append(hook)
    
    # 前向传播
    output = model(x)
    
    # 分析激活值
    for name, activation in activations.items():
        mean_val = activation.mean().item()
        std_val = activation.std().item()
        print(f"激活层 {name}: 均值 = {mean_val:.4f}, 标准差 = {std_val:.4f}")
        
        # 检查激活值饱和
        if isinstance(model.get_submodule(name.split('.')[0]), nn.Sigmoid):
            if mean_val > 0.9 or mean_val < 0.1:
                print(f"警告: 激活层 {name} 可能饱和!")
    
    # 清理钩子
    for hook in hooks:
        hook.remove()
    
    return output
```
