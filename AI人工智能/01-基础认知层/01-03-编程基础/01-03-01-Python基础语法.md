# Python基础语法

## 🐍 Python核心语法

### 1. 基础数据类型

**数值类型：**
```python
# 整数
a = 42
b = 0x2A  # 十六进制
c = 0o52  # 八进制

# 浮点数
d = 3.14
e = 1.23e-4  # 科学计数法

# 复数
f = 3 + 4j
g = complex(3, 4)
```

**字符串类型：**
```python
# 字符串创建
s1 = "Hello World"
s2 = 'Python'
s3 = """多行字符串
可以跨行"""

# 字符串操作
s = "Python"
print(s[0])      # P
print(s[-1])      # n
print(s[1:4])     # yth
print(len(s))     # 6
print(s.upper())  # PYTHON
```

**布尔类型：**
```python
# 布尔值
True, False

# 布尔运算
a = True
b = False
print(a and b)  # False
print(a or b)   # True
print(not a)    # False
```

### 2. 数据结构

**列表（List）：**
```python
# 列表创建
lst = [1, 2, 3, 4, 5]
lst2 = list(range(5))

# 列表操作
lst.append(6)        # 添加元素
lst.insert(0, 0)     # 插入元素
lst.remove(3)        # 删除元素
lst.pop()            # 弹出元素
lst.sort()           # 排序
lst.reverse()        # 反转

# 列表推导式
squares = [x**2 for x in range(10)]
evens = [x for x in range(20) if x % 2 == 0]
```

**字典（Dictionary）：**
```python
# 字典创建
d = {'name': 'Alice', 'age': 25}
d2 = dict(name='Bob', age=30)

# 字典操作
d['city'] = 'Beijing'    # 添加键值对
d.get('name')            # 获取值
d.keys()                 # 获取所有键
d.values()               # 获取所有值
d.items()                # 获取所有键值对

# 字典推导式
squares_dict = {x: x**2 for x in range(5)}
```

**元组（Tuple）：**
```python
# 元组创建
t = (1, 2, 3)
t2 = tuple([1, 2, 3])

# 元组操作
print(t[0])      # 1
print(t.count(2)) # 1
print(t.index(3)) # 2

# 元组解包
a, b, c = t
```

**集合（Set）：**
```python
# 集合创建
s = {1, 2, 3, 4}
s2 = set([1, 2, 3, 4])

# 集合操作
s.add(5)           # 添加元素
s.remove(1)        # 删除元素
s.union(s2)        # 并集
s.intersection(s2) # 交集
s.difference(s2)  # 差集
```

## 🔄 控制流语句

### 1. 条件语句

```python
# if-elif-else
age = 20
if age < 18:
    print("未成年")
elif age < 60:
    print("成年人")
else:
    print("老年人")

# 三元运算符
status = "成年" if age >= 18 else "未成年"
```

### 2. 循环语句

**for循环：**
```python
# 基本for循环
for i in range(5):
    print(i)

# 遍历列表
fruits = ['apple', 'banana', 'orange']
for fruit in fruits:
    print(fruit)

# 带索引的遍历
for i, fruit in enumerate(fruits):
    print(f"{i}: {fruit}")

# 字典遍历
person = {'name': 'Alice', 'age': 25}
for key, value in person.items():
    print(f"{key}: {value}")
```

**while循环：**
```python
# 基本while循环
count = 0
while count < 5:
    print(count)
    count += 1

# 带条件的循环
while True:
    user_input = input("输入quit退出: ")
    if user_input == 'quit':
        break
    print(f"你输入了: {user_input}")
```

## 🏗️ 函数定义

### 1. 基本函数

```python
# 函数定义
def greet(name):
    return f"Hello, {name}!"

# 函数调用
message = greet("Alice")
print(message)

# 默认参数
def greet_with_title(name, title="Mr."):
    return f"Hello, {title} {name}!"

# 关键字参数
greet_with_title("Smith", title="Dr.")
```

### 2. 高级函数特性

**可变参数：**
```python
# *args - 位置参数
def sum_all(*args):
    return sum(args)

print(sum_all(1, 2, 3, 4, 5))

# **kwargs - 关键字参数
def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

print_info(name="Alice", age=25, city="Beijing")
```

**lambda函数：**
```python
# lambda表达式
square = lambda x: x**2
print(square(5))  # 25

# 与map结合使用
numbers = [1, 2, 3, 4, 5]
squares = list(map(lambda x: x**2, numbers))
```

**装饰器：**
```python
# 装饰器定义
def timer(func):
    def wrapper(*args, **kwargs):
        import time
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"函数执行时间: {end - start:.2f}秒")
        return result
    return wrapper

# 使用装饰器
@timer
def slow_function():
    import time
    time.sleep(1)
    return "完成"

slow_function()
```

## 🏛️ 面向对象编程

### 1. 类定义

```python
# 基本类定义
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    def greet(self):
        return f"Hello, I'm {self.name}, {self.age} years old"
    
    def have_birthday(self):
        self.age += 1

# 创建对象
person = Person("Alice", 25)
print(person.greet())
person.have_birthday()
print(person.age)
```

### 2. 继承和多态

```python
# 继承
class Student(Person):
    def __init__(self, name, age, student_id):
        super().__init__(name, age)
        self.student_id = student_id
    
    def greet(self):
        return f"Hello, I'm {self.name}, student ID: {self.student_id}"

# 多态
def introduce(person):
    print(person.greet())

alice = Person("Alice", 25)
bob = Student("Bob", 20, "12345")

introduce(alice)  # Hello, I'm Alice, 25 years old
introduce(bob)    # Hello, I'm Bob, student ID: 12345
```

## 🔧 异常处理

### 1. 基本异常处理

```python
# try-except
try:
    result = 10 / 0
except ZeroDivisionError:
    print("除零错误")

# 多个异常
try:
    value = int("abc")
except ValueError:
    print("值错误")
except Exception as e:
    print(f"其他错误: {e}")

# finally块
try:
    file = open("data.txt", "r")
    data = file.read()
except FileNotFoundError:
    print("文件不存在")
finally:
    if 'file' in locals():
        file.close()
```

### 2. 自定义异常

```python
# 自定义异常类
class CustomError(Exception):
    def __init__(self, message):
        self.message = message
        super().__init__(self.message)

# 抛出异常
def validate_age(age):
    if age < 0:
        raise CustomError("年龄不能为负数")
    return age

try:
    validate_age(-5)
except CustomError as e:
    print(f"自定义错误: {e.message}")
```

### 2. 生成器

```python
# 生成器函数
def fibonacci(n):
    a, b = 0, 1
    for _ in range(n):
        yield a
        a, b = b, a + b

# 使用生成器
for num in fibonacci(10):
    print(num)

# 生成器表达式
squares_gen = (x**2 for x in range(10))
```

### 3. 上下文管理器

```python
# 使用with语句
with open('data.txt', 'r') as file:
    content = file.read()

# 自定义上下文管理器
class Timer:
    def __enter__(self):
        import time
        self.start = time.time()
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        import time
        print(f"执行时间: {time.time() - self.start:.2f}秒")

with Timer():
    # 执行一些操作
    sum(range(1000000))
```
