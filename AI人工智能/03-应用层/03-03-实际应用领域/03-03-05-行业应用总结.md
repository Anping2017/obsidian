import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.feature_selection import SelectKBest, f_classif

class DataDrivenPattern:
    def __init__(self):
        self.scaler = StandardScaler()
        self.encoder = LabelEncoder()
        self.feature_selector = SelectKBest(f_classif, k=10)
        
    def extract_features(self, data, target_column=None):
        """
        提取特征
        """
        # 数值特征
        numerical_features = data.select_dtypes(include=[np.number])
        
        # 分类特征
        categorical_features = data.select_dtypes(include=['object'])
        
        # 时间特征
        time_features = self._extract_time_features(data)
        
        # 组合特征
        combined_features = self._create_combined_features(data)
        
        # 特征选择
        if target_column:
            selected_features = self._select_features(
                combined_features, data[target_column])
        else:
            selected_features = combined_features
        
        return selected_features
    
    def _extract_time_features(self, data):
        """
        提取时间特征
        """
        time_features = pd.DataFrame()
        
        # 查找时间列
        time_columns = data.select_dtypes(include=['datetime64']).columns
        
        for col in time_columns:
            time_features[f'{col}_year'] = data[col].dt.year
            time_features[f'{col}_month'] = data[col].dt.month
            time_features[f'{col}_day'] = data[col].dt.day
            time_features[f'{col}_hour'] = data[col].dt.hour
            time_features[f'{col}_dayofweek'] = data[col].dt.dayofweek
            time_features[f'{col}_is_weekend'] = data[col].dt.dayofweek.isin([5, 6])
        
        return time_features
    
    def _create_combined_features(self, data):
        """
        创建组合特征
        """
        combined_features = pd.DataFrame()
        
        # 数值特征组合
        numerical_cols = data.select_dtypes(include=[np.number]).columns
        
        for i, col1 in enumerate(numerical_cols):
            for col2 in numerical_cols[i+1:]:
                combined_features[f'{col1}_x_{col2}'] = data[col1] * data[col2]
                combined_features[f'{col1}_div_{col2}'] = data[col1] / (data[col2] + 1e-8)
        
        return combined_features
    
    def _select_features(self, features, target):
        """
        特征选择
        """
        # 处理缺失值
        features = features.fillna(features.median())
        
        # 特征选择
        selected_features = self.feature_selector.fit_transform(features, target)
        
        # 获取选中的特征名
        selected_feature_names = features.columns[self.feature_selector.get_support()]
        
        return pd.DataFrame(selected_features, columns=selected_feature_names)
```

from sklearn.model_selection import GridSearchCV, cross_val_score
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.svm import SVC
from sklearn.linear_model import LogisticRegression
import xgboost as xgb

class ModelDrivenPattern:
    def __init__(self):
        self.models = {
            'random_forest': RandomForestClassifier(),
            'gradient_boosting': GradientBoostingClassifier(),
            'svm': SVC(),
            'logistic_regression': LogisticRegression(),
            'xgboost': xgb.XGBClassifier()
        }
        
    def select_best_model(self, X, y, cv=5):
        """
        选择最佳模型
        """
        model_scores = {}
        
        for name, model in self.models.items():
            # 交叉验证
            scores = cross_val_score(model, X, y, cv=cv, scoring='accuracy')
            model_scores[name] = {
                'mean_score': scores.mean(),
                'std_score': scores.std(),
                'scores': scores
            }
        
        # 选择最佳模型
        best_model_name = max(model_scores.keys(), 
                            key=lambda x: model_scores[x]['mean_score'])
        
        return best_model_name, model_scores
    
    def optimize_model(self, model, X, y, param_grid, cv=5):
        """
        优化模型参数
        """
        grid_search = GridSearchCV(
            model, param_grid, cv=cv, scoring='accuracy', n_jobs=-1
        )
        
        grid_search.fit(X, y)
        
        return grid_search.best_estimator_, grid_search.best_params_
    
    def ensemble_models(self, X, y, models):
        """
        集成模型
        """
        from sklearn.ensemble import VotingClassifier
        
        ensemble = VotingClassifier(
            estimators=list(models.items()),
            voting='soft'
        )
        
        ensemble.fit(X, y)
        
        return ensemble
```

#### 深度学习架构
```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

class EndToEndLearningPattern:
    def __init__(self, input_size, hidden_sizes, output_size):
        self.model = self._build_model(input_size, hidden_sizes, output_size)
        self.optimizer = optim.Adam(self.model.parameters(), lr=0.001)
        self.criterion = nn.CrossEntropyLoss()
        
    def _build_model(self, input_size, hidden_sizes, output_size):
        """
        构建深度学习模型
        """
        layers = []
        
        # 输入层
        layers.append(nn.Linear(input_size, hidden_sizes[0]))
        layers.append(nn.ReLU())
        layers.append(nn.Dropout(0.2))
        
        # 隐藏层
        for i in range(len(hidden_sizes) - 1):
            layers.append(nn.Linear(hidden_sizes[i], hidden_sizes[i+1]))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(0.2))
        
        # 输出层
        layers.append(nn.Linear(hidden_sizes[-1], output_size))
        
        return nn.Sequential(*layers)
    
    def train(self, X, y, epochs=100, batch_size=32):
        """
        训练模型
        """
        # 转换为张量
        X_tensor = torch.FloatTensor(X)
        y_tensor = torch.LongTensor(y)
        
        # 创建数据加载器
        dataset = TensorDataset(X_tensor, y_tensor)
        dataloader = DataLoader(dataset, batch_size=batch_size, shuffle=True)
        
        # 训练循环
        self.model.train()
        for epoch in range(epochs):
            total_loss = 0
            
            for batch_X, batch_y in dataloader:
                self.optimizer.zero_grad()
                
                # 前向传播
                outputs = self.model(batch_X)
                loss = self.criterion(outputs, batch_y)
                
                # 反向传播
                loss.backward()
                self.optimizer.step()
                
                total_loss += loss.item()
            
            if epoch % 10 == 0:
                print(f'Epoch {epoch}, Loss: {total_loss/len(dataloader):.4f}')
    
    def predict(self, X):
        """
        预测
        """
        self.model.eval()
        
        with torch.no_grad():
            X_tensor = torch.FloatTensor(X)
            outputs = self.model(X_tensor)
            predictions = torch.argmax(outputs, dim=1)
            
            return predictions.numpy()
```

                years, timeline['start_year'], 
                timeline['peak_year'], timeline['maturity_year']
            )
            
            # 绘制曲线
            color = colors[timeline['current_status']]
            ax.plot(years, maturity_curve, label=tech, color=color, linewidth=2)
            
            # 标记关键点
            ax.scatter(timeline['peak_year'], 0.8, color=color, s=100, zorder=5)

        ax.grid(True, alpha=0.3)

        plt.show()
    
    def _calculate_maturity_curve(self, years, start_year, peak_year, maturity_year):
        """
        计算技术成熟度曲线
        """
        curve = np.zeros_like(years, dtype=float)
        
        for i, year in enumerate(years):
            if year < start_year:
                curve[i] = 0
            elif year < peak_year:
                # 增长阶段
                curve[i] = 0.8 * (year - start_year) / (peak_year - start_year)

## 总结

AI技术在各行业的应用呈现出以下特点：

1. **技术成熟度差异**：不同技术在不同行业的成熟度存在显著差异
2. **应用模式多样**：从自动化到生成，从预测到优化，应用模式丰富多样
3. **跨行业迁移**：技术在不同行业间的迁移和适配成为重要趋势
4. **社会影响深远**：AI技术对社会、经济、环境等方面产生深远影响

未来AI技术发展将更加注重：
- 技术伦理和可解释性
- 跨行业应用和迁移
- 社会影响和治理
- 人机协作和融合

通过系统化的分析和规划，可以更好地把握AI技术在各行业的应用机会和挑战，实现技术价值和社会价值的统一。
