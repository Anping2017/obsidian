# 团队协作与项目管理

## 概述

团队协作与项目管理是AI项目成功的关键因素，涉及团队组织、沟通协作、项目管理、质量控制等多个方面。通过系统化的团队协作和项目管理实践，可以提高项目效率、降低风险、确保项目成功交付。本章将介绍团队协作与项目管理的核心概念、实践方法和最佳实践。

## 核心概念

### 1. 团队协作定义
- **目标**：建立高效的团队协作机制，提升项目执行效率
- **核心要素**：沟通、协调、分工、责任、信任
- **协作模式**：敏捷开发、DevOps、跨职能团队
- **关键挑战**：沟通障碍、角色冲突、技术债务、时间管理

### 2. 项目管理框架
- **项目生命周期**：启动、规划、执行、监控、收尾
- **管理维度**：范围、时间、成本、质量、风险、资源
- **方法论**：瀑布模型、敏捷开发、Scrum、看板
- **工具支持**：项目管理软件、版本控制、CI/CD

        """
        获取团队技能矩阵
        """
        skills_matrix = {}
        for member in self.team_members:
            for skill in member.skills:
                if skill not in skills_matrix:
                    skills_matrix[skill] = []
                skills_matrix[skill].append(member.name)
        return skills_matrix
    
    def analyze_team_gaps(self, required_skills: List[str]) -> Dict[str, List[str]]:
        """
        分析团队技能缺口
        """
        team_skills = set()
        for member in self.team_members:
            team_skills.update(member.skills)
        
        missing_skills = set(required_skills) - team_skills
        weak_skills = []
        
        for skill in required_skills:
            if skill in team_skills:
                skill_holders = [m.name for m in self.team_members if skill in m.skills]
                if len(skill_holders) < 2:  # 少于2人掌握
                    weak_skills.append(skill)
        
        return {
            'missing_skills': list(missing_skills),
            'weak_skills': weak_skills,
            'recommendations': self._generate_skill_recommendations(missing_skills, weak_skills)
        }
    
    def _generate_skill_recommendations(self, missing_skills: List[str], weak_skills: List[str]) -> List[str]:
        """
        生成技能提升建议
        """
        recommendations = []
        
        if missing_skills:
            recommendations.append(f"需要招聘或培训以下技能: {', '.join(missing_skills)}")
        
        if weak_skills:
            recommendations.append(f"需要加强以下技能的团队覆盖: {', '.join(weak_skills)}")
        
        return recommendations
    
    def set_team_hierarchy(self, hierarchy: Dict[str, List[str]]):
        """
        设置团队层级结构
        """
        self.team_hierarchy = hierarchy
    
    def set_communication_channels(self, channels: Dict[str, List[str]]):
        """
        设置沟通渠道
        """
        self.communication_channels = channels
    
    def get_team_summary(self) -> Dict:
        """
        获取团队摘要
        """
        role_counts = {}
        for member in self.team_members:
            role = member.role.value
            role_counts[role] = role_counts.get(role, 0) + 1
        
        return {
            'total_members': len(self.team_members),
            'role_distribution': role_counts,
            'avg_availability': sum(m.availability for m in self.team_members) / len(self.team_members),
            'skills_count': len(self.get_team_skills_matrix())
        }

from datetime import datetime, timedelta
from typing import Dict, List, Optional
import uuid

class CollaborationWorkflow:
    def __init__(self):
        self.workflows = {}
        self.active_tasks = {}
        self.task_history = []
        
    def create_workflow(self, workflow_name: str, stages: List[Dict]) -> str:
        """
        创建工作流
        """
        workflow_id = str(uuid.uuid4())
        
        self.workflows[workflow_id] = {
            'name': workflow_name,
            'stages': stages,
            'created_at': datetime.now(),
            'status': 'active'
        }
        
        print(f"工作流 {workflow_name} 已创建")
        return workflow_id
    
    def start_task(self, workflow_id: str, task_name: str, assignee: str, 
                   dependencies: List[str] = None) -> str:
        """
        开始任务
        """
        task_id = str(uuid.uuid4())
        
        self.active_tasks[task_id] = {
            'workflow_id': workflow_id,
            'task_name': task_name,
            'assignee': assignee,
            'status': 'in_progress',
            'started_at': datetime.now(),
            'dependencies': dependencies or [],
            'progress': 0
        }
        
        print(f"任务 {task_name} 已开始，分配给 {assignee}")
        return task_id
    
    def update_task_progress(self, task_id: str, progress: int, notes: str = ""):
        """
        更新任务进度
        """
        if task_id not in self.active_tasks:
            print(f"任务 {task_id} 不存在")
            return
        
        self.active_tasks[task_id]['progress'] = progress
        self.active_tasks[task_id]['last_updated'] = datetime.now()
        
        if notes:
            if 'notes' not in self.active_tasks[task_id]:
                self.active_tasks[task_id]['notes'] = []
            self.active_tasks[task_id]['notes'].append({
                'timestamp': datetime.now(),
                'content': notes
            })
        
        print(f"任务 {task_id} 进度更新为 {progress}%")
    
    def complete_task(self, task_id: str, completion_notes: str = ""):
        """
        完成任务
        """
        if task_id not in self.active_tasks:
            print(f"任务 {task_id} 不存在")
            return
        
        task = self.active_tasks[task_id]
        task['status'] = 'completed'
        task['completed_at'] = datetime.now()
        task['progress'] = 100
        
        if completion_notes:
            if 'notes' not in task:
                task['notes'] = []
            task['notes'].append({
                'timestamp': datetime.now(),
                'content': completion_notes
            })
        
        # 移动到历史记录
        self.task_history.append(task)
        del self.active_tasks[task_id]
        
        print(f"任务 {task_id} 已完成")
    
    def get_task_status(self, task_id: str) -> Optional[Dict]:
        """
        获取任务状态
        """
        if task_id in self.active_tasks:
            return self.active_tasks[task_id]
        elif task_id in [task['task_id'] for task in self.task_history]:
            return next(task for task in self.task_history if task.get('task_id') == task_id)
        return None
    
    def get_workflow_status(self, workflow_id: str) -> Dict:
        """
        获取工作流状态
        """
        if workflow_id not in self.workflows:
            return {}
        
        workflow = self.workflows[workflow_id]
        
        # 获取相关任务
        workflow_tasks = [
            task for task in self.active_tasks.values()
            if task['workflow_id'] == workflow_id
        ]
        
        workflow_tasks.extend([
            task for task in self.task_history
            if task.get('workflow_id') == workflow_id
        ])
        
        # 计算统计信息
        total_tasks = len(workflow_tasks)
        completed_tasks = len([task for task in workflow_tasks if task['status'] == 'completed'])
        in_progress_tasks = len([task for task in workflow_tasks if task['status'] == 'in_progress'])
        
        return {
            'workflow_id': workflow_id,
            'name': workflow['name'],
            'status': workflow['status'],
            'total_tasks': total_tasks,
            'completed_tasks': completed_tasks,
            'in_progress_tasks': in_progress_tasks,
            'completion_rate': completed_tasks / total_tasks if total_tasks > 0 else 0
        }

class CommunicationManager:
    def __init__(self):
        self.communication_channels = {}
        self.meeting_schedule = {}
        self.communication_logs = []
        
    def setup_communication_channel(self, channel_name: str, channel_type: str, 
                                   participants: List[str], purpose: str):
        """
        设置沟通渠道
        """
        self.communication_channels[channel_name] = {
            'type': channel_type,
            'participants': participants,
            'purpose': purpose,
            'created_at': datetime.now(),
            'active': True
        }
        
        print(f"沟通渠道 {channel_name} 已设置")
    
    def schedule_meeting(self, meeting_name: str, participants: List[str], 
                        start_time: datetime, duration: int, agenda: List[str]):
        """
        安排会议
        """
        meeting_id = str(uuid.uuid4())
        
        self.meeting_schedule[meeting_id] = {
            'name': meeting_name,
            'participants': participants,
            'start_time': start_time,
            'duration': duration,
            'end_time': start_time + timedelta(minutes=duration),
            'agenda': agenda,
            'status': 'scheduled'
        }
        
        print(f"会议 {meeting_name} 已安排")
        return meeting_id
    
    def log_communication(self, channel: str, sender: str, message: str, 
                         message_type: str = "text"):
        """
        记录沟通日志
        """
        log_entry = {
            'timestamp': datetime.now(),
            'channel': channel,
            'sender': sender,
            'message': message,
            'type': message_type
        }
        
        self.communication_logs.append(log_entry)
    
    def get_communication_summary(self, time_window: int = 7) -> Dict:
        """
        获取沟通摘要
        """
        cutoff_time = datetime.now() - timedelta(days=time_window)
        
        recent_logs = [
            log for log in self.communication_logs
            if log['timestamp'] >= cutoff_time
        ]
        
        # 按渠道统计
        channel_stats = {}
        for log in recent_logs:
            channel = log['channel']
            if channel not in channel_stats:
                channel_stats[channel] = 0
            channel_stats[channel] += 1
        
        # 按发送者统计
        sender_stats = {}
        for log in recent_logs:
            sender = log['sender']
            if sender not in sender_stats:
                sender_stats[sender] = 0
            sender_stats[sender] += 1
        
        return {
            'total_messages': len(recent_logs),
            'channel_stats': channel_stats,
            'sender_stats': sender_stats,
            'time_window_days': time_window
        }
    
    def analyze_communication_patterns(self) -> Dict:
        """
        分析沟通模式
        """
        if not self.communication_logs:
            return {}
        
        # 按小时统计消息分布
        hourly_stats = {}
        for log in self.communication_logs:
            hour = log['timestamp'].hour
            if hour not in hourly_stats:
                hourly_stats[hour] = 0
            hourly_stats[hour] += 1
        
        # 按星期几统计
        daily_stats = {}
        for log in self.communication_logs:
            weekday = log['timestamp'].weekday()
            if weekday not in daily_stats:
                daily_stats[weekday] = 0
            daily_stats[weekday] += 1
        
        return {
            'hourly_distribution': hourly_stats,
            'daily_distribution': daily_stats,
            'peak_hour': max(hourly_stats.keys(), key=lambda h: hourly_stats[h]),
            'peak_day': max(daily_stats.keys(), key=lambda d: daily_stats[d])
        }

from datetime import datetime, timedelta
from typing import List, Dict, Optional
import json

class Sprint:
    def __init__(self, sprint_number: int, duration_days: int = 14):
        self.sprint_number = sprint_number
        self.duration_days = duration_days
        self.start_date = datetime.now()
        self.end_date = self.start_date + timedelta(days=duration_days)
        self.user_stories = []
        self.sprint_goal = ""
        self.status = "planned"
        
    def add_user_story(self, story_id: str, title: str, description: str, 
                      story_points: int, assignee: str):
        """
        添加用户故事
        """
        user_story = {
            'id': story_id,
            'title': title,
            'description': description,
            'story_points': story_points,
            'assignee': assignee,
            'status': 'todo',
            'created_at': datetime.now()
        }
        
        self.user_stories.append(user_story)
        print(f"用户故事 {title} 已添加到Sprint {self.sprint_number}")
    
    def update_story_status(self, story_id: str, status: str):
        """
        更新故事状态
        """
        for story in self.user_stories:
            if story['id'] == story_id:
                story['status'] = status
                story['updated_at'] = datetime.now()
                print(f"用户故事 {story_id} 状态更新为 {status}")
                break
    
    def get_sprint_burndown(self) -> Dict:
        """
        获取Sprint燃尽图数据
        """
        total_points = sum(story['story_points'] for story in self.user_stories)
        completed_points = sum(
            story['story_points'] for story in self.user_stories 
            if story['status'] == 'done'
        )
        
        remaining_points = total_points - completed_points
        
        return {
            'total_points': total_points,
            'completed_points': completed_points,
            'remaining_points': remaining_points,
            'completion_rate': completed_points / total_points if total_points > 0 else 0
        }
    
    def get_sprint_summary(self) -> Dict:
        """
        获取Sprint摘要
        """
        return {
            'sprint_number': self.sprint_number,
            'start_date': self.start_date.isoformat(),
            'end_date': self.end_date.isoformat(),
            'status': self.status,
            'total_stories': len(self.user_stories),
            'completed_stories': len([s for s in self.user_stories if s['status'] == 'done']),
            'burndown': self.get_sprint_burndown()
        }

class ProductBacklog:
    def __init__(self):
        self.backlog_items = []
        self.epics = []
        
    def add_epic(self, epic_id: str, title: str, description: str, 
                business_value: int, priority: int):
        """
        添加Epic
        """
        epic = {
            'id': epic_id,
            'title': title,
            'description': description,
            'business_value': business_value,
            'priority': priority,
            'status': 'planned',
            'created_at': datetime.now(),
            'user_stories': []
        }
        
        self.epics.append(epic)
        print(f"Epic {title} 已添加到产品待办列表")
    
    def add_user_story(self, epic_id: str, story_id: str, title: str, 
                      description: str, acceptance_criteria: List[str],
                      story_points: int, priority: int):
        """
        添加用户故事
        """
        user_story = {
            'id': story_id,
            'epic_id': epic_id,
            'title': title,
            'description': description,
            'acceptance_criteria': acceptance_criteria,
            'story_points': story_points,
            'priority': priority,
            'status': 'backlog',
            'created_at': datetime.now()
        }
        
        # 添加到Epic
        for epic in self.epics:
            if epic['id'] == epic_id:
                epic['user_stories'].append(user_story)
                break
        
        self.backlog_items.append(user_story)
        print(f"用户故事 {title} 已添加到产品待办列表")
    
    def prioritize_backlog(self):
        """
        对产品待办列表进行优先级排序
        """
        # 按优先级和业务价值排序
        self.backlog_items.sort(key=lambda x: (x['priority'], x['story_points']), reverse=True)
        print("产品待办列表已按优先级排序")
    
    def get_ready_for_sprint(self, team_capacity: int) -> List[Dict]:
        """
        获取准备进入Sprint的待办项
        """
        ready_items = []
        total_points = 0
        
        for item in self.backlog_items:
            if item['status'] == 'backlog' and total_points + item['story_points'] <= team_capacity:
                ready_items.append(item)
                total_points += item['story_points']
        
        return ready_items

class ScrumMaster:
    def __init__(self):
        self.sprints = []
        self.current_sprint = None
        self.team_velocity = []
        
    def create_sprint(self, sprint_number: int, duration_days: int = 14) -> Sprint:
        """
        创建Sprint
        """
        sprint = Sprint(sprint_number, duration_days)
        self.sprints.append(sprint)
        
        if not self.current_sprint:
            self.current_sprint = sprint
        
        print(f"Sprint {sprint_number} 已创建")
        return sprint
    
    def start_sprint(self, sprint_number: int):
        """
        开始Sprint
        """
        for sprint in self.sprints:
            if sprint.sprint_number == sprint_number:
                sprint.status = "active"
                self.current_sprint = sprint
                print(f"Sprint {sprint_number} 已开始")
                break
    
    def end_sprint(self, sprint_number: int):
        """
        结束Sprint
        """
        for sprint in self.sprints:
            if sprint.sprint_number == sprint_number:
                sprint.status = "completed"
                
                # 计算团队速度
                burndown = sprint.get_sprint_burndown()
                self.team_velocity.append(burndown['completed_points'])
                
                print(f"Sprint {sprint_number} 已结束")
                break

            'updates': updates
        }
        
        print(f"每日站会已进行，参与人员: {', '.join(participants)}")
        return standup
    
    def conduct_sprint_review(self, sprint_number: int, demo_items: List[Dict]):
        """
        进行Sprint评审
        """
        review = {
            'sprint_number': sprint_number,
            'date': datetime.now(),
            'demo_items': demo_items,
            'feedback': []
        }
        
        print(f"Sprint {sprint_number} 评审已进行")
        return review
    
    def conduct_sprint_retrospective(self, sprint_number: int, 
                                   what_went_well: List[str],
                                   what_could_improve: List[str],
                                   action_items: List[str]):
        """
        进行Sprint回顾
        """
        retrospective = {
            'sprint_number': sprint_number,
            'date': datetime.now(),
            'what_went_well': what_went_well,
            'what_could_improve': what_could_improve,
            'action_items': action_items
        }
        
        print(f"Sprint {sprint_number} 回顾已进行")
        return retrospective
    
    def calculate_team_velocity(self) -> Dict:
        """
        计算团队速度
        """
        if not self.team_velocity:
            return {'average_velocity': 0, 'velocity_trend': 'stable'}
        
        avg_velocity = sum(self.team_velocity) / len(self.team_velocity)
        
        # 计算趋势
        if len(self.team_velocity) >= 3:
            recent_avg = sum(self.team_velocity[-3:]) / 3
            if recent_avg > avg_velocity * 1.1:
                trend = 'increasing'
            elif recent_avg < avg_velocity * 0.9:
                trend = 'decreasing'
            else:
                trend = 'stable'
        else:
            trend = 'stable'
        
        return {
            'average_velocity': avg_velocity,
            'velocity_trend': trend,
            'velocity_history': self.team_velocity
        }

from enum import Enum
from typing import Dict, List, Optional

class RiskLevel(Enum):

class RiskCategory(Enum):

class Risk:

    title: str
    description: str
    category: RiskCategory
    level: RiskLevel
    probability: float  # 0-1
    impact: float  # 0-1

    updated_at: datetime
    
    def calculate_risk_score(self) -> float:

            'description': self.description,

        }

class RiskManager:
    def __init__(self):
        self.risks = []
        self.risk_history = []
        self.mitigation_actions = []
        
    def identify_risk(self, title: str, description: str, category: RiskCategory,
                    probability: float, impact: float, owner: str,
                    mitigation_plan: str = "", contingency_plan: str = "") -> Risk:
        """
        识别风险
        """
        risk_id = str(uuid.uuid4())
        
        # 确定风险等级
        risk_score = probability * impact
        if risk_score >= 0.8:
            level = RiskLevel.CRITICAL
        elif risk_score >= 0.6:
            level = RiskLevel.HIGH
        elif risk_score >= 0.3:
            level = RiskLevel.MEDIUM
        else:
            level = RiskLevel.LOW
        
        risk = Risk(
            id=risk_id,
            title=title,
            description=description,
            category=category,
            level=level,
            probability=probability,
            impact=impact,
            owner=owner,
            mitigation_plan=mitigation_plan,
            contingency_plan=contingency_plan,
            created_at=datetime.now(),
            updated_at=datetime.now()
        )
        
        self.risks.append(risk)
        print(f"风险 {title} 已识别，等级: {level.value}")
        
        return risk
    
    def update_risk(self, risk_id: str, **kwargs):
        """
        更新风险
        """
        for risk in self.risks:
            if risk.id == risk_id:
                for key, value in kwargs.items():
                    if hasattr(risk, key):
                        setattr(risk, key, value)
                
                risk.updated_at = datetime.now()
                
                # 重新计算风险等级
                risk_score = risk.calculate_risk_score()
                if risk_score >= 0.8:
                    risk.level = RiskLevel.CRITICAL
                elif risk_score >= 0.6:
                    risk.level = RiskLevel.HIGH
                elif risk_score >= 0.3:
                    risk.level = RiskLevel.MEDIUM
                else:
                    risk.level = RiskLevel.LOW
                
                print(f"风险 {risk_id} 已更新")
                break
    
    def implement_mitigation(self, risk_id: str, action: str, owner: str, 
                           due_date: datetime):
        """
        实施缓解措施
        """
        mitigation_action = {
            'risk_id': risk_id,
            'action': action,
            'owner': owner,
            'due_date': due_date,
            'status': 'in_progress',
            'created_at': datetime.now()
        }
        
        self.mitigation_actions.append(mitigation_action)
        print(f"风险 {risk_id} 的缓解措施已实施")
    
    def get_risk_dashboard(self) -> Dict:
        """
        获取风险仪表板
        """
        # 按等级统计
        level_counts = {}
        for level in RiskLevel:
            level_counts[level.value] = len([r for r in self.risks if r.level == level])
        
        # 按类别统计
        category_counts = {}
        for category in RiskCategory:
            category_counts[category.value] = len([r for r in self.risks if r.category == category])
        
        # 高风险项目
        high_risk_items = [r for r in self.risks if r.level in [RiskLevel.HIGH, RiskLevel.CRITICAL]]
        
        return {
            'total_risks': len(self.risks),
            'level_distribution': level_counts,
            'category_distribution': category_counts,
            'high_risk_count': len(high_risk_items),
            'high_risk_items': [r.to_dict() for r in high_risk_items],
            'mitigation_actions_count': len(self.mitigation_actions)
        }
    
    def generate_risk_report(self) -> Dict:
        """
        生成风险报告
        """
        report = {
            'generated_at': datetime.now(),
            'summary': self.get_risk_dashboard(),
            'detailed_risks': [r.to_dict() for r in self.risks],
            'mitigation_actions': self.mitigation_actions,
            'recommendations': self._generate_risk_recommendations()

        if pending_mitigations > 0:

        # 定期审查建议

class QualityAssurance:
    def __init__(self):
        self.quality_metrics = {}
        self.test_cases = []
        self.defects = []
        self.quality_gates = []
        
    def define_quality_gate(self, gate_name: str, criteria: Dict, 
                          threshold: float, mandatory: bool = True):
        """
        定义质量门
        """
        quality_gate = {
            'name': gate_name,
            'criteria': criteria,
            'threshold': threshold,
            'mandatory': mandatory,
            'created_at': datetime.now()
        }
        
        self.quality_gates.append(quality_gate)
        print(f"质量门 {gate_name} 已定义")
    
    def add_test_case(self, test_id: str, title: str, description: str,
                     test_type: str, priority: str, steps: List[str],
                     expected_result: str):
        """
        添加测试用例
        """
        test_case = {
            'id': test_id,
            'title': title,
            'description': description,
            'type': test_type,
            'priority': priority,
            'steps': steps,
            'expected_result': expected_result,
            'status': 'draft',
            'created_at': datetime.now()
        }
        
        self.test_cases.append(test_case)
        print(f"测试用例 {title} 已添加")
    
    def execute_test_case(self, test_id: str, result: str, 
                         actual_result: str, tester: str):
        """
        执行测试用例
        """
        for test_case in self.test_cases:
            if test_case['id'] == test_id:
                test_case['result'] = result
                test_case['actual_result'] = actual_result
                test_case['tester'] = tester
                test_case['executed_at'] = datetime.now()
                test_case['status'] = 'executed'
                
                # 如果测试失败，记录缺陷
                if result == 'failed':
                    self._record_defect(test_id, actual_result, tester)
                
                print(f"测试用例 {test_id} 已执行，结果: {result}")
                break
    
    def _record_defect(self, test_id: str, description: str, reporter: str):
        """
        记录缺陷
        """
        defect_id = str(uuid.uuid4())
        
        defect = {
            'id': defect_id,
            'test_id': test_id,
            'description': description,
            'reporter': reporter,
            'status': 'open',
            'priority': 'medium',
            'created_at': datetime.now()
        }
        
        self.defects.append(defect)
        print(f"缺陷 {defect_id} 已记录")
    
    def calculate_quality_metrics(self) -> Dict:
        """
        计算质量指标
        """
        total_test_cases = len(self.test_cases)
        executed_test_cases = len([tc for tc in self.test_cases if tc['status'] == 'executed'])
        passed_test_cases = len([tc for tc in self.test_cases if tc.get('result') == 'passed'])
        
        test_coverage = executed_test_cases / total_test_cases if total_test_cases > 0 else 0
        test_pass_rate = passed_test_cases / executed_test_cases if executed_test_cases > 0 else 0
        
        open_defects = len([d for d in self.defects if d['status'] == 'open'])
        critical_defects = len([d for d in self.defects if d['priority'] == 'critical'])
        
        return {
            'test_coverage': test_coverage,
            'test_pass_rate': test_pass_rate,
            'total_test_cases': total_test_cases,
            'executed_test_cases': executed_test_cases,
            'passed_test_cases': passed_test_cases,
            'open_defects': open_defects,
            'critical_defects': critical_defects,
            'defect_density': open_defects / total_test_cases if total_test_cases > 0 else 0
        }
    
    def check_quality_gates(self) -> Dict:
        """
        检查质量门
        """
        gate_results = {}
        
        for gate in self.quality_gates:
            gate_name = gate['name']
            criteria = gate['criteria']
            threshold = gate['threshold']
            
            # 根据标准计算指标
            if 'test_pass_rate' in criteria:
                metrics = self.calculate_quality_metrics()
                actual_value = metrics['test_pass_rate']
            elif 'defect_density' in criteria:
                metrics = self.calculate_quality_metrics()
                actual_value = metrics['defect_density']
            else:
                actual_value = 0
            
            passed = actual_value >= threshold if criteria.get('min_threshold', True) else actual_value <= threshold
            
            gate_results[gate_name] = {
                'passed': passed,
                'threshold': threshold,
                'actual_value': actual_value,
                'mandatory': gate['mandatory']
            }
        
        return gate_results
