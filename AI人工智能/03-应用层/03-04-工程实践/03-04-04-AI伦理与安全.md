# AI伦理与安全

## 概述

AI伦理与安全是人工智能发展中的重要议题，涉及算法公平性、隐私保护、透明度、责任归属等多个方面。随着AI技术的广泛应用，确保AI系统的安全性、公平性和可解释性变得越来越重要。本章将介绍AI伦理与安全的核心概念、实践方法和治理框架。

## 核心概念

### 1. AI伦理定义
- **目标**：确保AI系统的设计、开发和使用符合道德标准和社会价值观
- **核心原则**：公平性、透明度、可解释性、隐私保护、安全性
- **关键挑战**：算法偏见、数据隐私、决策透明度、责任归属
- **应用范围**：算法设计、数据收集、模型训练、系统部署、用户交互

### 2. AI安全框架
- **技术安全**：系统稳定性、对抗攻击防护、数据完整性
- **伦理安全**：算法公平性、隐私保护、透明度
- **社会安全**：就业影响、社会公平、人权保护
- **治理安全**：监管合规、责任归属、审计机制

```python
import numpy as np
import pandas as pd
from sklearn.metrics import confusion_matrix
import matplotlib.pyplot as plt
import seaborn as sns

class FairnessMetrics:
    def __init__(self):
        self.metrics = {}
        self.bias_detection_results = {}
        
    def calculate_demographic_parity(self, y_true, y_pred, sensitive_attribute):
        """
        计算人口统计均等性
        """
        results = {}
        
        # 按敏感属性分组
        groups = np.unique(sensitive_attribute)
        
        for group in groups:
            group_mask = sensitive_attribute == group
            group_y_true = y_true[group_mask]
            group_y_pred = y_pred[group_mask]
            
            # 计算正预测率
            positive_rate = np.mean(group_y_pred)
            results[group] = positive_rate
        
        # 计算最大差异
        rates = list(results.values())
        max_difference = max(rates) - min(rates)
        
        return {
            'group_rates': results,
            'max_difference': max_difference,
            'fair': max_difference < 0.1  # 10%的阈值
        }
    
    def calculate_equalized_odds(self, y_true, y_pred, sensitive_attribute):
        """
        计算均等机会
        """
        results = {}
        
        # 按敏感属性分组
        groups = np.unique(sensitive_attribute)
        
        for group in groups:
            group_mask = sensitive_attribute == group
            group_y_true = y_true[group_mask]
            group_y_pred = y_pred[group_mask]
            
            # 计算真正率和假正率
            tn, fp, fn, tp = confusion_matrix(group_y_true, group_y_pred).ravel()
            
            tpr = tp / (tp + fn) if (tp + fn) > 0 else 0
            fpr = fp / (fp + tn) if (fp + tn) > 0 else 0
            
            results[group] = {
                'true_positive_rate': tpr,
                'false_positive_rate': fpr
            }
        
        # 计算差异
        tprs = [results[group]['true_positive_rate'] for group in groups]
        fprs = [results[group]['false_positive_rate'] for group in groups]
        
        tpr_difference = max(tprs) - min(tprs)
        fpr_difference = max(fprs) - min(fprs)
        
        return {
            'group_metrics': results,
            'tpr_difference': tpr_difference,
            'fpr_difference': fpr_difference,
            'fair': tpr_difference < 0.1 and fpr_difference < 0.1
        }
    
    def calculate_equal_opportunity(self, y_true, y_pred, sensitive_attribute):
        """
        计算均等机会
        """
        results = {}
        
        # 按敏感属性分组
        groups = np.unique(sensitive_attribute)
        
        for group in groups:
            group_mask = sensitive_attribute == group
            group_y_true = y_true[group_mask]
            group_y_pred = y_pred[group_mask]
            
            # 只考虑正例
            positive_mask = group_y_true == 1
            if np.sum(positive_mask) > 0:
                group_positive_y_true = group_y_true[positive_mask]
                group_positive_y_pred = group_y_pred[positive_mask]
                
                # 计算真正率
                tn, fp, fn, tp = confusion_matrix(group_positive_y_true, group_positive_y_pred).ravel()
                tpr = tp / (tp + fn) if (tp + fn) > 0 else 0
                
                results[group] = tpr
            else:
                results[group] = 0
        
        # 计算最大差异
        tprs = list(results.values())
        max_difference = max(tprs) - min(tprs)
        
        return {
            'group_tprs': results,
            'max_difference': max_difference,
            'fair': max_difference < 0.1
        }
    
    def calculate_calibration(self, y_true, y_pred_proba, sensitive_attribute):
        """
        计算校准公平性
        """
        results = {}
        
        # 按敏感属性分组
        groups = np.unique(sensitive_attribute)
        
        for group in groups:
            group_mask = sensitive_attribute == group
            group_y_true = y_true[group_mask]
            group_y_pred_proba = y_pred_proba[group_mask]
            
            # 计算校准误差
            calibration_error = np.mean(np.abs(group_y_pred_proba - group_y_true))
            results[group] = calibration_error
        
        # 计算最大差异
        errors = list(results.values())
        max_difference = max(errors) - min(errors)
        
        return {
            'group_errors': results,
            'max_difference': max_difference,
            'fair': max_difference < 0.05
        }
    
    def comprehensive_fairness_assessment(self, y_true, y_pred, y_pred_proba, sensitive_attribute):
        """
        综合公平性评估
        """
        assessment = {
            'demographic_parity': self.calculate_demographic_parity(y_true, y_pred, sensitive_attribute),
            'equalized_odds': self.calculate_equalized_odds(y_true, y_pred, sensitive_attribute),
            'equal_opportunity': self.calculate_equal_opportunity(y_true, y_pred, sensitive_attribute),
            'calibration': self.calculate_calibration(y_true, y_pred_proba, sensitive_attribute)
        }
        
        # 计算总体公平性分数
        fairness_scores = []
        for metric_name, metric_result in assessment.items():
            if 'fair' in metric_result:
                fairness_scores.append(1 if metric_result['fair'] else 0)
        
        overall_fairness = np.mean(fairness_scores) if fairness_scores else 0
        
        assessment['overall_fairness'] = {
            'score': overall_fairness,
            'fair': overall_fairness >= 0.75
        }
        
        return assessment

class BiasDetector:
    def __init__(self):
        self.detection_results = {}
        
    def detect_data_bias(self, data, sensitive_attributes):
        """
        检测数据偏见
        """
        bias_results = {}
        
        for attr in sensitive_attributes:
            if attr in data.columns:
                # 计算分布差异
                value_counts = data[attr].value_counts(normalize=True)
                
                # 检查分布是否均匀
                max_proportion = value_counts.max()
                min_proportion = value_counts.min()
                distribution_bias = max_proportion - min_proportion
                
                bias_results[attr] = {
                    'distribution': value_counts.to_dict(),
                    'bias_score': distribution_bias,
                    'biased': distribution_bias > 0.3  # 30%的阈值
                }
        
        return bias_results
    
    def detect_algorithmic_bias(self, model, X_test, y_test, sensitive_attribute):
        """
        检测算法偏见
        """
        # 获取预测结果
        y_pred = model.predict(X_test)
        y_pred_proba = model.predict_proba(X_test)[:, 1] if hasattr(model, 'predict_proba') else None
        
        # 计算公平性指标
        fairness_metrics = FairnessMetrics()
        
        if y_pred_proba is not None:
            assessment = fairness_metrics.comprehensive_fairness_assessment(
                y_test, y_pred, y_pred_proba, sensitive_attribute
            )
        else:
            assessment = {
                'demographic_parity': fairness_metrics.calculate_demographic_parity(
                    y_test, y_pred, sensitive_attribute
                ),
                'equalized_odds': fairness_metrics.calculate_equalized_odds(
                    y_test, y_pred, sensitive_attribute
                ),
                'equal_opportunity': fairness_metrics.calculate_equal_opportunity(
                    y_test, y_pred, sensitive_attribute
                )
            }
        
        return assessment
    
    def detect_historical_bias(self, data, target_column, sensitive_attributes):
        """
        检测历史偏见
        """
        bias_results = {}
        
        for attr in sensitive_attributes:
            if attr in data.columns:
                # 按敏感属性分组计算目标变量分布
                group_stats = data.groupby(attr)[target_column].agg(['count', 'mean', 'std'])
                
                # 计算组间差异
                means = group_stats['mean']
                max_mean = means.max()
                min_mean = means.min()
                mean_difference = max_mean - min_mean
                
                bias_results[attr] = {
                    'group_stats': group_stats.to_dict(),
                    'mean_difference': mean_difference,
                    'biased': mean_difference > 0.1  # 10%的阈值
                }
        
        return bias_results
    
    def generate_bias_report(self, data, model, target_column, sensitive_attributes):
        """
        生成偏见报告
        """
        report = {
            'data_bias': self.detect_data_bias(data, sensitive_attributes),
            'historical_bias': self.detect_historical_bias(data, target_column, sensitive_attributes),
            'recommendations': []
        }
        
        # 生成建议
        if any(result['biased'] for result in report['data_bias'].values()):
            report['recommendations'].append("数据存在分布偏见，建议重新采样或收集更多数据")
        
        if any(result['biased'] for result in report['historical_bias'].values()):
            report['recommendations'].append("历史数据存在偏见，建议使用公平性约束进行训练")
        
        return report
```

### 2. 偏见缓解技术

#### 公平性约束优化
```python
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
import cvxpy as cp

class FairnessConstrainedModel:
    def __init__(self, fairness_constraint='demographic_parity', constraint_value=0.1):
        self.fairness_constraint = fairness_constraint
        self.constraint_value = constraint_value
        self.model = None
        self.scaler = StandardScaler()
        
    def fit(self, X, y, sensitive_attribute):
        """
        训练带公平性约束的模型
        """
        # 标准化特征
        X_scaled = self.scaler.fit_transform(X)
        
        if self.fairness_constraint == 'demographic_parity':
            self.model = self._fit_demographic_parity_constrained(X_scaled, y, sensitive_attribute)
        elif self.fairness_constraint == 'equalized_odds':
            self.model = self._fit_equalized_odds_constrained(X_scaled, y, sensitive_attribute)
        else:
            raise ValueError(f"不支持的公平性约束: {self.fairness_constraint}")
        
        return self
    
    def _fit_demographic_parity_constrained(self, X, y, sensitive_attribute):
        """
        人口统计均等性约束
        """
        n_samples, n_features = X.shape
        groups = np.unique(sensitive_attribute)
        
        # 定义优化变量
        w = cp.Variable(n_features)
        b = cp.Variable()
        
        # 目标函数：最小化损失
        loss = cp.sum(cp.logistic(-y * (X @ w + b)))
        
        # 公平性约束
        constraints = []
        
        for group in groups:
            group_mask = sensitive_attribute == group
            group_X = X[group_mask]
            group_y = y[group_mask]
            
            # 计算该组的正预测率
            group_predictions = group_X @ w + b
            group_positive_rate = cp.mean(cp.logistic(group_predictions))
            
            # 添加约束
            constraints.append(group_positive_rate <= self.constraint_value)
        
        # 求解优化问题
        problem = cp.Problem(cp.Minimize(loss), constraints)
        problem.solve()
        
        # 创建模型
        model = LogisticRegression()
        model.coef_ = w.value.reshape(1, -1)
        model.intercept_ = b.value
        
        return model
    
    def _fit_equalized_odds_constrained(self, X, y, sensitive_attribute):
        """
        均等机会约束
        """
        n_samples, n_features = X.shape
        groups = np.unique(sensitive_attribute)
        
        # 定义优化变量
        w = cp.Variable(n_features)
        b = cp.Variable()
        
        # 目标函数
        loss = cp.sum(cp.logistic(-y * (X @ w + b)))
        
        # 约束
        constraints = []
        
        for group in groups:
            group_mask = sensitive_attribute == group
            group_X = X[group_mask]
            group_y = y[group_mask]
            
            # 正例和负例
            positive_mask = group_y == 1
            negative_mask = group_y == -1
            
            if np.sum(positive_mask) > 0:
                positive_X = group_X[positive_mask]
                positive_predictions = positive_X @ w + b
                tpr = cp.mean(cp.logistic(positive_predictions))
                constraints.append(tpr <= self.constraint_value)
            
            if np.sum(negative_mask) > 0:
                negative_X = group_X[negative_mask]
                negative_predictions = negative_X @ w + b
                fpr = cp.mean(cp.logistic(negative_predictions))
                constraints.append(fpr <= self.constraint_value)
        
        # 求解
        problem = cp.Problem(cp.Minimize(loss), constraints)
        problem.solve()
        
        # 创建模型
        model = LogisticRegression()
        model.coef_ = w.value.reshape(1, -1)
        model.intercept_ = b.value
        
        return model
    
    def predict(self, X):
        """
        预测
        """
        X_scaled = self.scaler.transform(X)
        return self.model.predict(X_scaled)
    
    def predict_proba(self, X):
        """
        预测概率
        """
        X_scaled = self.scaler.transform(X)
        return self.model.predict_proba(X_scaled)

class BiasMitigationTechniques:
    def __init__(self):
        self.techniques = {}
        
    def preprocess_data(self, data, target_column, sensitive_attributes, method='reweighing'):
        """
        预处理数据以减少偏见
        """
        if method == 'reweighing':
            return self._reweighing(data, target_column, sensitive_attributes)
        elif method == 'resampling':
            return self._resampling(data, target_column, sensitive_attributes)
        elif method == 'synthetic_generation':
            return self._synthetic_generation(data, target_column, sensitive_attributes)
        else:
            raise ValueError(f"不支持的预处理方法: {method}")
    
    def _reweighing(self, data, target_column, sensitive_attributes):
        """
        重新加权
        """
        # 计算每个组合的权重
        weights = np.ones(len(data))
        
        for attr in sensitive_attributes:
            if attr in data.columns:
                # 计算期望分布
                expected_dist = data[attr].value_counts(normalize=True)
                
                # 计算实际分布
                actual_dist = data[attr].value_counts(normalize=True)
                
                # 计算权重
                for value in data[attr].unique():
                    mask = data[attr] == value
                    weight = expected_dist[value] / actual_dist[value]
                    weights[mask] *= weight
        
        return data, weights
    
    def _resampling(self, data, target_column, sensitive_attributes):
        """
        重采样
        """
        # 按敏感属性和目标变量分组
        groups = data.groupby(sensitive_attributes + [target_column])
        
        # 计算每组的目标样本数
        target_counts = {}
        for name, group in groups:
            target_counts[name] = len(group)
        
        # 找到最大组
        max_count = max(target_counts.values())
        
        # 重采样
        resampled_data = []
        for name, group in groups:
            if len(group) < max_count:
                # 上采样
                resampled_group = group.sample(n=max_count, replace=True, random_state=42)
            else:
                # 下采样
                resampled_group = group.sample(n=max_count, random_state=42)
            
            resampled_data.append(resampled_group)
        
        return pd.concat(resampled_data, ignore_index=True)
    
    def _synthetic_generation(self, data, target_column, sensitive_attributes):
        """
        合成数据生成
        """
        # 使用SMOTE生成合成样本
        from imblearn.over_sampling import SMOTE
        
        # 分离特征和目标
        X = data.drop(columns=[target_column])
        y = data[target_column]
        
        # 应用SMOTE
        smote = SMOTE(random_state=42)
        X_resampled, y_resampled = smote.fit_resample(X, y)
        
        # 重新组合数据
        resampled_data = pd.DataFrame(X_resampled, columns=X.columns)
        resampled_data[target_column] = y_resampled
        
        return resampled_data
    
    def postprocess_predictions(self, y_pred, y_pred_proba, sensitive_attribute, method='threshold_optimization'):
        """
        后处理预测结果以减少偏见
        """
        if method == 'threshold_optimization':
            return self._threshold_optimization(y_pred, y_pred_proba, sensitive_attribute)
        elif method == 'calibration':
            return self._calibration(y_pred, y_pred_proba, sensitive_attribute)
        else:
            raise ValueError(f"不支持的后处理方法: {method}")
    
    def _threshold_optimization(self, y_pred, y_pred_proba, sensitive_attribute):
        """
        阈值优化
        """
        # 按敏感属性分组
        groups = np.unique(sensitive_attribute)
        
        # 为每组优化阈值
        optimized_predictions = y_pred.copy()
        
        for group in groups:
            group_mask = sensitive_attribute == group
            group_proba = y_pred_proba[group_mask]
            
            # 找到最优阈值
            thresholds = np.linspace(0, 1, 100)
            best_threshold = 0.5
            best_score = float('inf')
            
            for threshold in thresholds:
                group_pred = (group_proba >= threshold).astype(int)
                # 这里可以添加具体的优化目标
                score = np.mean(np.abs(group_pred - group_proba))
                
                if score < best_score:
                    best_score = score
                    best_threshold = threshold
            
            # 应用优化后的阈值
            optimized_predictions[group_mask] = (group_proba >= best_threshold).astype(int)
        
        return optimized_predictions
    
    def _calibration(self, y_pred, y_pred_proba, sensitive_attribute):
        """
        校准
        """
        # 按敏感属性分组校准
        groups = np.unique(sensitive_attribute)
        calibrated_predictions = y_pred.copy()
        
        for group in groups:
            group_mask = sensitive_attribute == group
            group_proba = y_pred_proba[group_mask]
            
            # 简单的校准：调整概率分布
            group_mean = np.mean(group_proba)
            group_std = np.std(group_proba)
            
            # 标准化
            calibrated_proba = (group_proba - group_mean) / group_std
            
            # 重新映射到[0,1]
            calibrated_proba = (calibrated_proba - calibrated_proba.min()) / (calibrated_proba.max() - calibrated_proba.min())
            
            # 应用校准后的阈值
            calibrated_predictions[group_mask] = (calibrated_proba >= 0.5).astype(int)
        
        return calibrated_predictions
```

## 隐私保护技术

### 1. 差分隐私

#### 差分隐私实现
```python
import numpy as np
from scipy.stats import laplace
import math

class DifferentialPrivacy:
    def __init__(self, epsilon=1.0):
        self.epsilon = epsilon
        
    def add_laplace_noise(self, data, sensitivity=1.0):
        """
        添加拉普拉斯噪声
        """
        noise = laplace.rvs(scale=sensitivity/self.epsilon, size=data.shape)
        return data + noise
    
    def add_gaussian_noise(self, data, sensitivity=1.0, delta=1e-5):
        """
        添加高斯噪声
        """
        sigma = sensitivity * math.sqrt(2 * math.log(1.25/delta)) / self.epsilon
        noise = np.random.normal(0, sigma, data.shape)
        return data + noise
    
    def private_mean(self, data, sensitivity=1.0):
        """
        私有均值计算
        """
        true_mean = np.mean(data)
        noise = laplace.rvs(scale=sensitivity/self.epsilon)
        return true_mean + noise
    
    def private_count(self, data, sensitivity=1.0):
        """
        私有计数
        """
        true_count = len(data)
        noise = laplace.rvs(scale=sensitivity/self.epsilon)
        return max(0, true_count + noise)
    
    def private_sum(self, data, sensitivity=1.0):
        """
        私有求和
        """
        true_sum = np.sum(data)
        noise = laplace.rvs(scale=sensitivity/self.epsilon)
        return true_sum + noise
    
    def private_histogram(self, data, bins, sensitivity=1.0):
        """
        私有直方图
        """
        # 计算真实直方图
        true_hist, _ = np.histogram(data, bins=bins)
        
        # 添加噪声
        noise = laplace.rvs(scale=sensitivity/self.epsilon, size=len(true_hist))
        private_hist = true_hist + noise
        
        # 确保非负
        private_hist = np.maximum(private_hist, 0)
        
        return private_hist
    
    def private_quantile(self, data, quantile, sensitivity=1.0):
        """
        私有分位数
        """
        # 排序数据
        sorted_data = np.sort(data)
        n = len(sorted_data)
        
        # 计算真实分位数
        index = int(quantile * n)
        true_quantile = sorted_data[index]
        
        # 添加噪声
        noise = laplace.rvs(scale=sensitivity/self.epsilon)
        return true_quantile + noise
    
    def compose_privacy_budget(self, operations):
        """
        组合隐私预算
        """
        total_epsilon = sum(op['epsilon'] for op in operations)
        total_delta = sum(op.get('delta', 0) for op in operations)
        
        return {
            'epsilon': total_epsilon,
            'delta': total_delta
        }

class PrivacyPreservingML:
    def __init__(self, epsilon=1.0):
        self.epsilon = epsilon
        self.dp = DifferentialPrivacy(epsilon)
        
    def private_logistic_regression(self, X, y, learning_rate=0.01, epochs=100):
        """
        私有逻辑回归
        """
        n_samples, n_features = X.shape
        
        # 初始化权重
        weights = np.random.normal(0, 0.01, n_features)
        bias = 0
        
        for epoch in range(epochs):
            # 计算梯度
            z = X @ weights + bias
            predictions = 1 / (1 + np.exp(-z))
            
            # 梯度
            gradient_w = X.T @ (predictions - y) / n_samples
            gradient_b = np.mean(predictions - y)
            
            # 添加噪声
            noise_w = laplace.rvs(scale=1.0/self.epsilon, size=n_features)
            noise_b = laplace.rvs(scale=1.0/self.epsilon)
            
            gradient_w += noise_w
            gradient_b += noise_b
            
            # 更新权重
            weights -= learning_rate * gradient_w
            bias -= learning_rate * gradient_b
        
        return weights, bias
    
    def private_kmeans(self, X, k, max_iters=100):
        """
        私有K-means聚类
        """
        n_samples, n_features = X.shape
        
        # 随机初始化中心
        centers = X[np.random.choice(n_samples, k, replace=False)]
        
        for iteration in range(max_iters):
            # 分配点到最近的中心
            distances = np.sqrt(((X - centers[:, np.newaxis])**2).sum(axis=2))
            labels = np.argmin(distances, axis=0)
            
            # 更新中心
            new_centers = np.zeros_like(centers)
            for i in range(k):
                cluster_points = X[labels == i]
                if len(cluster_points) > 0:
                    # 添加噪声到中心
                    noise = laplace.rvs(scale=1.0/self.epsilon, size=n_features)
                    new_centers[i] = np.mean(cluster_points, axis=0) + noise
            
            # 检查收敛
            if np.allclose(centers, new_centers):
                break
            
            centers = new_centers
        
        return centers, labels
    
    def private_pca(self, X, n_components):
        """
        私有PCA
        """
        # 中心化数据
        X_centered = X - np.mean(X, axis=0)
        
        # 计算协方差矩阵
        cov_matrix = X_centered.T @ X_centered / (len(X) - 1)
        
        # 添加噪声到协方差矩阵
        noise = laplace.rvs(scale=1.0/self.epsilon, size=cov_matrix.shape)
        private_cov_matrix = cov_matrix + noise
        
        # 特征值分解
        eigenvalues, eigenvectors = np.linalg.eigh(private_cov_matrix)
        
        # 选择前n_components个主成分
        indices = np.argsort(eigenvalues)[::-1][:n_components]
        principal_components = eigenvectors[:, indices]
        
        return principal_components, eigenvalues[indices]
```

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
import numpy as np

class FederatedLearningServer:
    def __init__(self, model, learning_rate=0.01):
        self.global_model = model
        self.learning_rate = learning_rate
        self.client_models = {}
        self.client_weights = {}
        
    def initialize_global_model(self):
        """
        初始化全局模型
        """
        # 随机初始化模型参数
        for param in self.global_model.parameters():
            param.data.normal_(0, 0.01)
        
        print("全局模型已初始化")
    
    def aggregate_client_updates(self, client_updates):
        """
        聚合客户端更新
        """
        # 计算加权平均
        total_samples = sum(client_updates[client]['num_samples'] for client in client_updates)
        
        # 初始化聚合参数
        aggregated_params = {}
        for name, param in self.global_model.named_parameters():
            aggregated_params[name] = torch.zeros_like(param.data)
        
        # 加权聚合
        for client_id, update in client_updates.items():
            weight = update['num_samples'] / total_samples
            
            for name, param in update['model'].named_parameters():
                aggregated_params[name] += weight * param.data
        
        # 更新全局模型
        for name, param in self.global_model.named_parameters():
            param.data = aggregated_params[name]
        
        print(f"已聚合 {len(client_updates)} 个客户端更新")
    
    def distribute_global_model(self, client_ids):
        """
        分发全局模型
        """
        for client_id in client_ids:
            # 复制全局模型
            client_model = type(self.global_model)()
            client_model.load_state_dict(self.global_model.state_dict())
            
            self.client_models[client_id] = client_model
        
        print(f"已向 {len(client_ids)} 个客户端分发模型")
    
    def federated_round(self, client_ids, client_data, epochs=5):
        """
        执行一轮联邦学习
        """
        # 分发全局模型
        self.distribute_global_model(client_ids)
        
        # 收集客户端更新
        client_updates = {}
        
        for client_id in client_ids:
            if client_id in self.client_models:
                # 客户端训练
                client_model = self.client_models[client_id]
                client_update = self._train_client_model(
                    client_model, client_data[client_id], epochs
                )
                
                client_updates[client_id] = client_update
        
        # 聚合更新
        if client_updates:
            self.aggregate_client_updates(client_updates)
        
        return client_updates
    
    def _train_client_model(self, model, data, epochs):
        """
        训练客户端模型
        """
        X, y = data
        dataset = TensorDataset(torch.FloatTensor(X), torch.LongTensor(y))
        dataloader = DataLoader(dataset, batch_size=32, shuffle=True)
        
        optimizer = optim.SGD(model.parameters(), lr=self.learning_rate)
        criterion = nn.CrossEntropyLoss()
        
        model.train()
        for epoch in range(epochs):
            for batch_X, batch_y in dataloader:
                optimizer.zero_grad()
                outputs = model(batch_X)
                loss = criterion(outputs, batch_y)
                loss.backward()
                optimizer.step()
        
        return {
            'model': model,
            'num_samples': len(X)
        }

class PrivacyPreservingFederatedLearning:
    def __init__(self, epsilon=1.0):
        self.epsilon = epsilon
        
    def add_differential_privacy_noise(self, model_updates, sensitivity=1.0):
        """
        添加差分隐私噪声
        """
        noisy_updates = {}
        
        for client_id, update in model_updates.items():
            noisy_model = type(update['model'])()
            noisy_model.load_state_dict(update['model'].state_dict())
            
            # 为每个参数添加噪声
            for param in noisy_model.parameters():
                noise = torch.normal(0, sensitivity/self.epsilon, param.shape)
                param.data += noise
            
            noisy_updates[client_id] = {
                'model': noisy_model,
                'num_samples': update['num_samples']
            }
        
        return noisy_updates
    
    def secure_aggregation(self, client_updates, threshold=3):
        """
        安全聚合
        """
        if len(client_updates) < threshold:
            raise ValueError(f"客户端数量不足，需要至少 {threshold} 个客户端")
        
        # 简化的安全聚合实现
        # 实际应用中应该使用更复杂的安全多方计算协议
        
        # 计算聚合结果
        total_samples = sum(update['num_samples'] for update in client_updates.values())
        
        aggregated_params = {}
        for name, param in list(client_updates.values())[0]['model'].named_parameters():
            aggregated_params[name] = torch.zeros_like(param.data)
        
        for update in client_updates.values():
            weight = update['num_samples'] / total_samples
            
            for name, param in update['model'].named_parameters():
                aggregated_params[name] += weight * param.data
        
        return aggregated_params
```

```python
import shap
import lime
import lime.lime_tabular
from sklearn.inspection import permutation_importance

import seaborn as sns

class ModelExplainability:
    def __init__(self, model, feature_names=None):
        self.model = model
        self.feature_names = feature_names
        
    def explain_with_shap(self, X, X_background=None):
        """
        使用SHAP解释模型
        """
        try:
            # 创建SHAP解释器
            if X_background is None:
                X_background = X[:100]  # 使用前100个样本作为背景
            
            explainer = shap.Explainer(self.model, X_background)
            shap_values = explainer(X)
            
            return {
                'shap_values': shap_values,
                'explainer': explainer,
                'feature_names': self.feature_names
            }
        except Exception as e:
            print(f"SHAP解释失败: {e}")
            return None
    
    def explain_with_lime(self, X, instance_index=0):
        """
        使用LIME解释模型
        """
        try:
            # 创建LIME解释器
            explainer = lime.lime_tabular.LimeTabularExplainer(
                X, 
                feature_names=self.feature_names,
                mode='classification'
            )
            
            # 解释单个实例
            explanation = explainer.explain_instance(
                X[instance_index], 
                self.model.predict_proba,
                num_features=len(self.feature_names)
            )
            
            return {
                'explanation': explanation,
                'instance_index': instance_index
            }
        except Exception as e:
            print(f"LIME解释失败: {e}")
            return None

        使用置换重要性解释模型
        """

            perm_importance = permutation_importance(
                self.model, X, y, n_repeats=n_repeats, random_state=42
            )

                'feature_index': feature_index,
```
