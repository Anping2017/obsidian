# 性能优化技巧

## 概述

性能优化是AI系统开发中的重要环节，涉及模型性能、推理速度、内存使用、计算效率等多个方面。通过系统化的性能优化，可以显著提升AI系统的效率和用户体验。本章将介绍性能优化的核心概念、技术方法和实践技巧。

## 核心概念

### 1. 性能优化定义
- **目标**：提升AI系统的整体性能，包括速度、精度、资源利用率
- **关键指标**：推理延迟、吞吐量、内存使用、CPU/GPU利用率
- **优化层次**：算法优化、模型优化、系统优化、硬件优化
- **平衡考虑**：性能与精度、速度与资源、开发与维护成本

### 2. 性能瓶颈分析
- **计算瓶颈**：模型复杂度、算法效率
- **内存瓶颈**：模型大小、数据加载、缓存策略
- **I/O瓶颈**：数据读取、模型加载、结果输出
- **网络瓶颈**：数据传输、分布式计算

```python
class ModelQuantizer:
    def __init__(self):
        self.quantization_configs = {}

        """
        量化模型
        """
        if method == 'dynamic':
            return self._dynamic_quantization(model)
        elif method == 'static':
            return self._static_quantization(model, calibration_data)
        elif method == 'qat':

class ModelPruner:
    def __init__(self):
        self.pruning_configs = {}
        
    def prune_model(self, model, pruning_ratio=0.2, method='magnitude'):
        """
        剪枝模型
        """
        if method == 'magnitude':
            return self._magnitude_pruning(model, pruning_ratio)
        elif method == 'gradient':
            return self._gradient_pruning(model, pruning_ratio)
        elif method == 'random':
            return self._random_pruning(model, pruning_ratio)
        else:
            raise ValueError(f"不支持的剪枝方法: {method}")
    
    def _magnitude_pruning(self, model, pruning_ratio):
        """
        基于幅度的剪枝
        """
        try:
            # 获取需要剪枝的参数
            parameters_to_prune = []
            for name, module in model.named_modules():
                if isinstance(module, torch.nn.Linear):
                    parameters_to_prune.append((module, 'weight'))
            
            # 应用剪枝
            quantization.prune.global_unstructured(
                parameters_to_prune,
                pruning_method=quantization.prune.L1Unstructured,
                amount=pruning_ratio,
            )
            
            print(f"幅度剪枝完成，剪枝比例: {pruning_ratio}")
            return model
            
        except Exception as e:
            print(f"幅度剪枝失败: {e}")
            raise
    
    def _gradient_pruning(self, model, pruning_ratio):
        """
        基于梯度的剪枝
        """
        try:
            # 计算梯度
            model.train()
            for param in model.parameters():
                if param.grad is not None:
                    param.grad.data.zero_()
            
            # 这里需要实际的训练数据来计算梯度
            # 简化实现
            parameters_to_prune = []
            for name, module in model.named_modules():
                if isinstance(module, torch.nn.Linear):
                    parameters_to_prune.append((module, 'weight'))
            
            # 应用剪枝
            quantization.prune.global_unstructured(
                parameters_to_prune,
                pruning_method=quantization.prune.L2Unstructured,
                amount=pruning_ratio,
            )
            
            print(f"梯度剪枝完成，剪枝比例: {pruning_ratio}")
            return model
            
        except Exception as e:
            print(f"梯度剪枝失败: {e}")
            raise
    
    def _random_pruning(self, model, pruning_ratio):
        """
        随机剪枝
        """
        try:
            parameters_to_prune = []
            for name, module in model.named_modules():
                if isinstance(module, torch.nn.Linear):
                    parameters_to_prune.append((module, 'weight'))
            
            # 应用剪枝
            quantization.prune.global_unstructured(
                parameters_to_prune,
                pruning_method=quantization.prune.RandomUnstructured,
                amount=pruning_ratio,
            )
            
            print(f"随机剪枝完成，剪枝比例: {pruning_ratio}")
            return model
            
        except Exception as e:
            print(f"随机剪枝失败: {e}")
            raise
    
    def iterative_pruning(self, model, target_sparsity=0.8, num_iterations=5):
        """
        迭代剪枝
        """
        try:
            current_model = model
            
            for iteration in range(num_iterations):
                current_sparsity = self._calculate_sparsity(current_model)
                
                if current_sparsity >= target_sparsity:
                    break
                
                # 计算当前迭代的剪枝比例
                remaining_sparsity = target_sparsity - current_sparsity
                pruning_ratio = remaining_sparsity / (num_iterations - iteration)
                
                # 剪枝
                current_model = self._magnitude_pruning(current_model, pruning_ratio)
                
                print(f"迭代 {iteration + 1}: 稀疏度 {self._calculate_sparsity(current_model):.3f}")
            
            return current_model
            
        except Exception as e:
            print(f"迭代剪枝失败: {e}")
            raise
    
    def _calculate_sparsity(self, model):
        """
        计算模型稀疏度
        """
        total_params = 0
        zero_params = 0
        
        for param in model.parameters():
            total_params += param.numel()
            zero_params += (param == 0).sum().item()
        
        return zero_params / total_params if total_params > 0 else 0

class ModelDistiller:
    def __init__(self):
        self.distillation_configs = {}
        
    def distill_model(self, teacher_model, student_model, train_data, 
                     temperature=3.0, alpha=0.7):
        """
        知识蒸馏
        """
        try:
            student_model.train()
            teacher_model.eval()
            
            optimizer = torch.optim.Adam(student_model.parameters(), lr=0.001)
            
            for epoch in range(50):
                total_loss = 0
                
                for data, target in train_data:
                    optimizer.zero_grad()
                    
                    # 教师模型预测
                    with torch.no_grad():
                        teacher_output = teacher_model(data)
                        teacher_soft = torch.softmax(teacher_output / temperature, dim=1)
                    
                    # 学生模型预测
                    student_output = student_model(data)
                    student_soft = torch.log_softmax(student_output / temperature, dim=1)
                    
                    # 计算蒸馏损失
                    distillation_loss = torch.nn.functional.kl_div(
                        student_soft, teacher_soft, reduction='batchmean'
                    ) * (temperature ** 2)
                    
                    # 计算学生损失
                    student_loss = torch.nn.functional.cross_entropy(student_output, target)
                    
                    # 总损失
                    total_loss = alpha * distillation_loss + (1 - alpha) * student_loss
                    
                    total_loss.backward()
                    optimizer.step()
                
                if epoch % 10 == 0:
                    print(f"Epoch {epoch}: Loss {total_loss.item():.4f}")
            
            print("知识蒸馏完成")
            return student_model
            
        except Exception as e:
            print(f"知识蒸馏失败: {e}")
            raise
    
    def compare_models(self, teacher_model, student_model, test_data):
        """
        比较教师和学生模型
        """
        teacher_model.eval()
        student_model.eval()
        
        teacher_correct = 0
        student_correct = 0
        total = 0
        
        with torch.no_grad():
            for data, target in test_data:
                teacher_output = teacher_model(data)
                student_output = student_model(data)
                
                teacher_pred = teacher_output.argmax(dim=1)
                student_pred = student_output.argmax(dim=1)
                
                teacher_correct += (teacher_pred == target).sum().item()
                student_correct += (student_pred == target).sum().item()
                total += target.size(0)
        
        teacher_accuracy = teacher_correct / total
        student_accuracy = student_correct / total
        
        return {
            'teacher_accuracy': teacher_accuracy,
            'student_accuracy': student_accuracy,
            'accuracy_drop': teacher_accuracy - student_accuracy
        }
```

```python
import torch.nn as nn
import torch.nn.functional as F

class EfficientCNN(nn.Module):
    def __init__(self, num_classes=10):
        super(EfficientCNN, self).__init__()
        
        # 使用深度可分离卷积
        self.conv1 = nn.Conv2d(3, 32, 3, padding=1)
        self.bn1 = nn.BatchNorm2d(32)
        
        # 深度可分离卷积块
        self.depthwise_conv1 = nn.Conv2d(32, 32, 3, padding=1, groups=32)
        self.pointwise_conv1 = nn.Conv2d(32, 64, 1)
        self.bn2 = nn.BatchNorm2d(64)
        
        self.depthwise_conv2 = nn.Conv2d(64, 64, 3, padding=1, groups=64)
        self.pointwise_conv2 = nn.Conv2d(64, 128, 1)
        self.bn3 = nn.BatchNorm2d(128)
        
        # 全局平均池化
        self.global_avg_pool = nn.AdaptiveAvgPool2d(1)
        
        # 分类器
        self.classifier = nn.Linear(128, num_classes)
        
    def forward(self, x):
        # 第一个卷积块
        x = F.relu(self.bn1(self.conv1(x)))
        x = F.max_pool2d(x, 2)
        
        # 深度可分离卷积块1
        x = F.relu(self.bn2(self.pointwise_conv1(self.depthwise_conv1(x))))
        x = F.max_pool2d(x, 2)
        
        # 深度可分离卷积块2
        x = F.relu(self.bn3(self.pointwise_conv2(self.depthwise_conv2(x))))
        x = F.max_pool2d(x, 2)
        
        # 全局平均池化
        x = self.global_avg_pool(x)
        x = x.view(x.size(0), -1)
        
        # 分类
        x = self.classifier(x)
        
        return x

class MobileNetBlock(nn.Module):
    def __init__(self, in_channels, out_channels, stride=1):
        super(MobileNetBlock, self).__init__()
        
        # 深度可分离卷积
        self.depthwise = nn.Conv2d(in_channels, in_channels, 3, stride, 1, groups=in_channels)
        self.pointwise = nn.Conv2d(in_channels, out_channels, 1)
        
        self.bn1 = nn.BatchNorm2d(in_channels)
        self.bn2 = nn.BatchNorm2d(out_channels)
        
        self.stride = stride
        
    def forward(self, x):
        # 深度卷积
        x = F.relu(self.bn1(self.depthwise(x)))
        
        # 点卷积
        x = F.relu(self.bn2(self.pointwise(x)))
        
        return x

class EfficientTransformer(nn.Module):
    def __init__(self, d_model=512, nhead=8, num_layers=6):
        super(EfficientTransformer, self).__init__()
        
        self.d_model = d_model
        self.nhead = nhead
        
        # 多头注意力
        self.attention = nn.MultiheadAttention(d_model, nhead, batch_first=True)
        
        # 前馈网络
        self.feed_forward = nn.Sequential(
            nn.Linear(d_model, d_model * 4),
            nn.ReLU(),
            nn.Linear(d_model * 4, d_model)
        )
        
        # 层归一化
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        
        # Dropout
        self.dropout = nn.Dropout(0.1)
        
    def forward(self, x):
        # 自注意力
        attn_output, _ = self.attention(x, x, x)
        x = self.norm1(x + self.dropout(attn_output))
        
        # 前馈网络
        ff_output = self.feed_forward(x)
        x = self.norm2(x + self.dropout(ff_output))
        
        return x
```

```python
import torch
import numpy as np
from torch.utils.data import DataLoader, TensorDataset
import time

class BatchProcessor:
    def __init__(self, model, batch_size=32):
        self.model = model
        self.batch_size = batch_size
        self.model.eval()
        
    def process_batch(self, batch_data):
        """
        处理批次数据
        """
        with torch.no_grad():
            if isinstance(batch_data, torch.Tensor):
                return self.model(batch_data)
            else:
                # 处理不同格式的数据
                processed_batch = self._preprocess_batch(batch_data)
                return self.model(processed_batch)
    
    def _preprocess_batch(self, batch_data):
        """
        预处理批次数据
        """
        if isinstance(batch_data, list):
            # 转换为张量
            return torch.stack([torch.tensor(item) for item in batch_data])
        elif isinstance(batch_data, np.ndarray):
            # 转换为张量
            return torch.from_numpy(batch_data).float()
        else:
            return batch_data
    
    def process_dataset(self, dataset, dataloader=None):
        """
        处理整个数据集
        """
        if dataloader is None:
            dataloader = DataLoader(dataset, batch_size=self.batch_size, shuffle=False)
        
        results = []
        
        for batch in dataloader:
            if isinstance(batch, (list, tuple)):
                batch_data = batch[0]
            else:
                batch_data = batch
            
            batch_result = self.process_batch(batch_data)
            results.append(batch_result)
        
        return torch.cat(results, dim=0)
    
    def benchmark_batch_sizes(self, test_data, batch_sizes=[1, 8, 16, 32, 64, 128]):
        """
        基准测试不同批次大小
        """
        results = {}
        
        for batch_size in batch_sizes:
            # 创建数据加载器
            dataloader = DataLoader(test_data, batch_size=batch_size, shuffle=False)
            
            # 预热
            for i, batch in enumerate(dataloader):
                if i >= 3:  # 预热3个批次
                    break
                self.process_batch(batch[0] if isinstance(batch, (list, tuple)) else batch)
            
            # 测试
            start_time = time.time()
            total_samples = 0
            
            for batch in dataloader:
                batch_data = batch[0] if isinstance(batch, (list, tuple)) else batch
                self.process_batch(batch_data)
                total_samples += len(batch_data)
            
            end_time = time.time()
            
            # 计算指标
            total_time = end_time - start_time
            throughput = total_samples / total_time
            
            results[batch_size] = {
                'total_time': total_time,
                'throughput': throughput,
                'avg_latency': total_time / total_samples
            }
        
        return results
    
    def optimize_batch_size(self, test_data, target_latency=None, target_throughput=None):
        """
        优化批次大小
        """
        # 基准测试
        benchmark_results = self.benchmark_batch_sizes(test_data)
        
        # 找到最优批次大小
        if target_latency:
            # 基于延迟优化
            best_batch_size = min(
                benchmark_results.keys(),
                key=lambda bs: abs(benchmark_results[bs]['avg_latency'] - target_latency)
            )
        elif target_throughput:
            # 基于吞吐量优化
            best_batch_size = max(
                benchmark_results.keys(),
                key=lambda bs: benchmark_results[bs]['throughput']
            )
        else:
            # 基于吞吐量优化（默认）
            best_batch_size = max(
                benchmark_results.keys(),
                key=lambda bs: benchmark_results[bs]['throughput']
            )
        
        return {
            'best_batch_size': best_batch_size,
            'benchmark_results': benchmark_results,
            'recommendation': f"推荐批次大小: {best_batch_size}"
        }

class InferenceOptimizer:
```

```python
import hashlib
import pickle
import os
from functools import wraps
import time

class ModelCache:
    def __init__(self, cache_dir="model_cache", max_size=1000):
        self.cache_dir = cache_dir
        self.max_size = max_size
        self.cache_index = {}
        self.access_times = {}
        
        # 创建缓存目录
        os.makedirs(cache_dir, exist_ok=True)
        
        # 加载缓存索引
        self._load_cache_index()
    
    def _load_cache_index(self):
        """
        加载缓存索引
        """
        index_file = os.path.join(self.cache_dir, "cache_index.pkl")
        if os.path.exists(index_file):
            with open(index_file, 'rb') as f:
                self.cache_index = pickle.load(f)
    
    def _save_cache_index(self):
        """
        保存缓存索引
        """
        index_file = os.path.join(self.cache_dir, "cache_index.pkl")
        with open(index_file, 'wb') as f:
            pickle.dump(self.cache_index, f)
    
    def _generate_cache_key(self, data):
        """
        生成缓存键
        """
        # 将数据转换为字符串并计算哈希
        data_str = str(data)
        return hashlib.md5(data_str.encode()).hexdigest()
    
    def get(self, data):
        """
        获取缓存数据
        """
        cache_key = self._generate_cache_key(data)
        
        if cache_key in self.cache_index:
            # 更新访问时间
            self.access_times[cache_key] = time.time()
            
            # 加载缓存数据
            cache_file = os.path.join(self.cache_dir, f"{cache_key}.pkl")
            if os.path.exists(cache_file):
                with open(cache_file, 'rb') as f:
                    return pickle.load(f)
        
        return None
    
    def set(self, data, result):
        """
        设置缓存数据
        """
        cache_key = self._generate_cache_key(data)
        
        # 检查缓存大小
        if len(self.cache_index) >= self.max_size:
            self._evict_oldest()
        
        # 保存缓存数据
        cache_file = os.path.join(self.cache_dir, f"{cache_key}.pkl")
        with open(cache_file, 'wb') as f:
            pickle.dump(result, f)
        
        # 更新索引
        self.cache_index[cache_key] = {
            'file': cache_file,
            'size': os.path.getsize(cache_file),
            'created_at': time.time()
        }
        
        self.access_times[cache_key] = time.time()
        
        # 保存索引
        self._save_cache_index()
    
    def _evict_oldest(self):
        """
        驱逐最旧的缓存
        """
        if not self.access_times:
            return
        
        # 找到最旧的缓存
        oldest_key = min(self.access_times.keys(), key=lambda k: self.access_times[k])
        
        # 删除文件
        if oldest_key in self.cache_index:
            cache_file = self.cache_index[oldest_key]['file']
            if os.path.exists(cache_file):
                os.remove(cache_file)
        
        # 从索引中删除
        del self.cache_index[oldest_key]
        del self.access_times[oldest_key]
    
    def clear(self):
        """
        清空缓存
        """
        # 删除所有缓存文件
        for cache_key in self.cache_index:
            cache_file = self.cache_index[cache_key]['file']
            if os.path.exists(cache_file):
                os.remove(cache_file)
        
        # 清空索引
        self.cache_index.clear()
        self.access_times.clear()
        
        # 保存索引
        self._save_cache_index()
    
    def get_cache_stats(self):
        """
        获取缓存统计信息
        """
        total_size = sum(info['size'] for info in self.cache_index.values())
        num_items = len(self.cache_index)
        
        return {
            'num_items': num_items,
            'total_size': total_size,
            'max_size': self.max_size,
            'hit_rate': self._calculate_hit_rate()
        }
    
    def _calculate_hit_rate(self):
        """
        计算命中率
        """
        # 简化实现，实际应用中需要跟踪命中次数
        return 0.0

def cached_inference(cache_dir="model_cache"):
    """
    缓存推理装饰器
    """
    def decorator(func):
        cache = ModelCache(cache_dir)
        
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 生成缓存键
            cache_key = str(args) + str(sorted(kwargs.items()))
            
            # 尝试从缓存获取
            cached_result = cache.get(cache_key)
            if cached_result is not None:
                return cached_result
            
            # 执行函数
            result = func(*args, **kwargs)
            
            # 保存到缓存
            cache.set(cache_key, result)
            
            return result
        
        return wrapper
    return decorator

class PredictionCache:
    def __init__(self, max_size=10000):
        self.cache = {}
        self.max_size = max_size
        self.access_order = []
    
    def get(self, key):
        """
        获取预测结果
        """
        if key in self.cache:
            # 更新访问顺序
            self.access_order.remove(key)
            self.access_order.append(key)
            return self.cache[key]
        return None
    
    def set(self, key, value):
        """
        设置预测结果
        """
        if len(self.cache) >= self.max_size:
            # 删除最旧的缓存
            oldest_key = self.access_order.pop(0)
            del self.cache[oldest_key]
        
        self.cache[key] = value
        self.access_order.append(key)
    
    def clear(self):
        """
        清空缓存
        """
        self.cache.clear()
        self.access_order.clear()
    
    def get_stats(self):
        """
        获取缓存统计
        """
        return {
            'size': len(self.cache),
            'max_size': self.max_size,
            'hit_rate': 0.0  # 简化实现
        }
```

```python
import gc

        监控内存使用

        # Python内存
        python_memory = psutil.Process().memory_info()
        
        # PyTorch内存（如果可用）
        torch_memory = {}
        if torch.cuda.is_available():

                'reserved': torch.cuda.memory_reserved(),
                'max_allocated': torch.cuda.max_memory_allocated()
            }

            'system': {
                'total': system_memory.total,
                'available': system_memory.available,
                'used': system_memory.used,
                'percentage': system_memory.percent
            },
            'python': {
                'rss': python_memory.rss,
                'vms': python_memory.vms
            },
            'torch': torch_memory
        }
        
        return self.memory_usage

        """
        检查内存阈值
        """

        if system_usage > self.memory_threshold:
            return True, f"内存使用率过高: {system_usage:.2%}"
        
        return False, "内存使用正常"

        优化内存使用
        """

        # 清理NumPy缓存
        try:
            np.seterr(all='ignore')
            optimizations.append("清理NumPy缓存")
        except:
            pass
        
        return optimizations
    
    def get_memory_recommendations(self):
        """
```

```python
import multiprocessing as mp
from concurrent.futures import ProcessPoolExecutor, ThreadPoolExecutor
import threading
import queue
import time

class ParallelProcessor:
    def __init__(self, num_processes=None, num_threads=None):
        self.num_processes = num_processes or mp.cpu_count()
        self.num_threads = num_threads or mp.cpu_count() * 2
        
    def process_data_parallel(self, data, process_func, use_threads=False):
        """
        并行处理数据
        """
        if use_threads:
            return self._process_with_threads(data, process_func)
        else:
            return self._process_with_processes(data, process_func)
    
    def _process_with_threads(self, data, process_func):
        """
        使用线程并行处理
        """
        results = []
        
        with ThreadPoolExecutor(max_workers=self.num_threads) as executor:
            futures = [executor.submit(process_func, item) for item in data]
            
            for future in futures:
                try:
                    result = future.result()
                    results.append(result)
                except Exception as e:
                    print(f"线程处理失败: {e}")
                    results.append(None)
        
        return results
    
    def _process_with_processes(self, data, process_func):
        """
        使用进程并行处理
        """
        results = []
        
        with ProcessPoolExecutor(max_workers=self.num_processes) as executor:
            futures = [executor.submit(process_func, item) for item in data]
            
            for future in futures:
                try:
                    result = future.result()
                    results.append(result)
                except Exception as e:
                    print(f"进程处理失败: {e}")
                    results.append(None)
        
        return results
    
    def benchmark_parallel_processing(self, data, process_func, num_runs=5):
        """
        基准测试并行处理
        """
        results = {}
        
        # 测试串行处理
        start_time = time.time()
        for item in data:
            process_func(item)
        serial_time = time.time() - start_time
        
        results['serial'] = {
            'time': serial_time,
            'throughput': len(data) / serial_time
        }
        
        # 测试线程并行
        start_time = time.time()
        self._process_with_threads(data, process_func)
        thread_time = time.time() - start_time
        
        results['threads'] = {
            'time': thread_time,
            'throughput': len(data) / thread_time,
            'speedup': serial_time / thread_time
        }
        
        # 测试进程并行
        start_time = time.time()
        self._process_with_processes(data, process_func)
        process_time = time.time() - start_time
        
        results['processes'] = {
            'time': process_time,
            'throughput': len(data) / process_time,
            'speedup': serial_time / process_time
        }
        
        return results

class PipelineOptimizer:
    def __init__(self):
        self.pipeline_stages = []
        self.stage_times = {}
        
    def add_stage(self, stage_name, stage_func):
        """
        添加流水线阶段
        """
        self.pipeline_stages.append({
            'name': stage_name,
            'function': stage_func
        })
    
    def execute_pipeline(self, data, parallel=False):
        """
        执行流水线
        """
        if parallel:
            return self._execute_parallel_pipeline(data)
        else:
            return self._execute_serial_pipeline(data)
    
    def _execute_serial_pipeline(self, data):
        """
        串行执行流水线
        """
        result = data
        
        for stage in self.pipeline_stages:
            start_time = time.time()
            result = stage['function'](result)
            stage_time = time.time() - start_time
            
            self.stage_times[stage['name']] = stage_time
        
        return result
    
    def _execute_parallel_pipeline(self, data):
        """
        并行执行流水线
        """
        # 简化实现，实际应用中需要更复杂的并行策略
        return self._execute_serial_pipeline(data)
    
    def optimize_pipeline(self, data):
        """
        优化流水线
        """
        # 分析每个阶段的性能
        self._execute_serial_pipeline(data)
        
        # 识别瓶颈
        bottlenecks = self._identify_bottlenecks()
        
        # 生成优化建议
        recommendations = self._generate_optimization_recommendations(bottlenecks)
        
        return {
            'stage_times': self.stage_times,
            'bottlenecks': bottlenecks,
            'recommendations': recommendations
        }
    
    def _identify_bottlenecks(self):
        """
        识别瓶颈
        """
        total_time = sum(self.stage_times.values())
        bottlenecks = []
        
        for stage_name, stage_time in self.stage_times.items():
            if stage_time / total_time > 0.3:  # 30%阈值
                bottlenecks.append({
                    'stage': stage_name,
                    'time': stage_time,
                    'percentage': stage_time / total_time * 100
                })
        
        return bottlenecks
    
    def _generate_optimization_recommendations(self, bottlenecks):
        """
        生成优化建议
        """
        recommendations = []
        
        for bottleneck in bottlenecks:
            recommendations.append(
                f"阶段 '{bottleneck['stage']}' 占用 {bottleneck['percentage']:.1f}% 的时间，建议优化"
            )
        
        return recommendations
```
