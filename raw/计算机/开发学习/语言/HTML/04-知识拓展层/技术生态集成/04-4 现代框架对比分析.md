### 🔧 JSX vs Template引擎

```jsx
// React JSX
function UserProfile({ user }) {
  return (
    <article className="profile-card">
      <header>
        <img src={user.avatar} alt={`${user.name}的头像`} />
        <h2>{user.name}</h2>
      </header>
      <main>
        <p>{user.bio}</p>
      </main>
    </article>
  )
}
```

```vue
<!-- Vue Template -->
<template>
  <article class="profile-card">
    <header>
      <img :src="user.avatar" :alt="`${user.name}的头像`" />
      <h2>{{ user.name }}</h2>
    </header>
    <main>
      <p>{{ user.bio }}</p>
    </main>
  </article>
</template>
```
