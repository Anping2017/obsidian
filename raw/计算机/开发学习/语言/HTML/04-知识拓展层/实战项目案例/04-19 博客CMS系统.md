        // 📖 博客功能和交互管理
        class BlogCMS {
            constructor() {
                this.posts = [];
                this.currentPage = 1;
                this.filters = {
                    category: '',
                    search: '',
                    sort: 'date-desc'
                };
                
                this.init();
            }
            
            init() {
                this.setupSearch();
                this.setupFilters();
                this.setupInfiniteScroll();
                this.setupSocialSharing();
                this.setupStats();
            }
            
            // 🔍 搜索功能
            setupSearch() {
                const searchInput = document.getElementById('search-input');
                const searchForm = searchInput.closest('form');
                
                let searchTimeout;
                
                searchInput.addEventListener('input', (e) => {
                    clearTimeout(searchTimeout);
                    const query = e.target.value.trim();
                    
                    if (query.length >= 2) {
                        searchTimeout = setTimeout(() => {
                            this.performSearch(query);
                        }, 300);
                    } else if (query.length === 0) {
                        this.clearSearch();
                    }
                });
                
                searchForm.addEventListener('submit', (e) => {
                    e.preventDefault();
                    const query = searchInput.value.trim();
                    if (query) {
                        this.performSearch(query);
                    }
                });
            }
            
            async performSearch(query) {
                this.filters.search = query;
                
                // 高亮搜索结果
                this.highlightSearchResults(query);
                
                // 显示搜索状态
                this.showSearchStatus(query);
                
                // 模拟API调用
                try {
                    const response = await this.searchPosts(query);
                    this.displaySearchResults(response.results, response.total);
                } catch (error) {
                    console.error('搜索失败:', error);
                    this.showErrorMessage('搜索失败，请稍后重试');
                }
            }
            
            highlightSearchResults(query) {
                const postCards = document.querySelectorAll('.post-card');
                postCards.forEach(card => {
                    const textElements = card.querySelectorAll('.post-title, .post-excerpt, .post-tags');
                    textElements.forEach(element => {
                        this.highlightText(element, query);
                    });
                });
            }
            
            highlightText(element, query) {
                const text = element.textContent;
                const regex = new RegExp(`(${query})`, 'gi');
                const highlightedText = text.replace(regex, '<mark>$1</mark>');
                
                if (highlightedText !== text) {
                    element.innerHTML = highlightedText;
                }
            }
            
            showSearchStatus(query) {
                const filterSection = document.querySelector('.filter-section');
                let statusElement = document.querySelector('.ssearch-status');
                
                if (!statusElement) {
                    statusElement = document.createElement('div');

                                   new Date(b.querySelector('.post-date').dateTime);

                            return new Date(b.querySelector('.post-date').dateTime) - 
                                   new Date(a.querySelector('.post-date').dateTime);
                        case 'views-desc':
                            return parseInt(b.querySelector('.views').textContent) - 
                                   parseInt(a.querySelector('.views').textContent);
                        case 'likes-desc':
                            return parseInt(b.querySelector('.likes span').textContent) - 
                                   parseInt(a.querySelector('.likes span').textContent);

            throttle(func, delay) {
                let timeoutId;
                let lastExecTime = 0;
                
                return function (...args) {
                    const currentTime = Date.now();
                    
                    if (currentTime - lastExecTime > delay) {
                        func.apply(this, args);
                        lastExecTime = currentTime;
                    } else {
                        clearTimeout(timeoutId);
                        timeoutId = setTimeout(() => {
                            func.apply(this, args);
                            lastExecTime = Date.now();
                        }, delay - (currentTime - lastExecTime));
                    }
                };
