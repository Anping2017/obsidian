        def tcp_client_example():
            """TCP客户端示例"""
            
            def simple_tcp_client():
                """简单TCP客户端"""
                
                # 创建socket对象
                client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
                
                try:
                    # 连接到服务器
                    client_socket.connect(('127.0.0.1', 8080))

                except ConnectionRefusedError:

                except Exception as e:
                    print(f"连接错误: {e}")
                finally:
                    client_socket.close()
            
            simple_tcp_client()

                try:
                    # 绑定地址和端口
                    server_socket.bind(('127.0.0.1', 8080))
                    server_socket.listen(5)
                    
                    print("服务器启动，等待连接...")
                    
                    while True:
                        # 接受客户端连接
                        client_socket, address = server_socket.accept()
                        print(f"客户端连接: {address}")
                        
                        try:
                            # 接收客户端数据
                            data = client_socket.recv(1024)
                            print(f"接收数据: {data.decode('utf-8')}")
                            
                            # 发送响应
                            response = "Hello, Client!"
                            client_socket.send(response.encode('utf-8'))
                            
                        except Exception as e:
                            print(f"处理客户端错误: {e}")
                        finally:
                            client_socket.close()
                            
                except Exception as e:
                    print(f"服务器错误: {e}")
                finally:
                    server_socket.close()

                    for attempt in range(max_retries):
                        try:
                            response = requests.get(url, timeout=10)
                            response.raise_for_status()
                            return response
                        except requests.RequestException as e:
                            print(f"尝试 {attempt + 1}/{max_retries} 失败: {e}")
                            if attempt < max_retries - 1:
                                time.sleep(delay)
                            else:
                                raise e
                
                try:
                    response = retry_request('https://httpbin.org/get')
                    print(f"重试成功，状态码: {response.status_code}")

                except Exception as e:
                    print(f"SSL验证错误: {e}")
            
            verify_ssl_certificate()

        def secure_requests():
            """安全请求"""
            
            def safe_request_with_timeout():
                """安全超时请求"""
                
                # 设置分层超时
                timeout_config = (10, 30)  # (连接超时, 读取超时)
                
                try:
                    response = requests.get(
                        'https://httpbin.org/delay/2',
                        timeout=timeout_config,
                        stream=True  # 流式响应
                    )
                    
                    print(f"分层超时状态码: {response.status_code}")
                    
                    # 流式读取
                    for chunk in response.iter_content(chunk_size=1024):
                        if chunk:
                            break  # 读取第一块就停止
                    
                except requests.exceptions.Timeout as e:
                    print(f"超时错误: {e}")

class NetworkOptimization:
    """网络性能优化"""
    
    @staticmethod
    def connection_pooling():
        """连接池管理"""
        
        def requests_session_pool():
            """requests会话池"""
            
            # 创建会话池
            session = requests.Session()
            
            # 配置连接池
            adapter = requests.adapters.HTTPAdapter(
                pool_connections=10,   # 连接池数量
                pool_maxsize=20,      # 最大连接数
                max_retries=3         # 最大重试次数
            )
            
            session.mount('http://', adapter)
            session.mount('https://', adapter)
            
            # 批量请求测试
            urls = [
                'https://httpbin.org/get?id=1',
                'https://httpbin.org/get?id=2',
                'https://httpbin.org/get?id=3'
            ]
            
            try:
                start_time = time.time()
                
                # 批量请求
                responses = []
                for url in urls:
                    response = session.get(url, timeout=10)
                    responses.append(response)
                
                end_time = time.time()
                print(f"批量请求耗时: {end_time - start_time:.2f}秒")
                print(f"成功请求数: {len(responses)}")
                
            except Exception as e:
                print(f"连接池测试错误: {e}")
            finally:
                session.close()
        
        requests_session_pool()
    
    @staticmethod
    def async_batch_processing():
        """异步批量处理"""
        
        async def async_batch_requests():
            """异步批量请求"""
            
            urls = [
                'https://httpbin.org/get?id=1',
                'https://httpbin.org/get?id=2',
                'https://httpbin.org/get?id=3',
                'https://httpbin.org/get?id=4',
                'https://httpbin.org/get?id=5'
            ]
            
            async def fetch_url(session, url):
                """获取单个URL"""
                try:
                    async with session.get(url) as response:
                        data = await response.json()
                        return data['url']
                except Exception as e:
                    print(f"请求 {url} 失败: {e}")
                    return None
            
            try:
                start_time = time.time()
                
                async with aiohttp.ClientSession() as session:
                    # 并发请求
                    tasks = [fetch_url(session, url) for url in urls]
                    results = await asyncio.gather(*tasks)
                
                end_time = time.time()
                print(f"异步批量请求耗时: {end_time - start_time:.2f}秒")
                print(f"成功结果数: {len([r for r in results if r])}")
                
            except Exception as e:
                print(f"异步批量处理错误: {e}")
        
        try:
            asyncio.run(async_batch_requests())
        except Exception as e:

```python
# ✅ 网络请求封装器
class NetworkClient:
    """网络请求封装器"""
    
    def __init__(self, base_url: str = '', timeout: int = 10):
        self.base_url = base_url
        self.session = requests.Session()
        self.session.timeout = timeout
    
    def get(self, endpoint: str, **kwargs):
        """GET请求"""
        url = self.base_url + endpoint
        return self.session.get(url, **kwargs)
    
    def post(self, endpoint: str, data=None, json=None, **kwargs):
        """POST请求"""
        url = self.base_url + endpoint
        return self.session.post(url, data=data, json=json, **kwargs)
    
    def close(self):
        """关闭会话"""
        self.session.close()
```
