# 3.1.1 Python内存模型

## 🎯 核心理念

Python内存模型是"自动化的艺术"。理解它的工作原理，就能写出更高效的Python代码，避免常见的内存陷阱。

### 🧠 Python对象内存布局

```python
import sys
import gc
from weakref import WeakValueDictionary

class PythonMemoryModel:
    """Python内存模型分析"""
    
    @staticmethod
    def object_layout_analysis():
        """对象布局分析"""
        
        # 小整数对象池
        print("=== 小整数对象池分析 ===")
        a = 256
        b = 256
        print(f"256 is 256: {a is b}")  # True - 小整数池
        
        c = 257
        d = 257
        print(f"257 is 257: {c is d}")  # False - 大整数
        
        # 字符串实习（intern）
        print("\n=== 字符串实习分析 ===")
        s1 = "hello"
        s2 = "hello"
        print(f"短字符串 is: {s1 is s2}")
        
        # 空容器复用
        print("\n=== 空容器分析 ===")
        empty_list = []
        empty_tuple = ()
        print(f"空列表内存: {sys.getsizeof(empty_list)}")
        print(f"空元组内存: {sys.getsizeof(empty_tuple)}")
    
    @staticmethod
    def memory_allocation_patterns():
        """内存分配模式"""
        
        def analyze_list_allocation():
            """列表分配模式"""
            lst = []
            initial_size = sys.getsizeof(lst)
            
            print("列表扩容模式:")
            for i in range(20):
                lst.append(i)
                current_size = sys.getsizeof(lst)
                if current_size != initial_size:
                    print(f"长度{len(lst)}: {initial_size} -> {current_size} 字节")
                    initial_size = current_size
        
        def analyze_dict_allocation():
            """字典分配模式"""
            d = {}
            print("\n字典分配模式:")
            
            for i in range(20):
                d[i] = f"value_{i}"
                if i in [0, 5, 10, 15, 19]:
                    print(f"长度{len(d)}: {sys.getsizeof(d)} 字节")
        
        analyze_list_allocation()
        analyze_dict_allocation()
    
    @staticmethod
    def garbage_collection_model():
        """垃圾回收模型"""
        
        def generation_gc_demo():
            """分代垃圾回收集制"""
            # 启用垃圾回收调试
            gc.set_debug(gc.DEBUG_STATS)
            
            print("=== 分代垃圾回收 ===")
            
            # 创建对象进入不同代
            objects_by_generation = []
            
            for generation in range(3):
                # 每代创建一些对象
                generation_objects = []
                for i in range(100):
                    obj = {"generation": generation, "id": i}
                    generation_objects.append(obj)
                
                objects_by_generation.append(generation_objects)
                
                # 触发垃圾回收
                collected = gc.collect(generation)
                print(f"第{generation}代回收了 {collected} 个对象")
                    
                # 显示各代对象计数
                counts = gc.get_count()
                print(f"各代对象计数: {counts}")
            
            # 强制完整回收
            final_collected = gc.collect()
            print(f"完整回收收集了 {final_collected} 个对象")
        
        generation_gc_demo()

# ✅ 高级内存主题
class AdvancedMemoryTopics:
    """高级内存主题"""
    
    @staticmethod
    def circular_reference_resolution():
        """循环引用解决方案"""
        
        class Node:
            """可能产生循环引用的节点"""
            def __init__(self, value):
                self.value = value
                self.children = []
                self.parent = None
            
            def add_child(self, child):
                child.parent = self
                self.children.append(child)
        
        print("=== 循环引用分析 ===")
        
        # 创建循环引用
        parent = Node("parent")
        child1 = Node("child1")
        child2 = Node("child2")
        
        parent.add_child(child1)
        parent.add_child(child2)
        
        # 查看引用计数
        print(f"Parent引用计数: {sys.getrefcount(parent)}")
        print(f"Child1引用计数: {sys.getrefcount(child1)}")
        
        # 删除外部引用
        del parent, child1, child2
        
        # 检查垃圾回收
        collected = gc.collect()
        print(f"垃圾回收收集了 {collected} 个对象")
    
    @staticmethod
    def weak_references():
        """弱引用应用"""
        
        print("=== 弱引用分析 ===")
        
        class Data:
            def __init__(self, value):
                self.value = value
            
            def __repr__(self):
                return f"Data({self.value})"
        
        # 创建弱引用字典
        weak_dict = WeakValueDictionary()
        
        # 添加对象
        data1 = Data("important data")
        data2 = Data("temporary data")
        
        weak_dict["data1"] = data1
        weak_dict["data2"] = data2
        
        print(f"弱引用字典: {list(weak_dict.keys())}")
        
        # 删除data2引用
        del data2
        
        # 触发垃圾回收
        gc.collect()
        
        print(f"垃圾回收后: {list(weak_dict.keys())}")
    
    @staticmethod
    def memory_leak_detection():
        """内存泄漏检测"""
        
        import tracemalloc
        
        def memory_profiling_experiment():
            """内存分析实验"""
            
            tracemalloc.start()
            
            # 开始快照
            snapshot1 = tracemalloc.take_snapshot()
            
            # 创建一些数据
            data_structures = []
            for i in range(10):
                large_list = [j for j in range(i * 1000)]
                large_dict = {k: k*2 for k in range(i * 100)}
                data_structures.extend([large_list, large_dict])
            
            # 结束快照
            snapshot2 = tracemalloc.take_snapshot()
            
            # 分析内存使用
            top_stats = snapshot2.compare_to(snapshot1, 'lineno')
            
            print("=== 内存使用分析 ===")
            for stat in top_stats[:5]:
                print(f"{stat}")
        
        memory_profiling_experiment()
