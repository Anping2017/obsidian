class ErrorHandlingMastery:
    """错误处理精通技术"""
    
    @staticmethod
    def exception_hierarchy():
        """异常层次结构设计"""
        
        # 自定义异常基类
        class BusinessException(Exception):
            """业务异常基类"""
            def __init__(self, message: str, error_code: str = None, details: dict = None):
                super().__init__(message)
                self.message = message
                self.error_code = error_code or self.__class__.__name__
                self.details = details or {}
        
        # 具体业务异常
        class ValidationError(BusinessException):
            """验证异常"""
            pass
        
        class AuthorizationError(BusinessException):
            """授权异常"""
            pass
        
        class ResourceNotFoundError(BusinessException):
            """资源未找到异常"""
            pass
        
        class ExternalServiceError(BusinessException):
            """外部服务异常"""
            def __init__(self, message: str, service_name: str, **kwargs):
                super().__init__(message, **kwargs)
                self.service_name = service_name
        
        def business_logic_example():
            """业务逻辑示例"""
            def validate_user_id(user_id: str):
                if not user_id or not user_id.strip():
                    raise ValidationError("用户ID不能为空")
                
                if len(user_id) < 3:
                    raise ValidationError("用户ID长度不能少于3位", details={"current_length": len(user_id)})
            
            def fetch_user(user_id: str):
                validate_user_id(user_id)
                
                # 模拟数据库查找
                if user_id == "admin" and "auth_token" not in locals():
                    raise AuthorizationError("需要认证令牌")
                
                # 模拟资源未找到
                if user_id not in ["admin", "user1", "user2"]:
                    raise ResourceNotFoundError(f"用户 {user_id} 不存在", details={"user_id": user_id})
                
                return {"id": user_id, "name": f"User_{user_id}"}
            
            # 测试异常处理
            test_cases = ["", "ab", "user1", "unknown_user"]
            
            for user_id in test_cases:
                try:
                    user = fetch_user(user_id)
                    print(f"成功获取用户: {user}")
                except ValidationError as e:
                    print(f"验证失败: {e.error_code} - {e.message}")
                except AuthorizationError as e:
                    print(f"授权失败: {e.error_code} - {e.message}")
                except ResourceNotFoundError as e:
                    print(f"资源未找到: {e.error_code} - {e.message}")
                except Exception as e:
                    print(f"未知错误: {e}")
        
        business_logic_example()

### 📝 实践1：智能重试机制

```python
# ✅ 智能重试机制
import random
from functools import wraps
from typing import Callable, Any, List

def smart_retry(max_attempts: int = 3, 
                delay_range: tuple = (1, 3),
                backoff_factor: float = 2.0,
                exceptions: tuple = (Exception,)):
    """智能重试装饰器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs) -> Any:
            attempt = 0
            last_exception = None
            
            while attempt < max_attempts:
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    attempt += 1
                    
                    if attempt < max_attempts:
                        delay = random.uniform(*delay_range) * (backoff_factor ** attempt)
                        time.sleep(delay)
                        print(f"重试第 {attempt} 次，延迟 {delay:.2f} 秒")
            
            raise last_exception
        return wrapper
    return decorator

# 使用示例
@smart_retry(max_attempts=3, exceptions=(ConnectionError, TimeoutError))
def unreliable_api_call():
    if random.random() < 0.7:  # 70% 失败率
        raise ConnectionError("API连接失败")
    return "API调用成功"
```

```python
# ✅ 测试数据工厂
from dataclasses import dataclass
from typing import Optional

@dataclass
class User:
    id: str
    name: str
    email: str
    age: Optional[int] = None

class TestDataFactory:
    """测试数据工厂"""
    
    @staticmethod
    def create_user(**overrides):
        """创建用户测试数据"""
        defaults = {
            "id": "test_id",
            "name": "Test User",
            "email": "test@example.com",
            "age": 25
        }
        defaults.update(overrides)
        return User(**defaults)
    
    @staticmethod
    def create_user_list(count: int):
        """创建用户列表"""
        return [TestDataFactory.create_user(id=f"user_{ix}") for ix in range(count)]
```
