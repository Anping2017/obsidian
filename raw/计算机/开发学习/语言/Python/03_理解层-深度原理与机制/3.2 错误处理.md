# 3.2 错误处理

## 🎯 核心理念

错误处理是Python的"优雅失败艺术"。它不是隐藏错误，而是优雅地处理它们，让程序在异常情况下依然健壮可靠。

### 🏗️ Python异常体系

# ✅ 异常继承层次演示
def show_exception_hierarchy():
    """展示Python异常体系"""
    class ExceptionInspector:
        @staticmethod
        def inspect_exception_tree():
            """检查异常继承树"""
            import inspect
            
            # 基础异常类
            base_exceptions = [
                BaseException,
                Exception,
                ArithmeticError,
                LookupError,
                TypeError,
                ValueError,
            ]
            
            for exc in base_exceptions:
                print(f"{exc.__name__}: {exc.__doc__}")
                if exc.__subclasses__():
                    for sub in exc.__subclasses__()[:3]:  # 只显示前3个
                        print(f"  └─ {sub.__name__}")
                print()

# 现代异常处理最佳实践
class ErrorHandlingPatterns:
    """现代错误处理模式"""
    
    @staticmethod
    def comprehensive_error_handling():
        """完善的错误处理示例"""
        
        # ✅ Result模式实现
        from dataclasses import dataclass
        from typing import Union, Generic, TypeVar
        
        T = TypeVar('T')
        E = TypeVar('E')
        
        @dataclass
        class Result(Generic[T, E]):
            """函数式错误处理"""
            _value: Union[T, None] = None
            _error: Union[E, None] = None
            
            @classmethod
            def ok(cls, value: T) -> 'Result[T, E]':
                return cls(_value=value)
            
            @classmethod
            def err(cls, error: E) -> 'Result[T, E]':
                return cls(_error=error)
            
            def is_ok(self) -> bool:
                return self._error is None
            
            def is_err(self) -> bool:
                return self._error is not None
            
            def unwrap(self) -> T:
                if self.is_err():
                    raise ValueError(f"Called unwrap on Error: {self._error}")
                return self._value
            
            def map(self, func):
                """函数变换"""
                if self.is_ok():
                    try:
                        return Result.ok(func(self._value))
                    except Exception as e:
                        return Result.err(e)
                return self
        
        # ✅ 使用Result模式处理文件操作
        def safe_file_read(filename: str) -> Result[str, Exception]:
            """安全的文件读取"""
            try:
                with open(filename, 'r', encoding='utf-8') as f:
                    content = f.read()
                return Result.ok(content)
            except Exception as e:
                return Result.err(e)
        
        # ✅ 链式错误处理
        def process_text_content(filename: str) -> Result[int, Exception]:
            """处理文本内容"""
            return (safe_file_read(filename)
                   .map(str.lower)          # 转为小写
                   .map(str.strip)          # 去除空白
                   .map(lambda s: len(s)))  # 计算长度
        
        # 测试错误处理
        result = process_text_content("exampl.txt")  # 故意写错文件名
        
        if result.is_ok():
            print(f"文本长度: {result.unwrap()}")
        else:
            print(f"处理失败: {result._error}")
    
    @staticmethod
    def context_manager_patterns():
        """上下文管理器模式"""
        
        from contextlib import contextmanager, suppress, ExitStack
        import os
        
        # ✅ 自定义上下文管理器
        @contextmanager
        def temp_directory(name: str):
            """临时目录上下文管理器"""
            path = f"/tmp/{name}"
            os.makedirs(path, exist_ok=True)
            try:
                yield path
            finally:
                import shutil
                shutil.rmtree(path, ignore_errors=True)
        
        # ✅ 使用suppress忽略特定异常
        def safe_remove_files(files: list):
            """安全删除文件"""
            for file in files:
                with suppress(FileNotFoundError, PermissionError):
                    os.remove(file)
                    print(f"已删除: {file}")
        
        # ✅ ExitStack管理多个上下文
        def multiple_context_example():
            """多上下文管理示例"""
            with ExitStack() as stack:
                contexts = [
                    temp_directory("dir1"),
                    temp_directory("dir2")
                ]
                
                paths = []
                for ctx in contexts:
                    paths.append(stack.enter_context(ctx))
                
                print(f"创建的目录: {paths}")
                
                # 临时文件操作
                for path in paths:
                    test_file = os.path.join(path, "test.txt")
                    with open(test_file, 'w') as f:
                        f.write("test content")

# 测试现代异常处理
ErrorHandlingPatterns.comprehensive_error_handling()
ErrorHandlingPatterns.context_manager_patterns()

## 💡 费曼检验

**向新同事解释：Python的异常处理和其他语言有什么不同？为什么说异常处理是一个好的设计而不是错误？**

核心要点：
1. **异常不是错误**：异常是预期的异常情况，不是bug
2. **优雅失败**：让程序在异常时优雅退出，而不是崩溃
3. **信息传递**：异常携带了详细的错误信息
4. **控制流**：异常是一种特殊的控制流机制
