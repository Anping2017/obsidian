# 2.3.3 函数式编程技巧与陷阱

## 🎯 核心理念

函数式编程是"数学与编程的优雅结合"。掌握它的技巧和避免陷阱，能让我们写出更简洁、更可预测、更易测试的代码。

class FunctionalProgrammingTips:
    """函数式编程技巧"""
    
    @staticmethod
    def advanced_functional_patterns():
        """高级函数式模式"""
        
        # ✅ 函数组合
        def compose(*functions):
            """函数组合: f ∘ g ∘ h"""
            def inner(x):
                return functools.reduce(lambda acc, f: f(acc), reversed(functions), x)
            return inner
        
        # ✅ 管道操作
        def pipe(value, *functions):
            """管道操作"""
            return compose(*functions)(value)
        
        # 示例函数
        add_one = lambda x: x + 1
        square = lambda x: x * x
        double = lambda x: x * 2
        
        # 组合: square(add_one(double(x)))
        composed = compose(square, add_one, double)
        
        # 管道: 2 → 4 → 5 → 25
        pipeline_result = pipe(2, double, add_one, square)
        
        print(f"组合结果 (2): {composed(2)}")
        print(f"管道结果 (2): {pipeline_result}")
    
    @staticmethod
    def currying_and_partial():
        """柯里化与偏应用"""
        
        # ✅ 柯里化函数
        def curry(func, arity=None):
            """通用柯里化函数"""
            if arity is None:
                arity = func.__code__.co_argcount
            
            def curried(*args):
                if len(args) == arity:
                    return func(*args)
                else:
                    return curry(lambda *more_args: func(*(args + more_args)), 
                               arity - len(args))
            return curried
        
        # ✅ 偏应用示例
        def greet(greeting, name, punctuation="!"):
            return f"{greeting}, {name}{punctuation}"
        
        curried_greet = curry(greet)
        say_hello = curried_greet("Hello")
        say_hi_alice = curried_greet("Hi")("Alice")
        
        print(f"Hello Bob: {say_hello('Bob')}")
        print(f"Hi Alice: {say_hi_alice}")
        
        # ✅ functools.partial的使用
        from functools import partial
        
        multiply_by = lambda base, factor: base * factor
        multiply_by_2 = partial(multiply_by, factor=2)
        multiply_by_5 = partial(multiply_by, factor=5)
        
        print(f"2 * 3 = {multiply_by_2(3)}")
        print(f"5 * 4 = {multiply_by_5(4)}")

    @staticmethod
    def monad_like_patterns():
        """类Monad模式"""
        
        # ✅ Maybe/Option模式
        class Maybe:
            """Maybe类型 - 处理可能为None的值"""
            
            def __init__(self, value):
                self._value = value
                self._is_nothing = value is None
            
            @classmethod
            def just(cls, value):
                return cls(value)
            
            @classmethod
            def nothing(cls):
                return cls(None)
            
            def map(self, func):
                """映射操作"""
                if self._is_nothing:
                    return self
                return Maybe.just(func(self._value))
            
            def flatmap(self, func):
                """平展映射"""
                if self._is_nothing:
                    return self
                return func(self._value)
            
            def get_or_else(self, default):
                """获取值或默认值"""
                return self.get_or_else if not self._is_nothing else default
            
            def __str__(self):
                return f"Nothing" if self._is_nothing else f"Just({self._value})"
        
        # ✅ Result/Either模式
        class Result:
            """Result类型 - 处理成功或失败"""
            
            def __init__(self, value=None, error=None):
                self._value = value
                self.error = error
                self.is_success = error is None
            
            @classmethod
            def success(cls, value):
                return cls(value=value)
            
            @classmethod
            def failure(cls, error):
                return cls(error=error)
            
            def map(self, func):
                """映射成功值"""
                if not self.is_success:
                    return self
                try:
                    return Result.success(func(self._value))
                except Exception as e:
                    return Result.failure(e)
            
            def get_or_else(self, default):
                """获取成功值或默认值"""
                return self._value if self.is_success else default
        
        # ✅ Writer模式示例
        class Writer:
            """Writer类型 - 累积日志信息"""
            
            def __init__(self, value, logs=None):
                self._value = value
                self._logs = logs or []
            
            @classmethod
            def tell(cls, message):
                return cls(None, [message])
            
            def map(self, func, log_message=None):
                new_value = func(self._value)
                new_logs = self._logs[:]
                if log_message:
                    new_logs.append(log_message)
                return Writer(new_value, new_logs)
            
            def flatmap(self, func, log_message=None):
                other = func(self._value)
                combined_logs = self._logs + other._logs
                if log_message:
                    combined_logs.append(log_message)
                return Writer(other._value, combined_logs)
            
            def __str__(self):
                return f"Writer({self._value}, logs={self._logs})"

class FunctionalTrapsAndSolutions:
    """函数式编程陷阱与解决方案"""
    
    @staticmethod
    def common_functional_traps():
        """常见函数式陷阱"""
        
        # ❌ 陷阱1: 过度使用嵌套
        def bad_nested_style():
            """糟糕的嵌套风格"""
            data = list(range(100))
            
            # 过度嵌套，难以理解
            result = list(
                map(
                    lambda x: (x ** 2 if x % 2 == 0 else x) ** 2,
                    filter(
                        lambda x: x > 10 and x < 90,
                        sorted(data, reverse=True)
                    )
                )
            )
            return result
        
        # ✅ 解决方案: 分步处理
        def good_step_by_step():
            """良好的分步处理"""
            data = list(range(100))
            
            # 步骤1: 反转排序
            sorted_data = sorted(data, reverse=True)
            
            # 步骤2: 过滤
            filtered_data = [x for x in sorted_data if 10 < x < 90]
            
            # 步骤3: 变换
            transformed_data = [
                (x if x % 2 != 0 else x ** 2) for x in filtered_data
            ]
            
            # 步骤4: 最终变换
            final_result = [x ** 2 for x in transformed_data]
            
            return final_result
        
        # ❌ 陷阱2: 忽略异常处理
        def unsafe_functional_code():
            """不安全的函数式代码"""
            data = [1, 2, "3", 4, "invalid"]
            
            # 会抛出异常
            try:
                result = [x * 2 for x in data if isinstance(x, int)]
                return result
            except Exception as e:
                print(f"错误处理: {e}")
                return []
        
        # ✅ 解决方案: 安全的函数式处理
        def safe_functional_code():
            """安全的函数式代码"""
            data = [1, 2, "3", 4, "invalid"]
            
            def safe_multiply(x):
                try:
                    return x * 2 if isinstance(x, int) else None
                except Exception:
                    return None
            
            result = [
                x for item in data 
                for x in [safe_multiply(item)] 
                if x is not None
            ]
            
            return result
        
        # ❌ 陷阱3: 性能问题
        def inefficient_functional_code():
            """低效的函数式代码"""
            data = list(range(10000))
            
            # 多次遍历
            squared = [x ** 2 for x in data]
            even_squared = [x for x in squared if x as not 0]
            double_even = [x * 2 for x in even_squared]
            
            return sum(double_even)
        
        # ✅ 解决方案: 高效的单次遍历
        def efficient_functional_code():
            """高效的函数式代码"""
            data = list(range(10000))
            
            # 单次遍历解决问题
            result = sum(
                (x ** 2) * 2 
                for x in data 
                if (x ** 2) != 0
            )
            
            return result
        
        print("嵌套风格测试:")
        nested_result = bad_nested_style()
        step_result = good_step_by_step()
        print(f"嵌套结果长度: {len(nested_result)}")
        print(f"分步结果长度: {len(step_result)}")
        
        print("\n异常处理测试:")
        unsafe_result = unsafe_functional_code()
        safe_result = safe_functional_code()
        print(f"不安全结果: {unsafe_result}")
        print(f"安全结果: {safe_result}")
        
        print("\n性能测试:")
        import time
        
        start = time.perf_counter()
        inefficient_result = inefficient_functional_code()
        inefficient_time = time.perf_counter() - start
        
        start = time.perf_counter()
        efficient_result = efficient_functional_code()
        efficient_time = time.perf_counter() - start
        
        print(f"低效实现耗时: {inefficient_time:.4f}s")
        print(f"高效实现耗时: {efficient_time:.4f}s")
        print(f"性能提升: {inefficient_time/efficient_time:.1f}倍")
    
    @staticmethod
    def best_practices():
        """函数式编程最佳实践"""
        
        # ✅ 实践1: 纯函数设计
        def pure_function_example():
            """纯函数示例"""
            # 纯函数：相同输入总是产生相同输出，无副作用
            def pure_add(a, b):
                return a + b
            
            # 非纯函数：有副作用的例子
            def impure_counter():
                if not hasattr(impure_counter, 'count'):
                    impure_counter.count = 0
                impure_counter.count += 1
                return impure_counter.count
            
            print(f"纯函数结果: {pure_add(5, 3)}")
            print(f"第一次调用: {impure_counter()}")
            print(f"第二次调用: {impure_counter()}")
        
        # ✅ 实践2: 不可变性偏好
        def immutability_best_practices():
            """不可变性最佳实践"""
            
            # ❌ 可变数据结构
            def mutate_list(data):
                data.append(999)
                return data
            
            # ✅ 不可变操作
            def immutable_operation(data):
                return data + [999]
            
            original = [1, 2, 3]
            
            # 可变操作
            mutated = mutate_list(original.copy())
            print(f"原始列表: {original}")  # 如果传入引用，会被修改！
            print(f"变异后: {mutated}")
            
            # 不可变操作
            immutabel_result = immutable_operation(original)
            print(f"不可变操作结果: {immutable_result}")
        
        # ✅ 实践3: 组合优于继承
        def composition_over_inheritance():
            """组合优于继承"""
            
            # 基础行为
            def send_email(message):
                return f"Email sent: {message}"
            
            def send_sms(message):
                return f"SMS sent: {message}"
            
            def log_notification(message):
                return f"Logged: {message}"
            
            # 组合通知服务
            def create_notification_service(basic_sender, logging):
                def send_notification(message, log=True):
                    result = basic_sender(message)
                    if log:
                        logging(message)
                    return result
                return send_notification
            
            # 创建组合的服务
            email_service = create_notification_service(send_email, log_notification)
            sms_service = create_notification_service(send_sms, log_notification)
            
            print(email_service("Hello Alice!"))
            print(sms_service("Hello Bob!"))
        
        pure_function_example()
        immutability_best_practices()
        composition_over_inheritance()

## 💡 费曼检验

**解释：什么是函数式编程的"副作用"？为什么纯函数比非纯函数更容易测试和推理？**

核心要点：
1. **纯函数**：相同输入总是产生相同输出，无副作用
2. **副作用**：函数对外部状态的修改，如I/O操作、全局变量修改
3. **测试优势**：纯函数易于单元测试，不需要复杂的设置和清理
4. **推理优势**：纯函数的行为可以单独分析，不依赖外部环境
