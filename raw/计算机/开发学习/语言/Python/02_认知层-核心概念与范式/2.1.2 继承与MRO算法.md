# 2.1.2 继承与MRO算法

## 🎯 核心理念

Python的继承机制不仅仅是"子类继承父类"，而是通过MRO（Method Resolution Order）算法实现的多重继承解析，体现了"广度优先、左侧优先"的设计哲学。

### 🧠 Python继承层次与MRO

```python
# ✅ MRO算法演示
class MROAnalyser:
    """MRO算法分析器"""
    
    @staticmethod
    def demonstrate_simple_inheritance():
        """简单继承的MRO"""
        
        class Animal:
            def speak(self):
                return "Animal sound"
        
        class Dog(Animal):
            def speak(self):
                return "Woof!"
        
        print(f"Dog MRO: {Dog.__mro__}")
        # 输出: (<class '__main__.Dog'>, <class '__main__.Animal'>, <class 'object'>)
    
    @staticmethod
    def demonstrate_multiple_inheritance():
        """多重继承的MRO"""
        
        class A:
            def method(self):
                return "A.method()"
        
        class B:
            def method(self):
                return "B.method()"
        
        class C(A, B):  # 继承A和B
            pass
        
        class D(B, A):  # 继承B和A（顺序相反）
            pass
        
        obj_c = C()
        obj_d = D()
        
        print(f"C继承A,B: {obj_c.method()}")  # 调用A.method()
        print(f"D继承B,A: {obj_d.method()}")  # 调用B.method()
        print(f"C的MRO: {C.__mro__}")
    
    @staticmethod
    def diamond_inheritance_problem():
        """菱形继承问题"""
        
        class GrandParent:
            def method(self):
                return "GrandParent"
        
        class Parent1(GrandParent):
            def method(self):
                return "Parent1"
        
        class Parent2(GrandParent):
            def method(self):
                return "Parent2"
        
        class Child(Parent1, Parent2):
            pass
        
        obj = Child()
        print(f"Diamond inheritance result: {obj.method()}")
        print(f"Child MRO: {Child.__mro__}")
        
        # 验证不会调用GrandParent两次
        # MRO确保每个类在查找链中只出现一次

# C3算法的核心原理
class C3AlgorithmExplaination:
    """C3算法原理解释"""
    
    @staticmethod
    def c3_algorithm_demo():
        """C3算法演示"""
        
        def linearize(cls):
            """简化的C3线性化算法"""
            if not cls.__bases__:
                return [cls]
            
            # 获取所有父类的线性化结果
            parent_linearizations = [linearize(parent) for parent in cls.__bases__]
            parent_linearizations = parent_linearizations + [list(cls.__bases__)]
            
            merged = []
            while parent_linearizations:
                # 找到所有线性化序列的首项
                candidate = parent_linearizations[0][0]
                
                # 检查候选类是否在其他序列的非首位置
                for seq in parent_linearizations:
                    if candidate in seq[1:]:
                        break
                else:
                    # 候选类满足条件，添加到结果中
                    merged.append(candidate)
                    # 从所有序列中移除候选类
                    for seq in parent_linearizations:
                        if seq and seq[0] == candidate:
                            seq.pop(0)
                    # 移空空序列
                    parent_linearizations = [seq for seq in parent_linearizations if seq]
                    continue
                
                # 如果没有找到合适的候选类，说明存在冲突
                raise TypeError("Cannot resolve method resolution order")
            
            return merged + [cls]
        
        return linearize

# 高级继承模式
class AdvancedInheritancePatterns:
    """高级继承模式"""
    
    @staticmethod
    def metaclass_inheritance():
        """元类继承模式"""
        
        class SingletonMeta(type):
            """单例元类"""
            _instances = {}
            
            def __call__(cls, *args, **kwargs):
                if cls not in cls._instances:
                    cls._instances[cls] = super().__call__(*args, **kwargs)
                return cls._instances[cls]
        
        class MetaInheritanceExample(metaclass=SingletonMeta):
            def __init__(self, name):
                self.name = name
        
        # 验证单例行为
        obj1 = MetaInheritanceExample("test1")
        obj2 = MetaInheritanceExample("test2")
        print(f"Same instance: {obj1 is obj2}")
        print(f"Same data: {obj1.name is obj2.name}")
    
    @staticmethod
    def mixin_pattern():
        """Mixin模式实现"""
        
        class DebugMixin:
            """调试功能Mixin"""
            def debug_info(self):
                return f"{self.__class__.__name__}: {vars(self)}"
        
        class LoggingMixin:
            """日志功能Mixin"""
            def log(self, message):
                print(f"LOG: {message}")
        
        class CachedDataMixin:
            """缓存功能Mixin"""
            def __init__(self, *args, **kwargs):
                super().__init__(*args, **kwargs)
                self._cache = {}
            
            def cached_method(self, key):
                if key not in self._cache:
                    self._cache[key] = f"processed_{key}"
                return self._cache[key]
        
        # 组合多个Mixin
        class ProcessedClass(DebugMixin, LoggingMixin, CachedDataMixin):
            def __init__(self, data):
                super().__init__()
                self.data = data
                self.log(f"Created ProcessedClass with data: {data}")
        
        obj = ProcessedClass("test")
        print(obj.debug_info())
        print(obj.cached_method("example"))

## 💡 费曼检验

**解释：为什么Python的多重继承不会产生方法调用的歧义？MRO算法是如何解决"菱形问题"的？**

核心要点：
1. **C3算法**：确保继承图的线性化满足特定约束
2. **单调性**：子类的MRO是父类MRO的子序列
3. **局部优先**：同一类在MRO中优先级高于其父类
4. **无重复保证**：每个类在MRO中只出现一次
