# 2.2.2 迭代器与生成器机制

## 🎯 核心理念

迭代器是Python的"惰性求值艺术"。它让"按需产生"而不是"一次性存储"成为可能，体现了"用多少算多少"的智慧和内存效率的哲学。

### 🧠 迭代器协议深度解析

```python
class IteratorMechanism:
    """迭代器机制分析"""
    
    @staticmethod
    def iterator_protocol_deep_dive():
        """迭代器协议深入探讨"""
        
        # ✅ 自定义迭代器实现
        class CustomIterator:
            """自定义迭代器"""
            def __init__(self, data, step=1):
                self.data = data
                self.index = 0
                self.step = step
            
            def __iter__(self):
                return self
            
            def __next__(self):
                if self.index >= len(self.data):
                    raise StopIteration
                
                value = self.data[self.index]
                self.index += self.step
                return value
        
        # ✅ 可迭代对象但非迭代器
        class IterableCollection:
            """可迭代但非迭代器对象"""
            def __init__(self, items):
                self.items = items
            
            def __iter__(self):
                # 返回一个迭代器实例
                return CustomIterator(self.items)
        
        # 测试迭代器
        custom_iter = CustomIterator([1, 2, 3, 4, 5], step=2)
        print("自定义迭代器:")
        for item in custom_iter:
            print(item)
        
        # 测试可迭代对象
        collection = IterableCollection(['a', 'b', 'c', 'd', 'e'])
        print("\n可迭代对象:")
        for item in collection:
            print(item)
    
    @staticmethod
    def generator_fundamentals():
        """生成器基础"""
        
        # ✅ 函数生成器
        def fibonacci_generator(n):
            """斐波那契数列生成器"""
            a, b = 0, 1
            count = 0
            while count < n:
                yield a
                a, b = b, a + b
                count += 1
        
        # ✅ 生成器表达式
        squares_gen = (x**2 for x in range(10) if x % 2 == 0)
        
        print("斐波那契生成器:")
        for num in fibonacci_generator(10):
            print(num, end=" ")
        print()
        
        print("生成器表达式:")
        print(list(squares_gen))
    
    @staticmethod
    def advanced_generator_patterns():
        """高级生成器模式"""
        
        # ✅ 管道式生成器
        def data_pipeline():
            """数据管道生成器"""
            
            def read_data():
                """模拟数据读取"""
                data = [i for i in range(100)]
                for item in data:
                    yield item
            
            def filter_data(data_stream, condition):
                """过滤数据"""
                for item in data_stream:
                    if condition(item):
                        yield item
            
            def transform_data(data_stream, transformer):
                """转换数据"""
                for item in data_stream:
                    yield transformer(item)
            
            def process_data(data_stream, processor):
                """处理数据"""
                for item in data_stream:
                    yield processor(item)
            
            # 构建数据管道
            raw_data = read_data()
            filtered_data = filter_data(raw_data, lambda x: x % 2 == 0)
            transformed_data = transform_data(filtered_data, lambda x: x * 2)
            processed_data = process_data(transformed_data, lambda x: f"Processed: {x}")
            
            return processed_data
        
        # ✅ 生成器装饰器
        def generator_decorator(func):
            """生成器装饰器"""
            @functools.wraps(func)
            def wrapper(*args, **kwargs):
                gen = func(*args, **kwargs)
                yield from gen
            return wrapper
        
        @generator_decorator
        def decorator_generator(items):
            """装饰的生成器"""
            for item in items:
                yield f"Decorated: {item}"
        
        # 测试数据管道
        pipe = data_pipeline()
        print("数据管道结果:")
        for i, item in enumerate(pipe):
            if i < 10:  # 只显示前10个
                print(item)
            else:
                break
        
        # 测试装饰生成器
        print("\n装饰生成器:")
        decorated_gen = decorator_generator([1, 2, 3, 4, 5])
        for item in decorated_gen:
            print(item)

                yield item

        next(acc)  # 启动协程

        print("协程累计器:")

            next(acc)  # 准备接收下一个值

class ItertoolsAdvancedUsage:

            """无限序列生成器"""
            
            # 计数生成器
            counter = count(10, 2)  # 从10开始，步长为2
            print("计数生成器 (前5个):", [next(counter) for _ in range(5)])

            # 重复生成器

## 💡 费曼检验

**解释：为什么使用生成器比创建完整的列表更节省内存？迭代器和生成器有什么区别？在什么情况下应该使用生成器？**

核心要点：
1. **惰性求值**：生成器只在需要时计算下一个值，不预先生成整个序列
2. **内存效率**：生成器只保存当前状态，不需要存储整个结果集
3. **协议差异**：迭代器是实现了`__next__`的对象，生成器是特殊的迭代器
4. **适用场景**：大数据处理、无限序列、数据管道等场景的理想选择
