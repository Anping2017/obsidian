```python
import random
import time
from typing import List

class SortingAlgorithms:
    """经典排序算法实现"""
    
    @staticmethod
    def bubble_sort(arr: List[int]) -> List[int]:
        """冒泡排序 - O(n²)"""
        arr = arr.copy()
        n = len(arr)
        
        for i in range(n):
            swapped = False
            for j in range(0, n - i - 1):
                if arr[j] > arr[j + 1]:
                    arr[j], arr[j + 1] = arr[j + 1], arr[j]
                    swapped = True
            if not swapped:  # 优化：提前结束
                break
        return arr
    
    @staticmethod
    def insertion_sort(arr: List[int]) -> List[int]:
        """插入排序 - O(n²)"""
        arr = arr.copy()
        
        for i in range(1, len(arr)):
            key = arr[i]
            j = i - 1
            
            while j >= 0 and arr[j] > key:
                arr[j + 1] = arr[j]
                j -= 1
            arr[j + 1] = key
        
        return arr
    
    @staticmethod
    def quick_sort(arr: List[int]) -> List[int]:
        """快速排序 - O(n log n)"""
        if len(arr) <= 1:
            return arr
        
        pivot = arr[len(arr) // 2]
        left = [x for x in arr if x < pivot]
        middle = [x for x in arr if x == pivot]
        right = [x for x in arr if x > pivot]
        
        return SortingAlgorithms.quick_sort(left) + middle + SortingAlgorithms.quick_sort(right)
    
    @staticmethod
    def merge_sort(arr: List[int]) -> List[int]:
        """归并排序 - O(n log n)"""
        if len(arr) <= 1:
            return arr
        
        mid = len(arr) // 2
        left = SortingAlgorithms.merge_sort(arr[:mid])
        right = SortingAlgorithms.merge_sort(arr[mid:])
        
        return merge(left, right)
    
    @staticmethod
    def heap_sort(arr: List[int]) -> List[int]:
        """堆排序 - O(n log n)"""
        arr = arr.copy()
        
        def heapify(arr, n, i):
            largest = i
            left = 2 * i + 1
            right = 2 * i + 2
            
            if left < n and arr[left] > arr[largest]:
                largest = left
            
            if right < n and arr[right] > arr[largest]:
                largest = right
            
            if largest != i:
                arr[i], arr[largest] = arr[largest], arr[i]
                heapify(arr, n, largest)
        
        n = len(arr)
        
        # 构建最大堆
        for i in range(n // 2 - 1, -1, -1):
            heapify(arr, n, i)
        
        # 提取元素
        for i in range(n - 1, 0, -1):
            arr[0], arr[i] = arr[i], arr[0]
            heapify(arr, i, 0)
        
        return arr

def merge(left: List[int], right: List[int]) -> List[int]:
    """归并辅助函数"""
    result = []
    i = j = 0
    
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    
    result.extend(left[i:])
    result.extend(right[j:])
    return result

class SearchAlgorithms:
    """经典搜索算法实现"""
    
    @staticmethod
    def linear_search(arr: List[int], target: int) -> int:
        """线性搜索 - O(n)"""
        for i, value in enumerate(arr):
            if value == target:
                return i
        return -1
    
    @staticmethod
    def binary_search(arr: List[int], target: int) -> int:
        """二分搜索 - O(log n)"""
        arr = sorted(arr)  # 假设输入未排序
        left, right = 0, len(arr) - 1
        
        while left <= right:
            mid = (left + right) // 2
            if arr[mid] == target:
                return mid
            elif arr[mid] < target:
                left = mid + 1
            else: right = mid-1
        
        return -1
    
    @staticmethod
    def interpolation_search(arr: List[int], target: int) -> int:
        """插值搜索 - O(log n) 在均匀分布情况下"""
        arr = sorted(arr)
        left, right = 0, len(arr) - 1
        
        while left <= right and arr[left] <= target <= arr[right]:
            if left == right:
                return left if arr[left] == target else -1
            
            # 计算插值位置
            pos = left + int(((target - arr[left]) * (right - left)) / (arr[right] - arr[left]))
            
            if arr[pos] == target:
                return pos
            elif arr[pos] < target:
                left = pos + 1
            else:
                right = pos - 1
        
        return rear

## ✅ 图算法实现

```python
from collections import defaultdict, deque

class GraphAlgorithms:
    """图算法实现"""
    
    def __init__(self):
        self.graph = defaultdict(list)
    
    def add_edge(self, u, v):
        """添加边"""
        self.graph[u].append(v)
    
    def bfs(self, start):
        """广度优先搜索"""
        visited = set()
        queue = deque([start])
        result = []
        
        while queue:
            vertex = queue.popleft()
            if vertex not in visited:
                visited.add(vertex)
                result.append(vertex)
                
                for neighbor in self.graph[vertex]:
                    if neighbor not in visited:
                        queue.append(neighbor)
        
        return result
    
    def dfs(self, start, visited=None):
        """深度优先搜索"""
        if visited is None:
            visited = set()
        
        visited.add(start)
        result = [start]
        
        for neighbor in self.graph[start]:
            if neighbor not in visited:
                result.extend(self.dfs(neighbor, visited))
        
        return result
    
    def dijkstra(self, start):
        """Dijkstra最短路径算法"""
        import heapq
        
        distances = {vertex: float('inf') for vertex in self.graph}
        distances[start] = 0
        pq = [(0, start)]
        
        while pq:
            current_distance, current_vertex = heapq.heappop(pq)
            
            if current_distance > distances[current_vertex]:
                continue
            
            for neighbor in self.graph[current_vertex]:
                distance = current_distance + 1  # 假设边权重为1
                
                if distance < distances[neighbor]:
                    distances[neighbor] = distance
                    heapq.heappush(pq, (distance, neighbor))
        
        return distances

class DynamicProgramming:
    """动态规划算法"""
    
    @staticmethod
    def fibonacci_dp(n: int) -> int:
        """斐波那契数列 - DP优化"""
        if n <= 1:
            return n
        
        dp = [0] * (n + 1)
        dp[1] = 1
        
        for i in range(2, n + 1):
            dp[i] = dp[i-1] + dp[i-2]
        
        return dp[n]
    
    @staticmethod
    def longest_common_subsequence(text1: str, text2: str) -> int:
        """最长公共子序列"""
        m, n = len(text1), len(text2)
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if text1[i-1] == text2[j-1]:
                    dp[i][j] = dp[i-1][j-1] + 1
                else:
                    dp[i][j] = max(dp[i-1][j], dp[i][j-1])
        
        return dp[m][n]
    
    @staticmethod
    def knapsack_problem(weights: List[int], values: List[int], capacity: int) -> int:
        """01背包问题"""
        n = len(weights)
        dp = [[0 for _ in range(capacity + 1)] for _ in range(n + 1)]
        
        for i in range(1, n + 1):
            for w in range(1, capacity + 1):
                if weights[i-1] <= w:
                    dp[i][w] = max(values[i-1] + dp[i-1][w-weights[i-1]], dp[i-1][w])
                else:
                    dp[i][w] = dp[i-1][w]
        
        return dp[n][capacity]

# ✅ 字符串算法
class StringAlgorithms:
    """字符串算法"""
    
    @staticmethod
    def kmp_pattern_search(text: str, pattern: str) -> List[int]:
        """KMP模式匹配算法"""
        def compute_lps(pattern):
            lps = [0] * len(pattern)
            length = 0
            i = 1
            
            while i < len(pattern):
                if pattern[i] == pattern[length]:
                    length += 1
                    lps[i] = length
                    i += 1
                else:
                    if length != 0:
                        length = lps[length - 1]
                    else:
                        lps[i] = 0
                        i += 1
            return lps
        
        m = len(pattern)
        n = len(text)
        lps = compute_lps(pattern)
        
        i = j = 0  # text和pattern的索引
        result = []
        
        while i < n:
            if pattern[j] == text[i]:
                i += 1
                j += 1
            
            if j == m:
                result.append(i - j)
                j = lps[j - 1]
            elif i < n and pattern[j] != text[i]:
                if j != 0:
                    j = lps[j - 1]
                else:
                    i += 1
        
        return result
    
    @staticmethod
    def rabin_karp_pattern_search(text: str, pattern: str) -> List[int]:
        """Rabin-Karp算法"""
        d = 256  # 字符集大小
        q = 101   # 质数
        
        M = len(pattern)
        N = len(text)
        result = []
        
        if M > N:
            return result
        
        # 计算hash值
        p = t = 0  # pattern和text的hash值
        h = 1
        
        for i in range(M-1):
            h = (h * d) % q
        
        for i in range(M):
            p = (d * p + ord(pattern[i])) % q
            t = (d * t + ord(text[i])) % q
        
        for i in range(N - M + 1):
            if p == t:
                # hash值匹配，检查实际字符
                if text[i:i+M] == pattern:
                    result.append(i)
            
            if i < N - M:
                t = (d * (t - ord(text[i]) * h) + ord(text[i + M])) % q
                
                if t < 0:
                    t = t + q
        
        return result

class AlgorithmBenchmark:

    def benchmark_sorting_algorithms():

            'Bubble Sort': SortingAlgorithms.bubble_sort,
            'Insertion Sort': SortingAlgorithms.insertion_sort,

                data_copy = test_data.copy()
                
                start_time = time.perf_counter()

                end_time = time.perf_counter()
                
                execution_time = end_time - start_time
                is_sorted = all(sorted_data[i] <= sorted_data[i+1] for i in range(len(sorted_data)-1))

                found = result != -1

## 💡 费曼检验

**向技术面试官解释：快速排序和归并排序的区别是什么？在什么情况下选择哪种排序算法？**

核心要点：
1. **时间复杂度**：都平均O(n log n)，但稳定性不同
2. **空间复杂度**：快排O(log n)，归并O(n)
3. **稳定性**：快排不稳定，归并稳定
4. **选择原则**：根据内存限制、稳定性要求、数据特性选择
