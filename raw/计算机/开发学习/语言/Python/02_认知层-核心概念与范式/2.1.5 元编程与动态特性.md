# 2.1.5 元编程与动态特性

## 🎯 核心理念

元编程是Python的"魔法艺术"。它让我们能够编写编写程序的程序，体现了"代码即数据、数据即代码"的Lisp哲学在Python中的实现。

### 🧠 元类编程深度解析

```python
class MetaProgrammingFundamentals:
    """元编程基础"""
    
    @staticmethod
    def metaclass_deep_dive():
        """元类深度探索"""
        
        # ✅ 自动属性装饰器
        class AutoProperty(type):
            """自动创建属性的元类"""
            def __new__(cls, name, bases, attrs):
                # 处理私有属性的配置
                new_attrs = {}
                properties = {}
                
                for key, value in attrs.items():
                    if key.startswith('_property_'):
                        prop_name = key[10:]  # 移除 '_property_' 前缀
                        properties[prop_name] = value
                    else:
                        new_attrs[key] = value
                
                # 为每个property创建getter和setter
                for prop_name, default_value in properties.items():
                    # 创建私有存储名称
                    private_name = f"_property_{prop_name}"
                    
                    # 创建getter
                    def getter(self, name=prop_name, pname=private_name):
                        return getattr(self, pname, default_value)
                    
                    # 创建setter
                    def setter(self, value, name=prop_name, pname=private_name):
                        setattr(self, pname, value)
                    
                    # 创建property
                    new_attrs[prop_name] = property(getter, setter)
                
                return super().__new__(cls, name, bases, new_attrs)
        
        class AutoModel(metaclass=AutoProperty):
            """自动属性模型"""
            _property_name = "Unknown"
            _property_age = 0
            _property_active = True
        
        # 测试自动属性
        person = AutoModel()
        person.name = "Alice"
        person.age = 30
        person.active = False
        
        print(f"Name: {person.name}, Age: {person.age}, Active: {person.active}")
    
    @staticmethod
    def descriptor_protocol():
        """描述符协议深度应用"""
        
        class ValidatedProperty:
            """验证属性描述符"""
            def __init__(self, validator=None, default=None):
                self.validator = validator
                self.default = default
            
            def validation __get__(self, instance, owner):
                if instance is None:
                    return self
                return getattr(instance, self.private_name, self.default)
            
            def __set__(self, instance, value):
                if self.validator and not self.validator(value):
                    raise ValueError(f"Validation failed for {value}")
                setattr(instance, self.private_name, value)
            
            def __set_name__(self, owner, name):
                self.name = name
                self.private_name = f'_validated_{name}'
        
        class TypedProperty:
            """类型验证描述符"""
            def __init__(self, expected_type, default=None):
                self.expected_type = expected_type
                self.default = default
            
            def __get__(self, instance, owner):
                if instance is None:
                    return self
                return getattr(instance, self.private_name, self.default)
            
            def __set__(self, instance, value):
                if value is not None and not isinstance(value, self.expected_type):
                    raise TypeError(f"Expected {self.expected_type}, got {type(value)}")
                setattr(instance, self.private_name, value)
            
            def __set_name__(self, owner, name):
                self.name = name
                self.private_name = f'_typed_{name}'
        
        class UserAccount:
            """使用描述符的用户账户"""
            # 使用描述符
            username = TypedProperty(str)
            email = TypedProperty(str)
            age = ValidatedProperty(lambda x: isinstance(x, int) and x >= 0)
            score = TypedProperty((int, float), 0)
            
            def __init__(self, username, email, age=0):
                self.username = username
                self.email = email
                self.age = age
        
        # 测试描述符
        user = UserAccount("alice", "alice@example.com", 25)
        print(f"User: {user.username}, Age: {user.age}")
        
        # 测试类型验证
        try:
            user.username = 123  # 应该引发TypeError
        except TypeError as e:
            print(f"Type error: {e}")
        
        # 测试验证函数
        try:
            user.age = -5  # 应该引发ValueError
        except ValueError as e:
            print(f"Validation error: {e}")
    
    @staticmethod
    def dynamic_class_creation():
        """动态类创建"""
        
        def create_model_class(class_name, fields):
            """动态创建数据模型类"""
            # 准备类字典
            class_dict = {
                '__module__': __name__,
                '__doc__': f'动态创建的{class_name}模型'
            }
            
            def __init__(self, **kwargs):
                for field_name in fields:
                    setattr(self, field_name, kwargs.get(field_name, None))
            
            def __repr__(self):
                field_strs = []
                for field_name in fields:
                    value = getattr(self, field_name)
                    field_strs.append(f"{field_name}={value!r}")
                return f"{class_name}({', '.join(field_strs)})"
            
            def __eq__(self, other):
                if not isinstance(other, self.__class__):
                    return False
                return all(
                    getattr(self, field) == getattr(other, field)
                    for field in fields
                )
            
            class_dict['__init__'] = __init__
            class_dict['__repr__'] = __repr__
            class_dict['__eq__'] = __eq__
            
            # 创建类
            return type(class_name, (), class_dict)
        
        # 创建数据模型
        Person = create_model_class('Person', ['name', 'age', 'email'])
        
        # 测试动态创建的类
        alice = Person(name="Alice", age=30, email="alice@example.com")
        bob = Person(name="Bob", email="bob@example.com")  # age默认为None
        
        print(alice)  # Person(name='Alice', age=30, email='alice@example.com')
        print(bob)    # Person(name='Bob', age=None, email='bob@example.com')
        print(alice == Person(name="Alice", age=30, email="alice@example.com"))  # True
    
    @staticmethod
    def runtime_modification():
        """运行时修改"""
        
        class DynamicClass:
            """动态修改的类"""
            def __init__(self, name):
                self.name = name
        
        # 动态添加方法
        def say_hello(self):
            return f"Hello from {self.name}!"
        
        DynamicClass.say_hello = say_hello
        
        # 动态添加属性
        DynamicClass.version = "1.0"
        
        # 动态添加特殊方法
        def __str__(self):
            return f"DynamicClass({self.name})"
        
        DynamicClass.__str__ = __str__
        
        # 动态添加类方法
        @classmethod
        def get_class_info(cls):
            return f"Class: {cls.__name__}, Version: {getattr(cls, 'version', 'unknown')}"
        
        DynamicClass.get_class_info = get_class_info
        
        # 测试动态修改
        obj = DynamicClass("Test")
        print(obj.say_hello())
        print(str(obj))
        print(DynamicClass.get_class_info())
        print(f"Class version: {DynamicClass.version}")


# ✅ 高级元编程技巧
class AdvancedMetaProgramming:
    """高级元编程技巧"""
    
    @staticmethod
    def method_interception():
        """方法拦截"""
        
        class MethodInterceptor:
            """方法拦截器元类"""
            def __new__(cls, name, bases, attrs):
                new_attrs = {}
                
                for key, value in attrs.items():
                    if callable(value) and not key.startswith('_'):  # 拦截公共方法
                        # 创建包装方法
                        def create_wrapper(original_method):
                            def wrapper(self, *args, **kwargs):
                                print(f"CALLING: {original_method.__name__} with args={args}, kwargs={kwargs}")
                                
                                # 前置处理
                                print(f"PRE: Processing before {original_method.__name__}")
                                
                                try:
                                    result = original_method(self, *args, **kwargs)
                                    # 后置处理
                                    print(f"POST: Successfully completed {original_method.__name__}")
                                    return result
                                except Exception as e:
                                    # 异常处理
                                    print(f"ERROR: Exception in {original_method.__name__}: {e}")
                                    raise
                            
                            return wrapper
                        
                        new_attrs[key] = create_wrapper(value)
                    else:
                        new_attrs[key] = value
                
                return super().__new__(cls, name, bases, new_attrs)
        
        class ProcessedClass(metaclass=MethodInterceptor):
            def add(self, a, b):
                return a + b
            
            def multiply(self, a, b):
                return a * b
            
            def divide(self, a, b):
                return a / b
        
        # 测试方法拦截
        calc = ProcessedClass()
        print("Result:", calc.add(2, 3))
        print("Result:", calc.multiply(4, 5))
        
        try:
            calc.divide(10, 0)
        except ZeroDivisionError:
            print("Expected division by zero error")
    
    @staticmethod
    def dynamic_inheritance():
        """动态继承"""
        
        def create_mixin(name, methods_dict):
            """创建Mixin类"""
            class TempMixin:
                pass
            
            for method_name, method_func in methods_dict.items():
                setattr(TempMixin, method_name, method_func)
            
            TempMixin.__name__ = name
            return TempMixin
            
        # 定义一些Mixin方法
        LoggingMixin = create_mixin('LoggingMixin', {
            'log': lambda self, message: print(f"[LOG] {message}"),
            'debug': lambda self, message: print(f"[DEBUG] {message}")
        })
        
        CachingMixin = create_mixin('CachingMixin', {
            'cache': {},
            'get_cached': lambda self, key: self.cache.get(key),
            'set_cache': lambda self, key, value: setattr(self.cache, key, value)
        })
        
        ValidationMixin = create_mixin('ValidationMixin', {
            'validate': lambda self, data: isinstance(data, dict),
            'sanitize': lambda self, data: {k: str(v) for k, v in data.items()}
        })
        
        # 动态组合类
        class DynamicComposite:
            def __init__(self):
                # 动态添加Mixin功能
                for mixin_class in [LoggingMixin, CachingMixin, ValidationMixin]:
                    for attr_name in dir(mixin_class):
                        if not attr_name.startswith('_'):
                            attr_value = getattr(mixin_class, attr_name)
                            setattr(self, attr_name, attr_value)
                self.log("DynamicComposite initialized")
        
        # 测试动态组合
        composite = DynamicComposite()
        composite.log("This is a log message")
        composite.debug("This is a debug message")
        
        data = {"name": "Alice", "age": 30}
        if composite.validate(data):
            sanitized = composite.sanitize(data)
            composite.set_cache("user_data", sanitized)
            print("Cached:", composite.get_cached("user_data"))
    

# ✅ 元编程在框架中的应用
class FrameworkMetaProgramming:
    """框架中的元编程应用"""
    
    @staticmethod
    def orm_meta_pattern():
        """ORM元模式"""
        
        class Field:
            """字段描述符"""
            def __init__(self, field_type, required=False, default=None):
                self.field_type = field_type
                self.required = required
                self.default = default
                self.name = None
            
            def __get__(self, instance, owner):
                if instance is None:
                    return self
                return getattr(instance, f"_field_{self.name}", self.default)
            
            def __set__(self, instance, value):
                if self.required and value is None:
                    raise ValueError(f"Field {self.name} is required")
                if value is not None and not isinstance(value, self.field_type):
                    raise TypeError(f"Expected {self.field_type}, got {type(value)}")
                setattr(instance, f"_field_{self.name}", value)
            
            def __set_name__(self, owner, name):
                self.name = name
        
        class ModelMeta(type):
            """模型元类"""
            def __new__(cls, name, bases, attrs):
                fields = {}
                
                # 收集字段信息
                for key, value in attrs.items():
                    if isinstance(value, Field):
                        fields[key] = value
                
                # 添加表名
                table_name = attrs.get('table_name', name.lower())
                
                new_attrs = attrs.copy()
                new_attrs['_fields'] = fields
                new_attrs['table_name'] = table_name
                
                return super().__new__(cls, name, bases, new_attrs)
        
        class BaseModel(metaclass=ModelMeta):
            """基础模型"""
            def __init__(self, **kwargs):
                for field_name, field in self._fields.items():
                    setattr(self, f"_field_{field_name}", kwargs.get(field_name, None))
            
            def to_dict(self):
                result = {}
                for field_name in self._fields:
                    result[field_name] = getattr(self, field_name)
                return result
        
        # 使用ORM模式
        class User(BaseModel):
            table_name = "users"
            id = Field(int, required=True)
            username = Field(str, required=True)
            email = Field(str, required=True)
            age = Field(int, default=0)
            is_active = Field(bool, default=True)
        
        # 测试ORM模式
        user = User(id=1, username="alice", email="alice@example.com", age=30)
        print(f"User table: {user.table_name}")
        print(f"User data: {user.get_dict()}")
        print(f"User active: {user.is_active}")
        
        # 动态访问字段信息
        print("Fields:", [(name, field.field_type) for name, field in user._fields.items()])
