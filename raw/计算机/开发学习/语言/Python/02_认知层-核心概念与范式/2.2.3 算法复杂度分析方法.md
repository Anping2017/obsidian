# 2.2.3 算法复杂度分析方法

## 🎯 核心理念

算法复杂度分析是"性能预测的艺术"。通过数学分析方法，我们可以预测算法在不同规模数据下的表现，为选择最优解决方案提供科学依据。

### 🧠 复杂度分析基础

```python
import time
import random
import math
from itertools import combinations

class ComplexityAnalysis:
    """算法复杂度分析"""
    
    @staticmethod
    def big_o_analysis():
        """大O符号分析"""
        
        # ✅ O(1) 常数时间 - 数组随机访问
        def constant_time_operations():
            """常数时间操作"""
            data = list(range(1000000))
            
            # 随机访问 -> O(1)
            start = time.perf_counter()
            for _ in range(1000):
                random_index = random.randint(0, len(data) - 1)
                _ = data[random_index]
            end = time.perf_counter()
            print(f"O(1) 随机访问耗时: {end - start:.4f}s")
            
            # 字典查找 -> O(1)
            d = {i: i*2 for i in range(100000)}
            start = time.perf_counter()
            for _ in range(1000):
                random_key = random.randint(0, 99999)
                _ = d[random_key]
            end = time.perf_counter()
            print(f"O(1) 字典查找耗时: {end - start:.4f}s")
        
        # ✅ O(log n) 对数时间 - 二分查找
        def logarithmic_time_operations():
            """对数时间操作"""
            
            def binary_search(arr, target):
                left, right = 0, len(arr) - 1
                while left <= right:
                    mid = (left + right) // 2
                    if arr[mid] == target:
                        return mid
                    elif arr[mid] < target:
                        left = mid + 1
                    else:
                        right = mid - 1
                return -1
            
            # 测试不同规模的二分查找
            sizes = [1000, 10000, 100000]
            for size in sizes:
                sorted_data = list(range(size))
                target = size // 2
                
                start = time.perf_counter()
                for _ in range(1000):
                    _ = binary_search(sorted_data, target)
                end = time.perf_counter()
                
                print(f"O(log n) 二分查找 (规模{size}): {end - start:.4f}s")
        
        # ✅ O(n) 线性时间 - 线性搜索
        def linear_time_operations():
            """线性时间操作"""
            
            def linear_search(arr, target):
                for i, item in enumerate(arr):
                    if item == target:
                        return i
                return -1
            
            sizes = [1000, 10000, 100000]
            for size in sizes:
                data = list(range(size))
                target = size - 1  # 最坏情况
                
                start = time.perf_counter()
                for _ in range(100):  # 减少迭代次数
                    _ = linear_search(data, target)
                end = time.perf_counter()
                
                print(f"O(n) 线性搜索 (规模{size}): {end - start:.4f}s")
        
        # ✅ O(n²) 二次时间 - 选择排序
        def quadratic_time_operations():
            """二次时间操作"""
            
            def selection_sort(arr):
                n = len(arr)
                for i in range(n):
                    min_idx = i
                    for j in range(i+1, n):
                        if arr[j] < arr[min_idx]:
                            min_idx = j
                    arr[i], arr[min_idx] = arr[min_idx], arr[i]
                return arr
            
            sizes = [500, 1000, 2000]
            for size in sizes:
                data = list(range(size, 0, -1))  # 逆序最坏情况
                
                start = time.perf_counter()
                _ = selection_sort(data.copy())
                end = time.perf_counter()
                
                print(f"O(n²) 选择排序 (规模{size}): {end - start:.4f}s")
        
        constant_time_operations()
        logarithmic_time_operations()
        linear_time_operations()
        quadratic_time_operations()
    
    @staticmethod
    def space_complexity_analysis():
        """空间复杂度分析"""
        
        # ✅ O(1) 常数空间
        def constant_space_example():
            """常数空间示例"""
            def iterative_factorial(n):
                result = 1
                for i in range(1, n + 1):
                    result *= i
                return result
            
            print("常数空间递归 -> 迭代")
            print(f"10! = {iterative_factorial(10)}")
        
        # ✅ O(n) 线性空间
        def linear_space_example():
            """线性空间示例"""
            def recursive_factorial(n):
                if n <= 1:
                    return 1
                return n * recursive_factorial(n - 1)
            
            # 递归调用栈深度 = O(n)
            print("线性空间递归")
            print(f"5! = {recursive_factorial(5)}")
        
        # ✅ O(n²) 二次空间
        def quadratic_space_example():
            """二次空间示例"""
            def create_matrix(n):
                """创建n×n矩阵"""
                matrix = []
                for i in range(n):
                    row = []
                    for j in range(n):
                        row.append(i * j)
                    matrix.append(row)
                return matrix
            
            matrix = create_matrix(1000)  # 占用 O(n²) 空间
            print(f"二次空间矩阵创建: {len(matrix)}x{len(matrix[0])}")
        
        constant_space_example()
        linear_space_example()
        quadratic_space_example()
    
    @staticmethod
    def complexity_pattern_analysis():
        """复杂度模式分析"""
        
        # ✅ 循环嵌套分析
        def nested_loop_analysis():
            """循环嵌套复杂度分析"""
            
            def triple_nested_example(n):
                """三重嵌套 - O(n³)"""
                count = 0
                for i in range(n):
                    for j in range(n):
                        for k in range(n):
                            count += 1
                return count
            
            sizes = [50, 100, 200]
            for size in sizes:
                start = time.perf_counter()
                result = triple_nested_example(size)
                end = time.perf_counter()
                
                expected = size ** 3
                print(f"O(n³) 三重嵌套 (n={size}): {end - start:.4f}s")
                print(f"结果验证: {result} == {expected}")
        
        # ✅ 分治算法分析
        def divide_conquer_analysis():
            """分治算法复杂度"""
            
            def merge_sort_recursive(arr):
                """归并排序递归实现"""
                if len(arr) <= 1:
                    return arr
                
                mid = len(arr) // 2
                left = merge_sort_recursive(arr[:mid])
                right = merge_sort_recursive(arr[mid:])
                
                return merge(left, right)
            
            def merge(left, right):
                """合并两个有序数组"""
                result = []
                i = j = 0
                
                while i < len(left) and j < len(right):
                    if left[i] <= right[j]:
                        result.append(left[i])
                        i += 1
                    else:
                        result.append(right[j])
                        j += 1
                
                result.extend(left[i:])
                result.extend(right[j:])
                return result
            
            sizes = [1000, 5000, 10000]
            for size in sizes:
                data = list(range(size, 0, -1))  # 逆序
                
                start = time.perf_counter()
                sorted_data = merge_sort_recursive(data)
                end = time.perf_counter()
                
                print(f"O(n log n) 归并排序 (n={size}): {end - start:.4f}s")
        
        nested_loop_analysis()
        divide_conquer_analysis()

# ✅ 实际性能测量与分析
class PerformanceMeasurement:
    """性能测量与分析"""
    
    @staticmethod
    def comparative_performance_analysis():
        """比较性能分析"""
        
        algorithms = {
            'bubble_sort': lambda arr: bubble_sort(arr.copy()),
            'insertion_sort': lambda arr: insertion_sort(arr.copy()),
            'selection_sort': lambda arr: selection_sort(arr.copy()),
            'merge_sort': lambda arr: merge_sort(arr.copy()),
            'quicksort': lambda arr: quicksort(arr.copy())
        }
        
        def bubble_sort(arr):
            """冒泡排序 O(n²)"""
            n = len(arr)
            for i in range(n):
                for j in range(0, n-i-1):
                    if arr[j] > arr[j+1]:
                        arr[j], arr[j+1] = arr[j+1], arr[j]
            return arr
        
        def insertion_sort(arr):
            """插入排序 O(n²)"""
            for i in range(1, len(arr)):
                key = arr[i]
                j = i - 1
                while j >= 0 and arr[j] > key:
                    arr[j+1] = arr[j]
                    j -= 1
                arr[j+1] = key
            return arr
        
        def selection_sort(arr):
            """选择排序 O(n²)"""
            n = len(arr)
            for i in range(n):
                min_idx = i
                for j in range(i+1, n):
                    if arr[j] < arr[min_idx]:
                        min_idx = j
                arr[i], arr[min_idx] = arr[min_idx], arr[i]
            return arr
        
        def merge_sort(arr):
            """归并排序 O(n log n)"""
            if len(arr) <= 1:
                return arr
            
            mid = len(arr) // 2
            left = merge_sort(arr[:mid])
            right = merge_sort(arr[mid:])
            
            return merge(left, right)
        
        def merge(left, right):
            """合并函数"""
            result = []
            i = j = 0
            
            while i < len(left) and j < len(right):
                if left[i] <= right[j]:
                    result.append(left[i])
                    i += 1
                else:
                    result.append(right[j])
                    j += 1
            
            result.extend(left[i:])
            result.extend(right j:])
            return result
        
        def quicksort(arr):
            """快速排序 O(n log n) 平均情况"""
            if len(arr) <= 1:
                return arr
            
            pivot = arr[len(arr) // 2]
            left = [x for x in arr if x < pivot]
            middle = [x for x in arr if x == pivot]
            right = [x for x in arr if x > pivot]
            
            return quicksort(left) + middle + quicksort(right)
        
        # 测试不同规模的数据
        sizes = [1000, 5000]
        
        print("算法性能比较:")
        print("-" * 70)
        
        for size in sizes:
            data = [random.randint(1, 10000) for _ in range(size)]
            print(f"\n数据规模: {size}")
            print("-" * 50)
            
            for name, algorithm in algorithms.items():
                data_copy = data.copy()
                
                start = time.perf_counter()
                result = algorithm(data_copy)
                end = time.perf_counter()
                
                # 验证排序正确性
                is_sorted = all(result[i] <= result[i+1] for i in range(len(result)-1))
                
                print(f"{name:15}: {end-start:.4f}s {'✓' if is_sorted else '✗'}")

# ✅ 复杂度可视化
def visualize_complexity():
    """复杂度可视化"""
    
    sizes = list(range(1, 101))
    
    # 计算各复杂度函数值

    print("-" * 60)
    
    for i in range(0, len(sizes), 20):  # 每20个显示一次

**向技术面试官解释：如何快速判断一个算法的复杂度？大O符号在实际工程中的意义是什么？**

核心要点：
1. **识别模式**：嵌套循环通常意味着平方或更高次复杂度
2. **分治识别**：递归分治算法通常是O(n log n)
3. **实际意义**：复杂度帮助我们预测算法在大规模数据下的表现
4. **权衡考虑**：有时候更高的复杂度换来的是更简单的实现或更好的空间效率
