# 1.9 装饰器高级应用

## 🎯 核心理念

装饰器是Python中最优雅的设计模式之一，它不仅简化了代码，更是函数式和面向对象编程的完美结合。掌握高级装饰器技巧，能让代码更加简洁、可维护、可扩展。

    @staticmethod
    def decorator_patterns():

        # ✅ 无参数装饰器
        def basic_decorator(func: Callable) -> Callable:
            """基础装饰器"""
            @wraps(func)
            def wrapper(*args, **kwargs):
                print(f"执行前: {func.__name__}")
                result = func(*args, **kwargs)
                print(f"执行后: {func.__name__}")
                return result
            return wrapper
        
        # ✅ 带参数装饰器
        def parameterized_decorator(repeat_times: int = 2):
            """参数化装饰器"""
            def decorator(func: Callable) -> Callable:
                @wraps(func)
                def wrapper(*args, **kwargs):
                    print(f"重复执行 {repeat_times} 次: {func.__name__}")
                    
                    for i in range(repeat_times):
                        print(f"第 {i + 1} 次执行")
                        result = func(*args, **kwargs)
                    
                    return result
                return wrapper
            return decorator
        
        # ✅ 类装饰器
        class ClassDecorator:
            """类装饰器"""
            def __init__(self, func: Callable):
                self.func = func
                self.call_count = 0
            
            def __call__(self, *args, **kwargs):
                self.call_count += 1
                print(f"类装饰器调用次数: {self.call_count}")
                return self.func(*args, **kwargs)
        
        # ✅ 函数原型装饰器元类
        class MethodDecorator:
            """方法装饰器"""
            def __call__(self, func: Callable):
                @wraps(func)
                def wrapper(self_instance, *args, **kwargs):
                    print(f"方法装饰器: {func.__name__} 在 {self_instance.__class__.__name__}")
                    return func(self_instance, *args, **kwargs)
                return wrapper
        
        # 使用示例
        @basic_decorator
        def simple_function():
            print("简单函数执行")
            return "success"
        
        @parameterized_decorator(repeat_times=3)
        def repeated_function():
            print("重复函数执行")
            return "repeated"

        def counted_function():
            print("计数函数执行")
            return "counted"

        print()
        counted_function()

            cache = {}

            def wrapper(*args, **kwargs):
                # 创建缓存键
                cache_key = str(args) + str(sorted(kwargs.items()))

                    print(f"缓存命中: {func.__name__}({args})")
                    return cache[cache_key]
                
                print(f"计算执行: {func.__name__}({args})")
                result = func(*args, **kwargs)
                cache[cache_key] = result
                return result
            
            wrapper.cache_clear = lambda: cache.clear()
            wrapper.cache_info = lambda: len(cache)
            return wrapper
        
        # ✅ 重试装饰器
        def retry(max_attempts: int = 3, delay: float = 1.0, exceptions: tuple = (Exception,)):
            """重试装饰器"""
            def decorator(func: Callable) -> Callable:
                @wraps(func)
                def wrapper(*args, **kwargs):
                    last_exception = None
                    
                    for attempt in range(max_attempts):
                        try:
                            return func(*args, **kwargs)
                        except exceptions as e:
                            last_exception = e
                            if attempt < max_attempts - 1:
                                print(f"尝试 {attempt + 1}/{max_attempts} 失败: {e}")
                                time.sleep(delay)
                            else:
                                print(f"所有尝试失败，抛出异常: {e}")
                    
                    raise last_exception
                return wrapper
            return decorator

        def performance_monitor(func: Callable) -> Callable:

                    end_memory = memory_profiling_end()

                    memory_used = end_memory - start_memory
                    
                    print(f"性能监控 - {func.__name__}:")
                    print(f"  执行时间: {execution_time:.4f}s")
                    print(f"  内存使用: {memory_used:.2f}MB")
                
                return result
            return wrapper

                @wraps(func)

                    # 注入依赖

class PracticalDecorators:
    """实用装饰器集合"""
    
    @staticmethod
    def api_decorators():
        """API装饰器集合"""
        
        # ✅ API认证装饰器
        def api_authentication(api_key_required: bool = True):
            """API认证装饰器"""
            def decorator(func: Callable) -> Callable:
                @wraps(func)
                def wrapper(*args, **kwargs):
                    headers = kwargs.get('headers', {})
                    api_key = headers.get('X-API-Key')
                    
                    if api_key_required and not api_key:
                        return {"error": "Missing API Key", "status": 401}
                    
                    if api_key != 'valid_key':
                        return {"error": "Invalid API Key", "status": 401}
                    
                    print("API认证成功")
                    return func(*args, **kwargs)
                return wrapper
            return decorator
        
        # ✅ API限流装饰器
        def rate_limiter(max_requests: int = 100, time_window: int = 3600):
            """API限流装饰器"""
            def decorator(func: Callable) -> Callable:
                request_times = []
                
                @wraps(func)
                def wrapper(*args, **kwargs):
                    current_time = time.time()
                    
                    # 清理过期请求
                    request_times[:] = [t for t in request_times if current_time - t < time_window]
                    
                    if len(request_times) >= max_requests:
                        return {"error": "Rate limit exceeded", "status": 429}
                    
                    request_times.append(current_time)
                    return func(*args, **kwargs)
                return wrapper
            return decorator
        
        # ✅ API缓存装饰器
        def api_cache(cache_seconds: int = 300):
            """API缓存装饰器"""
            def decorator(func: Callable) -> Callable:
                cache = {}
                
                @wraps(func)
                def wrapper(*args, **kwargs):
                    # 创建缓存键（排除可变参数）
                    cache_key = hash(str(sorted(kwargs.items())))
                    
                    if cache_key in cache:
                        cached_result, timestamp = cache[cache_key]
                        if time.time() - timestamp < cache_seconds:
                            print("API缓存命中")
                            return cached_result
                    
                    result = func(*args, **kwargs)
                    cache[cache_key] = (result, time.time())
                    print("API结果已缓存")
                    
                    return result
                return wrapper
            return decorator
        
        @api_authentication()
        @rate_limiter(max_requests=10, time_window=60)
        @api_cache(cache_seconds=120)
        def protected_api_endpoint(data: str):
            """受保护的API端点"""
            print(f"处理API请求: {data}")
            return {"status": "success", "data": data, "timestamp": time.time()}
        
        print("\n=== API装饰器测试 ===")
        
        # 测试没有API Key的情况

        # 测试有效API Key

        print(f"有效API Key结果: {result2}")

            'authentication': api_authentication,
            'rate_limiter': rate_limiter,
            'cache': api_cache
        }
    
    @staticmethod
    def validation_decorators():

        # ✅ 类型验证装饰器
        def type_validator(**type_hints):
            """類型驗證裝飾器"""
            def decorator(func: Callable) -> Callable:
                @wraps(func)
                def wrapper(*args, **kwargs):
                    import inspect
                    sig = inspect.signature(func)
                    bound_args = sig.bind(*args, **kwargs)
                    
                    # 验证参数类型
                    for param_name, expected_type in type_hints.items():
                        if param_name in bound_args.arguments:
                            value = bound_args.arguments[param_name]
                            if not isinstance(value, expected_type):
                                raise TypeError(f"参数 {param_name} 期望类型 {expected_type.__name__}, 实际类型 {type(value).__name__}")
                    
                    return func(*args, **kwargs)
                return wrapper
            return decorator
        
        # ✅ 范围验证装饰器
        def range_validator(min_value: float = None, max_value: float = None):
            """范围验证装饰器"""
            def decorator(func: Callable) -> Callable:
                @wraps(func)
                def wrapper(*args, **kwargs):
                    for arg in args:
                        if isinstance(arg, (int, float)):
                            if min_value is not None and arg < min_value:
                                raise ValueError(f"值 {arg} 小于最小值 {min_value}")
                            if max_value is not None and arg > max_value:
                                raise ValueError(f"值 {arg} 大于最大值 {max_value}")
                    
                    return func(*args, **kwargs)
                return wrapper
            return decorator
        
        @type_validator(value=str, multiplier=float)
        @range_validator(min_value=0.0, max_value=100.0)
        def validated_function(value: str, multiplier: float):
            """验证示例函数"""
            return f"处理结果: {value} * {multiplier}"
        
        print("\n=== 验证装饰器测试 ===")
        
        try:
            result = validated_function("hello", 5.5)
            print(f"验证成功: {result}")
        except Exception as e:
            print(f"验证失败: {e}")
        
        return type_validator, range_validator

```python
class DecoratorFactory:
    """装饰器工厂类"""
    
    @staticmethod
    def create_logging_decorator(log_level: str = 'INFO'):
        """创建日志装饰器"""
        def logging_decorator(func):
            logger = logging.getLogger(func.__module__)
            
            @wraps(func)
            def wrapper(*args, **kwargs):
                logger.info(f"调用函数: {func.__name__}")
                start_time = time.time()
                
                try:
                    result = func(*args, **kwargs)
                    execution_time = time.time() - start_time
                    logger.info(f"函数 {func.__name__} 执行完成，耗时: {execution_time:.4f}s")
                    return result
                except Exception as e:
                    logger.error(f"函数 {func.__name__} 执行失败: {e}")
                    raise
            
            return wrapper
        return logging_decorator
    
    @staticmethod
    def create_validation_decorator(schema: dict):
        """创建验证装饰器"""
        def validation_decorator(func):
            @wraps(func)
            def wrapper(*args, **kwargs):
                # 验证输入参数
                for param_name, validation_rule in schema.items():
                    if param_name in kwargs:
                        value = kwargs[param_name]
                        
                        # 类型检查
                        if 'type' in validation_rule:
                            if not isinstance(value, validation_rule['type']):
                                raise ValueError(f"参数 {param_name} 类型错误")
                        
                        # 最小值检查
                        if 'min' in validation_rule:
                            if value < validation_rule['min']:
                                raise ValueError(f"参数 {param_name} 小于最小值")
                
                return func(*args, **kwargs)
            return wrapper
        return validation_decorator
```

```python
# ✅ 装饰器管理器
class DecoratorManager:
    """装饰器管理器"""
    
    def __init__(self):
        self.registered_decorators = {}
    
    def register(self, name: str, decorator: Callable):
        """注册装饰器"""
        self.registered_decorators[name] = decorator
    
    def apply(self, *decorator_names):
        """应用多个装饰器"""
        def apply_decorators(func):
            result = func
            for name in reversed(decorator_names):
                if name in self.registered_decorators:
                    result = self.registered_decorators[name](result)
                else:
                    raise NameError(f"未找到装饰器: {name}")
            return result
        return apply_decorators
    
    def list_available(self):
        """列出可用装饰器"""
        return list(self.registered_decorators.keys())
```
