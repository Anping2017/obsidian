# 实战案例分析

## 案例1：模块化开发中的闭包应用

### 问题描述
在大型前端应用中，需要创建多个独立的模块，每个模块都有自己的私有状态和公共接口，同时要避免全局变量污染。

### 解决方案
```javascript
// 1. 基础模块模式
var UserModule = (function() {
    // 私有变量
    var users = [];
    var currentUser = null;
    var isAuthenticated = false;
    
    // 私有函数
    function validateUser(user) {
        return user && user.id && user.name;
    }
    
    function hashPassword(password) {
        // 简单的密码哈希（实际应用中应使用更安全的方法）
        return btoa(password);
    }
    
    // 公共接口
    return {
        // 用户注册
        register: function(userData) {
            if (!validateUser(userData)) {
                throw new Error('Invalid user data');
            }
            
            var user = {
                id: Date.now(),
                name: userData.name,
                email: userData.email,
                password: hashPassword(userData.password),
                createdAt: new Date()
            };
            
            users.push(user);
            return user;
        },
        
        // 用户登录
        login: function(email, password) {
            var user = users.find(u => u.email === email);
            if (user && user.password === hashPassword(password)) {
                currentUser = user;
                isAuthenticated = true;
                return user;
            }
            throw new Error('Invalid credentials');
        },
        
        // 用户登出
        logout: function() {
            currentUser = null;
            isAuthenticated = false;
        },
        
        // 获取当前用户
        getCurrentUser: function() {
            return currentUser ? { ...currentUser } : null;
        },
        
        // 检查认证状态
        isAuthenticated: function() {
            return isAuthenticated;
        },
        
        // 获取用户数量
        getUserCount: function() {
            return users.length;
        }
    };
})();

// 使用示例
try {
    var user = UserModule.register({
        name: 'John Doe',
        email: 'john@example.com',
        password: 'password123'
    });
    console.log('User registered:', user);
    
    var loggedInUser = UserModule.login('john@example.com', 'password123');
    console.log('User logged in:', loggedInUser);
    
    console.log('Is authenticated:', UserModule.isAuthenticated());
    console.log('User count:', UserModule.getUserCount());
} catch (error) {
    console.error('Error:', error.message);
}
```

### 案例分析
- **闭包优势**: 私有变量和函数无法从外部直接访问
- **状态管理**: 模块内部状态被安全地封装
- **接口清晰**: 只暴露必要的公共方法
- **内存安全**: 避免了全局变量污染

## 案例2：事件处理中的闭包应用

### 问题描述
在动态生成的列表中，需要为每个项目添加点击事件，同时要传递项目的数据和索引。

### 解决方案
```javascript
// 1. 基础事件处理
function createDynamicList(items) {
    var container = document.getElementById('list-container');
    var selectedItems = [];
    
    // 清理现有内容
    container.innerHTML = '';
    
    items.forEach(function(item, index) {
        var listItem = document.createElement('div');
        listItem.className = 'list-item';
        listItem.textContent = item.name;
        
        // 使用闭包保存item和index
        listItem.addEventListener('click', function() {
            handleItemClick(item, index);
        });
        
        container.appendChild(listItem);
    });
    
    function handleItemClick(item, index) {
        console.log('Clicked item:', item);
        console.log('Item index:', index);
        
        // 更新选中状态
        var isSelected = selectedItems.includes(index);
        if (isSelected) {
            selectedItems = selectedItems.filter(i => i !== index);
        } else {
            selectedItems.push(index);
        }
        
        updateSelection();
    }
    
    function updateSelection() {
        var items = container.querySelectorAll('.list-item');
        items.forEach(function(item, index) {
            if (selectedItems.includes(index)) {
                item.classList.add('selected');
            } else {
                item.classList.remove('selected');
            }
        });
    }
    
    return {
        getSelectedItems: function() {
            return selectedItems.map(index => items[index]);
        },
        clearSelection: function() {
            selectedItems = [];
            updateSelection();
        }
    };
}

// 2. 高级事件处理 - 使用事件委托
function createAdvancedList(items) {
    var container = document.getElementById('list-container');
    var selectedItems = [];
    var itemData = new Map();
    
    // 清理现有内容
    container.innerHTML = '';
    
    // 创建项目元素
    items.forEach(function(item, index) {
        var listItem = document.createElement('div');
        listItem.className = 'list-item';
        listItem.dataset.index = index;
        listItem.textContent = item.name;
        
        // 存储项目数据
        itemData.set(index, item);
        
        container.appendChild(listItem);
    });
    
    // 使用事件委托
    container.addEventListener('click', function(event) {
        var target = event.target;
        if (target.classList.contains('list-item')) {
            var index = parseInt(target.dataset.index);
            var item = itemData.get(index);
            handleItemClick(item, index, target);
        }
    });
    
    function handleItemClick(item, index, element) {
        console.log('Clicked item:', item);
        console.log('Item index:', index);
        
        // 更新选中状态
        var isSelected = selectedItems.includes(index);
        if (isSelected) {
            selectedItems = selectedItems.filter(i => i !== index);
            element.classList.remove('selected');
        } else {
            selectedItems.push(index);
            element.classList.add('selected');
        }
    }
    
    return {
        getSelectedItems: function() {
            return selectedItems.map(index => itemData.get(index));
        },
        clearSelection: function() {
            selectedItems = [];
            var items = container.querySelectorAll('.list-item');
            items.forEach(function(item) {
                item.classList.remove('selected');
            });
        },
        cleanup: function() {
            container.removeEventListener('click', arguments.callee);
            itemData.clear();
        }
    };
}

// 使用示例
var items = [
    { name: 'Item 1', value: 100 },
    { name: 'Item 2', value: 200 },
    { name: 'Item 3', value: 300 }
];

var listManager = createAdvancedList(items);
```

### 案例分析
- **闭包优势**: 每个事件处理器都能访问对应的item和index
- **内存效率**: 事件委托减少了事件监听器的数量
- **状态管理**: 选中状态被封装在闭包中
- **清理机制**: 提供了清理方法避免内存泄漏

## 案例3：缓存和记忆化应用

### 问题描述
在计算密集型应用中，需要缓存计算结果以提高性能，同时要管理缓存大小和生命周期。

### 解决方案
```javascript
// 1. 基础缓存实现
function createCache(fn, maxSize = 100) {
    var cache = new Map();
    var accessOrder = [];
    
    function get(key) {
        if (cache.has(key)) {
            // 更新访问顺序
            var index = accessOrder.indexOf(key);
            if (index > -1) {
                accessOrder.splice(index, 1);
            }
            accessOrder.push(key);
            return cache.get(key);
        }
        return undefined;
    }
    
    function set(key, value) {
        if (cache.size >= maxSize) {
            // 移除最久未使用的项
            var oldestKey = accessOrder.shift();
            cache.delete(oldestKey);
        }
        
        cache.set(key, value);
        accessOrder.push(key);
    }
    
    return function(...args) {
        var key = JSON.stringify(args);
        var cached = get(key);
        
        if (cached !== undefined) {
            console.log('Cache hit for:', key);
            return cached;
        }
        
        console.log('Cache miss for:', key);
        var result = fn.apply(this, args);
        set(key, result);
        return result;
    };
}

// 2. 高级缓存实现
class AdvancedCache {
    constructor(options = {}) {
        this.maxSize = options.maxSize || 100;
        this.ttl = options.ttl || 0; // 生存时间（毫秒）
        this.cache = new Map();
        this.accessOrder = [];
        this.timestamps = new Map();
    }
    
    get(key) {
        if (this.cache.has(key)) {
            // 检查TTL
            if (this.ttl > 0) {
                var timestamp = this.timestamps.get(key);
                if (Date.now() - timestamp > this.ttl) {
                    this.delete(key);
                    return undefined;
                }
            }
            
            // 更新访问顺序
            this.updateAccessOrder(key);
            return this.cache.get(key);
        }
        return undefined;
    }
    
    set(key, value) {
        // 检查大小限制
        if (this.cache.size >= this.maxSize) {
            this.evictLRU();
        }
        
        this.cache.set(key, value);
        this.timestamps.set(key, Date.now());
        this.updateAccessOrder(key);
    }
    
    delete(key) {
        this.cache.delete(key);
        this.timestamps.delete(key);
        var index = this.accessOrder.indexOf(key);
        if (index > -1) {
            this.accessOrder.splice(index, 1);
        }
    }
    
    updateAccessOrder(key) {
        var index = this.accessOrder.indexOf(key);
        if (index > -1) {
            this.accessOrder.splice(index, 1);
        }
        this.accessOrder.push(key);
    }
    
    evictLRU() {
        var oldestKey = this.accessOrder.shift();
        if (oldestKey) {
            this.delete(oldestKey);
        }
    }
    
    clear() {
        this.cache.clear();
        this.timestamps.clear();
        this.accessOrder = [];
    }
    
    size() {
        return this.cache.size;
    }
    
    stats() {
        return {
            size: this.cache.size,
            maxSize: this.maxSize,
            ttl: this.ttl
        };
    }
}

// 3. 使用缓存的实际应用
function createFibonacciCache() {
    var cache = new AdvancedCache({ maxSize: 1000, ttl: 300000 }); // 5分钟TTL
    
    function fibonacci(n) {
        if (n < 0) throw new Error('Negative numbers not allowed');
        if (n <= 1) return n;
        
        // 检查缓存
        var cached = cache.get(n);
        if (cached !== undefined) {
            return cached;
        }
        
        // 计算并缓存
        var result = fibonacci(n - 1) + fibonacci(n - 2);
        cache.set(n, result);
        return result;
    }
    
    return {
        fibonacci: fibonacci,
        getCacheStats: () => cache.stats(),
        clearCache: () => cache.clear()
    };
}

// 使用示例
var fibCache = createFibonacciCache();

console.time('Fibonacci 40');
console.log(fibCache.fibonacci(40));
console.timeEnd('Fibonacci 40');

console.time('Fibonacci 40 (cached)');
console.log(fibCache.fibonacci(40));
console.timeEnd('Fibonacci 40 (cached)');

console.log('Cache stats:', fibCache.getCacheStats());
```

### 案例分析
- **性能提升**: 缓存避免了重复计算
- **内存管理**: LRU策略控制缓存大小
- **生命周期**: TTL机制管理缓存过期
- **统计信息**: 提供缓存使用统计

## 案例4：状态管理应用

### 问题描述
在复杂的单页应用中，需要管理多个组件的状态，同时要支持状态订阅和更新通知。

### 解决方案
```javascript
// 1. 基础状态管理器
function createStateManager(initialState = {}) {
    var state = { ...initialState };
    var listeners = [];
    var isUpdating = false;
    
    function getState() {
        return { ...state };
    }
    
    function setState(newState) {
        if (isUpdating) {
            console.warn('State update already in progress');
            return;
        }
        
        isUpdating = true;
        
        try {
            var prevState = { ...state };
            state = { ...state, ...newState };
            
            // 通知所有监听器
            listeners.forEach(function(listener) {
                try {
                    listener(state, prevState);
                } catch (error) {
                    console.error('State listener error:', error);
                }
            });
        } finally {
            isUpdating = false;
        }
    }
    
    function subscribe(listener) {
        if (typeof listener !== 'function') {
            throw new Error('Listener must be a function');
        }
        
        listeners.push(listener);
        
        // 返回取消订阅函数
        return function unsubscribe() {
            var index = listeners.indexOf(listener);
            if (index > -1) {
                listeners.splice(index, 1);
            }
        };
    }
    
    function select(selector) {
        if (typeof selector !== 'function') {
            throw new Error('Selector must be a function');
        }
        
        var selectedValue = selector(state);
        var hasChanged = false;
        
        return function(prevSelectedValue) {
            if (prevSelectedValue === undefined) {
                return selectedValue;
            }
            
            var newSelectedValue = selector(state);
            hasChanged = newSelectedValue !== prevSelectedValue;
            return hasChanged ? newSelectedValue : prevSelectedValue;
        };
    }
    
    return {
        getState: getState,
        setState: setState,
        subscribe: subscribe,
        select: select
    };
}

// 2. 高级状态管理器
class AdvancedStateManager {
    constructor(initialState = {}, options = {}) {
        this.state = { ...initialState };
        this.listeners = new Map();
        this.middlewares = [];
        this.isUpdating = false;
        this.updateQueue = [];
        this.batchUpdates = options.batchUpdates || false;
    }
    
    getState() {
        return { ...this.state };
    }
    
    setState(newState, callback) {
        if (this.isUpdating) {
            if (this.batchUpdates) {
                this.updateQueue.push({ newState, callback });
                return;
            } else {
                console.warn('State update already in progress');
                return;
            }
        }
        
        this.isUpdating = true;
        
        try {
            var prevState = { ...this.state };
            var nextState = { ...this.state, ...newState };
            
            // 应用中间件
            var finalState = this.applyMiddlewares(nextState, prevState);
            
            this.state = finalState;
            
            // 通知监听器
            this.notifyListeners(finalState, prevState);
            
            // 执行回调
            if (callback) {
                callback(finalState, prevState);
            }
        } finally {
            this.isUpdating = false;
            
            // 处理队列中的更新
            if (this.batchUpdates && this.updateQueue.length > 0) {
                var queuedUpdate = this.updateQueue.shift();
                this.setState(queuedUpdate.newState, queuedUpdate.callback);
            }
        }
    }
    
    applyMiddlewares(nextState, prevState) {
        return this.middlewares.reduce(function(state, middleware) {
            return middleware(state, prevState);
        }, nextState);
    }
    
    notifyListeners(nextState, prevState) {
        this.listeners.forEach(function(listener, key) {
            try {
                listener(nextState, prevState);
            } catch (error) {
                console.error(`State listener error for key ${key}:`, error);
            }
        });
    }
    
    subscribe(key, listener) {
        if (typeof listener !== 'function') {
            throw new Error('Listener must be a function');
        }
        
        this.listeners.set(key, listener);
        
        return function unsubscribe() {
            this.listeners.delete(key);
        }.bind(this);
    }
    
    addMiddleware(middleware) {
        if (typeof middleware !== 'function') {
            throw new Error('Middleware must be a function');
        }
        
        this.middlewares.push(middleware);
    }
    
    select(selector) {
        if (typeof selector !== 'function') {
            throw new Error('Selector must be a function');
        }
        
        var selectedValue = selector(this.state);
        
        return function(prevSelectedValue) {
            var newSelectedValue = selector(this.state);
            return newSelectedValue !== prevSelectedValue ? newSelectedValue : prevSelectedValue;
        }.bind(this);
    }
}

// 3. 使用状态管理器的实际应用
function createTodoApp() {
    var stateManager = new AdvancedStateManager({
        todos: [],
        filter: 'all',
        loading: false
    });
    
    // 添加中间件
    stateManager.addMiddleware(function(state, prevState) {
        // 日志中间件
        console.log('State changed:', {
            prev: prevState,
            next: state
        });
        return state;
    });
    
    stateManager.addMiddleware(function(state, prevState) {
        // 持久化中间件
        localStorage.setItem('todos', JSON.stringify(state.todos));
        return state;
    });
    
    // 动作创建器
    var actions = {
        addTodo: function(text) {
            var todo = {
                id: Date.now(),
                text: text,
                completed: false,
                createdAt: new Date()
            };
            
            stateManager.setState({
                todos: [...stateManager.getState().todos, todo]
            });
        },
        
        toggleTodo: function(id) {
            var todos = stateManager.getState().todos.map(function(todo) {
                return todo.id === id ? { ...todo, completed: !todo.completed } : todo;
            });
            
            stateManager.setState({ todos: todos });
        },
        
        deleteTodo: function(id) {
            var todos = stateManager.getState().todos.filter(function(todo) {
                return todo.id !== id;
            });
            
            stateManager.setState({ todos: todos });
        },
        
        setFilter: function(filter) {
            stateManager.setState({ filter: filter });
        },
        
        clearCompleted: function() {
            var todos = stateManager.getState().todos.filter(function(todo) {
                return !todo.completed;
            });
            
            stateManager.setState({ todos: todos });
        }
    };
    
    // 选择器
    var selectors = {
        getVisibleTodos: function(state) {
            var todos = state.todos;
            var filter = state.filter;
            
            switch (filter) {
                case 'active':
                    return todos.filter(todo => !todo.completed);
                case 'completed':
                    return todos.filter(todo => todo.completed);
                default:
                    return todos;
            }
        },
        
        getTodoCount: function(state) {
            return state.todos.length;
        },
        
        getCompletedCount: function(state) {
            return state.todos.filter(todo => todo.completed).length;
        }
    };
    
    return {
        stateManager: stateManager,
        actions: actions,
        selectors: selectors
    };
}

// 使用示例
var todoApp = createTodoApp();

// 订阅状态变化
var unsubscribe = todoApp.stateManager.subscribe('todos', function(state, prevState) {
    console.log('Todos updated:', state.todos);
});

// 执行动作
todoApp.actions.addTodo('Learn JavaScript');
todoApp.actions.addTodo('Build a todo app');
todoApp.actions.toggleTodo(1);

// 使用选择器
var visibleTodos = todoApp.selectors.getVisibleTodos(todoApp.stateManager.getState());
console.log('Visible todos:', visibleTodos);

// 清理
unsubscribe();
```

### 案例分析
- **状态封装**: 状态被安全地封装在闭包中
- **订阅机制**: 支持状态变化监听
- **中间件**: 支持状态处理中间件
- **选择器**: 提供高效的状态选择
