```javascript
// 1. 代码安全扫描器
class SecurityScanner {
  constructor() {
    this.rules = [
      {
        name: 'XSS检测',
        pattern: /innerHTML\s*=\s*[^;]+/g,
        severity: 'high',
        message: '使用innerHTML可能导致XSS攻击，建议使用textContent'
      },
      {
        name: 'SQL注入检测',
        pattern: /query\s*\(\s*['"`][^'"`]*\$\{/g,
        severity: 'high',
        message: '字符串拼接可能导致SQL注入，建议使用参数化查询'
      },
      {
        name: '硬编码密码',
        pattern: /password\s*[:=]\s*['"`][^'"`]{3,}['"`]/gi,
        severity: 'critical',
        message: '发现硬编码密码，存在严重安全风险'
      },
      {
        name: '不安全的随机数',
        pattern: /Math\.random\(\)/g,
        severity: 'medium',
        message: 'Math.random()不适用于安全场景，建议使用crypto.getRandomValues()'
      },
      {
        name: 'eval使用',
        pattern: /eval\s*\(/g,
        severity: 'high',
        message: 'eval()函数存在代码注入风险，应避免使用'
      }
    ];
  }
  
  // 扫描代码文件
  scanFile(filePath, content) {
    const issues = [];
    
    this.rules.forEach(rule => {
      const matches = content.match(rule.pattern);
      if (matches) {
        matches.forEach(match => {
          const lineNumber = this.getLineNumber(content, match);
          issues.push({
            rule: rule.name,
            severity: rule.severity,
            message: rule.message,
            line: lineNumber,
            code: match.trim(),
            file: filePath
          });
        });
      }
    });
    
    return issues;
  }
  
  // 获取行号
  getLineNumber(content, match) {
    const beforeMatch = content.substring(0, content.indexOf(match));
    return beforeMatch.split('\n').length;
  }
  
  // 生成安全报告
  generateReport(issues) {
    const report = {
      summary: {
        total: issues.length,
        critical: issues.filter(i => i.severity === 'critical').length,
        high: issues.filter(i => i.severity === 'high').length,
        medium: issues.filter(i => i.severity === 'medium').length,
        low: issues.filter(i => i.severity === 'low').length
      },
      issues: issues.sort((a, b) => {
        const severityOrder = { critical: 4, high: 3, medium: 2, low: 1 };
        return severityOrder[b.severity] - severityOrder[a.severity];
      })
    };
    
    return report;
  }
}

// 2. 使用示例
const scanner = new SecurityScanner();

// 扫描代码
const code = `
function renderUser(user) {
  document.getElementById('name').innerHTML = user.name;
  const query = \`SELECT * FROM users WHERE id = \${userId}\`;
  const password = 'admin123';
  const randomId = Math.random();
  eval('console.log("test")');
}`;

const issues = scanner.scanFile('example.js', code);
const report = scanner.generateReport(issues);

console.log('安全扫描报告:', report);
```

```javascript
          '| ping -c 1 127.0.0.1'

        expected: 'safe'

  // 测试输入验证
  async testInputValidation(validator, testCase) {
    const results = [];
    
    for (const input of testCase.inputs) {
      try {
        const result = await validator.validate(input);
        results.push({
          input,
          result: result.isValid ? 'safe' : 'blocked',
          message: result.message
        });
      } catch (error) {
        results.push({
          input,
          result: 'error',
          message: error.message
        });
      }
    }
    
    return {
      testCase: testCase.name,
      results,
      passed: results.every(r => r.result === testCase.expected)
    };
  }
  
  // 运行所有测试
  async runAllTests(validator) {
    const testResults = [];
    
    for (const testCase of this.testCases) {
      const result = await this.testInputValidation(validator, testCase);
      testResults.push(result);
    }
    
    return testResults;
  }

// 2. 输入验证器实现
class SecureInputValidator {
  constructor() {
    this.xssPatterns = [
      /<script[^>]*>.*?<\/script>/gi,
      /javascript:/gi,
      /on\w+\s*=/gi,
      /<iframe[^>]*>.*?<\/iframe>/gi,
      /<object[^>]*>.*?<\/object>/gi,
      /<embed[^>]*>.*?<\/embed>/gi
    ];
    
    this.sqlPatterns = [
      /('|(\\')|(;)|(\-\-)|(\/\*)|(\*\/)|(\|)|(\*)|(\%27)|(\%3B)|(\%2D)|(\%2D))/gi,
      /(union|select|insert|update|delete|drop|create|alter|exec|execute)/gi
    ];
    
    this.pathPatterns = [
      /\.\./g,
      /\/etc\/passwd/gi,
      /\/windows\/system32/gi,
      /C:\\windows/gi
    ];
    
    this.commandPatterns = [
      /[;&|`$]/g,
      /(rm|del|format|fdisk|mkfs)/gi,
      /(cat|type|more|less|head|tail)/gi,
      /(ping|traceroute|nslookup)/gi
    ];
  }
  
  // 验证输入
  async validate(input) {
    if (typeof input !== 'string') {
      return { isValid: false, message: '输入必须是字符串' };
    }
    
    // XSS检测
    if (this.detectXSS(input)) {
      return { isValid: false, message: '检测到XSS攻击尝试' };
    }
    
    // SQL注入检测
    if (this.detectSQLInjection(input)) {
      return { isValid: false, message: '检测到SQL注入攻击尝试' };
    }
    
    // 路径遍历检测
    if (this.detectPathTraversal(input)) {
      return { isValid: false, message: '检测到路径遍历攻击尝试' };
    }
    
    // 命令注入检测
    if (this.detectCommandInjection(input)) {
      return { isValid: false, message: '检测到命令注入攻击尝试' };
    }
    
    return { isValid: true, message: '输入安全' };
  }
  
  // XSS检测
  detectXSS(input) {
    return this.xssPatterns.some(pattern => pattern.test(input));
  }
  
  // SQL注入检测
  detectSQLInjection(input) {
    return this.sqlPatterns.some(pattern => pattern.test(input));
  }
  
  // 路径遍历检测
  detectPathTraversal(input) {
    return this.pathPatterns.some(pattern => pattern.test(input));
  }
  
  // 命令注入检测
  detectCommandInjection(input) {
    return this.commandPatterns.some(pattern => pattern.test(input));
  }
}
```

```javascript
class AuthenticationTester {
  constructor() {
    this.testUsers = [
      { username: 'admin', password: 'admin123', role: 'admin' },
      { username: 'user', password: 'user123', role: 'user' },
      { username: 'guest', password: 'guest123', role: 'guest' }
    ];
  }
  
  // 测试登录功能
  async testLogin(authService) {
    const results = [];
    
    // 测试有效登录
    for (const user of this.testUsers) {
      try {
        const result = await authService.login(user.username, user.password);
        results.push({
          test: `有效登录 - ${user.username}`,
          passed: result.success && result.user.role === user.role,
          message: result.success ? '登录成功' : result.message
        });
      } catch (error) {
        results.push({
          test: `有效登录 - ${user.username}`,
          passed: false,
          message: error.message
        });
      }
    }
    
    // 测试无效登录
    const invalidCredentials = [
      { username: 'admin', password: 'wrongpassword' },
      { username: 'nonexistent', password: 'password' },
      { username: '', password: 'password' },
      { username: 'admin', password: '' }
    ];
    
    for (const cred of invalidCredentials) {
      try {
        const result = await authService.login(cred.username, cred.password);
        results.push({
          test: `无效登录 - ${cred.username}`,
          passed: !result.success,
          message: result.success ? '应该失败但成功了' : '正确拒绝登录'
        });
      } catch (error) {
        results.push({
          test: `无效登录 - ${cred.username}`,
          passed: true,
          message: '正确拒绝登录'
        });
      }
    }
    
    return results;
  }
  
  // 测试会话管理
  async testSessionManagement(authService) {
    const results = [];
    
    // 登录获取会话
    const loginResult = await authService.login('admin', 'admin123');
    if (!loginResult.success) {
      return [{ test: '会话测试', passed: false, message: '无法登录' }];
    }
    
    const sessionId = loginResult.sessionId;
    
    // 测试会话验证
    try {
      const isValid = await authService.validateSession(sessionId);
      results.push({
        test: '会话验证',
        passed: isValid,
        message: isValid ? '会话有效' : '会话无效'
      });
    } catch (error) {
      results.push({
        test: '会话验证',
        passed: false,
        message: error.message
      });
    }
    
    // 测试会话过期
    try {
      // 模拟会话过期
      await authService.expireSession(sessionId);
      const isValid = await authService.validateSession(sessionId);
      results.push({
        test: '会话过期',
        passed: !isValid,
        message: isValid ? '会话应该过期但仍然有效' : '会话正确过期'
      });
    } catch (error) {
      results.push({
        test: '会话过期',
        passed: true,
        message: '会话正确过期'
      });
    }
    
    return results;
  }

    const roleTests = [

            const hasAccess = await authService.checkPermission(sessionId, resource);

              test: `${roleTest.role}访问${resource}`,
              passed: hasAccess,
              message: hasAccess ? '有权限访问' : '无权限访问'

          } catch (error) {
            results.push({
              test: `${roleTest.role}访问${resource}`,
              passed: false,
              message: error.message
            });
          }

        // 测试不允许的资源
        const forbiddenResources = ['/admin', '/users', '/settings'].filter(
          r => !roleTest.allowedResources.includes(r)
        );
        
        for (const resource of forbiddenResources) {

// 2. 认证服务实现
class SecureAuthService {
  constructor() {
    this.sessions = new Map();
    this.users = new Map([
      ['admin', { username: 'admin', password: 'admin123', role: 'admin' }],
      ['user', { username: 'user', password: 'user123', role: 'user' }],
      ['guest', { username: 'guest', password: 'guest123', role: 'guest' }]
    ]);
    this.sessionTimeout = 30 * 60 * 1000; // 30分钟
  }
  
  // 用户登录
  async login(username, password) {
    if (!username || !password) {
      return { success: false, message: '用户名和密码不能为空' };
    }
    
    const user = this.users.get(username);
    if (!user || user.password !== password) {
      return { success: false, message: '用户名或密码错误' };
    }
    
    const sessionId = this.generateSessionId();
    const session = {
      userId: username,
      role: user.role,
      createdAt: Date.now(),
      expiresAt: Date.now() + this.sessionTimeout
    };
    
    this.sessions.set(sessionId, session);
    
    return {
      success: true,
      sessionId,
      user: { username: user.username, role: user.role }
    };
  }
  
  // 验证会话
  async validateSession(sessionId) {
    const session = this.sessions.get(sessionId);
    
    if (!session) {
      return false;
    }
    
    if (Date.now() > session.expiresAt) {
      this.sessions.delete(sessionId);
      return false;
    }
    
    // 更新过期时间
    session.expiresAt = Date.now() + this.sessionTimeout;
    return true;
  }
  
  // 检查权限
  async checkPermission(sessionId, resource) {
    const isValid = await this.validateSession(sessionId);
    if (!isValid) {
      throw new Error('会话无效');
    }
    
    const session = this.sessions.get(sessionId);
    const role = session.role;
    
    // 权限规则
    const permissions = {
      admin: ['/admin', '/users', '/settings', '/profile', '/dashboard', '/public'],
      user: ['/profile', '/dashboard', '/public'],
      guest: ['/public']
    };
    
    return permissions[role]?.includes(resource) || false;
  }
  
  // 过期会话
  async expireSession(sessionId) {
    const session = this.sessions.get(sessionId);
    if (session) {
      session.expiresAt = Date.now() - 1;
    }
  }
  
  // 生成会话ID
  generateSessionId() {
    return Math.random().toString(36).substring(2) + Date.now().toString(36);
  }
}
```

```javascript
// 2. 加密服务实现
class SecureEncryptionService {
  constructor() {
    this.algorithm = 'AES-GCM';
    this.keyLength = 256;
  }
  
  // 生成密钥
  async generateKey() {
    return await crypto.subtle.generateKey(
      {
        name: this.algorithm,
        length: this.keyLength
      },
      true,
      ['encrypt', 'decrypt']
    );
  }
  
  // 加密数据
  async encrypt(data, key) {
    const encoder = new TextEncoder();
    const dataBuffer = encoder.encode(data);
    
    const iv = crypto.getRandomValues(new Uint8Array(12));
    
    const encryptedData = await crypto.subtle.encrypt(
      {
        name: this.algorithm,
        iv: iv
      },
      key,
      dataBuffer
    );
    
    return {
      data: Array.from(new Uint8Array(encryptedData)),
      iv: Array.from(iv)
    };
  }
  
  // 解密数据
  async decrypt(encryptedData, key) {
    const { data, iv } = encryptedData;
    
    const decryptedData = await crypto.subtle.decrypt(
      {
        name: this.algorithm,
        iv: new Uint8Array(iv)
      },
      key,
      new Uint8Array(data)
    );
    
    const decoder = new TextDecoder();
    return decoder.decode(decryptedData);
  }
  
  // 哈希数据
  async hash(data) {
    const encoder = new TextEncoder();
    const dataBuffer = encoder.encode(data);
    const hashBuffer = await crypto.subtle.digest('SHA-256', dataBuffer);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
  }
  
  // 密码哈希
  async hashPassword(password) {
    const encoder = new TextEncoder();
    const data = encoder.encode(password);
    const hashBuffer = await crypto.subtle.digest('SHA-256', data);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
  }
  
  // 验证密码
  async verifyPassword(password, hash) {
    const passwordHash = await this.hashPassword(password);
    return passwordHash === hash;
  }
}
```

## 安全配置测试

### 安全头测试
```javascript
// 1. 安全头测试套件
class SecurityHeadersTester {
  constructor() {
    this.requiredHeaders = [
      'X-Content-Type-Options',
      'X-Frame-Options',
      'X-XSS-Protection',
      'Strict-Transport-Security',
      'Content-Security-Policy',
      'Referrer-Policy'
    ];
    
    this.headerValues = {
      'X-Content-Type-Options': 'nosniff',
      'X-Frame-Options': ['DENY', 'SAMEORIGIN'],
      'X-XSS-Protection': '1; mode=block',
      'Strict-Transport-Security': /max-age=\d+/,
      'Content-Security-Policy': /default-src/,
      'Referrer-Policy': ['strict-origin-when-cross-origin', 'no-referrer']
    };
  }
  
  // 测试安全头
  async testSecurityHeaders(url) {
    const results = [];
    
    try {
      const response = await fetch(url, { method: 'HEAD' });
      const headers = response.headers;
      
      for (const headerName of this.requiredHeaders) {
        const headerValue = headers.get(headerName);
        
        if (!headerValue) {
          results.push({
            header: headerName,
            passed: false,
            message: '安全头缺失'
          });
        } else {
          const expectedValues = this.headerValues[headerName];
          let isValid = false;
          
          if (Array.isArray(expectedValues)) {
            isValid = expectedValues.includes(headerValue);
          } else if (expectedValues instanceof RegExp) {
            isValid = expectedValues.test(headerValue);
          } else {
            isValid = headerValue === expectedValues;
          }
          
          results.push({
            header: headerName,
            passed: isValid,
            message: isValid ? '安全头配置正确' : `安全头配置错误: ${headerValue}`
          });
        }
      }
    } catch (error) {
      results.push({
        header: 'ALL',
        passed: false,
        message: `无法访问URL: ${error.message}`
      });
    }
    
    return results;
  }
  
  // 测试CSP策略
  async testCSPPolicy(url) {
    const results = [];
    
    try {
      const response = await fetch(url, { method: 'HEAD' });
      const cspHeader = response.headers.get('Content-Security-Policy');
      
      if (!cspHeader) {
        results.push({
          test: 'CSP头存在',
          passed: false,
          message: 'Content-Security-Policy头缺失'
        });
        return results;
      }
      
      // 检查CSP指令
      const cspDirectives = cspHeader.split(';').map(d => d.trim());
      const requiredDirectives = ['default-src', 'script-src', 'style-src'];
      
      for (const directive of requiredDirectives) {
        const hasDirective = cspDirectives.some(d => d.startsWith(directive));
        results.push({
          test: `CSP指令 - ${directive}`,
          passed: hasDirective,
          message: hasDirective ? '指令存在' : '指令缺失'
        });
      }
      
      // 检查危险配置
      const dangerousConfigs = [
        "script-src 'unsafe-inline'",
        "script-src 'unsafe-eval'",
        "default-src *"
      ];
      
      for (const config of dangerousConfigs) {
        const hasDangerous = cspDirectives.some(d => d.includes(config));
        results.push({
          test: `CSP安全 - ${config}`,
          passed: !hasDangerous,
          message: hasDangerous ? '存在危险配置' : '配置安全'
        });
      }
      
    } catch (error) {
      results.push({
        test: 'CSP测试',
        passed: false,
        message: `测试失败: ${error.message}`
      });
    }
    
    return results;
  }
}

// 2. 使用示例
const securityTester = new SecurityHeadersTester();

// 测试安全头
async function testSecurity() {
  const url = 'https://example.com';
  
  console.log('=== 安全头测试 ===');
  const headerResults = await securityTester.testSecurityHeaders(url);
  headerResults.forEach(result => {
    console.log(`${result.header}: ${result.passed ? '通过' : '失败'} - ${result.message}`);
  });
  
  console.log('\n=== CSP策略测试 ===');
  const cspResults = await securityTester.testCSPPolicy(url);
  cspResults.forEach(result => {
    console.log(`${result.test}: ${result.passed ? '通过' : '失败'} - ${result.message}`);
  });
}

testSecurity();
```
