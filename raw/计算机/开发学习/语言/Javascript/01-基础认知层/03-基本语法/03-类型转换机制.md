# 类型转换机制

## 类型转换概述

### 转换类型分类
```mermaid
graph TD
    A[类型转换] --> B[隐式转换]
    A --> C[显式转换]
    
    B --> B1[自动类型转换]

    C --> C1[String()]
```

## 隐式类型转换

### 字符串转换
```javascript
// 字符串拼接触发转换
console.log('5' + 3); // "53" (数字转字符串)
console.log('5' + true); // "5true" (布尔值转字符串)
console.log('5' + null); // "5null" (null转字符串)
console.log('5' + undefined); // "5undefined" (undefined转字符串)

// 对象转字符串
const obj = { name: 'John' };
console.log('User: ' + obj); // "User: [object Object]"
console.log('User: ' + obj.toString()); // "User: [object Object]"

// 数组转字符串
const arr = [1, 2, 3];
console.log('Array: ' + arr); // "Array: 1,2,3"
```

### 数值转换
```javascript
// 算术运算符触发数值转换
console.log('5' - 3); // 2 (字符串转数字)
console.log('5' * 2); // 10 (字符串转数字)
console.log('5' / 2); // 2.5 (字符串转数字)
console.log('5' % 2); // 1 (字符串转数字)

// 一元运算符
console.log(+'123'); // 123 (字符串转数字)
console.log(-'123'); // -123 (字符串转数字)
console.log(+true); // 1 (布尔值转数字)
console.log(+false); // 0 (布尔值转数字)
console.log(+null); // 0 (null转数字)
console.log(+undefined); // NaN (undefined转数字)

// 比较运算符
console.log('5' > 3); // true (字符串转数字)
console.log('5' == 5); // true (类型转换后比较)
console.log('5' === 5); // false (严格比较，不转换)
```

### 布尔值转换
```javascript
// 条件判断中的转换
if ('hello') {
    console.log('truthy'); // 执行
}

if ('') {
    console.log('falsy'); // 不执行
}

// 逻辑运算符
console.log('hello' && 'world'); // "world" (返回最后一个truthy值)
console.log('' || 'default'); // "default" (返回第一个truthy值)
console.log(!'hello'); // false (字符串转布尔值后取反)
console.log(!!'hello'); // true (双重否定，显式转换)
```

## 显式类型转换

### 字符串转换
```javascript
// String() 函数
console.log(String(123)); // "123"
console.log(String(true)); // "true"
console.log(String(null)); // "null"
console.log(String(undefined)); // "undefined"

// toString() 方法
console.log((123).toString()); // "123"
console.log(true.toString()); // "true"
// console.log(null.toString()); // TypeError
// console.log(undefined.toString()); // TypeError

// 模板字符串
const num = 42;
console.log(`${num}`); // "42"
```

### 数值转换
```javascript
// Number() 函数
console.log(Number('123')); // 123
console.log(Number('123.45')); // 123.45
console.log(Number('')); // 0
console.log(Number(' ')); // 0
console.log(Number('abc')); // NaN
console.log(Number(true)); // 1
console.log(Number(false)); // 0
console.log(Number(null)); // 0
console.log(Number(undefined)); // NaN

// parseInt() 函数
console.log(parseInt('123')); // 123
console.log(parseInt('123.45')); // 123 (截取整数部分)
console.log(parseInt('123px')); // 123 (忽略非数字字符)
console.log(parseInt('abc123')); // NaN (开头不是数字)
console.log(parseInt('0x10')); // 16 (十六进制)
console.log(parseInt('10', 2)); // 2 (二进制)

// parseFloat() 函数
console.log(parseFloat('123.45')); // 123.45
console.log(parseFloat('123.45px')); // 123.45
console.log(parseFloat('abc123')); // NaN
```

### 布尔值转换
```javascript
// Boolean() 函数
console.log(Boolean(1)); // true
console.log(Boolean(0)); // false
console.log(Boolean('')); // false
console.log(Boolean('hello')); // true
console.log(Boolean(null)); // false
console.log(Boolean(undefined)); // false
console.log(Boolean([])); // true (空数组是truthy)
console.log(Boolean({})); // true (空对象是truthy)

// 双重否定
console.log(!!1); // true
console.log(!!0); // false
console.log(!!'hello'); // true
console.log(!!''); // false
```

## 对象类型转换

### 对象转字符串
```javascript
// 默认行为
const obj = { name: 'John' };
console.log(String(obj)); // "[object Object]"
console.log(obj.toString()); // "[object Object]"

// 自定义toString方法
const customObj = {
    name: 'John',
    toString() {
        return `User: ${this.name}`;
    }
};
console.log(String(customObj)); // "User: John"
```

### 对象转数值
```javascript
// 默认行为
const obj = { name: 'John' };
console.log(Number(obj)); // NaN
console.log(+obj); // NaN

// 自定义valueOf方法
const customObj = {
    value: 42,
    valueOf() {
        return this.value;
    }
};
console.log(Number(customObj)); // 42
console.log(+customObj); // 42
```

### 对象转布尔值
```javascript
// 所有对象都是truthy
console.log(Boolean({})); // true
console.log(Boolean([])); // true
console.log(Boolean(new Date())); // true
console.log(Boolean(/regex/)); // true
console.log(Boolean(function() {})); // true
```

## 转换优先级

### 转换顺序
```javascript
// 1. 对象转原始值
const obj = {
    valueOf() {
        console.log('valueOf called');
        return 42;
    },
    toString() {
        console.log('toString called');
        return '42';
    }
};

// 数值转换：优先调用valueOf
console.log(Number(obj)); // valueOf called, 42

// 字符串转换：优先调用toString
console.log(String(obj)); // toString called, "42"

// 2. 原始值转换
console.log(Number('123')); // 123
console.log(String(123)); // "123"
console.log(Boolean(1)); // true
```

## 最佳实践

### 避免隐式转换
```javascript
// 使用严格比较
console.log('5' === 5); // false (推荐)
console.log('5' == 5); // true (不推荐)

// 显式转换
const userInput = '123';
const number = Number(userInput);
if (!isNaN(number)) {
    console.log('Valid number:', number);
}
```

### 类型检查
```javascript
// 安全的类型检查
function isNumber(value) {
    return typeof value === 'number' && !isNaN(value);
}

function isString(value) {
    return typeof value === 'string';
}

function isArray(value) {
    return Array.isArray(value);
}

// 使用
if (isNumber(userInput)) {
    console.log('Processing number:', userInput);
}
```

### 转换工具函数
```javascript
// 安全的转换函数
function safeParseInt(value, defaultValue = 0) {
    const parsed = parseInt(value, 10);
    return isNaN(parsed) ? defaultValue : parsed;
}

function safeParseFloat(value, defaultValue = 0) {
    const parsed = parseFloat(value);
    return isNaN(parsed) ? defaultValue : parsed;
}

// 使用
console.log(safeParseInt('123')); // 123
console.log(safeParseInt('abc')); // 0
console.log(safeParseInt('abc', -1)); // -1
```
