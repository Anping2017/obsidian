# 代码示例库-基础语法

## 变量声明示例

### var声明示例
```javascript
// 1. 函数作用域演示
function varScopeDemo() {
    console.log('Before block:', x); // undefined (变量提升)
    
    if (true) {
        var x = 1;
        var y = 2;
    }
    
    console.log('After block:', x); // 1 (函数作用域)
    console.log('After block:', y); // 2 (函数作用域)
}

// 2. 变量提升演示
console.log('Before declaration:', hoistedVar); // undefined
var hoistedVar = 'I am hoisted';
console.log('After declaration:', hoistedVar); // 'I am hoisted'

// 3. 重复声明演示
var repeatedVar = 'first';
var repeatedVar = 'second';
console.log(repeatedVar); // 'second'
```

### let声明示例
```javascript
// 1. 块级作用域演示
function letScopeDemo() {
    if (true) {
        let blockVar = 'block scope';
        const blockConst = 'block constant';
    }
    
    // console.log(blockVar); // ReferenceError
    // console.log(blockConst); // ReferenceError
}

// 2. 暂时性死区演示
// console.log(temporalDeadZone); // ReferenceError
let temporalDeadZone = 'I am in TDZ';
console.log(temporalDeadZone); // 'I am in TDZ'

// 3. 循环中的块级作用域
for (let i = 0; i < 3; i++) {
    setTimeout(() => {
        console.log('let i:', i); // 0, 1, 2
    }, 100);
}

// 对比var的行为
for (var j = 0; j < 3; j++) {
    setTimeout(() => {
        console.log('var j:', j); // 3, 3, 3
    }, 100);
}
```

## 数据类型示例

### 基本数据类型示例
```javascript
// 1. number类型
const integer = 42;
const float = 3.14;
const scientific = 1e6; // 1000000
const binary = 0b1010; // 10
const octal = 0o755; // 493
const hex = 0xFF; // 255

// 特殊数值
const infinity = Infinity;
const negativeInfinity = -Infinity;
const notANumber = NaN;

// 数值方法
console.log(Number.isInteger(42)); // true
console.log(Number.isNaN(NaN)); // true
console.log(Number.parseFloat('3.14px')); // 3.14
console.log(Number.parseInt('42px')); // 42

// 2. string类型
const singleQuote = 'Hello';
const doubleQuote = "World";
const templateLiteral = `Hello ${singleQuote}`;
const multiline = `This is a
multiline string`;

// 字符串方法
const str = 'Hello World';
console.log(str.length); // 11
console.log(str.charAt(0)); // 'H'
console.log(str.indexOf('World')); // 6
console.log(str.slice(0, 5)); // 'Hello'
console.log(str.toUpperCase()); // 'HELLO WORLD'
console.log(str.split(' ')); // ['Hello', 'World']

// 3. boolean类型
const isTrue = true;
const isFalse = false;

// 布尔转换
console.log(Boolean(1)); // true
console.log(Boolean(0)); // false
console.log(Boolean('')); // false
console.log(Boolean('hello')); // true
console.log(Boolean(null)); // false
console.log(Boolean(undefined)); // false

// 4. undefined和null
let undefinedVar;
console.log(undefinedVar); // undefined
console.log(typeof undefinedVar); // "undefined"

const nullVar = null;
console.log(nullVar); // null
console.log(typeof nullVar); // "object" (历史遗留)

// 5. symbol类型
const sym1 = Symbol();
const sym2 = Symbol('description');
const sym3 = Symbol('description');

console.log(sym2 === sym3); // false (每个Symbol都是唯一的)

// 全局Symbol
const globalSym = Symbol.for('global');
const sameGlobalSym = Symbol.for('global');
console.log(globalSym === sameGlobalSym); // true

// Symbol作为对象属性
const obj = {};
const key = Symbol('key');
obj[key] = 'value';
console.log(obj[key]); // 'value'

// 6. bigint类型
const bigInt1 = 123n;
const bigInt2 = BigInt(123);
const bigInt3 = BigInt('123');

console.log(bigInt1 + bigInt2); // 246n
console.log(bigInt1 * 2n); // 246n
console.log(Number(bigInt1)); // 123
```

### 引用数据类型示例
```javascript
// 1. object类型
const obj1 = {}; // 字面量
const obj2 = new Object(); // 构造函数
const obj3 = Object.create(null); // 原型创建

// 对象属性
const person = {
    name: 'John',
    age: 30,
    'full name': 'John Doe', // 包含空格的属性名
    address: {
        city: 'New York',
        country: 'USA'
    },
    greet() {
        return `Hello, I'm ${this.name}`;
    }
};

// 属性访问
console.log(person.name); // 'John'
console.log(person['full name']); // 'John Doe'
console.log(person.address.city); // 'New York'
console.log(person.greet()); // 'Hello, I'm John'

// 属性操作
person.city = 'New York'; // 添加属性
delete person.age; // 删除属性
console.log('name' in person); // true
console.log(Object.keys(person)); // ['name', 'full name', 'address', 'greet', 'city']

// 2. array类型
const arr1 = []; // 空数组
const arr2 = [1, 2, 3]; // 字面量
const arr3 = new Array(3); // 构造函数
const arr4 = Array.from('hello'); // ['h', 'e', 'l', 'l', 'o']

// 数组方法
const numbers = [1, 2, 3, 4, 5];
console.log(numbers.length); // 5
console.log(numbers.push(6)); // 6 (返回新长度)
console.log(numbers.pop()); // 6 (返回删除的元素)
console.log(numbers.slice(1, 3)); // [2, 3]
console.log(numbers.map(x => x * 2)); // [2, 4, 6, 8, 10]
console.log(numbers.filter(x => x > 2)); // [3, 4, 5]
console.log(numbers.reduce((sum, x) => sum + x, 0)); // 15

// 3. function类型
function regularFunction() {
    return 'regular';
}

const arrowFunction = () => {
    return 'arrow';
};

const functionExpression = function() {
    return 'expression';
};

// 函数特性
console.log(typeof regularFunction); // "function"
console.log(regularFunction.name); // "regularFunction"
console.log(regularFunction.length); // 0 (参数个数)
```

## 类型转换示例

### 隐式转换示例
```javascript
// 1. 字符串转换
console.log('5' + 3); // "53" (数字转字符串)
console.log('5' + true); // "5true" (布尔值转字符串)
console.log('5' + null); // "5null" (null转字符串)
console.log('5' + undefined); // "5undefined" (undefined转字符串)

// 对象转字符串
const obj = { name: 'John' };
console.log('User: ' + obj); // "User: [object Object]"

// 数组转字符串
const arr = [1, 2, 3];
console.log('Array: ' + arr); // "Array: 1,2,3"

// 2. 数值转换
console.log('5' - 3); // 2 (字符串转数字)
console.log('5' * 2); // 10 (字符串转数字)
console.log('5' / 2); // 2.5 (字符串转数字)
console.log('5' % 2); // 1 (字符串转数字)

// 一元运算符
console.log(+'123'); // 123 (字符串转数字)
console.log(-'123'); // -123 (字符串转数字)
console.log(+true); // 1 (布尔值转数字)
console.log(+false); // 0 (布尔值转数字)
console.log(+null); // 0 (null转数字)
console.log(+undefined); // NaN (undefined转数字)

// 3. 布尔值转换
if ('hello') {
    console.log('truthy'); // 执行
}

if ('') {
    console.log('falsy'); // 不执行
}

// 逻辑运算符
console.log('hello' && 'world'); // "world" (返回最后一个truthy值)
console.log('' || 'default'); // "default" (返回第一个truthy值)
console.log(!'hello'); // false (字符串转布尔值后取反)
console.log(!!'hello'); // true (双重否定，显式转换)
```

## 运算符优先级示例

### 算术运算符优先级
```javascript
// 乘除取余优先级高于加减
console.log(2 + 3 * 4); // 14 (不是20)
console.log(2 + 3 * 4); // 等价于 2 + (3 * 4)

// 使用括号改变优先级
console.log((2 + 3) * 4); // 20

// 一元运算符优先级
console.log(-2 + 3); // 1 (等价于 (-2) + 3)
console.log(-(2 + 3)); // -5

// 递增递减运算符
let x = 5;
console.log(++x); // 6 (先递增，后使用)
console.log(x++); // 6 (先使用，后递增)
console.log(x); // 7

let y = 5;
console.log(y++ * 2); // 10 (等价于 (y++) * 2)
console.log(y); // 6

let z = 5;
console.log(++z * 2); // 12 (等价于 (++z) * 2)
console.log(z); // 6
```

### 逻辑运算符优先级
```javascript
// 逻辑与优先级高于逻辑或
console.log(true || false && false); // true (等价于 true || (false && false))
console.log(false && true || true); // true (等价于 (false && true) || true)

// 短路求值
let x = 0;
console.log(x && x.toString()); // 0 (不会调用toString)
console.log(x || 'default'); // 'default'

// 实际应用
function getUserName(user) {
    return user && user.name || 'Anonymous';
}

console.log(getUserName({ name: 'John' })); // 'John'
console.log(getUserName(null)); // 'Anonymous'
```

### 赋值运算符优先级
```javascript
// 赋值运算符优先级最低
let x = 2 + 3 * 4; // 14 (等价于 x = (2 + 3 * 4))
let y = (2 + 3) * 4; // 20

// 复合赋值运算符
let z = 10;
z += 2 * 3; // 16 (等价于 z = z + (2 * 3))
console.log(z); // 16

// 链式赋值
let a, b, c;
a = b = c = 5; // 等价于 a = (b = (c = 5))
console.log(a, b, c); // 5 5 5
```

## 实际应用示例

### 类型检查工具函数
```javascript
// 安全的类型检查
function isString(value) {
    return typeof value === 'string';
}

function isNumber(value) {
    return typeof value === 'number' && !isNaN(value);
}

function isArray(value) {
    return Array.isArray(value);
}

function isObject(value) {
    return value !== null && typeof value === 'object';
}

function isFunction(value) {
    return typeof value === 'function';
}

// 使用Object.prototype.toString进行精确类型检查
function getType(value) {
    return Object.prototype.toString.call(value).slice(8, -1);
}

console.log(getType([])); // 'Array'
console.log(getType({})); // 'Object'
console.log(getType(null)); // 'Null'
console.log(getType(undefined)); // 'Undefined'
```

### 安全的转换函数
```javascript
// 安全的转换函数
function safeParseInt(value, defaultValue = 0) {
    const parsed = parseInt(value, 10);
    return isNaN(parsed) ? defaultValue : parsed;
}

function safeParseFloat(value, defaultValue = 0) {
    const parsed = parseFloat(value);
    return isNaN(parsed) ? defaultValue : parsed;
}

function safeString(value, defaultValue = '') {
    return value != null ? String(value) : defaultValue;
}

function safeBoolean(value, defaultValue = false) {
    return value != null ? Boolean(value) : defaultValue;
}

// 使用示例
console.log(safeParseInt('123')); // 123
console.log(safeParseInt('abc')); // 0
console.log(safeParseInt('abc', -1)); // -1

console.log(safeParseFloat('123.45')); // 123.45
console.log(safeParseFloat('abc')); // 0
console.log(safeParseFloat('abc', -1)); // -1
```

### 参数验证函数
```javascript
// 参数验证
function validateUser(user) {
    if (!isObject(user)) {
        throw new TypeError('Expected an object');
    }
    
    if (!isString(user.name)) {
        throw new TypeError('Expected user.name to be a string');
    }
    
    if (!isNumber(user.age)) {
        throw new TypeError('Expected user.age to be a number');
    }
    
    return true;
}

// 使用示例
try {
    validateUser({ name: 'John', age: 30 }); // ✅ 有效
    validateUser({ name: 'John' }); // ❌ 缺少age
    validateUser('invalid'); // ❌ 不是对象
} catch (error) {
    console.error('Validation error:', error.message);
}
```
