## 条件语句示例

### if语句示例
```javascript
// 1. 基本if语句
let age = 18;
if (age >= 18) {
    console.log('成年人');
}

// 2. if-else语句
if (age >= 18) {
    console.log('成年人');
} else {
    console.log('未成年人');
}

// 3. if-else if-else语句
let score = 85;
if (score >= 90) {
    console.log('优秀');
} else if (score >= 80) {
    console.log('良好');
} else if (score >= 70) {
    console.log('中等');
} else if (score >= 60) {
    console.log('及格');
} else {
    console.log('不及格');
}

// 4. 复杂条件判断
let user = {
    name: 'John',
    age: 25,
    isVip: true,
    hasPermission: false
};

if (user.age >= 18 && user.isVip) {
    console.log('VIP成年人');
} else if (user.age >= 18 || user.hasPermission) {
    console.log('有权限用户');
} else {
    console.log('无权限用户');
}

// 5. 嵌套if语句
if (user.age >= 18) {
    if (user.isVip) {
        console.log('VIP成年人');
    } else {
        console.log('普通成年人');
    }
} else {
    console.log('未成年人');
}

// 6. 早期返回模式
function processUser(user) {
    if (!user) return '用户不存在';
    if (!user.name) return '用户名不能为空';
    if (user.age < 0) return '年龄不能为负数';
    
    return `处理用户: ${user.name}`;
}
```

```javascript
// 4. 现代switch用法
function processUser(user) {
    switch (user.role) {
        case 'admin': {
            const permissions = ['read', 'write', 'delete'];
            return { ...user, permissions };
        }
        case 'user': {
            const permissions = ['read'];
            return { ...user, permissions };
        }
        case 'guest': {
            const permissions = [];
            return { ...user, permissions };
        }
        default:
```

## 循环语句示例

### for循环示例
```javascript
// 1. 基本for循环
for (let i = 0; i < 5; i++) {
    console.log(i); // 0, 1, 2, 3, 4
}

// 2. 倒序循环
for (let i = 4; i >= 0; i--) {
    console.log(i); // 4, 3, 2, 1, 0
}

// 3. 步长循环
for (let i = 0; i < 10; i += 2) {
    console.log(i); // 0, 2, 4, 6, 8
}

// 4. 多变量循环
for (let i = 0, j = 10; i < j; i++, j--) {
    console.log(`i: ${i}, j: ${j}`);
}

// 5. 嵌套循环
for (let i = 0; i < 3; i++) {
    for (let j = 0; j < 3; j++) {
        console.log(`(${i}, ${j})`);
    }
}

// 6. 条件循环
for (let i = 0; i < 10; i++) {
    if (i % 2 === 0) {
        console.log(`${i} 是偶数`);
    }
}

// 7. 提前终止
for (let i = 0; i < 10; i++) {
    if (i === 5) {
        break; // 跳出循环
    }
    console.log(i); // 0, 1, 2, 3, 4
}

// 8. 跳过当前迭代
for (let i = 0; i < 10; i++) {
    if (i % 2 === 0) {
        continue; // 跳过偶数
    }
    console.log(i); // 1, 3, 5, 7, 9
}

// 9. 使用let避免变量提升问题
for (let i = 0; i < 3; i++) {
    setTimeout(() => {
        console.log(i); // 0, 1, 2
    }, 100);
}

// 10. 缓存数组长度
const arr = [1, 2, 3, 4, 5];
for (let i = 0, len = arr.length; i < len; i++) {
    console.log(arr[i]);
}
```

### while循环示例

```javascript
let i = 0;
while (i < 5) {
    console.log(i); // 0, 1, 2, 3, 4

while (count < 10) {

    count++;

function getValidNumber() {
    let input;
    while (true) {
        input = prompt('请输入一个数字:');
        if (!isNaN(input) && input !== '') {
            return Number(input);
        }
        console.log('请输入有效的数字!');
    }
}

// 5. 文件读取模拟
function readFile() {
    let content = '';
    let line;
    while ((line = getNextLine()) !== null) {
        content += line + '\n';
    }
    return content;
}

// 6. 游戏循环
function gameLoop() {
    let gameRunning = true;
    let score = 0;

        switch (action) {
            case 'move':
                score += 10;
                break;
            case 'quit':
                gameRunning = false;
                break;

        if (score >= 100) {
            console.log('游戏胜利!');
            gameRunning = false;
        }
    }
```

### break语句示例
```javascript
// 1. 在循环中使用break
for (let i = 0; i < 10; i++) {
    if (i === 5) {
        break; // 跳出循环
    }
    console.log(i); // 0, 1, 2, 3, 4
}

// 2. 在while循环中使用break
let i = 0;
while (i < 10) {
    if (i === 5) {
        break; // 跳出循环
    }
    console.log(i); // 0, 1, 2, 3, 4
    i++;
}

// 3. 在switch中使用break
let day = 3;
switch (day) {
    case 1:
        console.log('星期一');
        break; // 防止穿透
    case 2:
        console.log('星期二');
        break;
    case 3:
        console.log('星期三');
        break;
    default:
        console.log('其他');
}

// 4. 标签break语句
outerLoop: for (let i = 0; i < 3; i++) {
    innerLoop: for (let j = 0; j < 3; j++) {
        if (i === 1 && j === 1) {
            break outerLoop; // 跳出外层循环
        }
        console.log(`(${i}, ${j})`);
    }
}

// 5. 复杂嵌套结构
function processMatrix(matrix) {
    rowLoop: for (let i = 0; i < matrix.length; i++) {
        colLoop: for (let j = 0; j < matrix[i].length; j++) {
            if (matrix[i][j] === null) {
                console.log(`发现空值 at (${i}, ${j})`);
                break rowLoop; // 跳出行循环
            }
            console.log(`处理 (${i}, ${j}): ${matrix[i][j]}`);
        }
    }
}
```

```javascript
// 1. 用户输入验证
function validateUserInput(input) {
    const errors = [];
    
    if (!input.name) {
        errors.push('姓名不能为空');
    }
    
    if (!input.email) {
        errors.push('邮箱不能为空');
    } else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(input.email)) {
        errors.push('邮箱格式不正确');
    }
    
    if (!input.age) {
        errors.push('年龄不能为空');
    } else if (input.age < 0 || input.age > 150) {
        errors.push('年龄必须在0-150之间');
    }
    
    return {
        isValid: errors.length === 0,
        errors
    };
}

// 2. 数据过滤
function filterValidUsers(users) {
    const validUsers = [];
    
    for (let user of users) {
        if (!user.name || !user.email) {
            continue; // 跳过无效用户
        }
        
        if (user.age < 18) {
            continue; // 跳过未成年用户
        }
        
        validUsers.push(user);
    }
    
    return validUsers;
}

// 3. 数据查找
function findUserById(users, id) {
    for (let user of users) {
        if (user.id === id) {
            return user; // 找到后立即返回
        }
    }
    return null; // 未找到
}
```

### 算法实现示例
```javascript
// 1. 冒泡排序
function bubbleSort(arr) {
    const n = arr.length;
    
    for (let i = 0; i < n - 1; i++) {
        let swapped = false;
        
        for (let j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                // 交换元素
                [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
                swapped = true;
            }
        }
        
        if (!swapped) {
            break; // 如果没有交换，说明已经排序完成
        }
    }
    
    return arr;
}

// 2. 二分查找
function binarySearch(arr, target) {
    let left = 0;
    let right = arr.length - 1;
    
    while (left <= right) {
        const mid = Math.floor((left + right) / 2);
        
        if (arr[mid] === target) {
            return mid; // 找到目标
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    
    return -1; // 未找到
}

// 3. 质数判断
function isPrime(n) {
    if (n < 2) return false;
    if (n === 2) return true;
    if (n % 2 === 0) return false;
    
    for (let i = 3; i <= Math.sqrt(n); i += 2) {
        if (n % i === 0) {
            return false;
        }
    }
    
    return true;
}
```

```javascript
function stateMachine() {
    let currentState = 'idle';
    let stateCount = 0;
    
    while (stateCount < 100) {
        switch (currentState) {
            case 'idle':
                if (Math.random() > 0.7) {
                    currentState = 'working';
                }
                break;
                
            case 'working':
                if (Math.random() > 0.8) {

                break;

                    currentState = 'idle';

    }
}
```
