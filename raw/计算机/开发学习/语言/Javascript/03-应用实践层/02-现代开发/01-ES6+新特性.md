# ES6+新特性

## ES6核心特性

### 变量声明
```javascript
// 1. let 和 const
// let - 块级作用域变量
function example1() {
    if (true) {
        let x = 1;
        console.log(x); // 1
    }
    // console.log(x); // ReferenceError: x is not defined
}

// const - 常量声明
const PI = 3.14159;
// PI = 3.14; // TypeError: Assignment to constant variable

// const 对象和数组
const person = { name: 'John', age: 30 };
person.age = 31; // 可以修改对象属性
// person = {}; // TypeError: Assignment to constant variable

const numbers = [1, 2, 3];
numbers.push(4); // 可以修改数组
// numbers = []; // TypeError: Assignment to constant variable
```

### 箭头函数
```javascript
// 1. 基本语法
// 传统函数
function add(a, b) {
    return a + b;
}

// 箭头函数
const add = (a, b) => a + b;

// 2. 不同形式
// 单个参数
const square = x => x * x;

// 无参数
const greet = () => 'Hello World';

// 多行函数体
const processData = (data) => {
    const result = data.map(item => item * 2);
    return result.filter(item => item > 10);
};

// 3. this 绑定
class Timer {
    constructor() {
        this.seconds = 0;
    }
    
    start() {
        // 传统函数 - this 指向调用者
        setInterval(function() {
            this.seconds++; // this 指向全局对象
        }, 1000);
        
        // 箭头函数 - this 指向定义时的上下文
        setInterval(() => {
            this.seconds++; // this 指向 Timer 实例
        }, 1000);
    }
}
```

### 模板字符串
```javascript
// 1. 基本用法
const name = 'John';
const age = 30;

// 传统字符串拼接
const message1 = 'Hello, ' + name + '! You are ' + age + ' years old.';

// 模板字符串
const message2 = `Hello, ${name}! You are ${age} years old.`;

// 2. 多行字符串
const html = `
    <div class="user">
        <h2>${name}</h2>
        <p>Age: ${age}</p>
        <p>Status: ${age >= 18 ? 'Adult' : 'Minor'}</p>
    </div>
`;

// 3. 标签模板
function highlight(strings, ...values) {
    return strings.reduce((result, string, i) => {
        const value = values[i] ? `<mark>${values[i]}</mark>` : '';
        return result + string + value;
    }, '');
}

const name = 'John';
const age = 30;
const highlighted = highlight`Hello ${name}, you are ${age} years old.`;
// 结果: "Hello <mark>John</mark>, you are <mark>30</mark> years old."
```

### 解构赋值
```javascript
// 1. 数组解构
const numbers = [1, 2, 3, 4, 5];

// 基本解构
const [first, second, third] = numbers;
console.log(first, second, third); // 1 2 3

// 跳过元素
const [a, , c] = numbers;
console.log(a, c); // 1 3

// 剩余元素
const [head, ...tail] = numbers;
console.log(head, tail); // 1 [2, 3, 4, 5]

// 默认值
const [x, y, z = 0] = [1, 2];
console.log(x, y, z); // 1 2 0

// 2. 对象解构
const person = {
    name: 'John',
    age: 30,
    address: {
        city: 'New York',
        country: 'USA'
    }
};

// 基本解构
const { name, age } = person;
console.log(name, age); // John 30

// 重命名
const { name: personName, age: personAge } = person;
console.log(personName, personAge); // John 30

// 嵌套解构
const { address: { city, country } } = person;
console.log(city, country); // New York USA

// 默认值
const { name, age, email = 'no-email@example.com' } = person;
console.log(email); // no-email@example.com

// 3. 函数参数解构
function greet({ name, age = 0 }) {
    return `Hello ${name}, you are ${age} years old.`;
}

const user = { name: 'John', age: 30 };
console.log(greet(user)); // Hello John, you are 30 years old.
```

### 扩展运算符
```javascript
// 1. 数组扩展
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];

// 合并数组
const combined = [...arr1, ...arr2];
console.log(combined); // [1, 2, 3, 4, 5, 6]

// 复制数组
const copy = [...arr1];
console.log(copy); // [1, 2, 3]

// 2. 对象扩展
const obj1 = { a: 1, b: 2 };
const obj2 = { c: 3, d: 4 };

// 合并对象
const merged = { ...obj1, ...obj2 };
console.log(merged); // { a: 1, b: 2, c: 3, d: 4 }

// 复制对象
const copy = { ...obj1 };
console.log(copy); // { a: 1, b: 2 }

// 3. 函数参数
function sum(...numbers) {
    return numbers.reduce((total, num) => total + num, 0);
}

console.log(sum(1, 2, 3, 4, 5)); // 15

// 4. 实际应用
// 数组去重
const numbers = [1, 2, 2, 3, 3, 4];
const unique = [...new Set(numbers)];
console.log(unique); // [1, 2, 3, 4]

// 对象属性覆盖
const defaultConfig = { theme: 'light', language: 'en' };
const userConfig = { theme: 'dark' };
const finalConfig = { ...defaultConfig, ...userConfig };
console.log(finalConfig); // { theme: 'dark', language: 'en' }
```

## ES7-ES9特性

### ES7特性
```javascript
// 1. Array.prototype.includes()
const numbers = [1, 2, 3, 4, 5];

// 传统方式
if (numbers.indexOf(3) !== -1) {
    console.log('Found 3');
}

// ES7方式
if (numbers.includes(3)) {
    console.log('Found 3');
}

// 2. 指数运算符
// 传统方式
const result1 = Math.pow(2, 3); // 8

// ES7方式
const result2 = 2 ** 3; // 8
const result3 = 2 ** 3 ** 2; // 512 (从右到左计算)
```

### ES8特性
```javascript
// 1. async/await
async function fetchUserData(userId) {
    try {
        const response = await fetch(`/api/users/${userId}`);
        const user = await response.json();
        return user;
    } catch (error) {
        console.error('Error fetching user:', error);
        throw error;
    }
}

// 2. Object.entries() 和 Object.values()
const person = { name: 'John', age: 30, city: 'New York' };

// Object.entries() - 返回键值对数组
const entries = Object.entries(person);
console.log(entries); // [['name', 'John'], ['age', 30], ['city', 'New York']]

// Object.values() - 返回值数组
const values = Object.values(person);
console.log(values); // ['John', 30, 'New York']

// 3. String padding
const str = 'Hello';
console.log(str.padStart(10, ' ')); // '     Hello'
console.log(str.padEnd(10, '!'));   // 'Hello!!!!!'

// 4. 函数参数列表和调用中的尾随逗号
function greet(
    name,
    age,
    city, // 尾随逗号
) {
    return `Hello ${name}, age ${age}, from ${city}`;
}
```

### ES9特性
```javascript
// 1. 异步迭代
async function* asyncGenerator() {
    yield 1;
    yield 2;
    yield 3;
}

async function processAsyncData() {
    for await (const value of asyncGenerator()) {
        console.log(value);
    }
}

// 2. Promise.finally()
fetch('/api/data')
    .then(response => response.json())
    .then(data => console.log(data))
    .catch(error => console.error(error))
    .finally(() => {
        console.log('请求完成');
    });

// 3. 正则表达式改进
// 命名捕获组
const regex = /(?<year>\d{4})-(?<month>\d{2})-(?<day>\d{2})/;
const match = regex.exec('2023-12-25');
console.log(match.groups.year);  // 2023
console.log(match.groups.month); // 12
console.log(match.groups.day);   // 25

// 反向断言
const text = 'Hello World';
const lookbehind = /(?<=Hello )\w+/;
console.log(text.match(lookbehind)); // ['World']

// 4. Rest/Spread 属性
const { a, b, ...rest } = { a: 1, b: 2, c: 3, d: 4 };
console.log(rest); // { c: 3, d: 4 }

const obj = { a: 1, b: 2 };
const newObj = { ...obj, c: 3, d: 4 };
console.log(newObj); // { a: 1, b: 2, c: 3, d: 4 }
```

## ES10-ES13特性

### ES10特性
```javascript
// 1. Array.prototype.flat()
const nested = [1, [2, 3], [4, [5, 6]]];
console.log(nested.flat());     // [1, 2, 3, 4, [5, 6]]
console.log(nested.flat(2));    // [1, 2, 3, 4, 5, 6]
console.log(nested.flat(Infinity)); // [1, 2, 3, 4, 5, 6]

// 2. Array.prototype.flatMap()
const numbers = [1, 2, 3, 4];
const doubled = numbers.flatMap(x => [x, x * 2]);
console.log(doubled); // [1, 2, 2, 4, 3, 6, 4, 8]

// 3. Object.fromEntries()
const entries = [['name', 'John'], ['age', 30]];
const obj = Object.fromEntries(entries);
console.log(obj); // { name: 'John', age: 30 }

// 4. String.prototype.trimStart() 和 trimEnd()
const str = '   Hello World   ';
console.log(str.trimStart()); // 'Hello World   '
console.log(str.trimEnd());   // '   Hello World'
```

### ES11特性
```javascript
// 1. 可选链操作符
const user = {
    name: 'John',
    address: {
        street: '123 Main St',
        city: 'New York'
    }
};

// 传统方式
const street = user && user.address && user.address.street;

// ES11方式
const street = user?.address?.street;
const phone = user?.phone?.number; // undefined

// 2. 空值合并操作符
const name = user.name ?? 'Anonymous';
const age = user.age ?? 0;
const city = user.address?.city ?? 'Unknown';

// 3. BigInt
const bigNumber = 123456789012345678901234567890n;
const anotherBig = BigInt('123456789012345678901234567890');
console.log(bigNumber + anotherBig);

// 4. 动态导入
async function loadModule() {
    const module = await import('./my-module.js');
    return module.default;
}
```

### ES12特性
```javascript
// 1. 逻辑赋值操作符
let x = 1;
let y = null;
let z = undefined;

// 逻辑或赋值
x ||= 2; // x = x || 2
y ||= 2; // y = 2

// 逻辑与赋值
x &&= 3; // x = x && 3
y &&= 3; // y = null

// 空值合并赋值
z ??= 4; // z = 4
x ??= 4; // x = 3 (不变)

// 2. 数字分隔符
const million = 1_000_000;
const binary = 0b1010_0001;
const hex = 0xFF_EC_DE_5E;

// 3. String.prototype.replaceAll()
const text = 'Hello World Hello';
const newText = text.replaceAll('Hello', 'Hi');
console.log(newText); // 'Hi World Hi'
```

### ES13特性
```javascript
// 1. 顶层 await
// 在模块顶层直接使用 await
const data = await fetch('/api/data').then(r => r.json());
console.log(data);

// 2. 私有字段和方法
class BankAccount {
    #balance = 0; // 私有字段
    #accountNumber; // 私有字段
    
    constructor(accountNumber) {
        this.#accountNumber = accountNumber;
    }
    
    #validateAmount(amount) { // 私有方法
        if (amount < 0) {
            throw new Error('Amount cannot be negative');
        }
    }
    
    deposit(amount) {
        this.#validateAmount(amount);
        this.#balance += amount;
    }
    
    getBalance() {
        return this.#balance;
    }
}

// 3. 静态私有字段和方法
class MathUtils {
    static #PI = 3.14159;
    
    static #validateNumber(num) {
        if (typeof num !== 'number') {
            throw new Error('Input must be a number');
        }
    }
    
    static getPI() {
        return this.#PI;
    }
    
    static calculateArea(radius) {
        this.#validateNumber(radius);
        return this.#PI * radius * radius;
    }
}
```

## 实际应用示例

### 现代JavaScript项目结构
```javascript
// 1. 模块化组织
// utils/helpers.js
export const formatDate = (date) => {
    return new Intl.DateTimeFormat('zh-CN').format(date);
};

export const debounce = (func, delay) => {
    let timeoutId;
    return (...args) => {
        clearTimeout(timeoutId);
        timeoutId = setTimeout(() => func.apply(null, args), delay);
    };
};

// 2. 类定义
// models/User.js
export class User {
    #id;
    #name;
    #email;
    
    constructor(id, name, email) {
        this.#id = id;
        this.#name = name;
        this.#email = email;
    }
    
    get id() { return this.#id; }
    get name() { return this.#name; }
    get email() { return this.#email; }
    
    toJSON() {
        return {
            id: this.#id,
            name: this.#name,
            email: this.#email
        };
    }
}

// 3. 服务层
// services/UserService.js
export class UserService {
    constructor(apiClient) {
        this.apiClient = apiClient;
    }
    
    async getUsers() {
        try {
            const response = await this.apiClient.get('/users');
            return response.map(user => new User(user.id, user.name, user.email));
        } catch (error) {
            console.error('获取用户失败:', error);
            throw error;
        }
    }
    
    async createUser(userData) {
        const response = await this.apiClient.post('/users', userData);
        return new User(response.id, response.name, response.email);
    }
}

// 4. 主应用
// app.js
import { UserService } from './services/UserService.js';
import { formatDate, debounce } from './utils/helpers.js';

class App {
    constructor() {
        this.userService = new UserService(apiClient);
        this.setupEventListeners();
    }
    
    setupEventListeners() {
        const searchInput = document.getElementById('search');
        const debouncedSearch = debounce(this.searchUsers.bind(this), 300);
        searchInput.addEventListener('input', debouncedSearch);
    }
    
    async searchUsers(event) {
        const query = event.target.value;
        if (query.length < 2) return;
        
        try {
            const users = await this.userService.searchUsers(query);
            this.renderUsers(users);
        } catch (error) {
            this.showError('搜索失败');
        }
    }
    
    renderUsers(users) {
        const container = document.getElementById('users');
        container.innerHTML = users.map(user => `
            <div class="user-card">
                <h3>${user.name}</h3>
                <p>${user.email}</p>
                <p>创建时间: ${formatDate(user.createdAt)}</p>
            </div>
        `).join('');
    }
}

// 启动应用
const app = new App();
```
