# 元素选择与操作

## 元素选择方法

### 基础选择器
```javascript
// 1. getElementById - 通过ID选择
function selectById() {
    var element = document.getElementById('myId');
    console.log('Element by ID:', element);
    
    // 检查元素是否存在
    if (element) {
        element.style.color = 'red';
    }
}

// 2. getElementsByClassName - 通过类名选择
function selectByClassName() {
    var elements = document.getElementsByClassName('myClass');
    console.log('Elements by class:', elements);
    console.log('Length:', elements.length);
    
    // 遍历HTMLCollection
    for (var i = 0; i < elements.length; i++) {
        elements[i].style.backgroundColor = 'yellow';
    }
}

// 3. getElementsByTagName - 通过标签名选择
function selectByTagName() {
    var elements = document.getElementsByTagName('div');
    console.log('Elements by tag:', elements);
    
    // 转换为数组
    var elementsArray = Array.from(elements);
    elementsArray.forEach(function(element) {
        element.style.border = '1px solid black';
    });
}

// 4. getElementsByName - 通过name属性选择
function selectByName() {
    var elements = document.getElementsByName('myName');
    console.log('Elements by name:', elements);
    
    // 处理表单元素
    Array.from(elements).forEach(function(element) {
        if (element.type === 'checkbox') {
            element.checked = true;
        }
    });
}
```

### 现代选择器
```javascript
// 1. querySelector - 选择单个元素
function selectSingle() {
    // CSS选择器语法
    var element1 = document.querySelector('#myId');
    var element2 = document.querySelector('.myClass');
    var element3 = document.querySelector('div.container');
    var element4 = document.querySelector('[data-value="123"]');
    var element5 = document.querySelector('div:first-child');
    
    console.log('Selected elements:', {
        byId: element1,
        byClass: element2,
        byTag: element3,
        byAttribute: element4,
        byPseudo: element5
    });
}

// 2. querySelectorAll - 选择多个元素
function selectMultiple() {
    // 选择所有匹配的元素
    var elements1 = document.querySelectorAll('.item');
    var elements2 = document.querySelectorAll('div[data-type="card"]');
    var elements3 = document.querySelectorAll('li:nth-child(odd)');
    var elements4 = document.querySelectorAll('input[type="text"]');
    
    console.log('Multiple selections:', {
        items: elements1.length,
        cards: elements2.length,
        oddItems: elements3.length,
        textInputs: elements4.length
    });
    
    // 转换为数组并操作
    Array.from(elements1).forEach(function(element, index) {
        element.dataset.index = index;
        element.style.opacity = 0.8;
    });
}

// 3. 复杂选择器
function complexSelectors() {
    // 后代选择器
    var descendants = document.querySelectorAll('.container .item');
    
    // 子元素选择器
    var children = document.querySelectorAll('.container > .item');
    
    // 相邻兄弟选择器
    var adjacent = document.querySelectorAll('.item + .item');
    
    // 通用兄弟选择器
    var general = document.querySelectorAll('.item ~ .item');
    
    // 属性选择器
    var withAttribute = document.querySelectorAll('[data-value]');
    var exactValue = document.querySelectorAll('[data-value="123"]');
    var startsWith = document.querySelectorAll('[data-value^="12"]');
    var endsWith = document.querySelectorAll('[data-value$="23"]');
    var contains = document.querySelectorAll('[data-value*="2"]');
    
    console.log('Complex selectors:', {
        descendants: descendants.length,
        children: children.length,
        adjacent: adjacent.length,
        general: general.length,
        withAttribute: withAttribute.length,
        exactValue: exactValue.length,
        startsWith: startsWith.length,
        endsWith: endsWith.length,
        contains: contains.length
    });
}
```

## 元素操作

### 属性操作
```javascript
// 1. 标准属性
function standardAttributes() {
    var element = document.getElementById('myElement');
    
    // 设置属性
    element.id = 'newId';
    element.className = 'new-class';
    element.title = 'New Title';
    element.href = 'https://example.com';
    
    // 获取属性
    console.log('ID:', element.id);
    console.log('Class:', element.className);
    console.log('Title:', element.title);
    
    // 检查属性
    console.log('Has ID:', element.hasAttribute('id'));
    console.log('Has Class:', element.hasAttribute('class'));
}

// 2. 自定义属性
function customAttributes() {
    var element = document.getElementById('myElement');
    
    // 使用setAttribute/getAttribute
    element.setAttribute('data-custom', 'custom-value');
    element.setAttribute('aria-label', 'Custom label');
    element.setAttribute('role', 'button');
    
    console.log('Custom attribute:', element.getAttribute('data-custom'));
    console.log('Aria label:', element.getAttribute('aria-label'));
    
    // 使用dataset（推荐方式）
    element.dataset.userId = '12345';
    element.dataset.role = 'admin';
    element.dataset.status = 'active';
    
    console.log('User ID:', element.dataset.userId);
    console.log('Role:', element.dataset.role);
    console.log('Status:', element.dataset.status);
    
    // 删除属性
    element.removeAttribute('data-custom');
    delete element.dataset.role;
}

// 3. 属性遍历
function iterateAttributes() {
    var element = document.getElementById('myElement');
    
    // 遍历所有属性
    console.log('All attributes:');
    for (var i = 0; i < element.attributes.length; i++) {
        var attr = element.attributes[i];
        console.log(attr.name + '=' + attr.value);
    }
    
    // 使用for...of遍历
    console.log('Using for...of:');
    for (var attr of element.attributes) {
        console.log(attr.name + '=' + attr.value);
    }
    
    // 转换为对象
    var attrObj = {};
    for (var attr of element.attributes) {
        attrObj[attr.name] = attr.value;
    }
    console.log('Attributes object:', attrObj);
}
```

```javascript
// 1. 内联样式
function inlineStyles() {
    var element = document.getElementById('myElement');
    
    // 设置样式
    element.style.color = 'red';
    element.style.backgroundColor = 'yellow';
    element.style.fontSize = '16px';
    element.style.marginTop = '10px';
    element.style.border = '1px solid black';
    
    // 获取样式
    console.log('Color:', element.style.color);
    console.log('Background:', element.style.backgroundColor);
    console.log('Font size:', element.style.fontSize);
    
    // 移除样式
    element.style.color = '';
    element.style.removeProperty('background-color');
    
    // 批量设置样式
    var styles = {
        color: 'blue',
        fontSize: '18px',
        fontWeight: 'bold',
        textAlign: 'center'
    };
    
    Object.assign(element.style, styles);
}

// 2. 类名操作
function classNameOperations() {
    var element = document.getElementById('myElement');
    
    // 添加类名
    element.classList.add('active');
    element.classList.add('highlight', 'selected');
    
    // 移除类名
    element.classList.remove('inactive');
    element.classList.remove('old-class', 'deprecated');
    
    // 切换类名
    element.classList.toggle('visible');
    element.classList.toggle('hidden', true); // 强制添加
    element.classList.toggle('shown', false); // 强制移除
    
    // 检查类名
    console.log('Has active:', element.classList.contains('active'));
    console.log('Has inactive:', element.classList.contains('inactive'));
    
    // 替换类名
    element.classList.replace('old-class', 'new-class');
    
    // 获取所有类名
    console.log('All classes:', element.classList.toString());
    console.log('Class list:', Array.from(element.classList));
    
    // 条件添加类名
    if (element.dataset.status === 'active') {
        element.classList.add('status-active');
    }
}

// 3. 计算样式
function computedStyles() {
    var element = document.getElementById('myElement');
    
    // 获取计算样式
    var computedStyle = window.getComputedStyle(element);
    
    console.log('Computed styles:');
    console.log('Color:', computedStyle.color);
    console.log('Font size:', computedStyle.fontSize);
    console.log('Margin:', computedStyle.margin);
    console.log('Display:', computedStyle.display);
    console.log('Position:', computedStyle.position);
    
    // 获取特定属性
    var color = computedStyle.getPropertyValue('color');
    var fontSize = computedStyle.getPropertyValue('font-size');
    var margin = computedStyle.getPropertyValue('margin');
    
    console.log('Specific properties:', { color, fontSize, margin });
    
    // 检查样式是否应用
    function isStyleApplied(element, property, value) {
        var computed = window.getComputedStyle(element);
        return computed.getPropertyValue(property) === value;
    }
    
    console.log('Is color red:', isStyleApplied(element, 'color', 'rgb(255, 0, 0)'));
}
```

```javascript
function dynamicContent() {
    var container = document.getElementById('container');
    var data = [
        { id: 1, name: 'Item 1', status: 'active' },
        { id: 2, name: 'Item 2', status: 'inactive' },
        { id: 3, name: 'Item 3', status: 'active' }
    ];

            item.querySelector('.name').textContent = newData.name;
            item.querySelector('.status').textContent = newData.status;
            item.className = 'item ' + newData.status;
        }
    }
    
    // 使用更新函数
    updateItem(1, { name: 'Updated Item 1', status: 'inactive' });
```

### 遍历方法
```javascript
// 1. 子元素遍历
function traverseChildren() {
    var parent = document.getElementById('parent');
    
    // 遍历所有子节点
    console.log('All child nodes:');
    for (var i = 0; i < parent.childNodes.length; i++) {
        var child = parent.childNodes[i];
        console.log('Node type:', child.nodeType, 'Content:', child.textContent);
    }
    
    // 遍历元素子节点
    console.log('Element children:');
    for (var i = 0; i < parent.children.length; i++) {
        var child = parent.children[i];
        console.log('Tag:', child.tagName, 'Class:', child.className);
    }
    
    // 使用for...of遍历
    console.log('Using for...of:');
    for (var child of parent.children) {
        console.log('Child:', child.tagName);
    }
}

// 2. 父元素遍历
function traverseParents() {
    var element = document.getElementById('child');
    
    // 遍历所有父节点
    var current = element;
    while (current.parentNode) {
        current = current.parentNode;
        console.log('Parent:', current.tagName, 'ID:', current.id);
        
        // 避免遍历到document
        if (current === document) break;
    }
    
    // 使用closest方法（现代浏览器）
    var closest = element.closest('.container');
    console.log('Closest container:', closest);
    
    // 检查是否包含在特定元素中
    var container = document.getElementById('container');
    var isContained = container.contains(element);
    console.log('Is contained:', isContained);
}

// 3. 兄弟元素遍历
function traverseSiblings() {
    var element = document.getElementById('middle');
    
    // 遍历前面的兄弟元素
    var prev = element.previousElementSibling;
    while (prev) {
        console.log('Previous sibling:', prev.tagName);
        prev = prev.previousElementSibling;
    }
    
    // 遍历后面的兄弟元素
    var next = element.nextElementSibling;
    while (next) {
        console.log('Next sibling:', next.tagName);
        next = next.nextElementSibling;
    }
    
    // 获取所有兄弟元素
    var parent = element.parentNode;
    var siblings = Array.from(parent.children).filter(function(child) {
        return child !== element;
    });
    console.log('All siblings:', siblings);
}

// 4. 深度遍历
function deepTraverse() {
    var root = document.getElementById('root');
    
    function traverse(node, level = 0) {
        var indent = '  '.repeat(level);
        console.log(indent + node.tagName + ' (' + node.nodeType + ')');
        
        if (node.nodeType === Node.ELEMENT_NODE) {
            console.log(indent + '  ID:', node.id);
            console.log(indent + '  Class:', node.className);
            console.log(indent + '  Attributes:', node.attributes.length);
        }
        
        if (node.nodeType === Node.TEXT_NODE) {
            var text = node.textContent.trim();
            if (text) {
                console.log(indent + '  Text: "' + text + '"');
            }
        }
        
        // 递归遍历子节点
        for (var i = 0; i < node.childNodes.length; i++) {
            traverse(node.childNodes[i], level + 1);
        }
    }
    
    traverse(root);
}
```

```javascript
// 1. 在元素内查找
function findWithinElement() {
    var container = document.getElementById('container');
    
    // 在容器内查找
    var items = container.querySelectorAll('.item');
    var activeItems = container.querySelectorAll('.item.active');
    var firstItem = container.querySelector('.item:first-child');
    
    console.log('Items in container:', items.length);
    console.log('Active items:', activeItems.length);
    console.log('First item:', firstItem);
    
    // 查找特定类型的元素
    var buttons = container.getElementsByTagName('button');
    var inputs = container.getElementsByTagName('input');
    
    console.log('Buttons:', buttons.length);
    console.log('Inputs:', inputs.length);
}

// 2. 条件查找
function conditionalFind() {
    var container = document.getElementById('container');
    
    // 查找满足条件的元素
    var elements = Array.from(container.children).filter(function(element) {
        return element.dataset.value > 50;
    });
    
    console.log('Elements with value > 50:', elements);
    
    // 查找包含特定文本的元素
    var textElements = Array.from(container.querySelectorAll('*')).filter(function(element) {
        return element.textContent.includes('specific text');
    });
    
    console.log('Elements with specific text:', textElements);
    
    // 查找可见元素
    var visibleElements = Array.from(container.children).filter(function(element) {
        var style = window.getComputedStyle(element);
        return style.display !== 'none' && style.visibility !== 'hidden';
    });
    
    console.log('Visible elements:', visibleElements);
}

// 3. 查找优化
function findOptimization() {
    var container = document.getElementById('container');
    
    // 缓存查找结果
    var cachedElements = {
        items: container.querySelectorAll('.item'),
        buttons: container.querySelectorAll('button'),
        inputs: container.querySelectorAll('input')
    };
    
    // 使用缓存的结果
    function useCachedElements() {
        cachedElements.items.forEach(function(item) {
            item.style.opacity = 0.8;
        });
        
        cachedElements.buttons.forEach(function(button) {
            button.disabled = false;
        });
    }
    
    // 更新缓存
    function updateCache() {
        cachedElements.items = container.querySelectorAll('.item');
        cachedElements.buttons = container.querySelectorAll('button');
        cachedElements.inputs = container.querySelectorAll('input');
    }
    
    return { useCachedElements, updateCache };
}
```
