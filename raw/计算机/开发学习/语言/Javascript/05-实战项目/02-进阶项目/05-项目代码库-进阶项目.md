```javascript
class App {
  constructor() {
    this.router = new Router();
    this.store = new Store();
    this.init();
  }
  
  init() {
    this.setupRoutes();
    this.setupEventListeners();
    this.render();
  }
  
  setupRoutes() {
    this.router
      .addRoute('/', () => new HomePage())
      .addRoute('/about', () => new AboutPage())
      .addRoute('/contact', () => new ContactPage())
      .addRoute('/user/:id', (route) => new UserPage(route.params.id));
  }
  
  setupEventListeners() {
    // 全局事件监听
    document.addEventListener('click', (e) => {
      if (e.target.matches('[data-route]')) {
        e.preventDefault();
        const path = e.target.getAttribute('data-route');
        this.router.navigate(path);
      }
    });
  }
  
  render() {
    const container = document.getElementById('app');
    container.innerHTML = `
      <header class="app-header">
        <nav class="main-nav">
          <a href="#" data-route="/">首页</a>
          <a href="#" data-route="/about">关于</a>
          <a href="#" data-route="/contact">联系</a>
        </nav>
      </header>
      <main class="app-main" id="main-content"></main>
      <footer class="app-footer">
        <p>&copy; 2023 SPA应用示例</p>
      </footer>
    `;
  }
}
```

```javascript
class ChartManager {
  constructor() {
    this.charts = new Map();
    this.dataSource = new DataSource();
  }
  
  // 创建折线图
  createLineChart(containerId, data) {
    const ctx = document.getElementById(containerId).getContext('2d');
    const chart = new Chart(ctx, {
      type: 'line',
      data: {
        labels: data.labels,
        datasets: [{
          label: data.label,
          data: data.values,
          borderColor: 'rgb(75, 192, 192)',
          backgroundColor: 'rgba(75, 192, 192, 0.2)',
          tension: 0.4
        }]
      },
      options: {
        responsive: true,
        plugins: {
          title: {
            display: true,
            text: data.title
          }
        }
      }
    });
    
    this.charts.set(containerId, chart);
    return chart;
  }
  
  // 创建柱状图
  createBarChart(containerId, data) {
    const ctx = document.getElementById(containerId).getContext('2d');
    const chart = new Chart(ctx, {
      type: 'bar',
      data: {
        labels: data.labels,
        datasets: [{
          label: data.label,
          data: data.values,
          backgroundColor: 'rgba(54, 162, 235, 0.8)',
          borderColor: 'rgba(54, 162, 235, 1)',
          borderWidth: 1
        }]
      },
      options: {
        responsive: true,
        plugins: {
          title: {
            display: true,
            text: data.title
          }
        }
      }
    });
    
    this.charts.set(containerId, chart);
    return chart;
  }
  
  // 更新图表数据
  updateChart(containerId, newData) {
    const chart = this.charts.get(containerId);
    if (chart) {
      chart.data = newData;
      chart.update();
    }
  }
}

class DataSource {
  constructor() {
    this.cache = new Map();
  }
  
  // 获取销售数据
  async getSalesData() {
    if (this.cache.has('sales')) {
      return this.cache.get('sales');
    }
    
    try {
      const response = await fetch('/api/sales');
      const data = await response.json();
      this.cache.set('sales', data);
      return data;
    } catch (error) {
      console.error('获取销售数据失败:', error);
      return this.getMockSalesData();
    }
  }
  
  // 获取用户数据
  async getUserData() {
    if (this.cache.has('users')) {
      return this.cache.get('users');
    }
    
    try {
      const response = await fetch('/api/users');
      const data = await response.json();
      this.cache.set('users', data);
      return data;
    } catch (error) {
      console.error('获取用户数据失败:', error);
      return this.getMockUserData();
    }
  }
  
  // 模拟销售数据
```

```javascript
// src/client/ChatClient.js
class ChatClient {
  constructor() {
    this.ws = null;
    this.user = null;
    this.currentRoom = null;
    this.messages = [];
    this.users = [];
    this.init();
  }
  
  init() {
    this.setupUI();
    this.connect();
  }
  
  setupUI() {
    document.body.innerHTML = `
      <div class="chat-container">
        <div class="chat-sidebar">
          <div class="user-info">
            <input type="text" id="usernameInput" placeholder="输入用户名" />
            <button id="loginBtn">登录</button>
          </div>
          <div class="room-list">
            <h3>房间列表</h3>
            <div id="roomList"></div>
            <button id="createRoomBtn">创建房间</button>
          </div>
          <div class="user-list">
            <h3>在线用户</h3>
            <div id="userList"></div>
          </div>
        </div>
        <div class="chat-main">
          <div class="chat-header">
            <h2 id="roomTitle">选择房间</h2>
          </div>
          <div class="chat-messages" id="messageContainer"></div>
          <div class="chat-input">
            <input type="text" id="messageInput" placeholder="输入消息..." disabled />
            <button id="sendBtn" disabled>发送</button>
          </div>
        </div>
      </div>
    `;
    
    this.bindEvents();
  }
  
  bindEvents() {
    // 登录
    document.getElementById('loginBtn').addEventListener('click', () => {
      this.login();
    });
    
    // 发送消息
    document.getElementById('sendBtn').addEventListener('click', () => {
      this.sendMessage();
    });
    
    // 回车发送
    document.getElementById('messageInput').addEventListener('keypress', (e) => {
      if (e.key === 'Enter') {
        this.sendMessage();
      }
    });
    
    // 创建房间
    document.getElementById('createRoomBtn').addEventListener('click', () => {
      this.createRoom();
    });
  }
  
  connect() {
    this.ws = new WebSocket('ws://localhost:8080');
    
    this.ws.onopen = () => {
      console.log('已连接到服务器');
    };
    
    this.ws.onmessage = (event) => {
      const message = JSON.parse(event.data);
      this.handleMessage(message);
    };
    
    this.ws.onclose = () => {
      console.log('与服务器断开连接');
      setTimeout(() => this.connect(), 3000);
    };
  }
  
  handleMessage(message) {
    switch (message.type) {
      case 'loginSuccess':
        this.user = message.user;
        this.updateUI();
        break;
      case 'roomList':
        this.updateRoomList(message.rooms);
        break;
      case 'userList':
        this.updateUserList(message.users);
        break;
      case 'chat':
        this.addMessage(message);
        break;
      case 'userJoined':
        this.addSystemMessage(`${message.user.name} 加入了房间`);
        break;
      case 'userLeft':
        this.addSystemMessage(`${message.user.name} 离开了房间`);
        break;
    }
  }
  
  login() {
    const username = document.getElementById('usernameInput').value.trim();
    if (username) {
      this.send({
        type: 'login',
        username: username
      });
    }
  }
  
  sendMessage() {
    const input = document.getElementById('messageInput');
    const content = input.value.trim();
    
    if (content && this.currentRoom) {
      this.send({
        type: 'chat',
        roomId: this.currentRoom,
        content: content
      });
      input.value = '';
    }
  }
  
  createRoom() {
    const name = prompt('房间名称:');
    if (name) {
      this.send({
        type: 'createRoom',
        name: name
      });
    }
  }
  
  joinRoom(roomId) {
    this.currentRoom = roomId;
    this.send({
      type: 'joinRoom',
      roomId: roomId
    });
    
    // 更新UI
    const room = this.rooms.find(r => r.id === roomId);
    document.getElementById('roomTitle').textContent = room ? room.name : '未知房间';
    
    // 启用输入
    document.getElementById('messageInput').disabled = false;
    document.getElementById('sendBtn').disabled = false;
  }
  
  send(data) {
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify(data));
    }
  }
  
  addMessage(message) {
    this.messages.push(message);
    this.renderMessage(message);
  }
  
  addSystemMessage(content) {
    const message = {
      type: 'system',
      content: content,
      timestamp: Date.now()
    };
    this.renderMessage(message);
  }
  
  renderMessage(message) {
    const container = document.getElementById('messageContainer');
    const messageElement = document.createElement('div');
    messageElement.className = `message ${message.type}`;
    
    if (message.type === 'system') {
      messageElement.innerHTML = `<div class="system-message">${message.content}</div>`;
    } else {
      messageElement.innerHTML = `
        <div class="message-header">
          <span class="username">${message.user.name}</span>
          <span class="timestamp">${new Date(message.timestamp).toLocaleTimeString()}</span>
        </div>
        <div class="message-content">${message.content}</div>
      `;
    }
    
    container.appendChild(messageElement);
    container.scrollTop = container.scrollHeight;
  }
  
  updateRoomList(rooms) {
    this.rooms = rooms;
    const roomList = document.getElementById('roomList');
    roomList.innerHTML = rooms.map(room => `
      <div class="room-item" data-room-id="${room.id}">
        <div class="room-name">${room.name}</div>
        <div class="room-count">${room.userCount} 人</div>
      </div>
    `).join('');
    
    // 绑定点击事件
    roomList.querySelectorAll('.room-item').forEach(item => {
      item.addEventListener('click', () => {
        const roomId = item.getAttribute('data-room-id');
        this.joinRoom(roomId);
      });
    });
  }
  
  updateUserList(users) {
    this.users = users;
    const userList = document.getElementById('userList');
    userList.innerHTML = users.map(user => `
      <div class="user-item">
        <div class="user-name">${user.name}</div>
        <div class="user-status ${user.status}"></div>
      </div>
    `).join('');
  }
  
  updateUI() {
    document.getElementById('usernameInput').disabled = true;
    document.getElementById('loginBtn').disabled = true;
  }
}
```

### 服务器端代码
```javascript
// src/server/index.js
const WebSocket = require('ws');
const express = require('express');
const { v4: uuidv4 } = require('uuid');

class ChatServer {
  constructor() {
    this.app = express();
    this.server = null;
    this.wss = null;
    this.users = new Map();
    this.rooms = new Map();
    this.init();
  }
  
  init() {
    this.setupExpress();
    this.setupWebSocket();
    this.start();
  }
  
  setupExpress() {
    this.app.use(express.static('public'));
    this.server = this.app.listen(8080, () => {
      console.log('服务器运行在 http://localhost:8080');
    });
  }
  
  setupWebSocket() {
    this.wss = new WebSocket.Server({ server: this.server });
    
    this.wss.on('connection', (ws) => {
      console.log('新用户连接');
      
      ws.on('message', (data) => {
        try {
          const message = JSON.parse(data);
          this.handleMessage(ws, message);
        } catch (error) {
          console.error('解析消息失败:', error);
        }
      });
      
      ws.on('close', () => {
        this.handleDisconnect(ws);
      });
    });
  }
  
  handleMessage(ws, message) {
    switch (message.type) {
      case 'login':
        this.handleLogin(ws, message);
        break;
      case 'joinRoom':
        this.handleJoinRoom(ws, message);
        break;
      case 'leaveRoom':
        this.handleLeaveRoom(ws, message);
        break;
      case 'chat':
        this.handleChat(ws, message);
        break;
      case 'createRoom':
        this.handleCreateRoom(ws, message);
        break;
    }
  }
  
  handleLogin(ws, message) {
    const user = {
      id: uuidv4(),
      name: message.username,
      ws: ws,
      currentRoom: null
    };
    
    this.users.set(ws, user);
    
    ws.send(JSON.stringify({
      type: 'loginSuccess',
      user: user
    }));
    
    // 发送房间列表
    this.sendRoomList(ws);
  }
  
  handleJoinRoom(ws, message) {
    const user = this.users.get(ws);
    if (!user) return;
    
    // 离开当前房间
    if (user.currentRoom) {
      this.leaveRoom(user, user.currentRoom);
    }
    
    // 加入新房间
    const room = this.rooms.get(message.roomId);
    if (room) {
      room.users.add(user);
      user.currentRoom = message.roomId;
      
      // 通知房间内其他用户
      this.broadcastToRoom(message.roomId, {
        type: 'userJoined',
        user: user
      }, ws);
      
      // 发送用户列表
      this.sendUserList(message.roomId);
    }
  }
  
  handleLeaveRoom(ws, message) {
    const user = this.users.get(ws);
    if (user && user.currentRoom) {
      this.leaveRoom(user, user.currentRoom);
    }
  }
  
  handleChat(ws, message) {
    const user = this.users.get(ws);
    if (!user || !user.currentRoom) return;
    
    const chatMessage = {
      type: 'chat',
      user: user,
      content: message.content,
      timestamp: Date.now()
    };
    
    this.broadcastToRoom(user.currentRoom, chatMessage);
  }
  
  handleCreateRoom(ws, message) {
    const room = {
      id: uuidv4(),
      name: message.name,
      users: new Set()
    };
    
    this.rooms.set(room.id, room);
    
    // 通知所有用户
    this.broadcastToAll({
      type: 'roomCreated',
      room: room
    });
  }
  
  handleDisconnect(ws) {
    const user = this.users.get(ws);
    if (user) {
      if (user.currentRoom) {
        this.leaveRoom(user, user.currentRoom);
      }
      this.users.delete(ws);
    }
  }
  
  leaveRoom(user, roomId) {
    const room = this.rooms.get(roomId);
    if (room) {
      room.users.delete(user);
      
      // 通知房间内其他用户
      this.broadcastToRoom(roomId, {
        type: 'userLeft',
        user: user
      });
      
      // 发送用户列表
      this.sendUserList(roomId);
    }
    
    user.currentRoom = null;
  }
  
  broadcastToRoom(roomId, message, excludeWs = null) {
    const room = this.rooms.get(roomId);
    if (room) {
      room.users.forEach(user => {
        if (user.ws !== excludeWs && user.ws.readyState === WebSocket.OPEN) {
          user.ws.send(JSON.stringify(message));
        }
      });
    }
  }
  
  broadcastToAll(message) {
    this.users.forEach(user => {
      if (user.ws.readyState === WebSocket.OPEN) {
        user.ws.send(JSON.stringify(message));
      }
    });
  }
  
  sendRoomList(ws) {
    const rooms = Array.from(this.rooms.values()).map(room => ({
      id: room.id,
      name: room.name,
      userCount: room.users.size
    }));
    
    ws.send(JSON.stringify({
      type: 'roomList',
      rooms: rooms
    }));
  }
  
  sendUserList(roomId) {
    const room = this.rooms.get(roomId);
    if (room) {
      const users = Array.from(room.users).map(user => ({
        id: user.id,
        name: user.name,
        status: 'online'
      }));
      
      this.broadcastToRoom(roomId, {
        type: 'userList',
        users: users
      });
    }
  }
  
  start() {
    console.log('聊天服务器启动完成');
  }
}

// 启动服务器
new ChatServer();
```

```javascript
// src/components/FileUpload.js
class FileUpload {
  constructor(container, options = {}) {
    this.container = container;
    this.options = {
      maxFileSize: 10 * 1024 * 1024, // 10MB
      allowedTypes: ['image/*', 'application/pdf'],
      multiple: true,
      chunkSize: 1024 * 1024, // 1MB
      ...options
    };
    
    this.files = [];
    this.uploadQueue = [];
    this.isUploading = false;
    this.init();
  }
  
  init() {
    this.render();
    this.bindEvents();
  }
  
  render() {
    this.container.innerHTML = `
      <div class="file-upload">
        <div class="upload-area">
          <div class="drop-zone">
            <div class="drop-icon">📁</div>
            <div class="drop-text">
              <h3>拖拽文件到此处</h3>
              <p>或点击选择文件</p>
            </div>
            <input type="file" class="file-input" multiple 
                   accept="${this.options.allowedTypes.join(',')}" />
          </div>
        </div>
        <div class="file-list"></div>
        <div class="upload-controls">
          <button class="upload-all-btn" disabled>上传全部</button>
          <button class="clear-all-btn" disabled>清空</button>
        </div>
      </div>
    `;
  }
  
  bindEvents() {
    const dropZone = this.container.querySelector('.drop-zone');
    const fileInput = this.container.querySelector('.file-input');
    const uploadAllBtn = this.container.querySelector('.upload-all-btn');
    const clearAllBtn = this.container.querySelector('.clear-all-btn');
    
    // 文件选择
    fileInput.addEventListener('change', (e) => {
      this.handleFiles(e.target.files);
    });
    
    // 拖拽事件
    dropZone.addEventListener('dragover', (e) => {
      e.preventDefault();
      dropZone.classList.add('drag-over');
    });
    
    dropZone.addEventListener('dragleave', (e) => {
      e.preventDefault();
      dropZone.classList.remove('drag-over');
    });
    
    dropZone.addEventListener('drop', (e) => {
      e.preventDefault();
      dropZone.classList.remove('drag-over');
      this.handleFiles(e.dataTransfer.files);
    });
    
    // 点击选择文件
    dropZone.addEventListener('click', () => {
      fileInput.click();
    });
    
    // 上传全部
    uploadAllBtn.addEventListener('click', () => {
      this.uploadAll();
    });
    
    // 清空全部
    clearAllBtn.addEventListener('click', () => {
      this.clearAll();
    });
  }
  
  handleFiles(fileList) {
    const files = Array.from(fileList);
    
    files.forEach(file => {
      if (this.validateFile(file)) {
        this.addFile(file);
      }
    });
  }
  
  validateFile(file) {
    // 检查文件大小
    if (file.size > this.options.maxFileSize) {
      this.showError(`文件 ${file.name} 超过最大大小限制`);
      return false;
    }
    
    // 检查文件类型
    const isValidType = this.options.allowedTypes.some(type => {
      if (type.endsWith('/*')) {
        return file.type.startsWith(type.slice(0, -1));
      }
      return file.type === type;
    });
    
    if (!isValidType) {
      this.showError(`文件 ${file.name} 类型不支持`);
      return false;
    }
    
    return true;
  }
  
  addFile(file) {
    const fileItem = {
      id: Date.now() + Math.random(),
      file: file,
      status: 'pending',
      progress: 0,
      error: null
    };
    
    this.files.push(fileItem);
    this.renderFileItem(fileItem);
    this.updateControls();
  }
  
  renderFileItem(fileItem) {
    const fileList = this.container.querySelector('.file-list');
    const fileElement = document.createElement('div');
    fileElement.className = 'file-item';
    fileElement.dataset.fileId = fileItem.id;
    
    fileElement.innerHTML = `
      <div class="file-info">
        <div class="file-name">${fileItem.file.name}</div>
        <div class="file-size">${this.formatFileSize(fileItem.file.size)}</div>
        <div class="file-status">${fileItem.status}</div>
      </div>
      <div class="file-preview">
        ${this.generatePreview(fileItem.file)}
      </div>
      <div class="file-actions">
        <button class="upload-btn" data-action="upload">上传</button>
        <button class="remove-btn" data-action="remove">删除</button>
      </div>
      <div class="file-progress" style="display: none;">
        <div class="progress-bar">
          <div class="progress-fill"></div>
        </div>
        <div class="progress-text">0%</div>
      </div>
    `;
    
    fileList.appendChild(fileElement);
    
    // 绑定事件
    fileElement.querySelector('.upload-btn').addEventListener('click', () => {
      this.uploadFile(fileItem.id);
    });
    
    fileElement.querySelector('.remove-btn').addEventListener('click', () => {
      this.removeFile(fileItem.id);
    });
  }
  
  generatePreview(file) {
    if (file.type.startsWith('image/')) {
      return `<img src="${URL.createObjectURL(file)}" class="preview-image" />`;
    } else if (file.type === 'application/pdf') {
      return `<div class="preview-pdf">PDF文件</div>`;
    } else {
      return `<div class="preview-file">${file.name}</div>`;
    }
  }
  
  async uploadFile(fileId) {
    const fileItem = this.files.find(f => f.id === fileId);
    if (!fileItem || fileItem.status === 'uploading') {
      return;
    }
    
    fileItem.status = 'uploading';
    this.updateFileStatus(fileId, 'uploading');
    
    try {
      if (fileItem.file.size > this.options.chunkSize) {
        await this.uploadChunked(fileItem);
      } else {
        await this.uploadSimple(fileItem);
      }
      
      fileItem.status = 'completed';
      this.updateFileStatus(fileId, 'completed');
      
    } catch (error) {
      fileItem.status = 'error';
      fileItem.error = error.message;
      this.updateFileStatus(fileId, 'error');
      this.showError(`上传失败: ${error.message}`);
    }
  }
  
  async uploadSimple(fileItem) {
    const formData = new FormData();
    formData.append('file', fileItem.file);
    
    const response = await fetch('/api/upload', {
      method: 'POST',
      body: formData
    });
    
    if (!response.ok) {
      throw new Error(`上传失败: ${response.status}`);
    }
    
    return await response.json();
  }
  
  async uploadChunked(fileItem) {
    const chunks = this.createChunks(fileItem.file);
    const uploadedChunks = [];
    
    for (let i = 0; i < chunks.length; i++) {
      const chunk = chunks[i];
      
      const formData = new FormData();
      formData.append('file', chunk.blob);
      formData.append('chunkIndex', chunk.index);
      formData.append('totalChunks', chunks.length);
      formData.append('fileName', fileItem.file.name);
      
      const response = await fetch('/api/upload-chunk', {
        method: 'POST',
        body: formData
      });
      
      if (!response.ok) {
        throw new Error(`分片上传失败: ${response.status}`);
      }
      
      uploadedChunks.push(chunk.index);
      
      // 更新进度
      const progress = ((i + 1) / chunks.length) * 100;
      fileItem.progress = progress;
      this.updateFileProgress(fileItem.id, progress);
    }
    
    // 合并分片
    const mergeResponse = await fetch('/api/merge-chunks', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        fileName: fileItem.file.name,
        totalChunks: chunks.length
      })
    });
    
    if (!mergeResponse.ok) {
      throw new Error(`合并分片失败: ${mergeResponse.status}`);
    }
    
    return await mergeResponse.json();
  }
  
  createChunks(file) {
    const chunks = [];
    const chunkSize = this.options.chunkSize;
    
    for (let i = 0; i < file.size; i += chunkSize) {
      const end = Math.min(i + chunkSize, file.size);
      chunks.push({
        index: Math.floor(i / chunkSize),
        blob: file.slice(i, end)
      });
    }
    
    return chunks;
  }
  
  updateFileStatus(fileId, status) {
    const fileElement = document.querySelector(`[data-file-id="${fileId}"]`);
    if (fileElement) {
      fileElement.className = `file-item ${status}`;
      fileElement.querySelector('.file-status').textContent = status;
      
      const progressElement = fileElement.querySelector('.file-progress');
      if (status === 'uploading') {
        progressElement.style.display = 'block';
      } else {
        progressElement.style.display = 'none';
      }
    }
  }
  
  updateFileProgress(fileId, progress) {
    const fileElement = document.querySelector(`[data-file-id="${fileId}"]`);
    if (fileElement) {
      const progressFill = fileElement.querySelector('.progress-fill');
      const progressText = fileElement.querySelector('.progress-text');
      
      progressFill.style.width = `${progress}%`;
      progressText.textContent = `${Math.round(progress)}%`;
    }
  }
  
  removeFile(fileId) {
    const index = this.files.findIndex(f => f.id === fileId);
    if (index > -1) {
      this.files.splice(index, 1);
      const fileElement = document.querySelector(`[data-file-id="${fileId}"]`);
      if (fileElement) {
        fileElement.remove();
      }
      this.updateControls();
    }
  }
  
  async uploadAll() {
    const pendingFiles = this.files.filter(f => f.status === 'pending');
    
    for (const fileItem of pendingFiles) {
      await this.uploadFile(fileItem.id);
    }
  }
  
  clearAll() {
    this.files = [];
    this.container.querySelector('.file-list').innerHTML = '';
    this.updateControls();
  }
  
  updateControls() {
    const uploadAllBtn = this.container.querySelector('.upload-all-btn');
    const clearAllBtn = this.container.querySelector('.clear-all-btn');
    
    const hasFiles = this.files.length > 0;
    const hasPendingFiles = this.files.some(f => f.status === 'pending');
    
    uploadAllBtn.disabled = !hasPendingFiles;
    clearAllBtn.disabled = !hasFiles;
  }
  
  formatFileSize(bytes) {
    if (bytes === 0) return '0 Bytes';
    const k = 1024;
    const sizes = ['Bytes', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
  }
  
  showError(message) {
    // 简单的错误提示
    alert(message);
  }
}
```
