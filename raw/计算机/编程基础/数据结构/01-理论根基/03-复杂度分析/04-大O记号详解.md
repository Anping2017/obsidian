# 大O记号详解

## 📖 核心概念

**大O记号（Big O Notation）**是算法复杂度分析的标准表示法，用于描述算法在最坏情况下的性能上界。它是计算机科学中最重要的数学工具之一。

```mermaid
    D --> D1[上界分析]
    D --> D2[渐进行为]
    D --> D3[忽略常数因子]
```

## 📊 常见复杂度对比

### 时间复杂度对比表

| 复杂度 | 名称 | 示例算法 | 性能特征 | 适用场景 |
|--------|------|----------|----------|----------|
| **O(1)** | 常数时间 | 数组访问、哈希表查找 | 最优性能 | 频繁访问 |
| **O(log n)** | 对数时间 | 二分搜索、平衡树操作 | 高效搜索 | 有序数据 |
| **O(n)** | 线性时间 | 线性搜索、遍历 | 基础性能 | 小规模数据 |
| **O(n log n)** | 线性对数时间 | 归并排序、堆排序 | 排序最优 | 中等规模 |
| **O(n²)** | 平方时间 | 冒泡排序、选择排序 | 性能较差 | 小规模数据 |
| **O(2ⁿ)** | 指数时间 | 递归斐波那契 | 性能极差 | 避免使用 |

### 性能增长对比

```cpp
// 不同复杂度的性能对比
class ComplexityComparison {
public:
    // O(1) - 常数时间
    int constantTime(int n) {
        return n * 2;  // 无论n多大，执行时间相同
    }
    
    // O(log n) - 对数时间
    int logarithmicTime(int n) {
        int count = 0;
        while (n > 1) {
            n /= 2;
            count++;
        }
        return count;
    }
    
    // O(n) - 线性时间
    int linearTime(int n) {
        int sum = 0;
        for (int i = 0; i < n; ++i) {
            sum += i;
        }
        return sum;
    }
    
    // O(n log n) - 线性对数时间
    void mergeSort(vector<int>& arr, int left, int right) {
        if (left < right) {
            int mid = left + (right - left) / 2;
            mergeSort(arr, left, mid);
            mergeSort(arr, mid + 1, right);
            merge(arr, left, mid, right);
        }
    }
    
    // O(n²) - 平方时间
    void bubbleSort(vector<int>& arr) {
        int n = arr.size();
        for (int i = 0; i < n - 1; ++i) {
            for (int j = 0; j < n - i - 1; ++j) {
                if (arr[j] > arr[j + 1]) {
                    swap(arr[j], arr[j + 1]);
                }
            }
        }
    }
};
```

## 🔍 大O记号定义

### 数学定义

```
f(n) = O(g(n)) 当且仅当存在正常数 c 和 n₀，
使得对于所有 n ≥ n₀，都有 f(n) ≤ c·g(n)
```

### 实际应用

```cpp
// 大O记号的实际应用
class BigOExamples {
public:
    // O(1) - 常数时间
    int getFirstElement(const vector<int>& arr) {
        return arr[0];  // 无论数组多大，都是常数时间
    }
    
    // O(log n) - 对数时间
    int binarySearch(const vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (arr[mid] == target) return mid;
            if (arr[mid] < target) left = mid + 1;
            else right = mid - 1;
        }
        return -1;
    }
    
    // O(n) - 线性时间
    int findMax(const vector<int>& arr) {
        int maxVal = arr[0];
        for (int i = 1; i < arr.size(); ++i) {
            if (arr[i] > maxVal) {
                maxVal = arr[i];
            }
        }
        return maxVal;
    }
    
    // O(n log n) - 线性对数时间
    void heapSort(vector<int>& arr) {
        // 建堆 O(n)
        for (int i = arr.size() / 2 - 1; i >= 0; --i) {
            heapify(arr, arr.size(), i);
        }
        
        // 排序 O(n log n)
        for (int i = arr.size() - 1; i > 0; --i) {
            swap(arr[0], arr[i]);
            heapify(arr, i, 0);
        }
    }
    
    // O(n²) - 平方时间
    void selectionSort(vector<int>& arr) {
        int n = arr.size();
        for (int i = 0; i < n - 1; ++i) {
            int minIdx = i;
            for (int j = i + 1; j < n; ++j) {
                if (arr[j] < arr[minIdx]) {
                    minIdx = j;
                }
            }
            swap(arr[i], arr[minIdx]);
        }
    }
    
private:
    void heapify(vector<int>& arr, int n, int i) {
        int largest = i;
        int left = 2 * i + 1;
        int right = 2 * i + 2;
        
        if (left < n && arr[left] > arr[largest]) {
            largest = left;
        }
        
        if (right < n && arr[right] > arr[largest]) {
            largest = right;
        }
        
        if (largest != i) {
            swap(arr[i], arr[largest]);
            heapify(arr, n, largest);
        }
    }
};
```

## 📈 复杂度分析技巧

### 1. 循环分析

```cpp
// 单层循环 - O(n)
void singleLoop(int n) {
    for (int i = 0; i < n; ++i) {
        // 常数时间操作
        process(i);
    }
}

// 嵌套循环 - O(n²)
void nestedLoop(int n) {
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            // 常数时间操作
            process(i, j);
        }
    }
}

// 循环步长 - O(log n)
void logarithmicLoop(int n) {
    for (int i = 1; i < n; i *= 2) {
        // 常数时间操作
        process(i);
    }
}
```

### 2. 递归分析

```cpp
// 线性递归 - O(n)
int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

// 二分递归 - O(log n)
int binarySearchRecursive(const vector<int>& arr, int target, int left, int right) {
    if (left > right) return -1;
    
    int mid = left + (right - left) / 2;
    if (arr[mid] == target) return mid;
    
    if (arr[mid] > target) {
        return binarySearchRecursive(arr, target, left, mid - 1);
    } else {
        return binarySearchRecursive(arr, target, mid + 1, right);
    }
}

// 树形递归 - O(2ⁿ)
int fibonacci(int n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}
```

### 3. 主定理应用

```cpp
// 分治算法复杂度分析
class MasterTheorem {
public:
    // T(n) = aT(n/b) + f(n)
    // 其中 a ≥ 1, b > 1
    
    // 情况1: f(n) = O(n^c), c < log_b(a)
    // 则 T(n) = Θ(n^(log_b(a)))
    void case1() {
        // 例如: T(n) = 2T(n/2) + n
        // a=2, b=2, f(n)=n, c=1
        // log_2(2) = 1, c = 1 = log_b(a)
        // 属于情况2: T(n) = Θ(n log n)
    }
    
    // 情况2: f(n) = Θ(n^c), c = log_b(a)
    // 则 T(n) = Θ(n^c log n)
    void case2() {
        // 例如: 归并排序
        // T(n) = 2T(n/2) + n
        // 结果: O(n log n)
    }
    
    // 情况3: f(n) = Ω(n^c), c > log_b(a)
    // 则 T(n) = Θ(f(n))
    void case3() {
        // 例如: T(n) = T(n/2) + n²
        // a=1, b=2, f(n)=n², c=2
        // log_2(1) = 0, c = 2 > log_b(a)
        // 结果: O(n²)
    }
};
```

### 算法选择决策

```cpp
class AlgorithmSelector {
public:
    string selectAlgorithm(size_t dataSize, const string& operation) {
        if (operation == "search") {
            if (dataSize < 100) {
                return "线性搜索 O(n)";
            } else {
                return "二分搜索 O(log n)";
            }
        } else if (operation == "sort") {
            if (dataSize < 50) {
                return "插入排序 O(n²)";
            } else if (dataSize < 1000) {
                return "快速排序 O(n log n)";
            } else {
                return "归并排序 O(n log n)";
            }
        } else if (operation == "access") {
            return "数组访问 O(1)";
        }
        return "未知操作";
    }
    
    // 性能预测
    double predictExecutionTime(const string& complexity, size_t dataSize) {
        if (complexity == "O(1)") return 1.0;
        if (complexity == "O(log n)") return log2(dataSize);
        if (complexity == "O(n)") return dataSize;
        if (complexity == "O(n log n)") return dataSize * log2(dataSize);
        if (complexity == "O(n²)") return dataSize * dataSize;
        return 0.0;
    }
};
```

### 性能优化策略

```cpp
class PerformanceOptimizer {
public:
    // 从O(n²)优化到O(n log n)
    void optimizeQuadraticToLogLinear(vector<int>& data) {
        // 原来: 嵌套循环 O(n²)
        // for (int i = 0; i < n; ++i) {
        //     for (int j = i + 1; j < n; ++j) {
        //         if (data[i] > data[j]) swap(data[i], data[j]);
        //     }
        // }
        
        // 优化后: 排序 O(n log n)
        sort(data.begin(), data.end());
    }
    
    // 从O(n)优化到O(log n)
    int optimizeLinearToLogarithmic(const vector<int>& sortedData, int target) {
        // 原来: 线性搜索 O(n)
        // for (int i = 0; i < sortedData.size(); ++i) {
        //     if (sortedData[i] == target) return i;
        // }
        
        // 优化后: 二分搜索 O(log n)
        return binarySearch(sortedData, target);
    }
    
private:
    int binarySearch(const vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (arr[mid] == target) return mid;
            if (arr[mid] < target) left = mid + 1;
            else right = mid - 1;
        }
        return -1;
    }
};
```

## 💡 大O记号要点

1. **上界表示**：描述算法性能的上限
2. **忽略常数**：关注增长趋势而非具体数值
3. **最坏情况**：通常分析最坏情况下的性能
4. **实际应用**：用于算法选择和性能预测
