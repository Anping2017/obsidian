# 拓扑排序与关键路径

## 📖 核心概念

**拓扑排序**和**关键路径**是图论中的重要概念。拓扑排序用于有向无环图（DAG）的线性排序，关键路径用于项目管理中的时间分析。

### 🏗️ 拓扑排序与关键路径分类


## 🔧 拓扑排序与关键路径实现

### 拓扑排序

```cpp
class TopologicalSort {
private:
    vector<vector<int>> graph;
    vector<int> inDegree;
    vector<bool> visited;
    vector<int> finishTime;
    int vertices;
    int time;
    
public:
    TopologicalSort(int v) : vertices(v), time(0) {
        graph.resize(vertices);
        inDegree.resize(vertices, 0);
        visited.resize(vertices, false);
        finishTime.resize(vertices, 0);
    }
    
    void addEdge(int source, int destination) {
        graph[source].push_back(destination);
        inDegree[destination]++;
    }
    
    // Kahn算法
    vector<int> kahnTopologicalSort() {
        vector<int> result;
        queue<int> q;
        
        // 找到所有入度为0的顶点
        for (int i = 0; i < vertices; ++i) {
            if (inDegree[i] == 0) {
                q.push(i);
            }
        }
        
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            result.push_back(u);
            
            for (int v : graph[u]) {
                inDegree[v]--;
                if (inDegree[v] == 0) {
                    q.push(v);
                }
            }
        }
        
        return result;
    }
    
    // DFS拓扑排序
    vector<int> dfsTopologicalSort() {
        resetVisited();
        vector<int> result;
        
        for (int i = 0; i < vertices; ++i) {
            if (!visited[i]) {
                dfsTopological(i, result);
            }
        }
        
        reverse(result.begin(), result.end());
        return result;
    }
    
    void dfsTopological(int vertex, vector<int>& result) {
        visited[vertex] = true;
        
        for (int neighbor : graph[vertex]) {
            if (!visited[neighbor]) {
                dfsTopological(neighbor, result);
            }
        }
        
        result.push_back(vertex);
    }
    
    // 检测环
    bool hasCycle() {
        vector<int> result = kahnTopologicalSort();
        return result.size() != vertices;
    }
    
    // 显示拓扑排序过程
    void displayTopologicalSortProcess() {
        cout << "Topological Sort Process:" << endl;
        cout << "========================" << endl;
        
        vector<int> result = kahnTopologicalSort();
        
        if (result.size() != vertices) {
            cout << "Graph has cycle, cannot perform topological sort" << endl;
            return;
        }
        
        cout << "Topological sort: ";
        for (int vertex : result) {
            cout << vertex << " ";
        }
        cout << endl;
    }
    
    void resetVisited() {
        fill(visited.begin(), visited.end(), false);
        time = 0;
    }
    
    void displayTopologicalSort() {
        cout << "Topological Sort:" << endl;
        cout << "================" << endl;
        
        if (hasCycle()) {
            cout << "Graph has cycle, cannot perform topological sort" << endl;
            return;
        }
        
        vector<int> kahnResult = kahnTopologicalSort();
        vector<int> dfsResult = dfsTopologicalSort();
        
        cout << "Kahn algorithm: ";
        for (int vertex : kahnResult) {
            cout << vertex << " ";
        }
        cout << endl;
        
        cout << "DFS algorithm: ";
        for (int vertex : dfsResult) {
            cout << vertex << " ";
        }
        cout << endl;
    }
};
```

### 关键路径

```cpp
class CriticalPath {
private:
    struct Activity {
        int id;
        int duration;
        int earliestStart;
        int latestStart;
        int earliestFinish;
        int latestFinish;
        int slack;
        
        Activity(int i, int d) : id(i), duration(d), earliestStart(0), 
                                latestStart(0), earliestFinish(0), 
                                latestFinish(0), slack(0) {}
    };
    
    vector<vector<int>> graph;
    vector<Activity> activities;
    vector<int> inDegree;
    vector<int> outDegree;
    int vertices;
    
public:
    CriticalPath(int v) : vertices(v) {
        graph.resize(vertices);
        inDegree.resize(vertices, 0);
        outDegree.resize(vertices, 0);
    }
    
    void addActivity(int id, int duration) {
        activities.push_back(Activity(id, duration));
    }
    
    void addDependency(int source, int destination) {
        graph[source].push_back(destination);
        inDegree[destination]++;
        outDegree[source]++;
    }
    
    // 计算最早开始时间
    void calculateEarliestTimes() {
        queue<int> q;
        vector<int> earliestStart(vertices, 0);
        
        // 找到所有入度为0的顶点
        for (int i = 0; i < vertices; ++i) {
            if (inDegree[i] == 0) {
                q.push(i);
            }
        }
        
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            
            activities[u].earliestStart = earliestStart[u];
            activities[u].earliestFinish = earliestStart[u] + activities[u].duration;
            
            for (int v : graph[u]) {
                earliestStart[v] = max(earliestStart[v], activities[u].earliestFinish);
                inDegree[v]--;
                
                if (inDegree[v] == 0) {
                    q.push(v);
                }
            }
        }
    }
    
    // 计算最晚开始时间
    void calculateLatestTimes() {
        int projectDuration = 0;
        for (const Activity& activity : activities) {
            projectDuration = max(projectDuration, activity.earliestFinish);
        }
        
        vector<int> latestFinish(vertices, projectDuration);
        
        // 找到所有出度为0的顶点
        queue<int> q;
        for (int i = 0; i < vertices; ++i) {
            if (outDegree[i] == 0) {
                q.push(i);
            }
        }
        
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            
            activities[u].latestFinish = latestFinish[u];
            activities[u].latestStart = latestFinish[u] - activities[u].duration;
            
            for (int v = 0; v < vertices; ++v) {
                for (int neighbor : graph[v]) {
                    if (neighbor == u) {
                        latestFinish[v] = min(latestFinish[v], activities[u].latestStart);
                        outDegree[v]--;
                        
                        if (outDegree[v] == 0) {
                            q.push(v);
                        }
                    }
                }
            }
        }
    }
    
    // 计算松弛时间
    void calculateSlack() {
        for (Activity& activity : activities) {
            activity.slack = activity.latestStart - activity.earliestStart;
        }
    }
    
    // 找到关键路径
    vector<int> findCriticalPath() {
        calculateEarliestTimes();
        calculateLatestTimes();
        calculateSlack();
        
        vector<int> criticalPath;
        for (int i = 0; i < vertices; ++i) {
            if (activities[i].slack == 0) {
                criticalPath.push_back(i);
            }
        }
        
        return criticalPath;
    }
    
    // 显示关键路径分析
    void displayCriticalPathAnalysis() {
        cout << "Critical Path Analysis:" << endl;
        cout << "=====================" << endl;
        
        calculateEarliestTimes();
        calculateLatestTimes();
        calculateSlack();
        
        cout << "Activity\tDuration\tES\tEF\tLS\tLF\tSlack" << endl;
        cout << "--------\t--------\t--\t--\t--\t--\t-----" << endl;
        
        for (const Activity& activity : activities) {
            cout << activity.id << "\t\t" << activity.duration << "\t\t" 
                 << activity.earliestStart << "\t" << activity.earliestFinish << "\t"
                 << activity.latestStart << "\t" << activity.latestFinish << "\t"
                 << activity.slack << endl;
        }
        
        vector<int> criticalPath = findCriticalPath();
        cout << "\nCritical Path: ";
        for (int activity : criticalPath) {
            cout << activity << " ";
        }
        cout << endl;
        
        int projectDuration = 0;
        for (const Activity& activity : activities) {
            projectDuration = max(projectDuration, activity.earliestFinish);
        }
        cout << "Project Duration: " << projectDuration << endl;
    }
    
    // 显示关键路径过程
    void displayCriticalPathProcess() {
        cout << "Critical Path Process:" << endl;
        cout << "=====================" << endl;
        
        cout << "Step 1: Calculate earliest times" << endl;
        calculateEarliestTimes();
        
        cout << "Step 2: Calculate latest times" << endl;
        calculateLatestTimes();
        
        cout << "Step 3: Calculate slack" << endl;
        calculateSlack();
        
        cout << "Step 4: Find critical path" << endl;
        vector<int> criticalPath = findCriticalPath();
        
        cout << "Critical activities: ";
        for (int activity : criticalPath) {
            cout << activity << " ";
        }
        cout << endl;
    }
};
```

### 项目管理应用

```cpp
class ProjectManagement {
private:
    TopologicalSort topologicalSort;
    CriticalPath criticalPath;
    int vertices;
    
public:
    ProjectManagement(int v) : vertices(v), topologicalSort(v), criticalPath(v) {}
    
    void addTask(int id, int duration) {
        criticalPath.addActivity(id, duration);
    }
    
    void addDependency(int source, int destination) {
        topologicalSort.addEdge(source, destination);
        criticalPath.addDependency(source, destination);
    }
    
    // 任务调度
    void scheduleTasks() {
        cout << "Task Scheduling:" << endl;
        cout << "===============" << endl;
        
        if (topologicalSort.hasCycle()) {
            cout << "Circular dependencies detected! Cannot schedule tasks." << endl;
            return;
        }
        
        topologicalSort.displayTopologicalSort();
    }
    
    // 关键路径分析
    void analyzeCriticalPath() {
        cout << "Critical Path Analysis:" << endl;
        cout << "=====================" << endl;
        
        criticalPath.displayCriticalPathAnalysis();
    }
    
    // 项目管理
    void manageProject() {
        cout << "Project Management:" << endl;
        cout << "=================" << endl;
        
        scheduleTasks();
        analyzeCriticalPath();
    }
    
    // 显示项目状态
    void displayProjectStatus() {
        cout << "Project Status:" << endl;
        cout << "=============" << endl;
        
        if (topologicalSort.hasCycle()) {
            cout << "Project has circular dependencies" << endl;
        } else {
            cout << "Project can be scheduled" << endl;
            criticalPath.displayCriticalPathAnalysis();
        }
    }
};
```
