# 最小生成树算法

## 📖 核心概念

**最小生成树算法**是图论中的重要算法，用于找到连接所有顶点的最小权重边集合。主要包括Kruskal算法和Prim算法。

### 🏗️ 最小生成树算法分类

```mermaid

```

### Kruskal算法

```cpp
class KruskalAlgorithm {
private:
    struct Edge {
        int source;
        int destination;
        int weight;
        
        Edge(int s, int d, int w) : source(s), destination(d), weight(w) {}
    };
    
    struct UnionFind {
        vector<int> parent;
        vector<int> rank;
        
        UnionFind(int n) : parent(n), rank(n, 0) {
            for (int i = 0; i < n; ++i) {
                parent[i] = i;
            }
        }
        
        int find(int x) {
            if (parent[x] != x) {
                parent[x] = find(parent[x]);
            }
            return parent[x];
        }
        
        void unite(int x, int y) {
            int px = find(x);
            int py = find(y);
            
            if (px == py) return;
            
            if (rank[px] < rank[py]) {
                parent[px] = py;
            } else if (rank[px] > rank[py]) {
                parent[py] = px;
            } else {
                parent[py] = px;
                rank[px]++;
            }
        }
    };
    
    vector<Edge> edges;
    int vertices;
    
public:
    KruskalAlgorithm(int v) : vertices(v) {}
    
    void addEdge(int source, int destination, int weight) {
        edges.push_back(Edge(source, destination, weight));
    }
    
    vector<Edge> kruskalMST() {
        // 按权重排序
        sort(edges.begin(), edges.end(), [](const Edge& a, const Edge& b) {
            return a.weight < b.weight;
        });
        
        UnionFind uf(vertices);
        vector<Edge> mst;
        
        for (const Edge& edge : edges) {
            if (uf.find(edge.source) != uf.find(edge.destination)) {
                mst.push_back(edge);
                uf.unite(edge.source, edge.destination);
            }
        }
        
        return mst;
    }
    
    void displayMST() {
        vector<Edge> mst = kruskalMST();
        
        cout << "Minimum Spanning Tree (Kruskal):" << endl;
        cout << "=================================" << endl;
        
        int totalWeight = 0;
        for (const Edge& edge : mst) {
            cout << edge.source << " - " << edge.destination << " : " << edge.weight << endl;
            totalWeight += edge.weight;
        }
        
        cout << "Total weight: " << totalWeight << endl;
    }
    
    void displayKruskalProcess() {
        cout << "Kruskal Algorithm Process:" << endl;
        cout << "========================" << endl;
        
        // 按权重排序
        sort(edges.begin(), edges.end(), [](const Edge& a, const Edge& b) {
            return a.weight < b.weight;
        });
        
        cout << "Edges sorted by weight:" << endl;
        for (const Edge& edge : edges) {
            cout << edge.source << " - " << edge.destination << " : " << edge.weight << endl;
        }
        cout << endl;
        
        UnionFind uf(vertices);
        vector<Edge> mst;
        
        cout << "Building MST:" << endl;
        for (const Edge& edge : edges) {
            if (uf.find(edge.source) != uf.find(edge.destination)) {
                mst.push_back(edge);
                uf.unite(edge.source, edge.destination);
                
                cout << "Added edge: " << edge.source << " - " << edge.destination 
                     << " : " << edge.weight << endl;
            } else {
                cout << "Skipped edge: " << edge.source << " - " << edge.destination 
                     << " : " << edge.weight << " (creates cycle)" << endl;
            }
        }
        
        cout << endl << "Final MST:" << endl;
        int totalWeight = 0;
        for (const Edge& edge : mst) {
            cout << edge.source << " - " << edge.destination << " : " << edge.weight << endl;
            totalWeight += edge.weight;
        }
        cout << "Total weight: " << totalWeight << endl;
    }
};
```

### Prim算法

```cpp
class PrimAlgorithm {
private:
    struct Edge {
        int destination;
        int weight;
        
        Edge(int dest, int w) : destination(dest), weight(w) {}
    };
    
    struct Vertex {
        int id;
        int key;
        
        Vertex(int i, int k) : id(i), key(k) {}
    };
    
    vector<vector<Edge>> graph;
    vector<int> key;
    vector<bool> inMST;
    vector<int> parent;
    int vertices;
    
public:
    PrimAlgorithm(int v) : vertices(v) {
        graph.resize(vertices);
        key.resize(vertices, INT_MAX);
        inMST.resize(vertices, false);
        parent.resize(vertices, -1);
    }
    
    void addEdge(int source, int destination, int weight) {
        graph[source].push_back(Edge(destination, weight));
        graph[destination].push_back(Edge(source, weight)); // 无向图
    }
    
    vector<Edge> primMST() {
        key[0] = 0;
        
        priority_queue<Vertex, vector<Vertex>, function<bool(const Vertex&, const Vertex&)>> pq(
            [](const Vertex& a, const Vertex& b) {
                return a.key > b.key; // 最小堆
            }
        );
        
        pq.push(Vertex(0, 0));
        
        vector<Edge> mst;
        
        while (!pq.empty()) {
            Vertex current = pq.top();
            pq.pop();
            
            int u = current.id;
            
            if (inMST[u]) {
                continue;
            }
            
            inMST[u] = true;
            
            if (parent[u] != -1) {
                mst.push_back(Edge(parent[u], key[u]));
            }
            
            for (const Edge& edge : graph[u]) {
                int v = edge.destination;
                int weight = edge.weight;
                
                if (!inMST[v] && weight < key[v]) {
                    key[v] = weight;
                    parent[v] = u;
                    pq.push(Vertex(v, key[v]));
                }
            }
        }
        
        return mst;
    }
    
    void displayMST() {
        vector<Edge> mst = primMST();
        
        cout << "Minimum Spanning Tree (Prim):" << endl;
        cout << "=============================" << endl;
        
        int totalWeight = 0;
        for (const Edge& edge : mst) {
            cout << parent[edge.destination] << " - " << edge.destination << " : " << edge.weight << endl;
            totalWeight += edge.weight;
        }
        
        cout << "Total weight: " << totalWeight << endl;
    }
    
    void displayPrimProcess() {
        cout << "Prim Algorithm Process:" << endl;
        cout << "=====================" << endl;
        
        key[0] = 0;
        
        priority_queue<Vertex, vector<Vertex>, function<bool(const Vertex&, const Vertex&)>> pq(
            [](const Vertex& a, const Vertex& b) {
                return a.key > b.key;
            }
        );
        
        pq.push(Vertex(0, 0));
        
        vector<Edge> mst;
        int step = 0;
        
        while (!pq.empty()) {
            step++;
            Vertex current = pq.top();
            pq.pop();
            
            int u = current.id;
            
            if (inMST[u]) {
                continue;
            }
            
            inMST[u] = true;
            
            if (parent[u] != -1) {
                mst.push_back(Edge(parent[u], key[u]));
                cout << "Step " << step << ": Added edge " << parent[u] 
                     << " - " << u << " : " << key[u] << endl;
            } else {
                cout << "Step " << step << ": Starting with vertex " << u << endl;
            }
            
            for (const Edge& edge : graph[u]) {
                int v = edge.destination;
                int weight = edge.weight;
                
                if (!inMST[v] && weight < key[v]) {
                    key[v] = weight;
                    parent[v] = u;
                    pq.push(Vertex(v, key[v]));
                    
                    cout << "  Updated key[" << v << "] to " << key[v] << endl;
                }
            }
        }
        
        cout << endl << "Final MST:" << endl;
        int totalWeight = 0;
        for (const Edge& edge : mst) {
            cout << parent[edge.destination] << " - " << edge.destination << " : " << edge.weight << endl;
            totalWeight += edge.weight;
        }
        cout << "Total weight: " << totalWeight << endl;
    }
};
```

```cpp
class MSTAnalysis {
public:
    static void analyzePerformance() {
        cout << "Minimum Spanning Tree Algorithm Performance Analysis:" << endl;
        cout << "=================================================" << endl;
        
        cout << "1. Kruskal Algorithm:" << endl;
        cout << "   - Time Complexity: O(E log E)" << endl;
        cout << "   - Space Complexity: O(V + E)" << endl;
        cout << "   - Best for: Sparse graphs" << endl;
        cout << "   - Advantages: Simple, works with disconnected components" << endl;
        
        cout << "2. Prim Algorithm:" << endl;
        cout << "   - Time Complexity: O(E log V)" << endl;
        cout << "   - Space Complexity: O(V + E)" << endl;
        cout << "   - Best for: Dense graphs" << endl;
        cout << "   - Advantages: Faster for dense graphs" << endl;
        
        cout << "3. Performance Comparison:" << endl;
        cout << "   - Sparse graphs: Kruskal is better" << endl;
        cout << "   - Dense graphs: Prim is better" << endl;
        cout << "   - Both are optimal for their respective cases" << endl;
    }
    
    static void analyzeSpaceComplexity() {
        cout << "Minimum Spanning Tree Algorithm Space Complexity Analysis:" << endl;
        cout << "=======================================================" << endl;
        
        cout << "1. Kruskal Algorithm:" << endl;
        cout << "   - Edge list: O(E)" << endl;
        cout << "   - Union-Find: O(V)" << endl;
        cout << "   - Total: O(V + E)" << endl;
        
        cout << "2. Prim Algorithm:" << endl;
        cout << "   - Adjacency list: O(V + E)" << endl;
        cout << "   - Priority queue: O(V)" << endl;
        cout << "   - Total: O(V + E)" << endl;
    }
    
    static void selectAlgorithm(int vertices, int edges) {
        cout << "Algorithm Selection:" << endl;
        cout << "===================" << endl;
        
        cout << "Vertices: " << vertices << endl;
        cout << "Edges: " << edges << endl;
        
        double density = (double)edges / (vertices * (vertices - 1) / 2);
        cout << "Graph density: " << density << endl;
        
        cout << "Recommendation:" << endl;
        
        if (density < 0.5) {
            cout << "Use Kruskal Algorithm (sparse graph)" << endl;
        } else {
            cout << "Use Prim Algorithm (dense graph)" << endl;
        }
    }
};
```

## 💡 最小生成树算法要点

1. **Kruskal算法**: 边排序，并查集，O(E log E)
2. **Prim算法**: 顶点选择，优先队列，O(E log V)
3. **应用场景**: 网络设计、电路设计、聚类分析
4. **选择原则**: 稀疏图用Kruskal，稠密图用Prim
