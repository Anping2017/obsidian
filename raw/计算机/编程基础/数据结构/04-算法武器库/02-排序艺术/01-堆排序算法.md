# 堆排序算法

## 📖 核心概念

**堆排序（Heap Sort）**是一种基于堆数据结构的比较排序算法。它利用堆的性质（父节点总是大于等于或小于等于子节点）来实现排序。堆排序是一种原地排序算法，具有O(n log n)的时间复杂度。

### 🏗️ 堆排序的组成要素

```mermaid
graph TD
    A[堆排序Heap Sort] --> B[堆结构]
    A --> C[建堆过程]
    A --> D[排序过程]
    A --> E[堆维护]
    
    B --> B1[最大堆]
    B --> B2[最小堆]
    
    C --> C1[自底向上建堆]
    C --> C2[O(n)时间复杂度]
    
    D --> D1[提取最大值]
    D --> D2[堆化调整]
    
    E --> E1[下沉操作]
    E --> E2[堆序性质维护]
```

## 🔍 堆排序分类

### 按堆类型分类

| 堆类型 | 排序结果 | 特点 | 应用场景 |
|--------|----------|------|----------|
| **最大堆** | 升序排列 | 根节点最大 | 一般排序 |
| **最小堆** | 降序排列 | 根节点最小 | 特殊需求 |

### 按实现方式分类

| 实现方式 | 优点 | 缺点 | 适用场景 |
|----------|------|------|----------|
| **原地排序** | 空间效率高 | 不稳定 | 内存受限 |
| **辅助空间** | 稳定排序 | 需要额外空间 | 稳定性要求 |

## 💻 堆排序基础实现

### 最大堆排序（升序）

```cpp
template<typename T>
class HeapSort {
private:
    vector<T>& arr;
    int size;
    
    // 获取父节点索引
    int parent(int i) const {
        return (i - 1) / 2;
    }
    
    // 获取左子节点索引
    int leftChild(int i) const {
        return 2 * i + 1;
    }
    
    // 获取右子节点索引
    int rightChild(int i) const {
        return 2 * i + 2;
    }
    
    // 交换两个元素
    void swap(int i, int j) {
        T temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
    
    // 下沉操作（维护最大堆性质）
    void heapifyDown(int index, int heapSize) {
        int largest = index;
        int left = leftChild(index);
        int right = rightChild(index);
        
        // 找到最大的子节点
        if (left < heapSize && arr[left] > arr[largest]) {
            largest = left;
        }
        if (right < heapSize && arr[right] > arr[largest]) {
            largest = right;
        }
        
        // 如果最大子节点大于当前节点，则交换
        if (largest != index) {
            swap(index, largest);
            heapifyDown(largest, heapSize); // 递归下沉
        }
    }
    
    // 上浮操作（维护最大堆性质）
    void heapifyUp(int index) {
        while (index > 0) {
            int parentIndex = parent(index);
            
            if (arr[index] > arr[parentIndex]) {
                swap(index, parentIndex);
                index = parentIndex;
            } else {
                break;
            }
        }
    }
    
public:
    HeapSort(vector<T>& array) : arr(array), size(array.size()) {}
    
    // 建堆（自底向上）
    void buildMaxHeap() {
        // 从最后一个非叶子节点开始，自底向上调整
        for (int i = parent(size - 1); i >= 0; i--) {
            heapifyDown(i, size);
        }
    }
    
    // 堆排序主算法
    void heapSort() {
        // 第一步：建堆
        buildMaxHeap();
        
        // 第二步：排序
        for (int i = size - 1; i > 0; i--) {
            // 将最大值（根节点）与最后一个元素交换
            swap(0, i);
            
            // 减小堆大小，并对新的根节点进行堆化
            heapifyDown(0, i);
        }
    }
    
    // 获取排序结果
    const vector<T>& getSortedArray() const {
        return arr;
    }
    
    // 检查数组是否已排序
    bool isSorted() const {
        for (int i = 1; i < size; i++) {
            if (arr[i] < arr[i - 1]) {
                return false;
            }
        }
        return true;
    }
};
```

### 最小堆排序（降序）

```cpp
template<typename T>
class MinHeapSort {
private:
    vector<T>& arr;
    int size;
    
    int parent(int i) const { return (i - 1) / 2; }
    int leftChild(int i) const { return 2 * i + 1; }
    int rightChild(int i) const { return 2 * i + 2; }
    
    void swap(int i, int j) {
        T temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
    
    // 下沉操作（维护最小堆性质）
    void heapifyDown(int index, int heapSize) {
        int smallest = index;
        int left = leftChild(index);
        int right = rightChild(index);
        
        // 找到最小的子节点
        if (left < heapSize && arr[left] < arr[smallest]) {
            smallest = left;
        }
        if (right < heapSize && arr[right] < arr[smallest]) {
            smallest = right;
        }
        
        // 如果最小子节点小于当前节点，则交换
        if (smallest != index) {
            swap(index, smallest);
            heapifyDown(smallest, heapSize);
        }
    }
    
public:
    MinHeapSort(vector<T>& array) : arr(array), size(array.size()) {}
    
    // 建最小堆
    void buildMinHeap() {
        for (int i = parent(size - 1); i >= 0; i--) {
            heapifyDown(i, size);
        }
    }
    
    // 最小堆排序（降序）
    void heapSortDescending() {
        // 建最小堆
        buildMinHeap();
        
        // 排序
        for (int i = size - 1; i > 0; i--) {
            // 将最小值（根节点）与最后一个元素交换
            swap(0, i);
            
            // 减小堆大小，并对新的根节点进行堆化
            heapifyDown(0, i);
        }
    }
};
```

## 🎯 堆排序优化实现

### 1. 迭代版本（避免递归）

```cpp
template<typename T>
class IterativeHeapSort {
private:
    vector<T>& arr;
    int size;
    
    int parent(int i) const { return (i - 1) / 2; }
    int leftChild(int i) const { return 2 * i + 1; }
    int rightChild(int i) const { return 2 * i + 2; }
    
    void swap(int i, int j) {
        T temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
    
    // 迭代版本的下沉操作
    void heapifyDownIterative(int index, int heapSize) {
        while (true) {
            int largest = index;
            int left = leftChild(index);
            int right = rightChild(index);
            
            if (left < heapSize && arr[left] > arr[largest]) {
                largest = left;
            }
            if (right < heapSize && arr[right] > arr[largest]) {
                largest = right;
            }
            
            if (largest == index) break; // 堆序性质已满足
            
            swap(index, largest);
            index = largest;
        }
    }
    
public:
    IterativeHeapSort(vector<T>& array) : arr(array), size(array.size()) {}
    
    // 迭代版本建堆
    void buildMaxHeap() {
        for (int i = parent(size - 1); i >= 0; i--) {
            heapifyDownIterative(i, size);
        }
    }
    
    // 迭代版本堆排序
    void heapSort() {
        buildMaxHeap();
        
        for (int i = size - 1; i > 0; i--) {
            swap(0, i);
            heapifyDownIterative(0, i);
        }
    }
};
```

### 2. 稳定版本（使用辅助空间）

```cpp
template<typename T>
class StableHeapSort {
private:
    vector<T>& arr;
    int size;
    
    // 稳定版本的堆节点（包含原始索引）
    struct StableNode {
        T value;
        int originalIndex;
        
        StableNode(const T& v, int idx) : value(v), originalIndex(idx) {}
        
        bool operator>(const StableNode& other) const {
            if (value != other.value) {
                return value > other.value;
            }
            return originalIndex < other.originalIndex; // 保持稳定性
        }
    };
    
public:
    StableHeapSort(vector<T>& array) : arr(array), size(array.size()) {}
    
    // 稳定堆排序
    void stableHeapSort() {
        vector<StableNode> heap;
        
        // 构建稳定堆
        for (int i = 0; i < size; i++) {
            heap.push_back(StableNode(arr[i], i));
        }
        
        // 建堆
        for (int i = parent(size - 1); i >= 0; i--) {
            heapifyDownStable(heap, i, size);
        }
        
        // 排序
        vector<T> result;
        for (int i = size - 1; i >= 0; i--) {
            result.push_back(heap[0].value);
            swap(heap[0], heap[i]);
            heapifyDownStable(heap, 0, i);
        }
        
        // 复制结果回原数组
        for (int i = 0; i < size; i++) {
            arr[i] = result[size - 1 - i];
        }
    }
    
private:
    int parent(int i) const { return (i - 1) / 2; }
    int leftChild(int i) const { return 2 * i + 1; }
    int rightChild(int i) const { return 2 * i + 2; }
    
    void swap(StableNode& a, StableNode& b) {
        StableNode temp = a;
        a = b;
        b = temp;
    }
    
    void heapifyDownStable(vector<StableNode>& heap, int index, int heapSize) {
        while (true) {
            int largest = index;
            int left = leftChild(index);
            int right = rightChild(index);
            
            if (left < heapSize && heap[left] > heap[largest]) {
                largest = left;
            }
            if (right < heapSize && heap[right] > heap[largest]) {
                largest = right;
            }
            
            if (largest == index) break;
            
            swap(heap[index], heap[largest]);
            index = largest;
        }
    }
};
```

### 3. 并行堆排序

```cpp
template<typename T>
class ParallelHeapSort {
private:
    vector<T>& arr;
    int size;
    
    int parent(int i) const { return (i - 1) / 2; }
    int leftChild(int i) const { return 2 * i + 1; }
    int rightChild(int i) const { return 2 * i + 2; }
    
    void swap(int i, int j) {
        T temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
    
    // 并行建堆
    void parallelBuildMaxHeap() {
        int numThreads = thread::hardware_concurrency();
        int chunkSize = max(1, size / numThreads);
        
        vector<thread> threads;
        
        // 并行处理不同的子树
        for (int i = 0; i < numThreads; i++) {
            int start = i * chunkSize;
            int end = min((i + 1) * chunkSize, size);
            
            threads.emplace_back([this, start, end]() {
                for (int j = start; j < end; j++) {
                    heapifyDown(j, size);
                }
            });
        }
        
        // 等待所有线程完成
        for (auto& t : threads) {
            t.join();
        }
        
        // 最终调整
        for (int i = parent(size - 1); i >= 0; i--) {
            heapifyDown(i, size);
        }
    }
    
    void heapifyDown(int index, int heapSize) {
        while (true) {
            int largest = index;
            int left = leftChild(index);
            int right = rightChild(index);
            
            if (left < heapSize && arr[left] > arr[largest]) {
                largest = left;
            }
            if (right < heapSize && arr[right] > arr[largest]) {
                largest = right;
            }
            
            if (largest == index) break;
            
            swap(index, largest);
            index = largest;
        }
    }
    
public:
    ParallelHeapSort(vector<T>& array) : arr(array), size(array.size()) {}
    
    // 并行堆排序
    void parallelHeapSort() {
        parallelBuildMaxHeap();
        
        for (int i = size - 1; i > 0; i--) {
            swap(0, i);
            heapifyDown(0, i);
        }
    }
};
```

## 🔧 堆排序应用场景

### 1. Top-K问题

```cpp
template<typename T>
class TopKFinder {
private:
    vector<T> data;
    
public:
    TopKFinder(const vector<T>& input) : data(input) {}
    
    // 使用堆排序找Top-K
    vector<T> findTopK(int k) {
        if (k <= 0 || k > data.size()) {
            return vector<T>();
        }
        
        // 使用最小堆找最大的K个元素
        vector<T> heap(data.begin(), data.begin() + k);
        
        // 建最小堆
        MinHeapSort<T> minHeap(heap);
        minHeap.buildMinHeap();
        
        // 处理剩余元素
        for (int i = k; i < data.size(); i++) {
            if (data[i] > heap[0]) {
                heap[0] = data[i];
                minHeap.heapifyDown(0, k);
            }
        }
        
        // 对结果排序
        minHeap.heapSortDescending();
        return heap;
    }
    
    // 使用堆排序找Bottom-K
    vector<T> findBottomK(int k) {
        if (k <= 0 || k > data.size()) {
            return vector<T>();
        }
        
        // 使用最大堆找最小的K个元素
        vector<T> heap(data.begin(), data.begin() + k);
        
        // 建最大堆
        HeapSort<T> maxHeap(heap);
        maxHeap.buildMaxHeap();
        
        // 处理剩余元素
        for (int i = k; i < data.size(); i++) {
            if (data[i] < heap[0]) {
                heap[0] = data[i];
                maxHeap.heapifyDown(0, k);
            }
        }
        
        // 对结果排序
        maxHeap.heapSort();
        return heap;
    }
};
```

### 2. 优先队列实现

```cpp
template<typename T>
class PriorityQueue {
private:
    vector<T> heap;
    int size;
    
    int parent(int i) const { return (i - 1) / 2; }
    int leftChild(int i) const { return 2 * i + 1; }
    int rightChild(int i) const { return 2 * i + 2; }
    
    void swap(int i, int j) {
        T temp = heap[i];
        heap[i] = heap[j];
        heap[j] = temp;
    }
    
    void heapifyUp(int index) {
        while (index > 0) {
            int parentIndex = parent(index);
            if (heap[index] > heap[parentIndex]) {
                swap(index, parentIndex);
                index = parentIndex;
            } else {
                break;
            }
        }
    }
    
    void heapifyDown(int index) {
        while (true) {
            int largest = index;
            int left = leftChild(index);
            int right = rightChild(index);
            
            if (left < size && heap[left] > heap[largest]) {
                largest = left;
            }
            if (right < size && heap[right] > heap[largest]) {
                largest = right;
            }
            
            if (largest == index) break;
            
            swap(index, largest);
            index = largest;
        }
    }
    
public:
    PriorityQueue() : size(0) {}
    
    // 插入元素
    void push(const T& value) {
        heap.push_back(value);
        size++;
        heapifyUp(size - 1);
    }
    
    // 删除最大元素
    T pop() {
        if (size == 0) {
            throw std::runtime_error("Priority queue is empty");
        }
        
        T maxValue = heap[0];
        heap[0] = heap[size - 1];
        heap.pop_back();
        size--;
        
        if (size > 0) {
            heapifyDown(0);
        }
        
        return maxValue;
    }
    
    // 获取最大元素
    T top() const {
        if (size == 0) {
            throw std::runtime_error("Priority queue is empty");
        }
        return heap[0];
    }
    
    // 检查是否为空
    bool empty() const {
        return size == 0;
    }
    
    // 获取大小
    int getSize() const {
        return size;
    }
};
```

### 3. 外部排序

```cpp
template<typename T>
class ExternalHeapSort {
private:
    string inputFile;
    string outputFile;
    int bufferSize;
    
public:
    ExternalHeapSort(const string& input, const string& output, int buffer) 
        : inputFile(input), outputFile(output), bufferSize(buffer) {}
    
    // 外部堆排序
    void externalHeapSort() {
        // 第一阶段：将大文件分成小文件并排序
        vector<string> tempFiles = createSortedRuns();
        
        // 第二阶段：多路归并
        mergeSortedRuns(tempFiles);
        
        // 清理临时文件
        for (const string& file : tempFiles) {
            remove(file.c_str());
        }
    }
    
private:
    // 创建排序后的运行文件
    vector<string> createSortedRuns() {
        vector<string> tempFiles;
        ifstream input(inputFile);
        int runCount = 0;
        
        while (!input.eof()) {
            vector<T> buffer;
            T value;
            
            // 读取缓冲区大小的数据
            for (int i = 0; i < bufferSize && input >> value; i++) {
                buffer.push_back(value);
            }
            
            if (!buffer.empty()) {
                // 对缓冲区进行堆排序
                HeapSort<T> heapSort(buffer);
                heapSort.heapSort();
                
                // 写入临时文件
                string tempFile = "temp_run_" + to_string(runCount++) + ".txt";
                ofstream output(tempFile);
                
                for (const T& val : buffer) {
                    output << val << " ";
                }
                
                output.close();
                tempFiles.push_back(tempFile);
            }
        }
        
        input.close();
        return tempFiles;
    }
    
    // 多路归并
    void mergeSortedRuns(const vector<string>& tempFiles) {
        ofstream output(outputFile);
        vector<ifstream> inputs;
        vector<T> currentValues;
        
        // 打开所有临时文件
        for (const string& file : tempFiles) {
            inputs.emplace_back(file);
            T value;
            if (inputs.back() >> value) {
                currentValues.push_back(value);
            } else {
                currentValues.push_back(numeric_limits<T>::max());
            }
        }
        
        // 使用最小堆进行多路归并
        PriorityQueue<pair<T, int>> minHeap;
        
        for (int i = 0; i < currentValues.size(); i++) {
            if (currentValues[i] != numeric_limits<T>::max()) {
                minHeap.push({currentValues[i], i});
            }
        }
        
        while (!minHeap.empty()) {
            auto [value, fileIndex] = minHeap.pop();
            output << value << " ";
            
            if (inputs[fileIndex] >> currentValues[fileIndex]) {
                minHeap.push({currentValues[fileIndex], fileIndex});
            }
        }
        
        output.close();
        
        // 关闭所有输入文件
        for (auto& input : inputs) {
            input.close();
        }
    }
};
```

## ⚡ 复杂度分析

### 时间复杂度

| 操作 | 时间复杂度 | 说明 |
|------|------------|------|
| **建堆** | O(n) | 自底向上建堆 |
| **堆化** | O(log n) | 单次下沉操作 |
| **排序** | O(n log n) | n次堆化操作 |
| **总体** | O(n log n) | 建堆 + 排序 |

### 空间复杂度

| 实现方式 | 空间复杂度 | 说明 |
|----------|------------|------|
| **原地排序** | O(1) | 只使用常数额外空间 |
| **稳定版本** | O(n) | 需要辅助数组 |
| **并行版本** | O(1) | 线程开销忽略 |

### 性能特点

| 特点 | 描述 | 影响 |
|------|------|------|
| **稳定性** | 不稳定 | 相同元素的相对位置可能改变 |
| **原地性** | 原地排序 | 空间效率高 |
| **适应性** | 非自适应 | 性能与输入顺序无关 |
| **缓存性能** | 较差 | 随机访问模式 |
