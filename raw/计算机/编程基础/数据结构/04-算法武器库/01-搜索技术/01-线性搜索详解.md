# 线性搜索详解

## 📖 核心概念

**线性搜索**是最基础的搜索算法，通过逐个检查数组中的每个元素来查找目标值。虽然效率不高，但实现简单，适用于小规模数据。

### 🏗️ 线性搜索特性

## 🔧 基础实现

### 标准线性搜索

```cpp
class LinearSearch {
public:
    // 基础线性搜索
    int search(const vector<int>& arr, int target) {
        for (int i = 0; i < arr.size(); ++i) {
            if (arr[i] == target) {
                return i;  // 返回索引
            }
        }
        return -1;  // 未找到
    }
    
    // 查找所有匹配项
    vector<int> searchAll(const vector<int>& arr, int target) {
        vector<int> indices;
        for (int i = 0; i < arr.size(); ++i) {
            if (arr[i] == target) {
                indices.push_back(i);
            }
        }
        return indices;
    }
    
    // 查找第一个匹配项
    int searchFirst(const vector<int>& arr, int target) {
        for (int i = 0; i < arr.size(); ++i) {
            if (arr[i] == target) {
                return i;
            }
        }
        return -1;
    }
    
    // 查找最后一个匹配项
    int searchLast(const vector<int>& arr, int target) {
        for (int i = arr.size() - 1; i >= 0; --i) {
            if (arr[i] == target) {
                return i;
            }
        }
        return -1;
    }
    
    // 查找最接近的值
    int searchClosest(const vector<int>& arr, int target) {
        if (arr.empty()) return -1;
        
        int closestIndex = 0;
        int minDiff = abs(arr[0] - target);
        
        for (int i = 1; i < arr.size(); ++i) {
            int diff = abs(arr[i] - target);
            if (diff < minDiff) {
                minDiff = diff;
                closestIndex = i;
            }
        }
        
        return closestIndex;
    }
};
```

### 优化版本

```cpp
class OptimizedLinearSearch {
public:
    // 哨兵优化版本
    int searchWithSentinel(vector<int>& arr, int target) {
        int n = arr.size();
        if (n == 0) return -1;
        
        // 保存最后一个元素
        int lastElement = arr[n - 1];
        
        // 设置哨兵
        arr[n - 1] = target;
        
        int i = 0;
        while (arr[i] != target) {
            ++i;
        }
        
        // 恢复最后一个元素
        arr[n - 1] = lastElement;
        
        // 检查是否找到
        if (i < n - 1 || lastElement == target) {
            return i;
        }
        
        return -1;
    }
    
    // 双向搜索
    int searchBidirectional(const vector<int>& arr, int target) {
        int left = 0;
        int right = arr.size() - 1;
        
        while (left <= right) {
            if (arr[left] == target) {
                return left;
            }
            if (arr[right] == target) {
                return right;
            }
            ++left;
            --right;
        }
        
        return -1;
    }
    
    // 跳跃搜索（适用于有序数组）
    int jumpSearch(const vector<int>& arr, int target) {
        int n = arr.size();
        if (n == 0) return -1;
        
        int step = sqrt(n);
        int prev = 0;
        
        // 跳跃到合适的块
        while (arr[min(step, n) - 1] < target) {
            prev = step;
            step += sqrt(n);
            if (prev >= n) {
                return -1;
            }
        }
        
        // 在块内进行线性搜索
        while (arr[prev] < target) {
            ++prev;
            if (prev == min(step, n)) {
                return -1;
            }
        }
        
        if (arr[prev] == target) {
            return prev;
        }
        
        return -1;
    }
};
```

| 搜索算法 | 时间复杂度 | 空间复杂度 | 适用场景 | 优点 | 缺点 |
|----------|------------|------------|----------|------|------|
| **线性搜索** | O(n) | O(1) | 无序数组 | 简单、通用 | 效率低 |
| **二分搜索** | O(log n) | O(1) | 有序数组 | 效率高 | 需要排序 |
| **哈希搜索** | O(1) | O(n) | 任意数组 | 最快 | 空间开销大 |

## 🎯 应用场景

### 1. 简单查找

```cpp
class SimpleSearch {
public:
    // 查找字符串中的字符
    int findChar(const string& str, char target) {
        for (int i = 0; i < str.length(); ++i) {
            if (str[i] == target) {
                return i;
            }
        }
        return -1;
    }
    
    // 查找数组中的最大值
    int findMax(const vector<int>& arr) {
        if (arr.empty()) return -1;
        
        int maxIndex = 0;
        for (int i = 1; i < arr.size(); ++i) {
            if (arr[i] > arr[maxIndex]) {
                maxIndex = i;
            }
        }
        return maxIndex;
    }
    
    // 查找数组中的最小值
    int findMin(const vector<int>& arr) {
        if (arr.empty()) return -1;
        
        int minIndex = 0;
        for (int i = 1; i < arr.size(); ++i) {
            if (arr[i] < arr[minIndex]) {
                minIndex = i;
            }
        }
        return minIndex;
    }
    
    // 统计目标值出现次数
    int countOccurrences(const vector<int>& arr, int target) {
        int count = 0;
        for (int value : arr) {
            if (value == target) {
                ++count;
            }
        }
        return count;
    }
};
```

### 2. 数据验证

```cpp
class DataValidation {
public:
    // 检查数组是否包含重复元素
    bool hasDuplicates(const vector<int>& arr) {
        for (int i = 0; i < arr.size(); ++i) {
            for (int j = i + 1; j < arr.size(); ++j) {
                if (arr[i] == arr[j]) {
                    return true;
                }
            }
        }
        return false;
    }
    
    // 检查数组是否包含特定范围的值
    bool containsInRange(const vector<int>& arr, int minVal, int maxVal) {
        for (int value : arr) {
            if (value >= minVal && value <= maxVal) {
                return true;
            }
        }
        return false;
    }
    
    // 查找第一个满足条件的元素
    int findFirst(const vector<int>& arr, function<bool(int)> predicate) {
        for (int i = 0; i < arr.size(); ++i) {
            if (predicate(arr[i])) {
                return i;
            }
        }
        return -1;
    }
    
    // 查找所有满足条件的元素
    vector<int> findAll(const vector<int>& arr, function<bool(int)> predicate) {
        vector<int> result;
        for (int i = 0; i < arr.size(); ++i) {
            if (predicate(arr[i])) {
                result.push_back(i);
            }
        }
        return result;
    }
};
```

## 🔧 优化技巧

### 1. 循环优化

```cpp
class LoopOptimization {
public:
    // 减少循环开销
    int searchOptimized(const vector<int>& arr, int target) {
        int n = arr.size();
        for (int i = 0; i < n; ++i) {
            if (arr[i] == target) {
                return i;
            }
        }
        return -1;
    }
    
    // 使用指针优化
    int searchWithPointers(const vector<int>& arr, int target) {
        const int* ptr = arr.data();
        const int* end = ptr + arr.size();
        
        while (ptr != end) {
            if (*ptr == target) {
                return ptr - arr.data();
            }
            ++ptr;
        }
        
        return -1;
    }
    
    // 循环展开
    int searchUnrolled(const vector<int>& arr, int target) {
        int n = arr.size();
        int i = 0;
        
        // 每次处理4个元素
        for (; i + 3 < n; i += 4) {
            if (arr[i] == target) return i;
            if (arr[i + 1] == target) return i + 1;
            if (arr[i + 2] == target) return i + 2;
            if (arr[i + 3] == target) return i + 3;
        }
        
        // 处理剩余元素
        for (; i < n; ++i) {
            if (arr[i] == target) return i;
        }
        
        return -1;
    }
};
```

### 2. 并行优化

```cpp
class ParallelSearch {
public:
    // 并行线性搜索
    int searchParallel(const vector<int>& arr, int target) {
        int n = arr.size();
        int numThreads = thread::hardware_concurrency();
        int chunkSize = n / numThreads;
        
        vector<future<int>> futures;
        
        for (int i = 0; i < numThreads; ++i) {
            int start = i * chunkSize;
            int end = (i == numThreads - 1) ? n : (i + 1) * chunkSize;
            
            futures.push_back(async(launch::async, [&arr, target, start, end]() {
                for (int j = start; j < end; ++j) {
                    if (arr[j] == target) {
                        return j;
                    }
                }
                return -1;
            }));
        }
        
        for (auto& future : futures) {
            int result = future.get();
            if (result != -1) {
                return result;
            }
        }
        
        return -1;
    }
};
```

## 💡 线性搜索要点

1. **简单实现**：最容易理解和实现的搜索算法
2. **通用性强**：适用于任何数据结构
3. **效率较低**：时间复杂度O(n)
4. **优化空间**：通过哨兵、并行等方式优化
