# 三-分搜索与插值搜索

## 📖 核心概念

**三-分搜索**和**插值搜索**是两种重要的搜索算法，它们都是对二分搜索的改进和扩展。三-分搜索将搜索区间分成三部分，插值搜索根据数据分布进行智能猜测。

### 🏗️ 搜索算法分类

```mermaid

    D --> D2[均匀分布优化]
    
    E --> E1[线性搜索]
    E --> E2[哈希搜索]
```

## 🔧 三-分搜索实现

### 三-分搜索基础

```cpp
class TernarySearch {
public:
    // 三-分搜索（整数版本）
    int ternarySearch(vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right) {
            int mid1 = left + (right - left) / 3;
            int mid2 = right - (right - left) / 3;
            
            if (arr[mid1] == target) {
                return mid1;
            }
            if (arr[mid2] == target) {
                return mid2;
            }
            
            if (target < arr[mid1]) {
                right = mid1 - 1;
            } else if (target > arr[mid2]) {
                left = mid2 + 1;
            } else {
                left = mid1 + 1;
                right = mid2 - 1;
            }
        }
        
        return -1;
    }
    
    // 三-分搜索（浮点数版本）
    double ternarySearchFunction(double left, double right, function<double(double)> func) {
        const double EPS = 1e-9;
        
        while (right - left > EPS) {
            double mid1 = left + (right - left) / 3;
            double mid2 = right - (right - left) / 3;
            
            if (func(mid1) < func(mid2)) {
                right = mid2;
            } else {
                left = mid1;
            }
        }
        
        return (left + right) / 2;
    }
    
    // 三-分搜索最大值
    int ternarySearchMax(vector<int>& arr) {
        int left = 0, right = arr.size() - 1;
        
        while (left < right) {
            int mid1 = left + (right - left) / 3;
            int mid2 = right - (right - left) / 3;
            
            if (arr[mid1] < arr[mid2]) {
                left = mid1 + 1;
            } else {
                right = mid2 - 1;
            }
        }
        
        return left;
    }
    
    // 三-分搜索最小值
    int ternarySearchMin(vector<int>& arr) {
        int left = 0, right = arr.size() - 1;
        
        while (left < right) {
            int mid1 = left + (right - left) / 3;
            int mid2 = right - (right - left) / 3;
            
            if (arr[mid1] > arr[mid2]) {
                left = mid1 + 1;
            } else {
                right = mid2 - 1;
            }
        }
        
        return left;
    }
    
    // 显示三-分搜索过程
    void displayTernarySearch(vector<int>& arr, int target) {
        cout << "Ternary Search for " << target << ":" << endl;
        cout << "========================" << endl;
        
        int left = 0, right = arr.size() - 1;
        int steps = 0;
        
        while (left <= right) {
            steps++;
            int mid1 = left + (right - left) / 3;
            int mid2 = right - (right - left) / 3;
            
            cout << "Step " << steps << ": left=" << left << ", right=" << right 
                 << ", mid1=" << mid1 << ", mid2=" << mid2 << endl;
            cout << "Values: arr[" << mid1 << "]=" << arr[mid1] 
                 << ", arr[" << mid2 << "]=" << arr[mid2] << endl;
            
            if (arr[mid1] == target) {
                cout << "Found at index " << mid1 << endl;
                return;
            }
            if (arr[mid2] == target) {
                cout << "Found at index " << mid2 << endl;
                return;
            }
            
            if (target < arr[mid1]) {
                right = mid1 - 1;
                cout << "Target < arr[mid1], search left" << endl;
            } else if (target > arr[mid2]) {
                left = mid2 + 1;
                cout << "Target > arr[mid2], search right" << endl;
            } else {
                left = mid1 + 1;
                right = mid2 - 1;
                cout << "Target between mid1 and mid2, search middle" << endl;
            }
            cout << endl;
        }
        
        cout << "Target not found" << endl;
    }
};
```

```cpp
    int searchInRotatedArray(vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right) {
            int mid1 = left + (right - left) / 3;
            int mid2 = right - (right - left) / 3;
            
            if (arr[mid1] == target) return mid1;
            if (arr[mid2] == target) return mid2;
            
            if (arr[mid1] <= arr[mid2]) {
                if (target >= arr[mid1] && target <= arr[mid2]) {
                    left = mid1 + 1;
                    right = mid2 - 1;
                } else if (target < arr[mid1]) {
                    right = mid1 - 1;
                } else {
                    left = mid2 + 1;
                }
            } else {
                if (target >= arr[mid1] || target <= arr[mid2]) {
                    left = mid1 + 1;
                    right = mid2 - 1;
                } else if (target < arr[mid1]) {
                    right = mid1 - 1;
                } else {
                    left = mid2 + 1;
                }
            }
        }
        
        return -1;
    }
```

## 🔧 插值搜索实现

### 插值搜索基础

```cpp
class InterpolationSearch {
public:
    // 插值搜索
    int interpolationSearch(vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right && target >= arr[left] && target <= arr[right]) {
            if (left == right) {
                if (arr[left] == target) {
                    return left;
                }
                return -1;
            }
            
            // 计算插值位置
            int pos = left + ((target - arr[left]) * (right - left)) / (arr[right] - arr[left]);
            
            if (arr[pos] == target) {
                return pos;
            } else if (arr[pos] < target) {
                left = pos + 1;
            } else {
                right = pos - 1;
            }
        }
        
        return -1;
    }
    
    // 插值搜索（递归版本）
    int interpolationSearchRecursive(vector<int>& arr, int target, int left, int right) {
        if (left > right || target < arr[left] || target > arr[right]) {
            return -1;
        }
        
        if (left == right) {
            return (arr[left] == target) ? left : -1;
        }
        
        int pos = left + ((target - arr[left]) * (right - left)) / (arr[right] - arr[left]);
        
        if (arr[pos] == target) {
            return pos;
        } else if (arr[pos] < target) {
            return interpolationSearchRecursive(arr, target, pos + 1, right);
        } else {
            return interpolationSearchRecursive(arr, target, left, pos - 1);
        }
    }
    
    // 显示插值搜索过程
    void displayInterpolationSearch(vector<int>& arr, int target) {
        cout << "Interpolation Search for " << target << ":" << endl;
        cout << "=============================" << endl;
        
        int left = 0, right = arr.size() - 1;
        int steps = 0;
        
        while (left <= right && target >= arr[left] && target <= arr[right]) {
            steps++;
            
            if (left == right) {
                if (arr[left] == target) {
                    cout << "Step " << steps << ": Found at index " << left << endl;
                } else {
                    cout << "Step " << steps << ": Not found" << endl;
                }
                return;
            }
            
            int pos = left + ((target - arr[left]) * (right - left)) / (arr[right] - arr[left]);
            
            cout << "Step " << steps << ": left=" << left << ", right=" << right 
                 << ", pos=" << pos << ", arr[" << pos << "]=" << arr[pos] << endl;
            
            if (arr[pos] == target) {
                cout << "Found at index " << pos << endl;
                return;
            } else if (arr[pos] < target) {
                left = pos + 1;
                cout << "Target > arr[pos], search right" << endl;
            } else {
                right = pos - 1;
                cout << "Target < arr[pos], search left" << endl;
            }
            cout << endl;
        }
        
        cout << "Target not found" << endl;
    }
    
    // 插值搜索优化版本
    int interpolationSearchOptimized(vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right && target >= arr[left] && target <= arr[right]) {
            if (left == right) {
                return (arr[left] == target) ? left : -1;
            }
            
            // 使用更精确的插值公式
            double ratio = (double)(target - arr[left]) / (arr[right] - arr[left]);
            int pos = left + (int)(ratio * (right - left));
            
            // 确保pos在有效范围内
            pos = max(left, min(right, pos));
            
            if (arr[pos] == target) {
                return pos;
            } else if (arr[pos] < target) {
                left = pos + 1;
            } else {
                right = pos - 1;
            }
        }
        
        return -1;
    }
};
```

```cpp
class InterpolationSearchApplications {
public:
    // 查找均匀分布数组中的元素
    int searchUniformArray(vector<int>& arr, int target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right && target >= arr[left] && target <= arr[right]) {
            if (left == right) {
                return (arr[left] == target) ? left : -1;
            }
            
            int pos = left + ((target - arr[left]) * (right - left)) / (arr[right] - arr[left]);
            
            if (arr[pos] == target) {
                return pos;
            } else if (arr[pos] < target) {
                left = pos + 1;
            } else {
                right = pos - 1;
            }
        }
        
        return -1;
    }
    
    // 查找字符串数组中的元素
    int searchStringArray(vector<string>& arr, string target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right) {
            if (left == right) {
                return (arr[left] == target) ? left : -1;
            }
            
            // 使用字符串的哈希值进行插值
            int leftHash = hash<string>{}(arr[left]);
            int rightHash = hash<string>{}(arr[right]);
            int targetHash = hash<string>{}(target);
            
            int pos = left + ((targetHash - leftHash) * (right - left)) / (rightHash - leftHash);
            pos = max(left, min(right, pos));
            
            if (arr[pos] == target) {
                return pos;
            } else if (arr[pos] < target) {
                left = pos + 1;
            } else {
                right = pos - 1;
            }
        }
        
        return -1;
    }
    
    // 查找浮点数数组中的元素
    int searchFloatArray(vector<double>& arr, double target) {
        int left = 0, right = arr.size() - 1;
        
        while (left <= right && target >= arr[left] && target <= arr[right]) {
            if (left == right) {
                return (abs(arr[left] - target) < 1e-9) ? left : -1;
            }
            
            double ratio = (target - arr[left]) / (arr[right] - arr[left]);
            int pos = left + (int)(ratio * (right - left));
            pos = max(left, min(right, pos));
            
            if (abs(arr[pos] - target) < 1e-9) {
                return pos;
            } else if (arr[pos] < target) {
                left = pos + 1;
            } else {
                right = pos - 1;
            }
        }
        
        return -1;
    }
    
    // 显示插值搜索应用
    void displayApplications(vector<int>& arr, int target) {
        cout << "Interpolation Search Applications:" << endl;
        cout << "=================================" << endl;
        
        cout << "Array: ";
        for (int num : arr) {
            cout << num << " ";
        }
        cout << endl;
        
        int result = searchUniformArray(arr, target);
        if (result != -1) {
            cout << "Target " << target << " found at index " << result << endl;
        } else {
            cout << "Target " << target << " not found" << endl;
        }
    }
};
```

## 💡 搜索算法要点

1. **三-分搜索**: 将搜索区间分成三部分，适合单峰函数
2. **插值搜索**: 根据数据分布智能猜测，适合均匀分布
3. **性能**: 插值搜索在均匀分布下性能最好
4. **应用**: 根据数据特点选择合适的搜索算法
