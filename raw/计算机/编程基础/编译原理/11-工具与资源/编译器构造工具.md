# 编译器构造工具

## 🎯 编译器构造工具概述

**编译器构造工具**是帮助开发者快速构建编译器的专业工具集，包括词法分析器生成器、语法分析器生成器、代码生成器、优化器等。这些工具大大简化了编译器的开发过程，提高了开发效率。

## 🔍 工具分类

### 1. 词法分析器生成器
- **Flex/Lex**：经典的词法分析器生成器
- **ANTLR**：功能强大的语法分析器生成器
- **Ragel**：状态机编译器
- **RE2C**：正则表达式到C代码的转换器

### 2. 语法分析器生成器
- **Yacc/Bison**：经典的语法分析器生成器
- **ANTLR**：支持多种目标语言的语法分析器
- **Packrat**：PEG解析器生成器
- **Lemon**：轻量级语法分析器生成器

### 3. 代码生成器
- **LLVM**：现代化的编译器基础设施
- **GCC**：GNU编译器集合
- **TinyCC**：轻量级C编译器
- **Clang**：基于LLVM的C/C++编译器

### 4. 优化器
- **LLVM Opt**：LLVM优化器
- **GCC优化器**：GCC内置优化器
- **Intel编译器**：Intel C++编译器
- **MSVC优化器**：Microsoft Visual C++优化器

### ANTLR
```java
// 示例：使用ANTLR生成词法分析器
grammar SimpleCalc;

// 词法规则
NUMBER: [0-9]+;
IDENTIFIER: [a-zA-Z][a-zA-Z0-9]*;
PLUS: '+';
MINUS: '-';
MULTIPLY: '*';
DIVIDE: '/';
ASSIGN: '=';
LPAREN: '(';
RPAREN: ')';
SEMICOLON: ';';
WS: [ \t\r\n]+ -> skip;

// 语法规则
program: statement+;

statement: assignment | expression;

assignment: IDENTIFIER ASSIGN expression SEMICOLON;

expression: term (PLUS | MINUS term)*;

term: factor (MULTIPLY | DIVIDE factor)*;

factor: NUMBER | IDENTIFIER | LPAREN expression RPAREN;

// 访问者模式
public class CalcVisitor extends SimpleCalcBaseVisitor<Double> {
    private Map<String, Double> variables = new HashMap<>();
    
    @Override
    public Double visitAssignment(SimpleCalcParser.AssignmentContext ctx) {
        String varName = ctx.IDENTIFIER().getText();
        Double value = visit(ctx.expression());
        variables.put(varName, value);
        System.out.println(varName + " = " + value);
        return value;
    }
    
    @Override
    public Double visitExpression(SimpleCalcParser.ExpressionContext ctx) {
        Double result = visit(ctx.term(0));
        
        for (int i = 1; i < ctx.term().size(); i++) {
            String op = ctx.getChild(2 * i - 1).getText();
            Double term = visit(ctx.term(i));
            
            if (op.equals("+")) {
                result += term;
            } else if (op.equals("-")) {
                result -= term;
            }
        }
        
        return result;
    }
    
    @Override
    public Double visitTerm(SimpleCalcParser.TermContext ctx) {
        Double result = visit(ctx.factor(0));
        
        for (int i = 1; i < ctx.factor().size(); i++) {
            String op = ctx.getChild(2 * i - 1).getText();
            Double factor = visit(ctx.factor(i));
            
            if (op.equals("*")) {
                result *= factor;
            } else if (op.equals("/")) {
                result /= factor;
            }
        }
        
        return result;
    }
    
    @Override
    public Double visitFactor(SimpleCalcParser.FactorContext ctx) {
        if (ctx.NUMBER() != null) {
            return Double.parseDouble(ctx.NUMBER().getText());
        } else if (ctx.IDENTIFIER() != null) {
            String varName = ctx.IDENTIFIER().getText();
            return variables.getOrDefault(varName, 0.0);
        } else {
            return visit(ctx.expression());
        }
    }
}
```

```c
// 示例：使用Bison生成语法分析器
%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// Token定义
typedef enum {
    TOKEN_NUMBER,
    TOKEN_IDENTIFIER,
    TOKEN_OPERATOR,
    TOKEN_DELIMITER,
    TOKEN_EOF
} TokenType;

typedef struct {
    TokenType type;
    char* value;
    int line;
    int column;
} Token;

// 语法树节点
typedef struct ASTNode {
    char* type;
    char* value;
    struct ASTNode* left;
    struct ASTNode* right;
} ASTNode;

// 全局变量
Token currentToken;
ASTNode* root;
%}

// Token声明
%token TOKEN_NUMBER TOKEN_IDENTIFIER TOKEN_OPERATOR TOKEN_DELIMITER TOKEN_EOF

// 优先级和结合性
%left '+' '-'
%left '*' '/'
%right '^'

// 语法规则
%%

program:
    statement_list
    {
        root = $1;
        printf("语法分析完成\n");
    }
;

statement_list:
    statement_list statement
    {
        $$ = createNode("statement_list", NULL, $1, $2);
    }
|   statement
    {
        $$ = createNode("statement_list", NULL, $1, NULL);
    }
;

statement:
    expression ';'
    {
        $$ = createNode("statement", NULL, $1, NULL);
    }
|   IDENTIFIER '=' expression ';'
    {
        $$ = createNode("assignment", $1, $3, NULL);
    }
;

expression:
    expression '+' term
    {
        $$ = createNode("add", "+", $1, $3);
    }
|   expression '-' term
    {
        $$ = createNode("subtract", "-", $1, $3);
    }
|   term
    {
        $$ = $1;
    }
;

term:
    term '*' factor
    {
        $$ = createNode("multiply", "*", $1, $3);
    }
|   term '/' factor
    {
        $$ = createNode("divide", "/", $1, $3);
    }
|   factor
    {
        $$ = $1;
    }
;

factor:
    NUMBER
    {
        $$ = createNode("number", $1, NULL, NULL);
    }
|   IDENTIFIER
    {
        $$ = createNode("identifier", $1, NULL, NULL);
    }
|   '(' expression ')'
    {
        $$ = $2;
    }
;

%%

// 辅助函数
ASTNode* createNode(const char* type, const char* value, ASTNode* left, ASTNode* right) {
    ASTNode* node = (ASTNode*)malloc(sizeof(ASTNode));
    node->type = strdup(type);
    node->value = value ? strdup(value) : NULL;
    node->left = left;
    node->right = right;
    return node;
}

void printAST(ASTNode* node, int depth) {
    if (!node) return;
    
    for (int i = 0; i < depth; i++) {
        printf("  ");
    }
    
    printf("%s", node->type);
    if (node->value) {
        printf(": %s", node->value);
    }
    printf("\n");
    
    printAST(node->left, depth + 1);
    printAST(node->right, depth + 1);
}

void freeAST(ASTNode* node) {
    if (!node) return;
    
    free(node->type);
    if (node->value) {
        free(node->value);
    }
    
    freeAST(node->left);
    freeAST(node->right);
    free(node);
}

int yyerror(const char* s) {
    printf("语法错误: %s\n", s);
    return 0;
}

int main() {
    printf("Bison语法分析器示例\n");
    printf("===========================================\n");
    
    int result = yyparse();
    
    if (result == 0) {
        printf("语法分析成功\n");
        printf("抽象语法树:\n");
        printAST(root, 0);
    } else {
        printf("语法分析失败\n");
    }
    
    freeAST(root);
    return result;
}
```

### Packrat
```c
// 示例：使用Packrat生成PEG解析器
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// PEG语法定义
typedef struct PEGRule {
    char* name;
    char* pattern;
    struct PEGRule* next;
} PEGRule;

typedef struct PEGParser {
    PEGRule* rules;
    char* input;
    size_t position;
    size_t length;
} PEGParser;

// 解析结果
typedef struct ParseResult {
    int success;
    char* value;
    size_t position;
} ParseResult;

// 创建PEG解析器
PEGParser* createPEGParser(const char* input) {
    PEGParser* parser = (PEGParser*)malloc(sizeof(PEGParser));
    parser->rules = NULL;
    parser->input = strdup(input);
    parser->position = 0;
    parser->length = strlen(input);
    return parser;
}

// 添加规则
void addRule(PEGParser* parser, const char* name, const char* pattern) {
    PEGRule* rule = (PEGRule*)malloc(sizeof(PEGRule));
    rule->name = strdup(name);
    rule->pattern = strdup(pattern);
    rule->next = parser->rules;
    parser->rules = rule;
}

// 解析数字
ParseResult parseNumber(PEGParser* parser) {
    ParseResult result = {0, NULL, parser->position};
    
    if (parser->position >= parser->length) {
        return result;
    }
    
    size_t start = parser->position;
    while (parser->position < parser->length && 
           parser->input[parser->position] >= '0' && 
           parser->input[parser->position] <= '9') {
        parser->position++;
    }
    
    if (parser->position > start) {
        result.success = 1;
        result.position = parser->position;
        result.value = strndup(parser->input + start, parser->position - start);
    }
    
    return result;
}

// 解析标识符
ParseResult parseIdentifier(PEGParser* parser) {
    ParseResult result = {0, NULL, parser->position};
    
    if (parser->position >= parser->length) {
        return result;
    }
    
    size_t start = parser->position;
    if ((parser->input[parser->position] >= 'a' && parser->input[parser->position] <= 'z') ||
        (parser->input[parser->position] >= 'A' && parser->input[parser->position] <= 'Z')) {
        parser->position++;
        
        while (parser->position < parser->length &&
               ((parser->input[parser->position] >= 'a' && parser->input[parser->position] <= 'z') ||
                (parser->input[parser->position] >= 'A' && parser->input[parser->position] <= 'Z') ||
                (parser->input[parser->position] >= '0' && parser->input[parser->position] <= '9'))) {
            parser->position++;
        }
        
        result.success = 1;
        result.position = parser->position;
        result.value = strndup(parser->input + start, parser->position - start);
    }
    
    return result;
}

// 解析操作符
ParseResult parseOperator(PEGParser* parser) {
    ParseResult result = {0, NULL, parser->position};
    
    if (parser->position >= parser->length) {
        return result;
    }
    
    char op = parser->input[parser->position];
    if (op == '+' || op == '-' || op == '*' || op == '/') {
        result.success = 1;
        result.position = parser->position + 1;
        result.value = strndup(parser->input + parser->position, 1);
        parser->position++;
    }
    
    return result;
}

// 解析表达式
ParseResult parseExpression(PEGParser* parser) {
    ParseResult result = {0, NULL, parser->position};
    
    // 尝试解析项
    ParseResult termResult = parseTerm(parser);
    if (!termResult.success) {
        return result;
    }
    
    result = termResult;
    
    // 尝试解析操作符和下一个项
    while (parser->position < parser->length) {
        ParseResult opResult = parseOperator(parser);
        if (!opResult.success) {
            break;
        }
        
        ParseResult nextTermResult = parseTerm(parser);
        if (!nextTermResult.success) {
            parser->position = result.position; // 回滚
            break;
        }
        
        // 创建新的结果
        char* newValue = (char*)malloc(strlen(result.value) + strlen(opResult.value) + strlen(nextTermResult.value) + 3);
        sprintf(newValue, "(%s %s %s)", result.value, opResult.value, nextTermResult.value);
        
        free(result.value);
        result.value = newValue;
        result.position = nextTermResult.position;
        
        free(opResult.value);
        free(nextTermResult.value);
    }
    
    return result;
}

// 解析项
ParseResult parseTerm(PEGParser* parser) {
    ParseResult result = {0, NULL, parser->position};
    
    // 尝试解析因子
    ParseResult factorResult = parseFactor(parser);
    if (!factorResult.success) {
        return result;
    }
    
    result = factorResult;
    
    // 尝试解析操作符和下一个因子
    while (parser->position < parser->length) {
        ParseResult opResult = parseOperator(parser);
        if (!opResult.success) {
            break;
        }
        
        ParseResult nextFactorResult = parseFactor(parser);
        if (!nextFactorResult.success) {
            parser->position = result.position; // 回滚
            break;
        }
        
        // 创建新的结果
        char* newValue = (char*)malloc(strlen(result.value) + strlen(opResult.value) + strlen(nextFactorResult.value) + 3);
        sprintf(newValue, "(%s %s %s)", result.value, opResult.value, nextFactorResult.value);
        
        free(result.value);
        result.value = newValue;
        result.position = nextFactorResult.position;
        
        free(opResult.value);
        free(nextFactorResult.value);
    }
    
    return result;
}

// 解析因子
ParseResult parseFactor(PEGParser* parser) {
    ParseResult result = {0, NULL, parser->position};
    
    // 尝试解析数字
    ParseResult numberResult = parseNumber(parser);
    if (numberResult.success) {
        return numberResult;
    }
    
    // 尝试解析标识符
    ParseResult identifierResult = parseIdentifier(parser);
    if (identifierResult.success) {
        return identifierResult;
    }
    
    // 尝试解析括号表达式
    if (parser->position < parser->length && parser->input[parser->position] == '(') {
        parser->position++; // 跳过 '('
        
        ParseResult exprResult = parseExpression(parser);
        if (exprResult.success && 
            parser->position < parser->length && 
            parser->input[parser->position] == ')') {
            parser->position++; // 跳过 ')'
            
            result.success = 1;
            result.position = parser->position;
            result.value = exprResult.value;
            return result;
        }
    }
    
    return result;
}

// 清理资源
void freePEGParser(PEGParser* parser) {
    if (parser) {
        free(parser->input);
        
        PEGRule* rule = parser->rules;
        while (rule) {
            PEGRule* next = rule->next;
            free(rule->name);
            free(rule->pattern);
            free(rule);
            rule = next;
        }
        
        free(parser);
    }
}

int main() {
    printf("Packrat PEG解析器示例\n");
    printf("===========================================\n");
    
    const char* input = "1 + 2 * 3";
    PEGParser* parser = createPEGParser(input);
    
    printf("输入: %s\n", input);
    
    ParseResult result = parseExpression(parser);
    
    if (result.success) {
        printf("解析成功: %s\n", result.value);
    } else {
        printf("解析失败\n");
    }
    
    if (result.value) {
        free(result.value);
    }
    
    freePEGParser(parser);
    return 0;
}
```

```cpp
// 示例：使用LLVM生成代码
#include <llvm/IR/LLVMContext.h>
#include <llvm/IR/Module.h>
#include <llvm/IR/Function.h>
#include <llvm/IR/BasicBlock.h>
#include <llvm/IR/Instructions.h>
#include <llvm/IR/Constants.h>
#include <llvm/IR/Type.h>
#include <llvm/Support/raw_ostream.h>
#include <llvm/Support/TargetSelect.h>
#include <llvm/Target/TargetMachine.h>
#include <llvm/Target/TargetOptions.h>
#include <llvm/Support/FileSystem.h>
#include <llvm/IR/Verifier.h>

using namespace llvm;

class LLVMCodeGenerator {
private:
    LLVMContext context;
    Module* module;
    IRBuilder<>* builder;
    Function* currentFunction;
    BasicBlock* currentBlock;
    
public:
    LLVMCodeGenerator() : builder(nullptr), currentFunction(nullptr), currentBlock(nullptr) {
        // 初始化LLVM
        InitializeAllTargetInfos();
        InitializeAllTargets();
        InitializeAllTargetMCs();
        InitializeAllAsmParsers();
        InitializeAllAsmPrinters();
        
        // 创建模块
        module = new Module("SimpleCalc", context);
        
        // 创建IR构建器
        builder = new IRBuilder<>(context);
        
        cout << "LLVM代码生成器初始化完成" << endl;
    }
    
    ~LLVMCodeGenerator() {
        if (builder) {
            delete builder;
        }
        if (module) {
            delete module;
        }
    }
    
    // 创建主函数
    Function* createMainFunction() {
        cout << "创建主函数" << endl;
        
        // 创建函数类型
        FunctionType* funcType = FunctionType::get(
            Type::getInt32Ty(context), // 返回类型
            false // 无参数
        );
        
        // 创建函数
        Function* func = Function::Create(
            funcType,
            Function::ExternalLinkage,
            "main",
            module
        );
        
        // 创建基本块
        BasicBlock* block = BasicBlock::Create(context, "entry", func);
        builder->SetInsertPoint(block);
        
        currentFunction = func;
        currentBlock = block;
        
        return func;
    }
    
    // 生成加法指令
    Value* generateAdd(Value* left, Value* right) {
        cout << "生成加法指令" << endl;
        return builder->CreateAdd(left, right, "addtmp");
    }
    
    // 生成减法指令
    Value* generateSub(Value* left, Value* right) {
        cout << "生成减法指令" << endl;
        return builder->CreateSub(left, right, "subtmp");
    }
    
    // 生成乘法指令
    Value* generateMul(Value* left, Value* right) {
        cout << "生成乘法指令" << endl;
        return builder->CreateMul(left, right, "multmp");
    }
    
    // 生成除法指令
    Value* generateDiv(Value* left, Value* right) {
        cout << "生成除法指令" << endl;
        return builder->CreateSDiv(left, right, "divtmp");
    }
    
    // 生成整数常量
    Value* generateIntegerConstant(int value) {
        cout << "生成整数常量: " << value << endl;
        return ConstantInt::get(Type::getInt32Ty(context), value);
    }
    
    // 生成返回指令
    void generateReturn(Value* value) {
        cout << "生成返回指令" << endl;
        builder->CreateRet(value);
    }
    
    // 生成返回0指令
    void generateReturnZero() {
        cout << "生成返回0指令" << endl;
        Value* zero = generateIntegerConstant(0);
        builder->CreateRet(zero);
    }
    
    // 验证模块
    bool verifyModule() {
        cout << "验证LLVM模块" << endl;
        
        string error;
        raw_string_ostream errorStream(error);
        
        if (verifyModule(*module, &errorStream)) {
            cout << "模块验证失败: " << error << endl;
            return false;
        }
        
        cout << "模块验证成功" << endl;
        return true;
    }
    
    // 打印IR
    void printIR() {
        cout << "LLVM IR代码:" << endl;
        cout << "===========================================" << endl;
        module->print(outs(), nullptr);
        cout << endl;
    }
    
    // 生成目标代码
    bool generateTargetCode(const string& filename) {
        cout << "生成目标代码: " << filename << endl;
        
        // 获取目标三元组
        string targetTriple = sys::getDefaultTargetTriple();
        module->setTargetTriple(targetTriple);
        
        // 获取目标信息
        string error;
        const Target* target = TargetRegistry::lookupTarget(targetTriple, error);
        if (!target) {
            cout << "无法找到目标: " << error << endl;
            return false;
        }
        
        // 创建目标机器
        TargetOptions options;
        TargetMachine* targetMachine = target->createTargetMachine(
            targetTriple, "generic", "", options, Reloc::PIC_);
        
        module->setDataLayout(targetMachine->createDataLayout());
        
        // 创建输出文件
        error_code ec;
        raw_fd_ostream dest(filename, ec, sys::fs::OF_None);
        if (ec) {
            cout << "无法创建输出文件: " << ec.message() << endl;
            return false;
        }
        
        // 生成目标代码
        legacy::PassManager pass;
        auto fileType = CGFT_ObjectFile;
        
        if (targetMachine->addPassesToEmitFile(pass, dest, nullptr, fileType)) {
            cout << "无法生成目标代码" << endl;
            return false;
        }
        
        pass.run(*module);
        dest.flush();
        
        cout << "目标代码生成完成" << endl;
        return true;
    }
    
    // 获取模块
    Module* getModule() const {
        return module;
    }
};

int main() {
    cout << "LLVM代码生成器示例" << endl;
    cout << "===========================================" << endl;
    
    LLVMCodeGenerator generator;
    
    // 创建主函数
    Function* mainFunc = generator.createMainFunction();
    
    // 生成表达式: 1 + 2 * 3
    Value* one = generator.generateIntegerConstant(1);
    Value* two = generator.generateIntegerConstant(2);
    Value* three = generator.generateIntegerConstant(3);
    
    Value* mulResult = generator.generateMul(two, three);
    Value* addResult = generator.generateAdd(one, mulResult);
    
    // 生成返回指令
    generator.generateReturn(addResult);
    
    // 验证模块
    if (generator.verifyModule()) {
        // 打印IR
        generator.printIR();
        
        // 生成目标代码
        generator.generateTargetCode("output.o");
```
