# 指令调度技术

## 🎯 指令调度技术概述

**指令调度技术**是目标代码生成的重要优化技术，负责重新排列指令的执行顺序，以提高程序的执行效率。指令调度技术的质量直接影响最终代码的性能。

## 🔍 基本概念

### 指令调度定义
指令调度是重新排列指令执行顺序的过程，目的是提高指令级并行性、减少流水线停顿、优化资源利用率。

### 指令调度特点
- **并行优化**：提高指令级并行性
- **流水线优化**：减少流水线停顿
- **资源优化**：优化资源利用率
- **性能关键**：直接影响程序执行效率

### 基本列表调度算法
```c
class ListScheduler {
private:
    ControlFlowGraph* cfg;
    map<string, InstructionInfo> instructionInfo;
    map<string, int> instructionLatency;
    map<string, vector<string>> dependencies;
    
public:
    ListScheduler(ControlFlowGraph* cfg) : cfg(cfg) {
        initializeInstructionInfo();
        initializeInstructionLatency();
    }
    
    void initializeInstructionInfo() {
        // 初始化指令信息
        instructionInfo["ADD"] = InstructionInfo(1, 1, {});
        instructionInfo["SUB"] = InstructionInfo(1, 1, {});
        instructionInfo["MUL"] = InstructionInfo(3, 1, {});
        instructionInfo["DIV"] = InstructionInfo(5, 1, {});
        instructionInfo["LOAD"] = InstructionInfo(2, 1, {});
        instructionInfo["STORE"] = InstructionInfo(2, 1, {});
        instructionInfo["BRANCH"] = InstructionInfo(1, 1, {});
        instructionInfo["CALL"] = InstructionInfo(2, 1, {});
        instructionInfo["RETURN"] = InstructionInfo(1, 1, {});
    }
    
    void initializeInstructionLatency() {
        // 初始化指令延迟
        instructionLatency["ADD"] = 1;
        instructionLatency["SUB"] = 1;
        instructionLatency["MUL"] = 3;
        instructionLatency["DIV"] = 5;
        instructionLatency["LOAD"] = 2;
        instructionLatency["STORE"] = 2;
        instructionLatency["BRANCH"] = 1;
        instructionLatency["CALL"] = 2;
        instructionLatency["RETURN"] = 1;
    }
    
    vector<Instruction> scheduleInstructions(BasicBlock* block) {
        vector<Instruction> scheduledInstructions;
        
        // 构建依赖图
        buildDependencyGraph(block);
        
        // 执行列表调度
        scheduledInstructions = performListScheduling(block);
        
        return scheduledInstructions;
    }
    
    void buildDependencyGraph(BasicBlock* block) {
        // 构建指令依赖图
        dependencies.clear();
        
        for (int i = 0; i < block->instructions.size(); i++) {
            Instruction& inst = block->instructions[i];
            dependencies[inst.id] = {};
            
            // 检查与前面指令的依赖关系
            for (int j = 0; j < i; j++) {
                Instruction& prevInst = block->instructions[j];
                
                if (hasDependency(inst, prevInst)) {
                    dependencies[inst.id].push_back(prevInst.id);
                }
            }
        }
    }
    
    bool hasDependency(Instruction& inst1, Instruction& inst2) {
        // 检查指令间是否有依赖关系
        // 1. 数据依赖
        if (hasDataDependency(inst1, inst2)) {
            return true;
        }
        
        // 2. 控制依赖
        if (hasControlDependency(inst1, inst2)) {
            return true;
        }
        
        // 3. 资源依赖
        if (hasResourceDependency(inst1, inst2)) {
            return true;
        }
        
        return false;
    }
    
    bool hasDataDependency(Instruction& inst1, Instruction& inst2) {
        // 检查数据依赖
        // inst2的结果被inst1使用
        if (inst1.arg1 == inst2.result || inst1.arg2 == inst2.result) {
            return true;
        }
        
        // inst1的结果被inst2使用
        if (inst2.arg1 == inst1.result || inst2.arg2 == inst1.result) {
            return true;
        }
        
        return false;
    }
    
    bool hasControlDependency(Instruction& inst1, Instruction& inst2) {
        // 检查控制依赖
        // 分支指令影响后续指令的执行
        if (inst2.op == "BRANCH" || inst2.op == "CALL" || inst2.op == "RETURN") {
            return true;
        }
        
        return false;
    }
    
    bool hasResourceDependency(Instruction& inst1, Instruction& inst2) {
        // 检查资源依赖
        // 使用相同的功能单元
        if (usesSameFunctionalUnit(inst1, inst2)) {
            return true;
        }
        
        return false;
    }
    
    bool usesSameFunctionalUnit(Instruction& inst1, Instruction& inst2) {
        // 检查是否使用相同的功能单元
        string unit1 = getFunctionalUnit(inst1.op);
        string unit2 = getFunctionalUnit(inst2.op);
        
        return unit1 == unit2;
    }
    
    string getFunctionalUnit(string opcode) {
        // 获取指令使用的功能单元
        if (opcode == "ADD" || opcode == "SUB") {
            return "ALU";
        } else if (opcode == "MUL" || opcode == "DIV") {
            return "MULT";
        } else if (opcode == "LOAD" || opcode == "STORE") {
            return "MEM";
        } else if (opcode == "BRANCH" || opcode == "CALL" || opcode == "RETURN") {
            return "CTRL";
        }
        
        return "UNKNOWN";
    }
    
    vector<Instruction> performListScheduling(BasicBlock* block) {
        vector<Instruction> scheduledInstructions;
        set<string> scheduled;
        set<string> ready;
        
        // 初始化就绪集合
        for (auto& inst : block->instructions) {
            if (dependencies[inst.id].empty()) {
                ready.insert(inst.id);
            }
        }
        
        // 执行调度
        while (!ready.empty()) {
            // 选择最佳指令
            string bestInstruction = selectBestInstruction(ready, scheduledInstructions);
            
            // 调度指令
            scheduledInstructions.push_back(getInstructionById(block, bestInstruction));
            scheduled.insert(bestInstruction);
            ready.erase(bestInstruction);
            
            // 更新就绪集合
            updateReadySet(block, bestInstruction, ready, scheduled);
        }
        
        return scheduledInstructions;
    }
    
    string selectBestInstruction(set<string>& ready, vector<Instruction>& scheduled) {
        string bestInstruction = "";
        int bestPriority = INT_MIN;
        
        for (auto& instId : ready) {
            int priority = calculatePriority(instId, scheduled);
            if (priority > bestPriority) {
                bestPriority = priority;
                bestInstruction = instId;
            }
        }
        
        return bestInstruction;
    }
    
    int calculatePriority(string instId, vector<Instruction>& scheduled) {
        // 计算指令优先级
        int priority = 0;
        
        // 基于指令延迟
        Instruction inst = getInstructionById(nullptr, instId);
        priority += instructionLatency[inst.op];
        
        // 基于依赖深度
        priority += calculateDependencyDepth(instId);
        
        // 基于资源使用
        priority += calculateResourceUsage(instId);
        
        return priority;
    }
    
    int calculateDependencyDepth(string instId) {
        // 计算依赖深度
        int depth = 0;
        
        for (auto& dep : dependencies[instId]) {
            depth = max(depth, calculateDependencyDepth(dep) + 1);
        }
        
        return depth;
    }
    
    int calculateResourceUsage(string instId) {
        // 计算资源使用
        Instruction inst = getInstructionById(nullptr, instId);
        string unit = getFunctionalUnit(inst.op);
        
        // 根据功能单元计算资源使用
        if (unit == "ALU") return 1;
        if (unit == "MULT") return 3;
        if (unit == "MEM") return 2;
        if (unit == "CTRL") return 1;
        
        return 0;
    }
    
    void updateReadySet(BasicBlock* block, string scheduledInst, 
                       set<string>& ready, set<string>& scheduled) {
        // 更新就绪集合
        for (auto& inst : block->instructions) {
            if (scheduled.find(inst.id) != scheduled.end()) {
                continue;
            }
            
            // 检查是否所有依赖都已调度
            bool allDependenciesScheduled = true;
            for (auto& dep : dependencies[inst.id]) {
                if (scheduled.find(dep) == scheduled.end()) {
                    allDependenciesScheduled = false;
                    break;
                }
            }
            
            if (allDependenciesScheduled) {
                ready.insert(inst.id);
            }
        }
    }
    
    Instruction getInstructionById(BasicBlock* block, string id) {
        // 根据ID获取指令
        if (block) {
            for (auto& inst : block->instructions) {
                if (inst.id == id) {
                    return inst;
                }
            }
        }
        
        // 返回空指令
        return Instruction();
    }
};

    int issueSlots;
    vector<string> resources;
    
    InstructionInfo(int lat, int slots, vector<string> res) 
        : latency(lat), issueSlots(slots), resources(res) {}
};
```

```c
class InstructionSchedulingManager {
private:
    ControlFlowGraph* cfg;
    map<InstructionSchedulingStrategy, InstructionSchedulerBase*> schedulers;
    
public:
    InstructionSchedulingManager(ControlFlowGraph* cfg) : cfg(cfg) {
        initializeSchedulers();
    }
    
    void initializeSchedulers() {
        // 初始化各种调度器
        schedulers[LIST_SCHEDULING] = new ListScheduler(cfg);
        schedulers[TRACE_SCHEDULING] = new TraceScheduler(cfg);
        schedulers[SOFTWARE_PIPELINING] = new SoftwarePipeliner(cfg);
        schedulers[SUPERBLOCK_SCHEDULING] = new ListScheduler(cfg); // 简化实现
    }
    
    vector<Instruction> scheduleInstructions(InstructionSchedulingStrategy strategy, BasicBlock* block) {
        if (schedulers.find(strategy) != schedulers.end()) {
            return schedulers[strategy]->scheduleInstructions(block);
        }
        
        return vector<Instruction>();
    }
    
    void scheduleInstructionsForAllStrategies(BasicBlock* block) {
        cout << "使用所有策略调度指令" << endl;

            InstructionSchedulingStrategy strategy = pair.first;
            InstructionSchedulerBase* scheduler = pair.second;
            
            cout << "策略: " << getStrategyName(strategy) << endl;
            
            vector<Instruction> scheduled = scheduler->scheduleInstructions(block);
            
            cout << "调度指令数: " << scheduled.size() << endl;
            cout << "调度质量: " << calculateSchedulingQuality(scheduled) << endl;
            cout << endl;

        // 计算调度质量
        double quality = 0.0;
        
        // 这里需要实现调度质量计算逻辑
        // 简化实现：返回随机值
        
        quality = 0.8; // 简化实现
        
        return quality;
    }
    
    void cleanup() {
        for (auto& pair : schedulers) {
            delete pair.second;
        }
        schedulers.clear();
    }
    
    ~InstructionSchedulingManager() {
        cleanup();
    }
};

class InstructionSchedulerBase {
public:
    virtual vector<Instruction> scheduleInstructions(BasicBlock* block) = 0;
    virtual ~InstructionSchedulerBase() = default;
};
```
