# 指令选择算法

## 🎯 指令选择算法概述

**指令选择算法**是目标代码生成的核心技术，负责将中间代码转换为目标机器的指令序列。指令选择的质量直接影响最终代码的执行效率和资源使用。

## 🔍 基本概念

### 指令选择定义
指令选择是将中间表示（如三地址码、四元式）转换为目标机器指令的过程，需要为每个中间代码操作选择最合适的目标指令。

### 指令选择特点
- **机器相关**：需要了解目标机器的指令集和特性
- **优化关键**：是代码生成优化的基础
- **资源考虑**：需要考虑寄存器、内存等资源限制
- **性能影响**：直接影响最终程序的执行效率

            InstructionTemplate("divi %dst, %src1, %imm", {REGISTER, REGISTER, IMMEDIATE}, 4),
            InstructionTemplate("div %dst, %src1, %src2", {REGISTER, REGISTER, REGISTER}, 6)
        };
        
        // 内存操作模板
        instructionTemplates["LOAD"] = {
            InstructionTemplate("ld %dst, %offset(%base)", {REGISTER, IMMEDIATE, REGISTER}, 2),
            InstructionTemplate("ld %dst, (%base)", {REGISTER, REGISTER}, 1),
            InstructionTemplate("ld %dst, %offset(%base)", {REGISTER, IMMEDIATE, REGISTER}, 3)
        };
        
        instructionTemplates["STORE"] = {
            InstructionTemplate("st %src, %offset(%base)", {REGISTER, IMMEDIATE, REGISTER}, 2),
            InstructionTemplate("st %src, (%base)", {REGISTER, REGISTER}, 1),

        // 控制流模板

            InstructionTemplate("b %label", {LABEL}, 1),

            InstructionTemplate("bne %src1, %src2, %label", {REGISTER, REGISTER, LABEL}, 2)
        };
        
        instructionTemplates["CALL"] = {
            InstructionTemplate("call %func", {FUNCTION}, 2),
            InstructionTemplate("jal %func", {FUNCTION}, 1),
            InstructionTemplate("call %func", {FUNCTION}, 3)
        };
        
        instructionTemplates["RETURN"] = {
            InstructionTemplate("ret", {}, 1),
            InstructionTemplate("jr %ra", {REGISTER}, 1),
            InstructionTemplate("ret", {}, 2)
        };
    }
    
    void initializeInstructionCosts() {
        // 初始化指令代价
        instructionCosts["ADD"] = 1;
        instructionCosts["SUB"] = 1;
        instructionCosts["MUL"] = 3;
        instructionCosts["DIV"] = 5;
        instructionCosts["LOAD"] = 2;
        instructionCosts["STORE"] = 2;

        vector<SelectedInstruction> selectedInstructions;
        
        for (auto& inst : intermediateCode) {
            SelectedInstruction selected = selectInstruction(inst);

        return selectedInstructions;

        return selected;
    }

    SelectedInstruction selectReturnInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "RETURN";
        selected.operands = {};
        
        vector<InstructionTemplate>& templates = instructionTemplates["RETURN"];
        InstructionTemplate bestTemplate = selectBestTemplate(templates, inst);
        selected.template = bestTemplate;
        
        return selected;
    }
    
    SelectedInstruction selectDefaultInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        selected.opcode = "UNKNOWN";
        selected.operands = {inst.result, inst.arg1, inst.arg2};
        selected.template = InstructionTemplate("", {}, 0);
        return selected;
    }
    
    InstructionTemplate selectBestTemplate(vector<InstructionTemplate>& templates, 
                                          IntermediateInstruction& inst) {
        InstructionTemplate bestTemplate = templates[0];
        int bestCost = calculateTemplateCost(templates[0], inst);
        
        for (int i = 1; i < templates.size(); i++) {
            int cost = calculateTemplateCost(templates[i], inst);
            if (cost < bestCost) {
                bestCost = cost;
                bestTemplate = templates[i];
            }
        }
        
        return bestTemplate;
    }
    
    int calculateTemplateCost(InstructionTemplate& template, IntermediateInstruction& inst) {
        int cost = template.baseCost;
        
        // 根据操作数类型调整代价
        for (int i = 0; i < template.operandTypes.size(); i++) {
            OperandType type = template.operandTypes[i];
            
            if (type == REGISTER) {
                cost += 0; // 寄存器操作数代价为0
            } else if (type == IMMEDIATE) {
                cost += 1; // 立即数操作数代价为1
            } else if (type == LABEL) {
                cost += 2; // 标签操作数代价为2
            } else if (type == FUNCTION) {
                cost += 3; // 函数操作数代价为3
            }
        }
        
        return cost;
    }
};

class InstructionTemplate {
public:
    string format;
    vector<OperandType> operandTypes;
    int baseCost;
    
    InstructionTemplate(string fmt, vector<OperandType> types, int cost) 
        : format(fmt), operandTypes(types), baseCost(cost) {}
};

    void initializePatternCosts() {
        // 初始化模式代价
        patternCosts["ADD_REG_REG"] = 1;
        patternCosts["ADD_REG_IMM"] = 1;
        patternCosts["SUB_REG_REG"] = 1;
        patternCosts["SUB_REG_IMM"] = 1;
        patternCosts["MUL_REG_REG"] = 3;
        patternCosts["MUL_REG_IMM"] = 2;
        patternCosts["DIV_REG_REG"] = 5;
        patternCosts["DIV_REG_IMM"] = 4;
        patternCosts["LOAD_REG_OFFSET"] = 2;
        patternCosts["LOAD_REG_REG"] = 1;
        patternCosts["STORE_REG_OFFSET"] = 2;
        patternCosts["STORE_REG_REG"] = 1;
        patternCosts["BRANCH_LABEL"] = 1;
        patternCosts["BRANCH_COND"] = 2;
        patternCosts["CALL_FUNC"] = 2;
        patternCosts["RETURN"] = 1;
    }
    
    vector<SelectedInstruction> selectInstructions(vector<IntermediateInstruction>& intermediateCode) {
        vector<SelectedInstruction> selectedInstructions;
        
        for (auto& inst : intermediateCode) {
            SelectedInstruction selected = selectInstruction(inst);
            selectedInstructions.push_back(selected);
        }
        
        return selectedInstructions;
    }
    
    SelectedInstruction selectInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        
        // 构建指令树
        TreePattern instructionTree = buildInstructionTree(inst);
        
        // 匹配最佳模式
        string bestPattern = matchBestPattern(instructionTree);
        
        // 生成选择的指令
        selected = generateSelectedInstruction(inst, bestPattern);
        
        return selected;
    }
    
    TreePattern buildInstructionTree(IntermediateInstruction& inst) {
        TreePattern tree;
        tree.opcode = inst.op;
        tree.operands = {};
        
        // 添加操作数
        if (!inst.result.empty()) {
            tree.operands.push_back(TreePattern("REG", {}));
        }
        if (!inst.arg1.empty()) {
            tree.operands.push_back(TreePattern("REG", {}));
        }
        if (!inst.arg2.empty()) {
            tree.operands.push_back(TreePattern("REG", {}));
        }
        
        return tree;
    }
    
    string matchBestPattern(TreePattern& instructionTree) {
        string bestPattern = "";
        int bestCost = INT_MAX;
        
        for (auto& pair : instructionPatterns) {
            string patternName = pair.first;
            TreePattern pattern = pair.second;
            
            if (matchesPattern(instructionTree, pattern)) {
                int cost = patternCosts[patternName];
                if (cost < bestCost) {
                    bestCost = cost;
                    bestPattern = patternName;
                }
            }
        }
        
        return bestPattern;
    }
    
    bool matchesPattern(TreePattern& instructionTree, TreePattern& pattern) {
        // 检查操作码是否匹配
        if (instructionTree.opcode != pattern.opcode) {
            return false;
        }
        
        // 检查操作数数量是否匹配
        if (instructionTree.operands.size() != pattern.operands.size()) {
            return false;
        }
        
        // 检查操作数类型是否匹配
        for (int i = 0; i < instructionTree.operands.size(); i++) {
            if (instructionTree.operands[i].opcode != pattern.operands[i].opcode) {
                return false;
            }
        }
        
        return true;
    }
    
    SelectedInstruction generateSelectedInstruction(IntermediateInstruction& inst, string patternName) {
        SelectedInstruction selected;
        
        if (patternName.empty()) {
            selected.opcode = "UNKNOWN";
            selected.operands = {inst.result, inst.arg1, inst.arg2};
            selected.template = InstructionTemplate("", {}, 0);
            return selected;
        }

        } else if (patternName == "ADD_REG_IMM") {

        } else if (patternName == "SUB_REG_IMM") {

    void initializeRulePriorities() {
        // 初始化规则优先级
        rulePriorities["ADD_REG_REG"] = 1;
        rulePriorities["ADD_REG_IMM"] = 2;
        rulePriorities["SUB_REG_REG"] = 1;
        rulePriorities["SUB_REG_IMM"] = 2;
        rulePriorities["MUL_REG_REG"] = 1;
        rulePriorities["MUL_REG_IMM"] = 2;
        rulePriorities["DIV_REG_REG"] = 1;
        rulePriorities["DIV_REG_IMM"] = 2;
        rulePriorities["LOAD_REG_OFFSET"] = 2;
        rulePriorities["LOAD_REG_REG"] = 1;
        rulePriorities["STORE_REG_OFFSET"] = 2;
        rulePriorities["STORE_REG_REG"] = 1;
        rulePriorities["BRANCH_LABEL"] = 1;
        rulePriorities["BRANCH_COND"] = 2;
        rulePriorities["CALL_FUNC"] = 1;
        rulePriorities["RETURN"] = 1;
    }
    
    vector<SelectedInstruction> selectInstructions(vector<IntermediateInstruction>& intermediateCode) {
        vector<SelectedInstruction> selectedInstructions;
        
        for (auto& inst : intermediateCode) {
            SelectedInstruction selected = selectInstruction(inst);
            selectedInstructions.push_back(selected);
        }
        
        return selectedInstructions;
    }
    
    SelectedInstruction selectInstruction(IntermediateInstruction& inst) {
        SelectedInstruction selected;
        
        // 应用选择规则
        SelectionRule bestRule = applySelectionRules(inst);
        
        // 生成选择的指令
        selected = generateSelectedInstruction(inst, bestRule);
        
        return selected;
    }
    
    SelectionRule applySelectionRules(IntermediateInstruction& inst) {
        SelectionRule bestRule;
        int bestPriority = INT_MAX;
        
        for (auto& rule : selectionRules) {
            if (matchesRule(inst, rule)) {
                int priority = rulePriorities[rule.name];
                if (priority < bestPriority) {
                    bestPriority = priority;
                    bestRule = rule;
                }
            }
        }
        
        return bestRule;
    }
    
    bool matchesRule(IntermediateInstruction& inst, SelectionRule& rule) {
        // 检查操作码是否匹配
        if (inst.op != rule.opcode) {
            return false;
        }
        
        // 检查操作数类型是否匹配
        if (inst.result.empty() && rule.operandTypes.size() > 0 && rule.operandTypes[0] == REGISTER) {
            return false;
        }
        if (inst.arg1.empty() && rule.operandTypes.size() > 1 && rule.operandTypes[1] == REGISTER) {
            return false;
        }
        if (inst.arg2.empty() && rule.operandTypes.size() > 2 && rule.operandTypes[2] == REGISTER) {
            return false;
        }
        
        return true;
    }
    
    SelectedInstruction generateSelectedInstruction(IntermediateInstruction& inst, SelectionRule& rule) {
        SelectedInstruction selected;
        
        selected.opcode = rule.opcode;
        selected.operands = {inst.result, inst.arg1, inst.arg2};
        selected.template = InstructionTemplate(rule.format, rule.operandTypes, rule.cost);
        
        return selected;
    }
};

class SelectionRule {
public:
    string name;
    string opcode;
    vector<OperandType> operandTypes;
    string format;
    int cost;
    string description;
    
    SelectionRule(string n, string op, vector<OperandType> types, string fmt, int c, string desc) 
        : name(n), opcode(op), operandTypes(types), format(fmt), cost(c), description(desc) {}
};

class InstructionSelectionManager {
private:
    map<InstructionSelectionStrategy, InstructionSelectorBase*> selectors;
    TargetMachine* targetMachine;
    
public:
    InstructionSelectionManager(TargetMachine* machine) : targetMachine(machine) {
        initializeSelectors();
    }
    
    void initializeSelectors() {
        // 初始化各种选择器
        selectors[TEMPLATE_MATCHING] = new TemplateMatchingInstructionSelector(targetMachine);
        selectors[TREE_PATTERN_MATCHING] = new TreePatternMatchingInstructionSelector(targetMachine);
        selectors[RULE_BASED_SELECTION] = new RuleBasedInstructionSelector(targetMachine);
    }
    
    vector<SelectedInstruction> selectInstructions(InstructionSelectionStrategy strategy, 
                                                  vector<IntermediateInstruction>& intermediateCode) {
        if (selectors.find(strategy) != selectors.end()) {
            return selectors[strategy]->selectInstructions(intermediateCode);
        }
        
        return vector<SelectedInstruction>();
    }
    
    void selectInstructionsForAllStrategies(vector<IntermediateInstruction>& intermediateCode) {
        cout << "使用所有策略选择指令" << endl;
        cout << "===========================================" << endl;
        
        for (auto& pair : selectors) {
            InstructionSelectionStrategy strategy = pair.first;
            InstructionSelectorBase* selector = pair.second;
            
            cout << "策略: " << getStrategyName(strategy) << endl;
            
            vector<SelectedInstruction> selected = selector->selectInstructions(intermediateCode);
            
            cout << "选择指令数: " << selected.size() << endl;
            cout << "总代价: " << calculateTotalCost(selected) << endl;
            cout << endl;
        }
    }
    
    string getStrategyName(InstructionSelectionStrategy strategy) {
        switch (strategy) {
            case TEMPLATE_MATCHING: return "模板匹配";
            case TREE_PATTERN_MATCHING: return "树模式匹配";
            case RULE_BASED_SELECTION: return "基于规则的选择";
            case COST_BASED_SELECTION: return "基于代价的选择";
            default: return "未知策略";
        }
    }
    
    int calculateTotalCost(vector<SelectedInstruction>& instructions) {
        int totalCost = 0;
        for (auto& inst : instructions) {
            totalCost += inst.template.baseCost;
        }
        return totalCost;
    }
    
    void cleanup() {
        for (auto& pair : selectors) {
            delete pair.second;
        }
        selectors.clear();
    }
    
    ~InstructionSelectionManager() {
        cleanup();
    }
};

class InstructionSelectorBase {
public:
    virtual vector<SelectedInstruction> selectInstructions(vector<IntermediateInstruction>& intermediateCode) = 0;
    virtual ~InstructionSelectorBase() = default;
};
