### 1. DFA模拟算法
- **算法描述**：模拟DFA的状态转换过程
- **时间复杂度**：O(n)，其中n是输入字符串长度
- **空间复杂度**：O(1)
- **应用场景**：词法分析器核心算法

```c
// DFA模拟算法实现
class DFASimulator {
private:
    map<pair<int, char>, int> transitions;
    int startState;
    set<int> acceptStates;
    
public:
    bool accepts(const string& input) {
        int currentState = startState;
        
        for (char c : input) {
            auto key = make_pair(currentState, c);
            if (transitions.find(key) != transitions.end()) {
                currentState = transitions[key];
            } else {
                return false;
            }
        }
        
        return acceptStates.find(currentState) != acceptStates.end();
    }
};
```

### 2. NFA模拟算法
- **算法描述**：模拟NFA的状态转换过程
- **时间复杂度**：O(n·m)，其中n是输入长度，m是状态数
- **空间复杂度**：O(m)
- **应用场景**：NFA词法分析器

```c
// NFA模拟算法实现
class NFASimulator {
private:
    map<pair<int, char>, set<int>> transitions;
    int startState;
    set<int> acceptStates;
    
public:
    bool accepts(const string& input) {
        set<int> currentStates = {startState};
        
        for (char c : input) {
            set<int> nextStates;
            for (int state : currentStates) {
                auto key = make_pair(state, c);
                if (transitions.find(key) != transitions.end()) {
                    for (int nextState : transitions[key]) {
                        nextStates.insert(nextState);
                    }
                }
            }
            currentStates = nextStates;
        }
        
        for (int state : currentStates) {
            if (acceptStates.find(state) != acceptStates.end()) {
                return true;
            }
        }
        return false;
    }
};
```

### 3. Thompson构造算法
- **算法描述**：将正则表达式转换为NFA
- **时间复杂度**：O(n)，其中n是正则表达式长度
- **空间复杂度**：O(n)
- **应用场景**：正则表达式到NFA转换

```c
// Thompson构造算法实现
class ThompsonConstructor {
public:
    NFA* construct(const string& regex) {
        stack<NFA*> nfaStack;
        
        for (char c : regex) {
            if (c == 'a' || c == 'b') {
                // 基本字符
                nfaStack.push(createBasicNFA(c));
            } else if (c == '|') {
                // 并集
                NFA* nfa2 = nfaStack.top(); nfaStack.pop();
                NFA* nfa1 = nfaStack.top(); nfaStack.pop();
                nfaStack.push(unionNFA(nfa1, nfa2));
            } else if (c == '*') {
                // 闭包
                NFA* nfa = nfaStack.top(); nfaStack.pop();
                nfaStack.push(closureNFA(nfa));
            } else if (c == '+') {
                // 正闭包
                NFA* nfa = nfaStack.top(); nfaStack.pop();
                nfaStack.push(positiveClosureNFA(nfa));
            } else if (c == '?') {
                // 可选
                NFA* nfa = nfaStack.top(); nfaStack.pop();
                nfaStack.push(optionalNFA(nfa));
            }
        }
        
        return nfaStack.top();
    }
    
private:
    NFA* createBasicNFA(char c) {
        // 创建基本字符的NFA
        return new NFA();
    }
    
    NFA* unionNFA(NFA* nfa1, NFA* nfa2) {
        // 创建并集NFA
        return new NFA();
    }
    
    NFA* closureNFA(NFA* nfa) {
        // 创建闭包NFA
        return new NFA();
    }
    
    NFA* positiveClosureNFA(NFA* nfa) {
        // 创建正闭包NFA

    }
    
    NFA* optionalNFA(NFA* nfa) {
```

### 4. 子集构造算法
- **算法描述**：将NFA转换为DFA
- **时间复杂度**：O(2^n·m)，其中n是NFA状态数，m是字母表大小
- **空间复杂度**：O(2^n)
- **应用场景**：NFA到DFA转换

```c
// 子集构造算法实现
class SubsetConstructor {
public:
    DFA* construct(NFA* nfa) {
        DFA* dfa = new DFA();
        map<set<int>, int> stateMap;
        queue<set<int>> stateQueue;
        
        // 初始状态
        set<int> startState = epsilonClosure(nfa, {nfa->startState});
        stateMap[startState] = 0;
        stateQueue.push(startState);
        dfa->startState = 0;
        
        int nextStateId = 1;
        while (!stateQueue.empty()) {
            set<int> currentState = stateQueue.front();
            stateQueue.pop();
            
            for (char c : nfa->alphabet) {
                set<int> nextState = epsilonClosure(nfa, delta(nfa, currentState, c));
                if (!nextState.empty()) {
                    if (stateMap.find(nextState) == stateMap.end()) {
                        stateMap[nextState] = nextStateId++;
                        stateQueue.push(nextState);
                    }
                    
                    dfa->addTransition(stateMap[currentState], c, stateMap[nextState]);
                }
            }
        }
        
        // 设置接受状态
        for (auto& pair : stateMap) {
            for (int state : pair.first) {
                if (nfa->acceptStates.find(state) != nfa->acceptStates.end()) {
                    dfa->acceptStates.insert(pair.second);
                    break;
                }
            }
        }
        
        return dfa;
    }
    
private:
    set<int> epsilonClosure(NFA* nfa, set<int> states) {
        set<int> closure = states;
        queue<int> stateQueue;
        
        for (int state : states) {
            stateQueue.push(state);
        }
        
        while (!stateQueue.empty()) {
            int state = stateQueue.front();
            stateQueue.pop();
            
            auto key = make_pair(state, '\0'); // ε转换
            if (nfa->transitions.find(key) != nfa->transitions.end()) {
                for (int nextState : nfa->transitions[key]) {
                    if (closure.find(nextState) == closure.end()) {
                        closure.insert(nextState);
                        stateQueue.push(nextState);
                    }
                }
            }
        }
        
        return closure;
    }
    
    set<int> delta(NFA* nfa, set<int> states, char c) {
        set<int> result;
        for (int state : states) {
            auto key = make_pair(state, c);
            if (nfa->transitions.find(key) != nfa->transitions.end()) {
                for (int nextState : nfa->transitions[key]) {
                    result.insert(nextState);
                }
            }
        }
        return result;
    }
};
```

### 5. Hopcroft算法
- **算法描述**：DFA最小化算法
- **时间复杂度**：O(n log n)，其中n是状态数
- **空间复杂度**：O(n)
- **应用场景**：DFA最小化

```c
// Hopcroft算法实现
class HopcroftAlgorithm {
public:
    DFA* minimize(DFA* dfa) {
        // 初始分割：接受状态和非接受状态
        set<set<int>> partitions;
        set<int> acceptStates = dfa->acceptStates;
        set<int> nonAcceptStates;
        
        for (int state : dfa->states) {
            if (acceptStates.find(state) == acceptStates.end()) {
                nonAcceptStates.insert(state);
            }
        }
        
        if (!acceptStates.empty()) {
            partitions.insert(acceptStates);
        }
        if (!nonAcceptStates.empty()) {
            partitions.insert(nonAcceptStates);
        }
        
        // 迭代分割
        bool changed = true;
        while (changed) {
            changed = false;
            set<set<int>> newPartitions;
            
            for (const set<int>& partition : partitions) {
                map<map<char, int>, set<int>> groups;
                
                for (int state : partition) {
                    map<char, int> transitions;
                    for (char c : dfa->alphabet) {
                        auto key = make_pair(state, c);
                        if (dfa->transitions.find(key) != dfa->transitions.end()) {
                            int nextState = dfa->transitions[key];
                            // 找到nextState所属的分区
                            for (const set<int>& p : partitions) {
                                if (p.find(nextState) != p.end()) {
                                    transitions[c] = distance(partitions.begin(), 
                                                           find(partitions.begin(), partitions.end(), p));
                                    break;
                                }
                            }
                        }
                    }
                    groups[transitions].insert(state);
                }
                
                for (auto& pair : groups) {
                    newPartitions.insert(pair.second);
                }
            }
            
            if (newPartitions.size() != partitions.size()) {
                changed = true;
                partitions = newPartitions;
            }
        }
        
        // 构建最小化DFA
        return buildMinimizedDFA(dfa, partitions);
    }
    
private:
    DFA* buildMinimizedDFA(DFA* originalDFA, const set<set<int>>& partitions) {
        DFA* minimizedDFA = new DFA();
        map<set<int>, int> partitionToState;
        int stateId = 0;
        
        // 创建新状态
        for (const set<int>& partition : partitions) {
            partitionToState[partition] = stateId++;
        }
        
        // 设置初始状态
        for (const set<int>& partition : partitions) {
            if (partition.find(originalDFA->startState) != partition.end()) {
                minimizedDFA->startState = partitionToState[partition];
                break;
            }
        }
        
        // 设置接受状态
        for (const set<int>& partition : partitions) {
            for (int state : partition) {
                if (originalDFA->acceptStates.find(state) != originalDFA->acceptStates.end()) {
                    minimizedDFA->acceptStates.insert(partitionToState[partition]);
                    break;
                }
            }
        }
        
        // 构建转换函数
        for (const set<int>& partition : partitions) {
            int fromState = partitionToState[partition];
            int representative = *partition.begin();
            
            for (char c : originalDFA->alphabet) {
                auto key = make_pair(representative, c);
                if (originalDFA->transitions.find(key) != originalDFA->transitions.end()) {
                    int nextState = originalDFA->transitions[key];
                    
                    // 找到nextState所属的分区
                    for (const set<int>& p : partitions) {
                        if (p.find(nextState) != p.end()) {
                            int toState = partitionToState[p];
                            minimizedDFA->addTransition(fromState, c, toState);
                            break;
                        }
                    }
                }
            }
        }
        
        return minimizedDFA;
    }
};
```

## 📊 语法分析算法

### 1. LL(1)分析算法
- **算法描述**：自顶向下语法分析
- **时间复杂度**：O(n)，其中n是输入长度
- **空间复杂度**：O(n)
- **应用场景**：LL(1)文法分析

```c
// LL(1)分析算法实现
class LL1Parser {
private:
    map<pair<string, string>, string> parseTable;
    
public:
    bool parse(const vector<Token>& tokens) {
        stack<string> parseStack;
        parseStack.push("$");
        parseStack.push("S");
        
        int tokenIndex = 0;
        while (!parseStack.empty()) {
            string top = parseStack.top();
            parseStack.pop();
            
            if (top == "$") {
                return tokenIndex == tokens.size();
            } else if (top == tokens[tokenIndex].type) {
                tokenIndex++;
            } else {
                auto key = make_pair(top, tokens[tokenIndex].type);
                if (parseTable.find(key) != parseTable.end()) {
                    string production = parseTable[key];
                    // 将产生式右部逆序压入栈
                    vector<string> rhs = splitProduction(production);
                    for (int i = rhs.size() - 1; i >= 0; i--) {
                        if (rhs[i] != "ε") {
                            parseStack.push(rhs[i]);
                        }
                    }
                } else {
                    return false;
                }
            }
        }
        
        return true;
    }
    
private:
    vector<string> splitProduction(const string& production) {
        vector<string> result;
        stringstream ss(production);
        string token;
        while (ss >> token) {
            result.push_back(token);
        }
        return result;
    }
};
```

### 2. 递归下降分析算法
- **算法描述**：递归函数实现语法分析
- **时间复杂度**：O(n)，其中n是输入长度
- **空间复杂度**：O(n)
- **应用场景**：简单文法分析

```c
// 递归下降分析算法实现
class RecursiveDescentParser {
private:
    vector<Token> tokens;
    int currentToken;
    
public:
    bool parse(const vector<Token>& inputTokens) {
        tokens = inputTokens;
        currentToken = 0;
        return parseS();
    }
    
private:
    bool parseS() {
        // S -> aSb | ε
        if (currentToken < tokens.size() && tokens[currentToken].type == "a") {
            currentToken++; // 匹配 'a'
            if (parseS()) {
                if (currentToken < tokens.size() && tokens[currentToken].type == "b") {
                    currentToken++; // 匹配 'b'
                    return true;
                }
            }
            return false;
        }
        return true; // ε产生式
    }
    
    bool parseA() {
        // A -> aA | ε
        if (currentToken < tokens.size() && tokens[currentToken].type == "a") {
            currentToken++; // 匹配 'a'
            return parseA();
        }
        return true; // ε产生式
    }
};
```

### 3. LR(1)分析算法
- **算法描述**：自底向上语法分析
- **时间复杂度**：O(n)，其中n是输入长度
- **空间复杂度**：O(n)
- **应用场景**：LR(1)文法分析

```c
// LR(1)分析算法实现
class LR1Parser {
private:
    map<pair<int, string>, string> actionTable;
    map<pair<int, string>, int> gotoTable;
    
public:
    bool parse(const vector<Token>& tokens) {
        stack<int> stateStack;
        stack<string> symbolStack;
        stateStack.push(0);
        
        int tokenIndex = 0;
        while (tokenIndex < tokens.size()) {
            int currentState = stateStack.top();
            string currentToken = tokens[tokenIndex].type;
            
            auto key = make_pair(currentState, currentToken);
            if (actionTable.find(key) != actionTable.end()) {
                string action = actionTable[key];
                
                if (action[0] == 's') {
                    // 移进
                    int nextState = stoi(action.substr(1));
                    stateStack.push(nextState);
                    symbolStack.push(currentToken);
                    tokenIndex++;
                } else if (action[0] == 'r') {
                    // 归约
                    int production = stoi(action.substr(1));
                    if (reduce(production, stateStack, symbolStack)) {
                        // 归约成功，继续
                    } else {
                        return false;
                    }
                } else if (action == "accept") {
                    return true;
                } else {
                    return false;
                }
            } else {
                return false;
            }
        }
        
        return false;
    }
    
private:
    bool reduce(int production, stack<int>& stateStack, stack<string>& symbolStack) {
        // 根据产生式进行归约
        switch (production) {
            case 1: // S -> A
                if (symbolStack.size() >= 1) {
                    symbolStack.pop();
                    symbolStack.push("S");
                    
                    int currentState = stateStack.top();
                    auto key = make_pair(currentState, "S");
                    if (gotoTable.find(key) != gotoTable.end()) {
                        stateStack.push(gotoTable[key]);
                        return true;
                    }
                }
                return false;
            case 2: // A -> a
                if (symbolStack.size() >= 1) {
                    symbolStack.pop();
                    symbolStack.push("A");
                    
                    int currentState = stateStack.top();
                    auto key = make_pair(currentState, "A");
                    if (gotoTable.find(key) != gotoTable.end()) {
                        stateStack.push(gotoTable[key]);
                        return true;
                    }
                }
                return false;
            default:
                return false;
        }
    }
};
```

- **时间复杂度**：O(n)，其中n是AST节点数
- **空间复杂度**：O(n)

```c
// 类型检查算法实现
class TypeChecker {
public:
    string checkType(ASTNode* node) {
        if (node->type == "number") {
            return "int";
        } else if (node->type == "string") {
            return "string";
        } else if (node->type == "add") {
            string leftType = checkType(node->children[0]);
            string rightType = checkType(node->children[1]);
            
            if (leftType == "int" && rightType == "int") {
                return "int";
            } else if (leftType == "string" && rightType == "string") {
                return "string";
            } else {
                return "error";
            }
        } else if (node->type == "assign") {
            string leftType = checkType(node->children[0]);
            string rightType = checkType(node->children[1]);
            
            if (leftType == rightType) {
                return leftType;
            } else {
                return "error";
            }
        }
        
        return "unknown";
    }
};
```

### 2. 死代码消除算法
- **算法描述**：删除不可达代码
- **时间复杂度**：O(n)，其中n是基本块数
- **空间复杂度**：O(n)
- **应用场景**：全局优化

```c
// 死代码消除算法实现
class DeadCodeElimination {
public:
    void eliminateDeadCode(Program* program) {
        set<BasicBlock*> reachableBlocks;
        queue<BasicBlock*> workList;
        
        // 从入口块开始
        workList.push(program->entryBlock);
        reachableBlocks.insert(program->entryBlock);
        
        while (!workList.empty()) {
            BasicBlock* current = workList.front();
            workList.pop();
            
            // 添加可达的后继块
            for (BasicBlock* successor : current->successors) {
                if (reachableBlocks.find(successor) == reachableBlocks.end()) {
                    reachableBlocks.insert(successor);
                    workList.push(successor);
                }
            }
        }
        
        // 删除不可达块
        for (BasicBlock* block : program->basicBlocks) {
            if (reachableBlocks.find(block) == reachableBlocks.end()) {
                program->removeBlock(block);
            }
        }
    }
};
```

### 3. 公共子表达式消除算法
- **算法描述**：重用计算结果
- **时间复杂度**：O(n²)，其中n是表达式数
- **空间复杂度**：O(n)
- **应用场景**：局部优化

```c
// 公共子表达式消除算法实现
class CommonSubexpressionElimination {
public:
    void eliminateCommonSubexpressions(BasicBlock* block) {
        map<string, string> expressionToVariable;
        
        for (auto& instruction : block->instructions) {
            if (instruction.type == "add" || instruction.type == "sub" || 
                instruction.type == "mul" || instruction.type == "div") {
                string expression = instruction.type + " " + 
                                  instruction.operands[0].name + " " + 
                                  instruction.operands[1].name;
                
                if (expressionToVariable.find(expression) != expressionToVariable.end()) {
                    // 重用已有变量
                    instruction.type = "assign";
                    instruction.operands[0].name = instruction.result;
                    instruction.operands[1].name = expressionToVariable[expression];
                } else {
                    // 记录新表达式
                    expressionToVariable[expression] = instruction.result;
                }
            }
        }
    }
};
```

```c
class InstructionSelector {
public:
    string selectInstruction(const string& operation, 
                           const vector<string>& operands) {
        if (operation == "add" && operands.size() == 2) {
            return "add " + operands[0] + ", " + operands[1];

    }
```

### 2. 寄存器分配算法
- **算法描述**：分配寄存器给变量
- **时间复杂度**：O(n²)，其中n是变量数
- **空间复杂度**：O(n)
- **应用场景**：目标代码生成

```c
// 寄存器分配算法实现
class RegisterAllocator {
private:
    set<string> availableRegisters;
    map<string, string> variableToRegister;
    map<string, set<string>> interferenceGraph;
    
public:
    string allocateRegister(const string& variable) {
        if (variableToRegister.find(variable) != variableToRegister.end()) {
            return variableToRegister[variable];
        }
        
        // 找到不与变量冲突的寄存器
        for (const string& reg : availableRegisters) {
            bool conflict = false;
            for (const string& neighbor : interferenceGraph[variable]) {
                if (variableToRegister.find(neighbor) != variableToRegister.end() &&
                    variableToRegister[neighbor] == reg) {
                    conflict = true;
                    break;
                }
            }
            
            if (!conflict) {
                variableToRegister[variable] = reg;
                availableRegisters.erase(reg);
                return reg;
            }
        }
        
        return ""; // 需要溢出到内存
    }
    
    void deallocateRegister(const string& variable) {
        auto it = variableToRegister.find(variable);
        if (it != variableToRegister.end()) {
            availableRegisters.insert(it->second);
            variableToRegister.erase(it);
        }
    }
};
```
