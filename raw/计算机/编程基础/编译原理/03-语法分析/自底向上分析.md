# 自底向上分析

## 🎯 自底向上分析概述

**自底向上分析**是从输入字符串开始，逐步归约到开始符号的语法分析方法。它通过移进和归约的方式构建语法分析树。

## 🔍 基本概念

### 归约过程
从输入字符串开始，逐步应用产生式，直到得到开始符号：

```c
// 文法：E → E + T | T, T → T * F | F, F → (E) | id
// 输入：id + id * id

// 归约过程：
id + id * id
→ F + id * id
→ T + id * id
→ E + id * id
→ E + F * id
→ E + T * id
→ E + T * F
→ E + T
→ E
```

### 移进-归约分析
使用栈和输入缓冲区进行分析：
- **移进**：将输入符号移入栈
- **归约**：将栈顶的符号序列归约为非终结符

## 🔧 移进-归约分析器

### 基本结构
```c
class ShiftReduceParser {
private:
    stack<TokenType> parseStack;
    vector<Token> input;
    int position;
    
public:
    ShiftReduceParser(vector<Token> input) 
        : input(input), position(0) {}
    
    bool parse() {
        parseStack.push(END_OF_INPUT);
        
        while (position < input.size()) {
            Token current = input[position];
            
            // 尝试归约
            if (canReduce()) {
                reduce();
            } else {
                // 移进
                shift();
            }
        }
        
        // 最终归约到开始符号
        return finalReduce();
    }
    
    void shift() {
        parseStack.push(input[position].type);
        position++;
    }
    
    void reduce() {
        // 找到可以归约的产生式
        Production production = findReducibleProduction();
        
        // 从栈中弹出产生式右部的符号
        for (int i = 0; i < production.right.length(); i++) {
            parseStack.pop();
        }
        
        // 将产生式左部压入栈
        parseStack.push(production.left);
    }
    
    bool canReduce() {
        // 检查栈顶是否有可以归约的符号序列
        return findReducibleProduction().left != EMPTY;
    }
};
```

## 📊 LR分析

### LR分析概述
LR分析是自底向上分析的一种，其中：
- **L**：从左到右扫描输入
- **R**：最右推导的逆过程

### LR分析器结构
```c
class LRParser {
private:
    LRParseTable parseTable;
    stack<LRState> stateStack;
    vector<Token> input;
    int position;
    
public:
    LRParser(LRParseTable table, vector<Token> input) 
        : parseTable(table), input(input), position(0) {}
    
    bool parse() {
        stateStack.push(parseTable.getInitialState());
        
        while (position < input.size()) {
            LRState currentState = stateStack.top();
            Token currentToken = input[position];
            
            Action action = parseTable.getAction(currentState, currentToken.type);
            
            switch (action.type) {
                case SHIFT:
                    shift(action.nextState);
                    break;
                case REDUCE:
                    reduce(action.production);
                    break;
                case ACCEPT:
                    return true;
                case ERROR:
                    error("Parse error");
                    return false;
            }
        }
        
        return false;
    }
    
    void shift(LRState nextState) {
        stateStack.push(nextState);
        position++;
    }
    
    void reduce(Production production) {
        // 从栈中弹出产生式右部的状态
        for (int i = 0; i < production.right.length(); i++) {
            stateStack.pop();
        }
        
        // 获取归约后的状态
        LRState currentState = stateStack.top();
        LRState nextState = parseTable.getGoto(currentState, production.left);
        stateStack.push(nextState);
    }
};
```

## 🔄 LR(0)分析

### LR(0)项目
LR(0)项目是带有圆点的产生式，表示分析过程中的位置：

```c
struct LR0Item {
    TokenType left;
    string right;
    int dotPosition;
    
    LR0Item(TokenType left, string right, int dotPosition)
        : left(left), right(right), dotPosition(dotPosition) {}
    
    bool operator==(const LR0Item& other) const {
        return left == other.left && right == other.right && 
               dotPosition == other.dotPosition;
    }
};
```

### LR(0)项目集构造
```c
class LR0ItemSet {
private:
    set<LR0Item> items;
    
public:
    void addItem(LR0Item item) {
        items.insert(item);
    }
    
    void closure() {
        bool changed = true;
        while (changed) {
            changed = false;
            set<LR0Item> newItems;
            
            for (auto& item : items) {
                if (item.dotPosition < item.right.length()) {
                    TokenType nextSymbol = item.right[item.dotPosition];
                    if (isNonTerminal(nextSymbol)) {
                        // 添加所有以nextSymbol为左部的产生式
                        for (auto& production : productions) {
                            if (production.left == nextSymbol) {
                                LR0Item newItem(nextSymbol, production.right, 0);
                                if (items.find(newItem) == items.end()) {
                                    newItems.insert(newItem);
                                    changed = true;
                                }
                            }
                        }
                    }
                }
            }
            
            items.insert(newItems.begin(), newItems.end());
        }
    }
    
    LR0ItemSet gotoSet(TokenType symbol) {
        LR0ItemSet newSet;
        
        for (auto& item : items) {
            if (item.dotPosition < item.right.length() && 
                item.right[item.dotPosition] == symbol) {
                LR0Item newItem(item.left, item.right, item.dotPosition + 1);
                newSet.addItem(newItem);
            }
        }
        
        newSet.closure();
        return newSet;
    }
};
```

## 📊 SLR(1)分析

### SLR(1)分析表构造
```c
class SLR1ParseTable {
private:
    map<pair<LRState, TokenType>, Action> actionTable;
    map<pair<LRState, TokenType>, LRState> gotoTable;
    
public:
    void constructParseTable() {
        // 构造LR(0)项目集
        vector<LR0ItemSet> itemSets = constructItemSets();
        
        // 构造ACTION表
        for (int i = 0; i < itemSets.size(); i++) {
            LR0ItemSet itemSet = itemSets[i];
            
            for (auto& item : itemSet.getItems()) {
                if (item.dotPosition < item.right.length()) {
                    // 移进项目
                    TokenType nextSymbol = item.right[item.dotPosition];
                    LR0ItemSet nextSet = itemSet.gotoSet(nextSymbol);
                    int nextState = findItemSetIndex(nextSet, itemSets);
                    
                    if (isTerminal(nextSymbol)) {
                        actionTable[{i, nextSymbol}] = Action(SHIFT, nextState);
                    } else {
                        gotoTable[{i, nextSymbol}] = nextState;
                    }
                } else {
                    // 归约项目
                    if (item.left == START_SYMBOL) {
                        actionTable[{i, END_OF_INPUT}] = Action(ACCEPT);
                    } else {
                        set<TokenType> follow = computeFollow(item.left);
                        for (TokenType terminal : follow) {
                            actionTable[{i, terminal}] = Action(REDUCE, item.left, item.right);
                        }
                    }
                }
            }
        }
    }
};
```

## 🎯 自底向上分析特点

### 优点
- **表达能力**：能够处理更复杂的文法
- **无限制**：不需要消除左递归
- **效率高**：分析效率高

### 缺点
- **实现复杂**：实现相对复杂
- **错误处理**：错误处理复杂
- **调试困难**：调试相对困难

## 🔧 分析表构造

### 项目集构造
```c
vector<LR0ItemSet> constructItemSets() {
    vector<LR0ItemSet> itemSets;
    LR0ItemSet initialSet;
    
    // 添加初始项目
    LR0Item initialItem(START_SYMBOL, "E", 0);
    initialSet.addItem(initialItem);
    initialSet.closure();
    
    itemSets.push_back(initialSet);
    
    // 构造所有项目集
    for (int i = 0; i < itemSets.size(); i++) {
        LR0ItemSet currentSet = itemSets[i];
        
        // 对每个符号计算GOTO
        for (TokenType symbol : allSymbols) {
            LR0ItemSet nextSet = currentSet.gotoSet(symbol);
            if (!nextSet.isEmpty()) {
                int nextIndex = findItemSetIndex(nextSet, itemSets);
                if (nextIndex == -1) {
                    itemSets.push_back(nextSet);
                }
            }
        }
    }
    
    return itemSets;
}
```
