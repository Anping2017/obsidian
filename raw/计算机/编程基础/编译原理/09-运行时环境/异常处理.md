# 异常处理

## 🎯 异常处理概述

**异常处理**是编译原理在程序运行时错误处理中的应用，涉及异常捕获、异常传播、异常恢复、栈展开等异常处理技术。异常处理需要处理复杂的错误处理策略、性能优化和并发安全。

## 🔍 基本概念

### 异常处理定义
异常处理是指程序运行时自动识别和处理异常情况的过程，避免程序崩溃和提高程序健壮性。

### 异常处理特点
- **自动捕获**：需要自动捕获异常
- **异常传播**：需要传播异常信息
- **异常恢复**：需要恢复程序状态
- **栈展开**：需要展开调用栈

## 🔧 异常处理器

### 异常处理器实现
```c
class ExceptionHandler {
private:
    map<ExceptionType, string> exceptionMessages;
    vector<ExceptionInfo> exceptionStack;
    bool isRunning;
    
public:
    ExceptionHandler() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化异常处理器" << endl;
        isRunning = true;
        
        // 初始化异常消息
        exceptionMessages[RUNTIME_ERROR] = "运行时错误";
        exceptionMessages[LOGIC_ERROR] = "逻辑错误";
        exceptionMessages[SYSTEM_ERROR] = "系统错误";
        exceptionMessages[USER_ERROR] = "用户错误";
        exceptionMessages[RESOURCE_ERROR] = "资源错误";
    }
    
    void registerException(ExceptionType type, const string& message) {
        exceptionMessages[type] = message;
        cout << "注册异常类型: " << getTypeName(type) << " -> " << message << endl;
    }
    
    string getTypeName(ExceptionType type) {
        switch (type) {
            case RUNTIME_ERROR: return "运行时错误";
            case LOGIC_ERROR: return "逻辑错误";
            case SYSTEM_ERROR: return "系统错误";
            case USER_ERROR: return "用户错误";
            case RESOURCE_ERROR: return "资源错误";
            default: return "未知";
        }
    }
    
    void throwException(ExceptionType type, const string& message) {
        cout << "抛出异常: " << getTypeName(type) << " - " << message << endl;
        
        ExceptionInfo info;
        info.type = type;
        info.message = message;
        info.timestamp = getCurrentTime();
        info.stackTrace = getStackTrace();
        
        exceptionStack.push_back(info);
        
        cout << "异常已添加到异常栈" << endl;
    }
    
    bool catchException(ExceptionType type, function<void(const ExceptionInfo&)> handler) {
        cout << "尝试捕获异常: " << getTypeName(type) << endl;
        
        for (auto it = exceptionStack.rbegin(); it != exceptionStack.rend(); ++it) {
            if (it->type == type) {
                cout << "捕获到异常: " << it->message << endl;
                
                // 执行异常处理
                handler(*it);
                
                // 从异常栈中移除
                exceptionStack.erase(it.base() - 1);
                
                return true;
            }
        }
        
        cout << "未找到匹配的异常" << endl;
        return false;
    }
    
    void handleAllExceptions(function<void(const ExceptionInfo&)> handler) {
        cout << "处理所有异常" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& exception : exceptionStack) {
            cout << "处理异常: " << exception.message << endl;
            handler(exception);
        }
        
        exceptionStack.clear();
        cout << "所有异常处理完成" << endl;
    }
    
    void printExceptionStack() {
        cout << "异常栈信息" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (size_t i = 0; i < exceptionStack.size(); i++) {
            const auto& exception = exceptionStack[i];
            cout << "[" << i << "] " << getTypeName(exception.type) 
                 << " - " << exception.message << endl;
            cout << "    时间: " << exception.timestamp << endl;
            cout << "    栈跟踪: " << exception.stackTrace << endl;
        }
        
        cout << endl;
    }
    
    string getCurrentTime() {
        time_t now = time(0);
        char* timeStr = ctime(&now);
        return string(timeStr);
    }
    
    string getStackTrace() {
        // 简化实现：返回模拟的栈跟踪信息
        return "main() -> function1() -> function2()";
    }
    
    void cleanup() {
        cout << "清理异常处理器" << endl;
        
        exceptionStack.clear();
        exceptionMessages.clear();
        
        isRunning = false;
        
        cout << "异常处理器清理完成" << endl;
    }
    
    ~ExceptionHandler() {
        cleanup();
    }
};

struct ExceptionInfo {
    ExceptionType type;
    string message;
    string timestamp;
    string stackTrace;
};
```

## 🔧 异常传播器

### 异常传播器实现
```c
class ExceptionPropagator {
private:
    vector<ExceptionHandler*> handlers;
    map<ExceptionType, ExceptionPropagation> propagationRules;
    bool isRunning;
    
public:
    ExceptionPropagator() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化异常传播器" << endl;
        isRunning = true;
        
        // 初始化传播规则
        propagationRules[RUNTIME_ERROR] = IMMEDIATE;
        propagationRules[LOGIC_ERROR] = DEFERRED;
        propagationRules[SYSTEM_ERROR] = IMMEDIATE;
        propagationRules[USER_ERROR] = CONDITIONAL;
        propagationRules[RESOURCE_ERROR] = IMMEDIATE;
    }
    
    void addHandler(ExceptionHandler* handler) {
        handlers.push_back(handler);
        cout << "添加异常处理器: " << handler << endl;
    }
    
    void removeHandler(ExceptionHandler* handler) {
        handlers.erase(
            remove(handlers.begin(), handlers.end(), handler),
            handlers.end()
        );
        cout << "移除异常处理器: " << handler << endl;
    }
    
    void setPropagationRule(ExceptionType type, ExceptionPropagation propagation) {
        propagationRules[type] = propagation;
        cout << "设置传播规则: " << getTypeName(type) << " -> " << getPropagationName(propagation) << endl;
    }
    
    string getPropagationName(ExceptionPropagation propagation) {
        switch (propagation) {
            case IMMEDIATE: return "立即传播";
            case DEFERRED: return "延迟传播";
            case CONDITIONAL: return "条件传播";
            case MULTIPLE: return "多重传播";
            case CUSTOM: return "自定义传播";
            default: return "未知";
        }
    }
    
    string getTypeName(ExceptionType type) {
        switch (type) {
            case RUNTIME_ERROR: return "运行时错误";
            case LOGIC_ERROR: return "逻辑错误";
            case SYSTEM_ERROR: return "系统错误";
            case USER_ERROR: return "用户错误";
            case RESOURCE_ERROR: return "资源错误";
            default: return "未知";
        }
    }
    
    void propagateException(ExceptionType type, const string& message) {
        cout << "传播异常: " << getTypeName(type) << " - " << message << endl;
        cout << "-------------------------------------------" << endl;
        
        ExceptionPropagation propagation = propagationRules[type];
        
        switch (propagation) {
            case IMMEDIATE:
                propagateImmediate(type, message);
                break;
            case DEFERRED:
                propagateDeferred(type, message);
                break;
            case CONDITIONAL:
                propagateConditional(type, message);
                break;
            case MULTIPLE:
                propagateMultiple(type, message);
                break;
            case CUSTOM:
                propagateCustom(type, message);
                break;
        }
        
        cout << "异常传播完成" << endl;
        cout << endl;
    }
    
    void propagateImmediate(ExceptionType type, const string& message) {
        cout << "立即传播异常" << endl;
        
        for (auto& handler : handlers) {
            handler->throwException(type, message);
        }
    }
    
    void propagateDeferred(ExceptionType type, const string& message) {
        cout << "延迟传播异常" << endl;
        
        // 延迟传播：等待一段时间后再传播
        this_thread::sleep_for(chrono::milliseconds(100));
        
        for (auto& handler : handlers) {
            handler->throwException(type, message);
        }
    }
    
    void propagateConditional(ExceptionType type, const string& message) {
        cout << "条件传播异常" << endl;
        
        // 条件传播：根据条件决定是否传播
        if (shouldPropagate(type)) {
            for (auto& handler : handlers) {
                handler->throwException(type, message);
            }
        } else {
            cout << "条件不满足，不传播异常" << endl;
        }
    }
    
    void propagateMultiple(ExceptionType type, const string& message) {
        cout << "多重传播异常" << endl;
        
        // 多重传播：传播到多个处理器
        for (auto& handler : handlers) {
            handler->throwException(type, message);
        }
        
        // 额外传播到系统日志
        cout << "传播到系统日志: " << message << endl;
    }
    
    void propagateCustom(ExceptionType type, const string& message) {
        cout << "自定义传播异常" << endl;
        
        // 自定义传播：根据类型自定义传播逻辑
        switch (type) {
            case RUNTIME_ERROR:
                cout << "运行时错误：记录到错误日志" << endl;
                break;
            case LOGIC_ERROR:
                cout << "逻辑错误：发送到调试器" << endl;
                break;
            case SYSTEM_ERROR:
                cout << "系统错误：发送到系统监控" << endl;
                break;
            case USER_ERROR:
                cout << "用户错误：显示用户友好消息" << endl;
                break;
            case RESOURCE_ERROR:
                cout << "资源错误：尝试资源恢复" << endl;
                break;
        }
    }
    
    bool shouldPropagate(ExceptionType type) {
        // 简化实现：根据类型决定是否传播
        return type != USER_ERROR || rand() % 2 == 0;
    }
    
    void printPropagationRules() {
        cout << "异常传播规则" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pair : propagationRules) {
            cout << getTypeName(pair.first) << " -> " 
                 << getPropagationName(pair.second) << endl;
        }
        
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理异常传播器" << endl;
        
        handlers.clear();
        propagationRules.clear();
        
        isRunning = false;
        
        cout << "异常传播器清理完成" << endl;
    }
    
    ~ExceptionPropagator() {
        cleanup();
    }
};
```

## 🔧 异常恢复器

### 异常恢复器实现
```c
class ExceptionRecovery {
private:
    map<ExceptionType, function<bool()>> recoveryStrategies;
    vector<RecoveryAction> recoveryHistory;
    bool isRunning;
    
public:
    ExceptionRecovery() : isRunning(false) {}
    
    void initialize() {
        cout << "初始化异常恢复器" << endl;
        isRunning = true;
        
        // 初始化恢复策略
        recoveryStrategies[RUNTIME_ERROR] = [this]() { return recoverRuntimeError(); };
        recoveryStrategies[LOGIC_ERROR] = [this]() { return recoverLogicError(); };
        recoveryStrategies[SYSTEM_ERROR] = [this]() { return recoverSystemError(); };
        recoveryStrategies[USER_ERROR] = [this]() { return recoverUserError(); };
        recoveryStrategies[RESOURCE_ERROR] = [this]() { return recoverResourceError(); };
    }
    
    void registerRecoveryStrategy(ExceptionType type, function<bool()> strategy) {
        recoveryStrategies[type] = strategy;
        cout << "注册恢复策略: " << getTypeName(type) << endl;
    }
    
    string getTypeName(ExceptionType type) {
        switch (type) {
            case RUNTIME_ERROR: return "运行时错误";
            case LOGIC_ERROR: return "逻辑错误";
            case SYSTEM_ERROR: return "系统错误";
            case USER_ERROR: return "用户错误";
            case RESOURCE_ERROR: return "资源错误";
            default: return "未知";
        }
    }
    
    bool recover(ExceptionType type) {
        cout << "尝试恢复异常: " << getTypeName(type) << endl;
        cout << "-------------------------------------------" << endl;
        
        if (recoveryStrategies.find(type) != recoveryStrategies.end()) {
            bool success = recoveryStrategies[type]();
            
            // 记录恢复历史
            RecoveryAction action;
            action.type = type;
            action.success = success;
            action.timestamp = getCurrentTime();
            recoveryHistory.push_back(action);
            
            if (success) {
                cout << "异常恢复成功" << endl;
            } else {
                cout << "异常恢复失败" << endl;
            }
            
            return success;
        } else {
            cout << "未找到恢复策略" << endl;
            return false;
        }
    }
    
    bool recoverRuntimeError() {
        cout << "恢复运行时错误" << endl;
        
        // 恢复策略：重新初始化运行时环境
        cout << "重新初始化运行时环境" << endl;
        cout << "清理临时资源" << endl;
        cout << "重置状态变量" << endl;
        
        return true;
    }
    
    bool recoverLogicError() {
        cout << "恢复逻辑错误" << endl;
        
        // 恢复策略：回滚到上一个稳定状态
        cout << "回滚到上一个稳定状态" << endl;
        cout << "重新计算逻辑" << endl;
        cout << "验证数据完整性" << endl;
        
        return true;
    }
    
    bool recoverSystemError() {
        cout << "恢复系统错误" << endl;
        
        // 恢复策略：重启系统组件
        cout << "重启系统组件" << endl;
        cout << "检查系统资源" << endl;
        cout << "重新建立连接" << endl;
        
        return true;
    }
    
    bool recoverUserError() {
        cout << "恢复用户错误" << endl;
        
        // 恢复策略：提示用户重新输入
        cout << "提示用户重新输入" << endl;
        cout << "验证用户输入" << endl;
        cout << "提供帮助信息" << endl;
        
        return true;
    }
    
    bool recoverResourceError() {
        cout << "恢复资源错误" << endl;
        
        // 恢复策略：重新分配资源
        cout << "重新分配资源" << endl;
        cout << "清理无效资源" << endl;
        cout << "优化资源使用" << endl;
        
        return true;
    }
    
    void printRecoveryHistory() {
        cout << "异常恢复历史" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (size_t i = 0; i < recoveryHistory.size(); i++) {
            const auto& action = recoveryHistory[i];
            cout << "[" << i << "] " << getTypeName(action.type) 
                 << " - " << (action.success ? "成功" : "失败") << endl;
            cout << "    时间: " << action.timestamp << endl;
        }
        
        cout << endl;
    }
    
    string getCurrentTime() {
        time_t now = time(0);
        char* timeStr = ctime(&now);
        return string(timeStr);
    }
    
    void cleanup() {
        cout << "清理异常恢复器" << endl;
        
        recoveryStrategies.clear();
        recoveryHistory.clear();
        
        isRunning = false;
        
        cout << "异常恢复器清理完成" << endl;
    }
    
    ~ExceptionRecovery() {
        cleanup();
    }
};

struct RecoveryAction {
    ExceptionType type;
    bool success;
    string timestamp;
};
```

## 🔧 异常管理器

### 异常管理器实现
```c
class ExceptionManager {
private:
    ExceptionHandler* handler;
    ExceptionPropagator* propagator;
    ExceptionRecovery* recovery;
    bool isRunning;
    
public:
    ExceptionManager() {
        initializeManager();
    }
    
    void initializeManager() {
        cout << "初始化异常管理器" << endl;
        
        handler = new ExceptionHandler();
        propagator = new ExceptionPropagator();
        recovery = new ExceptionRecovery();
        
        // 初始化各个组件
        handler->initialize();
        propagator->initialize();
        recovery->initialize();
        
        // 将处理器添加到传播器
        propagator->addHandler(handler);
        
        isRunning = true;
    }
    
    void throwException(ExceptionType type, const string& message) {
        cout << "异常管理器抛出异常" << endl;
        cout << "===========================================" << endl;
        
        // 抛出异常
        handler->throwException(type, message);
        
        // 传播异常
        propagator->propagateException(type, message);
        
        // 尝试恢复
        if (recovery->recover(type)) {
            cout << "异常已恢复" << endl;
        } else {
            cout << "异常恢复失败" << endl;
        }
        
        cout << endl;
    }
    
    bool catchException(ExceptionType type, function<void(const ExceptionInfo&)> handlerFunc) {
        return handler->catchException(type, handlerFunc);
    }
    
    void handleAllExceptions(function<void(const ExceptionInfo&)> handlerFunc) {
        handler->handleAllExceptions(handlerFunc);
    }
    
    void printExceptionInfo() {
        cout << "异常管理器信息" << endl;
        cout << "===========================================" << endl;
        
        handler->printExceptionStack();
        propagator->printPropagationRules();
        recovery->printRecoveryHistory();
    }
    
    void cleanup() {
        cout << "清理异常管理器" << endl;
        
        if (handler) {
            delete handler;
            handler = nullptr;
        }
        
        if (propagator) {
            delete propagator;
            propagator = nullptr;
        }
        
        if (recovery) {
            delete recovery;
            recovery = nullptr;
        }
        
        isRunning = false;
        
        cout << "异常管理器清理完成" << endl;
    }
    
    ~ExceptionManager() {
        cleanup();
    }
};
```
