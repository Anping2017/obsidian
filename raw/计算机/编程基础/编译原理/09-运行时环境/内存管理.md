# 内存管理

## 🎯 内存管理概述

**内存管理**是编译原理在程序运行时内存分配和释放中的应用，涉及堆管理、栈管理、内存池、内存对齐等内存管理技术。内存管理需要处理复杂的内存分配策略、内存碎片和内存泄漏。

## 🔍 基本概念

### 内存管理定义
内存管理是指程序运行时对内存的分配、使用、释放和回收的管理过程，包括堆内存、栈内存、静态内存等的管理。

### 内存管理特点
- **动态分配**：需要支持动态内存分配
- **自动回收**：需要自动回收无用内存
- **内存对齐**：需要处理内存对齐
- **碎片管理**：需要管理内存碎片

## 🔧 堆内存管理器

### 堆内存管理器实现
```c
class HeapMemoryManager {
private:
    map<void*, MemoryBlock*> allocatedBlocks;
    vector<FreeBlock*> freeBlocks;
    size_t totalSize;
    size_t usedSize;
    AllocationStrategy strategy;
    
public:
    HeapMemoryManager(AllocationStrategy s = FIRST_FIT) 
        : totalSize(0), usedSize(0), strategy(s) {
        initializeManager();
    }
    
    void initializeManager() {
        cout << "初始化堆内存管理器" << endl;
        
        // 分配初始堆内存
        totalSize = 1024 * 1024; // 1MB
        void* heapMemory = malloc(totalSize);
        
        if (heapMemory) {
            FreeBlock* initialBlock = new FreeBlock(heapMemory, totalSize);
            freeBlocks.push_back(initialBlock);
            
            cout << "堆内存初始化完成: " << totalSize << " 字节" << endl;
        } else {
            cout << "堆内存初始化失败" << endl;
        }
    }
    
    void* allocate(size_t size) {
        cout << "分配堆内存: " << size << " 字节" << endl;
        
        // 对齐内存大小
        size = alignSize(size);
        
        // 查找合适的空闲块
        FreeBlock* block = findSuitableBlock(size);
        
        if (block) {
            void* ptr = block->getPtr();
            
            // 创建已分配块
            MemoryBlock* allocatedBlock = new MemoryBlock(ptr, size);
            allocatedBlocks[ptr] = allocatedBlock;
            
            // 更新空闲块
            updateFreeBlock(block, size);
            
            // 更新统计信息
            usedSize += size;
            
            cout << "堆内存分配成功: " << ptr << endl;
            return ptr;
        }
        
        cout << "堆内存分配失败" << endl;
        return nullptr;
    }
    
    void deallocate(void* ptr) {
        cout << "释放堆内存: " << ptr << endl;
        
        if (allocatedBlocks.find(ptr) != allocatedBlocks.end()) {
            MemoryBlock* block = allocatedBlocks[ptr];
            size_t size = block->getSize();
            
            // 从已分配表中移除
            allocatedBlocks.erase(ptr);
            
            // 创建空闲块
            FreeBlock* freeBlock = new FreeBlock(ptr, size);
            freeBlocks.push_back(freeBlock);
            
            // 合并相邻的空闲块
            mergeFreeBlocks();
            
            // 更新统计信息
            usedSize -= size;
            
            delete block;
            
            cout << "堆内存释放成功" << endl;
        } else {
            cout << "堆内存指针无效: " << ptr << endl;
        }
    }
    
    FreeBlock* findSuitableBlock(size_t size) {
        switch (strategy) {
            case FIRST_FIT:
                return findFirstFit(size);
            case BEST_FIT:
                return findBestFit(size);
            case WORST_FIT:
                return findWorstFit(size);
            case NEXT_FIT:
                return findNextFit(size);
            default:
                return findFirstFit(size);
        }
    }
    
    FreeBlock* findFirstFit(size_t size) {
        for (auto& block : freeBlocks) {
            if (block->getSize() >= size) {
                return block;
            }
        }
        return nullptr;
    }
    
    FreeBlock* findBestFit(size_t size) {
        FreeBlock* bestBlock = nullptr;
        size_t bestSize = SIZE_MAX;
        
        for (auto& block : freeBlocks) {
            if (block->getSize() >= size && block->getSize() < bestSize) {
                bestBlock = block;
                bestSize = block->getSize();
            }
        }
        
        return bestBlock;
    }
    
    FreeBlock* findWorstFit(size_t size) {
        FreeBlock* worstBlock = nullptr;
        size_t worstSize = 0;
        
        for (auto& block : freeBlocks) {
            if (block->getSize() >= size && block->getSize() > worstSize) {
                worstBlock = block;
                worstSize = block->getSize();
            }
        }
        
        return worstBlock;
    }
    
    FreeBlock* findNextFit(size_t size) {
        // 这里需要实现循环适应算法
        // 简化实现：使用首次适应
        return findFirstFit(size);
    }
    
    void updateFreeBlock(FreeBlock* block, size_t allocatedSize) {
        size_t remainingSize = block->getSize() - allocatedSize;
        
        if (remainingSize > 0) {
            // 更新块大小
            block->setSize(remainingSize);
            
            // 更新块指针
            void* newPtr = (char*)block->getPtr() + allocatedSize;
            block->setPtr(newPtr);
        } else {
            // 移除完全分配的空闲块
            freeBlocks.erase(
                remove(freeBlocks.begin(), freeBlocks.end(), block),
                freeBlocks.end()
            );
            delete block;
        }
    }
    
    void mergeFreeBlocks() {
        cout << "合并空闲块" << endl;
        
        // 按地址排序
        sort(freeBlocks.begin(), freeBlocks.end(),
            [](FreeBlock* a, FreeBlock* b) {
                return a->getPtr() < b->getPtr();
            });
        
        // 合并相邻块
        for (size_t i = 0; i < freeBlocks.size() - 1; i++) {
            FreeBlock* current = freeBlocks[i];
            FreeBlock* next = freeBlocks[i + 1];
            
            if ((char*)current->getPtr() + current->getSize() == next->getPtr()) {
                // 合并块
                current->setSize(current->getSize() + next->getSize());
                
                // 移除下一个块
                freeBlocks.erase(freeBlocks.begin() + i + 1);
                delete next;
                
                i--; // 重新检查当前块
            }
        }
    }
    
    size_t alignSize(size_t size) {
        // 对齐到8字节边界
        return (size + 7) & ~7;
    }
    
    void printMemoryStats() {
        cout << "堆内存统计信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "总大小: " << totalSize << " 字节" << endl;
        cout << "已使用: " << usedSize << " 字节" << endl;
        cout << "空闲大小: " << totalSize - usedSize << " 字节" << endl;
        cout << "已分配块数: " << allocatedBlocks.size() << endl;
        cout << "空闲块数: " << freeBlocks.size() << endl;
        cout << "使用率: " << (double)usedSize / totalSize * 100 << "%" << endl;
        cout << endl;
    }
    
    void printFreeBlocks() {
        cout << "空闲块列表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (size_t i = 0; i < freeBlocks.size(); i++) {
            FreeBlock* block = freeBlocks[i];
            cout << "块 " << i << ": " << block->getPtr() 
                 << " 大小: " << block->getSize() << " 字节" << endl;
        }
        
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理堆内存管理器" << endl;
        
        // 释放所有分配的内存
        for (auto& pair : allocatedBlocks) {
            delete pair.second;
        }
        allocatedBlocks.clear();
        
        // 清理空闲块
        for (auto& block : freeBlocks) {
            delete block;
        }
        freeBlocks.clear();
        
        totalSize = 0;
        usedSize = 0;
        
        cout << "堆内存管理器清理完成" << endl;
    }
    
    ~HeapMemoryManager() {
        cleanup();
    }
};

class MemoryBlock {
private:
    void* ptr;
    size_t size;
    
public:
    MemoryBlock(void* p, size_t s) : ptr(p), size(s) {}
    
    void* getPtr() { return ptr; }
    size_t getSize() { return size; }
};

class FreeBlock {
private:
    void* ptr;
    size_t size;
    
public:
    FreeBlock(void* p, size_t s) : ptr(p), size(s) {}
    
    void* getPtr() { return ptr; }
    size_t getSize() { return size; }
    
    void setPtr(void* p) { ptr = p; }
    void setSize(size_t s) { size = s; }
};
```

## 🔧 栈内存管理器

### 栈内存管理器实现
```c
class StackMemoryManager {
private:
    void* stackBase;
    void* stackTop;
    void* stackPointer;
    size_t stackSize;
    vector<StackFrame*> stackFrames;
    
public:
    StackMemoryManager(size_t size = 1024 * 1024) : stackSize(size) {
        initializeManager();
    }
    
    void initializeManager() {
        cout << "初始化栈内存管理器" << endl;
        
        // 分配栈内存
        stackBase = malloc(stackSize);
        if (stackBase) {
            stackTop = (char*)stackBase + stackSize;
            stackPointer = stackTop;
            
            cout << "栈内存初始化完成: " << stackSize << " 字节" << endl;
        } else {
            cout << "栈内存初始化失败" << endl;
        }
    }
    
    void* allocate(size_t size) {
        cout << "分配栈内存: " << size << " 字节" << endl;
        
        // 对齐内存大小
        size = alignSize(size);
        
        // 检查栈空间
        if ((char*)stackPointer - size < (char*)stackBase) {
            cout << "栈内存不足" << endl;
            return nullptr;
        }
        
        // 分配栈内存
        stackPointer = (char*)stackPointer - size;
        
        cout << "栈内存分配成功: " << stackPointer << endl;
        return stackPointer;
    }
    
    void deallocate(size_t size) {
        cout << "释放栈内存: " << size << " 字节" << endl;
        
        // 对齐内存大小
        size = alignSize(size);
        
        // 释放栈内存
        stackPointer = (char*)stackPointer + size;
        
        cout << "栈内存释放成功: " << stackPointer << endl;
    }
    
    void pushFrame(string functionName) {
        cout << "压入栈帧: " << functionName << endl;
        
        StackFrame* frame = new StackFrame(functionName, stackPointer);
        stackFrames.push_back(frame);
        
        cout << "栈帧压入成功" << endl;
    }
    
    void popFrame() {
        if (!stackFrames.empty()) {
            StackFrame* frame = stackFrames.back();
            stackFrames.pop_back();
            
            cout << "弹出栈帧: " << frame->getFunctionName() << endl;
            
            // 恢复栈指针
            stackPointer = frame->getStackPointer();
            
            delete frame;
            
            cout << "栈帧弹出成功" << endl;
        } else {
            cout << "栈帧为空" << endl;
        }
    }
    
    size_t alignSize(size_t size) {
        // 对齐到8字节边界
        return (size + 7) & ~7;
    }
    
    void printStackInfo() {
        cout << "栈信息" << endl;
        cout << "-------------------------------------------" << endl;
        cout << "栈基址: " << stackBase << endl;
        cout << "栈顶: " << stackTop << endl;
        cout << "栈指针: " << stackPointer << endl;
        cout << "栈大小: " << stackSize << " 字节" << endl;
        cout << "已使用: " << (char*)stackTop - (char*)stackPointer << " 字节" << endl;
        cout << "栈帧数: " << stackFrames.size() << endl;
        cout << endl;
    }
    
    void printStackFrames() {
        cout << "栈帧列表" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (size_t i = 0; i < stackFrames.size(); i++) {
            StackFrame* frame = stackFrames[i];
            cout << "帧 " << i << ": " << frame->getFunctionName() 
                 << " 栈指针: " << frame->getStackPointer() << endl;
        }
        
        cout << endl;
    }
    
    void cleanup() {
        cout << "清理栈内存管理器" << endl;
        
        // 清理栈帧
        for (auto& frame : stackFrames) {
            delete frame;
        }
        stackFrames.clear();
        
        // 释放栈内存
        if (stackBase) {
            free(stackBase);
            stackBase = nullptr;
        }
        
        stackTop = nullptr;
        stackPointer = nullptr;
        stackSize = 0;
        
        cout << "栈内存管理器清理完成" << endl;
    }
    
    ~StackMemoryManager() {
        cleanup();
    }
};

class StackFrame {
private:
    string functionName;
    void* stackPointer;
    
public:
    StackFrame(string name, void* sp) : functionName(name), stackPointer(sp) {}
    
    string getFunctionName() { return functionName; }
    void* getStackPointer() { return stackPointer; }
};
```

## 🔧 内存池管理器

### 内存池管理器实现
```c
class MemoryPoolManager {
private:
    vector<MemoryPool*> memoryPools;
    map<size_t, MemoryPool*> poolMap;
    
public:
    MemoryPoolManager() {
        initializeManager();
    }
    
    void initializeManager() {
        cout << "初始化内存池管理器" << endl;
        
        // 创建不同大小的内存池
        createMemoryPools();
    }
    
    void createMemoryPools() {
        cout << "创建内存池" << endl;
        
        // 创建不同大小的内存池
        size_t sizes[] = {16, 32, 64, 128, 256, 512, 1024, 2048, 4096};
        size_t counts[] = {1000, 1000, 1000, 1000, 1000, 500, 500, 500, 500};
        
        for (size_t i = 0; i < 9; i++) {
            MemoryPool* pool = new MemoryPool(sizes[i], counts[i]);
            memoryPools.push_back(pool);
            poolMap[sizes[i]] = pool;
            
            cout << "创建内存池: " << sizes[i] << " 字节, " << counts[i] << " 块" << endl;
        }
    }
    
    void* allocate(size_t size) {
        cout << "分配内存池内存: " << size << " 字节" << endl;
        
        // 对齐内存大小
        size = alignSize(size);
        
        // 查找合适的内存池
        MemoryPool* pool = findSuitablePool(size);
        
        if (pool) {
            void* ptr = pool->allocate();
            if (ptr) {
                cout << "内存池分配成功: " << ptr << endl;
                return ptr;
            }
        }
        
        cout << "内存池分配失败" << endl;
        return nullptr;
    }
    
    void deallocate(void* ptr, size_t size) {
        cout << "释放内存池内存: " << ptr << " 大小: " << size << " 字节" << endl;
        
        // 对齐内存大小
        size = alignSize(size);
        
        // 查找对应的内存池
        MemoryPool* pool = findSuitablePool(size);
        
        if (pool) {
            pool->deallocate(ptr);
            cout << "内存池释放成功" << endl;
        } else {
            cout << "内存池释放失败" << endl;
        }
    }
    
    MemoryPool* findSuitablePool(size_t size) {
        // 查找最接近的内存池
        for (auto& pair : poolMap) {
            if (pair.first >= size) {
                return pair.second;
            }
        }
        
        return nullptr;
    }
    
    size_t alignSize(size_t size) {
        // 对齐到8字节边界
        return (size + 7) & ~7;
    }
    
    void printPoolStats() {
        cout << "内存池统计信息" << endl;
        cout << "-------------------------------------------" << endl;
        
        for (auto& pool : memoryPools) {
            cout << "池大小: " << pool->getBlockSize() << " 字节" << endl;
            cout << "总块数: " << pool->getPoolSize() << endl;
            cout << "空闲块数: " << pool->getFreeBlocks() << endl;
            cout << "使用率: " << (double)(pool->getPoolSize() - pool->getFreeBlocks()) / pool->getPoolSize() * 100 << "%" << endl;
            cout << endl;
        }
    }
    
    void cleanup() {
        cout << "清理内存池管理器" << endl;
        
        // 清理所有内存池
        for (auto& pool : memoryPools) {
            delete pool;
        }
        memoryPools.clear();
        poolMap.clear();
        
        cout << "内存池管理器清理完成" << endl;
    }
    
    ~MemoryPoolManager() {
        cleanup();
    }
};

class MemoryPool {
private:
    size_t blockSize;
    size_t poolSize;
    vector<void*> freeBlocks;
    void* poolMemory;
    
public:
    MemoryPool(size_t blockSize, size_t poolSize) 
        : blockSize(blockSize), poolSize(poolSize) {
        initializePool();
    }
    
    void initializePool() {
        cout << "初始化内存池: " << blockSize << " 字节, " << poolSize << " 块" << endl;
        
        // 分配池内存
        poolMemory = malloc(blockSize * poolSize);
        
        if (poolMemory) {
            // 初始化空闲块列表
            for (size_t i = 0; i < poolSize; i++) {
                void* block = (char*)poolMemory + i * blockSize;
                freeBlocks.push_back(block);
            }
            
            cout << "内存池初始化完成" << endl;
        } else {
            cout << "内存池初始化失败" << endl;
        }
    }
    
    void* allocate() {
        if (!freeBlocks.empty()) {
            void* block = freeBlocks.back();
            freeBlocks.pop_back();
            
            cout << "从内存池分配: " << block << endl;
            return block;
        }
        
        cout << "内存池已满" << endl;
        return nullptr;
    }
    
    void deallocate(void* block) {
        freeBlocks.push_back(block);
        cout << "释放到内存池: " << block << endl;
    }
    
    size_t getBlockSize() { return blockSize; }
    size_t getPoolSize() { return poolSize; }
    size_t getFreeBlocks() { return freeBlocks.size(); }
    
    ~MemoryPool() {
        if (poolMemory) {
            free(poolMemory);
        }
    }
};
```

## 🔧 内存管理器

### 内存管理器实现
```c
class MemoryManager {
private:
    HeapMemoryManager* heapManager;
    StackMemoryManager* stackManager;
    MemoryPoolManager* poolManager;
    
public:
    MemoryManager() {
        initializeManager();
    }
    
    void initializeManager() {
        cout << "初始化内存管理器" << endl;
        
        heapManager = new HeapMemoryManager();
        stackManager = new StackMemoryManager();
        poolManager = new MemoryPoolManager();
    }
    
    void* allocateHeap(size_t size) {
        return heapManager->allocate(size);
    }
    
    void deallocateHeap(void* ptr) {
        heapManager->deallocate(ptr);
    }
    
    void* allocateStack(size_t size) {
        return stackManager->allocate(size);
    }
    
    void deallocateStack(size_t size) {
        stackManager->deallocate(size);
    }
    
    void* allocatePool(size_t size) {
        return poolManager->allocate(size);
    }
    
    void deallocatePool(void* ptr, size_t size) {
        poolManager->deallocate(ptr, size);
    }
    
    void pushStackFrame(string functionName) {
        stackManager->pushFrame(functionName);
    }
    
    void popStackFrame() {
        stackManager->popFrame();
    }
    
    void printMemoryStats() {
        cout << "内存管理器统计信息" << endl;
        cout << "===========================================" << endl;
        
        heapManager->printMemoryStats();
        stackManager->printStackInfo();
        poolManager->printPoolStats();
    }
    
    void cleanup() {
        cout << "清理内存管理器" << endl;
        
        if (heapManager) {
            delete heapManager;
            heapManager = nullptr;
        }
        
        if (stackManager) {
            delete stackManager;
            stackManager = nullptr;
        }
        
        if (poolManager) {
            delete poolManager;
            poolManager = nullptr;
        }
        
        cout << "内存管理器清理完成" << endl;
    }
    
    ~MemoryManager() {
        cleanup();
    }
};
```
