# 可用表达式分析

## 🎯 可用表达式分析概述

**可用表达式分析**是数据流分析的重要类型，用于确定程序中每个点处哪些表达式是可用的（即在该点之前已经被计算过且其操作数没有被重新定义）。可用表达式分析是公共子表达式消除等优化技术的基础。

## 🔍 基本概念

### 可用表达式定义
一个表达式e在程序点p处是可用的，当且仅当从程序入口到p的所有执行路径上，e都被计算过，且在该计算之后e的操作数没有被重新定义。

### 可用表达式分析特点
- **前向分析**：从程序入口向出口分析
- **必须分析**：提供保守的近似结果
- **迭代计算**：通过迭代计算收敛到不动点
- **优化基础**：为多种优化技术提供信息

## 🔧 可用表达式分析实现

### 基本可用表达式分析
```c
class AvailableExpressionAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, set<string>> availableIn;
    map<BasicBlock*, set<string>> availableOut;
    map<BasicBlock*, set<string>> genSets;
    map<BasicBlock*, set<string>> killSets;
    
public:
    AvailableExpressionAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 计算生成集和杀死集
        computeGenKillSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeGenKillSets() {
        for (auto block : cfg->getBlocks()) {
            set<string> gen;
            set<string> kill;
            
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    string expr = buildExpressionKey(inst);
                    
                    // 添加到生成集
                    gen.insert(expr);
                    
                    // 添加到杀死集（杀死包含结果变量的表达式）
                    for (auto& otherExpr : getAllExpressions()) {
                        if (containsVariable(otherExpr, inst.result)) {
                            kill.insert(otherExpr);
                        }
                    }
                }
            }
            
            genSets[block] = gen;
            killSets[block] = kill;
        }
    }
    
    string buildExpressionKey(Instruction& inst) {
        return inst.op + ":" + inst.arg1 + ":" + inst.arg2;
    }
    
    set<string> getAllExpressions() {
        set<string> expressions;
        
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    string expr = buildExpressionKey(inst);
                    expressions.insert(expr);
                }
            }
        }
        
        return expressions;
    }
    
    bool containsVariable(string expr, string variable) {
        return expr.find(variable) != string::npos;
    }
    
    void iterativeAnalysis() {
        // 初始化
        initialize();
        
        // 迭代计算
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 1000;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                set<string> oldAvailableOut = availableOut[block];
                set<string> newAvailableOut = computeAvailableOut(block);
                
                if (oldAvailableOut != newAvailableOut) {
                    availableOut[block] = newAvailableOut;
                    availableIn[block] = computeAvailableIn(block);
                    changed = true;
                }
            }
        }
    }
    
    void initialize() {
        for (auto block : cfg->getBlocks()) {
            availableIn[block] = set<string>();
            availableOut[block] = set<string>();
        }
    }
    
    set<string> computeAvailableOut(BasicBlock* block) {
        // OUT[B] = (IN[B] - KILL[B]) ∪ GEN[B]
        set<string> availableOut = availableIn[block];
        
        // 减去杀死集
        for (auto expr : killSets[block]) {
            availableOut.erase(expr);
        }
        
        // 加上生成集
        for (auto expr : genSets[block]) {
            availableOut.insert(expr);
        }
        
        return availableOut;
    }
    
    set<string> computeAvailableIn(BasicBlock* block) {
        set<string> availableIn;
        
        // 合并所有前驱块的availableOut
        for (auto pred : block->predecessors) {
            for (auto expr : availableOut[pred]) {
                availableIn.insert(expr);
            }
        }
        
        return availableIn;
    }
    
    set<string> getAvailableIn(BasicBlock* block) {
        return availableIn[block];
    }
    
    set<string> getAvailableOut(BasicBlock* block) {
        return availableOut[block];
    }
    
    set<string> getGenSet(BasicBlock* block) {
        return genSets[block];
    }
    
    set<string> getKillSet(BasicBlock* block) {
        return killSets[block];
    }
};
```

```c
class PreciseAvailableExpressionAnalysis {
private:
    ControlFlowGraph* cfg;
    map<BasicBlock*, map<int, set<string>>> availableIn;
    map<BasicBlock*, map<int, set<string>>> availableOut;
    map<BasicBlock*, map<int, set<string>>> genSets;
    map<BasicBlock*, map<int, set<string>>> killSets;
    
public:
    PreciseAvailableExpressionAnalysis(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void analyze() {
        // 计算生成集和杀死集
        computeGenKillSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeGenKillSets() {
        for (auto block : cfg->getBlocks()) {
            map<int, set<string>> gen;
            map<int, set<string>> kill;
            
            // 为每个指令位置计算生成集和杀死集
            for (int i = 0; i < block->instructions.size(); i++) {
                Instruction& inst = block->instructions[i];
                
                if (inst.type == ASSIGNMENT) {
                    string expr = buildExpressionKey(inst);
                    
                    // 添加到生成集
                    gen[i].insert(expr);
                    
                    // 添加到杀死集（杀死包含结果变量的表达式）
                    for (auto& otherExpr : getAllExpressions()) {
                        if (containsVariable(otherExpr, inst.result)) {
                            kill[i].insert(otherExpr);
                        }
                    }
                }
            }
            
            genSets[block] = gen;
            killSets[block] = kill;
        }
    }
    
    string buildExpressionKey(Instruction& inst) {
        return inst.op + ":" + inst.arg1 + ":" + inst.arg2;
    }
    
    set<string> getAllExpressions() {
        set<string> expressions;
        
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    string expr = buildExpressionKey(inst);
                    expressions.insert(expr);
                }
            }
        }
        
        return expressions;
    }
    
    bool containsVariable(string expr, string variable) {
        return expr.find(variable) != string::npos;
    }
    
    void iterativeAnalysis() {
        // 初始化
        initialize();
        
        // 迭代计算
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 1000;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                map<int, set<string>> oldAvailableOut = availableOut[block];
                map<int, set<string>> newAvailableOut = computeAvailableOut(block);
                
                if (oldAvailableOut != newAvailableOut) {
                    availableOut[block] = newAvailableOut;
                    availableIn[block] = computeAvailableIn(block);
                    changed = true;
                }
            }
        }
    }
    
    void initialize() {
        for (auto block : cfg->getBlocks()) {
            availableIn[block] = map<int, set<string>>();
            availableOut[block] = map<int, set<string>>();
        }
    }
    
    map<int, set<string>> computeAvailableOut(BasicBlock* block) {
        map<int, set<string>> availableOut;
        
        // 为每个指令位置计算availableOut
        for (int i = 0; i < block->instructions.size(); i++) {
            set<string> availableOutAtI;
            
            if (i == 0) {
                // 第一条指令的availableOut是前驱块的availableOut
                for (auto pred : block->predecessors) {
                    for (auto expr : availableOut[pred][block->instructions.size() - 1]) {
                        availableOutAtI.insert(expr);
                    }
                }
            } else {
                // 其他指令的availableOut是前一条指令的availableOut
                availableOutAtI = availableOut[block][i - 1];
            }
            
            availableOut[i] = availableOutAtI;
        }
        
        return availableOut;
    }
    
    map<int, set<string>> computeAvailableIn(BasicBlock* block) {
        map<int, set<string>> availableIn;
        
        // 为每个指令位置计算availableIn
        for (int i = 0; i < block->instructions.size(); i++) {
            set<string> availableInAtI = availableOut[block][i];
            
            // 减去杀死集
            for (auto expr : killSets[block][i]) {
                availableInAtI.erase(expr);
            }
            
            // 加上生成集
            for (auto expr : genSets[block][i]) {
                availableInAtI.insert(expr);
            }
            
            availableIn[i] = availableInAtI;
        }
        
        return availableIn;
    }
    
    set<string> getAvailableIn(BasicBlock* block, int position) {
        return availableIn[block][position];
    }
    
    set<string> getAvailableOut(BasicBlock* block, int position) {
        return availableOut[block][position];
    }
};
```

### 公共子表达式消除
```c
class CommonSubexpressionEliminationUsingAvailableExpression {
private:
    ControlFlowGraph* cfg;
    AvailableExpressionAnalysis* availableExpressionAnalysis;
    map<string, string> expressionToVariable;
    
public:
    CommonSubexpressionEliminationUsingAvailableExpression(ControlFlowGraph* cfg) : cfg(cfg) {
        availableExpressionAnalysis = new AvailableExpressionAnalysis(cfg);
    }
    
    void eliminateCommonSubexpressions() {
        // 执行可用表达式分析
        availableExpressionAnalysis->analyze();
        
        // 消除公共子表达式
        for (auto block : cfg->getBlocks()) {
            eliminateCommonSubexpressionsInBlock(block);
        }
    }
    
    void eliminateCommonSubexpressionsInBlock(BasicBlock* block) {
        set<string> availableExprs = availableExpressionAnalysis->getAvailableIn(block);
        
        // 为每个可用表达式分配临时变量
        for (auto expr : availableExprs) {
            if (expressionToVariable.find(expr) == expressionToVariable.end()) {
                string tempVar = generateTempVariable();
                expressionToVariable[expr] = tempVar;
            }
        }
        
        // 应用公共子表达式消除
        applyCommonSubexpressionElimination(block);
    }
    
    string generateTempVariable() {
        static int counter = 0;
        return "t" + to_string(counter++);
    }
    
    void applyCommonSubexpressionElimination(BasicBlock* block) {
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                string expr = buildExpressionKey(inst);
                
                if (expressionToVariable.find(expr) != expressionToVariable.end()) {
                    // 使用已计算的临时变量
                    string tempVar = expressionToVariable[expr];
                    optimizedInstructions.push_back(
                        Instruction(ASSIGNMENT, "=", tempVar, "", inst.result));
                } else {
                    // 计算新表达式
                    optimizedInstructions.push_back(inst);
                }
            } else {
                optimizedInstructions.push_back(inst);
            }
        }
        
        block->instructions = optimizedInstructions;
    }
    
    string buildExpressionKey(Instruction& inst) {
        return inst.op + ":" + inst.arg1 + ":" + inst.arg2;
    }
};
```

```c
class PartialRedundancyEliminationUsingAvailableExpression {
private:
    ControlFlowGraph* cfg;
    AvailableExpressionAnalysis* availableExpressionAnalysis;
    map<BasicBlock*, set<string>> anticipatedExpressions;
    
public:
    PartialRedundancyEliminationUsingAvailableExpression(ControlFlowGraph* cfg) : cfg(cfg) {
        availableExpressionAnalysis = new AvailableExpressionAnalysis(cfg);
    }
    
    void eliminatePartialRedundancy() {
        // 执行可用表达式分析
        availableExpressionAnalysis->analyze();
        
        // 计算预期表达式
        computeAnticipatedExpressions();
        
        // 消除部分冗余
        for (auto block : cfg->getBlocks()) {
            eliminatePartialRedundancyInBlock(block);
        }
    }
    
    void computeAnticipatedExpressions() {
        // 初始化预期表达式
        for (auto block : cfg->getBlocks()) {
            anticipatedExpressions[block] = set<string>();
        }
        
        // 迭代计算预期表达式
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 1000;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                if (computeAnticipatedForBlock(block)) {
                    changed = true;
                }
            }
        }
    }
    
    bool computeAnticipatedForBlock(BasicBlock* block) {
        bool changed = false;
        set<string>& anticipated = anticipatedExpressions[block];
        set<string> newAnticipated;
        
        // 计算基本块内生成的表达式
        set<string> generated = getGeneratedExpressions(block);
        
        // 计算基本块内杀死的表达式
        set<string> killed = getKilledExpressions(block);
        
        // 计算后继块的预期表达式
        set<string> successorAnticipated;
        for (auto succ : block->successors) {
            for (auto expr : anticipatedExpressions[succ]) {
                successorAnticipated.insert(expr);
            }
        }
        
        // 计算新的预期表达式
        for (auto expr : generated) {
            newAnticipated.insert(expr);
        }
        
        for (auto expr : successorAnticipated) {
            if (killed.find(expr) == killed.end()) {
                newAnticipated.insert(expr);
            }
        }
        
        if (newAnticipated != anticipated) {
            anticipated = newAnticipated;
            changed = true;
        }
        
        return changed;
    }
    
    set<string> getGeneratedExpressions(BasicBlock* block) {
        set<string> generated;
        
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                string expr = buildExpressionKey(inst);
                generated.insert(expr);
            }
        }
        
        return generated;
    }
    
    set<string> getKilledExpressions(BasicBlock* block) {
        set<string> killed;
        
        for (auto& inst : block->instructions) {
            if (inst.type == ASSIGNMENT) {
                // 杀死包含结果变量的表达式
                string result = inst.result;
                for (auto& expr : availableExpressionAnalysis->getAvailableIn(block)) {
                    if (containsVariable(expr, result)) {
                        killed.insert(expr);
                    }
                }
            }
        }
        
        return killed;
    }
    
    string buildExpressionKey(Instruction& inst) {
        return inst.op + ":" + inst.arg1 + ":" + inst.arg2;
    }
    
    bool containsVariable(string expr, string variable) {
        return expr.find(variable) != string::npos;
    }
    
    void eliminatePartialRedundancyInBlock(BasicBlock* block) {
        set<string> availableExprs = availableExpressionAnalysis->getAvailableIn(block);
        set<string> anticipatedExprs = anticipatedExpressions[block];
        
        // 计算可插入的表达式
```

## 🎯 可用表达式分析集成

### 可用表达式分析管理器
```c
class AvailableExpressionAnalysisManager {
private:
    ControlFlowGraph* cfg;
    map<AvailableExpressionAnalysisType, AvailableExpressionAnalysisBase*> analyses;
    
public:
    AvailableExpressionAnalysisManager(ControlFlowGraph* cfg) : cfg(cfg) {
        // 初始化各种分析
        analyses[MAY_AVAILABLE_EXPRESSION] = new AvailableExpressionAnalysis(cfg);
        analyses[MUST_AVAILABLE_EXPRESSION] = new PreciseAvailableExpressionAnalysis(cfg);
    }
    
    void runAnalysis(AvailableExpressionAnalysisType type) {
        if (analyses.find(type) != analyses.end()) {
            analyses[type]->analyze();
        }
    }
    
    void runAllAnalyses() {
        for (auto& pair : analyses) {
            pair.second->analyze();
        }
    }
    
    set<string> getAvailableExpressions(AvailableExpressionAnalysisType type, BasicBlock* block, bool isInSet) {
        if (analyses.find(type) != analyses.end()) {
            AvailableExpressionAnalysis* analysis = dynamic_cast<AvailableExpressionAnalysis*>(analyses[type]);
            if (analysis) {
                return isInSet ? analysis->getAvailableIn(block) : analysis->getAvailableOut(block);
            }
        }
        return set<string>();
    }
    
    void cleanup() {
        for (auto& pair : analyses) {
            delete pair.second;
        }
        analyses.clear();
    }
};

class AvailableExpressionAnalysisBase {
public:
    virtual void analyze() = 0;
    virtual ~AvailableExpressionAnalysisBase() = default;
};
```
