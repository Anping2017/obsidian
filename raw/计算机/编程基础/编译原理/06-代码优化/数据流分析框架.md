# 数据流分析框架

## 🎯 数据流分析框架概述

**数据流分析框架**是代码优化的理论基础，通过分析程序中数据的定义、使用和传播关系，为各种优化技术提供精确的数据流信息。数据流分析是全局优化的核心组件。

## 🔍 基本概念

### 数据流分析定义
数据流分析是一种静态程序分析方法，通过分析程序中数据的定义、使用和传播关系，计算程序执行过程中数据的流动情况。

### 数据流分析特点
- **静态分析**：在编译时进行分析
- **近似分析**：提供保守的近似结果
- **迭代计算**：通过迭代计算收敛到不动点
- **框架统一**：提供统一的分析框架

## 🔧 数据流分析框架

### 基本框架结构
```c
template<typename T>
class DataFlowAnalysis {
protected:
    ControlFlowGraph* cfg;
    map<BasicBlock*, T> inSets;
    map<BasicBlock*, T> outSets;
    DataFlowDirection direction;
    AnalysisPrecision precision;
    
public:
    DataFlowAnalysis(ControlFlowGraph* cfg, DataFlowDirection dir, AnalysisPrecision prec)
        : cfg(cfg), direction(dir), precision(prec) {}
    
    virtual void analyze() = 0;
    
    virtual T getInSet(BasicBlock* block) {
        return inSets[block];
    }
    
    virtual T getOutSet(BasicBlock* block) {
        return outSets[block];
    }
    
    virtual void setInSet(BasicBlock* block, T value) {
        inSets[block] = value;
    }
    
    virtual void setOutSet(BasicBlock* block, T value) {
        outSets[block] = value;
    }
    
protected:
    virtual T getInitialValue() = 0;
    virtual T getBoundaryValue() = 0;
    virtual T meet(T a, T b) = 0;
    virtual T transfer(BasicBlock* block, T input) = 0;
    virtual bool isEqual(T a, T b) = 0;
    
    void iterativeAnalysis() {
        // 初始化
        initialize();
        
        // 迭代计算
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 1000;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (auto block : cfg->getBlocks()) {
                T oldValue = getCurrentValue(block);
                T newValue = computeNewValue(block);
                
                if (!isEqual(oldValue, newValue)) {
                    setCurrentValue(block, newValue);
                    changed = true;
                }
            }
        }
    }
    
    void initialize() {
        T initialValue = getInitialValue();
        T boundaryValue = getBoundaryValue();
        
        for (auto block : cfg->getBlocks()) {
            if (direction == FORWARD_ANALYSIS) {
                setInSet(block, initialValue);
                setOutSet(block, initialValue);
            } else {
                setInSet(block, initialValue);
                setOutSet(block, initialValue);
            }
        }
        
        // 设置边界条件
        if (direction == FORWARD_ANALYSIS) {
            setInSet(cfg->getEntryBlock(), boundaryValue);
        } else {
            setOutSet(cfg->getExitBlock(), boundaryValue);
        }
    }
    
    T computeNewValue(BasicBlock* block) {
        if (direction == FORWARD_ANALYSIS) {
            // 前向分析：OUT[B] = f_B(IN[B])
            T input = getInSet(block);
            return transfer(block, input);
        } else {
            // 后向分析：IN[B] = f_B(OUT[B])
            T output = getOutSet(block);
            return transfer(block, output);
        }
    }
    
    T getCurrentValue(BasicBlock* block) {
        if (direction == FORWARD_ANALYSIS) {
            return getOutSet(block);
        } else {
            return getInSet(block);
        }
    }
    
    void setCurrentValue(BasicBlock* block, T value) {
        if (direction == FORWARD_ANALYSIS) {
            setOutSet(block, value);
        } else {
            setInSet(block, value);
        }
    }
    
    void updateInputSets() {
        for (auto block : cfg->getBlocks()) {
            if (direction == FORWARD_ANALYSIS) {
                // 前向分析：IN[B] = meet(OUT[P1], OUT[P2], ...)
                T input = getInitialValue();
                for (auto pred : block->predecessors) {
                    input = meet(input, getOutSet(pred));
                }
                setInSet(block, input);
            } else {
                // 后向分析：OUT[B] = meet(IN[S1], IN[S2], ...)
                T output = getInitialValue();
                for (auto succ : block->successors) {
                    output = meet(output, getInSet(succ));
                }
                setOutSet(block, output);
            }
        }
    }
};
```

## 🔧 到达定值分析

### 到达定值分析实现
```c
class ReachingDefinitionAnalysis : public DataFlowAnalysis<set<Definition>> {
private:
    map<BasicBlock*, set<Definition>> genSets;
    map<BasicBlock*, set<Definition>> killSets;
    
public:
    ReachingDefinitionAnalysis(ControlFlowGraph* cfg) 
        : DataFlowAnalysis<set<Definition>>(cfg, FORWARD_ANALYSIS, MAY_ANALYSIS) {}
    
    void analyze() {
        // 计算生成集和杀死集
        computeGenKillSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeGenKillSets() {
        for (auto block : cfg->getBlocks()) {
            set<Definition> gen;
            set<Definition> kill;
            
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    Definition def(inst.result, block, &inst);
                    
                    // 添加到生成集
                    gen.insert(def);
                    
                    // 添加到杀死集（杀死其他对同一变量的定义）
                    for (auto& otherDef : getAllDefinitions(inst.result)) {
                        if (otherDef != def) {
                            kill.insert(otherDef);
                        }
                    }
                }
            }
            
            genSets[block] = gen;
            killSets[block] = kill;
        }
    }
    
    set<Definition> getAllDefinitions(string variable) {
        set<Definition> definitions;
        
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT && inst.result == variable) {
                    Definition def(variable, block, &inst);
                    definitions.insert(def);
                }
            }
        }
        
        return definitions;
    }
    
    set<Definition> getInitialValue() override {
        return set<Definition>();
    }
    
    set<Definition> getBoundaryValue() override {
        return set<Definition>();
    }
    
    set<Definition> meet(set<Definition> a, set<Definition> b) override {
        set<Definition> result = a;
        result.insert(b.begin(), b.end());
        return result;
    }
    
    set<Definition> transfer(BasicBlock* block, set<Definition> input) override {
        // OUT[B] = (IN[B] - KILL[B]) ∪ GEN[B]
        set<Definition> output = input;
        
        // 减去杀死集
        for (auto def : killSets[block]) {
            output.erase(def);
        }
        
        // 加上生成集
        for (auto def : genSets[block]) {
            output.insert(def);
        }
        
        return output;
    }
    
    bool isEqual(set<Definition> a, set<Definition> b) override {
        return a == b;
    }
};

struct Definition {
    string variable;
    BasicBlock* block;
    Instruction* instruction;
    
    Definition(string var, BasicBlock* blk, Instruction* inst) 
        : variable(var), block(blk), instruction(inst) {}
    
    bool operator<(const Definition& other) const {
        if (variable != other.variable) {
            return variable < other.variable;
        }
        if (block != other.block) {
            return block < other.block;
        }
        return instruction < other.instruction;
    }
    
    bool operator==(const Definition& other) const {
        return variable == other.variable && 
               block == other.block && 
               instruction == other.instruction;
    }
};
```

## 🔧 活跃变量分析

### 活跃变量分析实现
```c
class LiveVariableAnalysis : public DataFlowAnalysis<set<string>> {
private:
    map<BasicBlock*, set<string>> useSets;
    map<BasicBlock*, set<string>> defSets;
    
public:
    LiveVariableAnalysis(ControlFlowGraph* cfg) 
        : DataFlowAnalysis<set<string>>(cfg, BACKWARD_ANALYSIS, MAY_ANALYSIS) {}
    
    void analyze() {
        // 计算使用集和定义集
        computeUseDefSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeUseDefSets() {
        for (auto block : cfg->getBlocks()) {
            set<string> use;
            set<string> def;
            
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    // 添加到定义集
                    def.insert(inst.result);
                    
                    // 添加到使用集（如果变量在使用前未定义）
                    if (def.find(inst.arg1) == def.end()) {
                        use.insert(inst.arg1);
                    }
                    if (def.find(inst.arg2) == def.end()) {
                        use.insert(inst.arg2);
                    }
                }
            }
            
            useSets[block] = use;
            defSets[block] = def;
        }
    }
    
    set<string> getInitialValue() override {
        return set<string>();
    }
    
    set<string> getBoundaryValue() override {
        return set<string>();
    }
    
    set<string> meet(set<string> a, set<string> b) override {
        set<string> result = a;
        result.insert(b.begin(), b.end());
        return result;
    }
    
    set<string> transfer(BasicBlock* block, set<string> input) override {
        // IN[B] = USE[B] ∪ (OUT[B] - DEF[B])
        set<string> output = input;
        
        // 减去定义集
        for (auto var : defSets[block]) {
            output.erase(var);
        }
        
        // 加上使用集
        for (auto var : useSets[block]) {
            output.insert(var);
        }
        
        return output;
    }
    
    bool isEqual(set<string> a, set<string> b) override {
        return a == b;
    }
};
```

```c
class AvailableExpressionAnalysis : public DataFlowAnalysis<set<string>> {
private:
    map<BasicBlock*, set<string>> genSets;
    map<BasicBlock*, set<string>> killSets;
    
public:
    AvailableExpressionAnalysis(ControlFlowGraph* cfg) 
        : DataFlowAnalysis<set<string>>(cfg, FORWARD_ANALYSIS, MUST_ANALYSIS) {}
    
    void analyze() {
        // 计算生成集和杀死集
        computeGenKillSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeGenKillSets() {
        for (auto block : cfg->getBlocks()) {
            set<string> gen;
            set<string> kill;
            
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    string expr = buildExpressionKey(inst);
                    
                    // 添加到生成集
                    gen.insert(expr);
                    
                    // 添加到杀死集（杀死包含结果变量的表达式）
                    for (auto& otherExpr : getAllExpressions()) {
                        if (containsVariable(otherExpr, inst.result)) {
                            kill.insert(otherExpr);
                        }
                    }
                }
            }
            
            genSets[block] = gen;
            killSets[block] = kill;
        }
    }
    
    string buildExpressionKey(Instruction& inst) {
        return inst.op + ":" + inst.arg1 + ":" + inst.arg2;
    }
    
    set<string> getAllExpressions() {
        set<string> expressions;
        
        for (auto block : cfg->getBlocks()) {
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    string expr = buildExpressionKey(inst);
                    expressions.insert(expr);
                }
            }
        }
        
        return expressions;
    }
    
    bool containsVariable(string expr, string variable) {
        return expr.find(variable) != string::npos;
    }
    
    set<string> getInitialValue() override {
        return set<string>();
    }
    
    set<string> getBoundaryValue() override {
        return set<string>();
    }
    
    set<string> meet(set<string> a, set<string> b) override {
        set<string> result;
        for (auto expr : a) {
            if (b.find(expr) != b.end()) {
                result.insert(expr);
            }
        }
        return result;
    }
    
    set<string> transfer(BasicBlock* block, set<string> input) override {
        // OUT[B] = (IN[B] - KILL[B]) ∪ GEN[B]
        set<string> output = input;
        
        // 减去杀死集
        for (auto expr : killSets[block]) {
            output.erase(expr);
        }
        
        // 加上生成集
        for (auto expr : genSets[block]) {
            output.insert(expr);
        }
        
        return output;
    }
    
    bool isEqual(set<string> a, set<string> b) override {
        return a == b;
    }
};
```

## 🔧 常量传播分析

### 常量传播分析实现
```c
class ConstantPropagationAnalysis : public DataFlowAnalysis<map<string, ConstantValue>> {
private:
    map<BasicBlock*, map<string, ConstantValue>> genSets;
    map<BasicBlock*, map<string, ConstantValue>> killSets;
    
public:
    ConstantPropagationAnalysis(ControlFlowGraph* cfg) 
        : DataFlowAnalysis<map<string, ConstantValue>>(cfg, FORWARD_ANALYSIS, MUST_ANALYSIS) {}
    
    void analyze() {
        // 计算生成集和杀死集
        computeGenKillSets();
        
        // 执行迭代分析
        iterativeAnalysis();
    }
    
    void computeGenKillSets() {
        for (auto block : cfg->getBlocks()) {
            map<string, ConstantValue> gen;
            map<string, ConstantValue> kill;
            
            for (auto& inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    string var = inst.result;
                    
                    // 计算变量的常量值
                    ConstantValue value = evaluateConstantExpression(inst);
                    
                    if (value.isConstant) {
                        gen[var] = value;
                    } else {
                        kill[var] = ConstantValue(false, 0);
                    }
                }
            }
            
            genSets[block] = gen;
            killSets[block] = kill;
        }
    }
    
    ConstantValue evaluateConstantExpression(Instruction& inst) {
        if (inst.op == "=") {
            // 简单赋值
            return getConstantValue(inst.arg1);
        } else {
            // 二元运算
            ConstantValue left = getConstantValue(inst.arg1);
            ConstantValue right = getConstantValue(inst.arg2);
            
            if (left.isConstant && right.isConstant) {
                int result = evaluateBinaryOperation(inst.op, left.value, right.value);
                return ConstantValue(true, result);
            }
        }
        
        return ConstantValue(false, 0);
    }
    
    ConstantValue getConstantValue(string var) {
        // 检查是否为字面常量
        if (isLiteralConstant(var)) {
            return ConstantValue(true, parseLiteralConstant(var));
        }
        
        return ConstantValue(false, 0);
    }
    
    int evaluateBinaryOperation(string op, int left, int right) {
        if (op == "+") return left + right;
        if (op == "-") return left - right;
        if (op == "*") return left * right;
        if (op == "/") return right != 0 ? left / right : 0;
        if (op == "%") return right != 0 ? left % right : 0;
        if (op == "&") return left & right;
        if (op == "|") return left | right;
        if (op == "^") return left ^ right;
        if (op == "<<") return left << right;
        if (op == ">>") return left >> right;
        if (op == "==") return left == right ? 1 : 0;
        if (op == "!=") return left != right ? 1 : 0;
        if (op == "<") return left < right ? 1 : 0;
        if (op == ">") return left > right ? 1 : 0;
        if (op == "<=") return left <= right ? 1 : 0;
        if (op == ">=") return left >= right ? 1 : 0;
        return 0;
    }
    
    bool isLiteralConstant(string str) {
        if (str.empty()) return false;
        return str.find_first_not_of("0123456789") == string::npos ||
               str == "true" || str == "false";
    }
    
    int parseLiteralConstant(string str) {
        if (str == "true") return 1;
        if (str == "false") return 0;
        return stoi(str);
    }
    
    map<string, ConstantValue> getInitialValue() override {
        return map<string, ConstantValue>();
    }
    
    map<string, ConstantValue> getBoundaryValue() override {
        return map<string, ConstantValue>();
    }
    
    map<string, ConstantValue> meet(map<string, ConstantValue> a, map<string, ConstantValue> b) override {
        map<string, ConstantValue> result;
        
        // 合并两个映射
        for (auto& pair : a) {
            string var = pair.first;
            ConstantValue value = pair.second;
            
            if (b.find(var) != b.end()) {
                ConstantValue otherValue = b[var];
                
                if (value.isConstant && otherValue.isConstant && value.value == otherValue.value) {
                    result[var] = value;
                } else {
                    result[var] = ConstantValue(false, 0);
                }
            } else {
                result[var] = value;
            }
        }
        
        for (auto& pair : b) {
            string var = pair.first;
            ConstantValue value = pair.second;
            
            if (a.find(var) == a.end()) {
                result[var] = value;
            }
        }
        
        return result;
    }
    
    map<string, ConstantValue> transfer(BasicBlock* block, map<string, ConstantValue> input) override {
        // OUT[B] = (IN[B] - KILL[B]) ∪ GEN[B]
        map<string, ConstantValue> output = input;
        
        // 减去杀死集
        for (auto& pair : killSets[block]) {
            string var = pair.first;
            output.erase(var);
        }
        
        // 加上生成集
        for (auto& pair : genSets[block]) {
            string var = pair.first;
            ConstantValue value = pair.second;
            output[var] = value;
        }
        
        return output;
    }
    
    bool isEqual(map<string, ConstantValue> a, map<string, ConstantValue> b) override {
        return a == b;
    }
};
```

## 🎯 数据流分析集成

### 数据流分析管理器
```c
class DataFlowAnalysisManager {
private:
    ControlFlowGraph* cfg;
    map<DataFlowAnalysisType, DataFlowAnalysisBase*> analyses;
    
public:
    DataFlowAnalysisManager(ControlFlowGraph* cfg) : cfg(cfg) {
        // 初始化各种分析
        analyses[REACHING_DEFINITION] = new ReachingDefinitionAnalysis(cfg);
        analyses[LIVE_VARIABLE] = new LiveVariableAnalysis(cfg);
        analyses[AVAILABLE_EXPRESSION] = new AvailableExpressionAnalysis(cfg);
        analyses[CONSTANT_PROPAGATION] = new ConstantPropagationAnalysis(cfg);
    }
    
    void runAnalysis(DataFlowAnalysisType type) {
        if (analyses.find(type) != analyses.end()) {
            analyses[type]->analyze();
        }
    }
    
    void runAllAnalyses() {
        for (auto& pair : analyses) {
            pair.second->analyze();
        }
    }
    
    template<typename T>
    T getAnalysisResult(DataFlowAnalysisType type, BasicBlock* block, bool isInSet) {
        if (analyses.find(type) != analyses.end()) {
            DataFlowAnalysis<T>* analysis = dynamic_cast<DataFlowAnalysis<T>*>(analyses[type]);
            if (analysis) {
                return isInSet ? analysis->getInSet(block) : analysis->getOutSet(block);
            }
        }
        return T();
    }
    
    void cleanup() {
        for (auto& pair : analyses) {
            delete pair.second;
        }
        analyses.clear();
    }
};

class DataFlowAnalysisBase {
public:
    virtual void analyze() = 0;
    virtual ~DataFlowAnalysisBase() = default;
};
```
