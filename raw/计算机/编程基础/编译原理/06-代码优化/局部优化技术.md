# 局部优化技术

## 🎯 局部优化技术概述

**局部优化技术**是在基本块内部进行的代码优化，不涉及跨基本块的数据流分析。局部优化是代码优化的基础，具有实现简单、效果明显、风险较低的特点。

## 🔍 基本概念

### 局部优化定义
局部优化是在单个基本块内进行的优化，只考虑基本块内部的指令依赖关系，不考虑与其他基本块的数据流关系。

### 局部优化特点
- **范围限制**：仅限于单个基本块
- **实现简单**：不需要复杂的数据流分析
- **效果明显**：能够显著改善基本块性能
- **风险较低**：不会影响程序的其他部分

## 🔧 常量折叠优化

### 常量折叠实现
```c
class ConstantFoldingOptimizer {
private:
    vector<Instruction> instructions;
    
public:
    ConstantFoldingOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 10;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            for (int i = 0; i < instructions.size(); i++) {
                if (canFold(instructions[i])) {
                    Instruction folded = foldInstruction(instructions[i]);
                    if (folded != instructions[i]) {
                        instructions[i] = folded;
                        changed = true;
                    }
                }
            }
        }
        
        return instructions;
    }
    
    bool canFold(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            return isConstant(inst.arg1) && isConstant(inst.arg2);
        } else if (inst.type == UNARY_OP) {
            return isConstant(inst.arg1);
        }
        return false;
    }
    
    Instruction foldInstruction(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            return foldBinaryOperation(inst);
        } else if (inst.type == UNARY_OP) {
            return foldUnaryOperation(inst);
        }
        return inst;
    }
    
    Instruction foldBinaryOperation(Instruction& inst) {
        int val1 = parseConstant(inst.arg1);
        int val2 = parseConstant(inst.arg2);
        int result = evaluateBinaryOperation(inst.op, val1, val2);
        
        return Instruction(ASSIGNMENT, "=", to_string(result), "", inst.result);
    }
    
    Instruction foldUnaryOperation(Instruction& inst) {
        int val = parseConstant(inst.arg1);
        int result = evaluateUnaryOperation(inst.op, val);
        
        return Instruction(ASSIGNMENT, "=", to_string(result), "", inst.result);
    }
    
    int evaluateBinaryOperation(string op, int left, int right) {
        if (op == "+") return left + right;
        if (op == "-") return left - right;
        if (op == "*") return left * right;
        if (op == "/") return right != 0 ? left / right : 0;
        if (op == "%") return right != 0 ? left % right : 0;
        if (op == "&") return left & right;
        if (op == "|") return left | right;
        if (op == "^") return left ^ right;
        if (op == "<<") return left << right;
        if (op == ">>") return left >> right;
        if (op == "==") return left == right ? 1 : 0;
        if (op == "!=") return left != right ? 1 : 0;
        if (op == "<") return left < right ? 1 : 0;
        if (op == ">") return left > right ? 1 : 0;
        if (op == "<=") return left <= right ? 1 : 0;
        if (op == ">=") return left >= right ? 1 : 0;
        return 0;
    }
    
    int evaluateUnaryOperation(string op, int val) {
        if (op == "-") return -val;
        if (op == "!") return val == 0 ? 1 : 0;
        if (op == "~") return ~val;
        return val;
    }
    
    bool isConstant(string str) {
        if (str.empty()) return false;
        
        // 检查是否为数字
        if (str.find_first_not_of("0123456789") == string::npos) {
            return true;
        }
        
        // 检查是否为布尔值
        if (str == "true" || str == "false") {
            return true;
        }
        
        return false;
    }
    
    int parseConstant(string str) {
        if (str == "true") return 1;
        if (str == "false") return 0;
        return stoi(str);
    }
};
```

### 死代码消除实现

```c
class DeadCodeEliminationOptimizer {
private:
    vector<Instruction> instructions;
    set<string> liveVariables;
    
public:
    DeadCodeEliminationOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        // 计算活跃变量
        computeLiveVariables();
        
        // 消除死代码
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : instructions) {
            if (isLive(inst)) {
                optimizedInstructions.push_back(inst);
            }
        }
        
        return optimizedInstructions;
    }
    
    void computeLiveVariables() {
        liveVariables.clear();
        
        // 从后往前遍历指令
        for (int i = instructions.size() - 1; i >= 0; i--) {
            Instruction& inst = instructions[i];
            
            if (inst.type == ASSIGNMENT) {
                // 如果结果变量不在活跃变量集合中，则指令是死代码
                if (liveVariables.find(inst.result) == liveVariables.end()) {
                    continue;
                }
                
                // 移除结果变量
                liveVariables.erase(inst.result);
                
                // 添加使用的变量
                addUsedVariables(inst);
            } else if (inst.type == FUNCTION_CALL) {
                // 函数调用可能使用全局变量
                addGlobalVariables(inst);
            } else if (inst.type == RETURN) {
                // 返回语句使用返回值
                if (!inst.arg1.empty()) {
                    liveVariables.insert(inst.arg1);
                }
            }
        }
    }
    
    void addUsedVariables(Instruction& inst) {
        if (!inst.arg1.empty() && !isConstant(inst.arg1)) {
            liveVariables.insert(inst.arg1);
        }
        if (!inst.arg2.empty() && !isConstant(inst.arg2)) {
            liveVariables.insert(inst.arg2);
        }
    }
    
    void addGlobalVariables(Instruction& inst) {
        // 添加可能被函数调用使用的全局变量
        // 这里需要根据具体的函数调用语义来实现
    }
    
    bool isLive(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            return liveVariables.find(inst.result) != liveVariables.end();
        }
        return true; // 非赋值指令都认为是活跃的
    }
    
    bool isConstant(string str) {
        if (str.empty()) return false;
        return str.find_first_not_of("0123456789") == string::npos ||
               str == "true" || str == "false";
    }
};
```

## 🔧 公共子表达式消除

### 公共子表达式消除实现
```c
class CommonSubexpressionEliminationOptimizer {
private:
    vector<Instruction> instructions;
    map<string, string> expressionMap;
    map<string, int> expressionCount;
    
public:
    CommonSubexpressionEliminationOptimizer(vector<Instruction>& insts) 
        : instructions(insts) {}
    
    vector<Instruction> optimize() {
        // 第一遍：收集所有表达式
        collectExpressions();
        
        // 第二遍：消除公共子表达式
        return eliminateCommonSubexpressions();
    }
    
    void collectExpressions() {
        expressionMap.clear();
        expressionCount.clear();
        
        for (auto& inst : instructions) {
            if (inst.type == ASSIGNMENT) {
                string expr = buildExpressionKey(inst);
                expressionCount[expr]++;
            }
        }
    }
    
    string buildExpressionKey(Instruction& inst) {
        return inst.op + ":" + inst.arg1 + ":" + inst.arg2;
    }
    
    vector<Instruction> eliminateCommonSubexpressions() {
        vector<Instruction> optimizedInstructions;
        map<string, string> tempVariables;
        int tempCounter = 0;
        
        for (auto& inst : instructions) {
            if (inst.type == ASSIGNMENT) {
                string expr = buildExpressionKey(inst);
                
                if (expressionCount[expr] > 1) {
                    // 检查是否已经计算过这个表达式
                    if (expressionMap.find(expr) != expressionMap.end()) {
                        // 使用已计算的临时变量
                        string tempVar = expressionMap[expr];
                        optimizedInstructions.push_back(
                            Instruction(ASSIGNMENT, "=", tempVar, "", inst.result));
                    } else {
                        // 第一次遇到这个表达式，计算并存储
                        string tempVar = "t" + to_string(tempCounter++);
                        expressionMap[expr] = tempVar;
                        
                        // 添加计算表达式的指令
                        optimizedInstructions.push_back(
                            Instruction(ASSIGNMENT, inst.op, inst.arg1, inst.arg2, tempVar));
                        
                        // 添加赋值指令
                        optimizedInstructions.push_back(
                            Instruction(ASSIGNMENT, "=", tempVar, "", inst.result));
                    }
                } else {
                    // 表达式只出现一次，直接保留
                    optimizedInstructions.push_back(inst);
                }
            } else {
                optimizedInstructions.push_back(inst);
            }
        }
        
        return optimizedInstructions;
    }
};
```

### 强度削弱实现
```c
class StrengthReductionOptimizer {
private:
    vector<Instruction> instructions;
    
public:
    StrengthReductionOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : instructions) {
            Instruction optimized = reduceStrength(inst);
            optimizedInstructions.push_back(optimized);
        }
        
        return optimizedInstructions;
    }
    
    Instruction reduceStrength(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            return reduceBinaryOperation(inst);
        }
        return inst;
    }
    
    Instruction reduceBinaryOperation(Instruction& inst) {
        string op = inst.op;
        string arg1 = inst.arg1;
        string arg2 = inst.arg2;
        string result = inst.result;
        
        // 乘法替换为加法
        if (op == "*" && isConstant(arg2)) {
            int val = parseConstant(arg2);
            if (val == 2) {
                return Instruction(ASSIGNMENT, "+", arg1, arg1, result);
            } else if (val == 4) {
                // x * 4 = x + x + x + x
                string temp1 = generateTempVar();
                string temp2 = generateTempVar();
                return Instruction(ASSIGNMENT, "+", arg1, arg1, temp1);
                // 这里需要生成多条指令，简化处理
            }
        }
        
        // 除法替换为右移
        if (op == "/" && isConstant(arg2) && isPowerOfTwo(parseConstant(arg2))) {
            int shiftAmount = log2(parseConstant(arg2));
            return Instruction(ASSIGNMENT, ">>", arg1, to_string(shiftAmount), result);
        }
        
        // 乘法替换为左移
        if (op == "*" && isConstant(arg2) && isPowerOfTwo(parseConstant(arg2))) {
            int shiftAmount = log2(parseConstant(arg2));
            return Instruction(ASSIGNMENT, "<<", arg1, to_string(shiftAmount), result);
        }
        
        // 模运算优化
        if (op == "%" && isConstant(arg2) && isPowerOfTwo(parseConstant(arg2))) {
            int mask = parseConstant(arg2) - 1;
            return Instruction(ASSIGNMENT, "&", arg1, to_string(mask), result);
        }
        
        return inst;
    }
    
    bool isConstant(string str) {
        if (str.empty()) return false;
        return str.find_first_not_of("0123456789") == string::npos;
    }
    
    int parseConstant(string str) {
        return stoi(str);
    }
    
    bool isPowerOfTwo(int n) {
        return n > 0 && (n & (n - 1)) == 0;
    }
    
    int log2(int n) {
        int result = 0;
        while (n > 1) {
            n >>= 1;
            result++;
        }
        return result;
    }
    
    string generateTempVar() {
        static int counter = 0;
        return "t" + to_string(counter++);
    }
};
```

### 代数简化实现
```c
class AlgebraicSimplificationOptimizer {
private:
    vector<Instruction> instructions;
    
public:
    AlgebraicSimplificationOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : instructions) {
            Instruction simplified = simplifyInstruction(inst);
            optimizedInstructions.push_back(simplified);
        }
        
        return optimizedInstructions;
    }
    
    Instruction simplifyInstruction(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            return simplifyBinaryOperation(inst);
        }
        return inst;
    }
    
    Instruction simplifyBinaryOperation(Instruction& inst) {
        string op = inst.op;
        string arg1 = inst.arg1;
        string arg2 = inst.arg2;
        string result = inst.result;
        
        // 加法恒等式
        if (op == "+") {
            if (isZero(arg1)) {
                return Instruction(ASSIGNMENT, "=", arg2, "", result);
            }
            if (isZero(arg2)) {
                return Instruction(ASSIGNMENT, "=", arg1, "", result);
            }
        }
        
        // 乘法恒等式
        if (op == "*") {
            if (isOne(arg1)) {
                return Instruction(ASSIGNMENT, "=", arg2, "", result);
            }
            if (isOne(arg2)) {
                return Instruction(ASSIGNMENT, "=", arg1, "", result);
            }
            if (isZero(arg1) || isZero(arg2)) {
                return Instruction(ASSIGNMENT, "=", "0", "", result);
            }
        }
        
        // 减法恒等式
        if (op == "-") {
            if (isZero(arg2)) {
                return Instruction(ASSIGNMENT, "=", arg1, "", result);
            }
            if (arg1 == arg2) {
                return Instruction(ASSIGNMENT, "=", "0", "", result);
            }
        }
        
        // 除法恒等式
        if (op == "/") {
            if (isOne(arg2)) {
                return Instruction(ASSIGNMENT, "=", arg1, "", result);
            }
            if (arg1 == arg2) {
                return Instruction(ASSIGNMENT, "=", "1", "", result);
            }
        }
        
        // 逻辑运算恒等式
        if (op == "&&") {
            if (isFalse(arg1) || isFalse(arg2)) {
                return Instruction(ASSIGNMENT, "=", "false", "", result);
            }
            if (isTrue(arg1)) {
                return Instruction(ASSIGNMENT, "=", arg2, "", result);
            }
            if (isTrue(arg2)) {
                return Instruction(ASSIGNMENT, "=", arg1, "", result);
            }
        }
        
        if (op == "||") {
            if (isTrue(arg1) || isTrue(arg2)) {
                return Instruction(ASSIGNMENT, "=", "true", "", result);
            }
            if (isFalse(arg1)) {
                return Instruction(ASSIGNMENT, "=", arg2, "", result);
            }
            if (isFalse(arg2)) {
                return Instruction(ASSIGNMENT, "=", arg1, "", result);
            }
        }
        
        return inst;
    }
    
    bool isZero(string str) {
        return str == "0";
    }
    
    bool isOne(string str) {
        return str == "1";
    }
    
    bool isTrue(string str) {
        return str == "true" || str == "1";
    }
    
    bool isFalse(string str) {
        return str == "false" || str == "0";
    }
};
```

```c
class CopyPropagationOptimizer {
private:
    vector<Instruction> instructions;
    map<string, string> copyMap;
    
public:
    CopyPropagationOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        // 第一遍：建立复制映射
        buildCopyMap();
        
        // 第二遍：应用复制传播
        return applyCopyPropagation();
    }
    
    void buildCopyMap() {
        copyMap.clear();
        
        for (auto& inst : instructions) {
            if (inst.type == ASSIGNMENT && inst.op == "=") {
                // 简单的复制操作
                copyMap[inst.result] = inst.arg1;
            }
        }
    }
    
    vector<Instruction> applyCopyPropagation() {
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : instructions) {
            Instruction optimized = propagateCopies(inst);
            optimizedInstructions.push_back(optimized);
        }
        
        return optimizedInstructions;
    }
    
    Instruction propagateCopies(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            string arg1 = propagateCopy(inst.arg1);
            string arg2 = propagateCopy(inst.arg2);
            
            return Instruction(inst.type, inst.op, arg1, arg2, inst.result);
        }
        return inst;
    }
    
    string propagateCopy(string var) {
        if (copyMap.find(var) != copyMap.end()) {
            return copyMap[var];
        }
        return var;
    }
};
```

## 🎯 局部优化集成

### 局部优化器
```c
class LocalOptimizer {
private:
    vector<Instruction> instructions;
    vector<LocalOptimizationType> enabledOptimizations;
    
public:
    LocalOptimizer(vector<Instruction>& insts) : instructions(insts) {
        // 默认启用所有局部优化
        enabledOptimizations = {
            CONSTANT_FOLDING,
            ALGEBRAIC_SIMPLIFICATION,
            STRENGTH_REDUCTION,
            COPY_PROPAGATION,
            CONSTANT_PROPAGATION,
            COMMON_SUBEXPRESSION_ELIMINATION,
            DEAD_CODE_ELIMINATION
        };
    }
    
    vector<Instruction> optimize() {
        vector<Instruction> current = instructions;
        bool changed = true;
        int iterations = 0;
        const int maxIterations = 5;
        
        while (changed && iterations < maxIterations) {
            changed = false;
            iterations++;
            
            vector<Instruction> previous = current;
            
            // 按优先级执行优化
            for (auto optType : enabledOptimizations) {
                current = applyOptimization(current, optType);
            }
            
            // 检查是否有变化
            changed = (current != previous);
        }
        
        return current;
    }
    
    vector<Instruction> applyOptimization(vector<Instruction>& insts, LocalOptimizationType type) {
        switch (type) {
            case CONSTANT_FOLDING:
                return ConstantFoldingOptimizer(insts).optimize();
            case ALGEBRAIC_SIMPLIFICATION:
                return AlgebraicSimplificationOptimizer(insts).optimize();
            case STRENGTH_REDUCTION:
                return StrengthReductionOptimizer(insts).optimize();
            case COPY_PROPAGATION:
                return CopyPropagationOptimizer(insts).optimize();
            case COMMON_SUBEXPRESSION_ELIMINATION:
                return CommonSubexpressionEliminationOptimizer(insts).optimize();
            case DEAD_CODE_ELIMINATION:
                return DeadCodeEliminationOptimizer(insts).optimize();
            default:
                return insts;
        }
    }
    
    void enableOptimization(LocalOptimizationType type) {
        if (find(enabledOptimizations.begin(), enabledOptimizations.end(), type) 
            == enabledOptimizations.end()) {
            enabledOptimizations.push_back(type);
        }
    }
    
    void disableOptimization(LocalOptimizationType type) {
        enabledOptimizations.erase(
            remove(enabledOptimizations.begin(), enabledOptimizations.end(), type),
            enabledOptimizations.end());
    }
};
```
