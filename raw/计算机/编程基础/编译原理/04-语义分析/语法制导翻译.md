# 语法制导翻译

## 🎯 语法制导翻译概述

**语法制导翻译**是在语法分析过程中同时进行语义分析的技术。它将语义动作与语法规则相结合，在推导或归约过程中执行相应的语义计算。

## 🔍 基本概念

### 语法制导翻译定义
语法制导翻译是一种将语义动作嵌入到语法分析过程中的技术，通过属性计算来实现语义分析。

### 属性
属性是与语法符号相关联的语义信息：

```c
// 属性示例
struct Attribute {
    string name;        // 属性名
    string value;      // 属性值
    AttributeType type; // 属性类型
};

enum AttributeType {
    SYNTHESIZED,  // 综合属性
    INHERITED     // 继承属性
};
```

## 📊 属性文法

### 属性文法定义
属性文法是在上下文无关文法的基础上，为每个语法符号添加属性的文法。

### 属性文法示例
```c
// 简单算术表达式的属性文法
// E → E + T { E.val = E1.val + T.val; E.type = E1.type; }
// E → T { E.val = T.val; E.type = T.type; }
// T → T * F { T.val = T1.val * F.val; T.type = T1.type; }
// T → F { T.val = F.val; T.type = F.type; }
// F → (E) { F.val = E.val; F.type = E.type; }
// F → id { F.val = lookup(id.name); F.type = lookupType(id.name); }
// F → num { F.val = num.val; F.type = int; }
```

### 属性计算
```c
class AttributeCalculator {
private:
    map<string, Attribute> attributes;
    
public:
    void calculateAttributes(ASTNode* node) {
        switch (node->getType()) {
            case ADDITION:
                calculateAddition(node);
                break;
            case MULTIPLICATION:
                calculateMultiplication(node);
                break;
            case IDENTIFIER:
                calculateIdentifier(node);
                break;
            case NUMBER:
                calculateNumber(node);
                break;
        }
    }
    
    void calculateAddition(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        // 计算子节点属性
        calculateAttributes(left);
        calculateAttributes(right);
        
        // 计算当前节点属性
        int leftVal = left->getAttribute("val").getIntValue();
        int rightVal = right->getAttribute("val").getIntValue();
        
        node->setAttribute("val", leftVal + rightVal);
        node->setAttribute("type", "int");
    }
    
    void calculateIdentifier(ASTNode* node) {
        string name = node->getValue();
        
        // 查找符号表
        SymbolEntry* entry = symbolTable.lookup(name);
        if (entry) {
            node->setAttribute("val", entry->getValue());
            node->setAttribute("type", entry->getType());
        } else {
            reportError("Undeclared identifier: " + name);
        }
    }
};
```

## 🔧 语法制导翻译实现

### 自顶向下翻译
```c
class TopDownTranslator {
private:
    vector<Token> tokens;
    int position;
    SymbolTable symbolTable;
    
public:
    ASTNode* parseExpression() {
        ASTNode* left = parseTerm();
        
        while (currentToken.type == PLUS) {
            match(PLUS);
            ASTNode* right = parseTerm();
            
            // 语义动作
            ASTNode* result = new BinaryOpNode(PLUS, left, right);
            result->setAttribute("type", "int");
            result->setAttribute("val", 
                left->getAttribute("val").getIntValue() + 
                right->getAttribute("val").getIntValue());
            
            left = result;
        }
        
        return left;
    }
    
    ASTNode* parseTerm() {
        ASTNode* left = parseFactor();
        
        while (currentToken.type == MULTIPLY) {
            match(MULTIPLY);
            ASTNode* right = parseFactor();
            
            // 语义动作
            ASTNode* result = new BinaryOpNode(MULTIPLY, left, right);
            result->setAttribute("type", "int");
            result->setAttribute("val", 
                left->getAttribute("val").getIntValue() * 
                right->getAttribute("val").getIntValue());
            
            left = result;
        }
        
        return left;
    }
    
    ASTNode* parseFactor() {
        if (currentToken.type == LPAREN) {
            match(LPAREN);
            ASTNode* expr = parseExpression();
            match(RPAREN);
            
            // 语义动作
            expr->setAttribute("type", "int");
            return expr;
        } else if (currentToken.type == IDENTIFIER) {
            string name = currentToken.value;
            match(IDENTIFIER);
            
            // 语义动作
            ASTNode* node = new IdentifierNode(name);
            SymbolEntry* entry = symbolTable.lookup(name);
            if (entry) {
                node->setAttribute("val", entry->getValue());
                node->setAttribute("type", entry->getType());
            } else {
                reportError("Undeclared identifier: " + name);
            }
            
            return node;
        } else if (currentToken.type == NUMBER) {
            int value = stoi(currentToken.value);
            match(NUMBER);
            
            // 语义动作
            ASTNode* node = new NumberNode(value);
            node->setAttribute("val", value);
            node->setAttribute("type", "int");
            
            return node;
        } else {
            error("Expected identifier, number, or '('");
            return nullptr;
        }
    }
};
```

### 自底向上翻译
```c
class BottomUpTranslator {
private:
    stack<ASTNode*> parseStack;
    vector<Token> tokens;
    int position;
    
public:
    void translate() {
        while (position < tokens.size()) {
            Token current = tokens[position];
            
            if (canReduce()) {
                reduce();
            } else {
                shift();
            }
        }
    }
    
    void reduce() {
        // 找到可以归约的产生式
        Production production = findReducibleProduction();
        
        // 创建新节点
        ASTNode* node = new ASTNode(production.left);
        
        // 弹出产生式右部的节点
        vector<ASTNode*> children;
        for (int i = 0; i < production.right.length(); i++) {
            children.push_back(parseStack.top());
            parseStack.pop();
        }
        
        // 设置子节点
        for (int i = children.size() - 1; i >= 0; i--) {
            node->addChild(children[i]);
        }
        
        // 执行语义动作
        executeSemanticAction(node, production);
        
        // 压入新节点
        parseStack.push(node);
    }
    
    void executeSemanticAction(ASTNode* node, Production production) {
        switch (production.action) {
            case ADDITION:
                executeAdditionAction(node);
                break;
            case MULTIPLICATION:
                executeMultiplicationAction(node);
                break;
            case IDENTIFIER:
                executeIdentifierAction(node);
                break;
            case NUMBER:
                executeNumberAction(node);
                break;
        }
    }
    
    void executeAdditionAction(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        int leftVal = left->getAttribute("val").getIntValue();
        int rightVal = right->getAttribute("val").getIntValue();
        
        node->setAttribute("val", leftVal + rightVal);
        node->setAttribute("type", "int");
    }
    
    void executeIdentifierAction(ASTNode* node) {
        string name = node->getValue();
        SymbolEntry* entry = symbolTable.lookup(name);
        
        if (entry) {
            node->setAttribute("val", entry->getValue());
            node->setAttribute("type", entry->getType());
        } else {
            reportError("Undeclared identifier: " + name);
        }
    }
};
```

### 类型检查
```c
void checkTypes(ASTNode* node) {
    switch (node->getType()) {
        case ADDITION:
            checkAdditionTypes(node);
            break;
        case ASSIGNMENT:
            checkAssignmentTypes(node);
            break;
        case FUNCTION_CALL:
            checkFunctionCallTypes(node);
            break;
    }
}

void checkAdditionTypes(ASTNode* node) {
    ASTNode* left = node->getChild(0);
    ASTNode* right = node->getChild(1);
    
    checkTypes(left);
    checkTypes(right);
    
    string leftType = left->getAttribute("type").getStringValue();
    string rightType = right->getAttribute("type").getStringValue();
    
    if (leftType != rightType) {
        reportError("Type mismatch in addition: " + leftType + " + " + rightType);
    }
    
    node->setAttribute("type", leftType);
}

void checkAssignmentTypes(ASTNode* node) {
    ASTNode* left = node->getChild(0);
    ASTNode* right = node->getChild(1);
    
    checkTypes(right);
    
    string leftType = left->getAttribute("type").getStringValue();
    string rightType = right->getAttribute("type").getStringValue();
    
    if (!isCompatible(leftType, rightType)) {
        reportError("Type mismatch in assignment: " + leftType + " = " + rightType);
    }
}
```

void generateCode(ASTNode* node) {
    switch (node->getType()) {
        case ADDITION:
            generateAdditionCode(node);
            break;
        case ASSIGNMENT:
            generateAssignmentCode(node);
            break;
        case FUNCTION_CALL:
            generateFunctionCallCode(node);
            break;
    }
}

void generateAdditionCode(ASTNode* node) {
    ASTNode* left = node->getChild(0);
    ASTNode* right = node->getChild(1);
    
    generateCode(left);
    generateCode(right);
    
    // 生成加法指令
    string temp = generateTemp();
    emit("add " + temp + ", " + left->getAttribute("temp").getStringValue() + 
         ", " + right->getAttribute("temp").getStringValue());
    
    node->setAttribute("temp", temp);
}

void generateAssignmentCode(ASTNode* node) {
    ASTNode* left = node->getChild(0);
    ASTNode* right = node->getChild(1);
    
    generateCode(right);
    
    // 生成赋值指令
    emit("mov " + left->getAttribute("name").getStringValue() + 
         ", " + right->getAttribute("temp").getStringValue());
}

### 属性计算优化
```c
class OptimizedAttributeCalculator {
private:
    map<string, Attribute> attributeCache;
    
public:
    Attribute calculateAttribute(ASTNode* node, string attrName) {
        string key = node->getId() + "." + attrName;
        
        if (attributeCache.find(key) != attributeCache.end()) {
            return attributeCache[key];
        }
        
        Attribute attr = computeAttribute(node, attrName);
        attributeCache[key] = attr;
        
        return attr;
    }
    
    void clearCache() {
        attributeCache.clear();
    }
};
```

### 语义动作优化
```c
class OptimizedSemanticAction {
private:
    vector<SemanticAction> actions;
    
public:
    void optimizeActions() {
        // 合并相同动作
        mergeSimilarActions();
        
        // 消除冗余动作
        eliminateRedundantActions();
        
        // 优化动作顺序
        optimizeActionOrder();
    }
    
    void mergeSimilarActions() {
        // 合并相同的语义动作
        for (int i = 0; i < actions.size() - 1; i++) {
            if (actions[i].isSimilar(actions[i + 1])) {
                actions[i].merge(actions[i + 1]);
                actions.erase(actions.begin() + i + 1);
                i--;
            }
        }
    }
};
```
