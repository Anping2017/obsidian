# 符号表管理

## 🎯 符号表管理概述

**符号表管理**是语义分析的重要组成部分，负责存储和管理程序中标识符的信息，包括变量、函数、类型等。符号表为类型检查和代码生成提供必要的信息。

## 🔍 基本概念

### 符号表定义
符号表是存储程序中标识符信息的表，每个标识符对应一个符号表项。

### 符号表项
```c
struct SymbolEntry {
    string name;           // 标识符名称
    SymbolType type;      // 符号类型
    DataType dataType;    // 数据类型
    int scope;           // 作用域
    int offset;          // 存储偏移
    bool isInitialized;  // 是否已初始化
    vector<DataType> parameters; // 函数参数类型
    
    SymbolEntry(string n, SymbolType t, DataType dt, int s)
        : name(n), type(t), dataType(dt), scope(s), offset(0), isInitialized(false) {}
};

enum SymbolType {
    VARIABLE,
    FUNCTION,
    TYPE,
    CONSTANT
};

enum DataType {
    INT,
    FLOAT,
    CHAR,
    BOOL,
    ARRAY,
    STRUCT,
    POINTER
};
```

## 📊 符号表结构

### 线性符号表
```c
class LinearSymbolTable {
private:
    vector<SymbolEntry> entries;
    
public:
    void insert(SymbolEntry entry) {
        // 检查是否已存在
        if (lookup(entry.name) != nullptr) {
            reportError("Symbol already declared: " + entry.name);
            return;
        }
        
        entries.push_back(entry);
    }
    
    SymbolEntry* lookup(string name) {
        // 从后往前查找（后声明的优先）
        for (int i = entries.size() - 1; i >= 0; i--) {
            if (entries[i].name == name) {
                return &entries[i];
            }
        }
        return nullptr;
    }
    
    void remove(string name) {
        for (auto it = entries.begin(); it != entries.end(); ++it) {
            if (it->name == name) {
                entries.erase(it);
                break;
            }
        }
    }
    
    void clear() {
        entries.clear();
    }
};
```

### 哈希符号表
```c
class HashSymbolTable {
private:
    map<string, SymbolEntry> entries;
    
public:
    void insert(SymbolEntry entry) {
        if (entries.find(entry.name) != entries.end()) {
            reportError("Symbol already declared: " + entry.name);
            return;
        }
        
        entries[entry.name] = entry;
    }
    
    SymbolEntry* lookup(string name) {
        auto it = entries.find(name);
        if (it != entries.end()) {
            return &it->second;
        }
        return nullptr;
    }
    
    void remove(string name) {
        entries.erase(name);
    }
    
    void clear() {
        entries.clear();
    }
};
```

## 🔧 作用域管理

### 作用域栈
```c
class ScopeManager {
private:
    stack<SymbolTable*> scopeStack;
    int currentScope;
    
public:
    ScopeManager() : currentScope(0) {
        // 创建全局作用域
        enterScope();
    }
    
    void enterScope() {
        SymbolTable* newScope = new SymbolTable();
        scopeStack.push(newScope);
        currentScope++;
    }
    
    void exitScope() {
        if (scopeStack.size() > 1) {
            SymbolTable* scope = scopeStack.top();
            scopeStack.pop();
            delete scope;
            currentScope--;
        }
    }
    
    void insert(SymbolEntry entry) {
        entry.scope = currentScope;
        scopeStack.top()->insert(entry);
    }
    
    SymbolEntry* lookup(string name) {
        // 从当前作用域开始查找
        stack<SymbolTable*> tempStack = scopeStack;
        
        while (!tempStack.empty()) {
            SymbolEntry* entry = tempStack.top()->lookup(name);
            if (entry != nullptr) {
                return entry;
            }
            tempStack.pop();
        }
        
        return nullptr;
    }
    
    SymbolEntry* lookupInCurrentScope(string name) {
        return scopeStack.top()->lookup(name);
    }
    
    int getCurrentScope() const {
        return currentScope;
    }
};
```

### 作用域树
```c
class ScopeTreeNode {
private:
    SymbolTable symbolTable;
    vector<ScopeTreeNode*> children;
    ScopeTreeNode* parent;
    int scopeId;
    
public:
    ScopeTreeNode(int id, ScopeTreeNode* p = nullptr) 
        : scopeId(id), parent(p) {}
    
    void addChild(ScopeTreeNode* child) {
        children.push_back(child);
    }
    
    void insert(SymbolEntry entry) {
        symbolTable.insert(entry);
    }
    
    SymbolEntry* lookup(string name) {
        // 在当前作用域查找
        SymbolEntry* entry = symbolTable.lookup(name);
        if (entry != nullptr) {
            return entry;
        }
        
        // 在父作用域查找
        if (parent != nullptr) {
            return parent->lookup(name);
        }
        
        return nullptr;
    }
    
    SymbolEntry* lookupInCurrentScope(string name) {
        return symbolTable.lookup(name);
    }
    
    int getScopeId() const {
        return scopeId;
    }
    
    ScopeTreeNode* getParent() const {
        return parent;
    }
};

class ScopeTreeManager {
private:
    ScopeTreeNode* root;
    ScopeTreeNode* current;
    int nextScopeId;
    
public:
    ScopeTreeManager() : nextScopeId(0) {
        root = new ScopeTreeNode(nextScopeId++);
        current = root;
    }
    
    void enterScope() {
        ScopeTreeNode* newScope = new ScopeTreeNode(nextScopeId++, current);
        current->addChild(newScope);
        current = newScope;
    }
    
    void exitScope() {
        if (current != root) {
            current = current->getParent();
        }
    }
    
    void insert(SymbolEntry entry) {
        current->insert(entry);
    }
    
    SymbolEntry* lookup(string name) {
        return current->lookup(name);
    }
    
    SymbolEntry* lookupInCurrentScope(string name) {
        return current->lookupInCurrentScope(name);
    }
};
```

### 插入操作

```c
class SymbolTableManager {
private:
    ScopeManager scopeManager;
    
public:
    void insertVariable(string name, DataType type) {
        SymbolEntry entry(name, VARIABLE, type, scopeManager.getCurrentScope());
        scopeManager.insert(entry);

        scopeManager.insert(entry);

        SymbolEntry entry(name, CONSTANT, type, scopeManager.getCurrentScope());
        scopeManager.insert(entry);
    }
};
```

### 查找操作
```c
class SymbolLookup {
private:
    ScopeManager scopeManager;
    
public:
    SymbolEntry* lookup(string name) {
        return scopeManager.lookup(name);
    }
    
    SymbolEntry* lookupVariable(string name) {
        SymbolEntry* entry = scopeManager.lookup(name);
        if (entry != nullptr && entry->type == VARIABLE) {
            return entry;
        }
        return nullptr;
    }
    
    SymbolEntry* lookupFunction(string name) {
        SymbolEntry* entry = scopeManager.lookup(name);
        if (entry != nullptr && entry->type == FUNCTION) {
            return entry;
        }
        return nullptr;
    }
    
    SymbolEntry* lookupType(string name) {
        SymbolEntry* entry = scopeManager.lookup(name);
        if (entry != nullptr && entry->type == TYPE) {
            return entry;
        }
        return nullptr;
    }
    
    bool isDeclared(string name) {
        return scopeManager.lookup(name) != nullptr;
    }
    
    bool isDeclaredInCurrentScope(string name) {
        return scopeManager.lookupInCurrentScope(name) != nullptr;
    }
};
```

## 🔄 符号表生命周期

### 符号表构建
```c
class SymbolTableBuilder {
private:
    SymbolTableManager symbolManager;
    ASTNode* ast;
    
public:
    SymbolTableBuilder(ASTNode* ast) : ast(ast) {}
    
    void buildSymbolTable() {
        traverseAST(ast);
    }
    
    void traverseAST(ASTNode* node) {
        switch (node->getType()) {
            case DECLARATION:
                handleDeclaration(node);
                break;
            case FUNCTION_DEFINITION:
                handleFunctionDefinition(node);
                break;
            case BLOCK:
                handleBlock(node);
                break;
            case VARIABLE_DECLARATION:
                handleVariableDeclaration(node);
                break;
        }
        
        // 递归遍历子节点
        for (auto child : node->getChildren()) {
            traverseAST(child);
        }
    }
    
    void handleDeclaration(ASTNode* node) {
        string type = node->getChild(0)->getValue();
        string name = node->getChild(1)->getValue();
        
        DataType dataType = parseDataType(type);
        symbolManager.insertVariable(name, dataType);
    }
    
    void handleFunctionDefinition(ASTNode* node) {
        string returnType = node->getChild(0)->getValue();
        string name = node->getChild(1)->getValue();
        ASTNode* parameters = node->getChild(2);
        
        DataType dataType = parseDataType(returnType);
        vector<DataType> paramTypes = parseParameters(parameters);
        
        symbolManager.insertFunction(name, dataType, paramTypes);
        
        // 进入函数作用域
        symbolManager.enterScope();
        
        // 处理参数
        handleParameters(parameters);
    }
    
    void handleBlock(ASTNode* node) {
        symbolManager.enterScope();
        
        // 处理块内声明
        for (auto child : node->getChildren()) {
            traverseAST(child);
        }
        
        symbolManager.exitScope();
    }
    
    void handleVariableDeclaration(ASTNode* node) {
        string type = node->getChild(0)->getValue();
        string name = node->getChild(1)->getValue();
        
        DataType dataType = parseDataType(type);
        symbolManager.insertVariable(name, dataType);
    }
};
```

### 符号表销毁
```c
class SymbolTableDestructor {
private:
    ScopeManager scopeManager;
    
public:
    void destroySymbolTable() {
        // 清理所有作用域
        while (scopeManager.getCurrentScope() > 0) {
            scopeManager.exitScope();
        }
        
        // 清理全局作用域
        scopeManager.clear();
    }
    
    void cleanupScope(int scopeId) {
        // 清理指定作用域
        scopeManager.cleanupScope(scopeId);
    }
};
```

## 🎯 符号表优化

### 符号表压缩
```c
class SymbolTableCompressor {
private:
    map<string, int> symbolMap;
    vector<SymbolEntry> compressedEntries;
    
public:
    void compressSymbolTable(SymbolTable& table) {
        // 构建符号映射
        buildSymbolMap(table);
        
        // 压缩符号表
        compressEntries(table);
    }
    
    void buildSymbolMap(SymbolTable& table) {
        int index = 0;
        for (auto& entry : table.getEntries()) {
            if (symbolMap.find(entry.name) == symbolMap.end()) {
                symbolMap[entry.name] = index++;
            }
        }
    }
    
    void compressEntries(SymbolTable& table) {
        compressedEntries.clear();
        compressedEntries.resize(symbolMap.size());
        
        for (auto& entry : table.getEntries()) {
            int index = symbolMap[entry.name];
            compressedEntries[index] = entry;
        }
    }
    
    SymbolEntry* lookup(string name) {
        auto it = symbolMap.find(name);
        if (it != symbolMap.end()) {
            return &compressedEntries[it->second];
        }
        return nullptr;
    }
};
```

### 符号表缓存
```c
class SymbolTableCache {
private:
    map<string, SymbolEntry*> cache;
    int maxCacheSize;
    
public:
    SymbolTableCache(int maxSize = 1000) : maxCacheSize(maxSize) {}
    
    SymbolEntry* lookup(string name) {
        auto it = cache.find(name);
        if (it != cache.end()) {
            return it->second;
        }
        
        // 缓存未命中，从符号表查找
        SymbolEntry* entry = symbolTable.lookup(name);
        if (entry != nullptr) {
            // 添加到缓存
            if (cache.size() >= maxCacheSize) {
                evictOldestEntry();
            }
            cache[name] = entry;
        }
        
        return entry;
    }
    
    void evictOldestEntry() {
        // 移除最旧的缓存项
        if (!cache.empty()) {
            cache.erase(cache.begin());
        }
    }
    
    void clearCache() {
        cache.clear();
    }
};
```

## 📊 符号表应用

### 类型检查
```c
class TypeChecker {
private:
    SymbolTableManager symbolManager;
    
public:
    bool checkType(ASTNode* node) {
        switch (node->getType()) {
            case IDENTIFIER:
                return checkIdentifierType(node);
            case ASSIGNMENT:
                return checkAssignmentType(node);
            case FUNCTION_CALL:
                return checkFunctionCallType(node);
        }
        return true;
    }
    
    bool checkIdentifierType(ASTNode* node) {
        string name = node->getValue();
        SymbolEntry* entry = symbolManager.lookup(name);
        
        if (entry == nullptr) {
            reportError("Undeclared identifier: " + name);
            return false;
        }
        
        node->setAttribute("type", entry->dataType);
        return true;
    }
    
    bool checkAssignmentType(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        if (!checkType(left) || !checkType(right)) {
            return false;
        }
        
        DataType leftType = left->getAttribute("type").getDataType();
        DataType rightType = right->getAttribute("type").getDataType();
        
        if (!isCompatible(leftType, rightType)) {
            reportError("Type mismatch in assignment");
            return false;
        }
        
        return true;
    }
};
```

### 代码生成
```c
class CodeGenerator {
private:
    SymbolTableManager symbolManager;
    int nextOffset;
    
public:
    CodeGenerator() : nextOffset(0) {}
    
    void generateCode(ASTNode* node) {
        switch (node->getType()) {
            case VARIABLE_DECLARATION:
                generateVariableDeclaration(node);
                break;
            case FUNCTION_DEFINITION:
                generateFunctionDefinition(node);
                break;
            case ASSIGNMENT:
                generateAssignment(node);
                break;
        }
    }
    
    void generateVariableDeclaration(ASTNode* node) {
        string name = node->getChild(1)->getValue();
        SymbolEntry* entry = symbolManager.lookup(name);
        
        if (entry != nullptr) {
            entry->offset = nextOffset;
            nextOffset += getTypeSize(entry->dataType);
            
            emit("; Variable " + name + " at offset " + to_string(entry->offset));
        }
    }
    
    void generateFunctionDefinition(ASTNode* node) {
        string name = node->getChild(1)->getValue();
        SymbolEntry* entry = symbolManager.lookup(name);
        
        if (entry != nullptr) {
            emit(name + ":");
            emit("push ebp");
            emit("mov ebp, esp");
            
            // 生成函数体代码
            for (auto child : node->getChildren()) {
                generateCode(child);
            }
            
            emit("pop ebp");
            emit("ret");
        }
    }
};
```
