# 语义分析错误处理

## 🎯 语义分析错误处理概述

**语义分析错误处理**是语义分析的重要组成部分，负责检测、报告和恢复语义错误。语义错误处理确保编译器能够提供准确的错误信息，帮助程序员快速定位和修复问题。

## 🔍 基本概念

### 语义错误定义
语义错误是程序在语法正确但语义不正确的情况下产生的错误，包括类型错误、作用域错误、语义不一致等。

### 语义错误类型
```c
enum SemanticErrorType {
    TYPE_ERROR,           // 类型错误
    SCOPE_ERROR,          // 作用域错误
    SEMANTIC_ERROR,       // 语义错误
    SYMBOL_ERROR,         // 符号错误
    DECLARATION_ERROR,    // 声明错误
    USAGE_ERROR,          // 使用错误
    CONSISTENCY_ERROR     // 一致性错误
};

struct SemanticError {
    SemanticErrorType type;
    string message;
    int line;
    int column;
    string context;
    ASTNode* node;
    
    SemanticError(SemanticErrorType t, string msg, int l, int c, string ctx = "", ASTNode* n = nullptr)
        : type(t), message(msg), line(l), column(c), context(ctx), node(n) {}
};
```

## 📊 语义错误检测

### 基本语义错误检测器
```c
class SemanticErrorDetector {
private:
    vector<SemanticError> errors;
    ASTNode* ast;
    
public:
    SemanticErrorDetector(ASTNode* ast) : ast(ast) {}
    
    void detectErrors() {
        traverseAST(ast);
    }
    
    void traverseAST(ASTNode* node) {
        switch (node->getType()) {
            case VARIABLE_DECLARATION:
                detectVariableDeclarationErrors(node);
                break;
            case FUNCTION_DEFINITION:
                detectFunctionDefinitionErrors(node);
                break;
            case ASSIGNMENT:
                detectAssignmentErrors(node);
                break;
            case FUNCTION_CALL:
                detectFunctionCallErrors(node);
                break;
            case RETURN_STATEMENT:
                detectReturnStatementErrors(node);
                break;
        }
        
        // 递归遍历子节点
        for (auto child : node->getChildren()) {
            traverseAST(child);
        }
    }
    
    void detectVariableDeclarationErrors(ASTNode* node) {
        string typeName = node->getChild(0)->getValue();
        string varName = node->getChild(1)->getValue();
        
        // 检查类型是否存在
        if (!isValidType(typeName)) {
            addError(TYPE_ERROR, "Unknown type: " + typeName, node->getLine(), node->getColumn());
        }
        
        // 检查变量是否已声明
        if (isVariableDeclared(varName)) {
            addError(DECLARATION_ERROR, "Variable already declared: " + varName, node->getLine(), node->getColumn());
        }
        
        // 检查变量名是否有效
        if (!isValidIdentifier(varName)) {
            addError(SYMBOL_ERROR, "Invalid variable name: " + varName, node->getLine(), node->getColumn());
        }
    }
    
    void detectFunctionDefinitionErrors(ASTNode* node) {
        string funcName = node->getChild(1)->getValue();
        string returnType = node->getChild(0)->getValue();
        
        // 检查返回类型
        if (!isValidType(returnType)) {
            addError(TYPE_ERROR, "Unknown return type: " + returnType, node->getLine(), node->getColumn());
        }
        
        // 检查函数名是否有效
        if (!isValidIdentifier(funcName)) {
            addError(SYMBOL_ERROR, "Invalid function name: " + funcName, node->getLine(), node->getColumn());
        }
        
        // 检查函数是否已定义
        if (isFunctionDefined(funcName)) {
            addError(DECLARATION_ERROR, "Function already defined: " + funcName, node->getLine(), node->getColumn());
        }
        
        // 检查参数
        ASTNode* parameters = node->getChild(2);
        detectParameterErrors(parameters);
    }
    
    void detectAssignmentErrors(ASTNode* node) {
        string varName = node->getChild(0)->getValue();
        ASTNode* value = node->getChild(1);
        
        // 检查变量是否已声明
        if (!isVariableDeclared(varName)) {
            addError(SCOPE_ERROR, "Undeclared variable: " + varName, node->getLine(), node->getColumn());
            return;
        }
        
        // 检查类型兼容性
        DataType varType = getVariableType(varName);
        DataType valueType = getExpressionType(value);
        
        if (!isTypeCompatible(varType, valueType)) {
            addError(TYPE_ERROR, "Type mismatch in assignment: " + varName, node->getLine(), node->getColumn());
        }
    }
    
    void detectFunctionCallErrors(ASTNode* node) {
        string funcName = node->getChild(0)->getValue();
        ASTNode* arguments = node->getChild(1);
        
        // 检查函数是否已声明
        if (!isFunctionDeclared(funcName)) {
            addError(SCOPE_ERROR, "Undeclared function: " + funcName, node->getLine(), node->getColumn());
            return;
        }
        
        // 检查参数数量
        int expectedArgs = getFunctionParameterCount(funcName);
        int actualArgs = arguments->getChildren().size();
        
        if (expectedArgs != actualArgs) {
            addError(SEMANTIC_ERROR, "Wrong number of arguments for function: " + funcName, node->getLine(), node->getColumn());
        }
        
        // 检查参数类型
        vector<DataType> expectedTypes = getFunctionParameterTypes(funcName);
        for (int i = 0; i < actualArgs; i++) {
            DataType expectedType = expectedTypes[i];
            DataType actualType = getExpressionType(arguments->getChild(i));
            
            if (!isTypeCompatible(expectedType, actualType)) {
                addError(TYPE_ERROR, "Type mismatch in function argument: " + funcName, node->getLine(), node->getColumn());
            }
        }
    }
    
    void detectReturnStatementErrors(ASTNode* node) {
        ASTNode* value = node->getChild(0);
        DataType returnType = getCurrentFunctionReturnType();
        DataType valueType = getExpressionType(value);
        
        if (!isTypeCompatible(returnType, valueType)) {
            addError(TYPE_ERROR, "Type mismatch in return statement", node->getLine(), node->getColumn());
        }
    }
    
    void detectParameterErrors(ASTNode* node) {
        for (auto param : node->getChildren()) {
            string typeName = param->getChild(0)->getValue();
            string paramName = param->getChild(1)->getValue();
            
            // 检查参数类型
            if (!isValidType(typeName)) {
                addError(TYPE_ERROR, "Unknown parameter type: " + typeName, param->getLine(), param->getColumn());
            }
            
            // 检查参数名
            if (!isValidIdentifier(paramName)) {
                addError(SYMBOL_ERROR, "Invalid parameter name: " + paramName, param->getLine(), param->getColumn());
            }
        }
    }
    
    void addError(SemanticErrorType type, string message, int line, int column, string context = "") {
        SemanticError error(type, message, line, column, context);
        errors.push_back(error);
    }
    
    vector<SemanticError> getErrors() {
        return errors;
    }
    
    bool hasErrors() {
        return !errors.empty();
    }
};
```

```c
class AdvancedSemanticErrorDetector {
private:
    SemanticErrorDetector baseDetector;
    map<string, vector<SemanticError>> errorGroups;
    
public:
    void detectAdvancedErrors(ASTNode* node) {
        switch (node->getType()) {
            case CLASS_DEFINITION:
                detectClassDefinitionErrors(node);
                break;
            case INHERITANCE:
                detectInheritanceErrors(node);
                break;
            case POLYMORPHISM:
                detectPolymorphismErrors(node);
                break;
            case TEMPLATE_DEFINITION:
                detectTemplateDefinitionErrors(node);
                break;
            case GENERIC_INSTANTIATION:
                detectGenericInstantiationErrors(node);
                break;
        }
        
        baseDetector.traverseAST(node);
    }
    
    void detectClassDefinitionErrors(ASTNode* node) {
        string className = node->getChild(0)->getValue();
        
        // 检查类名是否有效
        if (!isValidIdentifier(className)) {
            addError(SYMBOL_ERROR, "Invalid class name: " + className, node->getLine(), node->getColumn());
        }
        
        // 检查类是否已定义
        if (isClassDefined(className)) {
            addError(DECLARATION_ERROR, "Class already defined: " + className, node->getLine(), node->getColumn());
        }
        
        // 检查类成员
        ASTNode* members = node->getChild(1);
        detectClassMemberErrors(members);
    }
    
    void detectInheritanceErrors(ASTNode* node) {
        string derivedClass = node->getChild(0)->getValue();
        string baseClass = node->getChild(1)->getValue();
        
        // 检查基类是否存在
        if (!isClassDefined(baseClass)) {
            addError(SCOPE_ERROR, "Unknown base class: " + baseClass, node->getLine(), node->getColumn());
        }
        
        // 检查循环继承
        if (hasCircularInheritance(derivedClass, baseClass)) {
            addError(SEMANTIC_ERROR, "Circular inheritance detected", node->getLine(), node->getColumn());
        }
        
        // 检查多重继承冲突
        if (hasMultipleInheritanceConflict(derivedClass, baseClass)) {
            addError(SEMANTIC_ERROR, "Multiple inheritance conflict", node->getLine(), node->getColumn());
        }
    }
    
    void detectPolymorphismErrors(ASTNode* node) {
        string methodName = node->getChild(0)->getValue();
        string className = node->getChild(1)->getValue();
        
        // 检查方法是否已定义
        if (!isMethodDefined(methodName, className)) {
            addError(SCOPE_ERROR, "Method not defined: " + methodName, node->getLine(), node->getColumn());
        }
        
        // 检查虚函数重写
        if (isVirtualMethodOverride(methodName, className)) {
            detectVirtualMethodOverrideErrors(node);
        }
    }
    
    void detectTemplateDefinitionErrors(ASTNode* node) {
        string templateName = node->getChild(0)->getValue();
        ASTNode* templateParams = node->getChild(1);
        
        // 检查模板参数
        for (auto param : templateParams->getChildren()) {
            string paramName = param->getValue();
            
            if (!isValidTemplateParameter(paramName)) {
                addError(SYMBOL_ERROR, "Invalid template parameter: " + paramName, param->getLine(), param->getColumn());
            }
        }
        
        // 检查模板是否已定义
        if (isTemplateDefined(templateName)) {
            addError(DECLARATION_ERROR, "Template already defined: " + templateName, node->getLine(), node->getColumn());
        }
    }
    
    void detectGenericInstantiationErrors(ASTNode* node) {
        string templateName = node->getChild(0)->getValue();
        ASTNode* typeArgs = node->getChild(1);
        
        // 检查模板是否存在
        if (!isTemplateDefined(templateName)) {
            addError(SCOPE_ERROR, "Unknown template: " + templateName, node->getLine(), node->getColumn());
            return;
        }
        
        // 检查类型参数
        vector<DataType> expectedTypes = getTemplateParameterTypes(templateName);
        vector<DataType> actualTypes;
        
        for (auto typeArg : typeArgs->getChildren()) {
            DataType type = parseDataType(typeArg->getValue());
            actualTypes.push_back(type);
        }
        
        if (expectedTypes.size() != actualTypes.size()) {
            addError(SEMANTIC_ERROR, "Wrong number of type arguments for template: " + templateName, node->getLine(), node->getColumn());
        }
        
        // 检查类型兼容性
        for (int i = 0; i < expectedTypes.size(); i++) {
            if (!isTypeCompatible(expectedTypes[i], actualTypes[i])) {
                addError(TYPE_ERROR, "Type mismatch in template instantiation: " + templateName, node->getLine(), node->getColumn());
            }
        }
    }
    
    void detectClassMemberErrors(ASTNode* node) {
        for (auto member : node->getChildren()) {
            switch (member->getType()) {
                case MEMBER_VARIABLE:
                    detectMemberVariableErrors(member);
                    break;
                case MEMBER_FUNCTION:
                    detectMemberFunctionErrors(member);
                    break;
                case CONSTRUCTOR:
                    detectConstructorErrors(member);
                    break;
                case DESTRUCTOR:
                    detectDestructorErrors(member);
                    break;
            }
        }
    }
    
    void detectMemberVariableErrors(ASTNode* node) {
        string typeName = node->getChild(0)->getValue();
        string varName = node->getChild(1)->getValue();
        
        // 检查成员变量类型
        if (!isValidType(typeName)) {
            addError(TYPE_ERROR, "Unknown member variable type: " + typeName, node->getLine(), node->getColumn());
        }
        
        // 检查成员变量名
        if (!isValidIdentifier(varName)) {
            addError(SYMBOL_ERROR, "Invalid member variable name: " + varName, node->getLine(), node->getColumn());
        }
    }
    
    void detectMemberFunctionErrors(ASTNode* node) {
        string returnType = node->getChild(0)->getValue();
        string funcName = node->getChild(1)->getValue();
        
        // 检查返回类型
        if (!isValidType(returnType)) {
            addError(TYPE_ERROR, "Unknown member function return type: " + returnType, node->getLine(), node->getColumn());
        }
        
        // 检查函数名
        if (!isValidIdentifier(funcName)) {
            addError(SYMBOL_ERROR, "Invalid member function name: " + funcName, node->getLine(), node->getColumn());
        }
        
        // 检查参数
        ASTNode* parameters = node->getChild(2);
        detectParameterErrors(parameters);
    }
    
    void detectConstructorErrors(ASTNode* node) {
        string className = node->getChild(0)->getValue();
        
        // 检查构造函数名
        if (!isValidIdentifier(className)) {
            addError(SYMBOL_ERROR, "Invalid constructor name: " + className, node->getLine(), node->getColumn());
        }
        
        // 检查参数
        ASTNode* parameters = node->getChild(1);
        detectParameterErrors(parameters);
    }
    
    void detectDestructorErrors(ASTNode* node) {
        string className = node->getChild(0)->getValue();
        
        // 检查析构函数名
        if (!isValidIdentifier(className)) {
            addError(SYMBOL_ERROR, "Invalid destructor name: " + className, node->getLine(), node->getColumn());
        }
    }
    
    void detectVirtualMethodOverrideErrors(ASTNode* node) {
        string methodName = node->getChild(0)->getValue();
        string className = node->getChild(1)->getValue();
        
        // 检查虚函数重写
        if (isVirtualMethodOverride(methodName, className)) {
            // 检查方法签名是否匹配
            if (!isMethodSignatureMatch(methodName, className)) {
                addError(SEMANTIC_ERROR, "Method signature mismatch in override: " + methodName, node->getLine(), node->getColumn());
            }
        }
    }
};
```

## 🔧 语义错误报告

### 基本错误报告器
```c
class SemanticErrorReporter {
private:
    vector<SemanticError> errors;
    string sourceCode;
    
public:
    SemanticErrorReporter(string source) : sourceCode(source) {}
    
    void reportErrors() {
        for (auto error : errors) {
            reportError(error);
        }
    }
    
    void reportError(SemanticError error) {
        cout << "Semantic Error: " << error.message << endl;
        cout << "Location: Line " << error.line << ", Column " << error.column << endl;
        
        if (!error.context.empty()) {
            cout << "Context: " << error.context << endl;
        }
        
        // 显示错误上下文
        displayErrorContext(error);
        
        cout << endl;
    }
    
    void displayErrorContext(SemanticError error) {
        vector<string> lines = splitLines(sourceCode);
        
        if (error.line > 0 && error.line <= lines.size()) {
            string line = lines[error.line - 1];
            cout << "Source: " << line << endl;
            
            // 显示错误位置
            string pointer = string(error.column - 1, ' ') + "^";
            cout << "        " << pointer << endl;
        }
    }
    
    void addError(SemanticError error) {
        errors.push_back(error);
    }
    
    vector<SemanticError> getErrors() {
        return errors;
    }
    
    bool hasErrors() {
        return !errors.empty();
    }
    
    int getErrorCount() {
        return errors.size();
    }
    
    vector<string> splitLines(string text) {
        vector<string> lines;
        stringstream ss(text);
        string line;
        
        while (getline(ss, line)) {
            lines.push_back(line);
        }
        
        return lines;
    }
};
```

```c
class AdvancedSemanticErrorReporter {
private:
    SemanticErrorReporter baseReporter;
    map<SemanticErrorType, vector<SemanticError>> errorGroups;
    
public:
    void reportErrors() {
        // 按错误类型分组
        groupErrorsByType();
        
        // 报告每种类型的错误
        for (auto& pair : errorGroups) {
            SemanticErrorType type = pair.first;
            vector<SemanticError> errors = pair.second;
            
            reportErrorGroup(type, errors);
        }
    }
    
    void groupErrorsByType() {
        for (auto error : baseReporter.getErrors()) {
            errorGroups[error.type].push_back(error);
        }
    }
    
    void reportErrorGroup(SemanticErrorType type, vector<SemanticError> errors) {
        cout << "=== " << getErrorTypeName(type) << " Errors ===" << endl;
        cout << "Count: " << errors.size() << endl;
        cout << endl;
        
        for (auto error : errors) {
            reportError(error);
        }
        
        cout << endl;
    }
    
    void reportError(SemanticError error) {
        cout << "Error: " << error.message << endl;
        cout << "Location: Line " << error.line << ", Column " << error.column << endl;
        
        if (!error.context.empty()) {
            cout << "Context: " << error.context << endl;
        }
        
        // 显示错误上下文
        displayErrorContext(error);
        
        // 显示建议修复
        displayFixSuggestion(error);
        
        cout << endl;
    }
    
    void displayErrorContext(SemanticError error) {
        vector<string> lines = splitLines(baseReporter.getSourceCode());
        
        if (error.line > 0 && error.line <= lines.size()) {
            string line = lines[error.line - 1];
            cout << "Source: " << line << endl;
            
            // 显示错误位置
            string pointer = string(error.column - 1, ' ') + "^";
            cout << "        " << pointer << endl;
        }
    }
    
    void displayFixSuggestion(SemanticError error) {
        string suggestion = getFixSuggestion(error);
        if (!suggestion.empty()) {
            cout << "Suggestion: " << suggestion << endl;
        }
    }
    
    string getFixSuggestion(SemanticError error) {
        switch (error.type) {
            case TYPE_ERROR:
                return "Check the type declaration and ensure compatibility";
            case SCOPE_ERROR:
                return "Declare the symbol before using it";
            case SEMANTIC_ERROR:
                return "Check the semantic rules and constraints";
            case SYMBOL_ERROR:
                return "Use a valid identifier name";
            case DECLARATION_ERROR:
                return "Remove duplicate declaration or use different name";
            case USAGE_ERROR:
                return "Check the usage context and requirements";
            case CONSISTENCY_ERROR:
                return "Ensure consistency across the program";
            default:
                return "";
        }
    }
    
    string getErrorTypeName(SemanticErrorType type) {
        switch (type) {
            case TYPE_ERROR: return "Type";
            case SCOPE_ERROR: return "Scope";
            case SEMANTIC_ERROR: return "Semantic";
            case SYMBOL_ERROR: return "Symbol";
            case DECLARATION_ERROR: return "Declaration";
            case USAGE_ERROR: return "Usage";
            case CONSISTENCY_ERROR: return "Consistency";
            default: return "Unknown";
        }
    }
};
```

```c
    STRICT,      // 严格模式：遇到错误立即停止
    PERMISSIVE,  // 宽松模式：尝试恢复并继续
    INTERACTIVE  // 交互模式：询问用户如何处理
};

class SemanticErrorHandler {
private:
    ErrorHandlingStrategy strategy;
    vector<SemanticError> errors;
    
public:
    SemanticErrorHandler(ErrorHandlingStrategy s) : strategy(s) {}
    
    void handleErrors() {
        switch (strategy) {
            case STRICT:
                handleStrictErrors();
                break;
            case PERMISSIVE:
                handlePermissiveErrors();
                break;
            case INTERACTIVE:
                handleInteractiveErrors();
                break;
        }
    }
    
    void handleStrictErrors() {
        // 严格模式：遇到错误立即停止
        for (auto error : errors) {
            reportError(error);
            if (isFatalError(error)) {
                stopCompilation();
            }
        }
    }
    
    void handlePermissiveErrors() {
        // 宽松模式：尝试恢复并继续
        for (auto error : errors) {
            reportError(error);
            if (canRecoverFromError(error)) {
                recoverFromError(error);
            } else {
                reportWarning("Cannot recover from error: " + error.message);
            }
        }
    }
    
    void handleInteractiveErrors() {
        // 交互模式：询问用户如何处理
        for (auto error : errors) {
            reportError(error);
            
            cout << "How would you like to handle this error?" << endl;

            cout << "2. Try to recover" << endl;
```
