# 类型检查系统

## 🎯 类型检查系统概述

**类型检查系统**是语义分析的核心组件，负责检查程序中类型使用的正确性，确保类型安全和语义一致性。类型检查系统是编译器的重要组成部分。

## 🔍 基本概念

### 类型系统
类型系统是一组规则，用于定义程序中值的类型以及如何操作这些值。

### 类型检查
类型检查是验证程序中类型使用是否符合类型系统规则的过程。

### 类型定义
```c
enum DataType {
    INT,
    FLOAT,
    CHAR,
    BOOL,
    ARRAY,
    STRUCT,
    POINTER,
    FUNCTION,
    VOID
};

struct TypeInfo {
    DataType baseType;
    int size;
    bool isConst;
    bool isVolatile;
    vector<TypeInfo> arrayDimensions;
    vector<TypeInfo> structMembers;
    vector<TypeInfo> functionParameters;
    TypeInfo returnType;
    
    TypeInfo(DataType type) : baseType(type), size(0), isConst(false), isVolatile(false) {}
};
```

## 📊 类型检查规则

### 基本类型规则
```c
class BasicTypeRules {
public:
    bool isCompatible(TypeInfo type1, TypeInfo type2) {
        // 相同类型
        if (type1.baseType == type2.baseType) {
            return true;
        }
        
        // 数值类型兼容性
        if (isNumericType(type1.baseType) && isNumericType(type2.baseType)) {
            return true;
        }
        
        // 指针类型兼容性
        if (type1.baseType == POINTER && type2.baseType == POINTER) {
            return isCompatible(type1.returnType, type2.returnType);
        }
        
        return false;
    }
    
    bool isNumericType(DataType type) {
        return type == INT || type == FLOAT || type == CHAR;
    }
    
    bool isAssignable(TypeInfo left, TypeInfo right) {
        // 左值必须可赋值
        if (!isLValue(left)) {
            return false;
        }
        
        // 类型必须兼容
        return isCompatible(left, right);
    }
    
    bool isLValue(TypeInfo type) {
        // 常量不能作为左值
        if (type.isConst) {
            return false;
        }
        
        // 函数不能作为左值
        if (type.baseType == FUNCTION) {
            return false;
        }
        
        return true;
    }
};
```

### 表达式类型规则
```c
class ExpressionTypeRules {
public:
    TypeInfo getExpressionType(ASTNode* node) {
        switch (node->getType()) {
            case ADDITION:
                return getAdditionType(node);
            case SUBTRACTION:
                return getSubtractionType(node);
            case MULTIPLICATION:
                return getMultiplicationType(node);
            case DIVISION:
                return getDivisionType(node);
            case ASSIGNMENT:
                return getAssignmentType(node);
            case IDENTIFIER:
                return getIdentifierType(node);
            case NUMBER:
                return getNumberType(node);
            case FUNCTION_CALL:
                return getFunctionCallType(node);
        }
        return TypeInfo(VOID);
    }
    
    TypeInfo getAdditionType(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        TypeInfo leftType = getExpressionType(left);
        TypeInfo rightType = getExpressionType(right);
        
        // 检查类型兼容性
        if (!isCompatible(leftType, rightType)) {
            reportError("Type mismatch in addition");
            return TypeInfo(VOID);
        }
        
        // 返回结果类型
        return promoteType(leftType, rightType);
    }
    
    TypeInfo getAssignmentType(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        TypeInfo leftType = getExpressionType(left);
        TypeInfo rightType = getExpressionType(right);
        
        // 检查左值
        if (!isLValue(leftType)) {
            reportError("Left operand must be an lvalue");
            return TypeInfo(VOID);
        }
        
        // 检查类型兼容性
        if (!isCompatible(leftType, rightType)) {
            reportError("Type mismatch in assignment");
            return TypeInfo(VOID);
        }
        
        return leftType;
    }
    
    TypeInfo promoteType(TypeInfo type1, TypeInfo type2) {
        // 类型提升规则
        if (type1.baseType == FLOAT || type2.baseType == FLOAT) {
            return TypeInfo(FLOAT);
        }
        if (type1.baseType == INT || type2.baseType == INT) {
            return TypeInfo(INT);
        }
        return TypeInfo(CHAR);
    }
};
```

## 🔧 类型检查器实现

### 基本类型检查器
```c
class TypeChecker {
private:
    SymbolTableManager symbolManager;
    ExpressionTypeRules expressionRules;
    
public:
    bool checkTypes(ASTNode* node) {
        switch (node->getType()) {
            case DECLARATION:
                return checkDeclaration(node);
            case ASSIGNMENT:
                return checkAssignment(node);
            case EXPRESSION:
                return checkExpression(node);
            case FUNCTION_CALL:
                return checkFunctionCall(node);
            case RETURN_STATEMENT:
                return checkReturnStatement(node);
        }
        
        // 递归检查子节点
        for (auto child : node->getChildren()) {
            if (!checkTypes(child)) {
                return false;
            }
        }
        
        return true;
    }
    
    bool checkDeclaration(ASTNode* node) {
        string typeName = node->getChild(0)->getValue();
        string varName = node->getChild(1)->getValue();
        
        // 检查类型是否存在
        TypeInfo type = getTypeInfo(typeName);
        if (type.baseType == VOID) {
            reportError("Unknown type: " + typeName);
            return false;
        }
        
        // 检查变量是否已声明
        if (symbolManager.isDeclaredInCurrentScope(varName)) {
            reportError("Variable already declared: " + varName);
            return false;
        }
        
        // 添加到符号表
        symbolManager.insertVariable(varName, type);
        
        return true;
    }
    
    bool checkAssignment(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        // 检查左值
        if (left->getType() != IDENTIFIER) {
            reportError("Left operand must be an identifier");
            return false;
        }
        
        // 检查变量是否声明
        string varName = left->getValue();
        SymbolEntry* entry = symbolManager.lookup(varName);
        if (entry == nullptr) {
            reportError("Undeclared variable: " + varName);
            return false;
        }
        
        // 检查类型兼容性
        TypeInfo leftType = entry->dataType;
        TypeInfo rightType = expressionRules.getExpressionType(right);
        
        if (!isCompatible(leftType, rightType)) {
            reportError("Type mismatch in assignment");
            return false;
        }
        
        return true;
    }
    
    bool checkFunctionCall(ASTNode* node) {
        string funcName = node->getChild(0)->getValue();
        ASTNode* arguments = node->getChild(1);
        
        // 检查函数是否声明
        SymbolEntry* entry = symbolManager.lookupFunction(funcName);
        if (entry == nullptr) {
            reportError("Undeclared function: " + funcName);
            return false;
        }
        
        // 检查参数数量和类型
        vector<TypeInfo> expectedTypes = entry->parameters;
        vector<TypeInfo> actualTypes = getArgumentTypes(arguments);
        
        if (expectedTypes.size() != actualTypes.size()) {
            reportError("Wrong number of arguments");
            return false;
        }
        
        for (int i = 0; i < expectedTypes.size(); i++) {
            if (!isCompatible(expectedTypes[i], actualTypes[i])) {
                reportError("Type mismatch in argument " + to_string(i + 1));
                return false;
            }
        }
        
        return true;
    }
};
```

### 高级类型检查器
```c
class AdvancedTypeChecker {
private:
    TypeChecker basicChecker;
    map<string, TypeInfo> typeAliases;
    
public:
    bool checkTypes(ASTNode* node) {
        switch (node->getType()) {
            case TYPE_ALIAS:
                return checkTypeAlias(node);
            case STRUCT_DEFINITION:
                return checkStructDefinition(node);
            case ARRAY_ACCESS:
                return checkArrayAccess(node);
            case POINTER_DEREFERENCE:
                return checkPointerDereference(node);
            case TYPE_CAST:
                return checkTypeCast(node);
        }
        
        return basicChecker.checkTypes(node);
    }
    
    bool checkTypeAlias(ASTNode* node) {
        string aliasName = node->getChild(0)->getValue();
        string originalType = node->getChild(1)->getValue();
        
        TypeInfo type = getTypeInfo(originalType);
        if (type.baseType == VOID) {
            reportError("Unknown type: " + originalType);
            return false;
        }
        
        typeAliases[aliasName] = type;
        return true;
    }
    
    bool checkStructDefinition(ASTNode* node) {
        string structName = node->getChild(0)->getValue();
        ASTNode* members = node->getChild(1);
        
        TypeInfo structType(STRUCT);
        structType.size = 0;
        
        // 检查成员
        for (auto member : members->getChildren()) {
            string memberType = member->getChild(0)->getValue();
            string memberName = member->getChild(1)->getValue();
            
            TypeInfo type = getTypeInfo(memberType);
            if (type.baseType == VOID) {
                reportError("Unknown type: " + memberType);
                return false;
            }
            
            structType.structMembers.push_back(type);
            structType.size += type.size;
        }
        
        symbolManager.insertType(structName, structType);
        return true;
    }
    
    bool checkArrayAccess(ASTNode* node) {
        ASTNode* array = node->getChild(0);
        ASTNode* index = node->getChild(1);
        
        TypeInfo arrayType = expressionRules.getExpressionType(array);
        TypeInfo indexType = expressionRules.getExpressionType(index);
        
        // 检查数组类型
        if (arrayType.baseType != ARRAY) {
            reportError("Array access on non-array type");
            return false;
        }
        
        // 检查索引类型
        if (indexType.baseType != INT) {
            reportError("Array index must be integer");
            return false;
        }
        
        return true;
    }
    
    bool checkPointerDereference(ASTNode* node) {
        ASTNode* pointer = node->getChild(0);
        
        TypeInfo pointerType = expressionRules.getExpressionType(pointer);
        
        // 检查指针类型
        if (pointerType.baseType != POINTER) {
            reportError("Dereference on non-pointer type");
            return false;
        }
        
        return true;
    }
    
    bool checkTypeCast(ASTNode* node) {
        string targetType = node->getChild(0)->getValue();
        ASTNode* expression = node->getChild(1);
        
        TypeInfo targetTypeInfo = getTypeInfo(targetType);
        TypeInfo expressionType = expressionRules.getExpressionType(expression);
        
        // 检查类型转换是否合法
        if (!isCastable(expressionType, targetTypeInfo)) {
            reportError("Invalid type cast");
            return false;
        }
        
        return true;
    }
};
```

## 📈 类型推导

### 类型推导算法
```c
class TypeInference {
private:
    map<string, TypeInfo> typeConstraints;
    map<string, TypeInfo> typeVariables;
    
public:
    TypeInfo inferType(ASTNode* node) {
        switch (node->getType()) {
            case ADDITION:
                return inferAdditionType(node);
            case ASSIGNMENT:
                return inferAssignmentType(node);
            case FUNCTION_CALL:
                return inferFunctionCallType(node);
            case IDENTIFIER:
                return inferIdentifierType(node);
        }
        return TypeInfo(VOID);
    }
    
    TypeInfo inferAdditionType(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        TypeInfo leftType = inferType(left);
        TypeInfo rightType = inferType(right);
        
        // 创建类型约束
        addConstraint(leftType, rightType);
        
        // 推导结果类型
        return unifyTypes(leftType, rightType);
    }
    
    TypeInfo inferAssignmentType(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        TypeInfo leftType = inferType(left);
        TypeInfo rightType = inferType(right);
        
        // 添加约束：左值类型 = 右值类型
        addConstraint(leftType, rightType);
        
        return leftType;
    }
    
    void addConstraint(TypeInfo type1, TypeInfo type2) {
        // 添加类型约束
        string key = type1.toString() + "=" + type2.toString();
        typeConstraints[key] = type1;
    }
    
    TypeInfo unifyTypes(TypeInfo type1, TypeInfo type2) {
        // 类型统一算法
        if (type1.baseType == type2.baseType) {
            return type1;
        }
        
        // 数值类型统一
        if (isNumericType(type1.baseType) && isNumericType(type2.baseType)) {
            return promoteType(type1, type2);
        }
        
        // 无法统一
        return TypeInfo(VOID);
    }
};
```

### 类型检查缓存
```c
class TypeCheckCache {
private:
    map<string, TypeInfo> typeCache;
    map<string, bool> compatibilityCache;
    
public:
    TypeInfo getCachedType(ASTNode* node) {
        string key = node->toString();
        auto it = typeCache.find(key);
        if (it != typeCache.end()) {
            return it->second;
        }
        
        TypeInfo type = inferType(node);
        typeCache[key] = type;
        return type;
    }
    
    bool getCachedCompatibility(TypeInfo type1, TypeInfo type2) {
        string key = type1.toString() + "=" + type2.toString();
        auto it = compatibilityCache.find(key);
        if (it != compatibilityCache.end()) {
            return it->second;
        }
        
        bool compatible = isCompatible(type1, type2);
        compatibilityCache[key] = compatible;
        return compatible;
    }
    
    void clearCache() {
        typeCache.clear();
        compatibilityCache.clear();
    }
};
```

### 错误检测
```c
class TypeErrorDetector {
public:
    void detectTypeErrors(ASTNode* node) {
        switch (node->getType()) {
            case ASSIGNMENT:
                detectAssignmentErrors(node);
                break;
            case FUNCTION_CALL:
                detectFunctionCallErrors(node);
                break;
            case ARRAY_ACCESS:
                detectArrayAccessErrors(node);
                break;
        }
        
        // 递归检测子节点
        for (auto child : node->getChildren()) {
            detectTypeErrors(child);
        }
    }
    
    void detectAssignmentErrors(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        TypeInfo leftType = getExpressionType(left);
        TypeInfo rightType = getExpressionType(right);
        
        if (!isCompatible(leftType, rightType)) {
            reportError("Type mismatch in assignment", node);
        }
        
        if (!isLValue(leftType)) {
            reportError("Left operand must be an lvalue", left);
        }
    }
    
    void detectFunctionCallErrors(ASTNode* node) {
        string funcName = node->getChild(0)->getValue();
        ASTNode* arguments = node->getChild(1);
        
        SymbolEntry* entry = symbolManager.lookupFunction(funcName);
        if (entry == nullptr) {
            reportError("Undeclared function: " + funcName, node);
            return;
        }
        
        vector<TypeInfo> expectedTypes = entry->parameters;
        vector<TypeInfo> actualTypes = getArgumentTypes(arguments);
        
        if (expectedTypes.size() != actualTypes.size()) {
            reportError("Wrong number of arguments", node);
            return;
        }
        
        for (int i = 0; i < expectedTypes.size(); i++) {
            if (!isCompatible(expectedTypes[i], actualTypes[i])) {
                reportError("Type mismatch in argument " + to_string(i + 1), node);
            }
        }
    }
};
```

### 类型优化
```c
class TypeOptimizer {
public:
    void optimizeTypes(ASTNode* node) {
        switch (node->getType()) {
            case ADDITION:
                optimizeAddition(node);
                break;
            case MULTIPLICATION:
                optimizeMultiplication(node);
                break;
            case TYPE_CAST:
                optimizeTypeCast(node);
                break;
        }
        
        // 递归优化子节点
        for (auto child : node->getChildren()) {
            optimizeTypes(child);
        }
    }
    
    void optimizeAddition(ASTNode* node) {
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        TypeInfo leftType = getExpressionType(left);
        TypeInfo rightType = getExpressionType(right);
        
        // 如果两个操作数都是常量，进行常量折叠
        if (isConstant(left) && isConstant(right)) {
            int result = getConstantValue(left) + getConstantValue(right);
            node->setType(NUMBER);
            node->setValue(to_string(result));
        }
    }
    
    void optimizeTypeCast(ASTNode* node) {
        string targetType = node->getChild(0)->getValue();
        ASTNode* expression = node->getChild(1);
        
        TypeInfo targetTypeInfo = getTypeInfo(targetType);
        TypeInfo expressionType = getExpressionType(expression);
        
        // 如果类型相同，移除类型转换
        if (targetTypeInfo.baseType == expressionType.baseType) {
            node->replaceWith(expression);
        }
    }
};
```
