# 抽象语法树构建

## 🎯 抽象语法树构建概述

**抽象语法树构建**是中间代码生成的重要技术，将语法分析树转换为抽象语法树，去除语法细节，保留语义信息。抽象语法树为后续的语义分析、代码优化和目标代码生成提供统一的中间表示。

## 🔍 基本概念

### 抽象语法树定义
抽象语法树（AST）是源代码语法结构的一种抽象表示，它去除了语法分析树中的语法细节，只保留语义相关的信息。

### 抽象语法树特点
- **语义导向**：只保留语义相关信息
- **结构简化**：去除语法细节
- **便于处理**：便于进行各种操作
- **便于优化**：容易进行代码优化

```c
enum ASTNodeType {
    PROGRAM,           // 程序
    FUNCTION_DEF,     // 函数定义
    VARIABLE_DECL,    // 变量声明
    ASSIGNMENT,       // 赋值语句
    EXPRESSION,       // 表达式
    BINARY_OP,        // 二元操作
    UNARY_OP,         // 一元操作
    IDENTIFIER,       // 标识符
    NUMBER,           // 数字
    STRING,           // 字符串
    IF_STATEMENT,     // if语句
    WHILE_STATEMENT,  // while语句
    FOR_STATEMENT,    // for语句
    RETURN_STATEMENT, // return语句
    FUNCTION_CALL,    // 函数调用
    ARRAY_ACCESS,     // 数组访问
    MEMBER_ACCESS     // 成员访问
};

struct ASTNode {
    ASTNodeType type;
    string value;
    vector<ASTNode*> children;
    map<string, void*> attributes;
    
    ASTNode(ASTNodeType t, string v = "") : type(t), value(v) {}
    
    void addChild(ASTNode* child) {
        children.push_back(child);
    }
    
    void setAttribute(string key, void* value) {
        attributes[key] = value;
    }
    
    void* getAttribute(string key) {
        return attributes[key];
    }
};
```

## 🔧 抽象语法树构建器

### 基本AST构建器
```c
class ASTBuilder {
private:
    ASTNode* root;
    
public:
    ASTBuilder() : root(nullptr) {}
    
    ASTNode* buildAST(ParseTree* parseTree) {
        root = buildNode(parseTree->getRoot());
        return root;
    }
    
    ASTNode* buildNode(ParseTreeNode* node) {
        switch (node->getType()) {
            case PROGRAM:
                return buildProgram(node);
            case FUNCTION_DEFINITION:
                return buildFunctionDefinition(node);
            case VARIABLE_DECLARATION:
                return buildVariableDeclaration(node);
            case ASSIGNMENT:
                return buildAssignment(node);
            case EXPRESSION:
                return buildExpression(node);
            case IF_STATEMENT:
                return buildIfStatement(node);
            case WHILE_STATEMENT:
                return buildWhileStatement(node);
            case RETURN_STATEMENT:
                return buildReturnStatement(node);
            default:
                return buildGenericNode(node);
        }
    }
    
    ASTNode* buildProgram(ParseTreeNode* node) {
        ASTNode* program = new ASTNode(PROGRAM);
        
        for (auto child : node->getChildren()) {
            ASTNode* childAST = buildNode(child);
            program->addChild(childAST);
        }
        
        return program;
    }
    
    ASTNode* buildFunctionDefinition(ParseTreeNode* node) {
        string returnType = node->getChild(0)->getValue();
        string funcName = node->getChild(1)->getValue();
        ParseTreeNode* parameters = node->getChild(2);
        ParseTreeNode* body = node->getChild(3);
        
        ASTNode* funcDef = new ASTNode(FUNCTION_DEF, funcName);
        funcDef->setAttribute("returnType", new string(returnType));
        
        // 构建参数列表
        ASTNode* paramList = new ASTNode(PARAMETER_LIST);
        for (auto param : parameters->getChildren()) {
            ASTNode* paramAST = buildNode(param);
            paramList->addChild(paramAST);
        }
        funcDef->addChild(paramList);
        
        // 构建函数体
        ASTNode* bodyAST = buildNode(body);
        funcDef->addChild(bodyAST);
        
        return funcDef;
    }
    
    ASTNode* buildVariableDeclaration(ParseTreeNode* node) {
        string typeName = node->getChild(0)->getValue();
        string varName = node->getChild(1)->getValue();
        
        ASTNode* varDecl = new ASTNode(VARIABLE_DECL, varName);
        varDecl->setAttribute("type", new string(typeName));
        
        // 如果有初始化表达式
        if (node->getChildren().size() > 2) {
            ParseTreeNode* initExpr = node->getChild(2);
            ASTNode* initAST = buildNode(initExpr);
            varDecl->addChild(initAST);
        }
        
        return varDecl;
    }
    
    ASTNode* buildAssignment(ParseTreeNode* node) {
        string varName = node->getChild(0)->getValue();
        ParseTreeNode* value = node->getChild(1);
        
        ASTNode* assignment = new ASTNode(ASSIGNMENT, varName);
        ASTNode* valueAST = buildNode(value);
        assignment->addChild(valueAST);
        
        return assignment;
    }
    
    ASTNode* buildExpression(ParseTreeNode* node) {
        if (node->getChildren().size() == 1) {
            // 单操作数表达式
            return buildNode(node->getChild(0));
        } else if (node->getChildren().size() == 2) {
            // 一元操作
            string op = node->getChild(0)->getValue();
            ParseTreeNode* operand = node->getChild(1);
            
            ASTNode* unaryOp = new ASTNode(UNARY_OP, op);
            ASTNode* operandAST = buildNode(operand);
            unaryOp->addChild(operandAST);
            
            return unaryOp;
        } else if (node->getChildren().size() == 3) {
            // 二元操作
            ParseTreeNode* left = node->getChild(0);
            string op = node->getChild(1)->getValue();
            ParseTreeNode* right = node->getChild(2);
            
            ASTNode* binaryOp = new ASTNode(BINARY_OP, op);
            ASTNode* leftAST = buildNode(left);
            ASTNode* rightAST = buildNode(right);
            binaryOp->addChild(leftAST);
            binaryOp->addChild(rightAST);
            
            return binaryOp;
        }
        
        return nullptr;
    }
    
    ASTNode* buildIfStatement(ParseTreeNode* node) {
        ParseTreeNode* condition = node->getChild(0);
        ParseTreeNode* thenBlock = node->getChild(1);
        ParseTreeNode* elseBlock = node->getChildren().size() > 2 ? node->getChild(2) : nullptr;
        
        ASTNode* ifStmt = new ASTNode(IF_STATEMENT);
        ASTNode* conditionAST = buildNode(condition);
        ASTNode* thenAST = buildNode(thenBlock);
        ASTNode* elseAST = elseBlock ? buildNode(elseBlock) : nullptr;
        
        ifStmt->addChild(conditionAST);
        ifStmt->addChild(thenAST);
        if (elseAST) {
            ifStmt->addChild(elseAST);
        }
        
        return ifStmt;
    }
    
    ASTNode* buildWhileStatement(ParseTreeNode* node) {
        ParseTreeNode* condition = node->getChild(0);
        ParseTreeNode* body = node->getChild(1);
        
        ASTNode* whileStmt = new ASTNode(WHILE_STATEMENT);
        ASTNode* conditionAST = buildNode(condition);
        ASTNode* bodyAST = buildNode(body);
        
        whileStmt->addChild(conditionAST);
        whileStmt->addChild(bodyAST);
        
        return whileStmt;
    }
    
    ASTNode* buildReturnStatement(ParseTreeNode* node) {
        ASTNode* returnStmt = new ASTNode(RETURN_STATEMENT);
        
        if (node->getChildren().size() > 0) {
            ParseTreeNode* value = node->getChild(0);
            ASTNode* valueAST = buildNode(value);
            returnStmt->addChild(valueAST);
        }
        
        return returnStmt;
    }
    
    ASTNode* buildGenericNode(ParseTreeNode* node) {
        ASTNode* astNode = new ASTNode(node->getType(), node->getValue());
        
        for (auto child : node->getChildren()) {
            ASTNode* childAST = buildNode(child);
            astNode->addChild(childAST);
        }
        
        return astNode;
    }
};
```

### 高级AST构建器
```c
class AdvancedASTBuilder {
private:
    ASTBuilder baseBuilder;
    map<string, TypeInfo> typeTable;
    map<string, FunctionInfo> functionTable;
    
public:
    ASTNode* buildAdvancedAST(ParseTree* parseTree) {
        // 构建基本AST
        ASTNode* ast = baseBuilder.buildAST(parseTree);
        
        // 进行高级处理
        processAdvancedFeatures(ast);
        
        return ast;
    }
    
    void processAdvancedFeatures(ASTNode* node) {
        switch (node->getType()) {
            case FUNCTION_DEF:
                processFunctionDefinition(node);
                break;
            case CLASS_DEFINITION:
                processClassDefinition(node);
                break;
            case TEMPLATE_DEFINITION:
                processTemplateDefinition(node);
                break;
            case GENERIC_INSTANTIATION:
                processGenericInstantiation(node);
                break;
        }
        
        // 递归处理子节点
        for (auto child : node->getChildren()) {
            processAdvancedFeatures(child);
        }
    }
    
    void processFunctionDefinition(ASTNode* node) {
        string funcName = node->getValue();
        
        // 创建函数信息
        FunctionInfo funcInfo;
        funcInfo.name = funcName;
        funcInfo.returnType = *(string*)node->getAttribute("returnType");
        
        // 处理参数
        ASTNode* paramList = node->getChild(0);
        for (auto param : paramList->getChildren()) {
            string paramType = *(string*)param->getAttribute("type");
            string paramName = param->getValue();
            funcInfo.parameters.push_back({paramType, paramName});
        }
        
        // 处理函数体
        ASTNode* body = node->getChild(1);
        processFunctionBody(body, funcInfo);
        
        functionTable[funcName] = funcInfo;
    }
    
    void processClassDefinition(ASTNode* node) {
        string className = node->getValue();
        
        // 创建类信息
        ClassInfo classInfo;
        classInfo.name = className;
        
        // 处理成员变量和方法
        for (auto member : node->getChildren()) {
            if (member->getType() == MEMBER_VARIABLE) {
                processMemberVariable(member, classInfo);
            } else if (member->getType() == MEMBER_FUNCTION) {
                processMemberFunction(member, classInfo);
            }
        }
        
        typeTable[className] = classInfo;
    }
    
    void processTemplateDefinition(ASTNode* node) {
        string templateName = node->getValue();
        
        // 创建模板信息
        TemplateInfo templateInfo;
        templateInfo.name = templateName;
        
        // 处理模板参数
        ASTNode* templateParams = node->getChild(0);
        for (auto param : templateParams->getChildren()) {
            templateInfo.parameters.push_back(param->getValue());
        }
        
        // 处理模板体
        ASTNode* templateBody = node->getChild(1);
        processTemplateBody(templateBody, templateInfo);
        
        templateTable[templateName] = templateInfo;
    }
    
    void processGenericInstantiation(ASTNode* node) {
        string templateName = node->getChild(0)->getValue();
        
        // 查找模板定义
        auto it = templateTable.find(templateName);
        if (it != templateTable.end()) {
            TemplateInfo templateInfo = it->second;
            
            // 处理类型参数
            ASTNode* typeArgs = node->getChild(1);
            vector<string> typeArguments;
            for (auto typeArg : typeArgs->getChildren()) {
                typeArguments.push_back(typeArg->getValue());
            }
            
            // 实例化模板
            instantiateTemplate(templateInfo, typeArguments);
        }
    }
    
    void processFunctionBody(ASTNode* body, FunctionInfo& funcInfo) {
        // 处理函数体中的语句
        for (auto stmt : body->getChildren()) {
            processStatement(stmt, funcInfo);
        }
    }
    
    void processMemberVariable(ASTNode* member, ClassInfo& classInfo) {
        string typeName = *(string*)member->getAttribute("type");
        string varName = member->getValue();
        
        MemberVariableInfo varInfo;
        varInfo.name = varName;
        varInfo.type = typeName;
        varInfo.accessLevel = *(string*)member->getAttribute("accessLevel");
        
        classInfo.memberVariables.push_back(varInfo);
    }
    
    void processMemberFunction(ASTNode* member, ClassInfo& classInfo) {
        string returnType = *(string*)member->getAttribute("returnType");
        string funcName = member->getValue();
        
        MemberFunctionInfo funcInfo;
        funcInfo.name = funcName;
        funcInfo.returnType = returnType;
        funcInfo.accessLevel = *(string*)member->getAttribute("accessLevel");
        
        // 处理参数
        ASTNode* paramList = member->getChild(0);
        for (auto param : paramList->getChildren()) {
            string paramType = *(string*)param->getAttribute("type");
            string paramName = param->getValue();
            funcInfo.parameters.push_back({paramType, paramName});
        }
        
        classInfo.memberFunctions.push_back(funcInfo);
    }
    
    void processTemplateBody(ASTNode* body, TemplateInfo& templateInfo) {
        // 处理模板体
        for (auto child : body->getChildren()) {
            processTemplateNode(child, templateInfo);
        }
    }
```

```c
class ASTOptimizer {
private:
    ASTNode* root;
    
public:
    ASTOptimizer(ASTNode* ast) : root(ast) {}
    
    ASTNode* optimize() {
        // 常量折叠
        constantFolding(root);
        
        // 死代码消除
        deadCodeElimination(root);
        
        // 公共子表达式消除
        commonSubexpressionElimination(root);
        
        // 强度削弱
        strengthReduction(root);
        
        return root;
    }
    
    void constantFolding(ASTNode* node) {
        if (node->getType() == BINARY_OP) {
            ASTNode* left = node->getChild(0);
            ASTNode* right = node->getChild(1);
            
            // 递归处理子节点
            constantFolding(left);
            constantFolding(right);
            
            // 如果两个操作数都是常量，则进行常量折叠
            if (isConstant(left) && isConstant(right)) {
                int val1 = stoi(left->getValue());
                int val2 = stoi(right->getValue());
                int result = evaluateBinaryOp(node->getValue(), val1, val2);
                
                // 替换为常量节点
                ASTNode* constantNode = new ASTNode(NUMBER, to_string(result));
                replaceNode(node, constantNode);
            }
        }
        
        // 递归处理子节点
        for (auto child : node->getChildren()) {
            constantFolding(child);
        }
    }
    
    void deadCodeElimination(ASTNode* node) {
        if (node->getType() == IF_STATEMENT) {
            ASTNode* condition = node->getChild(0);
            
            // 如果条件是常量
            if (isConstant(condition)) {
                int condValue = stoi(condition->getValue());
                ASTNode* thenBlock = node->getChild(1);
                ASTNode* elseBlock = node->getChildren().size() > 2 ? node->getChild(2) : nullptr;
                
                if (condValue != 0) {
                    // 条件为真，只保留then块
                    replaceNode(node, thenBlock);
                } else if (elseBlock) {
                    // 条件为假，只保留else块
                    replaceNode(node, elseBlock);
                } else {
                    // 条件为假且没有else块，删除整个if语句
                    deleteNode(node);
                }
            }
        }
        
        // 递归处理子节点
        for (auto child : node->getChildren()) {
            deadCodeElimination(child);
        }
    }
    
    void commonSubexpressionElimination(ASTNode* node) {
        map<string, ASTNode*> expressionMap;
        
        // 收集所有表达式
        collectExpressions(node, expressionMap);
        
        // 替换重复的表达式
        replaceCommonExpressions(node, expressionMap);
    }
    
    void strengthReduction(ASTNode* node) {
        if (node->getType() == BINARY_OP) {
            string op = node->getValue();
            ASTNode* left = node->getChild(0);
            ASTNode* right = node->getChild(1);
            
            // 乘法替换为加法
            if (op == "*" && isConstant(right) && stoi(right->getValue()) == 2) {
                ASTNode* addNode = new ASTNode(BINARY_OP, "+");
                addNode->addChild(left);
                addNode->addChild(left);
                replaceNode(node, addNode);
            }
            
            // 除法替换为右移
            if (op == "/" && isConstant(right) && isPowerOfTwo(stoi(right->getValue()))) {
                int shiftAmount = log2(stoi(right->getValue()));
                ASTNode* shiftNode = new ASTNode(BINARY_OP, ">>");
                shiftNode->addChild(left);
                shiftNode->addChild(new ASTNode(NUMBER, to_string(shiftAmount)));
                replaceNode(node, shiftNode);
            }
        }
        
        // 递归处理子节点
        for (auto child : node->getChildren()) {
            strengthReduction(child);
        }
    }
    
    bool isConstant(ASTNode* node) {
        return node->getType() == NUMBER;
    }
    
    int evaluateBinaryOp(string op, int left, int right) {
        if (op == "+") return left + right;
        if (op == "-") return left - right;
        if (op == "*") return left * right;
        if (op == "/") return left / right;
        if (op == "%") return left % right;
        if (op == "&") return left & right;
        if (op == "|") return left | right;
        if (op == "^") return left ^ right;
        if (op == "<<") return left << right;
        if (op == ">>") return left >> right;
        return 0;
    }
    
    bool isPowerOfTwo(int n) {
        return n > 0 && (n & (n - 1)) == 0;
    }
    
    int log2(int n) {
        int result = 0;
        while (n > 1) {
            n >>= 1;
            result++;
        }
        return result;
    }
```

```c
                return convertVariableDeclToC(node);
            case ASSIGNMENT:
                return convertAssignmentToC(node);
            case BINARY_OP:
                return convertBinaryOpToC(node);
            case IF_STATEMENT:
                return convertIfStatementToC(node);
            case WHILE_STATEMENT:
                return convertWhileStatementToC(node);
            default:
                return "";
        }
    }
    
    string convertProgramToC(ASTNode* node) {
        string cCode;
        for (auto child : node->getChildren()) {
            cCode += convertNodeToC(child) + "\n";
        }
        return cCode;
    }
    
    string convertFunctionDefToC(ASTNode* node) {
        string returnType = *(string*)node->getAttribute("returnType");
        string funcName = node->getValue();
        
        string cCode = returnType + " " + funcName + "(";
        
        // 处理参数
        ASTNode* paramList = node->getChild(0);
        for (int i = 0; i < paramList->getChildren().size(); i++) {
            ASTNode* param = paramList->getChild(i);
            string paramType = *(string*)param->getAttribute("type");
            string paramName = param->getValue();
            
            if (i > 0) cCode += ", ";
            cCode += paramType + " " + paramName;
        }
        
        cCode += ") {\n";
        
        // 处理函数体
        ASTNode* body = node->getChild(1);
        cCode += convertNodeToC(body);
        
        cCode += "}\n";
        return cCode;
    }
    
    string convertVariableDeclToC(ASTNode* node) {
        string typeName = *(string*)node->getAttribute("type");
        string varName = node->getValue();
        
        string cCode = typeName + " " + varName;
        
        // 如果有初始化表达式
        if (node->getChildren().size() > 0) {
            ASTNode* initExpr = node->getChild(0);
            cCode += " = " + convertNodeToC(initExpr);
        }
        
        cCode += ";\n";
        return cCode;
    }
    
    string convertAssignmentToC(ASTNode* node) {
        string varName = node->getValue();
        ASTNode* value = node->getChild(0);
        
        return varName + " = " + convertNodeToC(value) + ";\n";
    }
    
    string convertBinaryOpToC(ASTNode* node) {
        string op = node->getValue();
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        return "(" + convertNodeToC(left) + " " + op + " " + convertNodeToC(right) + ")";
    }
    
    string convertIfStatementToC(ASTNode* node) {
        ASTNode* condition = node->getChild(0);
        ASTNode* thenBlock = node->getChild(1);
        ASTNode* elseBlock = node->getChildren().size() > 2 ? node->getChild(2) : nullptr;
        
        string cCode = "if (" + convertNodeToC(condition) + ") {\n";
        cCode += convertNodeToC(thenBlock);
        cCode += "}";
        
        if (elseBlock) {
            cCode += " else {\n";
            cCode += convertNodeToC(elseBlock);
            cCode += "}";
        }
        
        cCode += "\n";
        return cCode;
    }
    
    string convertWhileStatementToC(ASTNode* node) {
        ASTNode* condition = node->getChild(0);
        ASTNode* body = node->getChild(1);
        
        string cCode = "while (" + convertNodeToC(condition) + ") {\n";
        cCode += convertNodeToC(body);
        cCode += "}\n";
        
        return cCode;
    }
```
