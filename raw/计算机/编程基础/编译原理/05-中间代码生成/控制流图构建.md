# 控制流图构建

## 🎯 控制流图构建概述

**控制流图构建**是中间代码生成的重要技术，将程序的控制流结构表示为图的形式，便于进行数据流分析和代码优化。控制流图为后续的代码优化和目标代码生成提供基础。

## 🔍 基本概念

### 控制流图定义
控制流图（CFG）是程序控制流结构的一种图形表示，其中节点表示基本块，边表示控制流转移。

### 控制流图特点
- **节点表示**：基本块（Basic Block）
- **边表示**：控制流转移
- **便于分析**：便于进行数据流分析
- **便于优化**：便于进行代码优化

### 1. 基本块定义
```c
struct BasicBlock {
    int id;
    vector<Instruction> instructions;
    vector<BasicBlock*> predecessors;
    vector<BasicBlock*> successors;
    set<string> liveIn;
    set<string> liveOut;
    set<string> def;
    set<string> use;
    
    BasicBlock(int i) : id(i) {}
    
    void addInstruction(Instruction inst) {
        instructions.push_back(inst);
    }
    
    void addPredecessor(BasicBlock* pred) {
        predecessors.push_back(pred);
    }
    
    void addSuccessor(BasicBlock* succ) {
        successors.push_back(succ);
    }
    
    bool isEmpty() {
        return instructions.empty();
    }
    
    bool isTerminal() {
        return successors.empty();
    }
    
    bool isEntry() {
        return predecessors.empty();
    }
};
```

### 2. 指令定义
```c
enum InstructionType {
    ASSIGNMENT,     // 赋值指令
    JUMP,          // 跳转指令
    CONDITIONAL,   // 条件跳转指令
    FUNCTION_CALL, // 函数调用指令
    RETURN,        // 返回指令
    LABEL          // 标签指令
};

struct Instruction {
    InstructionType type;
    string op;
    string arg1;
    string arg2;
    string result;
    string label;
    
    Instruction(InstructionType t, string o, string a1 = "", string a2 = "", string r = "", string l = "")
        : type(t), op(o), arg1(a1), arg2(a2), result(r), label(l) {}
    
    bool isJump() {
        return type == JUMP || type == CONDITIONAL;
    }
    
    bool isTerminal() {
        return type == JUMP || type == CONDITIONAL || type == RETURN;
    }
    
    string toString() {
        switch (type) {
            case ASSIGNMENT:
                return result + " = " + arg1 + " " + op + " " + arg2;
            case JUMP:
                return "goto " + label;
            case CONDITIONAL:
                return "if " + arg1 + " " + op + " " + arg2 + " goto " + label;
            case FUNCTION_CALL:
                return result + " = call " + arg1 + "(" + arg2 + ")";
            case RETURN:
                return "return " + arg1;
            case LABEL:
                return label + ":";
            default:
                return "";
        }
    }
};
```

### 3. 控制流图定义
```c
class ControlFlowGraph {
private:
    vector<BasicBlock*> blocks;
    BasicBlock* entryBlock;
    BasicBlock* exitBlock;
    map<string, BasicBlock*> labelToBlock;
    
public:
    ControlFlowGraph() : entryBlock(nullptr), exitBlock(nullptr) {}
    
    BasicBlock* createBasicBlock() {
        BasicBlock* block = new BasicBlock(blocks.size());
        blocks.push_back(block);
        return block;
    }
    
    BasicBlock* getEntryBlock() {
        return entryBlock;
    }
    
    BasicBlock* getExitBlock() {
        return exitBlock;
    }
    
    void setEntryBlock(BasicBlock* block) {
        entryBlock = block;
    }
    
    void setExitBlock(BasicBlock* block) {
        exitBlock = block;
    }
    
    void addEdge(BasicBlock* from, BasicBlock* to) {
        from->addSuccessor(to);
        to->addPredecessor(from);
    }
    
    void addLabel(BasicBlock* block, string label) {
        labelToBlock[label] = block;
    }
    
    BasicBlock* getBlockByLabel(string label) {
        auto it = labelToBlock.find(label);
        return it != labelToBlock.end() ? it->second : nullptr;
    }
    
    vector<BasicBlock*> getBlocks() {
        return blocks;
    }
    
    void print() {
        for (auto block : blocks) {
            cout << "Block " << block->id << ":" << endl;
            for (auto inst : block->instructions) {
                cout << "  " << inst.toString() << endl;
            }
            cout << "  Predecessors: ";
            for (auto pred : block->predecessors) {
                cout << pred->id << " ";
            }
            cout << endl;
            cout << "  Successors: ";
            for (auto succ : block->successors) {
                cout << succ->id << " ";
            }
            cout << endl;
        }
    }
};
```

## 🔧 控制流图构建器

### 基本CFG构建器
```c
class CFGBuilder {
private:
    ControlFlowGraph* cfg;
    BasicBlock* currentBlock;
    map<string, BasicBlock*> labelMap;
    
public:
    CFGBuilder() {
        cfg = new ControlFlowGraph();
        currentBlock = nullptr;
    }
    
    ControlFlowGraph* buildCFG(vector<Instruction>& instructions) {
        // 创建入口块
        BasicBlock* entryBlock = cfg->createBasicBlock();
        cfg->setEntryBlock(entryBlock);
        currentBlock = entryBlock;
        
        // 处理指令序列
        for (auto& inst : instructions) {
            processInstruction(inst);
        }
        
        // 创建出口块
        BasicBlock* exitBlock = cfg->createBasicBlock();
        cfg->setExitBlock(exitBlock);
        
        // 连接所有终端块到出口块
        connectTerminalBlocksToExit(exitBlock);
        
        return cfg;
    }
    
    void processInstruction(Instruction& inst) {
        switch (inst.type) {
            case LABEL:
                processLabel(inst);
                break;
            case JUMP:
                processJump(inst);
                break;
            case CONDITIONAL:
                processConditional(inst);
                break;
            case RETURN:
                processReturn(inst);
                break;
            default:
                processRegular(inst);
                break;
        }
    }
    
    void processLabel(Instruction& inst) {
        string label = inst.label;
        
        // 如果当前块不为空，创建新块
        if (!currentBlock->isEmpty()) {
            currentBlock = cfg->createBasicBlock();
        }
        
        // 添加标签映射
        cfg->addLabel(currentBlock, label);
        labelMap[label] = currentBlock;
        
        // 添加标签指令
        currentBlock->addInstruction(inst);
    }
    
    void processJump(Instruction& inst) {
        // 添加跳转指令
        currentBlock->addInstruction(inst);
        
        // 创建新块
        currentBlock = cfg->createBasicBlock();
    }
    
    void processConditional(Instruction& inst) {
        // 添加条件跳转指令
        currentBlock->addInstruction(inst);
        
        // 创建新块
        currentBlock = cfg->createBasicBlock();
    }
    
    void processReturn(Instruction& inst) {
        // 添加返回指令
        currentBlock->addInstruction(inst);
        
        // 创建新块
        currentBlock = cfg->createBasicBlock();
    }
    
    void processRegular(Instruction& inst) {
        // 添加常规指令
        currentBlock->addInstruction(inst);
    }
    
    void connectTerminalBlocksToExit(BasicBlock* exitBlock) {
        for (auto block : cfg->getBlocks()) {
            if (block->isTerminal() && block != exitBlock) {
                cfg->addEdge(block, exitBlock);
            }
        }
    }
    
    void connectBlocks() {
        for (auto block : cfg->getBlocks()) {
            for (auto inst : block->instructions) {
                if (inst.isJump()) {
                    string label = inst.label;
                    BasicBlock* targetBlock = labelMap[label];
                    if (targetBlock) {
                        cfg->addEdge(block, targetBlock);
                    }
                }
            }
        }
    }
};
```

### 高级CFG构建器
```c
class AdvancedCFGBuilder {
private:
    CFGBuilder baseBuilder;
    map<string, FunctionInfo> functionTable;
    
public:
    ControlFlowGraph* buildAdvancedCFG(ASTNode* ast) {
        // 构建基本CFG
        ControlFlowGraph* cfg = baseBuilder.buildCFG(extractInstructions(ast));
        
        // 进行高级处理
        processAdvancedFeatures(cfg, ast);
        
        return cfg;
    }
    
    vector<Instruction> extractInstructions(ASTNode* ast) {
        vector<Instruction> instructions;
        
        // 遍历AST提取指令
        extractInstructionsFromNode(ast, instructions);
        
        return instructions;
    }
    
    void extractInstructionsFromNode(ASTNode* node, vector<Instruction>& instructions) {
        switch (node->getType()) {
            case FUNCTION_DEF:
                extractFunctionInstructions(node, instructions);
                break;
            case ASSIGNMENT:
                extractAssignmentInstructions(node, instructions);
                break;
            case IF_STATEMENT:
                extractIfStatementInstructions(node, instructions);
                break;
            case WHILE_STATEMENT:
                extractWhileStatementInstructions(node, instructions);
                break;
            case RETURN_STATEMENT:
                extractReturnInstructions(node, instructions);
                break;
            default:
                // 递归处理子节点
                for (auto child : node->getChildren()) {
                    extractInstructionsFromNode(child, instructions);
                }
                break;
        }
    }
    
    void extractFunctionInstructions(ASTNode* node, vector<Instruction>& instructions) {
        string funcName = node->getValue();
        
        // 添加函数标签
        instructions.push_back(Instruction(LABEL, "", "", "", "", funcName));
        
        // 处理函数体
        ASTNode* body = node->getChild(1);
        extractInstructionsFromNode(body, instructions);
        
        // 添加返回指令
        instructions.push_back(Instruction(RETURN, "", "", "", "", ""));
    }
    
    void extractAssignmentInstructions(ASTNode* node, vector<Instruction>& instructions) {
        string varName = node->getValue();
        ASTNode* value = node->getChild(0);
        
        // 提取值表达式
        string valueExpr = extractExpression(value);
        
        // 添加赋值指令
        instructions.push_back(Instruction(ASSIGNMENT, "=", valueExpr, "", varName));
    }
    
    void extractIfStatementInstructions(ASTNode* node, vector<Instruction>& instructions) {
        ASTNode* condition = node->getChild(0);
        ASTNode* thenBlock = node->getChild(1);
        ASTNode* elseBlock = node->getChildren().size() > 2 ? node->getChild(2) : nullptr;
        
        // 提取条件表达式
        string condExpr = extractExpression(condition);
        
        // 生成标签
        string thenLabel = generateLabel("then");
        string elseLabel = generateLabel("else");
        string endLabel = generateLabel("end");
        
        // 添加条件跳转指令
        instructions.push_back(Instruction(CONDITIONAL, "==", condExpr, "0", "", elseLabel));
        
        // 添加then块标签
        instructions.push_back(Instruction(LABEL, "", "", "", "", thenLabel));
        
        // 处理then块
        extractInstructionsFromNode(thenBlock, instructions);
        
        // 添加跳转到结束标签
        instructions.push_back(Instruction(JUMP, "", "", "", "", endLabel));
        
        // 添加else块标签
        instructions.push_back(Instruction(LABEL, "", "", "", "", elseLabel));
        
        // 处理else块
        if (elseBlock) {
            extractInstructionsFromNode(elseBlock, instructions);
        }
        
        // 添加结束标签
        instructions.push_back(Instruction(LABEL, "", "", "", "", endLabel));
    }
    
    void extractWhileStatementInstructions(ASTNode* node, vector<Instruction>& instructions) {
        ASTNode* condition = node->getChild(0);
        ASTNode* body = node->getChild(1);
        
        // 生成标签
        string loopLabel = generateLabel("loop");
        string bodyLabel = generateLabel("body");
        string endLabel = generateLabel("end");
        
        // 添加循环标签
        instructions.push_back(Instruction(LABEL, "", "", "", "", loopLabel));
        
        // 提取条件表达式
        string condExpr = extractExpression(condition);
        
        // 添加条件跳转指令
        instructions.push_back(Instruction(CONDITIONAL, "==", condExpr, "0", "", endLabel));
        
        // 添加body标签
        instructions.push_back(Instruction(LABEL, "", "", "", "", bodyLabel));
        
        // 处理循环体
        extractInstructionsFromNode(body, instructions);
        
        // 添加跳转到循环标签
        instructions.push_back(Instruction(JUMP, "", "", "", "", loopLabel));
        
        // 添加结束标签
        instructions.push_back(Instruction(LABEL, "", "", "", "", endLabel));
    }
    
    void extractReturnInstructions(ASTNode* node, vector<Instruction>& instructions) {
        if (node->getChildren().size() > 0) {
            ASTNode* value = node->getChild(0);
            string valueExpr = extractExpression(value);
            instructions.push_back(Instruction(RETURN, "", valueExpr, "", ""));
        } else {
            instructions.push_back(Instruction(RETURN, "", "", "", ""));
        }
    }
    
    string extractExpression(ASTNode* node) {
        switch (node->getType()) {
            case BINARY_OP:
                return extractBinaryOpExpression(node);
            case UNARY_OP:
                return extractUnaryOpExpression(node);
            case IDENTIFIER:
                return node->getValue();
            case NUMBER:
                return node->getValue();
            default:
                return "";
        }
    }
    
    string extractBinaryOpExpression(ASTNode* node) {
        string op = node->getValue();
        ASTNode* left = node->getChild(0);
        ASTNode* right = node->getChild(1);
        
        string leftExpr = extractExpression(left);
        string rightExpr = extractExpression(right);
        
        return "(" + leftExpr + " " + op + " " + rightExpr + ")";
    }
    
    string extractUnaryOpExpression(ASTNode* node) {
        string op = node->getValue();
        ASTNode* operand = node->getChild(0);
        
        string operandExpr = extractExpression(operand);
        
        return op + operandExpr;
    }
    
    string generateLabel(string prefix) {
        static int counter = 0;
        return prefix + "_" + to_string(counter++);
    }
    
    void processAdvancedFeatures(ControlFlowGraph* cfg, ASTNode* ast) {
        // 处理函数调用
        processFunctionCalls(cfg, ast);
        
        // 处理异常处理
        processExceptionHandling(cfg, ast);
        
        // 处理循环优化
        processLoopOptimization(cfg, ast);
    }
```

```c
class CFGConverter {
private:
    ControlFlowGraph* cfg;
    
public:
    CFGConverter(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    string convertToAssembly() {
        return convertCFGToAssembly(cfg);
    }
    
    string convertToC() {
        return convertCFGToC(cfg);
    }
    
    string convertToJava() {
        return convertCFGToJava(cfg);
    }
    
    string convertCFGToAssembly(ControlFlowGraph* cfg) {
        string assemblyCode;
        
        for (auto block : cfg->getBlocks()) {
            assemblyCode += "block_" + to_string(block->id) + ":\n";
            
            for (auto inst : block->instructions) {
                assemblyCode += "  " + convertInstructionToAssembly(inst) + "\n";
            }
            
            // 添加跳转指令
            for (auto succ : block->successors) {
                assemblyCode += "  jmp block_" + to_string(succ->id) + "\n";
            }
            
            assemblyCode += "\n";
        }
        
        return assemblyCode;
    }
    
    string convertInstructionToAssembly(Instruction& inst) {
        switch (inst.type) {
            case ASSIGNMENT:
                return convertAssignmentToAssembly(inst);
            case JUMP:
                return "jmp " + inst.label;
            case CONDITIONAL:
                return "cmp " + inst.arg1 + ", " + inst.arg2 + "\n  j" + inst.op + " " + inst.label;
            case FUNCTION_CALL:
                return "call " + inst.arg1;
            case RETURN:
                return "ret";
            case LABEL:
                return inst.label + ":";
            default:
                return "";
        }
    }
    
    string convertAssignmentToAssembly(Instruction& inst) {
        if (inst.op == "=") {
            return "mov " + inst.result + ", " + inst.arg1;
        } else if (inst.op == "+") {
            return "add " + inst.result + ", " + inst.arg1 + ", " + inst.arg2;
        } else if (inst.op == "-") {
            return "sub " + inst.result + ", " + inst.arg1 + ", " + inst.arg2;
        } else if (inst.op == "*") {
            return "mul " + inst.result + ", " + inst.arg1 + ", " + inst.arg2;
        } else if (inst.op == "/") {
            return "div " + inst.result + ", " + inst.arg1 + ", " + inst.arg2;
        }
        return "";
    }
    
    string convertCFGToC(ControlFlowGraph* cfg) {
        string cCode;
        
        for (auto block : cfg->getBlocks()) {
            cCode += "block_" + to_string(block->id) + ":\n";
            
            for (auto inst : block->instructions) {
                cCode += "  " + convertInstructionToC(inst) + "\n";
            }
            
            // 添加跳转指令
            for (auto succ : block->successors) {
                cCode += "  goto block_" + to_string(succ->id) + ";\n";
            }
            
            cCode += "\n";
        }
        
        return cCode;
    }

                return inst.result + " = call " + inst.arg1 + "(" + inst.arg2 + ");";
            case RETURN:
                return "return " + inst.arg1 + ";";
```
