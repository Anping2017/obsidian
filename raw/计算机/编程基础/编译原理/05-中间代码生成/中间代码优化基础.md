# 中间代码优化基础

## 🎯 中间代码优化基础概述

**中间代码优化基础**是中间代码生成的重要技术，通过分析和优化中间代码来提高程序性能。中间代码优化为后续的目标代码生成提供优化的基础。

## 🔍 基本概念

### 中间代码优化定义
中间代码优化是在中间代码生成后，目标代码生成前，对中间代码进行分析和优化的过程。

### 中间代码优化特点
- **性能提升**：提高程序执行效率
- **代码简化**：简化中间代码结构
- **资源优化**：优化资源使用
- **便于生成**：便于目标代码生成

## 🔧 基本优化技术

### 1. 常量折叠
```c
class ConstantFoldingOptimizer {
private:
    vector<Instruction> instructions;
    
public:
    ConstantFoldingOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        bool changed = true;
        
        while (changed) {
            changed = false;
            
            for (int i = 0; i < instructions.size(); i++) {
                if (canFold(instructions[i])) {
                    Instruction folded = foldInstruction(instructions[i]);
                    if (folded != instructions[i]) {
                        instructions[i] = folded;
                        changed = true;
                    }
                }
            }
        }
        
        return instructions;
    }
    
    bool canFold(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            return isConstant(inst.arg1) && isConstant(inst.arg2);
        }
        return false;
    }
    
    Instruction foldInstruction(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            int val1 = stoi(inst.arg1);
            int val2 = stoi(inst.arg2);
            int result = evaluateOperation(inst.op, val1, val2);
            
            return Instruction(ASSIGNMENT, "=", to_string(result), "", inst.result);
        }
        return inst;
    }
    
    bool isConstant(string str) {
        return !str.empty() && str.find_first_not_of("0123456789") == string::npos;
    }
    
    int evaluateOperation(string op, int left, int right) {
        if (op == "+") return left + right;
        if (op == "-") return left - right;
        if (op == "*") return left * right;
        if (op == "/") return right != 0 ? left / right : 0;
        if (op == "%") return right != 0 ? left % right : 0;
        if (op == "&") return left & right;
        if (op == "|") return left | right;
        if (op == "^") return left ^ right;
        if (op == "<<") return left << right;
        if (op == ">>") return left >> right;
        return 0;
    }
};
```

### 2. 死代码消除
```c
class DeadCodeEliminationOptimizer {
private:
    vector<Instruction> instructions;
    set<string> liveVariables;
    
public:
    DeadCodeEliminationOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        // 计算活跃变量
        computeLiveVariables();
        
        // 消除死代码
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : instructions) {
            if (isLive(inst)) {
                optimizedInstructions.push_back(inst);
            }
        }
        
        return optimizedInstructions;
    }
    
    void computeLiveVariables() {
        // 初始化活跃变量集合
        liveVariables.clear();
        
        // 从后往前遍历指令
        for (int i = instructions.size() - 1; i >= 0; i--) {
            Instruction& inst = instructions[i];
            
            if (inst.type == ASSIGNMENT) {
                // 如果结果变量不在活跃变量集合中，则指令是死代码
                if (liveVariables.find(inst.result) == liveVariables.end()) {
                    continue;
                }
                
                // 移除结果变量
                liveVariables.erase(inst.result);
                
                // 添加使用的变量
                if (!inst.arg1.empty() && !isConstant(inst.arg1)) {
                    liveVariables.insert(inst.arg1);
                }
                if (!inst.arg2.empty() && !isConstant(inst.arg2)) {
                    liveVariables.insert(inst.arg2);
                }
            } else if (inst.type == FUNCTION_CALL) {
                // 函数调用可能使用全局变量
                // 这里需要根据具体的函数调用语义来实现
            }
        }
    }
    
    bool isLive(Instruction& inst) {
        if (inst.type == ASSIGNMENT) {
            return liveVariables.find(inst.result) != liveVariables.end();
        }
        return true; // 非赋值指令都认为是活跃的
    }
    
    bool isConstant(string str) {
        return !str.empty() && str.find_first_not_of("0123456789") == string::npos;
    }
};
```

```c
class CommonSubexpressionEliminationOptimizer {
private:
    vector<Instruction> instructions;
    map<string, string> expressionMap;
    
public:
    CommonSubexpressionEliminationOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : instructions) {
            if (inst.type == ASSIGNMENT) {
                string expr = inst.arg1 + " " + inst.op + " " + inst.arg2;
                
                if (expressionMap.find(expr) != expressionMap.end()) {
                    // 找到公共子表达式，替换为已计算的变量
                    string existingVar = expressionMap[expr];
                    optimizedInstructions.push_back(Instruction(ASSIGNMENT, "=", existingVar, "", inst.result));
                } else {
                    // 记录新的表达式
                    expressionMap[expr] = inst.result;
                    optimizedInstructions.push_back(inst);
                }
            } else {
                optimizedInstructions.push_back(inst);
            }
        }
        
        return optimizedInstructions;
    }
};
```

### 4. 强度削弱
```c
class StrengthReductionOptimizer {
private:
    vector<Instruction> instructions;
    
public:
    StrengthReductionOptimizer(vector<Instruction>& insts) : instructions(insts) {}
    
    vector<Instruction> optimize() {
        vector<Instruction> optimizedInstructions;
        
        for (auto& inst : instructions) {
            if (inst.type == ASSIGNMENT) {
                Instruction optimized = reduceStrength(inst);
                optimizedInstructions.push_back(optimized);
            } else {
                optimizedInstructions.push_back(inst);
            }
        }
        
        return optimizedInstructions;
    }
    
    Instruction reduceStrength(Instruction& inst) {
        string op = inst.op;
        string arg1 = inst.arg1;
        string arg2 = inst.arg2;
        string result = inst.result;
        
        // 乘法替换为加法
        if (op == "*" && isConstant(arg2) && stoi(arg2) == 2) {
            return Instruction(ASSIGNMENT, "+", arg1, arg1, result);
        }
        
        // 除法替换为右移
        if (op == "/" && isConstant(arg2) && isPowerOfTwo(stoi(arg2))) {
            int shiftAmount = log2(stoi(arg2));
            return Instruction(ASSIGNMENT, ">>", arg1, to_string(shiftAmount), result);
        }
        
        // 乘法替换为左移
        if (op == "*" && isConstant(arg2) && isPowerOfTwo(stoi(arg2))) {
            int shiftAmount = log2(stoi(arg2));
            return Instruction(ASSIGNMENT, "<<", arg1, to_string(shiftAmount), result);
        }
        
        return inst;
    }
    
    bool isConstant(string str) {
        return !str.empty() && str.find_first_not_of("0123456789") == string::npos;
    }
    
    bool isPowerOfTwo(int n) {
        return n > 0 && (n & (n - 1)) == 0;
    }
    
    int log2(int n) {
        int result = 0;
        while (n > 1) {
            n >>= 1;
            result++;
        }
        return result;
    }
};
```

```c
class LoopOptimizationOptimizer {
private:
    ControlFlowGraph* cfg;
    
public:
    LoopOptimizationOptimizer(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void optimize() {
        // 识别循环
        vector<Loop> loops = identifyLoops();
        
        // 对每个循环进行优化
        for (auto& loop : loops) {
            optimizeLoop(loop);
        }
    }
    
    vector<Loop> identifyLoops() {
        vector<Loop> loops;
        
        // 使用深度优先搜索识别循环
        for (auto block : cfg->getBlocks()) {
            if (isLoopHeader(block)) {
                Loop loop;
                loop.header = block;
                loop.body = getLoopBody(block);
                loops.push_back(loop);
            }
        }
        
        return loops;
    }
    
    bool isLoopHeader(BasicBlock* block) {
        // 检查是否有回边指向该块
        for (auto pred : block->predecessors) {
            if (isReachable(pred, block)) {
                return true;
            }
        }
        return false;
    }
    
    vector<BasicBlock*> getLoopBody(BasicBlock* header) {
        vector<BasicBlock*> body;
        set<BasicBlock*> visited;
        
        // 使用深度优先搜索收集循环体
        dfsLoopBody(header, header, body, visited);
        
        return body;
    }
    
    void dfsLoopBody(BasicBlock* current, BasicBlock* header, vector<BasicBlock*>& body, set<BasicBlock*>& visited) {
        if (visited.find(current) != visited.end()) {
            return;
        }
        
        visited.insert(current);
        body.push_back(current);
        
        for (auto succ : current->successors) {
            if (succ == header || isReachable(succ, header)) {
                dfsLoopBody(succ, header, body, visited);
            }
        }
    }
    
    bool isReachable(BasicBlock* from, BasicBlock* to) {
        set<BasicBlock*> visited;
        return dfsReachable(from, to, visited);
    }
    
    bool dfsReachable(BasicBlock* current, BasicBlock* target, set<BasicBlock*>& visited) {
        if (current == target) {
            return true;
        }
        
        if (visited.find(current) != visited.end()) {
            return false;
        }
        
        visited.insert(current);
        
        for (auto succ : current->successors) {
            if (dfsReachable(succ, target, visited)) {
                return true;
            }
        }
        
        return false;
    }
    
    void optimizeLoop(Loop& loop) {
        // 循环不变式外提
        loopInvariantCodeMotion(loop);
        
        // 循环展开
        loopUnrolling(loop);
        
        // 循环融合
        loopFusion(loop);
    }
    
    void loopInvariantCodeMotion(Loop& loop) {
        // 识别循环不变式
        vector<Instruction> invariantInstructions = identifyInvariantInstructions(loop);
        
        // 将循环不变式外提
        for (auto& inst : invariantInstructions) {
            moveInstructionOutOfLoop(inst, loop);
        }
    }
    
    vector<Instruction> identifyInvariantInstructions(Loop& loop) {
        vector<Instruction> invariantInstructions;
        
        for (auto block : loop.body) {
            for (auto inst : block->instructions) {
                if (isInvariant(inst, loop)) {
                    invariantInstructions.push_back(inst);
                }
            }
        }
        
        return invariantInstructions;
    }
    
    bool isInvariant(Instruction& inst, Loop& loop) {
        if (inst.type == ASSIGNMENT) {
            // 检查操作数是否在循环内定义
            if (isDefinedInLoop(inst.arg1, loop) || isDefinedInLoop(inst.arg2, loop)) {
                return false;
            }
            return true;
        }
        return false;
    }
    
    bool isDefinedInLoop(string var, Loop& loop) {
        for (auto block : loop.body) {
            for (auto inst : block->instructions) {
                if (inst.type == ASSIGNMENT && inst.result == var) {
```

```c
class RegisterAllocationOptimizer {
private:
    ControlFlowGraph* cfg;
    int numRegisters;
    map<string, int> variableToRegister;
    
public:
    RegisterAllocationOptimizer(ControlFlowGraph* cfg, int numRegs) : cfg(cfg), numRegisters(numRegs) {}
    
    void optimize() {
        // 计算活跃变量
        computeLiveVariables();
        
        // 进行寄存器分配
        allocateRegisters();
    }
    
    void computeLiveVariables() {
        // 计算每个基本块的活跃变量
        for (auto block : cfg->getBlocks()) {
            computeLiveVariablesForBlock(block);
        }
    }
    
    void computeLiveVariablesForBlock(BasicBlock* block) {
        set<string> liveOut;
        
        // 初始化liveOut
        for (auto succ : block->successors) {
            for (auto var : succ->liveIn) {
                liveOut.insert(var);
            }
        }
        
        // 从后往前计算liveIn
        set<string> liveIn = liveOut;
        
        for (int i = block->instructions.size() - 1; i >= 0; i--) {
            Instruction& inst = block->instructions[i];
            
            if (inst.type == ASSIGNMENT) {
                // 移除定义的变量
                liveIn.erase(inst.result);
                
                // 添加使用的变量
                if (!inst.arg1.empty() && !isConstant(inst.arg1)) {
                    liveIn.insert(inst.arg1);
                }
                if (!inst.arg2.empty() && !isConstant(inst.arg2)) {
                    liveIn.insert(inst.arg2);
                }
            }
        }
        
        block->liveIn = liveIn;
        block->liveOut = liveOut;
    }
    
    void allocateRegisters() {
        // 使用图着色算法进行寄存器分配
        map<string, set<string>> interferenceGraph = buildInterferenceGraph();
        
        // 对变量进行着色
        colorVariables(interferenceGraph);
    }
    
    map<string, set<string>> buildInterferenceGraph() {
        map<string, set<string>> interferenceGraph;
        
        for (auto block : cfg->getBlocks()) {
            set<string> live = block->liveIn;
            
            for (auto inst : block->instructions) {
                if (inst.type == ASSIGNMENT) {
                    // 添加冲突边
                    for (auto var : live) {
                        if (var != inst.result) {
                            interferenceGraph[inst.result].insert(var);
                            interferenceGraph[var].insert(inst.result);
                        }
                    }
                    
                    // 更新活跃变量集合
                    live.erase(inst.result);
                    if (!inst.arg1.empty() && !isConstant(inst.arg1)) {
                        live.insert(inst.arg1);
                    }
                    if (!inst.arg2.empty() && !isConstant(inst.arg2)) {
                        live.insert(inst.arg2);
                    }
                }
            }
        }
        
        return interferenceGraph;
    }
    
    void colorVariables(map<string, set<string>>& interferenceGraph) {
        // 使用贪心算法进行着色
        for (auto& pair : interferenceGraph) {
            string var = pair.first;
            set<string>& neighbors = pair.second;
            
            // 找到可用的寄存器
            int registerNum = findAvailableRegister(neighbors);
            variableToRegister[var] = registerNum;
        }
    }
    
    int findAvailableRegister(set<string>& neighbors) {
        set<int> usedRegisters;
        
        for (auto neighbor : neighbors) {
            if (variableToRegister.find(neighbor) != variableToRegister.end()) {
                usedRegisters.insert(variableToRegister[neighbor]);
            }
        }
        
        // 找到第一个可用的寄存器
        for (int i = 0; i < numRegisters; i++) {
            if (usedRegisters.find(i) == usedRegisters.end()) {
                return i;
            }
        }
        
        // 如果没有可用寄存器，返回-1表示需要溢出
        return -1;
    }
    
    bool isConstant(string str) {
        return !str.empty() && str.find_first_not_of("0123456789") == string::npos;
    }
};
```

```c
class InstructionSchedulingOptimizer {
private:
    ControlFlowGraph* cfg;
    
public:
    InstructionSchedulingOptimizer(ControlFlowGraph* cfg) : cfg(cfg) {}
    
    void optimize() {
        // 对每个基本块进行指令调度
        for (auto block : cfg->getBlocks()) {
            scheduleInstructions(block);
        }
    }
    
    void scheduleInstructions(BasicBlock* block) {
        vector<Instruction> scheduledInstructions;
        vector<Instruction> remainingInstructions = block->instructions;
        
        while (!remainingInstructions.empty()) {
            // 找到可以调度的指令
            vector<Instruction> readyInstructions = findReadyInstructions(remainingInstructions, scheduledInstructions);
            
            if (readyInstructions.empty()) {
                // 没有可调度的指令，选择第一个
                scheduledInstructions.push_back(remainingInstructions[0]);
                remainingInstructions.erase(remainingInstructions.begin());
            } else {
                // 选择最优的指令
                Instruction bestInst = selectBestInstruction(readyInstructions);
                scheduledInstructions.push_back(bestInst);
                
                // 从剩余指令中移除
                for (auto it = remainingInstructions.begin(); it != remainingInstructions.end(); ++it) {
                    if (*it == bestInst) {
                        remainingInstructions.erase(it);
                        break;
                    }
                }
            }
        }
        
        block->instructions = scheduledInstructions;
    }
    
    vector<Instruction> findReadyInstructions(vector<Instruction>& remaining, vector<Instruction>& scheduled) {
        vector<Instruction> readyInstructions;
        
        for (auto inst : remaining) {
            if (isReady(inst, scheduled)) {
                readyInstructions.push_back(inst);
            }
        }
        
        return readyInstructions;
    }
    
    bool isReady(Instruction& inst, vector<Instruction>& scheduled) {
        if (inst.type == ASSIGNMENT) {
            // 检查操作数是否已经计算
            if (!inst.arg1.empty() && !isConstant(inst.arg1) && !isComputed(inst.arg1, scheduled)) {
                return false;
            }
            if (!inst.arg2.empty() && !isConstant(inst.arg2) && !isComputed(inst.arg2, scheduled)) {
                return false;
            }
        }
        return true;
    }
    
    bool isComputed(string var, vector<Instruction>& scheduled) {
        for (auto inst : scheduled) {
            if (inst.type == ASSIGNMENT && inst.result == var) {
                return true;
            }
        }
        return false;
    }
    
    Instruction selectBestInstruction(vector<Instruction>& readyInstructions) {
        // 选择延迟最小的指令
        Instruction bestInst = readyInstructions[0];
        int minLatency = getLatency(bestInst);
        
        for (auto inst : readyInstructions) {
            int latency = getLatency(inst);
            if (latency < minLatency) {
                minLatency = latency;
                bestInst = inst;
            }
        }
        
        return bestInst;
    }
    
    int getLatency(Instruction& inst) {
        // 根据指令类型返回延迟
        switch (inst.type) {
            case ASSIGNMENT:
                if (inst.op == "+" || inst.op == "-") return 1;
                if (inst.op == "*") return 3;
                if (inst.op == "/") return 10;
                return 1;
            case FUNCTION_CALL:
                return 20;
            default:
                return 1;
        }
    }
    
    bool isConstant(string str) {
        return !str.empty() && str.find_first_not_of("0123456789") == string::npos;
    }
};
```

```c
class IntermediateCodeOptimizer {
private:
    vector<Instruction> instructions;
    ControlFlowGraph* cfg;
    
public:
    IntermediateCodeOptimizer(vector<Instruction>& insts) : instructions(insts) {
        cfg = nullptr;
    }
    
    vector<Instruction> optimize() {
        // 基本优化
        instructions = optimizeBasic();
        
        // 构建控制流图
        cfg = buildControlFlowGraph();
        
        // 高级优化
        optimizeAdvanced();
        
        return instructions;
    }
    
    vector<Instruction> optimizeBasic() {
        // 常量折叠
        ConstantFoldingOptimizer constantFolding(instructions);
        instructions = constantFolding.optimize();
        
        // 死代码消除
        DeadCodeEliminationOptimizer deadCodeElimination(instructions);
        instructions = deadCodeElimination.optimize();
        
        // 公共子表达式消除
        CommonSubexpressionEliminationOptimizer commonSubexpressionElimination(instructions);
        instructions = commonSubexpressionElimination.optimize();
        
        // 强度削弱
        StrengthReductionOptimizer strengthReduction(instructions);
        instructions = strengthReduction.optimize();
        
        return instructions;
    }
    
    ControlFlowGraph* buildControlFlowGraph() {
        CFGBuilder cfgBuilder;
        return cfgBuilder.buildCFG(instructions);
    }
    
    void optimizeAdvanced() {
        if (cfg == nullptr) return;
        
        // 循环优化
        LoopOptimizationOptimizer loopOptimization(cfg);
        loopOptimization.optimize();
        
        // 寄存器分配
        RegisterAllocationOptimizer registerAllocation(cfg, 16);
        registerAllocation.optimize();
        
        // 指令调度
        InstructionSchedulingOptimizer instructionScheduling(cfg);
        instructionScheduling.optimize();
    }
};
```
