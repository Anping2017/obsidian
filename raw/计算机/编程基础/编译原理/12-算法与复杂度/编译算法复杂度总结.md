# 编译算法复杂度总结

## 🎯 编译算法复杂度概述

**编译算法复杂度总结**是编译器原理学习的重要组成部分，涵盖了编译过程中各个阶段算法的时间复杂度、空间复杂度以及性能特征。理解这些复杂度有助于选择合适的算法和优化编译器性能。

## 🔍 复杂度分类

### 1. 时间复杂度
- **最好情况**：算法在最理想情况下的时间复杂度
- **平均情况**：算法在平均情况下的时间复杂度
- **最坏情况**：算法在最不利情况下的时间复杂度

### 2. 空间复杂度
- **辅助空间**：算法执行过程中需要的额外空间
- **输入空间**：存储输入数据所需的空间
- **输出空间**：存储输出结果所需的空间

### 3. 性能特征
- **可扩展性**：算法处理大规模数据的能力
- **稳定性**：算法性能的稳定性
- **实用性**：算法在实际应用中的表现

### 有限自动机算法

#### DFA模拟算法
**时间复杂度**：O(n)  
**空间复杂度**：O(|Q|)  
**其中**：n为输入长度，|Q|为状态数

```c
// DFA模拟算法复杂度分析
class DFASimulator {
private:
    int currentState;
    int** transitionTable;
    int numStates;
    int numSymbols;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(|Q|)
    bool simulate(const string& input) {
        currentState = 0; // 初始状态
        
        for (char c : input) {
            int symbol = getSymbolIndex(c);
            if (symbol == -1) return false;
            
            currentState = transitionTable[currentState][symbol];
            if (currentState == -1) return false;
        }
        
        return isAcceptState(currentState);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    int getSymbolIndex(char c) {
        if (c >= '0' && c <= '9') return c - '0';
        if (c >= 'a' && c <= 'z') return c - 'a' + 10;
        if (c >= 'A' && c <= 'Z') return c - 'A' + 36;
        return -1;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool isAcceptState(int state) {
        return acceptStates[state];
    }
};
```

#### NFA到DFA转换算法
**时间复杂度**：O(2^|Q| × |Σ|)  
**空间复杂度**：O(2^|Q| × |Σ|)  
**其中**：|Q|为NFA状态数，|Σ|为字母表大小

```c
// NFA到DFA转换算法复杂度分析
class NFAToDFAConverter {
private:
    vector<set<int>> dfaStates;
    map<set<int>, int> stateMap;
    
public:
    // 时间复杂度：O(2^|Q| × |Σ|)
    // 空间复杂度：O(2^|Q| × |Σ|)
    DFA convert(NFA nfa) {
        DFA dfa;
        
        // 初始状态：ε-闭包
        set<int> initialState = epsilonClosure(nfa.startState);
        dfaStates.push_back(initialState);
        stateMap[initialState] = 0;
        
        queue<int> stateQueue;
        stateQueue.push(0);
        
        while (!stateQueue.empty()) {
            int currentStateIndex = stateQueue.front();
            stateQueue.pop();
            
            set<int> currentState = dfaStates[currentStateIndex];
            
            // 对每个输入符号
            for (char symbol : alphabet) {
                set<int> nextState = epsilonClosure(
                    move(currentState, symbol));
                
                if (!nextState.empty()) {
                    int nextStateIndex = getOrCreateState(nextState);
                    dfa.addTransition(currentStateIndex, symbol, nextStateIndex);
                    
                    if (nextStateIndex == dfaStates.size() - 1) {
                        stateQueue.push(nextStateIndex);
                    }
                }
            }
        }
        
        return dfa;
    }
    
    // 时间复杂度：O(|Q|)
    // 空间复杂度：O(|Q|)
    set<int> epsilonClosure(int state) {
        set<int> closure;
        stack<int> stateStack;
        stateStack.push(state);
        
        while (!stateStack.empty()) {
            int currentState = stateStack.top();
            stateStack.pop();
            
            if (closure.find(currentState) == closure.end()) {
                closure.insert(currentState);
                
                for (int nextState : nfa.getEpsilonTransitions(currentState)) {
                    stateStack.push(nextState);
                }
            }
        }
        
        return closure;
    }
    
    // 时间复杂度：O(|Q|)
    // 空间复杂度：O(|Q|)
    set<int> epsilonClosure(const set<int>& states) {
        set<int> closure;
        
        for (int state : states) {
            set<int> stateClosure = epsilonClosure(state);
            closure.insert(stateClosure.begin(), stateClosure.end());
        }
        
        return closure;
    }
};
```

### 正则表达式算法

#### Thompson构造算法
**时间复杂度**：O(|r|)  
**空间复杂度**：O(|r|)  
**其中**：|r|为正则表达式的长度

```c
// Thompson构造算法复杂度分析
class ThompsonConstructor {
public:
    // 时间复杂度：O(|r|)
    // 空间复杂度：O(|r|)
    NFA construct(const string& regex) {
        stack<NFA> nfaStack;
        
        for (char c : regex) {
            switch (c) {
                case '|':
                    if (nfaStack.size() >= 2) {
                        NFA nfa2 = nfaStack.top(); nfaStack.pop();
                        NFA nfa1 = nfaStack.top(); nfaStack.pop();
                        nfaStack.push(unionNFA(nfa1, nfa2));
                    }
                    break;
                    
                case '*':
                    if (!nfaStack.empty()) {
                        NFA nfa = nfaStack.top(); nfaStack.pop();
                        nfaStack.push(kleeneStar(nfa));
                    }
                    break;
                    
                case '+':
                    if (!nfaStack.empty()) {
                        NFA nfa = nfaStack.top(); nfaStack.pop();
                        nfaStack.push(kleenePlus(nfa));
                    }
                    break;
                    
                case '?':
                    if (!nfaStack.empty()) {
                        NFA nfa = nfaStack.top(); nfaStack.pop();
                        nfaStack.push(optional(nfa));
                    }
                    break;
                    
                default:
                    nfaStack.push(singleSymbolNFA(c));
                    break;
            }
        }
        
        return nfaStack.empty() ? NFA() : nfaStack.top();
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    NFA singleSymbolNFA(char symbol) {
        NFA nfa;
        int start = nfa.addState();
        int end = nfa.addState();
        nfa.addTransition(start, symbol, end);
        nfa.setStartState(start);
        nfa.addAcceptState(end);
        return nfa;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    NFA unionNFA(const NFA& nfa1, const NFA& nfa2) {
        NFA nfa;
        int start = nfa.addState();
        int end = nfa.addState();
        
        nfa.addTransition(start, EPSILON, nfa1.getStartState());
        nfa.addTransition(start, EPSILON, nfa2.getStartState());
        nfa.addTransition(nfa1.getAcceptState(), EPSILON, end);
        nfa.addTransition(nfa2.getAcceptState(), EPSILON, end);
        
        nfa.setStartState(start);
        nfa.addAcceptState(end);
        return nfa;
    }
};
```

### 自顶向下分析算法

#### LL(1)分析算法
**时间复杂度**：O(n)  
**空间复杂度**：O(n)  
**其中**：n为输入长度

```c
// LL(1)分析算法复杂度分析
class LL1Parser {
private:
    map<pair<string, string>, Production> parseTable;
    stack<string> parseStack;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    bool parse(const vector<Token>& tokens) {
        parseStack.push("$");
        parseStack.push(startSymbol);
        
        int tokenIndex = 0;
        
        while (!parseStack.empty()) {
            string top = parseStack.top();
            parseStack.pop();
            
            if (top == "$") {
                return tokenIndex == tokens.size();
            }
            
            if (isTerminal(top)) {
                if (tokenIndex < tokens.size() && 
                    top == tokens[tokenIndex].type) {
                    tokenIndex++;
                } else {
                    return false;
                }
            } else {
                string lookahead = tokenIndex < tokens.size() ? 
                    tokens[tokenIndex].type : "$";
                
                auto key = make_pair(top, lookahead);
                if (parseTable.find(key) != parseTable.end()) {
                    Production prod = parseTable[key];
                    for (int i = prod.rhs.size() - 1; i >= 0; i--) {
                        parseStack.push(prod.rhs[i]);
                    }
                } else {
                    return false;
                }
            }
        }
        
        return false;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool isTerminal(const string& symbol) {
        return terminals.find(symbol) != terminals.end();
    }
};
```

#### 递归下降分析算法
**时间复杂度**：O(n)  
**空间复杂度**：O(n)  
**其中**：n为输入长度

```c
// 递归下降分析算法复杂度分析
class RecursiveDescentParser {
private:
    vector<Token> tokens;
    int currentToken;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    bool parse(const vector<Token>& inputTokens) {
        tokens = inputTokens;
        currentToken = 0;
        
        try {
            parseProgram();
            return currentToken == tokens.size();
        } catch (const ParseError& e) {
            return false;
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void parseProgram() {
        while (currentToken < tokens.size()) {
            if (isTypeSpecifier()) {
                parseDeclaration();
            } else if (isFunctionDefinition()) {
                parseFunctionDefinition();
            } else {
                throw ParseError("Expected declaration or function");
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void parseDeclaration() {
        parseTypeSpecifier();
        parseDeclarator();
        
        if (match("=")) {
            parseInitializer();
        }
        
        expect(";");
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void parseFunctionDefinition() {
        parseTypeSpecifier();
        parseDeclarator();
        parseCompoundStatement();
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void parseCompoundStatement() {
        expect("{");
        
        while (currentToken < tokens.size() && 
               tokens[currentToken].type != "}") {
            if (isTypeSpecifier()) {
                parseDeclaration();
            } else {
                parseStatement();
            }
        }
        
        expect("}");
    }
};
```

### 自底向上分析算法

#### LR(1)分析算法
**时间复杂度**：O(n)  
**空间复杂度**：O(n)  
**其中**：n为输入长度

```c
// LR(1)分析算法复杂度分析
class LR1Parser {
private:
    vector<LR1State> states;
    map<pair<int, string>, Action> actionTable;
    map<pair<int, string>, int> gotoTable;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    bool parse(const vector<Token>& tokens) {
        stack<int> stateStack;
        stack<string> symbolStack;
        
        stateStack.push(0);
        
        int tokenIndex = 0;
        
        while (tokenIndex < tokens.size()) {
            int currentState = stateStack.top();
            string currentSymbol = tokens[tokenIndex].type;
            
            auto key = make_pair(currentState, currentSymbol);
            
            if (actionTable.find(key) != actionTable.end()) {
                Action action = actionTable[key];
                
                switch (action.type) {
                    case SHIFT:
                        stateStack.push(action.state);
                        symbolStack.push(currentSymbol);
                        tokenIndex++;
                        break;
                        
                    case REDUCE:
                        Production prod = action.production;
                        for (int i = 0; i < prod.rhs.size(); i++) {
                            stateStack.pop();
                            symbolStack.pop();
                        }
                        
                        symbolStack.push(prod.lhs);
                        
                        auto gotoKey = make_pair(stateStack.top(), prod.lhs);
                        if (gotoTable.find(gotoKey) != gotoTable.end()) {
                            stateStack.push(gotoTable[gotoKey]);
                        } else {
                            return false;
                        }
                        break;
                        
                    case ACCEPT:
                        return true;
                        
                    default:
                        return false;
                }
            } else {
                return false;
            }
        }
        
        return false;
    }
};
```

### 符号表管理算法

#### 哈希表符号表
**查找时间复杂度**：O(1)  
**插入时间复杂度**：O(1)  
**空间复杂度**：O(n)  
**其中**：n为符号数量

```c
// 哈希表符号表复杂度分析
class HashSymbolTable {
private:
    map<string, Symbol> symbolTable;
    int hashTableSize;
    
public:
    // 查找时间复杂度：O(1)
    // 插入时间复杂度：O(1)
    // 空间复杂度：O(n)
    SymbolTable() : hashTableSize(1000) {
        symbolTable.reserve(hashTableSize);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void insert(const string& name, const Symbol& symbol) {
        symbolTable[name] = symbol;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    Symbol* lookup(const string& name) {
        auto it = symbolTable.find(name);
        return (it != symbolTable.end()) ? &it->second : nullptr;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool contains(const string& name) {
        return symbolTable.find(name) != symbolTable.end();
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    void remove(const string& name) {
        symbolTable.erase(name);
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    vector<string> getAllSymbols() {
        vector<string> symbols;
        for (const auto& pair : symbolTable) {
            symbols.push_back(pair.first);
        }
        return symbols;
    }
};
```

#### 树形符号表
**查找时间复杂度**：O(log n)  
**插入时间复杂度**：O(log n)  
**空间复杂度**：O(n)  
**其中**：n为符号数量

```c
// 树形符号表复杂度分析
class TreeSymbolTable {
private:
    struct SymbolNode {
        string name;
        Symbol symbol;
        SymbolNode* left;
        SymbolNode* right;
        
        SymbolNode(const string& n, const Symbol& s) 
            : name(n), symbol(s), left(nullptr), right(nullptr) {}
    };
    
    SymbolNode* root;
    
public:
    // 查找时间复杂度：O(log n)
    // 插入时间复杂度：O(log n)
    // 空间复杂度：O(n)
    TreeSymbolTable() : root(nullptr) {}
    
    // 时间复杂度：O(log n)
    // 空间复杂度：O(1)
    void insert(const string& name, const Symbol& symbol) {
        root = insertNode(root, name, symbol);
    }
    
    // 时间复杂度：O(log n)
    // 空间复杂度：O(1)
    SymbolNode* insertNode(SymbolNode* node, const string& name, const Symbol& symbol) {
        if (node == nullptr) {
            return new SymbolNode(name, symbol);
        }
        
        if (name < node->name) {
            node->left = insertNode(node->left, name, symbol);
        } else if (name > node->name) {
            node->right = insertNode(node->right, name, symbol);
        } else {
            node->symbol = symbol; // 更新现有符号
        }
        
        return node;
    }
    
    // 时间复杂度：O(log n)
    // 空间复杂度：O(1)
    Symbol* lookup(const string& name) {
        SymbolNode* node = findNode(root, name);
        return (node != nullptr) ? &node->symbol : nullptr;
    }
    
    // 时间复杂度：O(log n)
    // 空间复杂度：O(1)
    SymbolNode* findNode(SymbolNode* node, const string& name) {
        if (node == nullptr || node->name == name) {
            return node;
        }
        
        if (name < node->name) {
            return findNode(node->left, name);
        } else {
            return findNode(node->right, name);
        }
    }
};
```

### 类型检查算法

#### 类型推断算法
**时间复杂度**：O(n²)  
**空间复杂度**：O(n)  
**其中**：n为表达式节点数

```c
// 类型推断算法复杂度分析
class TypeInference {
private:
    map<string, Type> typeEnvironment;
    map<ASTNode*, Type> nodeTypes;
    
public:
    // 时间复杂度：O(n²)
    // 空间复杂度：O(n)
    Type inferType(ASTNode* node) {
        if (nodeTypes.find(node) != nodeTypes.end()) {
            return nodeTypes[node];
        }
        
        Type type;
        
        switch (node->type) {
            case "number":
                type = Type::INT;
                break;
                
            case "string":
                type = Type::STRING;
                break;
                
            case "identifier":
                type = lookupType(node->value);
                break;
                
            case "add":
            case "sub":
            case "mul":
            case "div":
                type = inferArithmeticType(node);
                break;
                
            case "function_call":
                type = inferFunctionCallType(node);
                break;
                
            default:
                type = Type::UNKNOWN;
                break;
        }
        
        nodeTypes[node] = type;
        return type;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    Type inferArithmeticType(ASTNode* node) {
        Type leftType = inferType(node->children[0]);
        Type rightType = inferType(node->children[1]);
        
        if (leftType == Type::INT && rightType == Type::INT) {
            return Type::INT;
        } else if (leftType == Type::FLOAT || rightType == Type::FLOAT) {
            return Type::FLOAT;
        } else {
            return Type::UNKNOWN;
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    Type inferFunctionCallType(ASTNode* node) {
        string functionName = node->children[0]->value;
        FunctionType* funcType = lookupFunctionType(functionName);
        
        if (funcType == nullptr) {
            return Type::UNKNOWN;
        }
        
        return funcType->returnType;
    }
};
```

## 📊 代码优化算法复杂度

### 数据流分析算法

#### 活跃变量分析
**时间复杂度**：O(n × m)  
**空间复杂度**：O(n × m)  
**其中**：n为基本块数，m为变量数

```c
// 活跃变量分析算法复杂度分析
class LiveVariableAnalysis {
private:
    map<BasicBlock*, set<string>> liveIn;
    map<BasicBlock*, set<string>> liveOut;
    map<BasicBlock*, set<string>> def;
    map<BasicBlock*, set<string>> use;
    
public:
    // 时间复杂度：O(n × m)
    // 空间复杂度：O(n × m)
    void analyze(ControlFlowGraph* cfg) {
        initialize(cfg);
        
        bool changed = true;
        while (changed) {
            changed = false;
            
            for (BasicBlock* block : cfg->getBlocks()) {
                set<string> oldLiveIn = liveIn[block];
                set<string> oldLiveOut = liveOut[block];
                
                // LiveOut[B] = ∪(LiveIn[S]) for all successors S of B
                liveOut[block].clear();
                for (BasicBlock* successor : block->getSuccessors()) {
                    liveOut[block].insert(liveIn[successor].begin(), 
                                        liveIn[successor].end());
                }
                
                // LiveIn[B] = Use[B] ∪ (LiveOut[B] - Def[B])
                liveIn[block] = use[block];
                for (string var : liveOut[block]) {
                    if (def[block].find(var) == def[block].end()) {
                        liveIn[block].insert(var);
                    }
                }
                
                if (oldLiveIn != liveIn[block] || oldLiveOut != liveOut[block]) {
                    changed = true;
                }
            }
        }
    }
    
    // 时间复杂度：O(n × m)
    // 空间复杂度：O(n × m)
    void initialize(ControlFlowGraph* cfg) {
        for (BasicBlock* block : cfg->getBlocks()) {
            liveIn[block] = set<string>();
            liveOut[block] = set<string>();
            def[block] = computeDef(block);
            use[block] = computeUse(block);
        }
    }
    
    // 时间复杂度：O(m)
    // 空间复杂度：O(m)
    set<string> computeDef(BasicBlock* block) {
        set<string> defSet;
        
        for (Instruction* inst : block->getInstructions()) {
            if (inst->hasDestination()) {
                defSet.insert(inst->getDestination());
            }
        }
        
        return defSet;
    }
    
    // 时间复杂度：O(m)
    // 空间复杂度：O(m)
    set<string> computeUse(BasicBlock* block) {
        set<string> useSet;
        
        for (Instruction* inst : block->getInstructions()) {
            for (string operand : inst->getOperands()) {
                useSet.insert(operand);
            }
        }
        
        return useSet;
    }
};
```

#### 到达定值分析
**时间复杂度**：O(n × m)  
**空间复杂度**：O(n × m)  
**其中**：n为基本块数，m为定值数

```c
// 到达定值分析算法复杂度分析
class ReachingDefinitionAnalysis {
private:
    map<BasicBlock*, set<int>> reachIn;
    map<BasicBlock*, set<int>> reachOut;
    map<BasicBlock*, set<int>> gen;
    map<BasicBlock*, set<int>> kill;
    
public:
    // 时间复杂度：O(n × m)
    // 空间复杂度：O(n × m)
    void analyze(ControlFlowGraph* cfg) {
        initialize(cfg);
        
        bool changed = true;
        while (changed) {
            changed = false;
            
            for (BasicBlock* block : cfg->getBlocks()) {
                set<int> oldReachIn = reachIn[block];
                set<int> oldReachOut = reachOut[block];
                
                // ReachIn[B] = ∪(ReachOut[P]) for all predecessors P of B
                reachIn[block].clear();
                for (BasicBlock* predecessor : block->getPredecessors()) {
                    reachIn[block].insert(reachOut[predecessor].begin(), 
                                        reachOut[predecessor].end());
                }
                
                // ReachOut[B] = Gen[B] ∪ (ReachIn[B] - Kill[B])
                reachOut[block] = gen[block];
                for (int def : reachIn[block]) {
                    if (kill[block].find(def) == kill[block].end()) {
                        reachOut[block].insert(def);
                    }
                }
                
                if (oldReachIn != reachIn[block] || oldReachOut != reachOut[block]) {
                    changed = true;
                }
            }
        }
    }
    
    // 时间复杂度：O(n × m)
    // 空间复杂度：O(n × m)
    void initialize(ControlFlowGraph* cfg) {
        for (BasicBlock* block : cfg->getBlocks()) {
            reachIn[block] = set<int>();
            reachOut[block] = set<int>();
            gen[block] = computeGen(block);
            kill[block] = computeKill(block);
        }
    }
    
    // 时间复杂度：O(m)
    // 空间复杂度：O(m)
    set<int> computeGen(BasicBlock* block) {
        set<int> genSet;
        
        for (int i = 0; i < block->getInstructions().size(); i++) {
            Instruction* inst = block->getInstructions()[i];
            if (inst->hasDestination()) {
                genSet.insert(getDefinitionId(block, i));
            }
        }
        
        return genSet;
    }
    
    // 时间复杂度：O(m)
    // 空间复杂度：O(m)
    set<int> computeKill(BasicBlock* block) {
        set<int> killSet;
        
        for (int i = 0; i < block->getInstructions().size(); i++) {
            Instruction* inst = block->getInstructions()[i];
            if (inst->hasDestination()) {
                string var = inst->getDestination();
                for (BasicBlock* otherBlock : cfg->getBlocks()) {
                    for (int j = 0; j < otherBlock->getInstructions().size(); j++) {
                        Instruction* otherInst = otherBlock->getInstructions()[j];
                        if (otherInst->hasDestination() && 
                            otherInst->getDestination() == var) {
                            killSet.insert(getDefinitionId(otherBlock, j));
                        }
                    }
                }
            }
        }
        
        return killSet;
    }
};
```

#### 图着色算法
**时间复杂度**：O(n²)  
**空间复杂度**：O(n²)  
**其中**：n为变量数

```c
// 图着色算法复杂度分析
class GraphColoringAllocator {
private:
    map<string, set<string>> interferenceGraph;
    map<string, int> registerAssignment;
    int numRegisters;
    
public:
    // 时间复杂度：O(n²)
    // 空间复杂度：O(n²)
    bool allocateRegisters(const vector<string>& variables) {
        buildInterferenceGraph(variables);
        
        stack<string> variableStack;
        map<string, int> degree;
        
        // 计算每个变量的度数
        for (const string& var : variables) {
            degree[var] = interferenceGraph[var].size();
        }
        
        // 简化阶段：移除度数小于寄存器数的变量
        while (!variables.empty()) {
            bool found = false;
            
            for (const string& var : variables) {
                if (degree[var] < numRegisters) {
                    variableStack.push(var);
                    variables.erase(find(variables.begin(), variables.end(), var));
                    
                    // 更新相邻变量的度数
                    for (const string& neighbor : interferenceGraph[var]) {
                        degree[neighbor]--;
                    }
                    
                    found = true;
                    break;
                }
            }
            
            if (!found) {
                // 溢出：选择度数最大的变量
                string maxVar = *max_element(variables.begin(), variables.end(),
                    [&](const string& a, const string& b) {
                        return degree[a] < degree[b];
                    });
                
                variableStack.push(maxVar);
                variables.erase(find(variables.begin(), variables.end(), maxVar));
                
                for (const string& neighbor : interferenceGraph[maxVar]) {
                    degree[neighbor]--;
                }
            }
        }
        
        // 着色阶段：为变量分配寄存器
        while (!variableStack.empty()) {
            string var = variableStack.top();
            variableStack.pop();
            
            set<int> usedColors;
            for (const string& neighbor : interferenceGraph[var]) {
                if (registerAssignment.find(neighbor) != registerAssignment.end()) {
                    usedColors.insert(registerAssignment[neighbor]);
                }
            }
            
            // 找到第一个可用的颜色
            for (int color = 0; color < numRegisters; color++) {
                if (usedColors.find(color) == usedColors.end()) {
                    registerAssignment[var] = color;
                    break;
                }
            }
            
            if (registerAssignment.find(var) == registerAssignment.end()) {
                return false; // 分配失败
            }
        }
        
        return true;
    }
    
    // 时间复杂度：O(n²)
    // 空间复杂度：O(n²)
    void buildInterferenceGraph(const vector<string>& variables) {
        for (const string& var : variables) {
            interferenceGraph[var] = set<string>();
        }
        
        for (BasicBlock* block : cfg->getBlocks()) {
            set<string> liveVars = getLiveVariables(block);
            
            for (const string& var1 : liveVars) {
                for (const string& var2 : liveVars) {
                    if (var1 != var2) {
                        interferenceGraph[var1].insert(var2);
                        interferenceGraph[var2].insert(var1);
                    }
                }
            }
        }
    }
};
```

#### 线性扫描算法
**时间复杂度**：O(n log n)  
**空间复杂度**：O(n)  
**其中**：n为变量数

```c
// 线性扫描算法复杂度分析
class LinearScanAllocator {
private:
    vector<LiveInterval> intervals;
    vector<int> activeIntervals;
    int numRegisters;
    
public:
    // 时间复杂度：O(n log n)
    // 空间复杂度：O(n)
    void allocateRegisters(const vector<LiveInterval>& liveIntervals) {
        intervals = liveIntervals;
        sort(intervals.begin(), intervals.end(), 
             [](const LiveInterval& a, const LiveInterval& b) {
                 return a.start < b.start;
             });
        
        for (const LiveInterval& interval : intervals) {
            expireOldIntervals(interval.start);
            
            if (activeIntervals.size() < numRegisters) {
                // 有可用寄存器
                interval.registerId = activeIntervals.size();
                activeIntervals.push_back(interval.registerId);
            } else {
                // 溢出：选择结束时间最晚的区间
                int spillIndex = findSpillInterval();
                if (intervals[spillIndex].end > interval.end) {
                    // 溢出选中的区间
                    interval.registerId = intervals[spillIndex].registerId;
                    intervals[spillIndex].registerId = -1; // 标记为溢出
                    activeIntervals[spillIndex] = interval.registerId;
                } else {
                    // 溢出当前区间
                    interval.registerId = -1; // 标记为溢出
                }
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    void expireOldIntervals(int position) {
        for (int i = activeIntervals.size() - 1; i >= 0; i--) {
            int intervalIndex = activeIntervals[i];
            if (intervals[intervalIndex].end <= position) {
                activeIntervals.erase(activeIntervals.begin() + i);
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    int findSpillInterval() {
        int maxEnd = -1;
        int spillIndex = -1;
        
        for (int i = 0; i < activeIntervals.size(); i++) {
            int intervalIndex = activeIntervals[i];
            if (intervals[intervalIndex].end > maxEnd) {
                maxEnd = intervals[intervalIndex].end;
                spillIndex = i;
            }
        }
        
        return spillIndex;
    }
};
```
