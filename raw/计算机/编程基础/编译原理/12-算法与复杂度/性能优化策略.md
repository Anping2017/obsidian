# 性能优化策略

## 🎯 性能优化策略概述

**性能优化策略**是编译器原理学习的重要组成部分，涵盖了编译器性能优化的各种策略和方法。掌握这些策略有助于提高编译器的编译速度、减少内存使用、优化目标代码质量，并提高编译器的整体性能。

## 🔍 优化策略分类

### 1. 编译时间优化
- **并行编译**：多线程、多进程、分布式编译
- **增量编译**：只编译修改的部分
- **缓存优化**：中间结果缓存、预编译头文件
- **算法优化**：选择更高效的算法

### 2. 内存使用优化
- **内存池**：预分配内存、减少分配开销
- **对象池**：复用对象、减少构造析构
- **压缩存储**：位向量、稀疏矩阵
- **垃圾回收**：自动内存管理

### 3. 目标代码优化
- **指令选择**：选择最优指令序列
- **寄存器分配**：最小化内存访问
- **指令调度**：提高流水线效率
- **代码生成**：优化代码结构

### 并行编译策略

#### 多线程词法分析
**策略**：使用多线程并行处理不同文件

```c
// 多线程词法分析实现
class ParallelLexicalAnalysis {
private:
    vector<thread> workers;
    queue<string> fileQueue;
    mutex queueMutex;
    vector<Token> allTokens;
    mutex tokensMutex;
    
public:
    // 时间复杂度：O(n/p)
    // 空间复杂度：O(n)
    void analyzeFiles(const vector<string>& files) {
        // 将文件加入队列
        for (const string& file : files) {
            fileQueue.push(file);
        }
        
        // 创建工作线程
        int numThreads = thread::hardware_concurrency();
        for (int i = 0; i < numThreads; i++) {
            workers.emplace_back(&ParallelLexicalAnalysis::workerThread, this);
        }
        
        // 等待所有线程完成
        for (thread& worker : workers) {
            worker.join();
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void workerThread() {
        while (true) {
            string file;
            
            // 获取文件
            {
                lock_guard<mutex> lock(queueMutex);
                if (fileQueue.empty()) {
                    break;
                }
                file = fileQueue.front();
                fileQueue.pop();
            }
            
            // 分析文件
            vector<Token> tokens = analyzeFile(file);
            
            // 添加结果
            {
                lock_guard<mutex> lock(tokensMutex);
                allTokens.insert(allTokens.end(), tokens.begin(), tokens.end());
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    vector<Token> analyzeFile(const string& filename) {
        ifstream file(filename);
        string content((istreambuf_iterator<char>(file)),
                      istreambuf_iterator<char>());
        
        LexicalAnalyzer lexer;
        return lexer.analyze(content);
    }
};
```

```c
        }

    void workerThread() {

            ASTNode* node;
            
            // 获取节点
            {
                lock_guard<mutex> lock(queueMutex);
                if (nodeQueue.empty()) {
                    break;
                }

                nodeQueue.pop();

                lock_guard<mutex> lock(nodesMutex);
                parsedNodes.push_back(parsedNode);
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)

        SyntaxAnalyzer analyzer;
        return analyzer.analyze(node);
    }
};
```

### 增量编译策略

#### 文件依赖分析
**策略**：分析文件依赖关系，只编译修改的文件

```c
// 文件依赖分析实现
class FileDependencyAnalysis {
private:
    map<string, set<string>> dependencies;
    map<string, time_t> fileTimestamps;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void analyzeDependencies(const vector<string>& files) {
        for (const string& file : files) {
            set<string> deps = getFileDependencies(file);
            dependencies[file] = deps;
            fileTimestamps[file] = getFileTimestamp(file);
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    set<string> getFileDependencies(const string& filename) {
        set<string> deps;
        ifstream file(filename);
        string line;
        
        while (getline(file, line)) {
            // 查找#include语句
            if (line.find("#include") != string::npos) {
                string includeFile = extractIncludeFile(line);
                deps.insert(includeFile);
            }
        }
        
        return deps;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    vector<string> getModifiedFiles() {
        vector<string> modifiedFiles;
        
        for (const auto& pair : fileTimestamps) {
            const string& filename = pair.first;
            time_t currentTimestamp = getFileTimestamp(filename);
            
            if (currentTimestamp > pair.second) {
                modifiedFiles.push_back(filename);
            }
        }
        
        return modifiedFiles;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    vector<string> getFilesToRecompile(const vector<string>& modifiedFiles) {
        set<string> filesToRecompile;
        
        for (const string& file : modifiedFiles) {
            filesToRecompile.insert(file);
            
            // 添加依赖此文件的所有文件
            for (const auto& pair : dependencies) {
                const string& dependentFile = pair.first;
                const set<string>& deps = pair.second;
                
                if (deps.find(file) != deps.end()) {
                    filesToRecompile.insert(dependentFile);
                }
            }
        }
        
        return vector<string>(filesToRecompile.begin(), filesToRecompile.end());
    }
};
```

#### 增量语法分析
**策略**：只重新分析修改的语法单元

```c
// 增量语法分析实现
class IncrementalSyntaxAnalysis {
private:
    map<ASTNode*, ASTNode*> nodeCache;
    map<ASTNode*, bool> nodeModified;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void analyzeIncremental(ASTNode* root) {
        // 标记修改的节点
        markModifiedNodes(root);
        
        // 重新分析修改的节点
        for (const auto& pair : nodeModified) {
            if (pair.second) {
                ASTNode* node = pair.first;
                ASTNode* newNode = analyzeNode(node);
                nodeCache[node] = newNode;
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void markModifiedNodes(ASTNode* node) {
        if (node == nullptr) return;
        
        // 检查节点是否被修改
        bool modified = isNodeModified(node);
        nodeModified[node] = modified;
        
        // 递归处理子节点
        for (ASTNode* child : node->children) {
            markModifiedNodes(child);
        }
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool isNodeModified(ASTNode* node) {
        // 检查节点的时间戳或哈希值
        return node->timestamp > node->lastAnalyzed;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    ASTNode* analyzeNode(ASTNode* node) {
        // 检查缓存
        auto it = nodeCache.find(node);
        if (it != nodeCache.end()) {
            return it->second;
        }
        
        // 分析节点
        SyntaxAnalyzer analyzer;
        ASTNode* newNode = analyzer.analyze(node);
        
        // 缓存结果
        nodeCache[node] = newNode;
        return newNode;
    }
};
```

### 缓存优化策略

#### 中间结果缓存
**策略**：缓存中间分析结果，避免重复计算

```c
// 中间结果缓存实现
class IntermediateResultCache {
private:
    map<string, vector<Token>> lexicalCache;
    map<string, ASTNode*> syntaxCache;
    map<string, Type> typeCache;
    map<string, set<string>> dataFlowCache;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    vector<Token> getLexicalResult(const string& content) {
        auto it = lexicalCache.find(content);
        if (it != lexicalCache.end()) {
            return it->second;
        }
        
        LexicalAnalyzer lexer;
        vector<Token> tokens = lexer.analyze(content);
        lexicalCache[content] = tokens;
        return tokens;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    ASTNode* getSyntaxResult(const string& content) {
        auto it = syntaxCache.find(content);
        if (it != syntaxCache.end()) {
            return it->second;
        }
        
        SyntaxAnalyzer parser;
        ASTNode* ast = parser.parse(content);
        syntaxCache[content] = ast;
        return ast;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    Type getTypeResult(ASTNode* node) {
        string key = getNodeKey(node);
        auto it = typeCache.find(key);
        if (it != typeCache.end()) {
            return it->second;
        }
        
        TypeChecker checker;
        Type type = checker.checkType(node);
        typeCache[key] = type;
        return type;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    set<string> getDataFlowResult(BasicBlock* block) {
        string key = getBlockKey(block);
        auto it = dataFlowCache.find(key);
        if (it != dataFlowCache.end()) {
            return it->second;
        }
        
        DataFlowAnalyzer analyzer;
        set<string> result = analyzer.analyze(block);
        dataFlowCache[key] = result;
        return result;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    void clearCache() {
        lexicalCache.clear();
        syntaxCache.clear();
        typeCache.clear();
        dataFlowCache.clear();
    }
};
```

#### 预编译头文件
**策略**：预编译常用头文件，减少重复编译

```c
// 预编译头文件实现
class PrecompiledHeader {
private:
    map<string, vector<Token>> precompiledTokens;
    map<string, ASTNode*> precompiledAST;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void precompileHeaders(const vector<string>& headers) {
        for (const string& header : headers) {
            precompileHeader(header);
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void precompileHeader(const string& header) {
        ifstream file(header);
        string content((istreambuf_iterator<char>(file)),
                      istreambuf_iterator<char>());
        
        // 词法分析
        LexicalAnalyzer lexer;
        vector<Token> tokens = lexer.analyze(content);
        precompiledTokens[header] = tokens;
        
        // 语法分析
        SyntaxAnalyzer parser;
        ASTNode* ast = parser.parse(content);
        precompiledAST[header] = ast;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    vector<Token> getPrecompiledTokens(const string& header) {
        auto it = precompiledTokens.find(header);
        if (it != precompiledTokens.end()) {
            return it->second;
        }
        return vector<Token>();
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    ASTNode* getPrecompiledAST(const string& header) {
        auto it = precompiledAST.find(header);
        if (it != precompiledAST.end()) {
            return it->second;
        }
        return nullptr;
    }
};
```

#### 对象池管理
**策略**：使用对象池减少内存分配开销

```c
// 对象池管理实现
template<typename T>
class ObjectPool {
private:
    queue<T*> availableObjects;
    vector<T*> allObjects;
    mutex poolMutex;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(n)
    ObjectPool(int initialSize = 100) {
        for (int i = 0; i < initialSize; i++) {
            T* obj = new T();
            availableObjects.push(obj);
            allObjects.push_back(obj);
        }
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    T* acquire() {
        lock_guard<mutex> lock(poolMutex);
        
        if (availableObjects.empty()) {
            // 创建新对象
            T* obj = new T();
            allObjects.push_back(obj);
            return obj;
        }
        
        T* obj = availableObjects.front();
        availableObjects.pop();
        return obj;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void release(T* obj) {
        if (obj == nullptr) return;
        
        lock_guard<mutex> lock(poolMutex);
        availableObjects.push(obj);
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    ~ObjectPool() {
        for (T* obj : allObjects) {
            delete obj;
        }
    }
};
```

#### 内存池分配器
**策略**：使用内存池分配器优化内存分配

```c
// 内存池分配器实现
class MemoryPoolAllocator {
private:
    struct MemoryBlock {
        char* data;
        size_t size;
        bool inUse;
        
        MemoryBlock(char* d, size_t s) : data(d), size(s), inUse(false) {}
    };
    
    vector<MemoryBlock> blocks;
    size_t blockSize;
    size_t currentBlock;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(n)
    MemoryPoolAllocator(size_t blockSize = 4096) 
        : blockSize(blockSize), currentBlock(0) {
        allocateNewBlock();
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void* allocate(size_t size) {
        // 查找可用块
        for (size_t i = 0; i < blocks.size(); i++) {
            if (!blocks[i].inUse && blocks[i].size >= size) {
                blocks[i].inUse = true;
                return blocks[i].data;
            }
        }
        
        // 分配新块
        allocateNewBlock();
        blocks[currentBlock].inUse = true;
        return blocks[currentBlock].data;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void deallocate(void* ptr) {
        for (MemoryBlock& block : blocks) {
            if (block.data == ptr) {
                block.inUse = false;
                return;
            }
        }
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void allocateNewBlock() {
        char* data = new char[blockSize];
        blocks.emplace_back(data, blockSize);
        currentBlock = blocks.size() - 1;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    ~MemoryPoolAllocator() {
        for (MemoryBlock& block : blocks) {
            delete[] block.data;
        }
    }
};
```

#### 位向量压缩
**策略**：使用位向量压缩存储状态集合

```c
// 位向量压缩实现
class BitVectorCompression {
private:
    vector<uint64_t> bits;
    int numElements;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(n/64)
    BitVectorCompression(int elements) : numElements(elements) {
        int numWords = (elements + 63) / 64;
        bits.resize(numWords, 0);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void set(int index) {
        int wordIndex = index / 64;
        int bitIndex = index % 64;
        bits[wordIndex] |= (1ULL << bitIndex);
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool get(int index) {
        int wordIndex = index / 64;
        int bitIndex = index % 64;
        return (bits[wordIndex] & (1ULL << bitIndex)) != 0;
    }
    
    // 时间复杂度：O(n/64)
    // 空间复杂度：O(1)
    void unionWith(const BitVectorCompression& other) {
        for (int i = 0; i < bits.size(); i++) {
            bits[i] |= other.bits[i];
        }
    }
    
    // 时间复杂度：O(n/64)
    // 空间复杂度：O(1)
    void intersectWith(const BitVectorCompression& other) {
        for (int i = 0; i < bits.size(); i++) {
            bits[i] &= other.bits[i];
        }
    }
    
    // 时间复杂度：O(n/64)
    // 空间复杂度：O(1)
    int count() {
        int count = 0;
        for (uint64_t word : bits) {
            count += __builtin_popcountll(word);
        }
        return count;
    }
};
```

```c
// 稀疏矩阵压缩实现
class SparseMatrixCompression {
private:
    struct SparseEntry {
        int row;
        int col;
        int value;
        
        SparseEntry(int r, int c, int v) : row(r), col(c), value(v) {}
    };
    
    vector<SparseEntry> entries;
    int numRows;
    int numCols;
    
public:
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    SparseMatrixCompression(int rows, int cols) 
        : numRows(rows), numCols(cols) {}
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    void set(int row, int col, int value) {
        if (value != 0) {
            entries.emplace_back(row, col, value);
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    int get(int row, int col) {
        for (const SparseEntry& entry : entries) {
            if (entry.row == row && entry.col == col) {
                return entry.value;
            }
        }
        return 0;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    vector<int> getRow(int row) {
        vector<int> result(numCols, 0);
        
        for (const SparseEntry& entry : entries) {
            if (entry.row == row) {
                result[entry.col] = entry.value;
            }
        }
        
        return result;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    vector<int> getCol(int col) {
        vector<int> result(numRows, 0);
        
        for (const SparseEntry& entry : entries) {
            if (entry.col == col) {
                result[entry.row] = entry.value;
            }
        }
        
        return result;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    int getNonZeroCount() {
        return entries.size();
    }
};
```

#### 模板匹配优化

```c
// 模板匹配优化实现
class TemplateMatchingOptimizer {
private:
    struct InstructionTemplate {
        string pattern;
        string instruction;
        int cost;
        
        InstructionTemplate(const string& p, const string& i, int c)
            : pattern(p), instruction(i), cost(c) {}
    };
    
    vector<InstructionTemplate> templates;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    void addTemplate(const string& pattern, const string& instruction, int cost) {
        templates.emplace_back(pattern, instruction, cost);
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    string selectInstruction(ASTNode* node) {
        string bestInstruction;
        int minCost = INT_MAX;
        
        for (const InstructionTemplate& template : templates) {
            if (matchesPattern(node, template.pattern)) {
                if (template.cost < minCost) {
                    minCost = template.cost;
                    bestInstruction = template.instruction;
                }
            }
        }
        
        return bestInstruction;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    bool matchesPattern(ASTNode* node, const string& pattern) {
        // 简化的模式匹配
        return node->type == pattern;
    }
};
```

```c
// 树模式匹配实现
class TreePatternMatching {
private:
    struct TreePattern {
        ASTNode* pattern;
        string instruction;
        int cost;
        
        TreePattern(ASTNode* p, const string& i, int c)
            : pattern(p), instruction(i), cost(c) {}
    };
    
    vector<TreePattern> patterns;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    void addPattern(ASTNode* pattern, const string& instruction, int cost) {
        patterns.emplace_back(pattern, instruction, cost);
    }
    
    // 时间复杂度：O(n × m)
    // 空间复杂度：O(1)
    string matchPattern(ASTNode* node) {
        string bestInstruction;
        int minCost = INT_MAX;
        
        for (const TreePattern& pattern : patterns) {
            if (matchesTreePattern(node, pattern.pattern)) {
                if (pattern.cost < minCost) {
                    minCost = pattern.cost;
                    bestInstruction = pattern.instruction;
                }
            }
        }
        
        return bestInstruction;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    bool matchesTreePattern(ASTNode* node, ASTNode* pattern) {
        if (node == nullptr && pattern == nullptr) {
            return true;
        }
        
        if (node == nullptr || pattern == nullptr) {
            return false;
        }
        
        if (node->type != pattern->type) {
            return false;
        }
        
        if (node->children.size() != pattern->children.size()) {
            return false;
        }
        
        for (size_t i = 0; i < node->children.size(); i++) {
            if (!matchesTreePattern(node->children[i], pattern->children[i])) {
                return false;
            }
        }
        
        return true;
    }
};
```

#### 图着色优化
**策略**：使用图着色算法优化寄存器分配

```c
// 图着色优化实现
class GraphColoringOptimizer {
private:
    map<string, set<string>> interferenceGraph;
    map<string, int> registerAssignment;
    int numRegisters;
    
public:
    // 时间复杂度：O(n²)
    // 空间复杂度：O(n²)
    bool allocateRegisters(const vector<string>& variables) {
        buildInterferenceGraph(variables);
        
        // 简化阶段
        stack<string> variableStack;
        map<string, int> degree;
        
        for (const string& var : variables) {
            degree[var] = interferenceGraph[var].size();
        }
        
        while (!variables.empty()) {
            bool found = false;
            
            for (const string& var : variables) {
                if (degree[var] < numRegisters) {
                    variableStack.push(var);
                    variables.erase(find(variables.begin(), variables.end(), var));
                    
                    for (const string& neighbor : interferenceGraph[var]) {
                        degree[neighbor]--;
                    }
                    
                    found = true;
                    break;
                }
            }
            
            if (!found) {
                // 溢出处理
                string maxVar = *max_element(variables.begin(), variables.end(),
                    [&](const string& a, const string& b) {
                        return degree[a] < degree[b];
                    });
                
                variableStack.push(maxVar);
                variables.erase(find(variables.begin(), variables.end(), maxVar));
                
                for (const string& neighbor : interferenceGraph[maxVar]) {
                    degree[neighbor]--;
                }
            }
        }
        
        // 着色阶段
        while (!variableStack.empty()) {
            string var = variableStack.top();
            variableStack.pop();
            
            set<int> usedColors;
            for (const string& neighbor : interferenceGraph[var]) {
                if (registerAssignment.find(neighbor) != registerAssignment.end()) {
                    usedColors.insert(registerAssignment[neighbor]);
                }
            }
            
            for (int color = 0; color < numRegisters; color++) {
                if (usedColors.find(color) == usedColors.end()) {
                    registerAssignment[var] = color;
                    break;
                }
            }
            
            if (registerAssignment.find(var) == registerAssignment.end()) {
                return false; // 分配失败
            }
        }
        
        return true;
    }
    
    // 时间复杂度：O(n²)
    // 空间复杂度：O(n²)
    void buildInterferenceGraph(const vector<string>& variables) {
        for (const string& var : variables) {
            interferenceGraph[var] = set<string>();
        }
        
        for (BasicBlock* block : cfg->getBlocks()) {
            set<string> liveVars = getLiveVariables(block);
            
            for (const string& var1 : liveVars) {
                for (const string& var2 : liveVars) {
                    if (var1 != var2) {
                        interferenceGraph[var1].insert(var2);
                        interferenceGraph[var2].insert(var1);
                    }
                }
            }
        }
    }
};
```

```c
// 线性扫描优化实现
class LinearScanOptimizer {
private:
    vector<LiveInterval> intervals;
    vector<int> activeIntervals;
    int numRegisters;
    
public:
    // 时间复杂度：O(n log n)
    // 空间复杂度：O(n)
    void allocateRegisters(const vector<LiveInterval>& liveIntervals) {
        intervals = liveIntervals;
        sort(intervals.begin(), intervals.end(), 
             [](const LiveInterval& a, const LiveInterval& b) {
                 return a.start < b.start;
             });
        
        for (const LiveInterval& interval : intervals) {
            expireOldIntervals(interval.start);
            
            if (activeIntervals.size() < numRegisters) {
                // 有可用寄存器
                interval.registerId = activeIntervals.size();
                activeIntervals.push_back(interval.registerId);
            } else {
                // 溢出：选择结束时间最晚的区间
                int spillIndex = findSpillInterval();
                if (intervals[spillIndex].end > interval.end) {
                    // 溢出选中的区间
                    interval.registerId = intervals[spillIndex].registerId;
                    intervals[spillIndex].registerId = -1; // 标记为溢出
                    activeIntervals[spillIndex] = interval.registerId;
                } else {
                    // 溢出当前区间
                    interval.registerId = -1; // 标记为溢出
                }
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    void expireOldIntervals(int position) {
        for (int i = activeIntervals.size() - 1; i >= 0; i--) {
            int intervalIndex = activeIntervals[i];
            if (intervals[intervalIndex].end <= position) {
                activeIntervals.erase(activeIntervals.begin() + i);
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    int findSpillInterval() {
        int maxEnd = -1;
        int spillIndex = -1;
        
        for (int i = 0; i < activeIntervals.size(); i++) {
            int intervalIndex = activeIntervals[i];
            if (intervals[intervalIndex].end > maxEnd) {
                maxEnd = intervals[intervalIndex].end;
                spillIndex = i;
            }
        }
        
        return spillIndex;
    }
};
```

```c
// 流水线调度实现
class PipelineScheduler {
private:
    struct Instruction {
        string opcode;
        vector<string> operands;
        int latency;
        int issueCycle;
        
        Instruction(const string& op, const vector<string>& ops, int lat)
            : opcode(op), operands(ops), latency(lat), issueCycle(0) {}
    };
    
    vector<Instruction> instructions;
    int pipelineStages;
    
public:
    // 时间复杂度：O(n²)
    // 空间复杂度：O(n)
    void scheduleInstructions(const vector<Instruction>& insts) {
        instructions = insts;
        
        for (int i = 0; i < instructions.size(); i++) {
            scheduleInstruction(i);
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    void scheduleInstruction(int index) {
        Instruction& inst = instructions[index];
        
        // 查找最早的可用周期
        int earliestCycle = 0;
        
        for (int j = 0; j < index; j++) {
            const Instruction& prevInst = instructions[j];
            
            // 检查数据依赖
            if (hasDataDependency(inst, prevInst)) {
                int requiredCycle = prevInst.issueCycle + prevInst.latency;
                earliestCycle = max(earliestCycle, requiredCycle);
            }
        }
        
        inst.issueCycle = earliestCycle;
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    bool hasDataDependency(const Instruction& inst1, const Instruction& inst2) {
        // 检查操作数依赖
        for (const string& op1 : inst1.operands) {
            for (const string& op2 : inst2.operands) {
                if (op1 == op2) {
                    return true;
                }
            }
        }
        
        return false;
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(1)
    int getTotalCycles() {
        int maxCycle = 0;
        
        for (const Instruction& inst : instructions) {
            int endCycle = inst.issueCycle + inst.latency;
            maxCycle = max(maxCycle, endCycle);
        }
        
        return maxCycle;
    }
};
```

```c
// 循环展开优化实现
class LoopUnrollingOptimizer {
private:
    struct LoopInfo {
        BasicBlock* header;
        BasicBlock* body;
        BasicBlock* exit;
        int tripCount;
        bool canUnroll;
    };
    
    vector<LoopInfo> loops;
    
public:
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void analyzeLoops(ControlFlowGraph* cfg) {
        for (BasicBlock* block : cfg->getBlocks()) {
            if (isLoopHeader(block)) {
                LoopInfo loop;
                loop.header = block;
                loop.body = getLoopBody(block);
                loop.exit = getLoopExit(block);
                loop.tripCount = estimateTripCount(block);
                loop.canUnroll = canUnrollLoop(loop);
                
                loops.push_back(loop);
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void unrollLoops() {
        for (const LoopInfo& loop : loops) {
            if (loop.canUnroll) {
                unrollLoop(loop);
            }
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    void unrollLoop(const LoopInfo& loop) {
        int unrollFactor = getUnrollFactor(loop.tripCount);
        
        for (int i = 0; i < unrollFactor; i++) {
            BasicBlock* unrolledBlock = createUnrolledBlock(loop.body, i);
            // 插入展开的块
            insertUnrolledBlock(unrolledBlock, loop);
        }
    }
    
    // 时间复杂度：O(1)
    // 空间复杂度：O(1)
    int getUnrollFactor(int tripCount) {
        if (tripCount <= 4) {
            return tripCount;
        } else if (tripCount <= 8) {
            return 4;
        } else {
            return 8;
        }
    }
    
    // 时间复杂度：O(n)
    // 空间复杂度：O(n)
    BasicBlock* createUnrolledBlock(BasicBlock* original, int iteration) {
        BasicBlock* unrolled = new BasicBlock();
        
        for (Instruction* inst : original->getInstructions()) {
            Instruction* newInst = createUnrolledInstruction(inst, iteration);
            unrolled->addInstruction(newInst);
        }
        
        return unrolled;
    }
};
```
