# 滑动窗口

## 概述

滑动窗口（Sliding Window）是算法优化中的重要技巧，通过维护一个动态窗口来解决问题。滑动窗口技巧可以高效地处理子数组、子字符串等问题，将时间复杂度从O(n²)降低到O(n)。

## 核心概念

### 基本定义
- **滑动窗口**：维护一个动态窗口
- **固定窗口**：窗口大小固定
- **可变窗口**：窗口大小可变
- **窗口边界**：窗口的左右边界

### 关键特性
- **高效性**：时间复杂度通常为O(n)
- **空间效率**：空间复杂度通常为O(1)
- **动态性**：窗口可以动态调整
- **连续性**：窗口内的元素连续

### 基本滑动窗口算法

from typing import List, Tuple, Any, Callable, Optional
import time

class SlidingWindowTechniques:
    """滑动窗口技巧算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.data = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, data: List[int], target: int = None) -> Any:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        if target is not None:
            return algorithm(data, target)
        return algorithm(data)
    
    def set_data(self, data: List[int]) -> None:
        """设置数据"""
        self.data = data

# 最大子数组和（滑动窗口）
def max_subarray_sum(nums: List[int]) -> int:
    """最大子数组和（滑动窗口）"""
    if not nums:
        return 0
    
    max_sum = current_sum = nums[0]
    
    for i in range(1, len(nums)):
        current_sum = max(nums[i], current_sum + nums[i])
        max_sum = max(max_sum, current_sum)
    
    return max_sum

# 长度最小的子数组（滑动窗口）
def min_subarray_length(nums: List[int], target: int) -> int:
    """长度最小的子数组（滑动窗口）"""
    if not nums:
        return 0
    
    left = 0
    min_length = float('inf')
    current_sum = 0
    
    for right in range(len(nums)):
        current_sum += nums[right]
        
        while current_sum >= target:
            min_length = min(min_length, right - left + 1)
            current_sum -= nums[left]
            left += 1
    
    return min_length if min_length != float('inf') else 0

# 最长无重复字符子串（滑动窗口）
def longest_substring_without_repeating(s: str) -> int:
    """最长无重复字符子串（滑动窗口）"""
    if not s:
        return 0
    
    char_set = set()
    left = 0
    max_length = 0
    
    for right in range(len(s)):
        while s[right] in char_set:
            char_set.remove(s[left])
            left += 1
        
        char_set.add(s[right])
        max_length = max(max_length, right - left + 1)
    
    return max_length

# 最多K个不同字符的子串（滑动窗口）
def longest_substring_with_k_distinct(s: str, k: int) -> int:
    """最多K个不同字符的子串（滑动窗口）"""
    if not s or k <= 0:
        return 0
    
    char_count = {}
    left = 0
    max_length = 0
    
    for right in range(len(s)):
        char_count[s[right]] = char_count.get(s[right], 0) + 1
        
        while len(char_count) > k:
            char_count[s[left]] -= 1
            if char_count[s[left]] == 0:
                del char_count[s[left]]
            left += 1
        
        max_length = max(max_length, right - left + 1)
    
    return max_length

# 固定长度的滑动窗口最大值
def max_sliding_window(nums: List[int], k: int) -> List[int]:
    """固定长度的滑动窗口最大值"""
    if not nums or k <= 0:
        return []
    
    result = []
    window = []
    
    for i in range(len(nums)):
        # 移除窗口左边超出范围的元素
        while window and window[0] <= i - k:
            window.pop(0)
        
        # 移除窗口右边小于当前元素的元素
        while window and nums[window[-1]] < nums[i]:
            window.pop()
        
        window.append(i)
        
        # 当窗口大小达到k时，记录最大值
        if i >= k - 1:
            result.append(nums[window[0]])
    
    return result

# 使用示例
if __name__ == "__main__":

    techniques.add_algorithm("longest_substring", longest_substring_without_repeating)
    techniques.add_algorithm("k_distinct", longest_substring_with_k_distinct)
    techniques.add_algorithm("max_sliding_window", max_sliding_window)
    
    # 测试最大子数组和
    nums = [-2, 1, -3, 4, -1, 2, 1, -5, 4]
    result = techniques.execute_algorithm("max_subarray", nums)
    print(f"最大子数组和结果: {result}")
    
    # 测试长度最小的子数组
    nums = [2, 3, 1, 2, 4, 3]
    target = 7
    result = techniques.execute_algorithm("min_subarray_length", nums, target)
    print(f"长度最小的子数组结果: {result}")
    
    # 测试最长无重复字符子串
    s = "abcabcbb"
    result = techniques.execute_algorithm("longest_substring", s)

# 最小覆盖子串（滑动窗口）
def min_window_substring(s: str, t: str) -> str:
    """最小覆盖子串（滑动窗口）"""
    if not s or not t:
        return ""
    
    # 统计t中每个字符的出现次数
    t_count = {}
    for char in t:
        t_count[char] = t_count.get(char, 0) + 1
    
    # 滑动窗口
    left = 0
    min_len = float('inf')
    min_start = 0
    matched = 0
    
    for right in range(len(s)):
        if s[right] in t_count:
            t_count[s[right]] -= 1
            if t_count[s[right]] >= 0:
                matched += 1
        
        # 收缩窗口
        while matched == len(t):
            if right - left + 1 < min_len:
                min_len = right - left + 1
                min_start = left
            
            if s[left] in t_count:
                t_count[s[left]] += 1
                if t_count[s[left]] > 0:
                    matched -= 1
            
            left += 1
    
    return s[min_start:min_start + min_len] if min_len != float('inf') else ""

# 字符串的排列（滑动窗口）
def check_inclusion(s1: str, s2: str) -> bool:
    """字符串的排列（滑动窗口）"""
    if len(s1) > len(s2):
        return False
    
    # 统计s1中每个字符的出现次数
    s1_count = {}
    for char in s1:
        s1_count[char] = s1_count.get(char, 0) + 1
    
    # 滑动窗口
    window_count = {}
    left = 0
    
    for right in range(len(s2)):
        window_count[s2[right]] = window_count.get(s2[right], 0) + 1
        
        # 当窗口大小达到s1的长度时
        if right - left + 1 == len(s1):
            if window_count == s1_count:
                return True
            
            # 移动窗口
            window_count[s2[left]] -= 1
            if window_count[s2[left]] == 0:
                del window_count[s2[left]]
            left += 1
    
    return False

# 无重复字符的最长子串（滑动窗口）
def length_of_longest_substring(s: str) -> int:
    """无重复字符的最长子串（滑动窗口）"""
    if not s:
        return 0
    
    char_index = {}
    left = 0
    max_length = 0
    
    for right in range(len(s)):
        if s[right] in char_index and char_index[s[right]] >= left:
            left = char_index[s[right]] + 1
        
        char_index[s[right]] = right
        max_length = max(max_length, right - left + 1)
    
    return max_length

# 水果成篮（滑动窗口）
def total_fruit(fruits: List[int]) -> int:
    """水果成篮（滑动窗口）"""
    if not fruits:
        return 0
    
    fruit_count = {}
    left = 0
    max_fruits = 0
    
    for right in range(len(fruits)):
        fruit_count[fruits[right]] = fruit_count.get(fruits[right], 0) + 1
        
        # 当水果种类超过2种时，收缩窗口
        while len(fruit_count) > 2:
            fruit_count[fruits[left]] -= 1
            if fruit_count[fruits[left]] == 0:
                del fruit_count[fruits[left]]
            left += 1
        
        max_fruits = max(max_fruits, right - left + 1)
    
    return max_fruits
