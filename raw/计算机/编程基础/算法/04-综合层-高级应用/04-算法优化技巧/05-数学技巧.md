# 数学技巧

## 概述

数学技巧（Mathematical Techniques）是算法优化中的重要方法，通过运用数学知识和技巧来解决问题。数学技巧可以显著提高算法效率，简化问题复杂度，是算法设计和优化的重要工具。

## 核心概念

### 基本定义
- **数学技巧**：运用数学知识解决问题的方法
- **数论技巧**：运用数论知识解决问题
- **组合数学技巧**：运用组合数学知识解决问题
- **概率统计技巧**：运用概率统计知识解决问题

### 关键特性
- **高效性**：数学技巧通常效率很高
- **简洁性**：代码实现简洁
- **通用性**：适用于多种问题类型
- **可扩展性**：可以扩展到复杂问题

from typing import List, Tuple, Any, Callable, Optional
import time
import math

class MathematicalTechniques:
    """数学技巧算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.data = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, data: List[int], *args) -> Any:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(data, *args)
    
    def set_data(self, data: List[int]) -> None:
        """设置数据"""
        self.data = data

# 快速幂
def fast_power(base: int, exponent: int, modulus: int = None) -> int:
    """快速幂"""
    result = 1
    base = base % modulus if modulus else base
    
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % modulus if modulus else result * base
        
        base = (base * base) % modulus if modulus else base * base
        exponent //= 2
    
    return result

# 最大公约数
def gcd(a: int, b: int) -> int:
    """最大公约数"""
    while b:
        a, b = b, a % b
    return a

# 最小公倍数
def lcm(a: int, b: int) -> int:
    """最小公倍数"""
    return abs(a * b) // gcd(a, b)

# 模逆元
def modular_inverse(a: int, m: int) -> int:
    """模逆元"""
    def extended_gcd(a: int, b: int) -> Tuple[int, int, int]:
        if b == 0:
            return a, 1, 0
        
        gcd, x1, y1 = extended_gcd(b, a % b)
        x = y1
        y = x1 - (a // b) * y1
        
        return gcd, x, y
    
    gcd, x, y = extended_gcd(a, m)
    if gcd != 1:
        raise ValueError("模逆元不存在")
    
    return (x % m + m) % m

# 组合数
def combination(n: int, k: int) -> int:
    """组合数"""
    if k > n or k < 0:
        return 0
    
    if k > n - k:
        k = n - k
    
    result = 1
    for i in range(k):
        result = result * (n - i) // (i + 1)
    
    return result

# 排列数
def permutation(n: int, k: int) -> int:
    """排列数"""
    if k > n or k < 0:
        return 0
    
    result = 1
    for i in range(k):
        result *= (n - i)
    
    return result

# 阶乘
def factorial(n: int) -> int:
    """阶乘"""
    if n < 0:
        return 0
    if n <= 1:
        return 1
    
    result = 1
    for i in range(2, n + 1):
        result *= i
    
    return result

# 素数判断
def is_prime(n: int) -> bool:
    """素数判断"""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    
    for i in range(3, int(math.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    
    return True

# 使用示例
if __name__ == "__main__":

# 中国剩余定理
def chinese_remainder_theorem(remainders: List[int], moduli: List[int]) -> int:
    """中国剩余定理"""
    if len(remainders) != len(moduli):
        raise ValueError("余数和模数数量不匹配")
    
    # 计算所有模数的乘积
    M = 1
    for m in moduli:
        M *= m
    
    result = 0
    for i in range(len(remainders)):
        Mi = M // moduli[i]
        Mi_inv = modular_inverse(Mi, moduli[i])
        result += remainders[i] * Mi * Mi_inv
    
    return result % M

# 欧拉函数
def euler_phi(n: int) -> int:
    """欧拉函数"""
    if n <= 0:
        return 0
    
    result = n
    p = 2
    
    while p * p <= n:
        if n % p == 0:
            while n % p == 0:
                n //= p
            result -= result // p
        p += 1
    
    if n > 1:
        result -= result // n
    
    return result

# 费马小定理
def fermat_little_theorem(a: int, p: int) -> bool:
    """费马小定理验证"""
    if p <= 1:
        return False
    
    # 检查p是否为素数
    for i in range(2, int(math.sqrt(p)) + 1):
        if p % i == 0:
            return False
    
    # 费马小定理：a^(p-1) ≡ 1 (mod p)
    return fast_power(a, p - 1, p) == 1

# 欧拉定理
def euler_theorem(a: int, n: int) -> bool:
    """欧拉定理验证"""
    if gcd(a, n) != 1:
        return False
    
    # 计算欧拉函数
    phi_n = euler_phi(n)
    
    # 欧拉定理：a^φ(n) ≡ 1 (mod n)
    return fast_power(a, phi_n, n) == 1

# 矩阵快速幂
def matrix_fast_power(matrix: List[List[int]], exponent: int, modulus: int = None) -> List[List[int]]:
    """矩阵快速幂"""
    n = len(matrix)
    
    if exponent == 0:
        # 返回单位矩阵
        identity = [[0] * n for _ in range(n)]
        for i in range(n):
            identity[i][i] = 1
        return identity
    
    if exponent == 1:
        return matrix
    
    # 递归计算
    half = matrix_fast_power(matrix, exponent // 2, modulus)
    
    # 矩阵乘法
    def matrix_multiply(a: List[List[int]], b: List[List[int]]) -> List[List[int]]:
        result = [[0] * n for _ in range(n)]
        for i in range(n):
            for j in range(n):
                for k in range(n):
                    result[i][j] += a[i][k] * b[k][j]
                    if modulus:
                        result[i][j] %= modulus
        return result
    
    if exponent % 2 == 0:
        return matrix_multiply(half, half)
    else:
        return matrix_multiply(matrix_multiply(half, half), matrix)

# 斐波那契数列（矩阵快速幂）
def fibonacci_matrix(n: int) -> int:
    """使用矩阵快速幂计算斐波那契数列"""
    if n <= 1:
        return n
    
    # 斐波那契矩阵
    matrix = [[1, 1], [1, 0]]
    result_matrix = matrix_fast_power(matrix, n - 1)
    
    return result_matrix[0][0]

# 卡特兰数
def catalan_number(n: int) -> int:
    """卡特兰数"""
    if n <= 1:
        return 1
    
    # 使用递推公式：C(n) = (2n)! / ((n+1)! * n!)
    numerator = factorial(2 * n)
    denominator = factorial(n + 1) * factorial(n)
    
    return numerator // denominator

# 斯特林数
def stirling_number_second(n: int, k: int) -> int:
    """第二类斯特林数"""
    if k > n or k < 0:
        return 0
    if k == 0 or n == 0:
        return 1 if k == 0 and n == 0 else 0
    
    # 使用递推公式：S(n,k) = k*S(n-1,k) + S(n-1,k-1)
    dp = [[0] * (k + 1) for _ in range(n + 1)]
    
    for i in range(n + 1):
        for j in range(min(i, k) + 1):
            if j == 0 or j == i:
                dp[i][j] = 1
            else:
                dp[i][j] = j * dp[i - 1][j] + dp[i - 1][j - 1]
    
    return dp[n][k]
