# 前缀和与差分

## 概述

前缀和与差分（Prefix Sum and Difference）是算法优化中的重要技巧，用于高效地处理区间查询和区间更新问题。前缀和与差分技巧可以将区间操作的时间复杂度从O(n)降低到O(1)。

## 核心概念

### 基本定义
- **前缀和**：数组前i个元素的和
- **差分数组**：相邻元素的差值
- **区间查询**：查询区间内的和
- **区间更新**：更新区间内的值

### 关键特性
- **高效性**：区间操作时间复杂度O(1)
- **预处理**：需要O(n)时间预处理
- **空间效率**：空间复杂度O(n)
- **可逆性**：前缀和和差分互为逆运算

### 基本前缀和与差分算法

```python
from typing import List, Tuple, Any, Callable, Optional
import time

class PrefixSumAndDifferenceTechniques:
    """前缀和与差分技巧算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.data = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, data: List[int], *args) -> Any:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(data, *args)
    
    def set_data(self, data: List[int]) -> None:
        """设置数据"""
        self.data = data

# 一维前缀和
def one_dimensional_prefix_sum(nums: List[int]) -> List[int]:
    """一维前缀和"""
    if not nums:
        return []
    
    prefix_sum = [0] * (len(nums) + 1)
    for i in range(len(nums)):
        prefix_sum[i + 1] = prefix_sum[i] + nums[i]
    
    return prefix_sum

# 一维区间查询
def range_sum_query(prefix_sum: List[int], left: int, right: int) -> int:
    """一维区间查询"""
    if left < 0 or right >= len(prefix_sum) - 1 or left > right:
        return 0
    
    return prefix_sum[right + 1] - prefix_sum[left]

# 一维差分数组
def one_dimensional_difference(nums: List[int]) -> List[int]:
    """一维差分数组"""
    if not nums:
        return []
    
    diff = [0] * len(nums)
    diff[0] = nums[0]
    
    for i in range(1, len(nums)):
        diff[i] = nums[i] - nums[i - 1]
    
    return diff

# 一维区间更新
def range_update(diff: List[int], left: int, right: int, value: int) -> None:
    """一维区间更新"""
    if left < 0 or right >= len(diff) or left > right:
        return
    
    diff[left] += value
    if right + 1 < len(diff):
        diff[right + 1] -= value

# 从差分数组恢复原数组
def recover_from_difference(diff: List[int]) -> List[int]:
    """从差分数组恢复原数组"""
    if not diff:
        return []
    
    nums = [0] * len(diff)
    nums[0] = diff[0]
    
    for i in range(1, len(diff)):
        nums[i] = nums[i - 1] + diff[i]
    
    return nums

# 二维前缀和
def two_dimensional_prefix_sum(matrix: List[List[int]]) -> List[List[int]]:
    """二维前缀和"""
    if not matrix or not matrix[0]:
        return []
    
    rows, cols = len(matrix), len(matrix[0])
    prefix_sum = [[0] * (cols + 1) for _ in range(rows + 1)]
    
    for i in range(rows):
        for j in range(cols):
            prefix_sum[i + 1][j + 1] = (prefix_sum[i][j + 1] + 
                                       prefix_sum[i + 1][j] - 
                                       prefix_sum[i][j] + 
                                       matrix[i][j])
    
    return prefix_sum

# 二维区间查询
def two_dimensional_range_query(prefix_sum: List[List[int]], 
                               row1: int, col1: int, 
                               row2: int, col2: int) -> int:
    """二维区间查询"""
    if (row1 < 0 or row2 >= len(prefix_sum) - 1 or 
        col1 < 0 or col2 >= len(prefix_sum[0]) - 1 or 
        row1 > row2 or col1 > col2):
        return 0
    
    return (prefix_sum[row2 + 1][col2 + 1] - 
            prefix_sum[row1][col2 + 1] - 
            prefix_sum[row2 + 1][col1] + 
            prefix_sum[row1][col1])

# 使用示例
if __name__ == "__main__":
    techniques = PrefixSumAndDifferenceTechniques()

    techniques.add_algorithm("two_d_prefix_sum", two_dimensional_prefix_sum)

    # 测试一维前缀和

    # 测试一维区间查询

    print(f"一维区间查询结果: {result}")
    
    # 测试一维差分数组
    diff = techniques.execute_algorithm("one_d_difference", nums)
    print(f"一维差分数组结果: {diff}")
    
    # 测试一维区间更新
    techniques.execute_algorithm("range_update", diff, 1, 3, 2)
    print(f"一维区间更新后的差分数组: {diff}")
    
    # 测试从差分数组恢复原数组
    recovered = techniques.execute_algorithm("recover_from_diff", diff)
    print(f"从差分数组恢复的原数组: {recovered}")
    
    # 测试二维前缀和
```

```python
from typing import List, Tuple, Any, Callable, Optional
import time

class AdvancedPrefixSumAndDifferenceTechniques:
    """高级前缀和与差分技巧算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.data = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""

    def execute_algorithm(self, name: str, data: List[int], *args) -> Any:
        """执行算法"""

            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(data, *args)
    
    def set_data(self, data: List[int]) -> None:
        """设置数据"""
        self.data = data

# 二维差分数组
def two_dimensional_difference(matrix: List[List[int]]) -> List[List[int]]:
    """二维差分数组"""
    if not matrix or not matrix[0]:
        return []
    
    rows, cols = len(matrix), len(matrix[0])
    diff = [[0] * cols for _ in range(rows)]
    
    # 计算差分数组
    for i in range(rows):
        for j in range(cols):
            diff[i][j] = matrix[i][j]
            if i > 0:
                diff[i][j] -= matrix[i - 1][j]
            if j > 0:
                diff[i][j] -= matrix[i][j - 1]
            if i > 0 and j > 0:
                diff[i][j] += matrix[i - 1][j - 1]
    
    return diff

# 二维区间更新
def two_dimensional_range_update(diff: List[List[int]], 
                                row1: int, col1: int, 
                                row2: int, col2: int, 
                                value: int) -> None:
    """二维区间更新"""
    if (row1 < 0 or row2 >= len(diff) or 
        col1 < 0 or col2 >= len(diff[0]) or 
        row1 > row2 or col1 > col2):
        return
    
    diff[row1][col1] += value
    if row2 + 1 < len(diff):
        diff[row2 + 1][col1] -= value
    if col2 + 1 < len(diff[0]):
        diff[row1][col2 + 1] -= value
    if row2 + 1 < len(diff) and col2 + 1 < len(diff[0]):
        diff[row2 + 1][col2 + 1] += value

# 从二维差分数组恢复原数组
def recover_from_two_d_difference(diff: List[List[int]]) -> List[List[int]]:
    """从二维差分数组恢复原数组"""
    if not diff or not diff[0]:
        return []
    
    rows, cols = len(diff), len(diff[0])
    matrix = [[0] * cols for _ in range(rows)]
    
    # 恢复原数组
    for i in range(rows):
        for j in range(cols):
            matrix[i][j] = diff[i][j]
            if i > 0:
                matrix[i][j] += matrix[i - 1][j]
            if j > 0:
                matrix[i][j] += matrix[i][j - 1]
            if i > 0 and j > 0:
                matrix[i][j] -= matrix[i - 1][j - 1]
    
    return matrix

# 三维前缀和
def three_dimensional_prefix_sum(cube: List[List[List[int]]]) -> List[List[List[int]]]:
    """三维前缀和"""
    if not cube or not cube[0] or not cube[0][0]:
        return []
    
    depth, rows, cols = len(cube), len(cube[0]), len(cube[0][0])
    prefix_sum = [[[0] * (cols + 1) for _ in range(rows + 1)] for _ in range(depth + 1)]
    
    for d in range(depth):
        for i in range(rows):
            for j in range(cols):
                prefix_sum[d + 1][i + 1][j + 1] = (
                    prefix_sum[d][i + 1][j + 1] + 
                    prefix_sum[d + 1][i][j + 1] + 
                    prefix_sum[d + 1][i + 1][j] - 
                    prefix_sum[d][i][j + 1] - 
                    prefix_sum[d][i + 1][j] - 
                    prefix_sum[d + 1][i][j] + 
                    prefix_sum[d][i][j] + 
                    cube[d][i][j]
                )
    
    return prefix_sum

# 三维区间查询
def three_dimensional_range_query(prefix_sum: List[List[List[int]]], 
                                 d1: int, r1: int, c1: int, 
                                 d2: int, r2: int, c2: int) -> int:
    """三维区间查询"""
    if (d1 < 0 or d2 >= len(prefix_sum) - 1 or 
        r1 < 0 or r2 >= len(prefix_sum[0]) - 1 or 
        c1 < 0 or c2 >= len(prefix_sum[0][0]) - 1 or 
        d1 > d2 or r1 > r2 or c1 > c2):
        return 0
    
    return (prefix_sum[d2 + 1][r2 + 1][c2 + 1] - 
            prefix_sum[d1][r2 + 1][c2 + 1] - 
            prefix_sum[d2 + 1][r1][c2 + 1] - 
            prefix_sum[d2 + 1][r2 + 1][c1] + 
            prefix_sum[d1][r1][c2 + 1] + 
            prefix_sum[d1][r2 + 1][c1] + 
            prefix_sum[d2 + 1][r1][c1] - 
            prefix_sum[d1][r1][c1])

# 使用示例
if __name__ == "__main__":

    # 添加算法

    advanced_techniques.add_algorithm("two_d_range_update", two_dimensional_range_update)
    advanced_techniques.add_algorithm("recover_from_two_d_diff", recover_from_two_d_difference)
    advanced_techniques.add_algorithm("three_d_prefix_sum", three_dimensional_prefix_sum)
    advanced_techniques.add_algorithm("three_d_range_query", three_dimensional_range_query)

    print(f"二维差分数组结果: {diff_2d}")

    print(f"二维区间更新后的差分数组: {diff_2d}")
    
    # 测试从二维差分数组恢复原数组
    recovered_2d = advanced_techniques.execute_algorithm("recover_from_two_d_diff", diff_2d)
    print(f"从二维差分数组恢复的原数组: {recovered_2d}")
    
    # 测试三维前缀和
    cube = [[[1, 2], [3, 4]], [[5, 6], [7, 8]]]
    prefix_sum_3d = advanced_techniques.execute_algorithm("three_d_prefix_sum", cube)
    print(f"三维前缀和结果: {prefix_sum_3d}")
    
    # 测试三维区间查询
    d1, r1, c1, d2, r2, c2 = 0, 0, 0, 1, 1, 1
    result = advanced_techniques.execute_algorithm("three_d_range_query", prefix_sum_3d, d1, r1, c1, d2, r2, c2)
```
