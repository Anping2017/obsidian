# 双指针技巧

## 概述

双指针技巧（Two Pointers Technique）是算法优化中的重要方法，通过使用两个指针在数组或链表中移动来解决问题。双指针技巧可以显著提高算法效率，将时间复杂度从O(n²)降低到O(n)。

## 核心概念

### 基本定义
- **双指针**：使用两个指针遍历数据结构
- **对撞指针**：两个指针从两端向中间移动
- **快慢指针**：两个指针以不同速度移动
- **滑动窗口**：维护一个动态窗口

### 关键特性
- **高效性**：时间复杂度通常为O(n)
- **空间效率**：空间复杂度通常为O(1)
- **简洁性**：代码实现简洁
- **通用性**：适用于多种问题类型

### 基本双指针算法

from typing import List, Tuple, Any, Callable, Optional
import time

class TwoPointersTechniques:
    """双指针技巧算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.data = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, data: List[int], target: int = None) -> Any:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        if target is not None:
            return algorithm(data, target)
        return algorithm(data)
    
    def set_data(self, data: List[int]) -> None:
        """设置数据"""
        self.data = data

# 两数之和（对撞指针）
def two_sum_two_pointers(nums: List[int], target: int) -> List[int]:
    """两数之和（对撞指针）"""
    # 需要先排序
    sorted_nums = sorted(enumerate(nums), key=lambda x: x[1])
    left, right = 0, len(sorted_nums) - 1
    
    while left < right:
        current_sum = sorted_nums[left][1] + sorted_nums[right][1]
        if current_sum == target:
            return [sorted_nums[left][0], sorted_nums[right][0]]
        elif current_sum < target:
            left += 1
        else:
            right -= 1
    
    return []

# 三数之和（对撞指针）
def three_sum_two_pointers(nums: List[int], target: int) -> List[List[int]]:
    """三数之和（对撞指针）"""
    nums.sort()
    result = []
    
    for i in range(len(nums) - 2):
        if i > 0 and nums[i] == nums[i - 1]:
            continue
        
        left, right = i + 1, len(nums) - 1
        while left < right:
            current_sum = nums[i] + nums[left] + nums[right]
            if current_sum == target:
                result.append([nums[i], nums[left], nums[right]])
                while left < right and nums[left] == nums[left + 1]:
                    left += 1
                while left < right and nums[right] == nums[right - 1]:
                    right -= 1
                left += 1
                right -= 1
            elif current_sum < target:
                left += 1
            else:
                right -= 1
    
    return result

# 盛最多水的容器（对撞指针）
def container_with_most_water(height: List[int]) -> int:
    """盛最多水的容器（对撞指针）"""
    left, right = 0, len(height) - 1
    max_area = 0
    
    while left < right:
        area = min(height[left], height[right]) * (right - left)
        max_area = max(max_area, area)
        
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_area

# 移除重复元素（快慢指针）
def remove_duplicates(nums: List[int]) -> int:
    """移除重复元素（快慢指针）"""
    if not nums:
        return 0
    
    slow = 0
    for fast in range(1, len(nums)):
        if nums[fast] != nums[slow]:
            slow += 1
            nums[slow] = nums[fast]
    
    return slow + 1

# 移动零（快慢指针）
def move_zeros(nums: List[int]) -> None:
    """移动零（快慢指针）"""
    slow = 0
    for fast in range(len(nums)):
        if nums[fast] != 0:
            nums[slow], nums[fast] = nums[fast], nums[slow]
            slow += 1

# 回文串判断（对撞指针）
def is_palindrome(s: str) -> bool:
    """回文串判断（对撞指针）"""
    left, right = 0, len(s) - 1
    
    while left < right:
        if not s[left].isalnum():
            left += 1
            continue
        if not s[right].isalnum():
            right -= 1
            continue
        
        if s[left].lower() != s[right].lower():
            return False
        
        left += 1
        right -= 1
    
    return True

# 使用示例
if __name__ == "__main__":

    techniques.add_algorithm("move_zeros", move_zeros)

    # 测试两数之和

    print(f"两数之和结果: {result}")
    
    # 测试三数之和
    nums = [-1, 0, 1, 2, -1, -4]
    target = 0
    result = techniques.execute_algorithm("three_sum", nums, target)
    print(f"三数之和结果: {result}")
    
    # 测试盛最多水的容器
    height = [1, 8, 6, 2, 5, 4, 8, 3, 7]
    result = techniques.execute_algorithm("container", height)
    print(f"盛最多水的容器结果: {result}")
    
    # 测试移除重复元素

# 最长无重复子串（滑动窗口）
def longest_substring_without_repeating(s: str) -> int:
    """最长无重复子串（滑动窗口）"""
    char_set = set()
    left = 0
    max_length = 0
    
    for right in range(len(s)):
        while s[right] in char_set:
            char_set.remove(s[left])
            left += 1
        
        char_set.add(s[right])
        max_length = max(max_length, right - left + 1)
    
    return max_length

# 最小覆盖子串（滑动窗口）
def min_window_substring(s: str, t: str) -> str:
    """最小覆盖子串（滑动窗口）"""
    if not s or not t:
        return ""
    
    # 统计t中每个字符的出现次数
    t_count = {}
    for char in t:
        t_count[char] = t_count.get(char, 0) + 1
    
    # 滑动窗口
    left = 0
    min_len = float('inf')
    min_start = 0
    matched = 0
    
    for right in range(len(s)):
        if s[right] in t_count:
            t_count[s[right]] -= 1
            if t_count[s[right]] >= 0:
                matched += 1
        
        # 收缩窗口
        while matched == len(t):
            if right - left + 1 < min_len:
                min_len = right - left + 1
                min_start = left
            
            if s[left] in t_count:
                t_count[s[left]] += 1
                if t_count[s[left]] > 0:
                    matched -= 1
            
            left += 1
    
    return s[min_start:min_start + min_len] if min_len != float('inf') else ""

# 链表中环的检测（快慢指针）
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def detect_cycle(head: ListNode) -> ListNode:
    """链表中环的检测（快慢指针）"""
    if not head or not head.next:
        return None
    
    slow = fast = head
    
    # 检测环
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            break
    else:
        return None
    
    # 找到环的起点
    slow = head
    while slow != fast:
        slow = slow.next
        fast = fast.next
    
    return slow

# 链表中倒数第k个节点（快慢指针）
def find_kth_from_end(head: ListNode, k: int) -> ListNode:
    """链表中倒数第k个节点（快慢指针）"""
    if not head or k <= 0:
        return None
    
    fast = slow = head
    
    # 快指针先走k步
    for _ in range(k):
        if not fast:
            return None
        fast = fast.next
    
    # 快慢指针同时移动
    while fast:
        slow = slow.next
        fast = fast.next
    
    return slow

# 使用示例
if __name__ == "__main__":

    advanced_techniques.add_algorithm("detect_cycle", detect_cycle)
    advanced_techniques.add_algorithm("find_kth", find_kth_from_end)
    
    # 测试最长无重复子串

    print(f"最长无重复子串结果: {result}")
    
    # 测试最小覆盖子串
    s = "ADOBECODEBANC"
    t = "ABC"
    result = advanced_techniques.execute_algorithm("min_window", s, t)
    print(f"最小覆盖子串结果: {result}")
    
    # 测试链表中环的检测
    # 创建带环的链表
    head = ListNode(1)
    head.next = ListNode(2)
    head.next.next = ListNode(3)
    head.next.next.next = ListNode(4)
    head.next.next.next.next = head.next  # 创建环
    
    result = advanced_techniques.execute_algorithm("detect_cycle", head)
    print(f"链表中环的检测结果: {result.val if result else None}")
    
    # 测试链表中倒数第k个节点

    head.next = ListNode(2)
    head.next.next = ListNode(3)
    head.next.next.next = ListNode(4)
