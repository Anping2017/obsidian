# 多线程算法

## 概述

多线程算法（Multithreaded Algorithms）是利用多个线程并行执行来提高算法性能的算法设计方法。通过将任务分解为多个独立的子任务，并在不同的线程中并行执行，可以显著提高算法的执行效率。

## 核心概念

### 基本思想
- **任务分解**：将大任务分解为多个小任务
- **并行执行**：多个线程同时执行不同任务
- **结果合并**：将各线程的结果合并为最终结果
- **同步控制**：确保线程间的正确协作

### 关键特性
- **并发性**：多个线程同时执行
- **独立性**：线程间相对独立
- **同步性**：需要同步机制
- **可扩展性**：可随CPU核心数扩展

```python
import threading
import time
import random
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import List, Any, Callable, Optional
import queue

class ThreadManager:
    """线程管理器"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.threads = []
        self.results = []
        self.lock = threading.Lock()
    
    def create_thread(self, target: Callable, args: tuple = (), name: str = None) -> threading.Thread:
        """创建线程"""
        thread = threading.Thread(target=target, args=args, name=name)
        self.threads.append(thread)
        return thread
    
    def start_all_threads(self) -> None:
        """启动所有线程"""
        for thread in self.threads:
            thread.start()
    
    def join_all_threads(self) -> None:
        """等待所有线程完成"""
        for thread in self.threads:
            thread.join()
    
    def add_result(self, result: Any) -> None:
        """添加结果（线程安全）"""
        with self.lock:
            self.results.append(result)
    
    def get_results(self) -> List[Any]:
        """获取所有结果"""
        return self.results.copy()
    
    def clear(self) -> None:
        """清理线程和结果"""
        self.threads.clear()
        self.results.clear()
```

### 线程同步机制

```python
import threading
import time
from typing import List, Any
import random

class SynchronizedCounter:
    """同步计数器"""
    
    def __init__(self):
        self._value = 0
        self._lock = threading.Lock()
    
    def increment(self) -> int:
        """增加计数（线程安全）"""
        with self._lock:
            self._value += 1
            return self._value
    
    def get_value(self) -> int:
        """获取当前值（线程安全）"""
        with self._lock:
            return self._value

class ProducerConsumer:
    """生产者消费者模式"""
    
    def __init__(self, max_size: int = 10):
        self.queue = queue.Queue(maxsize=max_size)
        self.producer_lock = threading.Lock()
        self.consumer_lock = threading.Lock()
        self.produced_count = 0
        self.consumed_count = 0
    
    def producer(self, producer_id: int, items: List[Any]) -> None:
        """生产者"""
        for item in items:
            self.queue.put(item)
            with self.producer_lock:
                self.produced_count += 1
            print(f"生产者 {producer_id} 生产了 {item}")
            time.sleep(random.uniform(0.1, 0.3))
    
    def consumer(self, consumer_id: int) -> None:
        """消费者"""
        while True:
            try:
                item = self.queue.get(timeout=1)
                with self.consumer_lock:
                    self.consumed_count += 1
                print(f"消费者 {consumer_id} 消费了 {item}")
                self.queue.task_done()
                time.sleep(random.uniform(0.1, 0.3))
            except queue.Empty:
                break
    
    def get_stats(self) -> dict:
        """获取统计信息"""
        return {
            'produced': self.produced_count,
            'consumed': self.consumed_count,
            'queue_size': self.queue.qsize()
        }

# 使用示例
if __name__ == "__main__":
    # 同步计数器示例
    counter = SynchronizedCounter()
    
    def increment_worker(worker_id: int, count: int):
        for _ in range(count):
            value = counter.increment()
            print(f"线程 {worker_id} 计数: {value}")
            time.sleep(0.01)
    
    # 创建多个线程增加计数
    threads = []
    for i in range(3):
        thread = threading.Thread(target=increment_worker, args=(i, 5))
        threads.append(thread)
        thread.start()
    
    # 等待所有线程完成
    for thread in threads:
        thread.join()
    
    print(f"最终计数: {counter.get_value()}")
    
    # 生产者消费者示例
    pc = ProducerConsumer(max_size=5)
    
    # 创建生产者线程
    producer_threads = []
    for i in range(2):
        items = [f"item-{i}-{j}" for j in range(5)]
        thread = threading.Thread(target=pc.producer, args=(i, items))
        producer_threads.append(thread)
        thread.start()
    
    # 创建消费者线程
    consumer_threads = []
    for i in range(3):
        thread = threading.Thread(target=pc.consumer, args=(i,))
        consumer_threads.append(thread)
        thread.start()
    
    # 等待生产者完成
    for thread in producer_threads:
        thread.join()
    
    # 等待队列清空
    pc.queue.join()
    
    # 停止消费者
    for thread in consumer_threads:
        thread.join()
    
    print(f"统计信息: {pc.get_stats()}")
```

### 1. 并行归并排序

```python
import threading
import time
from typing import List, Any
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelMergeSort:
    """并行归并排序"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def merge_sort(self, data: List[Any]) -> List[Any]:
        """并行归并排序"""
        if len(data) <= 1:
            return data
        
        # 如果数据量小，使用串行排序
        if len(data) < 100:
            return self._serial_merge_sort(data)
        
        # 分解任务
        mid = len(data) // 2
        left_data = data[:mid]
        right_data = data[mid:]
        
        # 并行排序左右两部分
        future_left = self.executor.submit(self.merge_sort, left_data)
        future_right = self.executor.submit(self.merge_sort, right_data)
        
        # 获取结果
        left_sorted = future_left.result()
        right_sorted = future_right.result()
        
        # 合并结果
        return self._merge(left_sorted, right_sorted)
    
    def _serial_merge_sort(self, data: List[Any]) -> List[Any]:
        """串行归并排序"""
        if len(data) <= 1:
            return data
        
        mid = len(data) // 2
        left = self._serial_merge_sort(data[:mid])
        right = self._serial_merge_sort(data[mid:])
        
        return self._merge(left, right)
    
    def _merge(self, left: List[Any], right: List[Any]) -> List[Any]:
        """合并两个有序数组"""
        result = []
        i = j = 0
        
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        
        result.extend(left[i:])
        result.extend(right[j:])
        return result
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()

    print(f"并行归并排序时间: {parallel_time:.4f}秒")

    parallel_sorter.close()
    serial_sorter.close()
```

```python
import threading
import time
from typing import List, Any
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelQuickSort:
    """并行快速排序"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        self.lock = threading.Lock()
        self.thread_count = 0
    
    def quick_sort(self, data: List[Any]) -> List[Any]:
        """并行快速排序"""
        if len(data) <= 1:
            return data
        
        # 如果数据量小，使用串行排序
        if len(data) < 50:
            return self._serial_quick_sort(data)
        
        # 选择基准点
        pivot = data[len(data) // 2]
        
        # 分割数据
        left = [x for x in data if x < pivot]
        middle = [x for x in data if x == pivot]
        right = [x for x in data if x > pivot]
        
        # 并行排序左右两部分
        future_left = self.executor.submit(self.quick_sort, left)
        future_right = self.executor.submit(self.quick_sort, right)
        
        # 获取结果
        left_sorted = future_left.result()
        right_sorted = future_right.result()
        
        # 合并结果
        return left_sorted + middle + right_sorted
    
    def _serial_quick_sort(self, data: List[Any]) -> List[Any]:
        """串行快速排序"""
        if len(data) <= 1:
            return data
        
        pivot = data[len(data) // 2]
        left = [x for x in data if x < pivot]
        middle = [x for x in data if x == pivot]
        right = [x for x in data if x > pivot]
        
        return self._serial_quick_sort(left) + middle + self._serial_quick_sort(right)
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()
```

### 3. 并行矩阵乘法

```python
import threading
import time
from typing import List, List
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelMatrixMultiplication:
    """并行矩阵乘法"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def multiply(self, A: List[List[float]], B: List[List[float]]) -> List[List[float]]:
        """并行矩阵乘法"""
        if len(A[0]) != len(B):
            raise ValueError("矩阵维度不匹配")
        
        rows_A = len(A)
        cols_B = len(B[0])
        
        # 创建结果矩阵
        C = [[0.0 for _ in range(cols_B)] for _ in range(rows_A)]
        
        # 分解任务
        tasks = []
        for i in range(rows_A):
            for j in range(cols_B):
                task = self.executor.submit(self._multiply_element, A, B, i, j)
                tasks.append((i, j, task))
        
        # 收集结果
        for i, j, task in tasks:
            C[i][j] = task.result()
        
        return C
    
    def _multiply_element(self, A: List[List[float]], B: List[List[float]], 
                         row: int, col: int) -> float:
        """计算矩阵元素"""
        result = 0.0
        for k in range(len(A[0])):
            result += A[row][k] * B[k][col]
        return result
    
    def multiply_by_rows(self, A: List[List[float]], B: List[List[float]]) -> List[List[float]]:
        """按行并行矩阵乘法"""
        if len(A[0]) != len(B):
            raise ValueError("矩阵维度不匹配")
        
        rows_A = len(A)
        cols_B = len(B[0])
        
        # 创建结果矩阵
        C = [[0.0 for _ in range(cols_B)] for _ in range(rows_A)]
        
        # 按行分解任务
        tasks = []
        for i in range(rows_A):
            task = self.executor.submit(self._multiply_row, A, B, i)
            tasks.append((i, task))
        
        # 收集结果
        for i, task in tasks:
            C[i] = task.result()
        
        return C
    
    def _multiply_row(self, A: List[List[float]], B: List[List[float]], row: int) -> List[float]:
        """计算矩阵行"""
        result = [0.0 for _ in range(len(B[0]))]
        for j in range(len(B[0])):
            for k in range(len(A[0])):
                result[j] += A[row][k] * B[k][j]
        return result
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()
```

### 1. 并行图算法

```python
import threading
import time
from typing import List, Dict, Set, Tuple
from concurrent.futures import ThreadPoolExecutor
from collections import defaultdict, deque
import random

class ParallelGraphAlgorithms:
    """并行图算法"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def parallel_bfs(self, graph: Dict[int, List[int]], start: int) -> Dict[int, int]:
        """并行BFS"""
        distances = {start: 0}
        visited = {start}
        queue = deque([start])
        level = 0
        
        while queue:
            level_size = len(queue)
            level += 1
            
            # 并行处理当前层的所有节点
            tasks = []
            for _ in range(level_size):
                node = queue.popleft()
                task = self.executor.submit(self._process_bfs_node, graph, node, level)
                tasks.append(task)
            
            # 收集结果
            for task in tasks:
                new_nodes = task.result()
                for new_node in new_nodes:
                    if new_node not in visited:
                        visited.add(new_node)
                        distances[new_node] = level
                        queue.append(new_node)
        
        return distances
    
    def _process_bfs_node(self, graph: Dict[int, List[int]], node: int, level: int) -> List[int]:
        """处理BFS节点"""
        new_nodes = []
        for neighbor in graph.get(node, []):
            new_nodes.append(neighbor)
        return new_nodes
    
    def parallel_dfs(self, graph: Dict[int, List[int]], start: int) -> List[int]:
        """并行DFS"""
        visited = set()
        result = []
        lock = threading.Lock()
        
        def dfs_worker(node: int):
            if node in visited:
                return
            
            with lock:
                if node in visited:
                    return
                visited.add(node)
                result.append(node)
            
            # 并行处理邻居节点
            tasks = []
            for neighbor in graph.get(node, []):
                task = self.executor.submit(dfs_worker, neighbor)
                tasks.append(task)
            
            # 等待所有任务完成
            for task in tasks:
                task.result()
        
        dfs_worker(start)
        return result
    
    def parallel_connected_components(self, graph: Dict[int, List[int]]) -> List[List[int]]:
        """并行连通分量"""
        visited = set()
        components = []
        lock = threading.Lock()
        
        def find_component(start: int) -> List[int]:
            component = []
            stack = [start]
            
            while stack:
                node = stack.pop()
                if node in visited:
                    continue
                
                with lock:
                    if node in visited:
                        continue
                    visited.add(node)
                
                component.append(node)
                
                for neighbor in graph.get(node, []):
                    if neighbor not in visited:
                        stack.append(neighbor)
            
            return component
        
        # 并行处理未访问的节点
        tasks = []
        for node in graph:
            if node not in visited:
                task = self.executor.submit(find_component, node)
                tasks.append(task)
        
        # 收集结果
        for task in tasks:
            component = task.result()
            if component:
                components.append(component)
        
        return components
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()

# 使用示例
if __name__ == "__main__":
    # 生成测试图
    graph = defaultdict(list)
    for i in range(100):
        for j in range(i + 1, min(i + 10, 100)):
            graph[i].append(j)
            graph[j].append(i)
    
    # 并行图算法
    parallel_graph = ParallelGraphAlgorithms(max_workers=4)
    
    # 并行BFS
    start_time = time.time()
    distances = parallel_graph.parallel_bfs(graph, 0)
    bfs_time = time.time() - start_time
    
    print(f"并行BFS时间: {bfs_time:.4f}秒")
    print(f"BFS结果: {len(distances)} 个节点")
    
    # 并行DFS
    start_time = time.time()
    dfs_result = parallel_graph.parallel_dfs(graph, 0)
    dfs_time = time.time() - start_time
    
    print(f"并行DFS时间: {dfs_time:.4f}秒")
    print(f"DFS结果: {len(dfs_result)} 个节点")
    
    # 并行连通分量
    start_time = time.time()
    components = parallel_graph.parallel_connected_components(graph)
    cc_time = time.time() - start_time
    
    print(f"并行连通分量时间: {cc_time:.4f}秒")
    print(f"连通分量数量: {len(components)}")
    
    parallel_graph.close()
```

```python
import threading
import time
from typing import List, Any, Callable, Dict
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelDataProcessor:
    """并行数据处理器"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def parallel_map(self, data: List[Any], func: Callable) -> List[Any]:
        """并行映射"""
        # 分解数据
        chunk_size = max(1, len(data) // self.max_workers)
        chunks = [data[i:i + chunk_size] for i in range(0, len(data), chunk_size)]
        
        # 并行处理
        tasks = []
        for chunk in chunks:
            task = self.executor.submit(self._process_chunk, chunk, func)
            tasks.append(task)
        
        # 收集结果
        results = []
        for task in tasks:
            results.extend(task.result())
        
        return results
    
    def _process_chunk(self, chunk: List[Any], func: Callable) -> List[Any]:
        """处理数据块"""
        return [func(item) for item in chunk]
    
    def parallel_filter(self, data: List[Any], predicate: Callable) -> List[Any]:
        """并行过滤"""
        # 分解数据
        chunk_size = max(1, len(data) // self.max_workers)
        chunks = [data[i:i + chunk_size] for i in range(0, len(data), chunk_size)]
        
        # 并行处理
        tasks = []
        for chunk in chunks:
            task = self.executor.submit(self._filter_chunk, chunk, predicate)
            tasks.append(task)
        
        # 收集结果
        results = []
        for task in tasks:
            results.extend(task.result())
        
        return results
    
    def _filter_chunk(self, chunk: List[Any], predicate: Callable) -> List[Any]:
        """过滤数据块"""
        return [item for item in chunk if predicate(item)]
    
    def parallel_reduce(self, data: List[Any], func: Callable, initial: Any = None) -> Any:
        """并行归约"""
        if not data:
            return initial
        
        # 分解数据
        chunk_size = max(1, len(data) // self.max_workers)
        chunks = [data[i:i + chunk_size] for i in range(0, len(data), chunk_size)]
        
        # 并行处理
        tasks = []
        for chunk in chunks:
            task = self.executor.submit(self._reduce_chunk, chunk, func, initial)
            tasks.append(task)
        
        # 收集结果
        results = [task.result() for task in tasks]
        
        # 最终归约
        if initial is not None:
            result = initial
            for chunk_result in results:
                result = func(result, chunk_result)
        else:
            result = results[0]
            for chunk_result in results[1:]:
                result = func(result, chunk_result)
        
        return result
    
    def _reduce_chunk(self, chunk: List[Any], func: Callable, initial: Any = None) -> Any:
        """归约数据块"""
        if not chunk:
            return initial
        
        if initial is not None:
            result = initial
            for item in chunk:
                result = func(result, item)
        else:
            result = chunk[0]
            for item in chunk[1:]:
                result = func(result, item)
        
        return result
    
    def parallel_group_by(self, data: List[Any], key_func: Callable) -> Dict[Any, List[Any]]:
        """并行分组"""
        # 分解数据
        chunk_size = max(1, len(data) // self.max_workers)
        chunks = [data[i:i + chunk_size] for i in range(0, len(data), chunk_size)]
        
        # 并行处理
        tasks = []
        for chunk in chunks:
            task = self.executor.submit(self._group_chunk, chunk, key_func)
            tasks.append(task)
        
        # 收集结果
        results = [task.result() for task in tasks]
        
        # 合并结果
        final_result = {}
        for chunk_result in results:
            for key, values in chunk_result.items():
                if key not in final_result:
                    final_result[key] = []
                final_result[key].extend(values)
        
        return final_result
    
    def _group_chunk(self, chunk: List[Any], key_func: Callable) -> Dict[Any, List[Any]]:
        """分组数据块"""
        result = {}
        for item in chunk:
            key = key_func(item)
            if key not in result:
                result[key] = []
            result[key].append(item)
        return result
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()
```

## 性能分析

### 时间复杂度
- **线程创建**：O(1)
- **任务分解**：O(n/p)
- **并行执行**：O(n/p)
- **结果合并**：O(n)

### 空间复杂度
- **线程存储**：O(p)
- **任务存储**：O(n)
- **结果存储**：O(n)
