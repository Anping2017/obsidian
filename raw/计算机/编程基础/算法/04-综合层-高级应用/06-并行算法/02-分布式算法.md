# 分布式算法

## 概述

分布式算法（Distributed Algorithms）是在多个独立的计算节点上运行的算法，这些节点通过网络通信协作完成共同的任务。分布式算法解决了单机无法处理的大规模问题，提高了系统的可扩展性和可靠性。

## 核心概念

### 基本思想
- **节点协作**：多个节点协同工作
- **消息传递**：节点间通过消息通信
- **一致性**：保证数据的一致性
- **容错性**：处理节点故障

### 关键特性
- **分布式性**：跨多个节点执行
- **异步性**：节点独立运行
- **一致性**：数据状态一致
- **可扩展性**：随节点数扩展

## 分布式基础

### 节点通信

```python
import socket
import threading
import json
import time
from typing import Dict, Any, List, Optional
from dataclasses import dataclass
from enum import Enum

class MessageType(Enum):
    """消息类型"""
    REQUEST = "request"
    RESPONSE = "response"
    HEARTBEAT = "heartbeat"
    ELECTION = "election"
    COMMIT = "commit"
    ABORT = "abort"

@dataclass
class Message:
    """消息类"""
    type: MessageType
    sender: str
    receiver: str
    data: Any
    timestamp: float
    id: str

class Node:
    """分布式节点"""
    
    def __init__(self, node_id: str, host: str = "localhost", port: int = 8000):
        self.node_id = node_id
        self.host = host
        self.port = port
        self.socket = None
        self.connected_nodes = {}
        self.message_queue = []
        self.running = False
        self.lock = threading.Lock()
    
    def start(self) -> None:
        """启动节点"""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.bind((self.host, self.port))
        self.socket.listen(5)
        self.running = True
        
        # 启动接收线程
        receive_thread = threading.Thread(target=self._receive_messages)
        receive_thread.daemon = True
        receive_thread.start()
        
        print(f"节点 {self.node_id} 启动在 {self.host}:{self.port}")
    
    def connect_to_node(self, node_id: str, host: str, port: int) -> bool:
        """连接到其他节点"""
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.connect((host, port))
            self.connected_nodes[node_id] = sock
            print(f"节点 {self.node_id} 连接到 {node_id}")
            return True
        except Exception as e:
            print(f"连接失败: {e}")
            return False
    
    def send_message(self, message: Message) -> bool:
        """发送消息"""
        try:
            if message.receiver in self.connected_nodes:
                sock = self.connected_nodes[message.receiver]
                data = json.dumps({
                    'type': message.type.value,
                    'sender': message.sender,
                    'receiver': message.receiver,
                    'data': message.data,
                    'timestamp': message.timestamp,
                    'id': message.id
                })
                sock.send(data.encode())
                return True
            return False
        except Exception as e:
            print(f"发送消息失败: {e}")
            return False
    
    def _receive_messages(self) -> None:
        """接收消息"""
        while self.running:
            try:
                client_socket, address = self.socket.accept()
                data = client_socket.recv(1024).decode()
                message_data = json.loads(data)
                
                message = Message(
                    type=MessageType(message_data['type']),
                    sender=message_data['sender'],
                    receiver=message_data['receiver'],
                    data=message_data['data'],
                    timestamp=message_data['timestamp'],
                    id=message_data['id']
                )
                
                with self.lock:
                    self.message_queue.append(message)
                
                client_socket.close()
            except Exception as e:
                if self.running:
                    print(f"接收消息错误: {e}")
    
    def get_messages(self) -> List[Message]:
        """获取消息队列"""
        with self.lock:
            messages = self.message_queue.copy()
            self.message_queue.clear()
            return messages
    
    def stop(self) -> None:
        """停止节点"""
        self.running = False
        if self.socket:
            self.socket.close()
        for sock in self.connected_nodes.values():
            sock.close()

# 使用示例
if __name__ == "__main__":
    # 创建节点
    node1 = Node("node1", "localhost", 8001)
    node2 = Node("node2", "localhost", 8002)
    
    # 启动节点
    node1.start()
    node2.start()
    
    # 连接节点
    node1.connect_to_node("node2", "localhost", 8002)
    node2.connect_to_node("node1", "localhost", 8001)
    
    # 发送消息
    message = Message(
        type=MessageType.REQUEST,
        sender="node1",
        receiver="node2",
        data={"action": "hello", "value": 42},
        timestamp=time.time(),
        id="msg1"
    )
    
    node1.send_message(message)
    
    # 接收消息
    time.sleep(1)
    messages = node2.get_messages()
    for msg in messages:
        print(f"收到消息: {msg.data}")
    
    # 停止节点
    node1.stop()
    node2.stop()
```

### 一致性算法

```python
import threading
import time
from typing import Dict, Any, List, Optional
from dataclasses import dataclass
from enum import Enum
import random

class ConsensusState(Enum):
    """一致性状态"""
    PROPOSED = "proposed"
    ACCEPTED = "accepted"
    COMMITTED = "committed"
    ABORTED = "aborted"

@dataclass
class Proposal:
    """提案"""
    id: str
    value: Any
    proposer: str
    timestamp: float
    state: ConsensusState

class PaxosNode:
    """Paxos算法节点"""
    
    def __init__(self, node_id: str):
        self.node_id = node_id
        self.proposals = {}
        self.accepted_proposals = {}
        self.committed_proposals = {}
        self.lock = threading.Lock()
    
    def propose(self, proposal_id: str, value: Any) -> bool:
        """提出提案"""
        with self.lock:
            if proposal_id in self.proposals:
                return False
            
            proposal = Proposal(
                id=proposal_id,
                value=value,
                proposer=self.node_id,
                timestamp=time.time(),
                state=ConsensusState.PROPOSED
            )
            
            self.proposals[proposal_id] = proposal
            return True
    
    def accept_proposal(self, proposal_id: str, value: Any) -> bool:
        """接受提案"""
        with self.lock:
            if proposal_id in self.accepted_proposals:
                return False
            
            self.accepted_proposals[proposal_id] = {
                'value': value,
                'acceptor': self.node_id,
                'timestamp': time.time()
            }
            return True
    
    def commit_proposal(self, proposal_id: str, value: Any) -> bool:
        """提交提案"""
        with self.lock:
            if proposal_id in self.committed_proposals:
                return False
            
            self.committed_proposals[proposal_id] = {
                'value': value,
                'committer': self.node_id,
                'timestamp': time.time()
            }
            return True
    
    def get_proposal(self, proposal_id: str) -> Optional[Proposal]:
        """获取提案"""
        with self.lock:
            return self.proposals.get(proposal_id)
    
    def get_accepted_proposal(self, proposal_id: str) -> Optional[Dict]:
        """获取已接受的提案"""
        with self.lock:
            return self.accepted_proposals.get(proposal_id)
    
    def get_committed_proposal(self, proposal_id: str) -> Optional[Dict]:
        """获取已提交的提案"""
        with self.lock:
            return self.committed_proposals.get(proposal_id)

class RaftNode:
    """Raft算法节点"""
    
    def __init__(self, node_id: str):
        self.node_id = node_id
        self.current_term = 0
        self.voted_for = None
        self.log = []
        self.commit_index = 0
        self.last_applied = 0
        self.state = "follower"  # follower, candidate, leader
        self.lock = threading.Lock()
    
    def start_election(self) -> bool:
        """开始选举"""
        with self.lock:
            if self.state != "follower":
                return False
            
            self.current_term += 1
            self.voted_for = self.node_id
            self.state = "candidate"
            return True
    
    def vote_for_candidate(self, candidate_id: str, term: int) -> bool:
        """为候选人投票"""
        with self.lock:
            if term < self.current_term:
                return False
            
            if self.voted_for is None or self.voted_for == candidate_id:
                self.voted_for = candidate_id
                self.current_term = term
                return True
            
            return False
    
    def become_leader(self) -> bool:
        """成为领导者"""
        with self.lock:
            if self.state != "candidate":
                return False
            
            self.state = "leader"
            return True
    
    def become_follower(self, term: int) -> bool:
        """成为跟随者"""
        with self.lock:
            if term < self.current_term:
                return False
            
            self.current_term = term
            self.state = "follower"
            self.voted_for = None
            return True
    
    def append_log_entry(self, entry: Dict[str, Any]) -> bool:
        """添加日志条目"""
        with self.lock:
            if self.state != "leader":
                return False
            
            self.log.append({
                'term': self.current_term,
                'index': len(self.log),
                'data': entry
            })
            return True
    
    def commit_log_entry(self, index: int) -> bool:
        """提交日志条目"""
        with self.lock:
            if index >= len(self.log):
                return False
            
            self.commit_index = max(self.commit_index, index)
            return True

# 使用示例
if __name__ == "__main__":
    # Paxos算法示例
    paxos_node = PaxosNode("node1")
    
    # 提出提案
    proposal_id = "proposal1"
    value = "Hello World"
    
    if paxos_node.propose(proposal_id, value):
        print(f"提案 {proposal_id} 已提出")
    
    # 接受提案
    if paxos_node.accept_proposal(proposal_id, value):
        print(f"提案 {proposal_id} 已接受")
    
    # 提交提案
    if paxos_node.commit_proposal(proposal_id, value):
        print(f"提案 {proposal_id} 已提交")
    
    # Raft算法示例
    raft_node = RaftNode("node1")
    
    # 开始选举
    if raft_node.start_election():
        print(f"节点 {raft_node.node_id} 开始选举")
    
    # 成为领导者
    if raft_node.become_leader():
        print(f"节点 {raft_node.node_id} 成为领导者")
    
    # 添加日志条目
    if raft_node.append_log_entry({"action": "set", "key": "x", "value": 1}):
        print("日志条目已添加")
    
    # 提交日志条目
    if raft_node.commit_log_entry(0):
        print("日志条目已提交")
```

## 分布式算法实现

### 1. 分布式排序

```python
import threading
import time
from typing import List, Any, Dict
from dataclasses import dataclass
import random

@dataclass
class SortTask:
    """排序任务"""
    task_id: str
    data: List[Any]
    start_index: int
    end_index: int
    priority: int

class DistributedSorting:
    """分布式排序"""
    
    def __init__(self, num_workers: int = 4):
        self.num_workers = num_workers
        self.workers = {}
        self.task_queue = []
        self.result_queue = []
        self.lock = threading.Lock()
        self.running = False
    
    def start_workers(self) -> None:
        """启动工作节点"""
        self.running = True
        
        for i in range(self.num_workers):
            worker_id = f"worker_{i}"
            worker_thread = threading.Thread(
                target=self._worker_loop,
                args=(worker_id,)
            )
            worker_thread.daemon = True
            worker_thread.start()
            self.workers[worker_id] = worker_thread
        
        print(f"启动了 {self.num_workers} 个工作节点")
    
    def _worker_loop(self, worker_id: str) -> None:
        """工作节点循环"""
        while self.running:
            task = self._get_next_task()
            if task:
                result = self._process_sort_task(task)
                self._submit_result(result)
            else:
                time.sleep(0.01)
    
    def _get_next_task(self) -> Optional[SortTask]:
        """获取下一个任务"""
        with self.lock:
            if self.task_queue:
                return self.task_queue.pop(0)
            return None
    
    def _process_sort_task(self, task: SortTask) -> Dict[str, Any]:
        """处理排序任务"""
        # 模拟排序处理时间
        time.sleep(random.uniform(0.1, 0.5))
        
        # 对数据片段进行排序
        sorted_data = sorted(task.data[task.start_index:task.end_index])
        
        return {
            'task_id': task.task_id,
            'sorted_data': sorted_data,
            'start_index': task.start_index,
            'end_index': task.end_index,
            'worker_id': threading.current_thread().name
        }
    
    def _submit_result(self, result: Dict[str, Any]) -> None:
        """提交结果"""
        with self.lock:
            self.result_queue.append(result)
    
    def sort(self, data: List[Any]) -> List[Any]:
        """分布式排序"""
        if not data:
            return data
        
        # 分解任务
        chunk_size = max(1, len(data) // self.num_workers)
        tasks = []
        
        for i in range(0, len(data), chunk_size):
            end_index = min(i + chunk_size, len(data))
            task = SortTask(
                task_id=f"task_{i}",
                data=data,
                start_index=i,
                end_index=end_index,
                priority=1
            )
            tasks.append(task)
        
        # 提交任务
        with self.lock:
            self.task_queue.extend(tasks)
        
        # 等待结果
        results = []
        while len(results) < len(tasks):
            with self.lock:
                if self.result_queue:
                    results.extend(self.result_queue)
                    self.result_queue.clear()
            time.sleep(0.01)
        
        # 合并结果
        sorted_data = [None] * len(data)
        for result in results:
            start_idx = result['start_index']
            end_idx = result['end_index']
            sorted_data[start_idx:end_idx] = result['sorted_data']
        
        return sorted_data
    
    def stop_workers(self) -> None:
        """停止工作节点"""
        self.running = False
        for worker in self.workers.values():
            worker.join()
        print("所有工作节点已停止")

# 使用示例
if __name__ == "__main__":
    # 生成测试数据
    data = [random.randint(1, 1000) for _ in range(1000)]
    
    # 分布式排序
    distributed_sorter = DistributedSorting(num_workers=4)
    distributed_sorter.start_workers()
    
    start_time = time.time()
    sorted_data = distributed_sorter.sort(data.copy())
    distributed_time = time.time() - start_time
    
    print(f"分布式排序时间: {distributed_time:.4f}秒")
    print(f"排序结果正确: {sorted_data == sorted(data)}")
    
    distributed_sorter.stop_workers()
```

### 2. 分布式搜索

```python
import threading
import time
from typing import List, Any, Dict, Optional
from dataclasses import dataclass
import random

@dataclass
class SearchTask:
    """搜索任务"""
    task_id: str
    data: List[Any]
    target: Any
    start_index: int
    end_index: int

class DistributedSearch:
    """分布式搜索"""
    
    def __init__(self, num_workers: int = 4):
        self.num_workers = num_workers
        self.workers = {}
        self.task_queue = []
        self.result_queue = []
        self.lock = threading.Lock()
        self.running = False
    
    def start_workers(self) -> None:
        """启动工作节点"""
        self.running = True
        
        for i in range(self.num_workers):
            worker_id = f"worker_{i}"
            worker_thread = threading.Thread(
                target=self._worker_loop,
                args=(worker_id,)
            )
            worker_thread.daemon = True
            worker_thread.start()
            self.workers[worker_id] = worker_thread
        
        print(f"启动了 {self.num_workers} 个工作节点")
    
    def _worker_loop(self, worker_id: str) -> None:
        """工作节点循环"""
        while self.running:
            task = self._get_next_task()
            if task:
                result = self._process_search_task(task)
                if result:
                    self._submit_result(result)
            else:
                time.sleep(0.01)
    
    def _get_next_task(self) -> Optional[SearchTask]:
        """获取下一个任务"""
        with self.lock:
            if self.task_queue:
                return self.task_queue.pop(0)
            return None
    
    def _process_search_task(self, task: SearchTask) -> Optional[Dict[str, Any]]:
        """处理搜索任务"""
        # 模拟搜索处理时间
        time.sleep(random.uniform(0.01, 0.1))
        
        # 在数据片段中搜索目标
        for i in range(task.start_index, task.end_index):
            if task.data[i] == task.target:
                return {
                    'task_id': task.task_id,
                    'found_index': i,
                    'target': task.target,
                    'worker_id': threading.current_thread().name
                }
        
        return None
    
    def _submit_result(self, result: Dict[str, Any]) -> None:
        """提交结果"""
        with self.lock:
            self.result_queue.append(result)
    
    def search(self, data: List[Any], target: Any) -> Optional[int]:
        """分布式搜索"""
        if not data:
            return None
        
        # 分解任务
        chunk_size = max(1, len(data) // self.num_workers)
        tasks = []
        
        for i in range(0, len(data), chunk_size):
            end_index = min(i + chunk_size, len(data))
            task = SearchTask(
                task_id=f"task_{i}",
                data=data,
                target=target,
                start_index=i,
                end_index=end_index
            )
            tasks.append(task)
        
        # 提交任务
        with self.lock:
            self.task_queue.extend(tasks)
        
        # 等待结果
        while True:
            with self.lock:
                if self.result_queue:
                    result = self.result_queue.pop(0)
                    return result['found_index']
            
            # 检查是否所有任务都已完成
            with self.lock:
                if not self.task_queue:
                    break
            
            time.sleep(0.01)
        
        return None
    
    def stop_workers(self) -> None:
        """停止工作节点"""
        self.running = False
        for worker in self.workers.values():
            worker.join()
        print("所有工作节点已停止")

# 使用示例
if __name__ == "__main__":
    # 生成测试数据
    data = [random.randint(1, 1000) for _ in range(1000)]
    target = data[500]  # 选择一个存在的目标
    
    # 分布式搜索
    distributed_searcher = DistributedSearch(num_workers=4)
    distributed_searcher.start_workers()
    
    start_time = time.time()
    result = distributed_searcher.search(data, target)
    search_time = time.time() - start_time
    
    print(f"分布式搜索时间: {search_time:.4f}秒")
    print(f"搜索结果: {result}")
    print(f"目标值: {target}")
    
    distributed_searcher.stop_workers()
```

## 性能分析

### 时间复杂度
- **任务分解**：O(n/p)
- **并行执行**：O(n/p)
- **结果合并**：O(n)
- **通信开销**：O(p)

### 空间复杂度
- **节点存储**：O(p)
- **任务存储**：O(n)
- **结果存储**：O(n)

### 性能对比表

| 算法类型 | 串行复杂度 | 分布式复杂度 | 加速比 | 适用场景 |
|---------|-----------|-------------|--------|----------|
| 排序算法 | O(n log n) | O(n log n / p) | O(p) | 大数据量排序 |
| 搜索算法 | O(n) | O(n / p) | O(p) | 大数据量搜索 |
| 图算法 | O(V + E) | O((V + E) / p) | O(p) | 大图遍历 |
| 矩阵运算 | O(n³) | O(n³ / p) | O(p) | 大矩阵运算 |
