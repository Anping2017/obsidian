# 并行排序

## 概述

并行排序（Parallel Sorting）是利用多个处理单元并行执行排序操作的算法。通过将排序任务分解为多个独立的子任务，并在不同的处理单元中并行执行，可以显著提高排序算法的性能。

## 核心概念

### 基本思想
- **任务分解**：将排序任务分解为多个子任务
- **并行执行**：多个处理单元同时执行
- **结果合并**：将各处理单元的结果合并
- **负载均衡**：合理分配任务负载

### 关键特性
- **并行性**：多个处理单元同时工作
- **独立性**：子任务相对独立
- **可扩展性**：随处理单元数扩展
- **效率性**：提高排序性能

## 并行排序算法

### 1. 并行归并排序

```python
import threading
import time
from typing import List, Any
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelMergeSort:
    """并行归并排序"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def merge_sort(self, data: List[Any]) -> List[Any]:
        """并行归并排序"""
        if len(data) <= 1:
            return data
        
        # 如果数据量小，使用串行排序
        if len(data) < 100:
            return self._serial_merge_sort(data)
        
        # 分解任务
        mid = len(data) // 2
        left_data = data[:mid]
        right_data = data[mid:]
        
        # 并行排序左右两部分
        future_left = self.executor.submit(self.merge_sort, left_data)
        future_right = self.executor.submit(self.merge_sort, right_data)
        
        # 获取结果
        left_sorted = future_left.result()
        right_sorted = future_right.result()
        
        # 合并结果
        return self._merge(left_sorted, right_sorted)
    
    def _serial_merge_sort(self, data: List[Any]) -> List[Any]:
        """串行归并排序"""
        if len(data) <= 1:
            return data
        
        mid = len(data) // 2
        left = self._serial_merge_sort(data[:mid])
        right = self._serial_merge_sort(data[mid:])
        
        return self._merge(left, right)
    
    def _merge(self, left: List[Any], right: List[Any]) -> List[Any]:
        """合并两个有序数组"""
        result = []
        i = j = 0
        
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        
        result.extend(left[i:])
        result.extend(right[j:])
        return result
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()

# 使用示例
if __name__ == "__main__":
    # 生成测试数据
    data = [random.randint(1, 1000) for _ in range(1000)]
    
    # 并行归并排序
    parallel_sorter = ParallelMergeSort(max_workers=4)
    
    start_time = time.time()
    sorted_data = parallel_sorter.merge_sort(data.copy())
    parallel_time = time.time() - start_time
    
    print(f"并行归并排序时间: {parallel_time:.4f}秒")
    print(f"排序结果正确: {sorted_data == sorted(data)}")
    
    # 串行归并排序对比
    serial_sorter = ParallelMergeSort(max_workers=1)
    
    start_time = time.time()
    serial_sorted_data = serial_sorter.merge_sort(data.copy())
    serial_time = time.time() - start_time
    
    print(f"串行归并排序时间: {serial_time:.4f}秒")
    print(f"加速比: {serial_time / parallel_time:.2f}x")
    
    parallel_sorter.close()
    serial_sorter.close()
```

### 2. 并行快速排序

```python
import threading
import time
from typing import List, Any
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelQuickSort:
    """并行快速排序"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def quick_sort(self, data: List[Any]) -> List[Any]:
        """并行快速排序"""
        if len(data) <= 1:
            return data
        
        # 如果数据量小，使用串行排序
        if len(data) < 50:
            return self._serial_quick_sort(data)
        
        # 选择基准点
        pivot = data[len(data) // 2]
        
        # 分割数据
        left = [x for x in data if x < pivot]
        middle = [x for x in data if x == pivot]
        right = [x for x in data if x > pivot]
        
        # 并行排序左右两部分
        future_left = self.executor.submit(self.quick_sort, left)
        future_right = self.executor.submit(self.quick_sort, right)
        
        # 获取结果
        left_sorted = future_left.result()
        right_sorted = future_right.result()
        
        # 合并结果
        return left_sorted + middle + right_sorted
    
    def _serial_quick_sort(self, data: List[Any]) -> List[Any]:
        """串行快速排序"""
        if len(data) <= 1:
            return data
        
        pivot = data[len(data) // 2]
        left = [x for x in data if x < pivot]
        middle = [x for x in data if x == pivot]
        right = [x for x in data if x > pivot]
        
        return self._serial_quick_sort(left) + middle + self._serial_quick_sort(right)
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()
```

### 3. 并行堆排序

```python
import threading
import time
from typing import List, Any
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelHeapSort:
    """并行堆排序"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def heap_sort(self, data: List[Any]) -> List[Any]:
        """并行堆排序"""
        if len(data) <= 1:
            return data
        
        # 如果数据量小，使用串行排序
        if len(data) < 100:
            return self._serial_heap_sort(data)
        
        # 构建最大堆
        self._build_max_heap(data)
        
        # 并行提取元素
        for i in range(len(data) - 1, 0, -1):
            data[0], data[i] = data[i], data[0]
            self._heapify(data, i, 0)
        
        return data
    
    def _build_max_heap(self, data: List[Any]) -> None:
        """构建最大堆"""
        n = len(data)
        for i in range(n // 2 - 1, -1, -1):
            self._heapify(data, n, i)
    
    def _heapify(self, data: List[Any], n: int, i: int) -> None:
        """堆化操作"""
        largest = i
        left = 2 * i + 1
        right = 2 * i + 2
        
        if left < n and data[left] > data[largest]:
            largest = left
        
        if right < n and data[right] > data[largest]:
            largest = right
        
        if largest != i:
            data[i], data[largest] = data[largest], data[i]
            self._heapify(data, n, largest)
    
    def _serial_heap_sort(self, data: List[Any]) -> List[Any]:
        """串行堆排序"""
        if len(data) <= 1:
            return data
        
        # 构建最大堆
        self._build_max_heap(data)
        
        # 提取元素
        for i in range(len(data) - 1, 0, -1):
            data[0], data[i] = data[i], data[0]
            self._heapify(data, i, 0)
        
        return data
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()
```

## 高级并行排序

### 1. 并行基数排序

```python
import threading
import time
from typing import List, int
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelRadixSort:
    """并行基数排序"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def radix_sort(self, data: List[int]) -> List[int]:
        """并行基数排序"""
        if not data:
            return data
        
        # 找到最大数
        max_num = max(data)
        
        # 按位排序
        for exp in range(1, max_num + 1):
            data = self._counting_sort(data, exp)
        
        return data
    
    def _counting_sort(self, data: List[int], exp: int) -> List[int]:
        """计数排序"""
        n = len(data)
        output = [0] * n
        count = [0] * 10
        
        # 计算计数
        for i in range(n):
            index = (data[i] // exp) % 10
            count[index] += 1
        
        # 计算位置
        for i in range(1, 10):
            count[i] += count[i - 1]
        
        # 构建输出数组
        for i in range(n - 1, -1, -1):
            index = (data[i] // exp) % 10
            output[count[index] - 1] = data[i]
            count[index] -= 1
        
        return output
    
    def parallel_radix_sort(self, data: List[int]) -> List[int]:
        """并行基数排序"""
        if not data:
            return data
        
        # 如果数据量小，使用串行排序
        if len(data) < 1000:
            return self.radix_sort(data)
        
        # 分解任务
        chunk_size = max(1, len(data) // self.max_workers)
        chunks = [data[i:i + chunk_size] for i in range(0, len(data), chunk_size)]
        
        # 并行排序各块
        futures = []
        for chunk in chunks:
            future = self.executor.submit(self.radix_sort, chunk)
            futures.append(future)
        
        # 收集结果
        sorted_chunks = [future.result() for future in futures]
        
        # 合并结果
        return self._merge_sorted_chunks(sorted_chunks)
    
    def _merge_sorted_chunks(self, chunks: List[List[int]]) -> List[int]:
        """合并有序块"""
        if not chunks:
            return []
        
        if len(chunks) == 1:
            return chunks[0]
        
        # 使用归并排序合并
        while len(chunks) > 1:
            new_chunks = []
            for i in range(0, len(chunks), 2):
                if i + 1 < len(chunks):
                    merged = self._merge_two_chunks(chunks[i], chunks[i + 1])
                    new_chunks.append(merged)
                else:
                    new_chunks.append(chunks[i])
            chunks = new_chunks
        
        return chunks[0]
    
    def _merge_two_chunks(self, chunk1: List[int], chunk2: List[int]) -> List[int]:
        """合并两个有序块"""
        result = []
        i = j = 0
        
        while i < len(chunk1) and j < len(chunk2):
            if chunk1[i] <= chunk2[j]:
                result.append(chunk1[i])
                i += 1
            else:
                result.append(chunk2[j])
                j += 1
        
        result.extend(chunk1[i:])
        result.extend(chunk2[j:])
        return result
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()

# 使用示例
if __name__ == "__main__":
    # 生成测试数据
    data = [random.randint(1, 10000) for _ in range(10000)]
    
    # 并行基数排序
    parallel_sorter = ParallelRadixSort(max_workers=4)
    
    start_time = time.time()
    sorted_data = parallel_sorter.parallel_radix_sort(data.copy())
    parallel_time = time.time() - start_time
    
    print(f"并行基数排序时间: {parallel_time:.4f}秒")
    print(f"排序结果正确: {sorted_data == sorted(data)}")
    
    # 串行基数排序对比
    serial_sorter = ParallelRadixSort(max_workers=1)
    
    start_time = time.time()
    serial_sorted_data = serial_sorter.parallel_radix_sort(data.copy())
    serial_time = time.time() - start_time
    
    print(f"串行基数排序时间: {serial_time:.4f}秒")
    print(f"加速比: {serial_time / parallel_time:.2f}x")
    
    parallel_sorter.close()
    serial_sorter.close()
```

### 2. 并行桶排序

```python
import threading
import time
from typing import List, Any
from concurrent.futures import ThreadPoolExecutor
import random

class ParallelBucketSort:
    """并行桶排序"""
    
    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def bucket_sort(self, data: List[float]) -> List[float]:
        """桶排序"""
        if not data:
            return data
        
        # 创建桶
        num_buckets = len(data)
        buckets = [[] for _ in range(num_buckets)]
        
        # 分配元素到桶
        for num in data:
            bucket_index = int(num * num_buckets)
            if bucket_index >= num_buckets:
                bucket_index = num_buckets - 1
            buckets[bucket_index].append(num)
        
        # 排序每个桶
        for bucket in buckets:
            bucket.sort()
        
        # 合并结果
        result = []
        for bucket in buckets:
            result.extend(bucket)
        
        return result
    
    def parallel_bucket_sort(self, data: List[float]) -> List[float]:
        """并行桶排序"""
        if not data:
            return data
        
        # 如果数据量小，使用串行排序
        if len(data) < 1000:
            return self.bucket_sort(data)
        
        # 创建桶
        num_buckets = len(data)
        buckets = [[] for _ in range(num_buckets)]
        
        # 分配元素到桶
        for num in data:
            bucket_index = int(num * num_buckets)
            if bucket_index >= num_buckets:
                bucket_index = num_buckets - 1
            buckets[bucket_index].append(num)
        
        # 并行排序每个桶
        futures = []
        for bucket in buckets:
            if bucket:  # 只处理非空桶
                future = self.executor.submit(sorted, bucket)
                futures.append(future)
            else:
                futures.append(None)
        
        # 收集结果
        sorted_buckets = []
        for future in futures:
            if future:
                sorted_buckets.append(future.result())
            else:
                sorted_buckets.append([])
        
        # 合并结果
        result = []
        for bucket in sorted_buckets:
            result.extend(bucket)
        
        return result
    
    def close(self) -> None:
        """关闭线程池"""
        self.executor.shutdown()
```

## 性能分析

### 时间复杂度
- **并行归并排序**：O(n log n / p)
- **并行快速排序**：O(n log n / p)
- **并行堆排序**：O(n log n / p)
- **并行基数排序**：O(d * n / p)

### 空间复杂度
- **并行归并排序**：O(n)
- **并行快速排序**：O(log n)
- **并行堆排序**：O(1)
- **并行基数排序**：O(n + k)

### 性能对比表

| 算法类型 | 串行复杂度 | 并行复杂度 | 加速比 | 适用场景 |
|---------|-----------|-----------|--------|----------|
| 归并排序 | O(n log n) | O(n log n / p) | O(p) | 稳定排序 |
| 快速排序 | O(n log n) | O(n log n / p) | O(p) | 平均性能好 |
| 堆排序 | O(n log n) | O(n log n / p) | O(p) | 原地排序 |
| 基数排序 | O(d * n) | O(d * n / p) | O(p) | 整数排序 |
| 桶排序 | O(n + k) | O(n + k / p) | O(p) | 均匀分布 |
