# 多边形面积

## 概述

多边形面积（Polygon Area）是计算几何中的基础问题，用于计算多边形的面积。多边形面积算法在计算机图形学、CAD系统、地理信息系统等领域有重要应用。

## 核心概念

### 基本定义
- **多边形面积**：多边形围成的区域大小
- **简单多边形**：边不相交的多边形
- **复杂多边形**：边可能相交的多边形
- **有向面积**：带方向符号的面积

### 关键特性
- **唯一性**：多边形的面积是唯一的
- **可加性**：面积可以相加
- **方向性**：有向面积有正负
- **单位性**：面积有单位

## 多边形面积算法基础

### 基本多边形面积算法

```python
from typing import List, Tuple, Any, Callable, Optional
import time
import math

class PolygonAreaAlgorithms:
    """多边形面积算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.polygon = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, polygon: List[Tuple[float, float]]) -> float:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(polygon)
    
    def set_polygon(self, polygon: List[Tuple[float, float]]) -> None:
        """设置多边形"""
        self.polygon = polygon

# 鞋带公式（Shoelace Formula）
def shoelace_formula(polygon: List[Tuple[float, float]]) -> float:
    """鞋带公式计算多边形面积"""
    if len(polygon) < 3:
        return 0.0
    
    n = len(polygon)
    area = 0.0
    
    for i in range(n):
        j = (i + 1) % n
        area += polygon[i][0] * polygon[j][1]
        area -= polygon[j][0] * polygon[i][1]
    
    return abs(area) / 2.0

# 三角形分解法
def triangle_decomposition(polygon: List[Tuple[float, float]]) -> float:
    """三角形分解法计算多边形面积"""
    if len(polygon) < 3:
        return 0.0
    
    # 选择第一个顶点作为公共顶点
    common_vertex = polygon[0]
    area = 0.0
    
    for i in range(1, len(polygon) - 1):
        p1 = polygon[i]
        p2 = polygon[i + 1]
        
        # 计算三角形面积
        triangle_area = abs(
            (p1[0] - common_vertex[0]) * (p2[1] - common_vertex[1]) - 
            (p2[0] - common_vertex[0]) * (p1[1] - common_vertex[1])
        ) / 2.0
        
        area += triangle_area
    
    return area

# 梯形分解法
def trapezoid_decomposition(polygon: List[Tuple[float, float]]) -> float:
    """梯形分解法计算多边形面积"""
    if len(polygon) < 3:
        return 0.0
    
    area = 0.0
    
    for i in range(len(polygon)):
        j = (i + 1) % len(polygon)
        
        # 计算梯形的有向面积
        trapezoid_area = (polygon[i][0] + polygon[j][0]) * (polygon[j][1] - polygon[i][1]) / 2.0
        area += trapezoid_area
    
    return abs(area)

# 使用示例
if __name__ == "__main__":
    algorithms = PolygonAreaAlgorithms()
    
    # 添加算法
    algorithms.add_algorithm("shoelace_formula", shoelace_formula)
    algorithms.add_algorithm("triangle_decomposition", triangle_decomposition)
    algorithms.add_algorithm("trapezoid_decomposition", trapezoid_decomposition)
    
    # 测试多边形
    test_polygon = [(0, 0), (2, 0), (2, 2), (0, 2)]
    
    # 测试鞋带公式
    result = algorithms.execute_algorithm("shoelace_formula", test_polygon)
    print(f"鞋带公式结果: {result}")
    
    # 测试三角形分解法
    result = algorithms.execute_algorithm("triangle_decomposition", test_polygon)
    print(f"三角形分解法结果: {result}")
    
    # 测试梯形分解法
    result = algorithms.execute_algorithm("trapezoid_decomposition", test_polygon)
    print(f"梯形分解法结果: {result}")
```

```python
# 复杂多边形面积计算
def complex_polygon_area(polygon: List[Tuple[float, float]]) -> float:
    """复杂多边形面积计算"""
    if len(polygon) < 3:
        return 0.0
    
    # 使用鞋带公式计算有向面积
    n = len(polygon)
    signed_area = 0.0
    
    for i in range(n):
        j = (i + 1) % n
        signed_area += polygon[i][0] * polygon[j][1]
        signed_area -= polygon[j][0] * polygon[i][1]
    
    return abs(signed_area) / 2.0

# 带洞的多边形面积计算
def polygon_with_holes_area(outer_polygon: List[Tuple[float, float]], 
                           holes: List[List[Tuple[float, float]]]) -> float:
    """带洞的多边形面积计算"""
    # 计算外多边形面积
    outer_area = shoelace_formula(outer_polygon)
    
    # 计算所有洞的面积
    holes_area = 0.0
    for hole in holes:
        holes_area += shoelace_formula(hole)
    
    return outer_area - holes_area

# 三维多边形面积计算
def polygon_area_3d(polygon: List[Tuple[float, float, float]]) -> float:
    """三维多边形面积计算"""
    if len(polygon) < 3:
        return 0.0
    
    # 计算法向量
    def cross_product(p1: Tuple[float, float, float], p2: Tuple[float, float, float], p3: Tuple[float, float, float]) -> Tuple[float, float, float]:
        return (
            (p2[1] - p1[1]) * (p3[2] - p1[2]) - (p2[2] - p1[2]) * (p3[1] - p1[1]),
            (p2[2] - p1[2]) * (p3[0] - p1[0]) - (p2[0] - p1[0]) * (p3[2] - p1[2]),
            (p2[0] - p1[0]) * (p3[1] - p1[1]) - (p2[1] - p1[1]) * (p3[0] - p1[0])
        )
    
    # 计算多边形面积
    area = 0.0
    for i in range(len(polygon)):
        j = (i + 1) % len(polygon)
        k = (i + 2) % len(polygon)
        
        # 计算三角形的面积
        normal = cross_product(polygon[i], polygon[j], polygon[k])
        triangle_area = math.sqrt(normal[0]**2 + normal[1]**2 + normal[2]**2) / 2.0
        area += triangle_area
    
    return area

# 使用示例
if __name__ == "__main__":

    advanced_algorithms.add_algorithm("polygon_with_holes", polygon_with_holes_area)
    advanced_algorithms.add_algorithm("polygon_area_3d", polygon_area_3d)
    
    # 测试复杂多边形
    test_polygon = [(0, 0), (2, 0), (2, 2), (0, 2)]
    result = advanced_algorithms.execute_algorithm("complex_polygon", test_polygon)
    print(f"复杂多边形面积: {result}")
    
    # 测试带洞的多边形
    outer_polygon = [(0, 0), (4, 0), (4, 4), (0, 4)]
    holes = [[(1, 1), (3, 1), (3, 3), (1, 3)]]
    result = advanced_algorithms.execute_algorithm("polygon_with_holes", (outer_polygon, holes))
    print(f"带洞的多边形面积: {result}")
    
    # 测试三维多边形
    test_polygon_3d = [(0, 0, 0), (2, 0, 0), (2, 2, 0), (0, 2, 0)]
    result = advanced_algorithms.execute_algorithm("polygon_area_3d", test_polygon_3d)
    print(f"三维多边形面积: {result}")
```
