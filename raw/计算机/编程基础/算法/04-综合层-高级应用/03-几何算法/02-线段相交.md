# 线段相交

## 概述

线段相交（Line Segment Intersection）是计算几何中的基础问题，用于判断两条线段是否相交，以及找到所有线段的交点。线段相交在计算机图形学、CAD系统、地理信息系统等领域有重要应用。

## 核心概念

### 基本定义
- **线段相交**：两条线段有公共点
- **交点**：两条线段的公共点
- **方向向量**：线段的方向向量
- **叉积**：用于判断线段方向的向量运算

### 关键特性
- **唯一性**：两条线段最多有一个交点
- **包含性**：交点必须在两条线段上
- **方向性**：可以通过方向判断相交
- **边界性**：需要考虑线段端点

## 线段相交算法基础

### 基本线段相交算法

```python
from typing import List, Tuple, Any, Callable, Optional
import time
import math

class LineSegmentIntersection:
    """线段相交算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.segments = []
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, segments: List[Tuple[Tuple[float, float], Tuple[float, float]]]) -> List[Tuple[float, float]]:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(segments)
    
    def set_segments(self, segments: List[Tuple[Tuple[float, float], Tuple[float, float]]]) -> None:
        """设置线段"""
        self.segments = segments

# 两条线段相交判断
def two_segments_intersect(segment1: Tuple[Tuple[float, float], Tuple[float, float]], 
                          segment2: Tuple[Tuple[float, float], Tuple[float, float]]) -> Tuple[bool, Optional[Tuple[float, float]]]:
    """判断两条线段是否相交"""
    p1, p2 = segment1
    p3, p4 = segment2
    
    # 计算叉积
    def cross_product(p1: Tuple[float, float], p2: Tuple[float, float], p3: Tuple[float, float]) -> float:
        return (p2[0] - p1[0]) * (p3[1] - p1[1]) - (p2[1] - p1[1]) * (p3[0] - p1[0])
    
    # 计算方向
    def direction(p1: Tuple[float, float], p2: Tuple[float, float], p3: Tuple[float, float]) -> int:
        cross = cross_product(p1, p2, p3)
        if cross > 0:
            return 1  # 逆时针
        elif cross < 0:
            return -1  # 顺时针
        else:
            return 0  # 共线
    
    # 检查点是否在线段上
    def on_segment(p1: Tuple[float, float], p2: Tuple[float, float], p3: Tuple[float, float]) -> bool:
        return (min(p1[0], p2[0]) <= p3[0] <= max(p1[0], p2[0]) and
                min(p1[1], p2[1]) <= p3[1] <= max(p1[1], p2[1]))
    
    # 计算两条线段的交点
    def line_intersection(p1: Tuple[float, float], p2: Tuple[float, float], 
                         p3: Tuple[float, float], p4: Tuple[float, float]) -> Optional[Tuple[float, float]]:
        x1, y1 = p1
        x2, y2 = p2
        x3, y3 = p3
        x4, y4 = p4
        
        denom = (x1 - x2) * (y3 - y4) - (y1 - y2) * (x3 - x4)
        if abs(denom) < 1e-10:
            return None
        
        t = ((x1 - x3) * (y3 - y4) - (y1 - y3) * (x3 - x4)) / denom
        u = -((x1 - x2) * (y1 - y3) - (y1 - y2) * (x1 - x3)) / denom
        
        if 0 <= t <= 1 and 0 <= u <= 1:
            x = x1 + t * (x2 - x1)
            y = y1 + t * (y2 - y1)
            return (x, y)
        
        return None
    
    # 判断相交
    d1 = direction(p3, p4, p1)
    d2 = direction(p3, p4, p2)
    d3 = direction(p1, p2, p3)
    d4 = direction(p1, p2, p4)
    
    # 一般情况：两条线段相交
    if ((d1 > 0 and d2 < 0) or (d1 < 0 and d2 > 0)) and ((d3 > 0 and d4 < 0) or (d3 < 0 and d4 > 0)):
        intersection = line_intersection(p1, p2, p3, p4)
        return True, intersection
    
    # 特殊情况：共线
    if d1 == 0 and on_segment(p3, p4, p1):
        return True, p1
    if d2 == 0 and on_segment(p3, p4, p2):
        return True, p2
    if d3 == 0 and on_segment(p1, p2, p3):
        return True, p3
    if d4 == 0 and on_segment(p1, p2, p4):
        return True, p4
    
    return False, None

# 多条线段相交检测
def multiple_segments_intersect(segments: List[Tuple[Tuple[float, float], Tuple[float, float]]]) -> List[Tuple[int, int, Tuple[float, float]]]:
    """检测多条线段的相交情况"""
    intersections = []
    
    for i in range(len(segments)):
        for j in range(i + 1, len(segments)):
            segment1 = segments[i]
            segment2 = segments[j]
            
            intersects, intersection_point = two_segments_intersect(segment1, segment2)
            if intersects and intersection_point:
                intersections.append((i, j, intersection_point))
    
    return intersections

# 扫描线算法
def sweep_line_intersection(segments: List[Tuple[Tuple[float, float], Tuple[float, float]]]) -> List[Tuple[int, int, Tuple[float, float]]]:
    """扫描线算法检测线段相交"""
    events = []
    
    # 创建事件点
    for i, segment in enumerate(segments):
        p1, p2 = segment
        if p1[0] > p2[0] or (p1[0] == p2[0] and p1[1] > p2[1]):
            p1, p2 = p2, p1
        
        events.append((p1[0], 'start', i, p1, p2))
        events.append((p2[0], 'end', i, p1, p2))
    
    # 按x坐标排序事件
    events.sort(key=lambda x: x[0])
    
    # 活跃线段集合
    active_segments = set()
    intersections = []
    
    for event in events:
        x, event_type, segment_id, p1, p2 = event
        
        if event_type == 'start':
            # 检查与活跃线段的相交
            for active_id in active_segments:
                active_segment = segments[active_id]
                intersects, intersection_point = two_segments_intersect((p1, p2), active_segment)
                if intersects and intersection_point:
                    intersections.append((segment_id, active_id, intersection_point))
            
            active_segments.add(segment_id)
        
        elif event_type == 'end':
            active_segments.remove(segment_id)
    
    return intersections
```

```python
# 平面扫描算法
def plane_sweep_intersection(segments: List[Tuple[Tuple[float, float], Tuple[float, float]]]) -> List[Tuple[int, int, Tuple[float, float]]]:
    """平面扫描算法检测线段相交"""
    events = []
    
    # 创建事件点
    for i, segment in enumerate(segments):
        p1, p2 = segment
        if p1[0] > p2[0] or (p1[0] == p2[0] and p1[1] > p2[1]):
            p1, p2 = p2, p1
        
        events.append((p1[0], 'start', i, p1, p2))
        events.append((p2[0], 'end', i, p1, p2))
    
    # 按x坐标排序事件
    events.sort(key=lambda x: x[0])
    
    # 活跃线段集合（按y坐标排序）
    active_segments = []
    intersections = []
    
    def insert_segment(segment_id: int, p1: Tuple[float, float], p2: Tuple[float, float]):
        """插入线段到活跃集合"""
        y = p1[1] + (p2[1] - p1[1]) * (events[0][0] - p1[0]) / (p2[0] - p1[0]) if p2[0] != p1[0] else p1[1]
        
        # 找到插入位置
        insert_pos = 0
        for i, (active_id, active_p1, active_p2) in enumerate(active_segments):
            active_y = active_p1[1] + (active_p2[1] - active_p1[1]) * (events[0][0] - active_p1[0]) / (active_p2[0] - active_p1[0]) if active_p2[0] != active_p1[0] else active_p1[1]
            if y < active_y:
                insert_pos = i
                break
            insert_pos = i + 1
        
        active_segments.insert(insert_pos, (segment_id, p1, p2))
    
    def remove_segment(segment_id: int):
        """从活跃集合中移除线段"""
        for i, (active_id, _, _) in enumerate(active_segments):
            if active_id == segment_id:
                active_segments.pop(i)
                break
    
    def check_intersections(segment_id: int, p1: Tuple[float, float], p2: Tuple[float, float]):
        """检查线段与活跃线段的相交"""
        for active_id, active_p1, active_p2 in active_segments:
            if active_id == segment_id:
                continue
            
            intersects, intersection_point = two_segments_intersect((p1, p2), (active_p1, active_p2))
            if intersects and intersection_point:
                intersections.append((segment_id, active_id, intersection_point))
    
    for event in events:
        x, event_type, segment_id, p1, p2 = event
        
        if event_type == 'start':
            # 检查与活跃线段的相交
            check_intersections(segment_id, p1, p2)
            insert_segment(segment_id, p1, p2)
        
        elif event_type == 'end':
            remove_segment(segment_id)
    
    return intersections

# 线段树算法
def segment_tree_intersection(segments: List[Tuple[Tuple[float, float], Tuple[float, float]]]) -> List[Tuple[int, int, Tuple[float, float]]]:
    """线段树算法检测线段相交"""
    # 收集所有x坐标
    x_coords = set()
    for segment in segments:
        p1, p2 = segment
        x_coords.add(p1[0])
        x_coords.add(p2[0])
    
    x_coords = sorted(list(x_coords))
    
    # 构建线段树
    class SegmentTreeNode:
        def __init__(self, start: int, end: int):
            self.start = start
            self.end = end
            self.segments = []
            self.left = None
            self.right = None
    
    def build_segment_tree(start: int, end: int) -> SegmentTreeNode:
        node = SegmentTreeNode(start, end)
        if start == end:
            return node
        
        mid = (start + end) // 2
        node.left = build_segment_tree(start, mid)
        node.right = build_segment_tree(mid + 1, end)
        return node
    
    def insert_segment(node: SegmentTreeNode, segment_id: int, p1: Tuple[float, float], p2: Tuple[float, float]):
        """插入线段到线段树"""
        if p1[0] <= x_coords[node.start] and x_coords[node.end] <= p2[0]:
            node.segments.append(segment_id)
            return
        
        if node.left and p1[0] <= x_coords[node.left.end]:
            insert_segment(node.left, segment_id, p1, p2)
        
        if node.right and p2[0] >= x_coords[node.right.start]:
            insert_segment(node.right, segment_id, p1, p2)
    
    def query_intersections(node: SegmentTreeNode, segment_id: int, p1: Tuple[float, float], p2: Tuple[float, float]) -> List[Tuple[int, int, Tuple[float, float]]]:
        """查询线段相交"""
        intersections = []
        
        # 检查当前节点的线段
        for active_id in node.segments:
            if active_id == segment_id:
                continue
            
            active_segment = segments[active_id]
            intersects, intersection_point = two_segments_intersect((p1, p2), active_segment)
            if intersects and intersection_point:
                intersections.append((segment_id, active_id, intersection_point))
        
        # 递归查询子节点
        if node.left and p1[0] <= x_coords[node.left.end]:
            intersections.extend(query_intersections(node.left, segment_id, p1, p2))
        
        if node.right and p2[0] >= x_coords[node.right.start]:
            intersections.extend(query_intersections(node.right, segment_id, p1, p2))
        
        return intersections
    
    # 构建线段树
    root = build_segment_tree(0, len(x_coords) - 1)
    
    # 插入所有线段
    for i, segment in enumerate(segments):
        p1, p2 = segment
        insert_segment(root, i, p1, p2)
    
    # 查询相交
    intersections = []
    for i, segment in enumerate(segments):
        p1, p2 = segment
        intersections.extend(query_intersections(root, i, p1, p2))
    
    return intersections
```

## 性能分析

### 线段相交算法分析

| 算法类型 | 时间复杂度 | 空间复杂度 | 适用场景 | 学习难度 |
|---------|-----------|-----------|----------|----------|
| 两条线段相交 | O(1) | O(1) | 基础判断 | 简单 |
| 多条线段相交 | O(n²) | O(1) | 小规模 | 简单 |
