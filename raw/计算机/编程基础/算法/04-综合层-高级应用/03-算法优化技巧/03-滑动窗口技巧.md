# 滑动窗口技巧

## 🎯 核心概念

**滑动窗口技巧**是一种高效的算法优化技术，通过维护一个动态的窗口来解决问题。滑动窗口技巧通常能将O(n²)的暴力解法优化到O(n)，是算法优化的重要方法。

```python
def sliding_window_types():
    """滑动窗口类型"""
    
    def fixed_window(arr, k):
        """固定窗口"""
        if len(arr) < k:
            return []
        
        window_sum = sum(arr[:k])
        max_sum = window_sum
        
        for i in range(k, len(arr)):
            window_sum = window_sum - arr[i - k] + arr[i]
            max_sum = max(max_sum, window_sum)
        
        return max_sum
    
    def variable_window(arr, target):
        """可变窗口"""
        left = 0
        min_length = float('inf')
        current_sum = 0
        
        for right in range(len(arr)):
            current_sum += arr[right]
            
            while current_sum >= target:
                min_length = min(min_length, right - left + 1)
                current_sum -= arr[left]
                left += 1
        
        return min_length if min_length != float('inf') else 0
    
    def two_pointer_window(arr, target):
        """双指针窗口"""
        left = 0
        max_length = 0
        current_sum = 0
        
        for right in range(len(arr)):
            current_sum += arr[right]
            
            while current_sum > target:
                current_sum -= arr[left]
                left += 1
            
            max_length = max(max_length, right - left + 1)
        
        return max_length
    
    return fixed_window, variable_window, two_pointer_window
```

```python
def sliding_window_algorithms():
    """滑动窗口算法构建"""
    
    class SlidingWindow:
        """滑动窗口类"""
        def __init__(self):
            self.results = []
        
        def max_sum_subarray(self, arr, k):
            """最大子数组和"""
            if len(arr) < k:
                return 0
            
            window_sum = sum(arr[:k])
            max_sum = window_sum
            
            for i in range(k, len(arr)):
                window_sum = window_sum - arr[i - k] + arr[i]
                max_sum = max(max_sum, window_sum)
            
            return max_sum
        
        def min_subarray_length(self, arr, target):
            """最小子数组长度"""
            left = 0
            min_length = float('inf')
            current_sum = 0
            
            for right in range(len(arr)):
                current_sum += arr[right]
                
                while current_sum >= target:
                    min_length = min(min_length, right - left + 1)
                    current_sum -= arr[left]
                    left += 1
            
            return min_length if min_length != float('inf') else 0
        
        def longest_substring_without_repeating(self, s):
            """最长无重复字符子串"""
            char_set = set()
            left = 0
            max_length = 0
            
            for right in range(len(s)):
                while s[right] in char_set:
                    char_set.remove(s[left])
                    left += 1
                
                char_set.add(s[right])
                max_length = max(max_length, right - left + 1)
            
            return max_length
        
        def longest_substring_with_at_most_k_distinct(self, s, k):
            """最多k个不同字符的最长子串"""
            char_count = {}
            left = 0
            max_length = 0
            
            for right in range(len(s)):
                char_count[s[right]] = char_count.get(s[right], 0) + 1
                
                while len(char_count) > k:
                    char_count[s[left]] -= 1
                    if char_count[s[left]] == 0:
                        del char_count[s[left]]
                    left += 1
                
                max_length = max(max_length, right - left + 1)
            
            return max_length
        
        def longest_substring_with_at_least_k_repeating(self, s, k):
            """至少k个重复字符的最长子串"""
            max_length = 0
            
            for unique_chars in range(1, 27):
                char_count = {}
                left = 0
                valid_chars = 0
                
                for right in range(len(s)):
                    char_count[s[right]] = char_count.get(s[right], 0) + 1
                    
                    if char_count[s[right]] == 1:
                        valid_chars += 1
                    
                    while valid_chars > unique_chars:
                        char_count[s[left]] -= 1
                        if char_count[s[left]] == 0:
                            valid_chars -= 1
                        left += 1
                    
                    if valid_chars == unique_chars:
                        all_valid = True
                        for count in char_count.values():
                            if count > 0 and count < k:
                                all_valid = False
                                break
                        
                        if all_valid:
                            max_length = max(max_length, right - left + 1)
            
            return max_length
        
        def sliding_window_maximum(self, arr, k):
            """滑动窗口最大值"""
            if not arr or k == 0:
                return []
            
            result = []
            window = []
            
            for i in range(len(arr)):
                while window and window[0] <= i - k:
                    window.pop(0)
                
                while window and arr[window[-1]] <= arr[i]:
                    window.pop()
                
                window.append(i)
                
                if i >= k - 1:
                    result.append(arr[window[0]])
            
            return result
        
        def sliding_window_minimum(self, arr, k):
            """滑动窗口最小值"""
            if not arr or k == 0:
                return []
            
            result = []
            window = []
            
            for i in range(len(arr)):
                while window and window[0] <= i - k:
                    window.pop(0)
                
                while window and arr[window[-1]] >= arr[i]:
                    window.pop()
                
                window.append(i)
                
                if i >= k - 1:
                    result.append(arr[window[0]])
            
            return result
        
        def sliding_window_median(self, arr, k):
            """滑动窗口中位数"""
            if not arr or k == 0:
                return []
            
            result = []
            window = []
            
            for i in range(len(arr)):
                window.append(arr[i])
                
                if len(window) > k:
                    window.pop(0)
                
                if len(window) == k:
                    sorted_window = sorted(window)
                    if k % 2 == 1:
                        median = sorted_window[k // 2]
                    else:
                        median = (sorted_window[k // 2 - 1] + sorted_window[k // 2]) / 2
                    result.append(median)
            
            return result
        
        def sliding_window_average(self, arr, k):
            """滑动窗口平均值"""
            if not arr or k == 0:
                return []
            
            result = []
            window_sum = 0
            
            for i in range(len(arr)):
                window_sum += arr[i]
                
                if i >= k - 1:
                    result.append(window_sum / k)
                    window_sum -= arr[i - k + 1]
            
            return result
        
        def sliding_window_sum(self, arr, k):
            """滑动窗口和"""
            if not arr or k == 0:
                return []
            
            result = []
            window_sum = 0
            
            for i in range(len(arr)):
                window_sum += arr[i]
                
                if i >= k - 1:
                    result.append(window_sum)
                    window_sum -= arr[i - k + 1]
            
            return result
    
    return SlidingWindow
```

```python
        def longest_palindromic_substring(self, s):
            """最长回文子串"""
            if not s:
                return ""
            
            start = 0
            max_length = 1
            
            for i in range(len(s)):
                # 奇数长度回文
                left, right = i, i
                while left >= 0 and right < len(s) and s[left] == s[right]:
                    if right - left + 1 > max_length:
                        start = left
                        max_length = right - left + 1
                    left -= 1
                    right += 1
                
                # 偶数长度回文
                left, right = i, i + 1
                while left >= 0 and right < len(s) and s[left] == s[right]:
                    if right - left + 1 > max_length:
                        start = left
                        max_length = right - left + 1
                    left -= 1
                    right += 1
            
            return s[start:start + max_length]

                while left >= 0 and right < len(s) and s[left] == s[right]:

                    left -= 1
                    right += 1
                
                # 偶数长度回文
                left, right = i, i + 1
                while left >= 0 and right < len(s) and s[left] == s[right]:
                    count += 1
                    left -= 1
                    right += 1
            
            return count
    
    return SubstringSlidingWindow
```

```python
def continuous_sequence_problems():
    """连续序列问题"""
    
    class ContinuousSequenceSlidingWindow:
        """连续序列滑动窗口"""
        def __init__(self):
            self.results = []
        
        def longest_continuous_increasing_subsequence(self, arr):
            """最长连续递增子序列"""
            if not arr:
                return 0
            
            max_length = 1
            current_length = 1
            
            for i in range(1, len(arr)):
                if arr[i] > arr[i - 1]:
                    current_length += 1
                else:
                    max_length = max(max_length, current_length)
                    current_length = 1
            
            return max(max_length, current_length)
        
        def longest_continuous_decreasing_subsequence(self, arr):
            """最长连续递减子序列"""

        def longest_continuous_equal_subsequence(self, arr):
            """最长连续相等子序列"""
            if not arr:

            current_length = 1
            
            for i in range(1, len(arr)):
                if arr[i] == arr[i - 1]:
                    current_length += 1
                else:
                    max_length = max(max_length, current_length)
                    current_length = 1
            
            return max(max_length, current_length)
        
        def longest_continuous_alternating_subsequence(self, arr):
            """最长连续交替子序列"""
            if not arr:
                return 0
            
            max_length = 1
            current_length = 1
            
            for i in range(1, len(arr)):
                if (arr[i] > arr[i - 1] and arr[i - 1] < arr[i - 2]) or \
                   (arr[i] < arr[i - 1] and arr[i - 1] > arr[i - 2]):
                    current_length += 1
                else:
                    max_length = max(max_length, current_length)
                    current_length = 1
            
            return max(max_length, current_length)
        
        def longest_continuous_monotonic_subsequence(self, arr):
            """最长连续单调子序列"""
            if not arr:
                return 0
            
            max_length = 1
            current_length = 1
            is_increasing = None
            
            for i in range(1, len(arr)):
                if arr[i] > arr[i - 1]:
                    if is_increasing is None or is_increasing:
                        current_length += 1
                        is_increasing = True
                    else:
                        max_length = max(max_length, current_length)
                        current_length = 2
                        is_increasing = True
                elif arr[i] < arr[i - 1]:
                    if is_increasing is None or not is_increasing:
                        current_length += 1
                        is_increasing = False
                    else:
                        max_length = max(max_length, current_length)
                        current_length = 2
                        is_increasing = False
                else:
                    max_length = max(max_length, current_length)
                    current_length = 1
                    is_increasing = None
            
            return max(max_length, current_length)
        
        def longest_continuous_positive_subsequence(self, arr):
            """最长连续正数子序列"""
            if not arr:
                return 0
            
            max_length = 0
            current_length = 0
            
            for num in arr:
                if num > 0:
                    current_length += 1
                else:
                    max_length = max(max_length, current_length)
                    current_length = 0
            
            return max(max_length, current_length)
        
        def longest_continuous_negative_subsequence(self, arr):
            """最长连续负数子序列"""
            if not arr:
                return 0
            
            max_length = 0
            current_length = 0
            
            for num in arr:
                if num < 0:
                    current_length += 1
                else:
                    max_length = max(max_length, current_length)
                    current_length = 0
            
            return max(max_length, current_length)
        
        def longest_continuous_zero_subsequence(self, arr):
            """最长连续零子序列"""
            if not arr:
                return 0
            
            max_length = 0
            current_length = 0
            
            for num in arr:
                if num == 0:
                    current_length += 1
                else:
                    max_length = max(max_length, current_length)
                    current_length = 0
            
            return max(max_length, current_length)
    
    return ContinuousSequenceSlidingWindow
```
