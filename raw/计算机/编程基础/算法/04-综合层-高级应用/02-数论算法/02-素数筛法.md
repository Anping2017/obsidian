# 素数筛法

## 概述

素数筛法（Prime Sieve）是数论中的重要算法，用于高效地找出一定范围内的所有素数。素数筛法在密码学、数论研究、算法竞赛等领域有重要应用。

## 核心概念

### 基本定义
- **素数**：只能被1和自身整除的大于1的自然数
- **合数**：除了1和自身外还有其他因数的自然数
- **筛法**：通过排除合数来找出素数的方法
- **埃拉托斯特尼筛法**：最经典的素数筛法

### 关键特性
- **唯一性**：每个自然数要么是素数要么是合数
- **无穷性**：素数有无穷多个
- **分布性**：素数分布不规则
- **筛除性**：通过筛除合数找出素数

## 素数筛法算法基础

### 基本素数筛法算法

from typing import List, Tuple, Any, Callable, Optional
import time
import math

class PrimeSieveAlgorithms:
    """素数筛法算法示例"""
    
    def __init__(self):
        self.algorithms = {}
        self.limit = 0
    
    def add_algorithm(self, name: str, algorithm: Callable) -> None:
        """添加算法"""
        self.algorithms[name] = algorithm
    
    def execute_algorithm(self, name: str, limit: int) -> List[int]:
        """执行算法"""
        if name not in self.algorithms:
            raise ValueError(f"未知算法: {name}")
        
        algorithm = self.algorithms[name]
        return algorithm(limit)
    
    def set_limit(self, limit: int) -> None:
        """设置上限"""
        self.limit = limit

# 埃拉托斯特尼筛法
def eratosthenes_sieve(limit: int) -> List[int]:
    """埃拉托斯特尼筛法"""
    if limit < 2:
        return []
    
    # 创建布尔数组，初始化为True
    is_prime = [True] * (limit + 1)
    is_prime[0] = is_prime[1] = False
    
    # 筛除合数
    for i in range(2, int(math.sqrt(limit)) + 1):
        if is_prime[i]:
            for j in range(i * i, limit + 1, i):
                is_prime[j] = False
    
    # 收集素数
    primes = []
    for i in range(2, limit + 1):
        if is_prime[i]:
            primes.append(i)
    
    return primes

# 优化的埃拉托斯特尼筛法
def optimized_eratosthenes_sieve(limit: int) -> List[int]:
    """优化的埃拉托斯特尼筛法"""
    if limit < 2:
        return []
    
    # 创建布尔数组，初始化为True
    is_prime = [True] * (limit + 1)
    is_prime[0] = is_prime[1] = False
    
    # 筛除合数（优化版本）
    for i in range(2, int(math.sqrt(limit)) + 1):
        if is_prime[i]:
            # 从i*i开始，因为更小的倍数已经被更小的素数筛除了
            for j in range(i * i, limit + 1, i):
                is_prime[j] = False
    
    # 收集素数
    primes = []
    for i in range(2, limit + 1):
        if is_prime[i]:
            primes.append(i)
    
    return primes

# 分段筛法
def segmented_sieve(limit: int) -> List[int]:
    """分段筛法"""
    if limit < 2:
        return []
    
    # 计算分段大小
    segment_size = int(math.sqrt(limit))
    
    # 使用埃拉托斯特尼筛法找出小素数
    small_primes = eratosthenes_sieve(segment_size)
    
    # 分段处理
    primes = small_primes.copy()
    
    for low in range(segment_size, limit + 1, segment_size):
        high = min(low + segment_size - 1, limit)
        
        # 创建当前段的布尔数组
        is_prime = [True] * (high - low + 1)
        
        # 使用小素数筛除当前段的合数
        for prime in small_primes:
            # 找到当前段中第一个被prime整除的数
            start = max(prime * prime, (low + prime - 1) // prime * prime)
            
            for j in range(start, high + 1, prime):
                is_prime[j - low] = False
        
        # 收集当前段的素数
        for i in range(len(is_prime)):
            if is_prime[i]:
                primes.append(low + i)
    
    return primes

# 线性筛法
def linear_sieve(limit: int) -> List[int]:
    """线性筛法"""
    if limit < 2:
        return []
    
    # 创建布尔数组和最小素因子数组
    is_prime = [True] * (limit + 1)
    min_factor = [0] * (limit + 1)
    primes = []
    
    for i in range(2, limit + 1):
        if is_prime[i]:
            primes.append(i)
            min_factor[i] = i
        
        # 使用已找到的素数筛除合数
        for prime in primes:
            if prime > min_factor[i] or i * prime > limit:
                break
            
            is_prime[i * prime] = False
            min_factor[i * prime] = prime
    
    return primes

# 使用示例
if __name__ == "__main__":

def wheel_sieve(limit: int) -> List[int]:
    """轮式筛法"""
    if limit < 2:
        return []
    
    # 使用2,3,5作为轮子
    wheel = [2, 3, 5]
    wheel_size = 2 * 3 * 5
    
    # 创建轮子模式
    wheel_pattern = []
    for i in range(1, wheel_size + 1):
        if all(i % p != 0 for p in wheel):
            wheel_pattern.append(i)
    
    # 创建布尔数组
    is_prime = [True] * (limit + 1)
    is_prime[0] = is_prime[1] = False
    
    # 筛除合数
    for i in range(2, int(math.sqrt(limit)) + 1):
        if is_prime[i]:
            for j in range(i * i, limit + 1, i):
                is_prime[j] = False
    
    # 使用轮子模式进一步优化
    primes = [2, 3, 5]
    for i in range(7, limit + 1, 2):
        if is_prime[i]:
            primes.append(i)
    
    return primes

def miller_rabin_test(n: int, k: int = 10) -> bool:
    """Miller-Rabin概率素数测试"""
    if n < 2:
        return False
    if n == 2 or n == 3:
        return True
    if n % 2 == 0:
        return False
    
    # 将n-1写成d * 2^r的形式
    d = n - 1
    r = 0
    while d % 2 == 0:
        d //= 2
        r += 1
    
    # 进行k次测试
    for _ in range(k):
        a = random.randint(2, n - 2)
        x = pow(a, d, n)
        
        if x == 1 or x == n - 1:
            continue
        
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                break
        else:
            return False
    
    return True

# 费马素数测试
def fermat_test(n: int, k: int = 10) -> bool:
    """费马素数测试"""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    
    # 进行k次测试
    for _ in range(k):
        a = random.randint(2, n - 2)
        if pow(a, n - 1, n) != 1:
            return False
    
    return True

## 性能分析

### 素数筛法算法分析

| 算法类型 | 时间复杂度 | 空间复杂度 | 适用场景 | 学习难度 |
|---------|-----------|-----------|----------|----------|
| 埃拉托斯特尼筛法 | O(n log log n) | O(n) | 一般情况 | 简单 |
| 优化的埃拉托斯特尼筛法 | O(n log log n) | O(n) | 一般情况 | 简单 |
| 分段筛法 | O(n log log n) | O(√n) | 大范围 | 中等 |
| 线性筛法 | O(n) | O(n) | 需要最小素因子 | 中等 |
| 轮式筛法 | O(n log log n) | O(n) | 大范围 | 困难 |
| Miller-Rabin测试 | O(k log³n) | O(1) | 大数测试 | 困难 |

### 算法选择分析

| 问题特征 | 推荐算法 | 优势 | 劣势 |
|---------|---------|------|------|
| 小范围 | 埃拉托斯特尼筛法 | 简单、高效 | 空间复杂度高 |
| 大范围 | 分段筛法 | 空间效率高 | 实现复杂 |
| 需要最小素因子 | 线性筛法 | 线性时间 | 空间复杂度高 |
| 大数测试 | Miller-Rabin测试 | 概率正确 | 可能误判 |
| 性能要求高 | 轮式筛法 | 高效 | 实现复杂 |
