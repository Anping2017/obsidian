# 数论算法

## 🎯 核心概念

**数论算法**是研究整数性质的算法，涉及素数、最大公约数、模运算、同余等数学概念。数论算法在密码学、计算机科学、算法竞赛等领域有重要应用，是算法设计的重要基础。

### 2. 数论基础
```python
def number_theory_basics():
    """数论基础"""
    
    def gcd(a, b):
        """最大公约数"""
        while b:
            a, b = b, a % b
        return a
    
    def lcm(a, b):
        """最小公倍数"""
        return a * b // gcd(a, b)
    
    def extended_gcd(a, b):
        """扩展欧几里得算法"""
        if a == 0:
            return b, 0, 1
        
        gcd_val, x1, y1 = extended_gcd(b % a, a)
        x = y1 - (b // a) * x1
        y = x1
        
        return gcd_val, x, y
    
    def modular_inverse(a, m):
        """模逆元"""
        gcd_val, x, y = extended_gcd(a, m)
        if gcd_val != 1:
            return None  # 不存在模逆元
        return (x % m + m) % m
    
    def fast_power(base, exp, mod):
        """快速幂"""
        result = 1
        base = base % mod
        
        while exp > 0:
            if exp % 2 == 1:
                result = (result * base) % mod
            exp = exp >> 1
            base = (base * base) % mod
        
        return result
    
    def is_prime(n):
        """判断素数"""
        if n < 2:
            return False
        if n == 2:
            return True
        if n % 2 == 0:
            return False
        
        for i in range(3, int(n**0.5) + 1, 2):
            if n % i == 0:
                return False
        return True
    
    def sieve_of_eratosthenes(n):
        """埃拉托斯特尼筛法"""
        is_prime = [True] * (n + 1)
        is_prime[0] = is_prime[1] = False
        
        for i in range(2, int(n**0.5) + 1):
            if is_prime[i]:
                for j in range(i * i, n + 1, i):
                    is_prime[j] = False
        
        return [i for i in range(2, n + 1) if is_prime[i]]
    
    def prime_factors(n):
        """质因数分解"""
        factors = []
        d = 2
        
        while d * d <= n:
            while n % d == 0:
                factors.append(d)
                n //= d
            d += 1
        
        if n > 1:
            factors.append(n)
        
        return factors
    
    def euler_phi(n):
        """欧拉函数"""
        result = n
        p = 2
        
        while p * p <= n:
            if n % p == 0:
                while n % p == 0:
                    n //= p
                result -= result // p
            p += 1
        
        if n > 1:
            result -= result // n
        
        return result
    
    return gcd, lcm, extended_gcd, modular_inverse, fast_power, is_prime, sieve_of_eratosthenes, prime_factors, euler_phi
```

```python
                    factors.append(d)
                    n //= d
                d += 1
            
            if n > 1:
                factors.append(n)

                p += 1
            
            if n > 1:

                result = (result + (remainders[i] - result) * x * current_mod) % (current_mod * moduli[i])

            if k > n:
                return 0

            while n > 0 or k > 0:

                if k_mod > n_mod:
                    return 0

            numerator = 1
            denominator = 1
            
            for i in range(k):
                numerator = (numerator * (n - i)) % mod
```

```python
def advanced_number_theory():
    """高级数论算法"""
    
    def chinese_remainder_theorem(remainders, moduli):
        """中国剩余定理"""
        n = len(remainders)
        if n == 0:
            return 0
        
        result = remainders[0]
        current_mod = moduli[0]
        
        for i in range(1, n):
            gcd_val, x, y = extended_gcd(current_mod, moduli[i])
            if gcd_val != 1:
                return None  # 无解
            
            result = (result + (remainders[i] - result) * x * current_mod) % (current_mod * moduli[i])
            current_mod *= moduli[i]
        
        return result
    
    def lucas_theorem(n, k, p):
        """卢卡斯定理"""
        if k > n:
            return 0
        
        result = 1
        while n > 0 or k > 0:
            n_mod = n % p
            k_mod = k % p
            
            if k_mod > n_mod:
                return 0
            
            result = (result * combination(n_mod, k_mod, p)) % p
            n //= p
            k //= p
        
        return result
    
    def combination(n, k, mod):
        """组合数"""
        if k > n:
            return 0
        
        numerator = 1
        denominator = 1
        
        for i in range(k):
            numerator = (numerator * (n - i)) % mod
            denominator = (denominator * (i + 1)) % mod
        
        return (numerator * modular_inverse(denominator, mod)) % mod
    
    def miller_rabin(n, k=10):
        """米勒-拉宾素性测试"""
        if n < 2:
            return False
        if n == 2 or n == 3:
            return True
        if n % 2 == 0:
            return False
        
        # 将n-1写成d * 2^r的形式
        d = n - 1
        r = 0
        while d % 2 == 0:
            d //= 2
            r += 1
        
        # 进行k轮测试
        for _ in range(k):
            a = random.randint(2, n - 2)
            x = fast_power(a, d, n)
            
            if x == 1 or x == n - 1:
                continue
            
            for _ in range(r - 1):
                x = (x * x) % n
                if x == n - 1:
                    break
            else:
                return False
        
        return True
    
    def pollard_rho(n):
        """Pollard's Rho算法"""
        if n % 2 == 0:
            return 2
        
        x = random.randint(1, n - 1)
        y = x
        c = random.randint(1, n - 1)
        d = 1
        
        while d == 1:
            x = (x * x + c) % n
            y = (y * y + c) % n
            y = (y * y + c) % n
            d = gcd(abs(x - y), n)
        
        return d if d != n else None
    
    def quadratic_residue(a, p):
        """二次剩余"""
        return fast_power(a, (p - 1) // 2, p) == 1
    
    def tonelli_shanks(n, p):
        """Tonelli-Shanks算法"""
        if not quadratic_residue(n, p):
            return None
        
        if p == 2:
            return n
        
        if p % 4 == 3:
            return fast_power(n, (p + 1) // 4, p)
        
        # 找到Q和S使得p-1 = Q * 2^S
        Q = p - 1
        S = 0
        while Q % 2 == 0:
            Q //= 2
            S += 1
        
        # 找到非二次剩余z
        z = 2
        while quadratic_residue(z, p):
            z += 1
        
        c = fast_power(z, Q, p)
        x = fast_power(n, (Q + 1) // 2, p)
        t = fast_power(n, Q, p)
        m = S
        
        while t != 1:
            tt = t
            i = 1
            while i < m and tt != 1:
                tt = (tt * tt) % p
                i += 1
            
            b = fast_power(c, 1 << (m - i - 1), p)
            x = (x * b) % p
            t = (t * b * b) % p
            c = (b * b) % p
            m = i
        
        return x
    
    def discrete_logarithm(a, b, m):
        """离散对数"""
        n = int(m**0.5) + 1
        
        # 计算a^n mod m
        an = fast_power(a, n, m)
        
        # 存储a^(i*n) mod m的值
        table = {}
        current = an
        for i in range(1, n + 1):
            if current not in table:
                table[current] = i
            current = (current * an) % m
        
        # 查找b * a^j mod m
        current = b
        for j in range(n + 1):
            if current in table:
                return table[current] * n - j
            current = (current * a) % m
        
        return None
    
    return chinese_remainder_theorem, lucas_theorem, combination, miller_rabin, pollard_rho, quadratic_residue, tonelli_shanks, discrete_logarithm
```

```python
def cryptography_applications():
    """密码学应用"""
    
    class RSA:
        """RSA加密算法"""
        def __init__(self, p, q):
            self.p = p
            self.q = q
            self.n = p * q
            self.phi = (p - 1) * (q - 1)
            self.e = self.find_e()
            self.d = self.find_d()
        
        def find_e(self):
            """找到公钥指数e"""
            for e in range(2, self.phi):
                if gcd(e, self.phi) == 1:
                    return e
            return None
        
        def find_d(self):
            """找到私钥指数d"""
            _, x, _ = extended_gcd(self.e, self.phi)
            return (x % self.phi + self.phi) % self.phi
        
        def encrypt(self, message):
            """加密"""
            return fast_power(message, self.e, self.n)
        
        def decrypt(self, ciphertext):
            """解密"""
            return fast_power(ciphertext, self.d, self.n)
    
    class DiffieHellman:
        """Diffie-Hellman密钥交换"""
        def __init__(self, p, g):
            self.p = p
            self.g = g
        
        def generate_key(self, private_key):
            """生成公钥"""
            return fast_power(self.g, private_key, self.p)
        
        def generate_shared_secret(self, private_key, public_key):
            """生成共享密钥"""
            return fast_power(public_key, private_key, self.p)
    
    class ElGamal:
        """ElGamal加密算法"""
        def __init__(self, p, g):
            self.p = p
            self.g = g
        
        def generate_key(self, private_key):
            """生成公钥"""
            return fast_power(self.g, private_key, self.p)
        
        def encrypt(self, message, public_key, random_k):
            """加密"""
            c1 = fast_power(self.g, random_k, self.p)
            c2 = (message * fast_power(public_key, random_k, self.p)) % self.p
            return c1, c2
        
        def decrypt(self, c1, c2, private_key):
            """解密"""
            s = fast_power(c1, private_key, self.p)
            s_inv = modular_inverse(s, self.p)
            return (c2 * s_inv) % self.p
    
    class EllipticCurve:
        """椭圆曲线"""
        def __init__(self, a, b, p):
            self.a = a
            self.b = b
            self.p = p
        
        def point_add(self, P, Q):
            """点加法"""
            if P is None:
                return Q
            if Q is None:
                return P
            
            x1, y1 = P
            x2, y2 = Q
            
            if x1 == x2:
                if y1 == y2:
                    # 点倍乘
                    s = (3 * x1 * x1 + self.a) * modular_inverse(2 * y1, self.p) % self.p
                else:
                    # 无穷远点
                    return None
            else:
                s = (y2 - y1) * modular_inverse(x2 - x1, self.p) % self.p
            
            x3 = (s * s - x1 - x2) % self.p
            y3 = (s * (x1 - x3) - y1) % self.p
            
            return (x3, y3)
        
        def point_multiply(self, P, k):
            """点倍乘"""
            if k == 0:
                return None
            if k == 1:
                return P
            
            if k % 2 == 0:
                return self.point_multiply(self.point_add(P, P), k // 2)
            else:
                return self.point_add(P, self.point_multiply(self.point_add(P, P), k // 2))
    
    return RSA, DiffieHellman, ElGamal, EllipticCurve
```

```python
def competitive_programming_applications():
    """算法竞赛应用"""
    
    def count_divisors(n):
        """计算因子个数"""
        factors = prime_factors(n)
        count = 1
        current_factor = factors[0]
        current_count = 1
        
        for i in range(1, len(factors)):
            if factors[i] == current_factor:
                current_count += 1
            else:
                count *= (current_count + 1)
                current_factor = factors[i]
                current_count = 1
        
        count *= (current_count + 1)
        return count
    
    def sum_of_divisors(n):
        """计算因子和"""
        factors = prime_factors(n)
        sum_val = 1
        current_factor = factors[0]
        current_power = 1
        current_sum = 1 + current_factor
        
        for i in range(1, len(factors)):
            if factors[i] == current_factor:
                current_power += 1
                current_sum += current_factor ** current_power
            else:
                sum_val *= current_sum
                current_factor = factors[i]
                current_power = 1
                current_sum = 1 + current_factor
        
        sum_val *= current_sum
        return sum_val
    
    def totient_function(n):
        """欧拉函数"""
        result = n
        p = 2
        
        while p * p <= n:
            if n % p == 0:
                while n % p == 0:
                    n //= p
                result -= result // p
            p += 1
        
        if n > 1:
            result -= result // n
        
        return result
    
    def mobius_function(n):
        """莫比乌斯函数"""
        if n == 1:
            return 1
        
        factors = prime_factors(n)
        if len(factors) != len(set(factors)):
            return 0
        
        return (-1) ** len(factors)

            return 0

        denominator = 1

            numerator = (numerator * (n - i)) % mod
            denominator = (denominator * (i + 1)) % mod

    def catalan_number(n):
        """卡特兰数"""
        if n <= 1:
            return 1
        
        result = 1
        for i in range(n):
            result = result * (2 * n - i) // (i + 1)
        
        return result
    
    def fibonacci_number(n):
        """斐波那契数"""
        if n <= 1:
            return n
        
        a, b = 0, 1
        for _ in range(2, n + 1):
            a, b = b, a + b
        
        return b
    
    def matrix_exponentiation(matrix, n):
        """矩阵快速幂"""
        if n == 1:
            return matrix
        
        if n % 2 == 0:
            half = matrix_exponentiation(matrix, n // 2)
            return matrix_multiply(half, half)
        else:
            return matrix_multiply(matrix, matrix_exponentiation(matrix, n - 1))
    
    def matrix_multiply(A, B):
        """矩阵乘法"""
        n = len(A)
        result = [[0] * n for _ in range(n)]
        
        for i in range(n):
            for j in range(n):
                for k in range(n):
                    result[i][j] += A[i][k] * B[k][j]
        
        return result
    
    return count_divisors, sum_of_divisors, totient_function, mobius_function, chinese_remainder_theorem, lucas_theorem, combination, catalan_number, fibonacci_number, matrix_exponentiation, matrix_multiply
```

```python
def computer_science_applications():
    """计算机科学应用"""
    
    class HashFunction:
        """哈希函数"""
        def __init__(self, m):
            self.m = m
        
        def hash(self, key):
            """哈希函数"""
            return key % self.m
        
        def hash_string(self, s):
            """字符串哈希"""
            hash_val = 0
            for char in s:
                hash_val = (hash_val * 256 + ord(char)) % self.m
            return hash_val
    
    class RandomNumberGenerator:
        """随机数生成器"""
        def __init__(self, seed):
            self.seed = seed
        
        def next(self):
            """生成下一个随机数"""
            self.seed = (self.seed * 1103515245 + 12345) % (2**31)
            return self.seed
        
        def next_range(self, min_val, max_val):
            """生成指定范围的随机数"""
            return min_val + (self.next() % (max_val - min_val + 1))
    
    class Checksum:
        """校验和"""
        def __init__(self):
            self.polynomial = 0x1021  # CRC-16多项式
        
        def crc16(self, data):
            """CRC-16校验"""
            crc = 0xFFFF
            
            for byte in data:
                crc ^= byte << 8
                for _ in range(8):
                    if crc & 0x8000:
                        crc = (crc << 1) ^ self.polynomial
                    else:
                        crc <<= 1
                    crc &= 0xFFFF
            
            return crc
        
        def luhn_algorithm(self, number):
            """Luhn算法"""
            digits = [int(d) for d in str(number)]
            
            for i in range(len(digits) - 2, -1, -2):
                digits[i] *= 2
                if digits[i] > 9:
                    digits[i] = digits[i] // 10 + digits[i] % 10
            
            return sum(digits) % 10 == 0
    
    class ErrorCorrection:
        """错误纠正"""
        def __init__(self):
            self.generator_polynomial = 0x11D  # Reed-Solomon生成多项式
        
        def reed_solomon_encode(self, data, n, k):
            """Reed-Solomon编码"""
            if len(data) != k:
                raise ValueError("数据长度必须等于k")
            
            # 创建生成矩阵
            generator_matrix = []
            for i in range(k):
                row = [0] * n
                for j in range(k):
                    row[j] = 1 if i == j else 0
                for j in range(k, n):
                    row[j] = self.galois_multiply(data[i], self.galois_power(2, j - k))
                generator_matrix.append(row)
            
            # 计算校验位
            codeword = [0] * n
            for i in range(k):
                for j in range(n):
                    codeword[j] ^= self.galois_multiply(data[i], generator_matrix[i][j])
            
            return codeword
        
        def galois_multiply(self, a, b):
            """伽罗瓦域乘法"""
            result = 0
            for _ in range(8):
                if b & 1:
                    result ^= a
                a <<= 1
                if a & 0x100:
                    a ^= self.generator_polynomial
                b >>= 1
            return result & 0xFF
        
        def galois_power(self, base, exp):
            """伽罗瓦域幂运算"""
            result = 1
            for _ in range(exp):
                result = self.galois_multiply(result, base)
            return result
    
    class Compression:
        """压缩算法"""
        def __init__(self):
            self.frequency = {}
            self.codes = {}
        
        def huffman_encode(self, text):
            """霍夫曼编码"""
            # 计算频率
            for char in text:
                self.frequency[char] = self.frequency.get(char, 0) + 1
            
            # 构建霍夫曼树
            heap = []
            for char, freq in self.frequency.items():
                heap.append((freq, char))
            
            heap.sort()
            
            while len(heap) > 1:
                freq1, char1 = heap.pop(0)
                freq2, char2 = heap.pop(0)
                merged = (freq1 + freq2, char1 + char2)
                heap.append(merged)
                heap.sort()
            
            # 生成编码
            self.generate_codes(heap[0][1], "")
            
            # 编码文本
            encoded = ""
            for char in text:
                encoded += self.codes[char]
            
            return encoded
        
        def generate_codes(self, node, code):
            """生成编码"""
            if len(node) == 1:
                self.codes[node] = code
            else:
                self.generate_codes(node[0], code + "0")
                self.generate_codes(node[1], code + "1")
    
    return HashFunction, RandomNumberGenerator, Checksum, ErrorCorrection, Compression
```
