# 几何算法

## 🎯 核心概念

**几何算法**是处理几何图形和空间关系的算法，涉及点、线、面、多边形等几何对象。几何算法在计算机图形学、机器人学、地理信息系统等领域有重要应用。

### 2. 几何基础
```python
def geometry_basics():
    """几何基础"""
    
    class Point:
        """点类"""
        def __init__(self, x, y):
            self.x = x
            self.y = y
        
        def __add__(self, other):
            return Point(self.x + other.x, self.y + other.y)
        
        def __sub__(self, other):
            return Point(self.x - other.x, self.y - other.y)
        
        def __mul__(self, scalar):
            return Point(self.x * scalar, self.y * scalar)
        
        def dot(self, other):
            """点积"""
            return self.x * other.x + self.y * other.y
        
        def cross(self, other):
            """叉积"""
            return self.x * other.y - self.y * other.x
        
        def distance(self, other):
            """距离"""
            return ((self.x - other.x)**2 + (self.y - other.y)**2)**0.5
    
    class Line:
        """直线类"""
        def __init__(self, p1, p2):
            self.p1 = p1
            self.p2 = p2
        
        def direction(self):
            """方向向量"""
            return self.p2 - self.p1
        
        def length(self):
            """长度"""
            return self.p1.distance(self.p2)
    
    def point_in_polygon(point, polygon):
        """点在多边形内"""
        n = len(polygon)
        inside = False
        
        for i in range(n):
            j = (i + 1) % n
            if ((polygon[i].y > point.y) != (polygon[j].y > point.y)):
                if point.x < (polygon[j].x - polygon[i].x) * (point.y - polygon[i].y) / (polygon[j].y - polygon[i].y) + polygon[i].x:
                    inside = not inside
        
        return inside
    
    def line_intersection(line1, line2):
        """线段相交"""
        p1, p2 = line1.p1, line1.p2
        p3, p4 = line2.p1, line2.p2
        
        d1 = (p4 - p3).cross(p1 - p3)
        d2 = (p4 - p3).cross(p2 - p3)
        d3 = (p2 - p1).cross(p3 - p1)
        d4 = (p2 - p1).cross(p4 - p1)
        
        if ((d1 > 0 and d2 < 0) or (d1 < 0 and d2 > 0)) and ((d3 > 0 and d4 < 0) or (d3 < 0 and d4 > 0)):
            return True
        
        return False
    
    def polygon_area(polygon):
        """多边形面积"""
        n = len(polygon)
        area = 0
        
        for i in range(n):
            j = (i + 1) % n
            area += polygon[i].x * polygon[j].y
            area -= polygon[j].x * polygon[i].y
        
        return abs(area) / 2
    
    return Point, Line, point_in_polygon, line_intersection, polygon_area
```

### 3. 几何算法构建
```python
def geometry_algorithms():
    """几何算法构建"""
    
    class GeometryAlgorithms:
        """几何算法类"""
        def __init__(self):
            self.points = []
            self.polygons = []
        
        def convex_hull(self, points):
            """凸包算法"""
            if len(points) < 3:
                return points
            
            # 找到最左下角的点
            start = min(points, key=lambda p: (p.x, p.y))
            
            # 按极角排序
            sorted_points = sorted(points, key=lambda p: self.polar_angle(start, p))
            
            # 构建凸包
            hull = [start]
            for point in sorted_points[1:]:
                while len(hull) > 1 and self.orientation(hull[-2], hull[-1], point) <= 0:
                    hull.pop()
                hull.append(point)
            
            return hull
        
        def polar_angle(self, p1, p2):
            """极角"""
            return math.atan2(p2.y - p1.y, p2.x - p1.x)
        
        def orientation(self, p1, p2, p3):
            """方向"""
            return (p2 - p1).cross(p3 - p1)
        
        def closest_pair(self, points):
            """最近点对"""
            if len(points) < 2:
                return None, None, float('inf')
            
            if len(points) == 2:
                return points[0], points[1], points[0].distance(points[1])
            
            # 按x坐标排序
            points.sort(key=lambda p: p.x)
            
            # 分治
            mid = len(points) // 2
            left_points = points[:mid]
            right_points = points[mid:]
            
            p1, p2, d1 = self.closest_pair(left_points)
            p3, p4, d2 = self.closest_pair(right_points)
            
            if d1 < d2:
                min_dist = d1
                closest = (p1, p2)
            else:
                min_dist = d2
                closest = (p3, p4)
            
            # 检查跨分界线的点对
            mid_x = points[mid].x
            strip = [p for p in points if abs(p.x - mid_x) < min_dist]
            strip.sort(key=lambda p: p.y)
            
            for i in range(len(strip)):
                for j in range(i + 1, len(strip)):
                    if strip[j].y - strip[i].y >= min_dist:
                        break
                    dist = strip[i].distance(strip[j])
                    if dist < min_dist:
                        min_dist = dist
                        closest = (strip[i], strip[j])
            
            return closest[0], closest[1], min_dist
        
        def line_segment_intersection(self, line1, line2):
            """线段相交"""
            p1, p2 = line1.p1, line1.p2
            p3, p4 = line2.p1, line2.p2
            
            d1 = (p4 - p3).cross(p1 - p3)
            d2 = (p4 - p3).cross(p2 - p3)
            d3 = (p2 - p1).cross(p3 - p1)
            d4 = (p2 - p1).cross(p4 - p1)
            
            if ((d1 > 0 and d2 < 0) or (d1 < 0 and d2 > 0)) and ((d3 > 0 and d4 < 0) or (d3 < 0 and d4 > 0)):
                return True
            
            return False
        
        def point_in_polygon(self, point, polygon):
            """点在多边形内"""
            n = len(polygon)
            inside = False
            
            for i in range(n):
                j = (i + 1) % n
                if ((polygon[i].y > point.y) != (polygon[j].y > point.y)):
                    if point.x < (polygon[j].x - polygon[i].x) * (point.y - polygon[i].y) / (polygon[j].y - polygon[i].y) + polygon[i].x:
                        inside = not inside
            
            return inside
        
        def polygon_area(self, polygon):
            """多边形面积"""
            n = len(polygon)
            area = 0
            
            for i in range(n):
                j = (i + 1) % n
                area += polygon[i].x * polygon[j].y
                area -= polygon[j].x * polygon[i].y
            
            return abs(area) / 2
        
        def circle_intersection(self, circle1, circle2):
            """圆相交"""
            d = circle1.center.distance(circle2.center)
            r1, r2 = circle1.radius, circle2.radius
            
            if d > r1 + r2 or d < abs(r1 - r2):
                return False  # 不相交
            
            return True
        
        def triangle_area(self, p1, p2, p3):
            """三角形面积"""
            return abs((p2 - p1).cross(p3 - p1)) / 2
        
        def point_to_line_distance(self, point, line):
            """点到直线距离"""
            p1, p2 = line.p1, line.p2
            return abs((p2 - p1).cross(point - p1)) / line.length()
    
    return GeometryAlgorithms

def advanced_geometry():
    """高级几何算法"""
    
    def delaunay_triangulation(points):
        """Delaunay三角剖分"""
        if len(points) < 3:
            return []
        
        # 创建超级三角形
        super_triangle = self.create_super_triangle(points)
        triangles = [super_triangle]
        
        for point in points:
            # 找到包含该点的三角形
            bad_triangles = []
            for triangle in triangles:
                if self.point_in_circumcircle(point, triangle):
                    bad_triangles.append(triangle)
            
            # 找到边界
            polygon = []
            for triangle in bad_triangles:
                for edge in triangle.edges:
                    if not self.edge_shared_by_triangles(edge, bad_triangles):
                        polygon.append(edge)
            
            # 移除坏三角形
            for triangle in bad_triangles:
                triangles.remove(triangle)
            
            # 添加新三角形
            for edge in polygon:
                new_triangle = Triangle(edge.p1, edge.p2, point)
                triangles.append(new_triangle)
        
        # 移除包含超级三角形顶点的三角形
        triangles = [t for t in triangles if not self.triangle_contains_super_vertex(t, super_triangle)]
        
        return triangles
    
    def voronoi_diagram(points):
        """Voronoi图"""
        if len(points) < 2:
            return []
        
        # 使用Delaunay三角剖分
        triangles = delaunay_triangulation(points)
        
        # 构建Voronoi图
        voronoi_cells = []
        for point in points:
            cell = []
            for triangle in triangles:
                if point in triangle.vertices:
                    # 找到外接圆圆心
                    center = self.circumcenter(triangle)
                    cell.append(center)
            
            voronoi_cells.append(cell)
        
        return voronoi_cells
    
    def minkowski_sum(polygon1, polygon2):
        """Minkowski和"""
        result = []
        
        for p1 in polygon1:
            for p2 in polygon2:
                result.append(p1 + p2)
        
        return convex_hull(result)
    
    def polygon_clipping(polygon, clip_polygon):
        """多边形裁剪"""
        result = polygon
        
        for i in range(len(clip_polygon)):
            j = (i + 1) % len(clip_polygon)
            clip_edge = Line(clip_polygon[i], clip_polygon[j])
            
            new_result = []
            for k in range(len(result)):
                l = (k + 1) % len(result)
                edge = Line(result[k], result[l])
                
                if self.line_inside_clip_edge(edge, clip_edge):
                    new_result.append(result[l])
                elif self.line_intersects_clip_edge(edge, clip_edge):
                    intersection = self.line_intersection_point(edge, clip_edge)
                    new_result.append(intersection)
                    if not self.line_inside_clip_edge(edge, clip_edge):
                        new_result.append(result[l])
            
            result = new_result
        
        return result
    
    def convex_hull_3d(points):
        """3D凸包"""
        if len(points) < 4:
            return points
        
        # 使用Gift Wrapping算法
        hull = []
        
        # 找到最底部的点
        start = min(points, key=lambda p: p.z)
        hull.append(start)
        
        # 找到第一个面
        for i in range(len(points)):
            if points[i] != start:
                for j in range(len(points)):
                    if points[j] != start and points[j] != points[i]:
                        # 检查是否所有其他点都在这个面的同一侧
                        normal = (points[i] - start).cross(points[j] - start)
                        valid = True
                        for k in range(len(points)):
                            if points[k] != start and points[k] != points[i] and points[k] != points[j]:
                                if (points[k] - start).dot(normal) > 0:
                                    valid = False
                                    break
                        if valid:
                            hull.append(points[i])
                            hull.append(points[j])
                            break
                if len(hull) == 3:
                    break
        
        return hull
    
    return delaunay_triangulation, voronoi_diagram, minkowski_sum, polygon_clipping, convex_hull_3d

def computer_graphics_applications():
    """计算机图形学应用"""
    
    class Rasterizer:
        """光栅化器"""
        def __init__(self, width, height):
            self.width = width
            self.height = height
            self.pixels = [[0 for _ in range(width)] for _ in range(height)]
        
        def draw_line(self, p1, p2, color):
            """绘制直线"""
            x1, y1 = int(p1.x), int(p1.y)
            x2, y2 = int(p2.x), int(p2.y)
            
            dx = abs(x2 - x1)
            dy = abs(y2 - y1)
            
            if dx > dy:
                if x1 > x2:
                    x1, x2 = x2, x1
                    y1, y2 = y2, y1
                
                for x in range(x1, x2 + 1):
                    y = y1 + (y2 - y1) * (x - x1) // (x2 - x1)
                    if 0 <= x < self.width and 0 <= y < self.height:
                        self.pixels[y][x] = color
            else:
                if y1 > y2:
                    x1, x2 = x2, x1
                    y1, y2 = y2, y1
                
                for y in range(y1, y2 + 1):
                    x = x1 + (x2 - x1) * (y - y1) // (y2 - y1)
                    if 0 <= x < self.width and 0 <= y < self.height:
                        self.pixels[y][x] = color
        
        def draw_triangle(self, p1, p2, p3, color):
            """绘制三角形"""
            # 使用扫描线算法
            points = [p1, p2, p3]
            points.sort(key=lambda p: p.y)
            
            for y in range(int(points[0].y), int(points[2].y) + 1):
                if y < int(points[1].y):
                    # 上半部分
                    x1 = self.interpolate_x(points[0], points[1], y)
                    x2 = self.interpolate_x(points[0], points[2], y)
                else:
                    # 下半部分
                    x1 = self.interpolate_x(points[1], points[2], y)
                    x2 = self.interpolate_x(points[0], points[2], y)
                
                if x1 > x2:
                    x1, x2 = x2, x1
                
                for x in range(int(x1), int(x2) + 1):
                    if 0 <= x < self.width and 0 <= y < self.height:
                        self.pixels[y][x] = color
        
        def interpolate_x(self, p1, p2, y):
            """插值计算x坐标"""
            if p2.y == p1.y:
                return p1.x
            return p1.x + (p2.x - p1.x) * (y - p1.y) / (p2.y - p1.y)
    
    class Transform:
        """变换类"""
        def __init__(self):
            self.matrix = [[1, 0, 0], [0, 1, 0], [0, 0, 1]]
        
        def translate(self, dx, dy):
            """平移"""
            self.matrix[0][2] += dx
            self.matrix[1][2] += dy
        
        def rotate(self, angle):
            """旋转"""
            cos_a = math.cos(angle)
            sin_a = math.sin(angle)
            
            rotation_matrix = [[cos_a, -sin_a, 0], [sin_a, cos_a, 0], [0, 0, 1]]
            self.matrix = self.multiply_matrices(self.matrix, rotation_matrix)
        
        def scale(self, sx, sy):
            """缩放"""
            scale_matrix = [[sx, 0, 0], [0, sy, 0], [0, 0, 1]]
            self.matrix = self.multiply_matrices(self.matrix, scale_matrix)
        
        def multiply_matrices(self, m1, m2):
            """矩阵乘法"""
            result = [[0 for _ in range(3)] for _ in range(3)]
            for i in range(3):
                for j in range(3):
                    for k in range(3):
                        result[i][j] += m1[i][k] * m2[k][j]
            return result
        
        def transform_point(self, point):
            """变换点"""
            x = self.matrix[0][0] * point.x + self.matrix[0][1] * point.y + self.matrix[0][2]
            y = self.matrix[1][0] * point.x + self.matrix[1][1] * point.y + self.matrix[1][2]
            return Point(x, y)
    
    return Rasterizer, Transform

def robotics_applications():
    """机器人学应用"""
    
    class PathPlanner:
        """路径规划器"""
        def __init__(self):
            self.obstacles = []
            self.start = None
            self.goal = None
        
        def add_obstacle(self, polygon):
            """添加障碍物"""
            self.obstacles.append(polygon)
        
        def set_start_goal(self, start, goal):
            """设置起点和终点"""
            self.start = start
            self.goal = goal
        
        def find_path(self):
            """寻找路径"""
            # 使用A*算法
            open_set = [self.start]
            closed_set = set()
            came_from = {}
            g_score = {self.start: 0}
            f_score = {self.start: self.heuristic(self.start, self.goal)}
            
            while open_set:
                current = min(open_set, key=lambda p: f_score.get(p, float('inf')))
                
                if current == self.goal:
                    return self.reconstruct_path(came_from, current)
                
                open_set.remove(current)
                closed_set.add(current)
                
                for neighbor in self.get_neighbors(current):
                    if neighbor in closed_set:
                        continue
                    
                    tentative_g_score = g_score[current] + current.distance(neighbor)
                    
                    if neighbor not in open_set:
                        open_set.append(neighbor)
                    elif tentative_g_score >= g_score.get(neighbor, float('inf')):
                        continue
                    
                    came_from[neighbor] = current
                    g_score[neighbor] = tentative_g_score
                    f_score[neighbor] = g_score[neighbor] + self.heuristic(neighbor, self.goal)
            
            return None
        
        def heuristic(self, p1, p2):
            """启发式函数"""
            return p1.distance(p2)
        
        def get_neighbors(self, point):
            """获取邻居点"""
            neighbors = []
            for obstacle in self.obstacles:
                for vertex in obstacle:
                    if not self.line_intersects_obstacles(point, vertex):
                        neighbors.append(vertex)
            return neighbors
        
        def line_intersects_obstacles(self, p1, p2):
            """检查线段是否与障碍物相交"""
            line = Line(p1, p2)
            for obstacle in self.obstacles:
                for i in range(len(obstacle)):
                    j = (i + 1) % len(obstacle)
                    obstacle_edge = Line(obstacle[i], obstacle[j])
                    if self.line_segment_intersection(line, obstacle_edge):
                        return True
            return False
        
        def reconstruct_path(self, came_from, current):
            """重构路径"""
            path = [current]
            while current in came_from:
                current = came_from[current]
                path.append(current)
            return path[::-1]
    
    class CollisionDetector:
        """碰撞检测器"""
        def __init__(self):
            self.objects = []
        
        def add_object(self, polygon):
            """添加对象"""
            self.objects.append(polygon)
        
        def check_collision(self, obj1, obj2):
            """检查碰撞"""
            # 使用分离轴定理
            for i in range(len(obj1)):
                j = (i + 1) % len(obj1)
                edge = obj1[j] - obj1[i]
                normal = Point(-edge.y, edge.x)
                normal = normal * (1 / normal.distance(Point(0, 0)))
                
                if not self.separating_axis(obj1, obj2, normal):
                    return True
            
            for i in range(len(obj2)):
                j = (i + 1) % len(obj2)
                edge = obj2[j] - obj2[i]
                normal = Point(-edge.y, edge.x)
                normal = normal * (1 / normal.distance(Point(0, 0)))
                
                if not self.separating_axis(obj1, obj2, normal):
                    return True
            
            return False
        
        def separating_axis(self, obj1, obj2, normal):
            """分离轴测试"""
            min1 = max1 = obj1[0].dot(normal)
            for point in obj1[1:]:
                proj = point.dot(normal)
                min1 = min(min1, proj)
                max1 = max(max1, proj)
            
            min2 = max2 = obj2[0].dot(normal)
            for point in obj2[1:]:
                proj = point.dot(normal)
                min2 = min(min2, proj)
                max2 = max(max2, proj)
            
            return max1 < min2 or max2 < min1
    
    return PathPlanner, CollisionDetector
```

### 3. 地理信息系统
```python
def gis_applications():
    """地理信息系统应用"""
    
    class SpatialIndex:
        """空间索引"""
        def __init__(self, bounds):
            self.bounds = bounds
            self.objects = []
        
        def insert(self, obj):
            """插入对象"""
            self.objects.append(obj)
        
        def query(self, bounds):
            """查询"""
            result = []
            for obj in self.objects:
                if self.bounds_intersect(obj.bounds, bounds):
                    result.append(obj)
            return result
        
        def bounds_intersect(self, bounds1, bounds2):
            """边界相交"""
            return not (bounds1.max_x < bounds2.min_x or bounds2.max_x < bounds1.min_x or
                       bounds1.max_y < bounds2.min_y or bounds2.max_y < bounds1.min_y)
    
    class Geocoder:
        """地理编码器"""
        def __init__(self):
            self.locations = {}
        
        def add_location(self, name, point):
            """添加位置"""
            self.locations[name] = point
        
        def geocode(self, name):
            """地理编码"""
            return self.locations.get(name)
        
        def reverse_geocode(self, point):
            """反向地理编码"""
            min_dist = float('inf')
            closest_name = None
            
            for name, location in self.locations.items():
                dist = point.distance(location)
                if dist < min_dist:
                    min_dist = dist
                    closest_name = name
            
            return closest_name
    
    class RouteOptimizer:
        """路径优化器"""
        def __init__(self):
            self.waypoints = []
        
        def add_waypoint(self, point):
            """添加路径点"""
            self.waypoints.append(point)
        
        def optimize_route(self):
            """优化路径"""
            if len(self.waypoints) < 2:
                return self.waypoints
            
            # 使用最近邻算法
            route = [self.waypoints[0]]
            remaining = self.waypoints[1:]
            
            while remaining:
                current = route[-1]
                closest = min(remaining, key=lambda p: current.distance(p))
                route.append(closest)
                remaining.remove(closest)
            
            return route
    
    return SpatialIndex, Geocoder, RouteOptimizer
```
