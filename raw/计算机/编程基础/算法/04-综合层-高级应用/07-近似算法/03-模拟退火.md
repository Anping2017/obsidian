# 模拟退火

## 概述

模拟退火（Simulated Annealing, SA）是一种基于物理退火过程的优化算法，通过模拟金属退火过程中的原子运动来寻找问题的最优解。模拟退火算法能够跳出局部最优，在解空间中全局搜索，特别适用于复杂的优化问题。

## 核心概念

### 基本思想
- **温度控制**：通过温度控制搜索过程
- **概率接受**：以一定概率接受更差的解
- **冷却策略**：逐渐降低温度
- **全局搜索**：避免陷入局部最优

### 关键特性
- **全局性**：能够跳出局部最优
- **概率性**：基于概率的搜索策略
- **适应性**：适应不同问题类型
- **鲁棒性**：对初始解不敏感

### 基本模拟退火算法

```python
from typing import List, Tuple, Any, Callable, Optional
import random
import math
import time

class SimulatedAnnealing:
    """基本模拟退火算法"""
    
    def __init__(self, initial_temperature: float = 100.0,
                 final_temperature: float = 0.01,
                 cooling_rate: float = 0.95,
                 max_iterations: int = 1000):
        self.initial_temperature = initial_temperature
        self.final_temperature = final_temperature
        self.cooling_rate = cooling_rate
        self.max_iterations = max_iterations
        
        self.current_solution = []
        self.best_solution = []
        self.current_value = float('inf')
        self.best_value = float('inf')
        self.temperature_history = []
        self.value_history = []
    
    def solve(self, initial_solution: List[Any], objective_function: Callable,
              neighbor_function: Callable, bounds: Optional[List[Tuple[float, float]]] = None) -> Tuple[List[Any], float]:
        """求解优化问题"""
        # 初始化
        self.current_solution = initial_solution.copy()
        self.best_solution = initial_solution.copy()
        self.current_value = objective_function(initial_solution)
        self.best_value = self.current_value
        
        temperature = self.initial_temperature
        iteration = 0
        
        # 退火过程
        while temperature > self.final_temperature and iteration < self.max_iterations:
            # 生成邻域解
            neighbor = neighbor_function(self.current_solution)
            
            # 计算目标函数值
            neighbor_value = objective_function(neighbor)
            
            # 接受准则
            if self._accept_solution(neighbor_value, temperature):
                self.current_solution = neighbor
                self.current_value = neighbor_value
                
                # 更新最佳解
                if neighbor_value < self.best_value:
                    self.best_solution = neighbor.copy()
                    self.best_value = neighbor_value
            
            # 降低温度
            temperature *= self.cooling_rate
            iteration += 1
            
            # 记录历史
            self.temperature_history.append(temperature)
            self.value_history.append(self.current_value)
        
        return self.best_solution, self.best_value
    
    def _accept_solution(self, neighbor_value: float, temperature: float) -> bool:
        """接受准则"""
        # 如果新解更好，直接接受
        if neighbor_value < self.current_value:
            return True
        
        # 如果新解更差，以一定概率接受
        delta = neighbor_value - self.current_value
        probability = math.exp(-delta / temperature)
        return random.random() < probability
    
    def get_history(self) -> Tuple[List[float], List[float]]:
        """获取历史记录"""
        return self.temperature_history, self.value_history
```

```python
from typing import List, Tuple, Any, Callable, Optional
import random
import math
import time

class AdvancedSimulatedAnnealing:
    """高级模拟退火算法"""
    
    def __init__(self, initial_temperature: float = 100.0,
                 final_temperature: float = 0.01,
                 cooling_schedule: str = 'exponential',
                 max_iterations: int = 1000,
                 max_iterations_per_temperature: int = 10):
        self.initial_temperature = initial_temperature
        self.final_temperature = final_temperature
        self.cooling_schedule = cooling_schedule
        self.max_iterations = max_iterations
        self.max_iterations_per_temperature = max_iterations_per_temperature
        
        self.current_solution = []
        self.best_solution = []
        self.current_value = float('inf')
        self.best_value = float('inf')
        self.temperature_history = []
        self.value_history = []
        self.acceptance_rate_history = []
    
    def solve(self, initial_solution: List[Any], objective_function: Callable,
              neighbor_function: Callable, bounds: Optional[List[Tuple[float, float]]] = None) -> Tuple[List[Any], float]:
        """求解优化问题"""
        # 初始化
        self.current_solution = initial_solution.copy()
        self.best_solution = initial_solution.copy()
        self.current_value = objective_function(initial_solution)
        self.best_value = self.current_value
        
        temperature = self.initial_temperature
        iteration = 0
        
        # 退火过程
        while temperature > self.final_temperature and iteration < self.max_iterations:
            # 在当前温度下进行多次迭代
            accepted_count = 0
            total_count = 0
            
            for _ in range(self.max_iterations_per_temperature):
                # 生成邻域解
                neighbor = neighbor_function(self.current_solution)
                
                # 边界处理
                if bounds:
                    neighbor = self._apply_bounds(neighbor, bounds)
                
                # 计算目标函数值
                neighbor_value = objective_function(neighbor)
                
                # 接受准则
                if self._accept_solution(neighbor_value, temperature):
                    self.current_solution = neighbor
                    self.current_value = neighbor_value
                    accepted_count += 1
                    
                    # 更新最佳解
                    if neighbor_value < self.best_value:
                        self.best_solution = neighbor.copy()
                        self.best_value = neighbor_value
                
                total_count += 1
                iteration += 1
            
            # 计算接受率
            acceptance_rate = accepted_count / total_count if total_count > 0 else 0
            
            # 自适应温度调整
            temperature = self._adaptive_temperature_adjustment(temperature, acceptance_rate)
            
            # 记录历史
            self.temperature_history.append(temperature)
            self.value_history.append(self.current_value)
            self.acceptance_rate_history.append(acceptance_rate)
        
        return self.best_solution, self.best_value
    
    def _accept_solution(self, neighbor_value: float, temperature: float) -> bool:
        """接受准则"""
        # 如果新解更好，直接接受
        if neighbor_value < self.current_value:
            return True
        
        # 如果新解更差，以一定概率接受
        delta = neighbor_value - self.current_value
        probability = math.exp(-delta / temperature)
        return random.random() < probability
    
    def _apply_bounds(self, solution: List[float], bounds: List[Tuple[float, float]]) -> List[float]:
        """应用边界约束"""
        bounded_solution = solution.copy()
        for i in range(len(bounded_solution)):
            if i < len(bounds):
                lower, upper = bounds[i]
                bounded_solution[i] = max(lower, min(upper, bounded_solution[i]))
        return bounded_solution
    
    def _adaptive_temperature_adjustment(self, temperature: float, acceptance_rate: float) -> float:
        """自适应温度调整"""
        if self.cooling_schedule == 'exponential':
            return temperature * 0.95
        elif self.cooling_schedule == 'linear':
            return temperature - (self.initial_temperature - self.final_temperature) / self.max_iterations
        elif self.cooling_schedule == 'logarithmic':
            return self.initial_temperature / math.log(1 + self.max_iterations - len(self.temperature_history))
        elif self.cooling_schedule == 'adaptive':
            # 基于接受率自适应调整
            if acceptance_rate > 0.5:
                return temperature * 0.9  # 接受率太高，快速降温
            elif acceptance_rate < 0.1:
                return temperature * 0.99  # 接受率太低，慢速降温
            else:
                return temperature * 0.95  # 正常降温
        else:
            return temperature * 0.95
    
    def get_history(self) -> Tuple[List[float], List[float], List[float]]:
        """获取历史记录"""
        return self.temperature_history, self.value_history, self.acceptance_rate_history
```

```python
from typing import List, Tuple, Any, Callable, Optional
import random
import math
import time
import threading
from concurrent.futures import ThreadPoolExecutor

class ParallelSimulatedAnnealing:
    """并行模拟退火算法"""
    
    def __init__(self, num_threads: int = 4, initial_temperature: float = 100.0,
                 final_temperature: float = 0.01, cooling_rate: float = 0.95,
                 max_iterations: int = 1000):
        self.num_threads = num_threads
        self.initial_temperature = initial_temperature
        self.final_temperature = final_temperature
        self.cooling_rate = cooling_rate
        self.max_iterations = max_iterations
        
        self.best_solution = []
        self.best_value = float('inf')
        self.lock = threading.Lock()
    
    def solve(self, initial_solution: List[Any], objective_function: Callable,
              neighbor_function: Callable, bounds: Optional[List[Tuple[float, float]]] = None) -> Tuple[List[Any], float]:
        """并行求解优化问题"""
        # 创建多个初始解
        initial_solutions = []
        for i in range(self.num_threads):
            if bounds:
                solution = self._generate_random_solution(bounds)
            else:
                solution = initial_solution.copy()
                for j in range(len(solution)):
                    solution[j] += random.gauss(0, 0.1)
            initial_solutions.append(solution)
        
        # 并行执行模拟退火
        with ThreadPoolExecutor(max_workers=self.num_threads) as executor:
            futures = []
            for i in range(self.num_threads):
                future = executor.submit(
                    self._simulated_annealing_thread,
                    initial_solutions[i],
                    objective_function,
                    neighbor_function,
                    bounds
                )
                futures.append(future)
            
            # 收集结果
            for future in futures:
                solution, value = future.result()
                with self.lock:
                    if value < self.best_value:
                        self.best_solution = solution.copy()
                        self.best_value = value
        
        return self.best_solution, self.best_value
    
    def _simulated_annealing_thread(self, initial_solution: List[Any],
                                   objective_function: Callable,
                                   neighbor_function: Callable,
                                   bounds: Optional[List[Tuple[float, float]]]) -> Tuple[List[Any], float]:
        """模拟退火线程"""

        current_value = objective_function(current_solution)
        best_solution = current_solution.copy()
        best_value = current_value

                neighbor = self._apply_bounds(neighbor, bounds)
            
            # 计算目标函数值

            # 接受准则

                current_solution = neighbor
                current_value = neighbor_value
                
                # 更新最佳解
                if neighbor_value < best_value:
                    best_solution = neighbor.copy()
                    best_value = neighbor_value
            
            # 降低温度
            temperature *= self.cooling_rate
            iteration += 1
        
        return best_solution, best_value
    
    def _accept_solution(self, neighbor_value: float, current_value: float, temperature: float) -> bool:
        """接受准则"""
        if neighbor_value < current_value:
            return True
        
        delta = neighbor_value - current_value
        probability = math.exp(-delta / temperature)
        return random.random() < probability
    
    def _apply_bounds(self, solution: List[float], bounds: List[Tuple[float, float]]) -> List[float]:
        """应用边界约束"""
        bounded_solution = solution.copy()
        for i in range(len(bounded_solution)):
            if i < len(bounds):
                lower, upper = bounds[i]
                bounded_solution[i] = max(lower, min(upper, bounded_solution[i]))
        return bounded_solution
    
    def _generate_random_solution(self, bounds: List[Tuple[float, float]]) -> List[float]:
        """生成随机解"""
        solution = []
        for lower, upper in bounds:
            solution.append(random.uniform(lower, upper))
```

```python
        current_solution = self.current_solution.copy()
        current_value = self.current_value
        
        for _ in range(self.local_search_iterations):
            neighbor = neighbor_function(current_solution)
            
            if bounds:
                neighbor = self._apply_bounds(neighbor, bounds)
            
            neighbor_value = objective_function(neighbor)
            
            if neighbor_value < current_value:
                current_solution = neighbor
                current_value = neighbor_value
                
                if neighbor_value < self.best_value:
                    self.best_solution = neighbor.copy()
                    self.best_value = neighbor_value
        
        self.current_solution = current_solution
        self.current_value = current_value
    
    def get_history(self) -> Tuple[List[float], List[float]]:
        """获取历史记录"""
        return self.temperature_history, self.value_history
```

## 性能分析

### 时间复杂度
- **基本模拟退火**：O(max_iterations)
- **并行模拟退火**：O(max_iterations / num_threads)
- **混合模拟退火**：O(max_iterations * local_search_iterations)

### 空间复杂度
- **基本模拟退火**：O(solution_length)
- **并行模拟退火**：O(num_threads * solution_length)
- **混合模拟退火**：O(solution_length)
