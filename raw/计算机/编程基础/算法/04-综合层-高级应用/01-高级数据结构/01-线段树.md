# 线段树

## 🎯 核心概念

**线段树(Segment Tree)**是一种二叉树数据结构，用于处理区间查询和区间更新问题。线段树将区间递归地分割成更小的子区间，每个节点代表一个区间，支持高效的区间查询和更新操作。

### 2. 线段树构建
```python
def segment_tree_construction():
    """线段树构建"""
    
    class SegmentTree:
        """线段树"""
        def __init__(self, arr):
            self.n = len(arr)
            self.tree = [0] * (4 * self.n)
            self.lazy = [0] * (4 * self.n)
            self.build(arr, 1, 0, self.n - 1)
        
        def build(self, arr, node, start, end):
            """构建线段树"""
            if start == end:
                self.tree[node] = arr[start]
            else:
                mid = (start + end) // 2
                self.build(arr, 2 * node, start, mid)
                self.build(arr, 2 * node + 1, mid + 1, end)
                self.tree[node] = self.tree[2 * node] + self.tree[2 * node + 1]
        
        def query(self, node, start, end, l, r):
            """区间查询"""
            if r < start or end < l:
                return 0
            
            if l <= start and end <= r:
                return self.tree[node]
            
            mid = (start + end) // 2
            left = self.query(2 * node, start, mid, l, r)
            right = self.query(2 * node + 1, mid + 1, end, l, r)
            return left + right
        
        def update(self, node, start, end, idx, val):
            """单点更新"""
            if start == end:
                self.tree[node] = val
            else:
                mid = (start + end) // 2
                if idx <= mid:
                    self.update(2 * node, start, mid, idx, val)
                else:
                    self.update(2 * node + 1, mid + 1, end, idx, val)
                self.tree[node] = self.tree[2 * node] + self.tree[2 * node + 1]
        
        def range_update(self, node, start, end, l, r, val):
            """区间更新"""
            if self.lazy[node] != 0:
                self.tree[node] += (end - start + 1) * self.lazy[node]
                if start != end:
                    self.lazy[2 * node] += self.lazy[node]
                    self.lazy[2 * node + 1] += self.lazy[node]
                self.lazy[node] = 0
            
            if r < start or end < l:
                return
            
            if l <= start and end <= r:
                self.tree[node] += (end - start + 1) * val
                if start != end:
                    self.lazy[2 * node] += val
                    self.lazy[2 * node + 1] += val
                return
            
            mid = (start + end) // 2
            self.range_update(2 * node, start, mid, l, r, val)
            self.range_update(2 * node + 1, mid + 1, end, l, r, val)
            self.tree[node] = self.tree[2 * node] + self.tree[2 * node + 1]
    
    return SegmentTree
```

```python
def dynamic_programming_applications():
    """动态规划应用"""
    
    def longest_increasing_subsequence(arr):
        """最长递增子序列"""
        n = len(arr)
        
        # 离散化
        sorted_arr = sorted(set(arr))
        mapping = {val: i for i, val in enumerate(sorted_arr)}
        
        # 线段树
        st = SegmentTree([0] * len(sorted_arr))
        
        result = 0
        for val in arr:
            idx = mapping[val]
            # 查询小于当前值的最大长度
            max_len = st.query(1, 0, len(sorted_arr) - 1, 0, idx - 1)
            # 更新当前值
            st.update(1, 0, len(sorted_arr) - 1, idx, max_len + 1)
            result = max(result, max_len + 1)
        
        return result
    
    def range_sum_query_2d(matrix, queries):
        """二维区间和查询"""
        class SegmentTree2D:
            def __init__(self, matrix):
                self.m = len(matrix)
                self.n = len(matrix[0])
                self.tree = [[0] * (4 * self.n) for _ in range(4 * self.m)]
                self.build(matrix, 1, 0, self.m - 1, 1, 0, self.n - 1)
            
            def build(self, matrix, node_x, start_x, end_x, node_y, start_y, end_y):
                if start_x == end_x and start_y == end_y:
                    self.tree[node_x][node_y] = matrix[start_x][start_y]
                elif start_x == end_x:
                    mid_y = (start_y + end_y) // 2
                    self.build(matrix, node_x, start_x, end_x, 2 * node_y, start_y, mid_y)
                    self.build(matrix, node_x, start_x, end_x, 2 * node_y + 1, mid_y + 1, end_y)
                    self.tree[node_x][node_y] = self.tree[node_x][2 * node_y] + self.tree[node_x][2 * node_y + 1]
                elif start_y == end_y:
                    mid_x = (start_x + end_x) // 2
                    self.build(matrix, mid_x + 1, end_x, node_y, start_y, end_y)
                    self.build(matrix, mid_x + 1, end_x, node_y, start_y, end_y)
                    self.tree[node_x][node_y] = self.tree[2 * node_x][node_y] + self.tree[2 * node_x + 1][node_y]
                else:
                    mid_x = (start_x + end_x) // 2
                    mid_y = (start_y + end_y) // 2
                    self.build(matrix, 2 * node_x, start_x, mid_x, 2 * node_y, start_y, mid_y)
                    self.build(matrix, 2 * node_x, start_x, mid_x, 2 * node_y + 1, mid_y + 1, end_y)
                    self.build(matrix, 2 * node_x + 1, mid_x + 1, end_x, 2 * node_y, start_y, mid_y)
                    self.build(matrix, 2 * node_x + 1, mid_x + 1, end_x, 2 * node_y + 1, mid_y + 1, end_y)
                    self.tree[node_x][node_y] = (self.tree[2 * node_x][2 * node_y] + 
                                               self.tree[2 * node_x][2 * node_y + 1] + 
                                               self.tree[2 * node_x + 1][2 * node_y] + 
                                               self.tree[2 * node_x + 1][2 * node_y + 1])
            
            def query(self, node_x, start_x, end_x, node_y, start_y, end_y, lx, rx, ly, ry):
                if rx < start_x or end_x < lx or ry < start_y or end_y < ly:
                    return 0
                
                if lx <= start_x and end_x <= rx and ly <= start_y and end_y <= ry:
                    return self.tree[node_x][node_y]
                
                mid_x = (start_x + end_x) // 2
                mid_y = (start_y + end_y) // 2
                
                result = 0
                result += self.query(2 * node_x, start_x, mid_x, 2 * node_y, start_y, mid_y, lx, rx, ly, ry)
                result += self.query(2 * node_x, start_x, mid_x, 2 * node_y + 1, mid_y + 1, end_y, lx, rx, ly, ry)

                result += self.query(2 * node_x + 1, mid_x + 1, end_x, 2 * node_y + 1, mid_y + 1, end_y, lx, rx, ly, ry)
                
                return result
        
        st = SegmentTree2D(matrix)
```
