```python
def database_system_applications():
    """数据库系统应用"""
    
    class DatabaseSystem:
        """数据库系统"""
        def __init__(self):
            self.indexes = {}  # 索引表
            self.tables = {}   # 数据表
        
        def create_index(self, table_name, column_name, index_type="BTree"):
            """创建索引"""
            if index_type == "BTree":
                self.indexes[f"{table_name}.{column_name}"] = BTreeIndex()
            elif index_type == "Hash":
                self.indexes[f"{table_name}.{column_name}"] = HashIndex()
            elif index_type == "Bitmap":
                self.indexes[f"{table_name}.{column_name}"] = BitmapIndex()
        
        def insert_record(self, table_name, record):
            """插入记录"""
            if table_name not in self.tables:
                self.tables[table_name] = []
            
            self.tables[table_name].append(record)
            
            # 更新索引
            for index_key, index in self.indexes.items():
                if index_key.startswith(f"{table_name}."):
                    column_name = index_key.split(".")[1]
                    if column_name in record:
                        index.insert(record[column_name], len(self.tables[table_name]) - 1)
        
        def query_range(self, table_name, column_name, low, high):
            """范围查询"""
            index_key = f"{table_name}.{column_name}"
            if index_key in self.indexes:
                return self.indexes[index_key].range_query(low, high)
            else:
                # 全表扫描
                result = []
                for i, record in enumerate(self.tables[table_name]):
                    if low <= record[column_name] <= high:
                        result.append(i)
                return result
        
        def get_statistics(self):
            """获取统计信息"""
            stats = {
                "tables": len(self.tables),
                "indexes": len(self.indexes),
                "total_records": sum(len(table) for table in self.tables.values())
            }
            return stats
    
    class BTreeIndex:
        """B树索引"""
        def __init__(self, order=3):
            self.order = order
            self.root = None
        
        def insert(self, key, value):
            """插入键值对"""
            if self.root is None:
                self.root = BTreeNode([key], [value], True)
            else:
                if len(self.root.keys) == 2 * self.order - 1:
                    new_root = BTreeNode([], [], False)
                    new_root.children.append(self.root)
                    self.split_child(new_root, 0)
                    self.root = new_root
                
                self.insert_non_full(self.root, key, value)
        
        def range_query(self, low, high):
            """范围查询"""
            result = []
            self.range_query_helper(self.root, low, high, result)
            return result
        
        def range_query_helper(self, node, low, high, result):
            """范围查询辅助函数"""
            if node is None:
                return
            
            i = 0
            while i < len(node.keys) and node.keys[i] < low:
                i += 1
            
            if not node.is_leaf:
                if i < len(node.children):
                    self.range_query_helper(node.children[i], low, high, result)
            
            while i < len(node.keys) and node.keys[i] <= high:
                result.append(node.values[i])
                i += 1
            
            if not node.is_leaf and i < len(node.children):
                self.range_query_helper(node.children[i], low, high, result)
    
    class HashIndex:
        """哈希索引"""
        def __init__(self):
            self.table = {}
        
        def insert(self, key, value):
            """插入键值对"""
            if key not in self.table:
                self.table[key] = []
            self.table[key].append(value)
        
        def range_query(self, low, high):
            """范围查询"""
            result = []
            for key in self.table:
                if low <= key <= high:
                    result.extend(self.table[key])
            return result
    
    class BitmapIndex:
        """位图索引"""
        def __init__(self):
            self.bitmaps = {}
            self.record_count = 0
        
        def insert(self, key, value):
            """插入键值对"""
            if key not in self.bitmaps:
                self.bitmaps[key] = [False] * (self.record_count + 1)
            
            # 扩展位图
            while len(self.bitmaps[key]) <= value:
                for k in self.bitmaps:
                    self.bitmaps[k].extend([False] * (len(self.bitmaps[k]) + 1))
            
            self.bitmaps[key][value] = True
            self.record_count = max(self.record_count, value + 1)
        
        def range_query(self, low, high):
            """范围查询"""
            result = []
            for key in self.bitmaps:
                if low <= key <= high:
                    for i, bit in enumerate(self.bitmaps[key]):
                        if bit:
                            result.append(i)
            return result
    
    return DatabaseSystem, BTreeIndex, HashIndex, BitmapIndex
```

```python
def operating_system_applications():
    """操作系统应用"""
    
    class MemoryManager:
        """内存管理器"""
        def __init__(self, total_size):
            self.total_size = total_size
            self.free_tree = RedBlackTree()
            self.allocated_tree = RedBlackTree()
            
            # 初始化空闲内存
            self.free_tree.insert(0, total_size)
        
        def allocate(self, size):
            """分配内存"""
            # 找到第一个足够大的空闲块
            free_blocks = self.free_tree.range_query(0, self.total_size)
            
            for start, block_size in free_blocks:
                if block_size >= size:
                    # 分配内存
                    self.free_tree.delete(start)
                    
                    if block_size > size:
                        # 剩余部分放回空闲树
                        self.free_tree.insert(start + size, block_size - size)
                    
                    # 记录已分配的内存
                    self.allocated_tree.insert(start, size)
                    return start
            
            return None  # 内存不足
        
        def deallocate(self, start):
            """释放内存"""
            node = self.allocated_tree.search(start)
            if node == self.allocated_tree.nil:
                return False
            
            size = node.value
            self.allocated_tree.delete(start)
            
            # 合并相邻的空闲块
            self.merge_free_blocks(start, size)
            
            return True
        
        def merge_free_blocks(self, start, size):
            """合并相邻的空闲块"""
            # 检查前一个块
            prev_blocks = self.free_tree.range_query(0, start - 1)
            if prev_blocks:
                prev_start, prev_size = prev_blocks[-1]
                if prev_start + prev_size == start:
                    # 合并前一个块
                    self.free_tree.delete(prev_start)
                    start = prev_start
                    size += prev_size
            
            # 检查后一个块
            next_blocks = self.free_tree.range_query(start + size, self.total_size)
            if next_blocks:
                next_start, next_size = next_blocks[0]
                if start + size == next_start:
                    # 合并后一个块
                    self.free_tree.delete(next_start)
                    size += next_size
            
            # 插入合并后的块
            self.free_tree.insert(start, size)
        
        def get_memory_status(self):
            """获取内存状态"""
            free_blocks = self.free_tree.range_query(0, self.total_size)
            allocated_blocks = self.allocated_tree.range_query(0, self.total_size)
            
            total_free = sum(size for _, size in free_blocks)
            total_allocated = sum(size for _, size in allocated_blocks)
            
            return {
                "total_size": self.total_size,
                "total_free": total_free,
                "total_allocated": total_allocated,
                "free_blocks": free_blocks,
                "allocated_blocks": allocated_blocks
            }
    
    class ProcessScheduler:
        """进程调度器"""
        def __init__(self):
            self.ready_queue = []  # 就绪队列
            self.blocked_queue = []  # 阻塞队列
            self.running_process = None  # 运行进程
        
        def add_process(self, process):
            """添加进程"""
            self.ready_queue.append(process)
            self.ready_queue.sort(key=lambda p: p.priority, reverse=True)
        
        def schedule(self):
            """调度进程"""
            if self.running_process is None and self.ready_queue:
                self.running_process = self.ready_queue.pop(0)
                return self.running_process
            return None
        
        def block_process(self, process):
            """阻塞进程"""
            if self.running_process == process:
                self.running_process = None
                self.blocked_queue.append(process)
            else:
                self.ready_queue.remove(process)
                self.blocked_queue.append(process)
        
        def unblock_process(self, process):
            """解除阻塞进程"""
            if process in self.blocked_queue:
                self.blocked_queue.remove(process)
                self.ready_queue.append(process)
                self.ready_queue.sort(key=lambda p: p.priority, reverse=True)
        
        def get_queue_status(self):
            """获取队列状态"""
            return {
                "ready_count": len(self.ready_queue),
                "blocked_count": len(self.blocked_queue),
                "running": self.running_process is not None
            }
    
    class FileSystem:
        """文件系统"""
        def __init__(self):
            self.directory_tree = {}  # 目录树
            self.file_table = {}      # 文件表
            self.free_blocks = set()   # 空闲块
        
        def create_file(self, path, size):
            """创建文件"""
            # 分配块
            blocks = self.allocate_blocks(size)
            if not blocks:
                return False
            
            # 创建文件表项
            self.file_table[path] = {
                "size": size,
                "blocks": blocks,
                "created": time.time()
            }
            
            return True
        
        def delete_file(self, path):
            """删除文件"""
            if path in self.file_table:
                # 释放块
                self.free_blocks.update(self.file_table[path]["blocks"])
                del self.file_table[path]
                return True
            return False
        
        def allocate_blocks(self, size):
            """分配块"""
            blocks_needed = (size + 4095) // 4096  # 4KB块
            blocks = []
            
            for block in self.free_blocks:
                if len(blocks) < blocks_needed:
                    blocks.append(block)
                else:
                    break
            
            if len(blocks) == blocks_needed:
                self.free_blocks -= set(blocks)
                return blocks
            
            return []
        
        def get_file_system_status(self):
            """获取文件系统状态"""
            total_files = len(self.file_table)
            total_size = sum(file_info["size"] for file_info in self.file_table.values())
            free_blocks = len(self.free_blocks)
            
            return {
                "total_files": total_files,
                "total_size": total_size,
                "free_blocks": free_blocks
            }
    
    return MemoryManager, ProcessScheduler, FileSystem
```

```python
def compiler_applications():
    """编译器应用"""
    
    class SymbolTable:
        """符号表"""
        def __init__(self):
            self.table = RedBlackTree()
            self.scope_stack = []  # 作用域栈
        
        def enter_scope(self):
            """进入作用域"""
            self.scope_stack.append({})
        
        def exit_scope(self):
            """退出作用域"""
            if self.scope_stack:
                self.scope_stack.pop()
        
        def insert_symbol(self, name, symbol_info):
            """插入符号"""
            if self.scope_stack:
                self.scope_stack[-1][name] = symbol_info
        
        def lookup_symbol(self, name):
            """查找符号"""
            # 从当前作用域开始向上查找
            for scope in reversed(self.scope_stack):
                if name in scope:
                    return scope[name]
            return None
        
        def get_all_symbols(self):
            """获取所有符号"""
            all_symbols = {}
            for scope in self.scope_stack:
                all_symbols.update(scope)
            return all_symbols
    
    class SyntaxAnalyzer:
        """语法分析器"""
        def __init__(self):
            self.stack = []  # 分析栈
            self.grammar = {}  # 语法规则
        
        def parse(self, tokens):
            """解析词法单元"""
            self.stack = ["$", "S"]  # 初始状态
            i = 0
            
            while self.stack:
                top = self.stack[-1]
                current_token = tokens[i] if i < len(tokens) else "$"
                
                if top == current_token:
                    # 匹配成功
                    self.stack.pop()
                    i += 1
                elif top in self.grammar:
                    # 应用语法规则
                    rule = self.grammar[top].get(current_token)
                    if rule:
                        self.stack.pop()
                        for symbol in reversed(rule):
                            if symbol != "ε":  # 空符号
                                self.stack.append(symbol)
                    else:
                        return False  # 语法错误
                else:
                    return False  # 语法错误
            
            return i == len(tokens)
        
        def add_rule(self, non_terminal, terminal, production):
            """添加语法规则"""
            if non_terminal not in self.grammar:
                self.grammar[non_terminal] = {}
            self.grammar[non_terminal][terminal] = production
    
    class CodeOptimizer:
        """代码优化器"""
        def __init__(self):
            self.optimizations = []
        
        def add_optimization(self, optimization):
            """添加优化"""
            self.optimizations.append(optimization)
        
        def optimize(self, code):
            """优化代码"""
            optimized_code = code
            
            for optimization in self.optimizations:
                optimized_code = optimization.apply(optimized_code)
            
            return optimized_code
        
        def get_optimization_stats(self):
            """获取优化统计"""
            return {
                "total_optimizations": len(self.optimizations),
                "applied_optimizations": sum(1 for opt in self.optimizations if opt.applied)
            }
    
    class Optimization:
        """优化"""
        def __init__(self, name):
            self.name = name
            self.applied = False
        
        def apply(self, code):
            """应用优化"""
            # 这里应该实现具体的优化逻辑
            self.applied = True
            return code
    
    return SymbolTable, SyntaxAnalyzer, CodeOptimizer, Optimization
```
