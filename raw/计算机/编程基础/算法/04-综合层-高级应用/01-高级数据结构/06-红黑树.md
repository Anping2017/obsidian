# 红黑树

## 🎯 核心概念

**红黑树(Red-Black Tree)**是一种自平衡二叉搜索树，通过颜色标记和旋转操作来维护平衡性。红黑树在保证O(log n)时间复杂度的同时，具有相对简单的实现和良好的性能，广泛应用于C++ STL的map和set、Java的TreeMap和TreeSet等。

### 3. 红黑树构建
```python
def red_black_tree_construction():
    """红黑树构建"""
    
    class RedBlackTree:
        """红黑树"""
        def __init__(self):
            self.nil = RedBlackNode(None, 'BLACK')  # 哨兵节点
            self.root = self.nil
        
        def left_rotate(self, x):
            """左旋"""
            y = x.right
            x.right = y.left
            
            if y.left != self.nil:
                y.left.parent = x
            
            y.parent = x.parent
            
            if x.parent == self.nil:
                self.root = y
            elif x == x.parent.left:
                x.parent.left = y
            else:
                x.parent.right = y
            
            y.left = x
            x.parent = y
        
        def right_rotate(self, y):
            """右旋"""
            x = y.left
            y.left = x.right
            
            if x.right != self.nil:
                x.right.parent = y
            
            x.parent = y.parent
            
            if y.parent == self.nil:
                self.root = x
            elif y == y.parent.left:
                y.parent.left = x
            else:
                y.parent.right = x
            
            x.right = y
            y.parent = x
        
        def insert(self, key):
            """插入节点"""
            z = RedBlackNode(key)
            y = self.nil
            x = self.root
            
            # 找到插入位置
            while x != self.nil:
                y = x
                if z.key < x.key:
                    x = x.left
                else:
                    x = x.right
            
            z.parent = y
            
            if y == self.nil:
                self.root = z
            elif z.key < y.key:
                y.left = z
            else:
                y.right = z
            
            z.left = self.nil
            z.right = self.nil
            z.color = 'RED'
            
            self.insert_fixup(z)
        
        def insert_fixup(self, z):
            """插入修复"""
            while z.parent.color == 'RED':
                if z.parent == z.parent.parent.left:
                    y = z.parent.parent.right
                    if y.color == 'RED':
                        # Case 1: 叔叔节点是红色
                        z.parent.color = 'BLACK'
                        y.color = 'BLACK'
                        z.parent.parent.color = 'RED'
                        z = z.parent.parent
                    else:
                        if z == z.parent.right:
                            # Case 2: 叔叔节点是黑色，z是右子节点
                            z = z.parent
                            self.left_rotate(z)
                        # Case 3: 叔叔节点是黑色，z是左子节点
                        z.parent.color = 'BLACK'
                        z.parent.parent.color = 'RED'
                        self.right_rotate(z.parent.parent)
                else:
                    # 对称情况
                    y = z.parent.parent.left
                    if y.color == 'RED':
                        z.parent.color = 'BLACK'
                        y.color = 'BLACK'
                        z.parent.parent.color = 'RED'
                        z = z.parent.parent
                    else:
                        if z == z.parent.left:
                            z = z.parent
                            self.right_rotate(z)
                        z.parent.color = 'BLACK'
                        z.parent.parent.color = 'RED'
                        self.left_rotate(z.parent.parent)
            
            self.root.color = 'BLACK'
        
        def delete(self, key):
            """删除节点"""
            z = self.search(key)
            if z == self.nil:
                return False
            
            y = z
            y_original_color = y.color
            
            if z.left == self.nil:
                x = z.right
                self.transplant(z, z.right)
            elif z.right == self.nil:
                x = z.left
                self.transplant(z, z.left)
            else:
                y = self.minimum(z.right)
                y_original_color = y.color
                x = y.right
                
                if y.parent == z:
                    x.parent = y
                else:
                    self.transplant(y, y.right)
                    y.right = z.right
                    y.right.parent = y
                
                self.transplant(z, y)
                y.left = z.left
                y.left.parent = y
                y.color = z.color
            
            if y_original_color == 'BLACK':
                self.delete_fixup(x)
            
            return True
        
        def delete_fixup(self, x):
            """删除修复"""
            while x != self.root and x.color == 'BLACK':
                if x == x.parent.left:
                    w = x.parent.right
                    if w.color == 'RED':
                        # Case 1: 兄弟节点是红色
                        w.color = 'BLACK'
                        x.parent.color = 'RED'
                        self.left_rotate(x.parent)
                        w = x.parent.right
                    
                    if w.left.color == 'BLACK' and w.right.color == 'BLACK':
                        # Case 2: 兄弟节点的两个子节点都是黑色
                        w.color = 'RED'
                        x = x.parent
                    else:
                        if w.right.color == 'BLACK':
                            # Case 3: 兄弟节点的右子节点是黑色
                            w.left.color = 'BLACK'
                            w.color = 'RED'
                            self.right_rotate(w)
                            w = x.parent.right
                        
                        # Case 4: 兄弟节点的右子节点是红色
                        w.color = x.parent.color
                        x.parent.color = 'BLACK'
                        w.right.color = 'BLACK'
                        self.left_rotate(x.parent)
                        x = self.root
                else:
                    # 对称情况
                    w = x.parent.left
                    if w.color == 'RED':
                        w.color = 'BLACK'
                        x.parent.color = 'RED'
                        self.right_rotate(x.parent)
                        w = x.parent.left
                    
                    if w.right.color == 'BLACK' and w.left.color == 'BLACK':
                        w.color = 'RED'
                        x = x.parent
                    else:
                        if w.left.color == 'BLACK':
                            w.right.color = 'BLACK'
                            w.color = 'RED'
                            self.left_rotate(w)
                            w = x.parent.left
                        
                        w.color = x.parent.color
                        x.parent.color = 'BLACK'
                        w.left.color = 'BLACK'
                        self.right_rotate(x.parent)
                        x = self.root
            
            x.color = 'BLACK'
        
        def transplant(self, u, v):
            """替换子树"""
            if u.parent == self.nil:
                self.root = v
            elif u == u.parent.left:
                u.parent.left = v
            else:
                u.parent.right = v
            
            v.parent = u.parent
        
        def search(self, key):
            """搜索节点"""
            x = self.root
            while x != self.nil and x.key != key:
                if key < x.key:
                    x = x.left
                else:
                    x = x.right
            return x
        
        def minimum(self, x):
            """找到最小节点"""
            while x.left != self.nil:
                x = x.left
            return x
        
        def maximum(self, x):
            """找到最大节点"""
            while x.right != self.nil:
                x = x.right
            return x
        
        def inorder_traversal(self):
            """中序遍历"""
            result = []
            
            def inorder(x):
                if x != self.nil:
                    inorder(x.left)
                    result.append((x.key, x.color))
                    inorder(x.right)
            
            inorder(self.root)
            return result
    
    return RedBlackTree

        def left_rotate(self, x):
            """左旋"""
            y = x.right
            x.right = y.left
            
            if y.left != self.nil:
                y.left.parent = x

            if x.parent == self.nil:

            elif x == x.parent.left:
                x.parent.left = y
            else:
                x.parent.right = y
            
            y.left = x
            x.parent = y
            
            # 更新子树大小
            y.size = x.size
            x.size = x.left.size + x.right.size + 1
        
        def right_rotate(self, y):
            """右旋"""
            x = y.left
            y.left = x.right
            
            if x.right != self.nil:
                x.right.parent = y
            
            x.parent = y.parent
            
            if y.parent == self.nil:
                self.root = x
            elif y == y.parent.left:
                y.parent.left = x
            else:
                y.parent.right = x
            
            x.right = y
            y.parent = x
            
            # 更新子树大小
            x.size = y.size
            y.size = y.left.size + y.right.size + 1
        
        def select(self, i):
            """选择第i小的元素"""
            return self.select_helper(self.root, i)
        
        def select_helper(self, x, i):
            """选择辅助函数"""
            r = x.left.size + 1
            if i == r:
                return x
            elif i < r:
                return self.select_helper(x.left, i)
            else:
                return self.select_helper(x.right, i - r)
        
        def rank(self, x):
            """计算节点x的排名"""
            r = x.left.size + 1
            y = x
            while y != self.root:
                if y == y.parent.right:
                    r += y.parent.left.size + 1
                y = y.parent
            return r
        
        def range_query(self, low, high):
            """范围查询"""
            result = []
            
            def range_query_helper(x):
                if x == self.nil:
                    return
                
                if x.key > low:
                    range_query_helper(x.left)
                
                if low <= x.key <= high:
                    result.append((x.key, x.value))
                
                if x.key < high:
                    range_query_helper(x.right)
            
            range_query_helper(self.root)
            return result
        
        def get_statistics(self):
            """获取统计信息"""
            stats = {
                "total_nodes": self.root.size,
                "height": self.get_height(),
                "black_height": self.get_black_height()
            }
            return stats
        
        def get_height(self):
            """获取树高"""
            def height_helper(x):
                if x == self.nil:
                    return 0
                return 1 + max(height_helper(x.left), height_helper(x.right))
            
            return height_helper(self.root)
        
        def get_black_height(self):
            """获取黑高"""
            def black_height_helper(x):
                if x == self.nil:
                    return 0
                
                left_black_height = black_height_helper(x.left)
                right_black_height = black_height_helper(x.right)
                
                if left_black_height != right_black_height:
                    raise ValueError("红黑树性质被违反")
                
                return left_black_height + (1 if x.color == 'BLACK' else 0)
            
            return black_height_helper(self.root)
    
    return AdvancedRedBlackTree

def ordered_set_applications():
    """有序集合应用"""
    
    class OrderedSet:
        """有序集合"""
        def __init__(self):
            self.tree = AdvancedRedBlackTree()
        
        def add(self, key):
            """添加元素"""
            if not self.contains(key):
                self.tree.insert(key)
        
        def remove(self, key):
            """删除元素"""
            node = self.tree.search(key)
            if node != self.tree.nil:
                self.tree.delete(key)
                return True
            return False
        
        def contains(self, key):
            """检查元素是否存在"""
            return self.tree.search(key) != self.tree.nil
        
        def size(self):
            """获取集合大小"""
            return self.tree.root.size
        
        def first(self):
            """获取第一个元素"""
            if self.tree.root == self.tree.nil:
                return None
            return self.tree.minimum(self.tree.root).key
        
        def last(self):
            """获取最后一个元素"""
            if self.tree.root == self.tree.nil:
                return None
            return self.tree.maximum(self.tree.root).key
        
        def floor(self, key):
            """获取小于等于key的最大元素"""
            node = self.tree.search(key)
            if node != self.tree.nil:
                return key
            
            # 找到小于key的最大元素
            x = self.tree.root
            result = None
            
            while x != self.tree.nil:
                if x.key < key:
                    result = x.key
                    x = x.right
                else:
                    x = x.left
            
            return result
        
        def ceiling(self, key):
            """获取大于等于key的最小元素"""
            node = self.tree.search(key)
            if node != self.tree.nil:
                return key
            
            # 找到大于key的最小元素
            x = self.tree.root
            result = None
            
            while x != self.tree.nil:
                if x.key > key:
                    result = x.key
                    x = x.left
                else:
                    x = x.right
            
            return result
        
        def range_query(self, low, high):
            """范围查询"""
            return self.tree.range_query(low, high)
        
        def to_list(self):
            """转换为列表"""
            result = []
            
            def inorder(x):
                if x != self.tree.nil:
                    inorder(x.left)
                    result.append(x.key)
                    inorder(x.right)
            
            inorder(self.tree.root)
            return result
    
    return OrderedSet

    class BTreeIndex:

            self.root = None

            """插入键值对"""
            if self.root is None:
                self.root = BTreeNode([key], [value], True)
            else:
                if len(self.root.keys) == 2 * self.order - 1:
                    new_root = BTreeNode([], [], False)
                    new_root.children.append(self.root)
                    self.split_child(new_root, 0)
                    self.root = new_root
                
                self.insert_non_full(self.root, key, value)
        
        def insert_non_full(self, node, key, value):
            """非满节点插入"""
            i = len(node.keys) - 1
            
            if node.is_leaf:
                node.keys.append(None)
                node.values.append(None)
                
                while i >= 0 and key < node.keys[i]:
                    node.keys[i + 1] = node.keys[i]
                    node.values[i + 1] = node.values[i]
                    i -= 1
                
                node.keys[i + 1] = key
                node.values[i + 1] = value
            else:
                while i >= 0 and key < node.keys[i]:
                    i -= 1
                i += 1
                
                if len(node.children[i].keys) == 2 * self.order - 1:
                    self.split_child(node, i)
                    if key > node.keys[i]:
                        i += 1
                
                self.insert_non_full(node.children[i], key, value)
        
        def split_child(self, parent, index):
            """分裂子节点"""
            order = self.order
            child = parent.children[index]
            new_child = BTreeNode([], [], child.is_leaf)
            
            # 移动键值对
            new_child.keys = child.keys[order:]
            new_child.values = child.values[order:]
            child.keys = child.keys[:order-1]
            child.values = child.values[:order-1]
            
            # 移动子节点
            if not child.is_leaf:
                new_child.children = child.children[order:]
                child.children = child.children[:order]
            
            # 更新父节点
            parent.children.insert(index + 1, new_child)
            parent.keys.insert(index, child.keys[order-1])
            parent.values.insert(index, child.values[order-1])
    
    class BTreeNode:
        """B树节点"""
        def __init__(self, keys, values, is_leaf):
            self.keys = keys
            self.values = values
            self.children = []
            self.is_leaf = is_leaf
    
    return DatabaseIndex, BTreeIndex
```

### 3. 内存管理
```python
def memory_management_applications():
    """内存管理应用"""
    
    class MemoryManager:
        """内存管理器"""
        def __init__(self, total_size):
            self.total_size = total_size
            self.free_tree = AdvancedRedBlackTree()
            self.allocated_tree = AdvancedRedBlackTree()
            
            # 初始化空闲内存
            self.free_tree.insert(0, total_size)
        
        def allocate(self, size):
            """分配内存"""
            # 找到第一个足够大的空闲块
            free_blocks = self.free_tree.range_query(0, self.total_size)
            
            for start, block_size in free_blocks:
                if block_size >= size:
                    # 分配内存
                    self.free_tree.delete(start)
                    
                    if block_size > size:
                        # 剩余部分放回空闲树
                        self.free_tree.insert(start + size, block_size - size)
                    
                    # 记录已分配的内存
                    self.allocated_tree.insert(start, size)
                    return start
            
            return None  # 内存不足
        
        def deallocate(self, start):
            """释放内存"""
            node = self.allocated_tree.search(start)
            if node == self.allocated_tree.nil:
                return False
            
            size = node.value
            self.allocated_tree.delete(start)
            
            # 合并相邻的空闲块
            self.merge_free_blocks(start, size)
            
            return True
        
        def merge_free_blocks(self, start, size):
            """合并相邻的空闲块"""
            # 检查前一个块
            prev_blocks = self.free_tree.range_query(0, start - 1)
            if prev_blocks:
                prev_start, prev_size = prev_blocks[-1]
                if prev_start + prev_size == start:
                    # 合并前一个块
                    self.free_tree.delete(prev_start)
                    start = prev_start
                    size += prev_size
            
            # 检查后一个块
            next_blocks = self.free_tree.range_query(start + size, self.total_size)
            if next_blocks:
                next_start, next_size = next_blocks[0]
                if start + size == next_start:
                    # 合并后一个块
                    self.free_tree.delete(next_start)
                    size += next_size
            
            # 插入合并后的块
            self.free_tree.insert(start, size)
        
        def get_memory_status(self):
            """获取内存状态"""
            free_blocks = self.free_tree.range_query(0, self.total_size)
            allocated_blocks = self.allocated_tree.range_query(0, self.total_size)
            
            total_free = sum(size for _, size in free_blocks)
            total_allocated = sum(size for _, size in allocated_blocks)
            
            return {
                "total_size": self.total_size,
                "total_free": total_free,
                "total_allocated": total_allocated,
                "free_blocks": free_blocks,
                "allocated_blocks": allocated_blocks
            }
    
    class BuddySystem:
        """伙伴系统"""
        def __init__(self, total_size):
            self.total_size = total_size
            self.max_level = self.get_max_level(total_size)
            self.free_lists = [[] for _ in range(self.max_level + 1)]
            
            # 初始化最大块
            self.free_lists[self.max_level].append(0)
        
        def get_max_level(self, size):
            """获取最大级别"""
            level = 0
            while (1 << level) < size:
                level += 1
            return level
        
        def allocate(self, size):
            """分配内存"""
            level = self.get_level(size)
            
            # 找到合适的块
            block = self.find_block(level)
            if block is None:
                return None
            
            # 分割块
            self.split_block(block, level, size)
            
            return block
        
        def find_block(self, level):
            """找到合适的块"""
            # 从当前级别开始向上查找
            for i in range(level, self.max_level + 1):
                if self.free_lists[i]:
                    return self.free_lists[i].pop(0)
            
            return None
        
        def split_block(self, block, level, size):
            """分割块"""
            while level > 0 and (1 << (level - 1)) >= size:
                level -= 1
                buddy = block + (1 << level)
                self.free_lists[level].append(buddy)
            
            return block
        
        def deallocate(self, block, size):
            """释放内存"""
            level = self.get_level(size)
            self.free_lists[level].append(block)
            
            # 合并伙伴块
            self.merge_buddies(block, level)
        
        def merge_buddies(self, block, level):
            """合并伙伴块"""
            while level < self.max_level:
                buddy = block ^ (1 << level)
                
                if buddy in self.free_lists[level]:
                    self.free_lists[level].remove(buddy)
                    block = min(block, buddy)
                    level += 1
                else:
                    break
            
            self.free_lists[level].append(block)
        
        def get_level(self, size):
            """获取级别"""
            level = 0
            while (1 << level) < size:
                level += 1
            return level
    
    return MemoryManager, BuddySystem
```
