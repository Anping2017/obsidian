# 字典树

## 🎯 核心概念

**字典树(Trie)**是一种树形数据结构，用于高效地存储和搜索字符串集合。字典树的每个节点代表一个字符，从根节点到任意节点的路径表示一个字符串，在字符串匹配、前缀搜索、自动补全等领域有重要应用。

### 2. 字典树构建
```python
def trie_construction():
    """字典树构建"""
    
    class TrieNode:
        """字典树节点"""
        def __init__(self):
            self.children = {}
            self.is_end_of_word = False
            self.count = 0  # 以该节点结尾的单词数量
    
    class Trie:
        """字典树"""
        def __init__(self):
            self.root = TrieNode()
        
        def insert(self, word):
            """插入单词"""
            current = self.root
            
            for char in word:
                if char not in current.children:
                    current.children[char] = TrieNode()
                current = current.children[char]
            
            current.is_end_of_word = True
            current.count += 1
        
        def search(self, word):
            """搜索单词"""
            current = self.root
            
            for char in word:
                if char not in current.children:
                    return False
                current = current.children[char]
            
            return current.is_end_of_word
        
        def starts_with(self, prefix):
            """前缀搜索"""
            current = self.root
            
            for char in prefix:
                if char not in current.children:
                    return False
                current = current.children[char]
            
            return True
        
        def delete(self, word):
            """删除单词"""
            if not self.search(word):
                return False
            
            current = self.root
            path = [current]
            
            for char in word:
                current = current.children[char]
                path.append(current)
            
            # 标记为非单词结尾
            current.is_end_of_word = False
            current.count -= 1
            
            # 删除不必要的节点
            for i in range(len(path) - 1, 0, -1):
                if not path[i].children and not path[i].is_end_of_word:
                    char = word[i - 1]
                    del path[i - 1].children[char]
                else:
                    break
            
            return True
        
        def get_all_words(self):
            """获取所有单词"""
            words = []
            
            def dfs(node, current_word):
                if node.is_end_of_word:
                    words.append(current_word)
                
                for char, child in node.children.items():
                    dfs(child, current_word + char)
            
            dfs(self.root, "")
            return words
        
        def get_words_with_prefix(self, prefix):
            """获取指定前缀的所有单词"""
            current = self.root
            
            # 找到前缀对应的节点
            for char in prefix:
                if char not in current.children:
                    return []
                current = current.children[char]
            
            words = []
            
            def dfs(node, current_word):
                if node.is_end_of_word:
                    words.append(prefix + current_word)
                
                for char, child in node.children.items():
                    dfs(child, current_word + char)
            
            dfs(current, "")
            return words
    
    return Trie
```

```python
def advanced_trie():
    """高级字典树实现"""
    
    class AdvancedTrieNode:
        """高级字典树节点"""
        def __init__(self):
            self.children = {}
            self.is_end_of_word = False
            self.count = 0
            self.frequency = 0  # 词频
    
    class AdvancedTrie:
        """高级字典树"""
        def __init__(self):
            self.root = AdvancedTrieNode()
        
        def insert(self, word, frequency=1):
            """插入单词"""
            current = self.root
            
            for char in word:
                if char not in current.children:
                    current.children[char] = AdvancedTrieNode()
                current = current.children[char]
            
            current.is_end_of_word = True
            current.count += 1
            current.frequency += frequency
        
        def search(self, word):
            """搜索单词"""
            current = self.root
            
            for char in word:
                if char not in current.children:
                    return False
                current = current.children[char]
            
            return current.is_end_of_word
        
        def starts_with(self, prefix):
            """前缀搜索"""
            current = self.root
            
            for char in prefix:
                if char not in current.children:
                    return False
                current = current.children[char]
            
            return True
        
        def get_words_with_prefix(self, prefix):
            """获取指定前缀的所有单词"""
            current = self.root
            
            for char in prefix:
                if char not in current.children:
                    return []
                current = current.children[char]
            
            words = []
            
            def dfs(node, current_word):
                if node.is_end_of_word:
                    words.append((prefix + current_word, node.frequency))
                
                for char, child in node.children.items():
                    dfs(child, current_word + char)
            
            dfs(current, "")
            return words
        
        def get_top_k_words(self, prefix, k):
            """获取指定前缀的Top K单词"""
            words = self.get_words_with_prefix(prefix)
            words.sort(key=lambda x: x[1], reverse=True)
            return words[:k]
        
        def get_statistics(self):
            """获取统计信息"""
            stats = {
                "total_words": 0,
                "total_nodes": 0,
                "max_depth": 0
            }
            
            def dfs(node, depth):
                stats["total_nodes"] += 1
                stats["max_depth"] = max(stats["max_depth"], depth)
                
                if node.is_end_of_word:
                    stats["total_words"] += node.count
                
                for child in node.children.values():
                    dfs(child, depth + 1)
            
            dfs(self.root, 0)
            return stats
    
    return AdvancedTrie
```

```python
def string_matching_applications():
    """字符串匹配应用"""
    
    def word_search_ii(board, words):
        """单词搜索II"""
        trie = Trie()
        for word in words:
            trie.insert(word)
        
        m, n = len(board), len(board[0])
        result = set()
        
        def dfs(i, j, node, path):
            if i < 0 or i >= m or j < 0 or j >= n:
                return
            
            char = board[i][j]
            if char not in node.children:
                return
            
            node = node.children[char]
            path += char
            
            if node.is_end_of_word:
                result.add(path)
            
            # 标记为已访问
            board[i][j] = '#'
            
            # 四个方向搜索
            for di, dj in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
                dfs(i + di, j + dj, node, path)
            
            # 恢复
            board[i][j] = char
        
        for i in range(m):
            for j in range(n):
                dfs(i, j, trie.root, "")
        
        return list(result)
    
    def replace_words(dictionary, sentence):
        """单词替换"""
        trie = Trie()
        for word in dictionary:
            trie.insert(word)
        
        words = sentence.split()
        result = []
        
        for word in words:
            current = trie.root
            prefix = ""
            
            for char in word:
                if char not in current.children:
                    result.append(word)
                    break
                
                current = current.children[char]
                prefix += char
                
                if current.is_end_of_word:
                    result.append(prefix)
                    break
            else:
                result.append(word)
        
        return " ".join(result)
    
    def word_break(s, word_dict):
        """单词拆分"""
        trie = Trie()
        for word in word_dict:
            trie.insert(word)
        
        n = len(s)
        dp = [False] * (n + 1)
        dp[0] = True
        
        for i in range(n):
            if not dp[i]:
                continue
            
            current = trie.root
            for j in range(i, n):
                if s[j] not in current.children:
                    break
                
                current = current.children[s[j]]
                if current.is_end_of_word:
                    dp[j + 1] = True
        
        return dp[n]
    
    def word_break_ii(s, word_dict):
        """单词拆分II"""
        trie = Trie()
        for word in word_dict:
            trie.insert(word)
        
        n = len(s)
        result = []
        
        def dfs(start, path):
            if start == n:
                result.append(" ".join(path))
                return
            
            current = trie.root
            for i in range(start, n):
                if s[i] not in current.children:
                    break
                
                current = current.children[s[i]]
                if current.is_end_of_word:
                    path.append(s[start:i+1])
                    dfs(i + 1, path)
                    path.pop()
        
        dfs(0, [])
        return result
    
    return word_search_ii, replace_words, word_break, word_break_ii
```

```python
def prefix_search_applications():
    """前缀搜索应用"""
    
    def autocomplete_system(sentences, times):
        """自动补全系统"""
        trie = AdvancedTrie()
        
        # 插入句子和频率
        for sentence, time in zip(sentences, times):
            trie.insert(sentence, time)
        
        def input(c):
            if c == '#':
                return []
            
            # 获取前缀的所有句子
            sentences = trie.get_words_with_prefix(c)
            
            # 按频率排序，返回前3个
            sentences.sort(key=lambda x: x[1], reverse=True)
            return [sentence for sentence, _ in sentences[:3]]
        
        return input
    
    def search_suggestions_system(products, search_word):
        """搜索建议系统"""
        trie = Trie()
        for product in products:
            trie.insert(product)
        
        result = []
        prefix = ""
        
        for char in search_word:
            prefix += char
            suggestions = trie.get_words_with_prefix(prefix)
            suggestions.sort()
            result.append(suggestions[:3])
        
        return result
    
    def prefix_and_suffix_search(words):
        """前缀和后缀搜索"""
        trie = Trie()
        
        # 插入所有可能的前缀和后缀组合
        for i, word in enumerate(words):
            for j in range(len(word)):
                for k in range(len(word)):
                    prefix_suffix = word[:j+1] + "#" + word[k:]
                    trie.insert(prefix_suffix, i)
        
        def f(prefix, suffix):
            key = prefix + "#" + suffix
            words_with_prefix = trie.get_words_with_prefix(key)
            if words_with_prefix:
                return max(words_with_prefix)
            return -1
        
        return f
    
    def word_squares(words):
        """单词方块"""
        trie = Trie()
        for word in words:
            trie.insert(word)
        
        n = len(words[0])
        result = []
        
        def dfs(square):
            if len(square) == n:
                result.append(square[:])
                return
            
            # 构建当前行的前缀
            prefix = ""
            for i in range(len(square)):
                prefix += square[i][len(square)]
            
            # 找到所有以该前缀开头的单词
            candidates = trie.get_words_with_prefix(prefix)
            
            for candidate in candidates:
                square.append(candidate)
                dfs(square)
                square.pop()
        
        for word in words:
            dfs([word])
        
        return result
    
    return autocomplete_system, search_suggestions_system, prefix_and_suffix_search, word_squares
```

```python
def autocomplete_applications():
    """自动补全应用"""
    
    class AutocompleteSystem:
        """自动补全系统"""
        def __init__(self, sentences, times):
            self.trie = AdvancedTrie()
            self.current_input = ""
            
            for sentence, time in zip(sentences, times):
                self.trie.insert(sentence, time)
        
        def input(self, c):
            """输入字符"""
            if c == '#':
                # 结束输入，更新频率
                if self.current_input:
                    self.trie.insert(self.current_input, 1)
                self.current_input = ""
                return []
            
            self.current_input += c
            
            # 获取建议
            suggestions = self.trie.get_words_with_prefix(self.current_input)
            suggestions.sort(key=lambda x: x[1], reverse=True)
            
            return [sentence for sentence, _ in suggestions[:3]]
    
    def keyboard_autocomplete(words, frequencies):
        """键盘自动补全"""
        trie = AdvancedTrie()
        
        for word, freq in zip(words, frequencies):
            trie.insert(word, freq)
        
        def get_suggestions(prefix):
            suggestions = trie.get_words_with_prefix(prefix)
            suggestions.sort(key=lambda x: x[1], reverse=True)
            return [sentence for sentence, _ in suggestions[:5]]
        
        return get_suggestions
    
    def smart_autocomplete(words, context):
        """智能自动补全"""
        trie = AdvancedTrie()
        
        # 根据上下文调整频率
        for word in words:
            base_freq = 1
            if word in context:
                base_freq *= 2  # 提高上下文相关词的频率
            
            trie.insert(word, base_freq)
        
        def get_contextual_suggestions(prefix, context_words):
            suggestions = trie.get_words_with_prefix(prefix)
            
            # 根据上下文调整分数
            for i, (word, freq) in enumerate(suggestions):
                if word in context_words:
                    suggestions[i] = (word, freq * 1.5)
            
            suggestions.sort(key=lambda x: x[1], reverse=True)
            return [sentence for sentence, _ in suggestions[:3]]
        
        return get_contextual_suggestions
    
    def personalized_autocomplete(user_history, words):
        """个性化自动补全"""
        trie = AdvancedTrie()
        
        # 根据用户历史调整频率
        for word in words:
            base_freq = 1
            if word in user_history:
                base_freq *= user_history[word]  # 用户历史频率
            
            trie.insert(word, base_freq)
        
        def get_personalized_suggestions(prefix):
            suggestions = trie.get_words_with_prefix(prefix)
            suggestions.sort(key=lambda x: x[1], reverse=True)
            return [sentence for sentence, _ in suggestions[:5]]
        
        return get_personalized_suggestions
    
    return AutocompleteSystem, keyboard_autocomplete, smart_autocomplete, personalized_autocomplete
```
