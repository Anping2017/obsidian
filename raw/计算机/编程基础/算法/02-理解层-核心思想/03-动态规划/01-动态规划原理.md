# 动态规划原理

## 🎯 核心概念

**动态规划**是一种通过把原问题分解为相对简单的子问题的方式求解复杂问题的方法。它通过记忆化存储子问题的解，避免重复计算，从而提高算法效率。

## 🎨 经典问题

### 1. 斐波那契数列
```python
def fibonacci_recursive(n):
    """递归实现斐波那契数列 - O(2^n)"""
    if n <= 1:
        return n
    return fibonacci_recursive(n-1) + fibonacci_recursive(n-2)

def fibonacci_memoization(n, memo={}):
    """记忆化递归实现斐波那契数列 - O(n)"""
    if n in memo:
        return memo[n]
    
    if n <= 1:
        return n
    
    memo[n] = fibonacci_memoization(n-1, memo) + fibonacci_memoization(n-2, memo)
    return memo[n]

def fibonacci_dp(n):
    """动态规划实现斐波那契数列 - O(n)"""
    if n <= 1:
        return n
    
    dp = [0] * (n + 1)
    dp[0] = 0
    dp[1] = 1
    
    for i in range(2, n + 1):
        dp[i] = dp[i-1] + dp[i-2]
    
    return dp[n]

def fibonacci_optimized(n):
    """空间优化的斐波那契数列 - O(n)时间，O(1)空间"""
    if n <= 1:
        return n
    
    prev2 = 0
    prev1 = 1
    
    for i in range(2, n + 1):
        current = prev1 + prev2
        prev2 = prev1
        prev1 = current
    
    return prev1
```

```python
def climb_stairs(n):
    """爬楼梯问题 - O(n)"""

        return n
    
    prev2 = 1
    prev1 = 2

        current = prev1 + prev2

        prev1 = current

def climb_stairs_variants(n, steps):
    """爬楼梯变种：可以走steps中的任意步数 - O(n*k)"""
    dp = [0] * (n + 1)
    dp[0] = 1
    
    for i in range(1, n + 1):
        for step in steps:
            if i >= step:
                dp[i] += dp[i - step]
    
    return dp[n]
```

### 3. 最长公共子序列
```python
def longest_common_subsequence(text1, text2):
    """最长公共子序列 - O(m*n)"""
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i-1] == text2[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    
    return dp[m][n]

def longest_common_subsequence_optimized(text1, text2):
    """空间优化的最长公共子序列 - O(m*n)时间，O(min(m,n))空间"""
    if len(text1) < len(text2):
        text1, text2 = text2, text1
    
    m, n = len(text1), len(text2)
    prev = [0] * (n + 1)
    curr = [0] * (n + 1)
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i-1] == text2[j-1]:
                curr[j] = prev[j-1] + 1
            else:
                curr[j] = max(prev[j], curr[j-1])
        prev, curr = curr, prev
    
    return prev[n]

def lcs_reconstruction(text1, text2):
    """重构最长公共子序列"""
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    # 计算LCS长度
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i-1] == text2[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    
    # 重构LCS
    lcs = []
    i, j = m, n
    while i > 0 and j > 0:
        if text1[i-1] == text2[j-1]:
            lcs.append(text1[i-1])
            i -= 1
            j -= 1
        elif dp[i-1][j] > dp[i][j-1]:
            i -= 1
        else:
            j -= 1
    
    return ''.join(reversed(lcs))
```

### 4. 最长递增子序列
```python
def longest_increasing_subsequence(nums):
    """最长递增子序列 - O(n^2)"""
    if not nums:
        return 0
    
    n = len(nums)
    dp = [1] * n
    
    for i in range(1, n):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)
    
    return max(dp)

def longest_increasing_subsequence_optimized(nums):
    """优化的最长递增子序列 - O(n log n)"""
    if not nums:
        return 0
    
    tails = []
    
    for num in nums:
        left, right = 0, len(tails)
        while left < right:
            mid = (left + right) // 2
            if tails[mid] < num:
                left = mid + 1
            else:
                right = mid
        
        if left == len(tails):
            tails.append(num)
        else:
            tails[left] = num
    
    return len(tails)

def lis_reconstruction(nums):
    """重构最长递增子序列"""
    if not nums:
        return []
    
    n = len(nums)
    dp = [1] * n
    prev = [-1] * n
    
    for i in range(1, n):
        for j in range(i):
            if nums[j] < nums[i] and dp[j] + 1 > dp[i]:
                dp[i] = dp[j] + 1
                prev[i] = j
    
    # 找到最长序列的末尾
    max_len = max(dp)
    end = dp.index(max_len)
    
    # 重构序列
    lis = []
    while end != -1:
        lis.append(nums[end])
        end = prev[end]
    
    return lis[::-1]
```

```python
def house_robber(nums):
    """打家劫舍问题 - O(n)"""
    if not nums:
        return 0
    if len(nums) == 1:
        return nums[0]
    
    dp = [0] * len(nums)
    dp[0] = nums[0]
    dp[1] = max(nums[0], nums[1])
    
    for i in range(2, len(nums)):
        dp[i] = max(dp[i-1], dp[i-2] + nums[i])
    
    return dp[-1]

def house_robber_optimized(nums):
    """空间优化的打家劫舍问题 - O(n)时间，O(1)空间"""
    if not nums:
        return 0
    if len(nums) == 1:
        return nums[0]
    
    prev2 = nums[0]
    prev1 = max(nums[0], nums[1])
    
    for i in range(2, len(nums)):
        current = max(prev1, prev2 + nums[i])
        prev2 = prev1
        prev1 = current
    
    return prev1
```

```python
def unique_paths(m, n):
    """不同路径问题 - O(m*n)"""
    dp = [[1] * n for _ in range(m)]
    
    for i in range(1, m):
        for j in range(1, n):
            dp[i][j] = dp[i-1][j] + dp[i][j-1]
    
    return dp[m-1][n-1]

def unique_paths_optimized(m, n):
    """空间优化的不同路径问题 - O(m*n)时间，O(n)空间"""
    dp = [1] * n
    
    for i in range(1, m):
        for j in range(1, n):
            dp[j] += dp[j-1]
    
    return dp[n-1]

def min_path_sum(grid):
    """最小路径和问题 - O(m*n)"""
    m, n = len(grid), len(grid[0])
    dp = [[0] * n for _ in range(m)]
    
    dp[0][0] = grid[0][0]
    
    # 初始化第一行
    for j in range(1, n):
        dp[0][j] = dp[0][j-1] + grid[0][j]
    
    # 初始化第一列
    for i in range(1, m):
        dp[i][0] = dp[i-1][0] + grid[i][0]
    
    # 填充其余位置
    for i in range(1, m):
        for j in range(1, n):
            dp[i][j] = min(dp[i-1][j], dp[i][j-1]) + grid[i][j]
    
    return dp[m-1][n-1]
```

```python
def interval_dp():
    """区间动态规划"""
    # 状态转移方程通常涉及区间
    # 例如：dp[i][j] = f(dp[i][k], dp[k+1][j]) for k in [i, j)
    
    pass

def matrix_chain_multiplication(p):
    """矩阵链乘法问题 - O(n^3)"""
    n = len(p) - 1
    dp = [[0] * n for _ in range(n)]
    
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i][j] = float('inf')
            
            for k in range(i, j):
                cost = dp[i][k] + dp[k+1][j] + p[i] * p[k+1] * p[j+1]
                dp[i][j] = min(dp[i][j], cost)
    
    return dp[0][n-1]

def palindrome_substring(s):
    """回文子串问题 - O(n^2)"""
    n = len(s)
    dp = [[False] * n for _ in range(n)]
    count = 0
    
    # 单个字符都是回文
    for i in range(n):
        dp[i][i] = True
        count += 1
    
    # 两个字符的情况
    for i in range(n - 1):
        if s[i] == s[i + 1]:
            dp[i][i + 1] = True
            count += 1
    
    # 三个或更多字符的情况
    for length in range(3, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            if s[i] == s[j] and dp[i + 1][j - 1]:
                dp[i][j] = True
                count += 1
    
    return count
```

## 📈 复杂度分析

### 时间复杂度
| 问题类型 | 时间复杂度 | 说明 |
|----------|------------|------|
| 线性DP | O(n) | 一维状态转移 |
| 二维DP | O(m*n) | 二维状态转移 |
| 区间DP | O(n^3) | 区间长度和分割点 |
| 背包DP | O(n*W) | 物品数量和容量 |

### 空间复杂度
| 优化方法 | 空间复杂度 | 说明 |
|----------|------------|------|
| 无优化 | O(n) 或 O(m*n) | 存储所有状态 |
| 滚动数组 | O(n) | 只存储必要状态 |
| 空间优化 | O(1) | 只存储当前状态 |

## 🎯 应用场景

### 适用条件
| 条件 | 说明 | 示例 |
|------|------|------|
| 最优子结构 | 问题的最优解包含子问题的最优解 | 最短路径、最长子序列 |
| 重叠子问题 | 递归过程中重复计算相同子问题 | 斐波那契数列、爬楼梯 |
| 无后效性 | 当前状态只依赖于之前的状态 | 路径问题、序列问题 |

### 不适用场景
| 场景 | 原因 | 示例 |
|------|------|------|
| 无最优子结构 | 无法分解为子问题 | 某些NP完全问题 |
| 无重叠子问题 | 递归过程中不重复计算 | 分治算法 |
| 有后效性 | 当前状态依赖于如何到达 | 某些博弈问题 |
