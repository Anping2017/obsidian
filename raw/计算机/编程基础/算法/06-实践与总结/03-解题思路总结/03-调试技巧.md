# 调试技巧

## 概述

调试技巧是算法实现中的重要技能，通过系统性的调试方法可以快速定位和修复代码中的错误。掌握有效的调试技巧可以提高开发效率，减少调试时间，提高代码质量。

## 核心概念

### 基本定义
- **调试技巧**：定位和修复代码错误的方法
- **调试工具**：用于调试的软件工具
- **调试策略**：系统性的调试方法
- **错误定位**：快速找到错误位置

### 关键特性
- **系统性**：提供完整的调试流程
- **高效性**：快速定位错误
- **准确性**：准确找到错误原因
- **可重复性**：可以重复使用的方法

```python
def print_debugging_example():
    """打印调试示例"""
    def binary_search_debug(arr, target):
```

```python
def assertion_debugging_example():
    """断言调试示例"""
    def quick_sort_debug(arr, low, high):
        if low < high:
            # 断言：检查输入参数
            assert 0 <= low < len(arr), f"low={low} 超出范围"
            assert 0 <= high < len(arr), f"high={high} 超出范围"
            assert low <= high, f"low={low} > high={high}"
            
            # 分区操作
            pivot_index = partition(arr, low, high)
            
            # 断言：检查分区结果
            assert low <= pivot_index <= high, f"pivot_index={pivot_index} 超出范围"
            
            # 递归排序
            quick_sort_debug(arr, low, pivot_index - 1)
            quick_sort_debug(arr, pivot_index + 1, high)
    
    def partition(arr, low, high):
        pivot = arr[high]
        i = low - 1
        
        for j in range(low, high):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        
        arr[i + 1], arr[high] = arr[high], arr[i + 1]
        return i + 1
    
    # 测试
    arr = [64, 34, 25, 12, 22, 11, 90]
    print(f"排序前: {arr}")
    quick_sort_debug(arr, 0, len(arr) - 1)
    print(f"排序后: {arr}")
```

```python
import logging

def logging_debugging_example():
    """日志调试示例"""
    # 配置日志
    logging.basicConfig(level=logging.DEBUG, format='%(asctime)s - %(levelname)s - %(message)s')
    logger = logging.getLogger(__name__)
    
    def merge_sort_debug(arr):
        logger.debug(f"开始排序数组: {arr}")
        
        if len(arr) <= 1:
            logger.debug(f"数组长度<=1，直接返回: {arr}")
            return arr
        
        mid = len(arr) // 2
        left = arr[:mid]
        right = arr[mid:]
        
        logger.debug(f"分割数组: left={left}, right={right}")
        
        left_sorted = merge_sort_debug(left)
        right_sorted = merge_sort_debug(right)
        
        logger.debug(f"合并数组: left_sorted={left_sorted}, right_sorted={right_sorted}")
        
        result = merge(left_sorted, right_sorted)
        logger.debug(f"合并结果: {result}")
        
        return result
    
    def merge(left, right):
        result = []
        i = j = 0
        
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        
        result.extend(left[i:])
        result.extend(right[j:])
        
        return result
    
    # 测试
    arr = [38, 27, 43, 3, 9, 82, 10]
    result = merge_sort_debug(arr)
    print(f"排序结果: {result}")

# 使用示例
```

```python
def breakpoint_debugging_example():
    """断点调试示例"""
    def dfs_debug(graph, start, visited=None):
        if visited is None:
            visited = set()
        
        visited.add(start)
        print(f"访问节点: {start}")
        
        # 断点：检查访问状态
        # import pdb; pdb.set_trace()  # 设置断点
        
        for neighbor in graph[start]:
            if neighbor not in visited:
                dfs_debug(graph, neighbor, visited)
        
        return visited
    
    # 测试图
    graph = {
        'A': ['B', 'C'],
        'B': ['A', 'D', 'E'],
        'C': ['A', 'F'],
        'D': ['B'],
        'E': ['B', 'F'],
        'F': ['C', 'E']
    }
    
    result = dfs_debug(graph, 'A')
    print(f"DFS遍历结果: {result}")
```

```python
import unittest

def unit_test_debugging_example():
    """单元测试调试示例"""
    class TestAlgorithms(unittest.TestCase):
        def test_binary_search(self):
            """测试二分搜索"""
            arr = [1, 3, 5, 7, 9, 11, 13]
            
            # 测试正常情况
            self.assertEqual(binary_search(arr, 7), 3)
            self.assertEqual(binary_search(arr, 1), 0)
            self.assertEqual(binary_search(arr, 13), 6)
            
            # 测试边界情况
            self.assertEqual(binary_search(arr, 0), -1)
            self.assertEqual(binary_search(arr, 14), -1)
            
            # 测试空数组
            self.assertEqual(binary_search([], 5), -1)
            
            # 测试单元素数组
            self.assertEqual(binary_search([5], 5), 0)
            self.assertEqual(binary_search([5], 3), -1)
        
        def test_quick_sort(self):
            """测试快速排序"""
            # 测试正常情况
            arr = [64, 34, 25, 12, 22, 11, 90]
            expected = sorted(arr)
            quick_sort(arr, 0, len(arr) - 1)
            self.assertEqual(arr, expected)
            
            # 测试边界情况
            empty_arr = []
            quick_sort(empty_arr, 0, -1)
            self.assertEqual(empty_arr, [])
            
            single_arr = [5]
            quick_sort(single_arr, 0, 0)
            self.assertEqual(single_arr, [5])
    
    def binary_search(arr, target):
        left, right = 0, len(arr) - 1
        
        while left <= right:
            mid = (left + right) // 2
            if arr[mid] == target:
                return mid
            elif arr[mid] < target:
                left = mid + 1
            else:
                right = mid - 1
        
        return -1
    
    def quick_sort(arr, low, high):
        if low < high:
            pivot_index = partition(arr, low, high)
            quick_sort(arr, low, pivot_index - 1)
            quick_sort(arr, pivot_index + 1, high)
    
    def partition(arr, low, high):
        pivot = arr[high]
        i = low - 1
        
        for j in range(low, high):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        
        arr[i + 1], arr[high] = arr[high], arr[i + 1]
        return i + 1
    
    # 运行测试
    unittest.main(argv=[''], exit=False, verbosity=2)

# 使用示例
unit_test_debugging_example()
```

#### 2.3 性能调试
```python
import time
import cProfile
import pstats

def performance_debugging_example():
    """性能调试示例"""
    def fibonacci_slow(n):
        """慢速斐波那契数列"""
        if n <= 1:
            return n
        return fibonacci_slow(n - 1) + fibonacci_slow(n - 2)
    
    def fibonacci_fast(n):
        """快速斐波那契数列"""
        if n <= 1:
            return n
        
        a, b = 0, 1
        for _ in range(2, n + 1):
            a, b = b, a + b
        return b
    
    def fibonacci_dp(n):
        """动态规划斐波那契数列"""
        if n <= 1:
            return n
        
        dp = [0] * (n + 1)
        dp[1] = 1
        
        for i in range(2, n + 1):
            dp[i] = dp[i - 1] + dp[i - 2]
        
        return dp[n]
    
    # 时间测试
    n = 30
    
    print("=== 时间测试 ===")
    
    # 测试慢速版本
    start_time = time.time()
    result_slow = fibonacci_slow(n)
    slow_time = time.time() - start_time
    print(f"慢速版本: {result_slow}, 时间: {slow_time:.6f}秒")
    
    # 测试快速版本
    start_time = time.time()
    result_fast = fibonacci_fast(n)
    fast_time = time.time() - start_time
    print(f"快速版本: {result_fast}, 时间: {fast_time:.6f}秒")
    
    # 测试动态规划版本
    start_time = time.time()
    result_dp = fibonacci_dp(n)
    dp_time = time.time() - start_time
    print(f"动态规划版本: {result_dp}, 时间: {dp_time:.6f}秒")
    
    # 性能分析
    print("\n=== 性能分析 ===")
    cProfile.run('fibonacci_slow(20)', 'slow_profile.prof')
    cProfile.run('fibonacci_fast(20)', 'fast_profile.prof')
    
    # 分析结果
    slow_stats = pstats.Stats('slow_profile.prof')
    fast_stats = pstats.Stats('fast_profile.prof')
    
    print("慢速版本性能分析:")
    slow_stats.sort_stats('cumulative').print_stats(10)
    
    print("\n快速版本性能分析:")
    fast_stats.sort_stats('cumulative').print_stats(10)

# 使用示例
performance_debugging_example()
```

```python
def visualization_debugging_example():
    """可视化调试示例"""
    def visualize_sorting(arr):
        """可视化排序过程"""
        print(f"初始数组: {arr}")
        
        n = len(arr)
        for i in range(n):
            min_idx = i
            for j in range(i + 1, n):
                if arr[j] < arr[min_idx]:
                    min_idx = j
            
            if min_idx != i:
                arr[i], arr[min_idx] = arr[min_idx], arr[i]
                print(f"交换 {arr[i]} 和 {arr[min_idx]}: {arr}")
        
        return arr
    
    # 测试
    arr = [64, 25, 12, 22, 11]
    result = visualize_sorting(arr)
    print(f"最终结果: {result}")
```
