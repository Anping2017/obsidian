```python
def gcd(a, b):
    """最大公约数"""
    while b:
        a, b = b, a % b
    return a

def extended_gcd(a, b):
    """扩展欧几里得算法"""
    if b == 0:
        return a, 1, 0
    
    gcd, x1, y1 = extended_gcd(b, a % b)
    x = y1
    y = x1 - (a // b) * y1
    
    return gcd, x, y

def lcm(a, b):
    """最小公倍数"""
    return abs(a * b) // gcd(a, b)

# 使用示例
a, b = 48, 18
print(f"gcd({a}, {b}) = {gcd(a, b)}")
print(f"lcm({a}, {b}) = {lcm(a, b)}")
```

#### 素数判断模板
```python
def is_prime(n):
    """素数判断"""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    
    for i in range(3, int(n**0.5) + 1, 2):
        if n % i == 0:
            return False
    
    return True

def sieve_of_eratosthenes(n):
    """埃拉托斯特尼筛法"""
    if n < 2:
        return []
    
    is_prime = [True] * (n + 1)
    is_prime[0] = is_prime[1] = False
    
    for i in range(2, int(n**0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, n + 1, i):
                is_prime[j] = False
    
    primes = []
    for i in range(2, n + 1):
        if is_prime[i]:
            primes.append(i)
    
    return primes

# 使用示例
n = 100
primes = sieve_of_eratosthenes(n)
print(f"小于等于{n}的素数: {primes}")
```

#### 模运算模板
```python
def modular_inverse(a, m):
    """模逆元"""
    gcd, x, y = extended_gcd(a, m)
    if gcd != 1:
        raise ValueError("模逆元不存在")
    
    return (x % m + m) % m

def modular_power(base, exponent, modulus):
    """模幂运算"""
    result = 1
    base = base % modulus
    
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % modulus
        
        base = (base * base) % modulus
        exponent //= 2
    
    return result

def chinese_remainder_theorem(remainders, moduli):
    """中国剩余定理"""
    if len(remainders) != len(moduli):
        raise ValueError("余数和模数数量不匹配")
    
    M = 1
    for m in moduli:
        M *= m
    
    result = 0
    for i in range(len(remainders)):
        Mi = M // moduli[i]
        Mi_inv = modular_inverse(Mi, moduli[i])
        result += remainders[i] * Mi * Mi_inv
    
    return result % M

# 使用示例
a, m = 3, 11
inv = modular_inverse(a, m)
print(f"{a}在模{m}下的逆元: {inv}")

base, exp, mod = 2, 10, 1000000007
result = modular_power(base, exp, mod)
print(f"{base}^{exp} mod {mod} = {result}")
```

### 2. 组合数学模板

#### 组合数模板
```python
def combination(n, k):
    """组合数C(n,k)"""
    if k > n or k < 0:
        return 0
    
    if k > n - k:
        k = n - k
    
    result = 1
    for i in range(k):
        result = result * (n - i) // (i + 1)
    
    return result

def combination_with_mod(n, k, mod):
    """组合数C(n,k) mod m"""
    if k > n or k < 0:
        return 0
    
    if k > n - k:
        k = n - k
    
    numerator = 1
    denominator = 1
    
    for i in range(k):
        numerator = (numerator * (n - i)) % mod
        denominator = (denominator * (i + 1)) % mod
    
    return (numerator * modular_inverse(denominator, mod)) % mod

def pascal_triangle(n):
    """杨辉三角形"""
    triangle = []
    for i in range(n):
        row = []
        for j in range(i + 1):
            row.append(combination(i, j))
        triangle.append(row)
    return triangle

# 使用示例
n, k = 10, 3
print(f"C({n}, {k}) = {combination(n, k)}")

triangle = pascal_triangle(5)
for row in triangle:
    print(row)
```

#### 排列数模板
```python
def permutation(n, k):
    """排列数P(n,k)"""
    if k > n or k < 0:
        return 0
    
    result = 1
    for i in range(k):
        result *= (n - i)
    
    return result

def factorial(n):
    """阶乘n!"""
    if n < 0:
        return 0
    if n <= 1:
        return 1
    
    result = 1
    for i in range(2, n + 1):
        result *= i
    
    return result

def factorial_with_mod(n, mod):
    """阶乘n! mod m"""
    if n < 0:
        return 0
    if n <= 1:
        return 1
    
    result = 1
    for i in range(2, n + 1):
        result = (result * i) % mod
    
    return result

# 使用示例
n, k = 10, 3
print(f"P({n}, {k}) = {permutation(n, k)}")
print(f"{n}! = {factorial(n)}")
```

#### 卡特兰数模板
```python
def catalan_number(n):
    """卡特兰数"""
    if n <= 1:
        return 1
    
    # 使用递推公式：C(n) = (2n)! / ((n+1)! * n!)
    numerator = factorial(2 * n)
    denominator = factorial(n + 1) * factorial(n)
    
    return numerator // denominator

def catalan_number_dp(n):
    """卡特兰数（动态规划）"""
    if n <= 1:
        return 1
    
    dp = [0] * (n + 1)
    dp[0] = dp[1] = 1
    
    for i in range(2, n + 1):
        for j in range(i):
            dp[i] += dp[j] * dp[i - 1 - j]
    
    return dp[n]

# 使用示例
n = 10
print(f"第{n}个卡特兰数: {catalan_number(n)}")
print(f"第{n}个卡特兰数（DP）: {catalan_number_dp(n)}")
```

### 3. 矩阵运算模板

#### 矩阵乘法模板
```python
def matrix_multiply(A, B):
    """矩阵乘法"""
    m, n = len(A), len(A[0])
    p, q = len(B), len(B[0])
    
    if n != p:
        raise ValueError("矩阵维度不匹配")
    
    result = [[0] * q for _ in range(m)]
    
    for i in range(m):
        for j in range(q):
            for k in range(n):
                result[i][j] += A[i][k] * B[k][j]
    
    return result

def matrix_power(matrix, n):
    """矩阵快速幂"""
    if n == 0:
        # 返回单位矩阵
        size = len(matrix)
        identity = [[0] * size for _ in range(size)]
        for i in range(size):
            identity[i][i] = 1
        return identity
    
    if n == 1:
        return matrix
    
    if n % 2 == 0:
        half = matrix_power(matrix, n // 2)
        return matrix_multiply(half, half)
    else:
        half = matrix_power(matrix, n // 2)
        return matrix_multiply(matrix_multiply(half, half), matrix)

def matrix_power_with_mod(matrix, n, mod):
    """矩阵快速幂（带模运算）"""
    if n == 0:
        size = len(matrix)
        identity = [[0] * size for _ in range(size)]
        for i in range(size):
            identity[i][i] = 1
        return identity
    
    if n == 1:
        return [[val % mod for val in row] for row in matrix]
    
    if n % 2 == 0:
        half = matrix_power_with_mod(matrix, n // 2, mod)
        return matrix_multiply_with_mod(half, half, mod)
    else:
        half = matrix_power_with_mod(matrix, n // 2, mod)
        return matrix_multiply_with_mod(matrix_multiply_with_mod(half, half, mod), matrix, mod)

def matrix_multiply_with_mod(A, B, mod):
    """矩阵乘法（带模运算）"""
    m, n = len(A), len(A[0])
    p, q = len(B), len(B[0])
    
    if n != p:
        raise ValueError("矩阵维度不匹配")
    
    result = [[0] * q for _ in range(m)]
    
    for i in range(m):
        for j in range(q):
            for k in range(n):
                result[i][j] = (result[i][j] + A[i][k] * B[k][j]) % mod
    
    return result

# 使用示例
A = [[1, 2], [3, 4]]
B = [[5, 6], [7, 8]]
result = matrix_multiply(A, B)
print(f"矩阵乘法结果: {result}")

matrix = [[1, 1], [1, 0]]
power = matrix_power(matrix, 10)
print(f"矩阵的10次幂: {power}")
```

### 4. 几何算法模板

#### 点运算模板
```python
import math

class Point:
    """点类"""
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def distance_to(self, other):
        """计算到另一点的距离"""
        return math.sqrt((self.x - other.x)**2 + (self.y - other.y)**2)
    
    def __str__(self):
        return f"({self.x}, {self.y})"

def cross_product(p1, p2, p3):
    """叉积"""
    return (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x)

def dot_product(p1, p2, p3):
    """点积"""
    return (p2.x - p1.x) * (p3.x - p1.x) + (p2.y - p1.y) * (p3.y - p1.y)

def convex_hull(points):
    """凸包算法（Graham扫描）"""
    if len(points) < 3:
        return points
    
    # 找到最下面的点
    bottom = min(points, key=lambda p: (p.y, p.x))
    
    # 按极角排序
    def polar_angle(p):
        return math.atan2(p.y - bottom.y, p.x - bottom.x)
    
    points.sort(key=polar_angle)
    
    # 构建凸包
    hull = [points[0], points[1]]
    
    for i in range(2, len(points)):
        while len(hull) > 1 and cross_product(hull[-2], hull[-1], points[i]) <= 0:
            hull.pop()
        hull.append(points[i])
    
    return hull

# 使用示例
points = [Point(0, 0), Point(1, 1), Point(2, 0), Point(1, -1)]
hull = convex_hull(points)
print(f"凸包: {[str(p) for p in hull]}")
```

#### 线段相交模板
```python
class LineSegment:
    """线段类"""
    def __init__(self, p1, p2):
        self.p1 = p1
        self.p2 = p2
    
    def intersects(self, other):
        """判断两条线段是否相交"""
        def orientation(p, q, r):
            val = (q.y - p.y) * (r.x - q.x) - (q.x - p.x) * (r.y - q.y)
            if val == 0:
                return 0  # 共线
            return 1 if val > 0 else 2  # 顺时针或逆时针
        
        def on_segment(p, q, r):
            return (q.x <= max(p.x, r.x) and q.x >= min(p.x, r.x) and
                    q.y <= max(p.y, r.y) and q.y >= min(p.y, r.y))
        
        o1 = orientation(self.p1, self.p2, other.p1)
        o2 = orientation(self.p1, self.p2, other.p2)
        o3 = orientation(other.p1, other.p2, self.p1)
        o4 = orientation(other.p1, other.p2, self.p2)
        
        if o1 != o2 and o3 != o4:
            return True
        
        if o1 == 0 and on_segment(self.p1, other.p1, self.p2):
            return True
        if o2 == 0 and on_segment(self.p1, other.p2, self.p2):
            return True
        if o3 == 0 and on_segment(other.p1, self.p1, other.p2):
            return True
        if o4 == 0 and on_segment(other.p1, self.p2, other.p2):
            return True
        
        return False

# 使用示例
p1, p2 = Point(0, 0), Point(2, 2)
p3, p4 = Point(1, 0), Point(1, 2)
seg1 = LineSegment(p1, p2)
seg2 = LineSegment(p3, p4)
print(f"线段相交: {seg1.intersects(seg2)}")
```

### 5. 概率统计模板

#### 概率计算模板
```python
def binomial_probability(n, k, p):
    """二项分布概率"""
    if k > n or k < 0:
        return 0
    
    return combination(n, k) * (p ** k) * ((1 - p) ** (n - k))

def poisson_probability(k, lambda_param):
    """泊松分布概率"""
    if k < 0:
        return 0
    
    return (math.exp(-lambda_param) * (lambda_param ** k)) / factorial(k)

def normal_probability(x, mu, sigma):
    """正态分布概率密度"""
    return (1 / (sigma * math.sqrt(2 * math.pi))) * math.exp(-0.5 * ((x - mu) / sigma) ** 2)

# 使用示例
n, k, p = 10, 3, 0.5
prob = binomial_probability(n, k, p)
print(f"二项分布P(X={k}|n={n}, p={p}) = {prob}")
```
