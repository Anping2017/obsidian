# 性能测试

## 概述

性能测试是算法实现中的重要实践，通过系统性的性能测试可以验证算法的执行效率、资源消耗和可扩展性。良好的性能测试是高质量算法实现的基础，也是专业程序员必备的技能。

## 核心概念

### 基本定义
- **性能测试**：测试算法执行效率的过程
- **基准测试**：建立性能基准的测试
- **压力测试**：测试算法极限性能
- **性能分析**：分析性能瓶颈的过程

### 关键特性
- **可测量性**：可以量化的性能指标
- **可重复性**：测试结果可以重复
- **可比较性**：可以比较不同算法性能
- **可预测性**：可以预测性能表现

```python
import time
import random

def time_performance_tests():
    """时间性能测试示例"""
    
    def measure_execution_time(func, *args, **kwargs):
        """测量函数执行时间"""
        start_time = time.time()
        result = func(*args, **kwargs)
        end_time = time.time()
        execution_time = end_time - start_time
        return result, execution_time
    
    def bubble_sort(arr):
        """冒泡排序"""
        n = len(arr)
        for i in range(n):
            for j in range(0, n - i - 1):
                if arr[j] > arr[j + 1]:
                    arr[j], arr[j + 1] = arr[j + 1], arr[j]
        return arr
    
    def quick_sort(arr):
        """快速排序"""
        if len(arr) <= 1:
            return arr
        pivot = arr[len(arr) // 2]
        left = [x for x in arr if x < pivot]
        middle = [x for x in arr if x == pivot]
        right = [x for x in arr if x > pivot]
        return quick_sort(left) + middle + quick_sort(right)
    
    def merge_sort(arr):
        """归并排序"""
        if len(arr) <= 1:
            return arr
        
        mid = len(arr) // 2
        left = merge_sort(arr[:mid])
        right = merge_sort(arr[mid:])
        
        return merge(left, right)
    
    def merge(left, right):
        """合并两个有序数组"""
        result = []
        i = j = 0
        
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        
        result.extend(left[i:])
        result.extend(right[j:])
        return result
    
    # 测试不同数据规模
    test_sizes = [100, 500, 1000, 2000]
    
    print("=== 排序算法性能测试 ===")
    print(f"{'数据规模':<10} {'冒泡排序':<15} {'快速排序':<15} {'归并排序':<15}")
    print("-" * 60)
    
    for size in test_sizes:
        # 生成测试数据
        test_data = [random.randint(1, 1000) for _ in range(size)]
        
        # 测试冒泡排序
        _, bubble_time = measure_execution_time(bubble_sort, test_data.copy())
        
        # 测试快速排序
        _, quick_time = measure_execution_time(quick_sort, test_data.copy())
        
        # 测试归并排序
        _, merge_time = measure_execution_time(merge_sort, test_data.copy())
        
        print(f"{size:<10} {bubble_time:<15.6f} {quick_time:<15.6f} {merge_time:<15.6f}")

# 使用示例
time_performance_tests()
```

```python
import sys

def space_performance_tests():
    """空间性能测试示例"""
    
    def measure_memory_usage(func, *args, **kwargs):
        """测量函数内存使用"""
        # 记录初始内存使用
        initial_memory = sys.getsizeof(args[0]) if args else 0
        
        # 执行函数
        result = func(*args, **kwargs)
        
        # 记录结果内存使用
        result_memory = sys.getsizeof(result) if result else 0
        
        return result, result_memory - initial_memory
    
    def inefficient_fibonacci(n):
        """低效的斐波那契实现"""
        if n <= 1:
            return n
        return inefficient_fibonacci(n - 1) + inefficient_fibonacci(n - 2)
    
    def efficient_fibonacci(n):
        """高效的斐波那契实现"""
        if n <= 1:
            return n
        a, b = 0, 1
        for _ in range(2, n + 1):
            a, b = b, a + b
        return b
    
    def fibonacci_with_memoization(n, memo=None):
        """带记忆化的斐波那契实现"""
        if memo is None:
            memo = {}
        
        if n in memo:
            return memo[n]
        
        if n <= 1:
            return n
        
        memo[n] = fibonacci_with_memoization(n - 1, memo) + fibonacci_with_memoization(n - 2, memo)
        return memo[n]
    
    # 测试不同实现的内存使用
    test_values = [10, 20, 30]
    
    print("=== 斐波那契算法内存使用测试 ===")
    print(f"{'n值':<5} {'低效版本':<15} {'高效版本':<15} {'记忆化版本':<15}")

        # 测试低效版本

        _, efficient_memory = measure_memory_usage(efficient_fibonacci, n)
```

```python
def benchmark_tests():
    """基准测试示例"""
    
    class Benchmark:
        def __init__(self):
            self.results = {}
        
        def benchmark_function(self, func, name, *args, **kwargs):
            """对函数进行基准测试"""
            times = []
            
            # 多次运行取平均值
            for _ in range(10):
                start_time = time.time()
                result = func(*args, **kwargs)
                end_time = time.time()
                times.append(end_time - start_time)
            
            avg_time = sum(times) / len(times)
            min_time = min(times)
            max_time = max(times)
            
            self.results[name] = {
                'average': avg_time,
                'min': min_time,
                'max': max_time,
                'result': result
            }
            
            return result
        
        def compare_results(self):
            """比较测试结果"""
            print("=== 基准测试结果 ===")
            print(f"{'算法名称':<20} {'平均时间':<15} {'最小时间':<15} {'最大时间':<15}")
            print("-" * 70)
            
            for name, data in self.results.items():
                print(f"{name:<20} {data['average']:<15.6f} {data['min']:<15.6f} {data['max']:<15.6f}")
    
    def linear_search(arr, target):
        """线性搜索"""
        for i, val in enumerate(arr):
            if val == target:
                return i
        return -1
    
    def binary_search(arr, target):
        """二分搜索"""
        left, right = 0, len(arr) - 1
        
        while left <= right:
            mid = (left + right) // 2
            if arr[mid] == target:
                return mid
            elif arr[mid] < target:
                left = mid + 1
            else:
                right = mid - 1
        
        return -1
    
    def hash_search(arr, target):
        """哈希搜索"""
        hash_map = {val: i for i, val in enumerate(arr)}
        return hash_map.get(target, -1)
    
    # 创建基准测试
    benchmark = Benchmark()
    
    # 生成测试数据
    test_data = list(range(10000))
    target = 5000
    
    # 对搜索算法进行基准测试
    benchmark.benchmark_function(linear_search, "线性搜索", test_data, target)
    benchmark.benchmark_function(binary_search, "二分搜索", test_data, target)
    benchmark.benchmark_function(hash_search, "哈希搜索", test_data, target)
    
    # 比较结果
    benchmark.compare_results()

# 使用示例
benchmark_tests()
```

```python
import cProfile
import pstats

def performance_profiler_tests():
    """性能分析器测试示例"""
    
    def fibonacci_recursive(n):
        """递归斐波那契"""
        if n <= 1:
            return n
        return fibonacci_recursive(n - 1) + fibonacci_recursive(n - 2)
    
    def fibonacci_iterative(n):
        """迭代斐波那契"""
        if n <= 1:
            return n
        a, b = 0, 1
        for _ in range(2, n + 1):
            a, b = b, a + b
        return b
    
    def fibonacci_memoized(n, memo=None):
        """记忆化斐波那契"""
        if memo is None:
            memo = {}
        
        if n in memo:
            return memo[n]
        
        if n <= 1:
            return n
        
        memo[n] = fibonacci_memoized(n - 1, memo) + fibonacci_memoized(n - 2, memo)
        return memo[n]
    
    def profile_function(func, *args):
        """分析函数性能"""
        profiler = cProfile.Profile()
        profiler.enable()
        
        result = func(*args)
        
        profiler.disable()
        
        stats = pstats.Stats(profiler)
        stats.sort_stats('cumulative')
        
        return result, stats
    
    # 分析递归版本
    print("=== 递归斐波那契性能分析 ===")
    result_recursive, stats_recursive = profile_function(fibonacci_recursive, 20)
    stats_recursive.print_stats(10)
    
    # 分析迭代版本
    print("\n=== 迭代斐波那契性能分析 ===")
    result_iterative, stats_iterative = profile_function(fibonacci_iterative, 20)
    stats_iterative.print_stats(10)
    
    # 分析记忆化版本
    print("\n=== 记忆化斐波那契性能分析 ===")
    result_memoized, stats_memoized = profile_function(fibonacci_memoized, 20)
    stats_memoized.print_stats(10)
    
    # 验证结果正确性
    print(f"\n结果验证:")
    print(f"递归版本: {result_recursive}")
    print(f"迭代版本: {result_iterative}")
    print(f"记忆化版本: {result_memoized}")
    print(f"结果一致: {result_recursive == result_iterative == result_memoized}")

performance_profiler_tests()
```

```python
import tracemalloc

def memory_profiler_tests():
    """内存分析器测试示例"""
    
    def inefficient_list_creation(n):
        """低效的列表创建"""
        result = []
        for i in range(n):
            result.append(i ** 2)
        return result
    
    def efficient_list_creation(n):
        """高效的列表创建"""
        return [i ** 2 for i in range(n)]
    
    def inefficient_string_concatenation(n):
        """低效的字符串拼接"""
        result = ""
        for i in range(n):
            result += str(i)
        return result
    
    def efficient_string_concatenation(n):
        """高效的字符串拼接"""
        return "".join(str(i) for i in range(n))
    
    def profile_memory_usage(func, *args):
        """分析内存使用"""
        # 开始内存跟踪
        tracemalloc.start()
        
        # 执行函数
        result = func(*args)
        
        # 获取内存使用情况
        current, peak = tracemalloc.get_traced_memory()
        
        # 停止内存跟踪
        tracemalloc.stop()
        
        return result, current, peak
    
    # 分析列表创建的内存使用
    print("=== 列表创建内存分析 ===")
    
    n = 10000
    
    # 分析低效版本
    result_inefficient, current_inefficient, peak_inefficient = profile_memory_usage(inefficient_list_creation, n)
    
    # 分析高效版本
    result_efficient, current_efficient, peak_efficient = profile_memory_usage(efficient_list_creation, n)
    
    print(f"低效版本 - 当前内存: {current_inefficient / 1024 / 1024:.2f} MB, 峰值内存: {peak_inefficient / 1024 / 1024:.2f} MB")
    print(f"高效版本 - 当前内存: {current_efficient / 1024 / 1024:.2f} MB, 峰值内存: {peak_efficient / 1024 / 1024:.2f} MB")
    
    # 分析字符串拼接的内存使用
    print("\n=== 字符串拼接内存分析 ===")

    # 分析低效版本
    result_inefficient_str, current_inefficient_str, peak_inefficient_str = profile_memory_usage(inefficient_string_concatenation, n)

    result_efficient_str, current_efficient_str, peak_efficient_str = profile_memory_usage(efficient_string_concatenation, n)
```
