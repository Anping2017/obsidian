# 单元测试

## 概述

单元测试是算法实现中的重要实践，通过编写和执行单元测试可以验证算法的正确性、可靠性和性能。良好的单元测试是高质量算法实现的基础，也是专业程序员必备的技能。

## 核心概念

### 基本定义
- **单元测试**：对最小可测试单元进行测试
- **测试用例**：测试的具体场景和输入
- **断言**：验证测试结果的语句
- **测试覆盖率**：测试覆盖的代码比例

### 关键特性
- **独立性**：每个测试用例独立运行
- **可重复性**：测试结果可以重复
- **自动化**：可以自动执行测试
- **快速性**：测试执行速度快

## 单元测试分类

### 1. 基础单元测试

#### 1.1 简单函数测试
```python
import unittest

def simple_function_tests():
    """简单函数测试示例"""
    
    class TestSimpleFunctions(unittest.TestCase):
        def test_add_function(self):
            """测试加法函数"""
            def add(a, b):
                return a + b
            
            # 测试正常情况
            self.assertEqual(add(2, 3), 5)
            self.assertEqual(add(-1, 1), 0)
            self.assertEqual(add(0, 0), 0)
            
            # 测试边界情况
            self.assertEqual(add(-5, -3), -8)
            self.assertEqual(add(100, 200), 300)
        
        def test_factorial_function(self):
            """测试阶乘函数"""
            def factorial(n):
                if n < 0:
                    raise ValueError("n must be non-negative")
                if n <= 1:
                    return 1
                return n * factorial(n - 1)
            
            # 测试正常情况
            self.assertEqual(factorial(0), 1)
            self.assertEqual(factorial(1), 1)
            self.assertEqual(factorial(5), 120)
            
            # 测试异常情况
            with self.assertRaises(ValueError):
                factorial(-1)
        
        def test_is_prime_function(self):
            """测试素数判断函数"""
            def is_prime(n):
                if n < 2:
                    return False
                for i in range(2, int(n**0.5) + 1):
                    if n % i == 0:
                        return False
                return True
            
            # 测试素数
            self.assertTrue(is_prime(2))
            self.assertTrue(is_prime(3))
            self.assertTrue(is_prime(17))
            self.assertTrue(is_prime(97))
            
            # 测试非素数
            self.assertFalse(is_prime(1))
            self.assertFalse(is_prime(4))
            self.assertFalse(is_prime(15))
            self.assertFalse(is_prime(100))
    
    # 运行测试
    unittest.main(argv=[''], exit=False, verbosity=2)

# 使用示例
```

```python
def algorithm_function_tests():
    """算法函数测试示例"""
    
    class TestAlgorithms(unittest.TestCase):
        def test_binary_search(self):
            """测试二分搜索算法"""
            def binary_search(arr, target):
                left, right = 0, len(arr) - 1
                
                while left <= right:
                    mid = (left + right) // 2
                    if arr[mid] == target:
                        return mid
                    elif arr[mid] < target:
                        left = mid + 1
                    else:
                        right = mid - 1
                
                return -1
            
            # 测试正常情况
            arr = [1, 3, 5, 7, 9, 11, 13]
            self.assertEqual(binary_search(arr, 7), 3)
            self.assertEqual(binary_search(arr, 1), 0)
            self.assertEqual(binary_search(arr, 13), 6)
            
            # 测试边界情况
            self.assertEqual(binary_search(arr, 0), -1)
            self.assertEqual(binary_search(arr, 14), -1)
            
            # 测试空数组
            self.assertEqual(binary_search([], 5), -1)
            
            # 测试单元素数组
            self.assertEqual(binary_search([5], 5), 0)
            self.assertEqual(binary_search([5], 3), -1)
        
        def test_quick_sort(self):
            """测试快速排序算法"""
            def quick_sort(arr):
                if len(arr) <= 1:
                    return arr
                
                pivot = arr[len(arr) // 2]
                left = [x for x in arr if x < pivot]
                middle = [x for x in arr if x == pivot]
                right = [x for x in arr if x > pivot]
                
                return quick_sort(left) + middle + quick_sort(right)
            
            # 测试正常情况
            test_cases = [
                [64, 34, 25, 12, 22, 11, 90],
                [5, 2, 8, 1, 9],
                [1, 2, 3, 4, 5],
                [5, 4, 3, 2, 1]
            ]
            
            for test_case in test_cases:
                result = quick_sort(test_case.copy())
                expected = sorted(test_case)
                self.assertEqual(result, expected)
            
            # 测试边界情况
            self.assertEqual(quick_sort([]), [])
            self.assertEqual(quick_sort([5]), [5])
        
        def test_fibonacci(self):
            """测试斐波那契数列算法"""
            def fibonacci(n):
                if n < 0:
                    raise ValueError("n must be non-negative")
                if n <= 1:
                    return n
                return fibonacci(n - 1) + fibonacci(n - 2)
            
            # 测试正常情况
            self.assertEqual(fibonacci(0), 0)
            self.assertEqual(fibonacci(1), 1)
            self.assertEqual(fibonacci(5), 5)
            self.assertEqual(fibonacci(10), 55)
            
            # 测试异常情况
            with self.assertRaises(ValueError):
                fibonacci(-1)
    
    # 运行测试
    unittest.main(argv=[''], exit=False, verbosity=2)
```

#### 2.1 类测试
```python
def class_tests():
    """类测试示例"""
    
    class TestDataStructures(unittest.TestCase):
        def test_stack(self):
            """测试栈数据结构"""
            class Stack:
                def __init__(self):
                    self.items = []
                
                def push(self, item):
                    self.items.append(item)
                
                def pop(self):
                    if self.is_empty():
                        raise IndexError("Stack is empty")
                    return self.items.pop()
                
                def peek(self):
                    if self.is_empty():
                        raise IndexError("Stack is empty")
                    return self.items[-1]
                
                def is_empty(self):
                    return len(self.items) == 0
                
                def size(self):
                    return len(self.items)
            
            stack = Stack()
            
            # 测试空栈
            self.assertTrue(stack.is_empty())
            self.assertEqual(stack.size(), 0)
            
            # 测试入栈
            stack.push(1)
            stack.push(2)
            stack.push(3)
            
            self.assertFalse(stack.is_empty())
            self.assertEqual(stack.size(), 3)
            self.assertEqual(stack.peek(), 3)
            
            # 测试出栈
            self.assertEqual(stack.pop(), 3)
            self.assertEqual(stack.pop(), 2)
            self.assertEqual(stack.pop(), 1)
            
            self.assertTrue(stack.is_empty())
            
            # 测试异常情况
            with self.assertRaises(IndexError):
                stack.pop()
            
            with self.assertRaises(IndexError):
                stack.peek()
        
        def test_binary_tree(self):
            """测试二叉树数据结构"""
            class TreeNode:
                def __init__(self, val=0, left=None, right=None):
                    self.val = val
                    self.left = left
                    self.right = right
            
            class BinaryTree:
                def __init__(self):
                    self.root = None
                
                def insert(self, val):
                    if self.root is None:
                        self.root = TreeNode(val)
                    else:
                        self._insert_recursive(self.root, val)
                
                def _insert_recursive(self, node, val):
                    if val < node.val:
                        if node.left is None:
                            node.left = TreeNode(val)
                        else:
                            self._insert_recursive(node.left, val)
                    else:
                        if node.right is None:
                            node.right = TreeNode(val)
                        else:
                            self._insert_recursive(node.right, val)
                
                def search(self, val):
                    return self._search_recursive(self.root, val)
                
                def _search_recursive(self, node, val):
                    if node is None:
                        return False
                    if node.val == val:
                        return True
                    if val < node.val:
                        return self._search_recursive(node.left, val)
                    else:
                        return self._search_recursive(node.right, val)
            
            tree = BinaryTree()
            
            # 测试插入
            tree.insert(5)
            tree.insert(3)
            tree.insert(7)
            tree.insert(1)
            tree.insert(9)
            
            # 测试搜索
            self.assertTrue(tree.search(5))
            self.assertTrue(tree.search(3))
            self.assertTrue(tree.search(7))
            self.assertTrue(tree.search(1))
            self.assertTrue(tree.search(9))
            
            self.assertFalse(tree.search(2))
            self.assertFalse(tree.search(8))
            self.assertFalse(tree.search(10))
    
    # 运行测试
    unittest.main(argv=[''], exit=False, verbosity=2)
```

```python
def exception_tests():
    """异常测试示例"""
    
    class TestExceptions(unittest.TestCase):
        def test_division_by_zero(self):
            """测试除零异常"""
            def divide(a, b):
                if b == 0:
                    raise ValueError("Cannot divide by zero")
                return a / b
            
            # 测试正常情况
            self.assertEqual(divide(10, 2), 5)
            self.assertEqual(divide(15, 3), 5)
            
            # 测试异常情况
            with self.assertRaises(ValueError) as context:
                divide(10, 0)
            self.assertEqual(str(context.exception), "Cannot divide by zero")
        
        def test_invalid_input(self):
            """测试无效输入异常"""
            def validate_age(age):
                if not isinstance(age, int):
                    raise TypeError("Age must be an integer")
                if age < 0:
                    raise ValueError("Age cannot be negative")
                if age > 150:
                    raise ValueError("Age cannot be greater than 150")
                return True
            
            # 测试正常情况
            self.assertTrue(validate_age(25))
            self.assertTrue(validate_age(0))
            self.assertTrue(validate_age(150))
            
            # 测试类型异常
            with self.assertRaises(TypeError):
                validate_age("25")
            
            with self.assertRaises(TypeError):
                validate_age(25.5)
            
            # 测试值异常
            with self.assertRaises(ValueError):
                validate_age(-1)
            
            with self.assertRaises(ValueError):
                validate_age(151)
        
        def test_file_not_found(self):
            """测试文件未找到异常"""
            def read_file(filename):
                try:
                    with open(filename, 'r') as f:
                        return f.read()
                except FileNotFoundError:
                    raise FileNotFoundError(f"File '{filename}' not found")
            
            # 测试异常情况
            with self.assertRaises(FileNotFoundError):
                read_file("nonexistent_file.txt")
    
    # 运行测试
    unittest.main(argv=[''], exit=False, verbosity=2)
```

### 3. 性能测试

#### 3.1 时间复杂度测试
```python
import time

def performance_tests():
    """性能测试示例"""
    
    class TestPerformance(unittest.TestCase):
        def test_sorting_performance(self):
            """测试排序算法性能"""
            def bubble_sort(arr):
                n = len(arr)
                for i in range(n):
                    for j in range(0, n - i - 1):
                        if arr[j] > arr[j + 1]:
                            arr[j], arr[j + 1] = arr[j + 1], arr[j]
                return arr
            
            def quick_sort(arr):
                if len(arr) <= 1:
                    return arr
                pivot = arr[len(arr) // 2]
                left = [x for x in arr if x < pivot]
                middle = [x for x in arr if x == pivot]
                right = [x for x in arr if x > pivot]
                return quick_sort(left) + middle + quick_sort(right)
            
            # 测试小数据量
            small_data = [64, 34, 25, 12, 22, 11, 90]
            
            start_time = time.time()
            bubble_result = bubble_sort(small_data.copy())
            bubble_time = time.time() - start_time
            
            start_time = time.time()
            quick_result = quick_sort(small_data.copy())
            quick_time = time.time() - start_time
            
            # 验证结果正确性
            self.assertEqual(bubble_result, sorted(small_data))
            self.assertEqual(quick_result, sorted(small_data))
            
            # 验证性能差异
            print(f"冒泡排序时间: {bubble_time:.6f}秒")
            print(f"快速排序时间: {quick_time:.6f}秒")
        
        def test_search_performance(self):
            """测试搜索算法性能"""
            def linear_search(arr, target):
                for i, val in enumerate(arr):
                    if val == target:
                        return i
                return -1
            
            def binary_search(arr, target):
                left, right = 0, len(arr) - 1
                while left <= right:
                    mid = (left + right) // 2
                    if arr[mid] == target:
                        return mid
                    elif arr[mid] < target:
                        left = mid + 1
                    else:
                        right = mid - 1
                return -1
            
            # 测试大数据量
            large_data = list(range(10000))
            target = 5000
            
            start_time = time.time()
            linear_result = linear_search(large_data, target)
            linear_time = time.time() - start_time
            
            start_time = time.time()
            binary_result = binary_search(large_data, target)
            binary_time = time.time() - start_time
            
            # 验证结果正确性
            self.assertEqual(linear_result, target)
            self.assertEqual(binary_result, target)
            
            # 验证性能差异
            print(f"线性搜索时间: {linear_time:.6f}秒")
            print(f"二分搜索时间: {binary_time:.6f}秒")
    
    # 运行测试
    unittest.main(argv=[''], exit=False, verbosity=2)

# 使用示例
```

```python
import sys

def memory_tests():
    """内存使用测试示例"""
    
    class TestMemoryUsage(unittest.TestCase):
        def test_memory_efficiency(self):
            """测试内存效率"""
            def inefficient_fibonacci(n):
                """低效的斐波那契实现"""
                if n <= 1:
                    return n
                return inefficient_fibonacci(n - 1) + inefficient_fibonacci(n - 2)
            
            def efficient_fibonacci(n):
                """高效的斐波那契实现"""
                if n <= 1:
                    return n
                a, b = 0, 1
                for _ in range(2, n + 1):
                    a, b = b, a + b
                return b
            
            # 测试小数据量
            n = 20
            
            # 测试低效版本
            start_time = time.time()
            result_inefficient = inefficient_fibonacci(n)
            inefficient_time = time.time() - start_time
            
            # 测试高效版本
            start_time = time.time()
            result_efficient = efficient_fibonacci(n)
            efficient_time = time.time() - start_time
            
            # 验证结果正确性
            self.assertEqual(result_inefficient, result_efficient)
            
            # 验证性能差异
            print(f"低效版本时间: {inefficient_time:.6f}秒")
            print(f"高效版本时间: {efficient_time:.6f}秒")
            print(f"性能提升: {inefficient_time / efficient_time:.2f}倍")
        
        def test_data_structure_memory(self):
            """测试数据结构内存使用"""
            def create_list(n):
                return list(range(n))
            
            def create_generator(n):
                return (i for i in range(n))
            
            # 测试列表内存使用
            n = 1000
            my_list = create_list(n)
            list_size = sys.getsizeof(my_list)
            
            # 测试生成器内存使用
            my_generator = create_generator(n)
            generator_size = sys.getsizeof(my_generator)
            
            print(f"列表内存使用: {list_size} 字节")
            print(f"生成器内存使用: {generator_size} 字节")
            print(f"内存节省: {list_size / generator_size:.2f}倍")
    
    # 运行测试
    unittest.main(argv=[''], exit=False, verbosity=2)

memory_tests()
```

## 测试最佳实践

### 1. 测试设计原则
- **单一职责**：每个测试用例只测试一个功能
- **独立性**：测试用例之间相互独立
- **可重复性**：测试结果可以重复
- **快速性**：测试执行速度快

### 2. 测试用例设计
- **边界值测试**：测试边界情况
- **等价类测试**：测试等价类
- **异常测试**：测试异常情况
- **性能测试**：测试性能表现

### 3. 测试维护
- **及时更新**：代码修改时同步更新测试
- **重构测试**：定期重构测试代码
- **删除冗余**：删除重复的测试用例
- **优化测试**：优化测试执行效率
