# 贪心设计策略

## 概述

贪心设计策略（Greedy Design Strategy）是算法设计的重要方法，通过在每个步骤中做出局部最优选择来寻求全局最优解。贪心策略简单直观，在许多优化问题中都能得到很好的结果。

## 核心概念

### 基本定义
- **贪心策略**：在每个步骤中做出局部最优选择
- **贪心选择**：当前步骤的最优选择
- **最优子结构**：问题的最优解包含子问题的最优解
- **无后效性**：当前选择不影响后续选择

### 关键特性
- **局部最优**：每步选择局部最优解
- **全局最优**：局部最优解构成全局最优解
- **简单性**：算法简单直观
- **高效性**：通常具有较好的时间复杂度

# 活动选择问题的贪心策略
def activity_selection_strategy(activities: List[Tuple[int, int]]) -> Dict[str, Any]:
    """活动选择问题的贪心策略"""
    def activity_selection(acts: List[Tuple[int, int]]) -> List[Tuple[int, int]]:
        # 贪心选择：按结束时间排序
        acts.sort(key=lambda x: x[1])
        
        selected = [acts[0]]
        last_end_time = acts[0][1]
        
        for i in range(1, len(acts)):
            if acts[i][0] >= last_end_time:
                selected.append(acts[i])
                last_end_time = acts[i][1]
        
        return selected
    
    # 执行活动选择
    result = activity_selection(activities)
    
    return {
        'algorithm': 'activity_selection',
        'strategy': 'greedy',
        'greedy_choice': '选择结束时间最早的活动',
        'optimal_substructure': '剩余时间的最优解包含在全局最优解中',
        'time_complexity': 'O(n log n)',
        'space_complexity': 'O(1)',
        'result': result
    }

# 分数背包问题的贪心策略
def fractional_knapsack_strategy(items: List[Tuple[int, int]], capacity: int) -> Dict[str, Any]:
    """分数背包问题的贪心策略"""
    def fractional_knapsack(item_list: List[Tuple[int, int]], cap: int) -> Tuple[List[Tuple[int, float]], float]:
        # 贪心选择：按价值密度排序
        item_list.sort(key=lambda x: x[1] / x[0], reverse=True)
        
        selected = []
        total_value = 0
        remaining_capacity = cap
        
        for weight, value in item_list:
            if remaining_capacity >= weight:
                # 完全选择
                selected.append((weight, 1.0))
                total_value += value
                remaining_capacity -= weight
            else:
                # 部分选择
                fraction = remaining_capacity / weight
                selected.append((weight, fraction))
                total_value += value * fraction
                break
        
        return selected, total_value
    
    # 执行分数背包
    result, total_value = fractional_knapsack(items, capacity)
    
    return {
        'algorithm': 'fractional_knapsack',
        'strategy': 'greedy',
        'greedy_choice': '选择价值密度最高的物品',
        'optimal_substructure': '剩余容量的最优解包含在全局最优解中',
        'time_complexity': 'O(n log n)',
        'space_complexity': 'O(1)',
        'result': result,
        'total_value': total_value
    }

# 霍夫曼编码的贪心策略
def huffman_coding_strategy(frequencies: List[int]) -> Dict[str, Any]:
    """霍夫曼编码的贪心策略"""
    import heapq
    
    def huffman_coding(freqs: List[int]) -> List[str]:
        if len(freqs) == 1:
            return ['0']
        
        # 贪心选择：选择频率最小的两个节点
        heap = [(freq, i) for i, freq in enumerate(freqs)]
        heapq.heapify(heap)
        
        codes = [''] * len(freqs)
        
        while len(heap) > 1:
            freq1, node1 = heapq.heappop(heap)
            freq2, node2 = heapq.heappop(heap)
            
            # 合并节点
            merged_freq = freq1 + freq2
            merged_node = len(freqs)
            heapq.heappush(heap, (merged_freq, merged_node))
            
            # 更新编码
            for i in range(len(freqs)):
                if i == node1:
                    codes[i] = '0' + codes[i]
                elif i == node2:
                    codes[i] = '1' + codes[i]
        
        return codes
    
    # 执行霍夫曼编码
    result = huffman_coding(frequencies)
    
    return {
        'algorithm': 'huffman_coding',
        'strategy': 'greedy',
        'greedy_choice': '选择频率最小的两个节点合并',
        'optimal_substructure': '子问题的最优解包含在全局最优解中',
        'time_complexity': 'O(n log n)',
        'space_complexity': 'O(n)',
        'result': result
    }

# 最小生成树的贪心策略
def mst_strategy(edges: List[Tuple[int, int, int]], num_vertices: int) -> Dict[str, Any]:
    """最小生成树的贪心策略"""
    def kruskal_mst(edge_list: List[Tuple[int, int, int]], n: int) -> List[Tuple[int, int, int]]:
        # 贪心选择：按权重排序
        edge_list.sort(key=lambda x: x[2])
        
        parent = list(range(n))
        rank = [0] * n
        
        def find(x: int) -> int:
            if parent[x] != x:
                parent[x] = find(parent[x])
            return parent[x]
        
        def union(x: int, y: int) -> bool:
            px, py = find(x), find(y)
            if px == py:
                return False
            
            if rank[px] < rank[py]:
                parent[px] = py
            elif rank[px] > rank[py]:
                parent[py] = px
            else:
                parent[py] = px
                rank[px] += 1
            return True
        
        mst = []
        for u, v, weight in edge_list:
            if union(u, v):
                mst.append((u, v, weight))
                if len(mst) == n - 1:
                    break
        
        return mst
    
    # 执行Kruskal算法
    result = kruskal_mst(edges, num_vertices)
    
    return {
        'algorithm': 'kruskal_mst',
        'strategy': 'greedy',
        'greedy_choice': '选择权重最小的边',
        'optimal_substructure': '子图的最优解包含在全局最优解中',
        'time_complexity': 'O(E log E)',
        'space_complexity': 'O(V)',
        'result': result
    }

# 最短路径的贪心策略
def shortest_path_strategy(graph: List[List[int]], start: int) -> Dict[str, Any]:
    """最短路径的贪心策略"""
    def dijkstra(g: List[List[int]], s: int) -> List[int]:
        n = len(g)
        dist = [float('inf')] * n
        dist[s] = 0
        visited = [False] * n
        
        for _ in range(n):
            # 贪心选择：选择距离最小的未访问节点
            u = -1
            min_dist = float('inf')
            for v in range(n):
                if not visited[v] and dist[v] < min_dist:
                    min_dist = dist[v]
                    u = v
            
            if u == -1:
                break
            
            visited[u] = True
            
            # 更新邻接节点的距离
            for v in range(n):
                if not visited[v] and g[u][v] > 0:
                    dist[v] = min(dist[v], dist[u] + g[u][v])
        
        return dist
    
    # 执行Dijkstra算法
    result = dijkstra(graph, start)
    
    return {
        'algorithm': 'dijkstra',
        'strategy': 'greedy',
        'greedy_choice': '选择距离最小的未访问节点',
        'optimal_substructure': '子路径的最优解包含在全局最优解中',
        'time_complexity': 'O(V²)',
        'space_complexity': 'O(V)',
        'result': result
    }

# 任务调度的贪心策略
def task_scheduling_strategy(tasks: List[Tuple[int, int]]) -> Dict[str, Any]:
    """任务调度的贪心策略"""
    def task_scheduling(task_list: List[Tuple[int, int]]) -> List[Tuple[int, int]]:
        # 贪心选择：按截止时间排序
        task_list.sort(key=lambda x: x[1])
        
        scheduled = []
        current_time = 0
        
        for task in task_list:
            duration, deadline = task
            if current_time + duration <= deadline:
                scheduled.append(task)
                current_time += duration
        
        return scheduled
    
    # 执行任务调度
    result = task_scheduling(tasks)
    
    return {
        'algorithm': 'task_scheduling',
        'strategy': 'greedy',
        'greedy_choice': '选择截止时间最早的任务',

# 资源分配的贪心策略
def resource_allocation_strategy(resources: List[int], demands: List[int]) -> Dict[str, Any]:
    """资源分配的贪心策略"""
    def resource_allocation(resource_list: List[int], demand_list: List[int]) -> List[int]:
        # 贪心选择：按需求排序
        demand_indices = sorted(range(len(demand_list)), key=lambda i: demand_list[i])
        
        allocation = [0] * len(demand_list)
        remaining_resources = resource_list.copy()
        
        for i in demand_indices:
            demand = demand_list[i]
            for j in range(len(remaining_resources)):
                if remaining_resources[j] >= demand:
                    allocation[i] = j
                    remaining_resources[j] -= demand
                    break
        
        return allocation
    
    # 执行资源分配
    result = resource_allocation(resources, demands)
    
    return {
        'algorithm': 'resource_allocation',
        'strategy': 'greedy',
        'greedy_choice': '选择需求最小的任务',
        'optimal_substructure': '剩余资源的最优解包含在全局最优解中',
        'time_complexity': 'O(n log n)',
        'space_complexity': 'O(n)',
        'result': result
    }

## 优缺点分析

### 优点
- **简单直观**：算法思想简单易懂
- **高效性**：通常具有较好的时间复杂度
- **实现简单**：代码实现相对简单
- **适用广泛**：适用于多种优化问题

### 缺点
- **局部最优**：可能不是全局最优解
- **证明困难**：需要证明贪心选择的正确性
- **适用限制**：不适用于所有问题
- **依赖性质**：需要问题具有特定性质
