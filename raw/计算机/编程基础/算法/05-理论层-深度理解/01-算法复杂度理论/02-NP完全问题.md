# NP完全问题

## 概述

NP完全问题（NP-Complete Problems）是计算复杂性理论中的重要概念，指那些既属于NP类又具有NP难性的问题。NP完全问题是NP类中最困难的问题，如果能够找到一个NP完全问题的多项式时间算法，那么所有NP问题都可以在多项式时间内解决。

## 核心概念

### 基本定义
- **NP完全问题**：既属于NP类又具有NP难性的问题
- **NP难问题**：至少与NP类中最困难的问题一样难的问题
- **多项式时间归约**：将一个问题转换为另一个问题的方法
- **Cook-Levin定理**：证明SAT问题是NP完全的

### 关键特性
- **NP类**：可以在多项式时间内验证解
- **NP难性**：至少与NP类中最困难的问题一样难
- **归约性**：可以通过多项式时间归约相互转换
- **等价性**：所有NP完全问题在计算上等价

class NPCompleteProblems:
    """NP完全问题示例"""
    
    def __init__(self):
        self.problems = {}
        self.reductions = {}
    
    def add_problem(self, name: str, problem: Callable) -> None:
        """添加问题"""
        self.problems[name] = problem
    
    def add_reduction(self, from_problem: str, to_problem: str, reduction: Callable) -> None:
        """添加归约"""
        if from_problem not in self.reductions:
            self.reductions[from_problem] = {}
        self.reductions[from_problem][to_problem] = reduction
    
    def solve_problem(self, name: str, input_data: Any) -> Any:
        """解决问题"""
        if name not in self.problems:
            raise ValueError(f"未知问题: {name}")
        
        start_time = time.time()
        result = self.problems[name](input_data)
        end_time = time.time()
        
        return result, end_time - start_time
    
    def reduce_problem(self, from_problem: str, to_problem: str, input_data: Any) -> Any:
        """归约问题"""
        if from_problem not in self.reductions or to_problem not in self.reductions[from_problem]:
            raise ValueError(f"未知归约: {from_problem} -> {to_problem}")
        
        reduction = self.reductions[from_problem][to_problem]
        return reduction(input_data)

# SAT问题（可满足性问题）
def satisfiability_brute_force(clauses: List[List[int]], variables: int) -> Optional[List[bool]]:
    """可满足性问题 - O(2^n)"""
    # 尝试所有可能的赋值
    for i in range(2 ** variables):
        assignment = []
        for j in range(variables):
            assignment.append(bool(i & (1 << j)))
        
        # 检查是否满足所有子句
        satisfied = True
        for clause in clauses:
            clause_satisfied = False
            for literal in clause:
                var = abs(literal) - 1
                value = assignment[var]
                if literal > 0 and value:
                    clause_satisfied = True
                    break
                elif literal < 0 and not value:
                    clause_satisfied = True
                    break
            
            if not clause_satisfied:
                satisfied = False
                break
        
        if satisfied:
            return assignment
    
    return None

# 3-SAT问题
def three_sat_brute_force(clauses: List[List[int]], variables: int) -> Optional[List[bool]]:
    """3-SAT问题 - O(2^n)"""
    # 确保每个子句恰好有3个文字
    for clause in clauses:
        if len(clause) != 3:
            raise ValueError("每个子句必须有3个文字")
    
    return satisfiability_brute_force(clauses, variables)

# 子集和问题
def subset_sum_brute_force(numbers: List[int], target: int) -> Optional[List[int]]:
    """子集和问题 - O(2^n)"""
    n = len(numbers)
    best_subset = None
    best_sum = 0
    
    # 尝试所有可能的子集
    for i in range(2 ** n):
        subset = []
        current_sum = 0
        
        for j in range(n):
            if i & (1 << j):
                subset.append(numbers[j])
                current_sum += numbers[j]
        
        if current_sum == target:
            return subset
        elif current_sum < target and current_sum > best_sum:
            best_subset = subset
            best_sum = current_sum
    
    return best_subset

# 旅行商问题
def traveling_salesman_brute_force(distances: List[List[int]]) -> Optional[List[int]]:
    """旅行商问题 - O(n!)"""
    n = len(distances)
    if n == 0:
        return []
    
    best_path = None
    best_distance = float('inf')
    
    # 生成所有可能的路径
    def generate_paths(path: List[int], remaining: List[int]):
        nonlocal best_path, best_distance
        
        if not remaining:
            # 计算路径总距离
            total_distance = 0
            for i in range(len(path)):
                current_city = path[i]
                next_city = path[(i + 1) % len(path)]
                total_distance += distances[current_city][next_city]
            
            if total_distance < best_distance:
                best_path = path.copy()
                best_distance = total_distance
            return
        
        for city in remaining:
            new_path = path + [city]
            new_remaining = [c for c in remaining if c != city]
            generate_paths(new_path, new_remaining)
    
    generate_paths([0], list(range(1, n)))
    return best_path

# 图着色问题
def graph_coloring_brute_force(graph: List[List[int]], colors: int) -> Optional[List[int]]:
    """图着色问题 - O(colors^n)"""
    n = len(graph)
    best_coloring = None
    best_colors_used = float('inf')
    
    # 尝试所有可能的着色
    def try_coloring(coloring: List[int], vertex: int):
        nonlocal best_coloring, best_colors_used
        
        if vertex == n:
            # 检查着色是否有效
            is_valid = True
            for i in range(n):
                for j in graph[i]:
                    if coloring[i] == coloring[j]:
                        is_valid = False
                        break
                if not is_valid:
                    break
            
            if is_valid:
                colors_used = len(set(coloring))
                if colors_used < best_colors_used:
                    best_coloring = coloring.copy()
                    best_colors_used = colors_used
            return
        
        for color in range(colors):
            coloring[vertex] = color
            try_coloring(coloring, vertex + 1)
    
    try_coloring([0] * n, 0)
    return best_coloring

def hamiltonian_path_brute_force(graph: List[List[int]]) -> Optional[List[int]]:
    """哈密顿路径问题 - O(n!)"""
    n = len(graph)
    if n == 0:
        return []
    
    best_path = None
    
    # 生成所有可能的路径
    def generate_paths(path: List[int], remaining: List[int]):
        nonlocal best_path
        
        if not remaining:
            # 检查路径是否有效
            is_valid = True
            for i in range(len(path) - 1):
                current_city = path[i]
                next_city = path[i + 1]
                if next_city not in graph[current_city]:
                    is_valid = False
                    break
            
            if is_valid:
                best_path = path.copy()
            return
        
        for city in remaining:
            new_path = path + [city]
            new_remaining = [c for c in remaining if c != city]
            generate_paths(new_path, new_remaining)
    
    generate_paths([0], list(range(1, n)))
    return best_path

from typing import List, Tuple, Any, Callable, Optional
import time

class PolynomialTimeReduction:
    """多项式时间归约"""
    
    def __init__(self):
        self.reductions = {}
        self.reduction_graph = {}
    
    def add_reduction(self, from_problem: str, to_problem: str, reduction: Callable) -> None:
        """添加归约"""
        if from_problem not in self.reductions:
            self.reductions[from_problem] = {}
        self.reductions[from_problem][to_problem] = reduction
        
        # 构建归约图
        if from_problem not in self.reduction_graph:
            self.reduction_graph[from_problem] = []
        self.reduction_graph[from_problem].append(to_problem)
    
    def reduce(self, from_problem: str, to_problem: str, input_data: Any) -> Any:
        """执行归约"""
        if from_problem not in self.reductions or to_problem not in self.reductions[from_problem]:
            raise ValueError(f"未知归约: {from_problem} -> {to_problem}")
        
        reduction = self.reductions[from_problem][to_problem]
        return reduction(input_data)
    
    def find_reduction_path(self, from_problem: str, to_problem: str) -> List[str]:
        """查找归约路径"""
        if from_problem == to_problem:
            return [from_problem]
        
        # 使用BFS查找路径
        queue = [(from_problem, [from_problem])]
        visited = {from_problem}
        
        while queue:
            current_problem, path = queue.pop(0)
            
            if current_problem in self.reduction_graph:
                for next_problem in self.reduction_graph[current_problem]:
                    if next_problem == to_problem:
                        return path + [next_problem]
                    
                    if next_problem not in visited:
                        visited.add(next_problem)
                        queue.append((next_problem, path + [next_problem]))
        
        return []
    
    def is_reducible(self, from_problem: str, to_problem: str) -> bool:
        """判断是否可以归约"""
        return len(self.find_reduction_path(from_problem, to_problem)) > 0

# SAT到3-SAT的归约
def sat_to_3sat_reduction(clauses: List[List[int]], variables: int) -> Tuple[List[List[int]], int]:
    """SAT到3-SAT的归约"""
    new_clauses = []
    new_variables = variables
    
    for clause in clauses:
        if len(clause) == 1:
            # 单文字子句: x -> (x ∨ y ∨ z) ∧ (x ∨ ¬y ∨ z) ∧ (x ∨ y ∨ ¬z) ∧ (x ∨ ¬y ∨ ¬z)
            x = clause[0]
            y = new_variables + 1
            z = new_variables + 2
            new_variables += 2
            
            new_clauses.extend([
                [x, y, z],
                [x, -y, z],
                [x, y, -z],
                [x, -y, -z]
            ])
        elif len(clause) == 2:
            # 双文字子句: (x ∨ y) -> (x ∨ y ∨ z) ∧ (x ∨ y ∨ ¬z)
            x, y = clause
            z = new_variables + 1
            new_variables += 1
            
            new_clauses.extend([
                [x, y, z],
                [x, y, -z]
            ])
        elif len(clause) == 3:
            # 三文字子句: 直接添加
            new_clauses.append(clause)
        else:
            # 多文字子句: (x₁ ∨ x₂ ∨ ... ∨ xₖ) -> (x₁ ∨ x₂ ∨ y₁) ∧ (¬y₁ ∨ x₃ ∨ y₂) ∧ ... ∧ (¬yₖ₋₃ ∨ xₖ₋₁ ∨ xₖ)
            k = len(clause)
            new_clauses.append([clause[0], clause[1], new_variables + 1])
            
            for i in range(1, k - 2):
                new_clauses.append([-(new_variables + i), clause[i + 1], new_variables + i + 1])
            
            new_clauses.append([-(new_variables + k - 2), clause[k - 2], clause[k - 1]])
            new_variables += k - 2
    
    return new_clauses, new_variables

def three_sat_to_subset_sum_reduction(clauses: List[List[int]], variables: int) -> Tuple[List[int], int]:
    """3-SAT到子集和的归约"""
    n = len(clauses)
    m = variables
    
    # 创建数字列表
    numbers = []
    
    # 为每个变量创建两个数字（正文字和负文字）
    for i in range(m):
        # 正文字
        positive_number = 0
        for j in range(n):
            if i + 1 in clauses[j]:
                positive_number += 10 ** (n - j - 1)
        numbers.append(positive_number)
        
        # 负文字
        negative_number = 0
        for j in range(n):
            if -(i + 1) in clauses[j]:
                negative_number += 10 ** (n - j - 1)
        numbers.append(negative_number)
    
    # 为每个子句创建两个数字
    for j in range(n):
        clause_number = 10 ** (n - j - 1)
        numbers.append(clause_number)
        numbers.append(clause_number)
    
    # 计算目标值
    target = 0
    for j in range(n):
        target += 10 ** (n - j - 1)
    
    return numbers, target

def subset_sum_to_tsp_reduction(numbers: List[int], target: int) -> List[List[int]]:
    """子集和到旅行商的归约"""
    n = len(numbers)
    
    # 创建距离矩阵
    distances = [[0 for _ in range(n + 1)] for _ in range(n + 1)]
    
    # 设置距离
    for i in range(n):
        for j in range(n):
            if i == j:
                distances[i][j] = 0
            else:
                distances[i][j] = numbers[i] + numbers[j]
    
    # 设置到终点的距离
    for i in range(n):
        distances[i][n] = numbers[i]
        distances[n][i] = numbers[i]
    
    distances[n][n] = 0
    
    return distances
