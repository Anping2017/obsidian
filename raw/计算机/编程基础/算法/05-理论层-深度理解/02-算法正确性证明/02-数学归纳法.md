# 数学归纳法

## 概述

数学归纳法（Mathematical Induction）是证明算法正确性的重要工具，特别适用于递归算法和具有递推关系的算法。数学归纳法通过证明基础情况和归纳步骤来证明算法的正确性。

## 核心概念

### 基本定义
- **数学归纳法**：通过证明基础情况和归纳步骤来证明命题对所有自然数成立
- **基础情况**：证明命题对n=0或n=1成立
- **归纳假设**：假设命题对n=k成立
- **归纳步骤**：证明命题对n=k+1成立

### 关键特性
- **递归性**：适用于递归算法
- **递推性**：适用于具有递推关系的算法
- **完整性**：覆盖所有可能的情况
- **严谨性**：数学证明严谨

# 斐波那契数列的数学归纳法
def fibonacci_induction(n: int) -> Dict[str, Any]:
    """斐波那契数列的数学归纳法证明"""
    def fibonacci(n: int) -> int:
        if n <= 1:
            return n
        return fibonacci(n - 1) + fibonacci(n - 2)
    
    # 基础情况：n=0和n=1
    base_cases = {
        0: fibonacci(0),
        1: fibonacci(1)
    }
    
    # 归纳步骤：假设F(k)和F(k-1)正确，证明F(k+1)正确
    def induction_step(k: int) -> bool:
        if k <= 1:
            return True
        
        # 归纳假设：F(k) = F(k-1) + F(k-2)
        # 归纳步骤：F(k+1) = F(k) + F(k-1)
        f_k = fibonacci(k)
        f_k_minus_1 = fibonacci(k - 1)
        f_k_plus_1 = fibonacci(k + 1)
        
        return f_k_plus_1 == f_k + f_k_minus_1
    
    return {
        'algorithm': 'fibonacci',
        'base_cases': base_cases,
        'induction_hypothesis': 'F(k) = F(k-1) + F(k-2)',
        'induction_step': 'F(k+1) = F(k) + F(k-1)',
        'proof': '通过基础情况和归纳步骤证明斐波那契数列的正确性',
        'correctness': '算法正确计算斐波那契数列'
    }

# 阶乘的数学归纳法
def factorial_induction(n: int) -> Dict[str, Any]:
    """阶乘的数学归纳法证明"""
    def factorial(n: int) -> int:
        if n <= 1:
            return 1
        return n * factorial(n - 1)
    
    # 基础情况：n=0和n=1
    base_cases = {
        0: factorial(0),
        1: factorial(1)
    }
    
    # 归纳步骤：假设k!正确，证明(k+1)!正确
    def induction_step(k: int) -> bool:
        if k <= 1:
            return True
        
        # 归纳假设：k! = k * (k-1)!
        # 归纳步骤：(k+1)! = (k+1) * k!
        k_factorial = factorial(k)
        k_plus_1_factorial = factorial(k + 1)
        
        return k_plus_1_factorial == (k + 1) * k_factorial
    
    return {
        'algorithm': 'factorial',
        'base_cases': base_cases,
        'induction_hypothesis': 'k! = k * (k-1)!',
        'induction_step': '(k+1)! = (k+1) * k!',
        'proof': '通过基础情况和归纳步骤证明阶乘的正确性',
        'correctness': '算法正确计算阶乘'
    }

# 汉诺塔的数学归纳法
def hanoi_induction(n: int) -> Dict[str, Any]:
    """汉诺塔的数学归纳法证明"""
    def hanoi(n: int, source: str, destination: str, auxiliary: str) -> List[str]:
        moves = []
        
        if n == 1:
            moves.append(f"Move disk 1 from {source} to {destination}")
        else:
            moves.extend(hanoi(n - 1, source, auxiliary, destination))
            moves.append(f"Move disk {n} from {source} to {destination}")
            moves.extend(hanoi(n - 1, auxiliary, destination, source))
        
        return moves
    
    # 基础情况：n=1
    base_case = hanoi(1, "A", "C", "B")
    
    # 归纳步骤：假设n个盘子的移动正确，证明n+1个盘子的移动正确
    def induction_step(k: int) -> bool:
        if k <= 1:
            return True
        
        # 归纳假设：k个盘子需要2^k - 1步
        # 归纳步骤：k+1个盘子需要2^(k+1) - 1步
        k_moves = len(hanoi(k, "A", "C", "B"))
        k_plus_1_moves = len(hanoi(k + 1, "A", "C", "B"))
        
        return k_plus_1_moves == 2 * k_moves + 1
    
    return {
        'algorithm': 'hanoi',
        'base_cases': base_case,
        'induction_hypothesis': 'k个盘子需要2^k - 1步',
        'induction_step': 'k+1个盘子需要2^(k+1) - 1步',
        'proof': '通过基础情况和归纳步骤证明汉诺塔的正确性',
        'correctness': '算法正确解决汉诺塔问题'
    }

# 快速幂的数学归纳法
def fast_power_induction(base: int, exponent: int) -> Dict[str, Any]:
    """快速幂的数学归纳法证明"""
    def fast_power(base: int, exponent: int) -> int:
        if exponent == 0:
            return 1
        if exponent == 1:
            return base
        
        if exponent % 2 == 0:
            half = fast_power(base, exponent // 2)
            return half * half
        else:
            half = fast_power(base, exponent // 2)
            return half * half * base
    
    # 基础情况：exponent=0和exponent=1
    base_cases = {
        0: fast_power(base, 0),
        1: fast_power(base, 1)
    }
    
    # 归纳步骤：假设base^k正确，证明base^(k+1)正确
    def induction_step(k: int) -> bool:
        if k <= 1:
            return True
        
        # 归纳假设：base^k = base^(k//2) * base^(k//2) * base^(k%2)
        # 归纳步骤：base^(k+1) = base^k * base
        k_power = fast_power(base, k)
        k_plus_1_power = fast_power(base, k + 1)
        
        return k_plus_1_power == k_power * base
    
    return {
        'algorithm': 'fast_power',
        'base_cases': base_cases,
        'induction_hypothesis': 'base^k = base^(k//2) * base^(k//2) * base^(k%2)',
        'induction_step': 'base^(k+1) = base^k * base',
        'proof': '通过基础情况和归纳步骤证明快速幂的正确性',
        'correctness': '算法正确计算快速幂'
    }

# 最大公约数的数学归纳法
def gcd_induction(a: int, b: int) -> Dict[str, Any]:
    """最大公约数的数学归纳法证明"""
    def gcd(a: int, b: int) -> int:
        if b == 0:
            return a
        return gcd(b, a % b)
    
    # 基础情况：b=0
    base_case = gcd(a, b) if b == 0 else None
    
    # 归纳步骤：假设gcd(b, a%b)正确，证明gcd(a, b)正确
    def induction_step(a_val: int, b_val: int) -> bool:
        if b_val == 0:
            return True
        
        # 归纳假设：gcd(b, a%b) = gcd(a, b)
        # 归纳步骤：gcd(a, b) = gcd(b, a%b)
        gcd_result = gcd(a_val, b_val)
        gcd_recursive = gcd(b_val, a_val % b_val)
        
        return gcd_result == gcd_recursive
    
    return {
        'algorithm': 'gcd',
        'base_cases': base_case,
        'induction_hypothesis': 'gcd(b, a%b) = gcd(a, b)',
        'induction_step': 'gcd(a, b) = gcd(b, a%b)',
        'proof': '通过基础情况和归纳步骤证明最大公约数的正确性',
        'correctness': '算法正确计算最大公约数'
    }

# 归并排序的数学归纳法
def merge_sort_induction(data: List[int]) -> Dict[str, Any]:
    """归并排序的数学归纳法证明"""
    def merge_sort(arr: List[int], left: int, right: int) -> None:
        if left < right:
            mid = (left + right) // 2
            
            # 递归排序
            merge_sort(arr, left, mid)
            merge_sort(arr, mid + 1, right)
            
            # 合并
            merge(arr, left, mid, right)
    
    def merge(arr: List[int], left: int, mid: int, right: int) -> None:
        n1 = mid - left + 1
        n2 = right - mid
        
        L = [0] * n1
        R = [0] * n2
        
        for i in range(n1):
            L[i] = arr[left + i]
        for j in range(n2):
            R[j] = arr[mid + 1 + j]
        
        i = j = 0
        k = left
        
        while i < n1 and j < n2:
            if L[i] <= R[j]:
                arr[k] = L[i]
                i += 1
            else:
                arr[k] = R[j]
                j += 1
            k += 1
        
        while i < n1:
            arr[k] = L[i]
            i += 1
            k += 1
        
        while j < n2:
            arr[k] = R[j]
            j += 1
            k += 1
    
    # 基础情况：n=1
    base_case = len(data) == 1
    
    # 归纳步骤：假设长度为k的数组排序正确，证明长度为k+1的数组排序正确
    def induction_step(k: int) -> bool:
        if k <= 1:
            return True
        
        # 归纳假设：长度为k的数组排序正确
        # 归纳步骤：长度为k+1的数组排序正确
        test_data = list(range(k + 1, 0, -1))
        merge_sort(test_data, 0, len(test_data) - 1)
        
        return test_data == list(range(1, k + 2))
    
    merge_sort(data, 0, len(data) - 1)
    
    return {
        'algorithm': 'merge_sort',
        'base_cases': base_case,
        'induction_hypothesis': '长度为k的数组排序正确',
        'induction_step': '长度为k+1的数组排序正确',
        'proof': '通过基础情况和归纳步骤证明归并排序的正确性',
        'correctness': '算法正确排序整个数组'

def quicksort_induction(data: List[int]) -> Dict[str, Any]:
    """快速排序的数学归纳法证明"""
    def quicksort(arr: List[int], low: int, high: int) -> None:
        if low < high:
            # 分区
            pivot_index = partition(arr, low, high)
            
            # 递归排序
            quicksort(arr, low, pivot_index - 1)
            quicksort(arr, pivot_index + 1, high)
    
    def partition(arr: List[int], low: int, high: int) -> int:
        pivot = arr[high]
        i = low - 1
        
        for j in range(low, high):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        
        arr[i + 1], arr[high] = arr[high], arr[i + 1]
        return i + 1
    
    # 基础情况：n=1
    base_case = len(data) == 1
    
    # 归纳步骤：假设长度为k的数组排序正确，证明长度为k+1的数组排序正确

def heap_sort_induction(data: List[int]) -> Dict[str, Any]:
    """堆排序的数学归纳法证明"""
    def heapify(arr: List[int], n: int, i: int) -> None:
        largest = i
        left = 2 * i + 1
        right = 2 * i + 2
        
        if left < n and arr[left] > arr[largest]:
            largest = left
        
        if right < n and arr[right] > arr[largest]:
            largest = right
        
        if largest != i:
            arr[i], arr[largest] = arr[largest], arr[i]
            heapify(arr, n, largest)
    
    n = len(data)
    
    # 构建最大堆
    for i in range(n // 2 - 1, -1, -1):
        heapify(data, n, i)
    
    # 提取元素
    for i in range(n - 1, 0, -1):
        data[0], data[i] = data[i], data[0]
        heapify(data, i, 0)
    
    # 基础情况：n=1
    base_case = n == 1
    
    # 归纳步骤：假设长度为k的数组排序正确，证明长度为k+1的数组排序正确
    def induction_step(k: int) -> bool:

        # 归纳步骤：长度为k+1的数组排序正确

    def binary_search(arr: List[int], target: int, left: int, right: int) -> int:

            return -1

            return mid
        elif arr[mid] < target:
            return binary_search(arr, target, mid + 1, right)
        else:
            return binary_search(arr, target, left, mid - 1)

    base_case = len(data) == 1
    
    # 归纳步骤：假设长度为k的数组搜索正确，证明长度为k+1的数组搜索正确

        # 归纳步骤：长度为k+1的数组搜索正确
        test_data = list(range(1, k + 2))
        result = binary_search(test_data, k + 1, 0, len(test_data) - 1)
        
        return result == k
    
    result = binary_search(data, target, 0, len(data) - 1)
