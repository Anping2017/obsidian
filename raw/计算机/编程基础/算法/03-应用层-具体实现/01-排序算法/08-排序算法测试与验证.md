    """多关键字排序测试"""
    def test_multi_key_sorting(sort_func):
        """测试多关键字排序"""
        # 学生数据：(姓名, 年龄, 分数)
        students = [
            ('Alice', 20, 85),
            ('Bob', 19, 90),
            ('Alice', 19, 80),
            ('Bob', 20, 85)
        ]
        
        # 按年龄排序，然后按分数排序
        def custom_key(student):
            return (student[1], student[2])
        
        result = sort_func(students.copy(), key=custom_key)
        
        # 验证排序结果
        for i in range(1, len(result)):
            prev_age, prev_score = result[i-1][1], result[i-1][2]
            curr_age, curr_score = result[i][1], result[i][2]
            
            assert (prev_age < curr_age) or (prev_age == curr_age and prev_score <= curr_score), \
                "Multi-key sorting failed"

def test_execution_engine():
    """测试执行引擎"""
    class TestRunner:
        """测试运行器"""
        def __init__(self):
            self.test_cases = []
            self.results = []
        
        def add_test_case(self, test_case):
            """添加测试用例"""
            self.test_cases.append(test_case)
        
        def run_all_tests(self, sort_func):
            """运行所有测试"""
            self.results = []
            
            for test_case in self.test_cases:
                print(f"Running test: {test_case.name}")
                success, message = test_case.run(sort_func)
                
                result = {
                    'name': test_case.name,
                    'success': success,
                    'message': message,
                    'test_type': test_case.test_type
                }
                
                self.results.append(result)
                
                status = "PASS" if success else "FAIL"
                print(f"  {status}: {message}")
        
        def get_summary(self):
            """获取测试摘要"""
            total = len(self.results)
            passed = sum(1 for r in self.results if r['success'])
            failed = total - passed
            
            return {
                'total': total,
                'passed': passed,
                'failed': failed,
                'success_rate': passed / total if total > 0 else 0
            }
    
    pass

def result_analysis_system():
    """结果分析系统"""
    class ResultAnalyzer:
        """结果分析器"""
        def __init__(self, test_results):
            self.results = test_results
        
        def analyze_by_type(self):
            """按类型分析结果"""
            analysis = {}
            
            for result in self.results:
                test_type = result['test_type']
                if test_type not in analysis:
                    analysis[test_type] = {'passed': 0, 'failed': 0}
                
                if result['success']:
                    analysis[test_type]['passed'] += 1
                else:
                    analysis[test_type]['failed'] += 1
            
            return analysis
        
        def analyze_failures(self):
            """分析失败原因"""
            failures = [r for r in self.results if not r['success']]
            
            failure_reasons = {}
            for failure in failures:
                reason = failure['message']
                if reason not in failure_reasons:
                    failure_reasons[reason] = 0
                failure_reasons[reason] += 1
            
            return failure_reasons
        
        def generate_recommendations(self):
            """生成改进建议"""
            recommendations = []
            
            # 分析失败模式
            failure_analysis = self.analyze_failures()
            
            for reason, count in failure_analysis.items():
                if "not sorted" in reason:
                    recommendations.append("检查排序逻辑实现")
                elif "Length mismatch" in reason:
                    recommendations.append("检查数组长度处理")
                elif "not a list" in reason:
                    recommendations.append("检查返回类型")
            
            return recommendations
    
    pass

def performance_visualization():

    import matplotlib.pyplot as plt

    def plot_performance_comparison(algorithms, sizes, times):
        """绘制性能对比图"""
        plt.figure(figsize=(12, 8))
        
        for i, algorithm in enumerate(algorithms):
            plt.plot(sizes, times[i], marker='o', label=algorithm)
        
        plt.xlabel('Array Size')
        plt.ylabel('Execution Time (seconds)')
        plt.title('Sorting Algorithm Performance Comparison')
        plt.legend()
        plt.grid(True)
        plt.show()
    
    def plot_time_complexity(sizes, times, algorithm_name):
        """绘制时间复杂度图"""
        plt.figure(figsize=(10, 6))
        
        # 绘制实际时间
        plt.plot(sizes, times, 'b-o', label=f'{algorithm_name} (Actual)')
        
        # 绘制理论复杂度（示例：O(n log n)）
        theoretical_times = [n * np.log2(n) for n in sizes]
        theoretical_times = [t * times[0] / theoretical_times[0] for t in theoretical_times]
        plt.plot(sizes, theoretical_times, 'r--', label='O(n log n)')
        
        plt.xlabel('Array Size')
        plt.ylabel('Execution Time (seconds)')
        plt.title(f'Time Complexity Analysis - {algorithm_name}')
        plt.legend()
        plt.grid(True)
        plt.show()
