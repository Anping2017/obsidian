# 二分图匹配

## 🎯 核心概念

**二分图匹配(Bipartite Graph Matching)**是在二分图中寻找最大匹配的问题。二分图是指顶点集可以分成两个不相交的子集，使得每条边都连接两个不同子集中的顶点。匹配是指一组边，其中任意两条边都不共享顶点。二分图匹配在任务分配、资源调度、网络流等领域有重要应用。

### 1. 匈牙利算法
```python
def hungarian_algorithm():
    """匈牙利算法实现"""
    
    def hungarian_matching(graph, left_vertices, right_vertices):
        """匈牙利算法求最大匹配"""
        # 初始化匹配
        match_right = {v: None for v in right_vertices}
        match_left = {v: None for v in left_vertices}
        
        def dfs(u, visited):
            """DFS寻找增广路径"""
            for v in graph[u]:
                if v not in visited:
                    visited.add(v)
                    if match_right[v] is None or dfs(match_right[v], visited):
                        match_right[v] = u
                        match_left[u] = v
                        return True
            return False
        
        # 对每个左顶点寻找增广路径
        for u in left_vertices:
            visited = set()
            dfs(u, visited)
        
        return match_left, match_right
    
    def hungarian_with_path_reconstruction(graph, left_vertices, right_vertices):
        """带路径重构的匈牙利算法"""
        match_right = {v: None for v in right_vertices}
        match_left = {v: None for v in left_vertices}
        paths = []
        
        def dfs_with_path(u, visited, path):
            """DFS寻找增广路径并记录路径"""
            for v in graph[u]:
                if v not in visited:
                    visited.add(v)
                    new_path = path + [v]
                    if match_right[v] is None:
                        # 找到增广路径
                        paths.append(new_path)
                        match_right[v] = u
                        match_left[u] = v
                        return True
                    elif dfs_with_path(match_right[v], visited, new_path):
                        match_right[v] = u
                        match_left[u] = v
                        return True
            return False
        
        for u in left_vertices:
            visited = set()
            dfs_with_path(u, visited, [u])
        
        return match_left, match_right, paths
    
    def hungarian_with_weighted_matching(graph, weights, left_vertices, right_vertices):
        """带权重的匈牙利算法"""
        # 使用最小费用最大流算法
        def min_cost_max_flow(graph, weights, left_vertices, right_vertices):
            # 构建网络流图
            flow_graph = {}
            capacity = {}
            cost = {}
            
            # 添加源点和汇点
            source = 's'
            sink = 't'
            
            # 连接源点到左顶点
            for u in left_vertices:
                flow_graph[source] = flow_graph.get(source, []) + [u]
                capacity[(source, u)] = 1
                cost[(source, u)] = 0
            
            # 连接右顶点到汇点
            for v in right_vertices:
                flow_graph[v] = flow_graph.get(v, []) + [sink]
                capacity[(v, sink)] = 1
                cost[(v, sink)] = 0
            
            # 连接左右顶点
            for u in left_vertices:
                for v in graph[u]:
                    flow_graph[u] = flow_graph.get(u, []) + [v]
                    capacity[(u, v)] = 1
                    cost[(u, v)] = weights.get((u, v), 0)
            
            # 使用Ford-Fulkerson算法
            return ford_fulkerson(flow_graph, capacity, cost, source, sink)
        
        return min_cost_max_flow(graph, weights, left_vertices, right_vertices)
    
    return hungarian_matching, hungarian_with_path_reconstruction, hungarian_with_weighted_matching
```

### 2. Hopcroft-Karp算法
```python
def hopcroft_karp_algorithm():
    """Hopcroft-Karp算法实现"""
    
    def hopcroft_karp_matching(graph, left_vertices, right_vertices):
        """Hopcroft-Karp算法求最大匹配"""
        match_right = {v: None for v in right_vertices}
        match_left = {v: None for v in left_vertices}
        
        def bfs():
            """BFS寻找最短增广路径"""
            queue = []
            dist = {}
            
            for u in left_vertices:
                if match_left[u] is None:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = float('inf')
            
            dist[None] = float('inf')
            
            while queue:
                u = queue.pop(0)
                if u is not None:
                    for v in graph[u]:
                        if dist[match_right[v]] == float('inf'):
                            dist[match_right[v]] = dist[u] + 1
                            queue.append(match_right[v])
            
            return dist[None] != float('inf')
        
        def dfs(u, dist):
            """DFS寻找增广路径"""
            if u is None:
                return True
            
            for v in graph[u]:
                if dist[match_right[v]] == dist[u] + 1:
                    if dfs(match_right[v], dist):
                        match_right[v] = u
                        match_left[u] = v
                        return True
            
            dist[u] = float('inf')
            return False
        
        # 主算法
        matching_size = 0
        while bfs():
            for u in left_vertices:
                if match_left[u] is None:
                    if dfs(u, {}):
                        matching_size += 1
        
        return match_left, match_right, matching_size
    
    def hopcroft_karp_with_paths(graph, left_vertices, right_vertices):
        """带路径的Hopcroft-Karp算法"""
        match_right = {v: None for v in right_vertices}
        match_left = {v: None for v in left_vertices}
        paths = []
        
        def bfs_with_paths():
            """BFS寻找最短增广路径"""
            queue = []
            dist = {}
            parent = {}
            
            for u in left_vertices:
                if match_left[u] is None:
                    dist[u] = 0
                    queue.append(u)
                    parent[u] = None
                else:
                    dist[u] = float('inf')
            
            dist[None] = float('inf')
            
            while queue:
                u = queue.pop(0)
                if u is not None:
                    for v in graph[u]:
                        if dist[match_right[v]] == float('inf'):
                            dist[match_right[v]] = dist[u] + 1
                            parent[match_right[v]] = u
                            queue.append(match_right[v])
            
            return dist[None] != float('inf'), parent
        
        def dfs_with_paths(u, dist, parent):
            """DFS寻找增广路径并记录路径"""
            if u is None:
                return True
            
            for v in graph[u]:
                if dist[match_right[v]] == dist[u] + 1:
                    if dfs_with_paths(match_right[v], dist, parent):
                        match_right[v] = u
                        match_left[u] = v
                        return True
            
            dist[u] = float('inf')
            return False
        
        # 主算法
        matching_size = 0
        while True:
            has_path, parent = bfs_with_paths()
            if not has_path:
                break
            
            for u in left_vertices:
                if match_left[u] is None:
                    if dfs_with_paths(u, {}, parent):
                        matching_size += 1
        
        return match_left, match_right, matching_size, paths
    
    def hopcroft_karp_with_weights(graph, weights, left_vertices, right_vertices):
        """带权重的Hopcroft-Karp算法"""
        # 使用最小费用最大流算法
        def min_cost_max_flow_hopcroft(graph, weights, left_vertices, right_vertices):
            # 构建网络流图
            flow_graph = {}
            capacity = {}
            cost = {}
            
            source = 's'
            sink = 't'
            
            # 连接源点到左顶点
            for u in left_vertices:
                flow_graph[source] = flow_graph.get(source, []) + [u]
                capacity[(source, u)] = 1
                cost[(source, u)] = 0
            
            # 连接右顶点到汇点
            for v in right_vertices:
                flow_graph[v] = flow_graph.get(v, []) + [sink]
                capacity[(v, sink)] = 1
                cost[(v, sink)] = 0
            
            # 连接左右顶点
            for u in left_vertices:
                for v in graph[u]:
                    flow_graph[u] = flow_graph.get(u, []) + [v]
                    capacity[(u, v)] = 1
                    cost[(u, v)] = weights.get((u, v), 0)
            
            # 使用Ford-Fulkerson算法
            return ford_fulkerson(flow_graph, capacity, cost, source, sink)
        
        return min_cost_max_flow_hopcroft(graph, weights, left_vertices, right_vertices)
    
    return hopcroft_karp_matching, hopcroft_karp_with_paths, hopcroft_karp_with_weights
```

```python
def network_flow_algorithm():
    """网络流算法实现"""
    
    def ford_fulkerson(graph, capacity, source, sink):
        """Ford-Fulkerson最大流算法"""
        def bfs_path(graph, capacity, source, sink, parent):
            """BFS寻找增广路径"""
            visited = set()
            queue = [source]
            visited.add(source)
            
            while queue:
                u = queue.pop(0)
                for v in graph.get(u, []):
                    if v not in visited and capacity.get((u, v), 0) > 0:
                        visited.add(v)
                        parent[v] = u
                        if v == sink:
                            return True
                        queue.append(v)
            
            return False
        
        parent = {}
        max_flow = 0
        
        while bfs_path(graph, capacity, source, sink, parent):
            # 计算路径上的最小容量
            path_flow = float('inf')
            v = sink
            while v != source:
                u = parent[v]
                path_flow = min(path_flow, capacity.get((u, v), 0))
                v = u
            
            # 更新流量
            v = sink
            while v != source:
                u = parent[v]
                capacity[(u, v)] -= path_flow
                capacity[(v, u)] = capacity.get((v, u), 0) + path_flow
                v = u
            
            max_flow += path_flow
        
        return max_flow
    
    def edmonds_karp(graph, capacity, source, sink):
        """Edmonds-Karp最大流算法"""
        def bfs_path(graph, capacity, source, sink, parent):
            """BFS寻找最短增广路径"""
            visited = set()
            queue = [source]
            visited.add(source)
            
            while queue:
                u = queue.pop(0)
                for v in graph.get(u, []):
                    if v not in visited and capacity.get((u, v), 0) > 0:
                        visited.add(v)
                        parent[v] = u
                        if v == sink:
                            return True
                        queue.append(v)
            
            return False
        
        parent = {}
        max_flow = 0
        
        while bfs_path(graph, capacity, source, sink, parent):
            # 计算路径上的最小容量
            path_flow = float('inf')
            v = sink
            while v != source:
                u = parent[v]
                path_flow = min(path_flow, capacity.get((u, v), 0))
                v = u
            
            # 更新流量
            v = sink
            while v != source:
                u = parent[v]
                capacity[(u, v)] -= path_flow
                capacity[(v, u)] = capacity.get((v, u), 0) + path_flow
                v = u
            
            max_flow += path_flow
        
        return max_flow
    
    def dinic_algorithm(graph, capacity, source, sink):
        """Dinic最大流算法"""
        def bfs_level(graph, capacity, source, sink, level):
            """BFS计算层次"""
            visited = set()
            queue = [source]
            visited.add(source)
            level[source] = 0
            
            while queue:
                u = queue.pop(0)
                for v in graph.get(u, []):
                    if v not in visited and capacity.get((u, v), 0) > 0:
                        visited.add(v)
                        level[v] = level[u] + 1
                        if v == sink:
                            return True
                        queue.append(v)
            
            return False
        
        def dfs_blocking_flow(graph, capacity, source, sink, level, u, flow):
            """DFS寻找阻塞流"""
            if u == sink:
                return flow
            
            for v in graph.get(u, []):
                if level[v] == level[u] + 1 and capacity.get((u, v), 0) > 0:
                    path_flow = dfs_blocking_flow(graph, capacity, source, sink, level, v, min(flow, capacity.get((u, v), 0)))
                    if path_flow > 0:
                        capacity[(u, v)] -= path_flow
                        capacity[(v, u)] = capacity.get((v, u), 0) + path_flow
                        return path_flow
            
            return 0
        
        max_flow = 0
        level = {}
        
        while bfs_level(graph, capacity, source, sink, level):
            while True:
                flow = dfs_blocking_flow(graph, capacity, source, sink, level, source, float('inf'))
                if flow == 0:
                    break
                max_flow += flow
        
        return max_flow
    
    return ford_fulkerson, edmonds_karp, dinic_algorithm
```

```python
        """最大流问题"""
        # 使用Ford-Fulkerson算法
        max_flow = ford_fulkerson(graph, capacity, source, sink)
        
        # 分析流网络
        flow_analysis = analyze_flow_network(max_flow, graph, capacity)
        return flow_analysis
    
    def min_cost_flow(graph, capacity, cost, source, sink, demand):
        """最小费用流问题"""
        # 使用最小费用最大流算法
        min_cost = min_cost_max_flow(graph, capacity, cost, source, sink, demand)
        
        # 分析费用流
        cost_analysis = analyze_cost_flow(min_cost, graph, capacity, cost)
        return cost_analysis
    
    def multi_commodity_flow(graph, capacity, commodities):
        """多商品流问题"""
        # 使用多商品流算法
        multi_flow = multi_commodity_flow_algorithm(graph, capacity, commodities)
        
        # 分析多商品流
        flow_analysis = analyze_multi_commodity_flow(multi_flow, graph, capacity, commodities)
        return flow_analysis
    
    return max_flow_problem, min_cost_flow, multi_commodity_flow
```
