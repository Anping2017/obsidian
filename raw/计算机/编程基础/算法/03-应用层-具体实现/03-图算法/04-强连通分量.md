# 强连通分量

## 🎯 核心概念

**强连通分量(Strongly Connected Components, SCC)**是有向图中最大的强连通子图。在有向图中，如果从顶点u到顶点v存在路径，且从顶点v到顶点u也存在路径，则称u和v强连通。强连通分量在社交网络分析、编译器优化、网络分析等领域有重要应用。

### 1. Tarjan算法
```python
def tarjan_algorithm():
    """Tarjan算法实现"""
    
    def tarjan_scc(graph):
        """Tarjan强连通分量算法"""
        index = 0
        stack = []
        indices = {}
        lowlinks = {}
        on_stack = set()
        sccs = []
        
        def strongconnect(vertex):
            nonlocal index
            indices[vertex] = index
            lowlinks[vertex] = index
            index += 1
            stack.append(vertex)
            on_stack.add(vertex)
            
            for neighbor in graph[vertex]:
                if neighbor not in indices:
                    strongconnect(neighbor)
                    lowlinks[vertex] = min(lowlinks[vertex], lowlinks[neighbor])
                elif neighbor in on_stack:
                    lowlinks[vertex] = min(lowlinks[vertex], indices[neighbor])
            
            if lowlinks[vertex] == indices[vertex]:
                scc = []
                while True:
                    w = stack.pop()
                    on_stack.remove(w)
                    scc.append(w)
                    if w == vertex:
                        break
                sccs.append(scc)
        
        for vertex in graph:
            if vertex not in indices:
                strongconnect(vertex)
        
        return sccs
    
    def tarjan_with_component_graph(graph):
        """带分量图的Tarjan算法"""
        sccs = tarjan_scc(graph)
        
        # 构建分量图
        vertex_to_component = {}
        for i, scc in enumerate(sccs):
            for vertex in scc:
                vertex_to_component[vertex] = i
        
        component_graph = {i: [] for i in range(len(sccs))}
        
        for vertex in graph:
            for neighbor in graph[vertex]:
                comp_vertex = vertex_to_component[vertex]
                comp_neighbor = vertex_to_component[neighbor]
                if comp_vertex != comp_neighbor:
                    component_graph[comp_vertex].append(comp_neighbor)
        
        return sccs, component_graph
    
    def tarjan_with_topological_sort(graph):
        """带拓扑排序的Tarjan算法"""
        sccs = tarjan_scc(graph)
        
        # 构建分量图
        vertex_to_component = {}
        for i, scc in enumerate(sccs):
            for vertex in scc:
                vertex_to_component[vertex] = i
        
        component_graph = {i: [] for i in range(len(sccs))}
        
        for vertex in graph:
            for neighbor in graph[vertex]:
                comp_vertex = vertex_to_component[vertex]
                comp_neighbor = vertex_to_component[neighbor]
                if comp_vertex != comp_neighbor:
                    component_graph[comp_vertex].append(comp_neighbor)
        
        # 对分量图进行拓扑排序
        def topological_sort(comp_graph):
            in_degree = {i: 0 for i in comp_graph}
            for i in comp_graph:
                for j in comp_graph[i]:
                    in_degree[j] += 1
            
            queue = [i for i in comp_graph if in_degree[i] == 0]
            result = []
            
            while queue:
                current = queue.pop(0)
                result.append(current)
                
                for neighbor in comp_graph[current]:
                    in_degree[neighbor] -= 1
                    if in_degree[neighbor] == 0:
                        queue.append(neighbor)
            
            return result
        
        topo_order = topological_sort(component_graph)
        return sccs, component_graph, topo_order
    
    return tarjan_scc, tarjan_with_component_graph, tarjan_with_topological_sort
```

```python
def kosaraju_algorithm():
    """Kosaraju算法实现"""
    
    def kosaraju_scc(graph):
        """Kosaraju强连通分量算法"""
        # 第一步：计算图的转置
        def transpose_graph(graph):
            transposed = {vertex: [] for vertex in graph}
            for vertex in graph:
                for neighbor in graph[vertex]:
                    transposed[neighbor].append(vertex)
            return transposed
        
        # 第二步：对原图进行DFS，记录完成时间
        visited = set()
        finish_order = []
        
        def dfs1(vertex):
            if vertex in visited:
                return
            visited.add(vertex)
            for neighbor in graph[vertex]:
                dfs1(neighbor)
            finish_order.append(vertex)
        
        for vertex in graph:
            if vertex not in visited:
                dfs1(vertex)
        
        # 第三步：对转置图按完成时间逆序进行DFS
        transposed = transpose_graph(graph)
        visited = set()
        sccs = []
        
        def dfs2(vertex, scc):
            if vertex in visited:
                return
            visited.add(vertex)
            scc.append(vertex)
            for neighbor in transposed[vertex]:
                dfs2(neighbor, scc)
        
        for vertex in reversed(finish_order):
            if vertex not in visited:
                scc = []
                dfs2(vertex, scc)
                sccs.append(scc)
        
        return sccs
    
    def kosaraju_with_iterative_dfs(graph):
        """使用迭代DFS的Kosaraju算法"""
        def transpose_graph(graph):
            transposed = {vertex: [] for vertex in graph}
            for vertex in graph:
                for neighbor in graph[vertex]:
                    transposed[neighbor].append(vertex)
            return transposed
        
        # 迭代DFS
        def dfs1_iterative(vertex, visited, finish_order):
            stack = [vertex]
            while stack:
                current = stack.pop()
                if current in visited:
                    continue
                visited.add(current)
                stack.append(current)  # 重新入栈用于后序遍历
                
                for neighbor in graph[current]:
                    if neighbor not in visited:
                        stack.append(neighbor)
                
                # 检查是否所有邻居都已访问
                all_neighbors_visited = True
                for neighbor in graph[current]:
                    if neighbor not in visited:
                        all_neighbors_visited = False
                        break
                
                if all_neighbors_visited:
                    finish_order.append(current)
        
        visited = set()
        finish_order = []
        
        for vertex in graph:
            if vertex not in visited:
                dfs1_iterative(vertex, visited, finish_order)
        
        transposed = transpose_graph(graph)
        visited = set()
        sccs = []
        
        def dfs2_iterative(vertex, visited):
            stack = [vertex]
            scc = []
            while stack:
                current = stack.pop()
                if current in visited:
                    continue
                visited.add(current)
                scc.append(current)
                
                for neighbor in transposed[current]:
                    if neighbor not in visited:
                        stack.append(neighbor)
            
            return scc
        
        for vertex in reversed(finish_order):
            if vertex not in visited:
                scc = dfs2_iterative(vertex, visited)
                sccs.append(scc)
        
        return sccs
    
    def kosaraju_with_component_graph(graph):
        """带分量图的Kosaraju算法"""
        sccs = kosaraju_scc(graph)
        
        # 构建分量图
        vertex_to_component = {}
        for i, scc in enumerate(sccs):
            for vertex in scc:
                vertex_to_component[vertex] = i
        
        component_graph = {i: [] for i in range(len(sccs))}
        
        for vertex in graph:
            for neighbor in graph[vertex]:
                comp_vertex = vertex_to_component[vertex]
                comp_neighbor = vertex_to_component[neighbor]
                if comp_vertex != comp_neighbor:
                    component_graph[comp_vertex].append(comp_neighbor)
        
        return sccs, component_graph
    
    return kosaraju_scc, kosaraju_with_iterative_dfs, kosaraju_with_component_graph
```

### 3. Gabow算法
```python
def gabow_algorithm():
    """Gabow算法实现"""
    
    def gabow_scc(graph):
        """Gabow强连通分量算法"""
        index = 0
        path_stack = []
        scc_stack = []
        indices = {}
        sccs = []
        
        def strongconnect(vertex):
            nonlocal index
            indices[vertex] = index
            index += 1
            path_stack.append(vertex)
            scc_stack.append(vertex)
            
            for neighbor in graph[vertex]:
                if neighbor not in indices:
                    strongconnect(neighbor)
                elif neighbor in scc_stack:
                    # 找到路径栈中neighbor的位置
                    while path_stack and indices[path_stack[-1]] > indices[neighbor]:
                        path_stack.pop()
            
            if path_stack and path_stack[-1] == vertex:
                scc = []
                while True:
                    w = scc_stack.pop()
                    scc.append(w)
                    if w == vertex:
                        break
                sccs.append(scc)
                path_stack.pop()
        
        for vertex in graph:
            if vertex not in indices:
                strongconnect(vertex)
        
        return sccs
    
    def gabow_with_component_graph(graph):
        """带分量图的Gabow算法"""
        sccs = gabow_scc(graph)
        
        # 构建分量图

                result.append(current)
                
                for neighbor in comp_graph[current]:
                    in_degree[neighbor] -= 1
                    if in_degree[neighbor] == 0:
                        queue.append(neighbor)
            
            return result
        
        topo_order = topological_sort(component_graph)
        return sccs, component_graph, topo_order
    
    return gabow_scc, gabow_with_component_graph, gabow_with_topological_sort
```
