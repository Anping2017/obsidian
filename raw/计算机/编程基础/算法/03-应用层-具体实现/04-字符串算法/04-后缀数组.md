# 后缀数组

## 🎯 核心概念

**后缀数组(Suffix Array)**是一种字符串数据结构，存储字符串所有后缀的排序信息。后缀数组是后缀树的替代方案，具有更小的空间复杂度，在字符串匹配、最长公共子串、字符串排序等领域有重要应用。

### 2. 后缀数组构建
```python
def suffix_array_construction():
    """后缀数组构建"""
    
    def naive_suffix_array(text):
        """朴素后缀数组构建"""
        n = len(text)
        suffixes = []
        
        # 生成所有后缀
        for i in range(n):
            suffixes.append((text[i:], i))
        
        # 按字典序排序
        suffixes.sort()
        
        # 提取索引
        suffix_array = [suffix[1] for suffix in suffixes]
        return suffix_array
    
    def efficient_suffix_array(text):
        """高效后缀数组构建"""
        n = len(text)
        
        # 使用基数排序
        def radix_sort(suffixes, k):
            """基数排序"""
            # 按第k个字符排序
            suffixes.sort(key=lambda x: text[x + k] if x + k < n else -1)
            return suffixes
        
        # 初始后缀数组
        suffix_array = list(range(n))
        
        # 多轮排序
        k = 1
        while k < n:
            suffix_array = radix_sort(suffix_array, k)
            k *= 2
        
        return suffix_array
    
    def dc3_suffix_array(text):
        """DC3后缀数组构建"""
        n = len(text)
        
        # 添加哨兵字符
        text += '$'
        n += 1
        
        # 构建三元组
        triples = []
        for i in range(n):
            if i % 3 == 0:
                triples.append((i, text[i], text[i + 1] if i + 1 < n else '$', text[i + 2] if i + 2 < n else '$'))
        
        # 排序三元组
        triples.sort(key=lambda x: (x[1], x[2], x[3]))
        
        # 构建后缀数组
        suffix_array = [triple[0] for triple in triples]
        return suffix_array
    
    return naive_suffix_array, efficient_suffix_array, dc3_suffix_array
```

```python
def basic_suffix_array():
    """基本后缀数组实现"""
    
    def build_suffix_array(text):
        """构建后缀数组"""
        n = len(text)
        suffixes = []
        
        # 生成所有后缀
        for i in range(n):
            suffixes.append((text[i:], i))
        
        # 按字典序排序
        suffixes.sort()
        
        # 提取索引
        suffix_array = [suffix[1] for suffix in suffixes]
        return suffix_array
    
    def build_suffix_array_with_lcp(text):
        """构建后缀数组和LCP数组"""
        n = len(text)
        suffixes = []
        
        # 生成所有后缀
        for i in range(n):
            suffixes.append((text[i:], i))
        
        # 按字典序排序
        suffixes.sort()
        
        # 提取索引
        suffix_array = [suffix[1] for suffix in suffixes]
        
        # 构建LCP数组
        lcp = [0] * n
        for i in range(1, n):
            lcp[i] = longest_common_prefix(text, suffix_array[i - 1], suffix_array[i])
        
        return suffix_array, lcp
    
    def longest_common_prefix(text, i, j):
        """计算最长公共前缀"""
        n = len(text)
        length = 0
        
        while i + length < n and j + length < n and text[i + length] == text[j + length]:
            length += 1
        
        return length
    
    def search_suffix_array(text, pattern, suffix_array):
        """在后缀数组中搜索模式"""
        n = len(text)
        m = len(pattern)
        
        # 二分搜索
        left = 0
        right = n - 1
        
        while left <= right:
            mid = (left + right) // 2
            suffix_start = suffix_array[mid]
            
            # 比较模式和后缀
            comparison = 0
            for i in range(m):
                if suffix_start + i >= n:
                    comparison = -1
                    break
                if pattern[i] < text[suffix_start + i]:
                    comparison = -1
                    break
                elif pattern[i] > text[suffix_start + i]:
                    comparison = 1
                    break
            
            if comparison == 0:
                # 找到匹配
                return mid
            elif comparison < 0:
                right = mid - 1
            else:
                left = mid + 1
        
        return -1
    
    return build_suffix_array, build_suffix_array_with_lcp, longest_common_prefix, search_suffix_array
```

```python
def advanced_suffix_array():
    """高级后缀数组实现"""
    
    def build_suffix_array_dc3(text):
        """DC3算法构建后缀数组"""
        n = len(text)
        
        # 添加哨兵字符
        text += '$'
        n += 1
        
        # 构建三元组
        triples = []
        for i in range(n):
            if i % 3 == 0:
                triples.append((i, text[i], text[i + 1] if i + 1 < n else '$', text[i + 2] if i + 2 < n else '$'))
        
        # 排序三元组
        triples.sort(key=lambda x: (x[1], x[2], x[3]))
        
        # 构建后缀数组
        suffix_array = [triple[0] for triple in triples]
        return suffix_array
    
    def build_suffix_array_kasai(text, suffix_array):
        """Kasai算法构建LCP数组"""
        n = len(text)
        lcp = [0] * n
        
        # 构建逆后缀数组
        inv_suffix = [0] * n
        for i in range(n):
            inv_suffix[suffix_array[i]] = i
        
        # 计算LCP
        length = 0
        for i in range(n):
            if inv_suffix[i] == 0:
                length = 0
                continue
            
            j = suffix_array[inv_suffix[i] - 1]
            while i + length < n and j + length < n and text[i + length] == text[j + length]:
                length += 1
            
            lcp[inv_suffix[i]] = length
            if length > 0:
                length -= 1
        
        return lcp
    
    def build_suffix_array_manber(text):
        """Manber算法构建后缀数组"""
        n = len(text)
        
        # 初始排序
        suffix_array = list(range(n))
        suffix_array.sort(key=lambda x: text[x])
        
        # 多轮排序
        k = 1
        while k < n:
            # 按前k个字符排序
            suffix_array.sort(key=lambda x: (text[x], text[x + k] if x + k < n else '$'))
            k *= 2
        
        return suffix_array
    
    def build_suffix_array_skew(text):
        """Skew算法构建后缀数组"""
        n = len(text)
        
        # 添加哨兵字符
        text += '$'
        n += 1
        
        # 构建三元组
        triples = []
        for i in range(n):
            if i % 3 == 0:
                triples.append((i, text[i], text[i + 1] if i + 1 < n else '$', text[i + 2] if i + 2 < n else '$'))

        triples.sort(key=lambda x: (x[1], x[2], x[3]))
        
        # 构建后缀数组
        suffix_array = [triple[0] for triple in triples]
        return suffix_array
    
    return build_suffix_array_dc3, build_suffix_array_kasai, build_suffix_array_manber, build_suffix_array_skew
```

```python
def suffix_array_applications():
    """后缀数组应用"""
    
    def longest_common_substring(text1, text2):
        """最长公共子串"""
        # 连接两个字符串
        combined = text1 + '#' + text2
        n = len(combined)
        
        # 构建后缀数组
        suffix_array = build_suffix_array(combined)
        
        # 构建LCP数组
        lcp = [0] * n
        for i in range(1, n):
            lcp[i] = longest_common_prefix(combined, suffix_array[i - 1], suffix_array[i])
        
        # 找到最长公共子串
        max_length = 0
        max_start = 0
        
        for i in range(1, n):
            if lcp[i] > max_length:
                # 检查是否来自不同字符串
                pos1 = suffix_array[i - 1]
                pos2 = suffix_array[i]
                
                if (pos1 < len(text1) and pos2 >= len(text1) + 1) or (pos2 < len(text1) and pos1 >= len(text1) + 1):
                    max_length = lcp[i]
                    max_start = min(pos1, pos2)
        
        return combined[max_start:max_start + max_length]
    
    def longest_repeated_substring(text):
        """最长重复子串"""
        n = len(text)
        
        # 构建后缀数组
        suffix_array = build_suffix_array(text)
        
        # 构建LCP数组
        lcp = [0] * n
        for i in range(1, n):
            lcp[i] = longest_common_prefix(text, suffix_array[i - 1], suffix_array[i])
        
        # 找到最长重复子串
        max_length = 0
        max_start = 0
        
        for i in range(1, n):
            if lcp[i] > max_length:
                max_length = lcp[i]
                max_start = suffix_array[i]
        
        return text[max_start:max_start + max_length]
    
    def string_matching_suffix_array(text, pattern):
        """字符串匹配"""
        n = len(text)
        m = len(pattern)
        
        # 构建后缀数组
        suffix_array = build_suffix_array(text)
        
        # 二分搜索
        left = 0
        right = n - 1
        
        while left <= right:
            mid = (left + right) // 2
            suffix_start = suffix_array[mid]
            
            # 比较模式和后缀
            comparison = 0
            for i in range(m):
                if suffix_start + i >= n:
                    comparison = -1
                    break
                if pattern[i] < text[suffix_start + i]:
                    comparison = -1
                    break
                elif pattern[i] > text[suffix_start + i]:
                    comparison = 1
                    break
            
            if comparison == 0:
                # 找到匹配
                return mid
            elif comparison < 0:
                right = mid - 1
            else:
                left = mid + 1
        
        return -1
    
    def string_sorting_suffix_array(texts):
        """字符串排序"""
        # 为每个字符串构建后缀数组
        suffix_arrays = []
        for text in texts:
            suffix_array = build_suffix_array(text)
            suffix_arrays.append(suffix_array)
        
        # 按字典序排序
        sorted_texts = sorted(texts)
        return sorted_texts
    
    def text_compression_suffix_array(text):
        """文本压缩"""
        n = len(text)
        
        # 构建后缀数组
        suffix_array = build_suffix_array(text)
        
        # 构建LCP数组
        lcp = [0] * n
        for i in range(1, n):
            lcp[i] = longest_common_prefix(text, suffix_array[i - 1], suffix_array[i])
        
        # 找到重复子串
        repeated_substrings = []
        for i in range(1, n):
            if lcp[i] > 0:
                start = suffix_array[i]
                length = lcp[i]
                repeated_substrings.append((start, length))
        
        return repeated_substrings
    
    return longest_common_substring, longest_repeated_substring, string_matching_suffix_array, string_sorting_suffix_array, text_compression_suffix_array
```

```python
def text_analysis_applications():
    """文本分析应用"""
    
    def document_similarity(doc1, doc2):
        """文档相似度"""
        # 计算最长公共子串
        lcs = longest_common_substring(doc1, doc2)
        
        # 计算相似度
        similarity = len(lcs) / max(len(doc1), len(doc2))
        return similarity
    
    def plagiarism_detection(documents):
        """抄袭检测"""
        similarities = {}
        
        for i, doc1 in enumerate(documents):
            for j, doc2 in enumerate(documents):
                if i != j:
                    similarity = document_similarity(doc1, doc2)
                    similarities[(i, j)] = similarity
        
        return similarities
    
    def text_summarization(text, max_length=100):
        """文本摘要"""
        n = len(text)
        
        # 构建后缀数组
        suffix_array = build_suffix_array(text)
        
        # 构建LCP数组
        lcp = [0] * n
        for i in range(1, n):
            lcp[i] = longest_common_prefix(text, suffix_array[i - 1], suffix_array[i])
        
        # 找到重要子串
        important_substrings = []
        for i in range(1, n):
            if lcp[i] > 0:
                start = suffix_array[i]
                length = lcp[i]
                important_substrings.append((start, length))
        
        # 选择最重要的子串
        important_substrings.sort(key=lambda x: x[1], reverse=True)
        
        summary = ""
        for start, length in important_substrings:
            if len(summary) + length <= max_length:
                summary += text[start:start + length]
            else:
                break
        
        return summary
    
    def keyword_extraction(text, min_length=3):
        """关键词提取"""
        n = len(text)
        
        # 构建后缀数组
        suffix_array = build_suffix_array(text)
        
        # 构建LCP数组
        lcp = [0] * n
        for i in range(1, n):
            lcp[i] = longest_common_prefix(text, suffix_array[i - 1], suffix_array[i])
        
        # 找到重复子串
        keywords = []
        for i in range(1, n):
            if lcp[i] >= min_length:
                start = suffix_array[i]
                length = lcp[i]
                keyword = text[start:start + length]
                if keyword not in keywords:
                    keywords.append(keyword)
        
        return keywords
    
    return document_similarity, plagiarism_detection, text_summarization, keyword_extraction
```
