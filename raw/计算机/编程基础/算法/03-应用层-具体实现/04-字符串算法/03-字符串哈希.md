# 字符串哈希

## 🎯 核心概念

**字符串哈希**是将字符串映射为数值的技术，通过哈希函数将字符串转换为固定长度的数值，用于快速比较、查找和匹配。字符串哈希在字符串匹配、数据去重、密码学等领域有重要应用。

```python
def hash_function_types():
    """哈希函数类型"""
    
    def polynomial_hash(text, base=256, mod=1000000007):
        """多项式哈希函数"""
        hash_value = 0
        for char in text:
            hash_value = (hash_value * base + ord(char)) % mod
        return hash_value
    
    def djb2_hash(text):
        """DJB2哈希函数"""
        hash_value = 5381
        for char in text:
            hash_value = ((hash_value << 5) + hash_value) + ord(char)
        return hash_value
    
    def sdbm_hash(text):
        """SDBM哈希函数"""
        hash_value = 0
        for char in text:
            hash_value = ord(char) + (hash_value << 6) + (hash_value << 16) - hash_value
        return hash_value
    
    def fnv_hash(text):
        """FNV哈希函数"""
        hash_value = 2166136261
        for char in text:
            hash_value ^= ord(char)
            hash_value *= 16777619
        return hash_value
    
    def murmur_hash(text, seed=0):
        """Murmur哈希函数"""
        hash_value = seed
        for char in text:
            hash_value ^= ord(char)
            hash_value *= 0x85ebca6b
            hash_value ^= hash_value >> 13
            hash_value *= 0xc2b2ae35
            hash_value ^= hash_value >> 16
        return hash_value
    
    return polynomial_hash, djb2_hash, sdbm_hash, fnv_hash, murmur_hash
```

```python
def basic_string_hash():
    """基本字符串哈希实现"""
    
    def simple_hash(text):
        """简单哈希函数"""
        hash_value = 0
        for char in text:
            hash_value += ord(char)
        return hash_value
    
    def weighted_hash(text):
        """加权哈希函数"""
        hash_value = 0
        weight = 1
        for char in text:
            hash_value += ord(char) * weight
            weight *= 31
        return hash_value
    
    def modular_hash(text, mod=1000000007):
        """模运算哈希函数"""
        hash_value = 0
        base = 256
        for char in text:
            hash_value = (hash_value * base + ord(char)) % mod
        return hash_value
    
    def rolling_hash(text, window_size):
        """滚动哈希函数"""
        n = len(text)
        if n < window_size:
            return []
        
        base = 256
        mod = 1000000007
        base_power = pow(base, window_size - 1, mod)
        
        # 计算第一个窗口的哈希值
        window_hash = 0
        for i in range(window_size):
            window_hash = (window_hash * base + ord(text[i])) % mod
        
        hashes = [window_hash]
        
        # 滚动计算
        for i in range(window_size, n):
            window_hash = (window_hash - ord(text[i - window_size]) * base_power) % mod
            window_hash = (window_hash * base + ord(text[i])) % mod
            hashes.append(window_hash)
        
        return hashes
    
    return simple_hash, weighted_hash, modular_hash, rolling_hash
```

```python
def advanced_string_hash():
    """高级字符串哈希实现"""
    
    def double_hash(text, base1=256, mod1=1000000007, base2=257, mod2=1000000009):
        """双重哈希函数"""
        hash1 = 0
        hash2 = 0
        
        for char in text:
            hash1 = (hash1 * base1 + ord(char)) % mod1
            hash2 = (hash2 * base2 + ord(char)) % mod2
        
        return (hash1, hash2)
    
    def triple_hash(text, base1=256, mod1=1000000007, base2=257, mod2=1000000009, base3=258, mod3=1000000011):
        """三重哈希函数"""
        hash1 = 0
        hash2 = 0
        hash3 = 0
        
        for char in text:
            hash1 = (hash1 * base1 + ord(char)) % mod1
            hash2 = (hash2 * base2 + ord(char)) % mod2
            hash3 = (hash3 * base3 + ord(char)) % mod3
        
        return (hash1, hash2, hash3)
    
    def universal_hash(text, a, b, p, m):
        """通用哈希函数"""
        hash_value = 0
        for char in text:
            hash_value = (a * hash_value + ord(char) + b) % p
        return hash_value % m
    
    def cuckoo_hash(text, hash1_func, hash2_func):
        """布谷鸟哈希函数"""
        hash1 = hash1_func(text)
        hash2 = hash2_func(text)
        return (hash1, hash2)
    
    def consistent_hash(text, ring_size):
        """一致性哈希函数"""
        hash_value = 0
        for char in text:
            hash_value = (hash_value * 31 + ord(char)) % ring_size
        return hash_value
    
    return double_hash, triple_hash, universal_hash, cuckoo_hash, consistent_hash
```

### 3. 字符串哈希应用
```python
def string_hash_applications():
    """字符串哈希应用"""
    
    def string_matching_hash(text, pattern):
        """字符串匹配哈希"""
        n, m = len(text), len(pattern)
        if m == 0:
            return [i for i in range(n + 1)]
        if n < m:
            return []
        
        base = 256
        mod = 1000000007
        base_power = pow(base, m - 1, mod)
        
        # 计算模式串哈希值
        pattern_hash = 0
        for char in pattern:
            pattern_hash = (pattern_hash * base + ord(char)) % mod
        
        # 计算第一个文本子串哈希值
        text_hash = 0
        for i in range(m):
            text_hash = (text_hash * base + ord(text[i])) % mod
        
        matches = []
        
        # 检查第一个子串
        if text_hash == pattern_hash and text[:m] == pattern:
            matches.append(0)
        
        # 滚动匹配
        for i in range(m, n):
            text_hash = (text_hash - ord(text[i - m]) * base_power) % mod
            text_hash = (text_hash * base + ord(text[i])) % mod
            
            if text_hash == pattern_hash and text[i - m + 1:i + 1] == pattern:
                matches.append(i - m + 1)
        
        return matches
    
    def string_deduplication(strings):
        """字符串去重"""
        seen = set()
        unique_strings = []
        
        for string in strings:
            hash_value = polynomial_hash(string)
            if hash_value not in seen:
                seen.add(hash_value)
                unique_strings.append(string)
        
        return unique_strings
    
    def string_grouping(strings):
        """字符串分组"""
        groups = {}
        
        for string in strings:
            hash_value = polynomial_hash(string)
            if hash_value not in groups:
                groups[hash_value] = []
            groups[hash_value].append(string)
        
        return groups
    
    def string_similarity(text1, text2):
        """字符串相似度"""
        hash1 = polynomial_hash(text1)
        hash2 = polynomial_hash(text2)
        
        # 简单的相似度计算
        if hash1 == hash2:
            return 1.0
        
        # 计算编辑距离
        def edit_distance(s1, s2):
            m, n = len(s1), len(s2)
            dp = [[0] * (n + 1) for _ in range(m + 1)]
            
            for i in range(m + 1):
                dp[i][0] = i
            for j in range(n + 1):
                dp[0][j] = j
            
            for i in range(1, m + 1):
                for j in range(1, n + 1):
                    if s1[i - 1] == s2[j - 1]:
                        dp[i][j] = dp[i - 1][j - 1]
                    else:
                        dp[i][j] = 1 + min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1])
            
            return dp[m][n]
        
        distance = edit_distance(text1, text2)
        max_len = max(len(text1), len(text2))
        similarity = 1.0 - distance / max_len
        
        return similarity
    
    def string_fingerprinting(text, k):
        """字符串指纹"""
        n = len(text)
        if n < k:
            return []
        
        fingerprints = []
        base = 256
        mod = 1000000007
        base_power = pow(base, k - 1, mod)
        
        # 计算第一个k-gram的哈希值
        fingerprint = 0
        for i in range(k):
            fingerprint = (fingerprint * base + ord(text[i])) % mod
        
        fingerprints.append(fingerprint)
        
        # 滚动计算
        for i in range(k, n):
            fingerprint = (fingerprint - ord(text[i - k]) * base_power) % mod
            fingerprint = (fingerprint * base + ord(text[i])) % mod
            fingerprints.append(fingerprint)
        
        return fingerprints
    
    return string_matching_hash, string_deduplication, string_grouping, string_similarity, string_fingerprinting
```

```python
def string_matching_applications():
    """字符串匹配应用"""
    
    def rabin_karp_matching(text, pattern):
        """Rabin-Karp字符串匹配"""
        return string_matching_hash(text, pattern)
    
    def multiple_pattern_matching(text, patterns):
        """多模式串匹配"""
        results = {}
        
        for pattern in patterns:
            matches = string_matching_hash(text, pattern)
            results[pattern] = matches
        
        return results
    
    def approximate_matching(text, pattern, threshold=0.8):
        """近似匹配"""
        n, m = len(text), len(pattern)
        if m == 0:
            return [i for i in range(n + 1)]
        if n < m:
            return []
        
        matches = []
        
        for i in range(n - m + 1):
            substring = text[i:i + m]
            similarity = string_similarity(substring, pattern)
            
            if similarity >= threshold:
                matches.append(i)
        
        return matches
    
    def wildcard_matching(text, pattern):
        """通配符匹配"""
        n, m = len(text), len(pattern)
        if m == 0:
            return [i for i in range(n + 1)]
        if n < m:
            return []
        
        matches = []
        
        for i in range(n - m + 1):
            substring = text[i:i + m]
            match = True
            
            for j in range(m):
                if pattern[j] != '?' and pattern[j] != substring[j]:
                    match = False
                    break
            
            if match:
                matches.append(i)
        
        return matches
    
    return rabin_karp_matching, multiple_pattern_matching, approximate_matching, wildcard_matching
```

```python
def cryptographic_applications():
    """密码学应用"""
    
    def password_hashing(password, salt=""):
        """密码哈希"""
        # 使用SHA-256进行密码哈希
        import hashlib
        
        password_salt = password + salt
        hash_value = hashlib.sha256(password_salt.encode()).hexdigest()
        return hash_value
    
    def message_authentication(text, key):
        """消息认证"""
        # 使用HMAC进行消息认证
        import hmac
        import hashlib
        
        mac = hmac.new(key.encode(), text.encode(), hashlib.sha256)
        return mac.hexdigest()
    
    def digital_signature(text, private_key):
        """数字签名"""
        # 使用RSA进行数字签名
        import hashlib
        import rsa
        
        hash_value = hashlib.sha256(text.encode()).digest()
        signature = rsa.sign(hash_value, private_key, 'SHA-256')
        return signature
    
    def checksum_calculation(data):
        """校验和计算"""
        # 使用CRC32计算校验和
        import zlib
        
        checksum = zlib.crc32(data.encode())
        return checksum
    
    def integrity_verification(data, expected_hash):
        """完整性验证"""
        # 验证数据完整性
        actual_hash = polynomial_hash(data)
        return actual_hash == expected_hash
    
    return password_hashing, message_authentication, digital_signature, checksum_calculation, integrity_verification
```
