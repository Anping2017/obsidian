# 文本数据特征工程

## 概述

文本数据特征工程是将原始文本数据转换为机器学习模型可以理解和处理的数值特征的过程。这是NLP项目成功的关键步骤，直接影响模型的性能和效果。

## 核心概念

### 1. 特征工程定义
- **特征提取**：从原始文本中提取有意义的特征
- **特征选择**：选择对任务最有用的特征
- **特征变换**：将特征转换为更适合模型的形式

### 2. 特征类型
- **统计特征**：词频、文档长度、句子数量等
- **语义特征**：词向量、主题分布、情感倾向等
- **结构特征**：语法结构、句法树、依存关系等

## 特征提取方法

### 1. 词频特征
```python
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
import numpy as np

# 词频特征
def extract_count_features(texts, max_features=1000):
    """
    提取词频特征
    """
    vectorizer = CountVectorizer(
        max_features=max_features,
        stop_words='english',
        ngram_range=(1, 2)  # 包含1-gram和2-gram
    )
    
    features = vectorizer.fit_transform(texts)
    feature_names = vectorizer.get_feature_names_out()
    
    return features, feature_names

# TF-IDF特征
def extract_tfidf_features(texts, max_features=1000):
    """
    提取TF-IDF特征
    """
    vectorizer = TfidfVectorizer(
        max_features=max_features,
        stop_words='english',
        ngram_range=(1, 2),
        sublinear_tf=True  # 使用对数缩放
    )
    
    features = vectorizer.fit_transform(texts)
    feature_names = vectorizer.get_feature_names_out()
    
    return features, feature_names
```

### 2. 语义特征
```python
import gensim
from gensim.models import Word2Vec, Doc2Vec
import numpy as np

# Word2Vec特征
def extract_word2vec_features(texts, vector_size=100, window=5, min_count=2):
    """
    提取Word2Vec特征
    """
    # 预处理文本
    processed_texts = [text.lower().split() for text in texts]
    
    # 训练Word2Vec模型
    model = Word2Vec(
        processed_texts,
        vector_size=vector_size,
        window=window,
        min_count=min_count,
        workers=4
    )
    
    # 提取文档向量（平均词向量）
    doc_vectors = []
    for text in processed_texts:
        vectors = []
        for word in text:
            if word in model.wv:
                vectors.append(model.wv[word])
        
        if vectors:
            doc_vector = np.mean(vectors, axis=0)
        else:
            doc_vector = np.zeros(vector_size)
        
        doc_vectors.append(doc_vector)
    
    return np.array(doc_vectors), model

# Doc2Vec特征
def extract_doc2vec_features(texts, vector_size=100, window=5, min_count=2):
    """
    提取Doc2Vec特征
    """
    # 预处理文本
    processed_texts = []
    for i, text in enumerate(texts):
        processed_texts.append(gensim.models.doc2vec.TaggedDocument(
            text.lower().split(), [i]
        ))
    
    # 训练Doc2Vec模型
    model = Doc2Vec(
        processed_texts,
        vector_size=vector_size,
        window=window,
        min_count=min_count,
        workers=4
    )
    
    # 提取文档向量
    doc_vectors = []
    for i in range(len(texts)):
        doc_vector = model.dv[i]
        doc_vectors.append(doc_vector)
    
    return np.array(doc_vectors), model
```

### 3. 统计特征
```python
import re
from collections import Counter
import numpy as np

def extract_statistical_features(texts):
    """
    提取统计特征
    """
    features = []
    
    for text in texts:
        # 基本统计特征
        char_count = len(text)
        word_count = len(text.split())
        sentence_count = len(re.split(r'[.!?]+', text))
        
        # 词汇丰富度
        unique_words = len(set(text.lower().split()))
        lexical_diversity = unique_words / word_count if word_count > 0 else 0
        
        # 平均词长
        avg_word_length = np.mean([len(word) for word in text.split()]) if word_count > 0 else 0
        
        # 标点符号比例
        punctuation_count = len(re.findall(r'[^\w\s]', text))
        punctuation_ratio = punctuation_count / char_count if char_count > 0 else 0
        
        # 数字比例
        digit_count = len(re.findall(r'\d', text))
        digit_ratio = digit_count / char_count if char_count > 0 else 0
        
        # 大写字母比例
        upper_count = len(re.findall(r'[A-Z]', text))
        upper_ratio = upper_count / char_count if char_count > 0 else 0
        
        features.append([
            char_count, word_count, sentence_count,
            lexical_diversity, avg_word_length,
            punctuation_ratio, digit_ratio, upper_ratio
        ])
    
    return np.array(features)
```

## 特征选择方法

### 1. 基于统计的特征选择
```python
from sklearn.feature_selection import SelectKBest, chi2, mutual_info_classif
from sklearn.feature_selection import VarianceThreshold

def select_features_statistical(X, y, method='chi2', k=1000):
    """
    基于统计方法的特征选择
    """
    if method == 'chi2':
        selector = SelectKBest(chi2, k=k)
    elif method == 'mutual_info':
        selector = SelectKBest(mutual_info_classif, k=k)
    else:
        raise ValueError("Method must be 'chi2' or 'mutual_info'")
    
    X_selected = selector.fit_transform(X, y)
    selected_features = selector.get_support(indices=True)
    
    return X_selected, selected_features

# 方差阈值特征选择
def select_features_variance(X, threshold=0.01):
    """
    基于方差的特征选择
    """
    selector = VarianceThreshold(threshold=threshold)
    X_selected = selector.fit_transform(X)
    selected_features = selector.get_support(indices=True)
    
    return X_selected, selected_features
```

### 2. 基于模型的特征选择
```python
from sklearn.feature_selection import SelectFromModel
from sklearn.ensemble import RandomForestClassifier
from sklearn.linear_model import LassoCV

def select_features_model(X, y, method='random_forest', threshold='median'):
    """
    基于模型的特征选择
    """
    if method == 'random_forest':
        model = RandomForestClassifier(n_estimators=100, random_state=42)
    elif method == 'lasso':
        model = LassoCV(cv=5, random_state=42)
    else:
        raise ValueError("Method must be 'random_forest' or 'lasso'")
    
    selector = SelectFromModel(model, threshold=threshold)
    X_selected = selector.fit_transform(X, y)
    selected_features = selector.get_support(indices=True)
    
    return X_selected, selected_features
```

## 特征变换方法

### 1. 标准化和归一化
```python
from sklearn.preprocessing import StandardScaler, MinMaxScaler, RobustScaler

def transform_features(X, method='standard'):
    """
    特征变换
    """
    if method == 'standard':
        scaler = StandardScaler()
    elif method == 'minmax':
        scaler = MinMaxScaler()
    elif method == 'robust':
        scaler = RobustScaler()
    else:
        raise ValueError("Method must be 'standard', 'minmax', or 'robust'")
    
    X_transformed = scaler.fit_transform(X)
    
    return X_transformed, scaler
```

### 2. 降维技术
```python
from sklearn.decomposition import PCA, TruncatedSVD
from sklearn.manifold import TSNE

def reduce_dimensions(X, method='pca', n_components=50):
    """
    降维技术
    """
    if method == 'pca':
        reducer = PCA(n_components=n_components)
    elif method == 'svd':
        reducer = TruncatedSVD(n_components=n_components)
    elif method == 'tsne':
        reducer = TSNE(n_components=n_components, random_state=42)
    else:
        raise ValueError("Method must be 'pca', 'svd', or 'tsne'")
    
    X_reduced = reducer.fit_transform(X)
    
    return X_reduced, reducer
```

## 高级特征工程

### 1. 主题建模特征
```python
from sklearn.decomposition import LatentDirichletAllocation
import numpy as np

def extract_topic_features(texts, n_topics=10, max_features=1000):
    """
    提取主题建模特征
    """
    from sklearn.feature_extraction.text import CountVectorizer
    
    # 使用CountVectorizer提取词频
    vectorizer = CountVectorizer(max_features=max_features, stop_words='english')
    X = vectorizer.fit_transform(texts)
    
    # LDA主题建模
    lda = LatentDirichletAllocation(
        n_components=n_topics,
        random_state=42,
        max_iter=100
    )
    
    topic_features = lda.fit_transform(X)
    
    # 获取主题词
    feature_names = vectorizer.get_feature_names_out()
    topic_words = []
    for topic_idx, topic in enumerate(lda.components_):
        top_words_idx = topic.argsort()[-10:][::-1]
        top_words = [feature_names[i] for i in top_words_idx]
        topic_words.append(top_words)
    
    return topic_features, topic_words
```

### 2. 情感特征
```python
from textblob import TextBlob
import numpy as np

def extract_sentiment_features(texts):
    """
    提取情感特征
    """
    features = []
    
    for text in texts:
        blob = TextBlob(text)
        
        # 情感极性
        polarity = blob.sentiment.polarity
        
        # 情感主观性
        subjectivity = blob.sentiment.subjectivity
        
        # 情感强度
        sentiment_intensity = abs(polarity)
        
        features.append([polarity, subjectivity, sentiment_intensity])
    
    return np.array(features)
```

## 特征工程最佳实践

### 1. 特征工程流程
```python
class TextFeatureEngineer:
    def __init__(self):
        self.vectorizers = {}
        self.scalers = {}
        self.selectors = {}
    
    def fit_transform(self, texts, y=None):
        """
        完整的特征工程流程
        """
        features = []
        
        # 1. 提取多种特征
        # 词频特征
        count_features, _ = extract_count_features(texts)
        features.append(count_features)
        
        # TF-IDF特征
        tfidf_features, _ = extract_tfidf_features(texts)
        features.append(tfidf_features)
        
        # 统计特征
        stat_features = extract_statistical_features(texts)
        features.append(stat_features)
        
        # 情感特征
        sentiment_features = extract_sentiment_features(texts)
        features.append(sentiment_features)
        
        # 2. 合并特征
        X = np.hstack(features)
        
        # 3. 特征选择
        if y is not None:
            X_selected, _ = select_features_statistical(X, y, method='chi2', k=1000)
            X = X_selected
        
        # 4. 特征变换
        X_transformed, scaler = transform_features(X, method='standard')
        
        return X_transformed, scaler
    
    def transform(self, texts, scaler):
        """
        对新数据进行特征工程
        """
        features = []
        
        # 提取特征（与训练时相同）
        count_features, _ = extract_count_features(texts)
        tfidf_features, _ = extract_tfidf_features(texts)
        stat_features = extract_statistical_features(texts)
        sentiment_features = extract_sentiment_features(texts)
        
        features.append(count_features)
        features.append(tfidf_features)
        features.append(stat_features)
        features.append(sentiment_features)
        
        X = np.hstack(features)
        X_transformed = scaler.transform(X)
        
        return X_transformed
```

### 2. 特征工程评估
```python
from sklearn.model_selection import cross_val_score
from sklearn.ensemble import RandomForestClassifier

def evaluate_features(X, y, cv=5):
    """
    评估特征工程效果
    """
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    scores = cross_val_score(model, X, y, cv=cv, scoring='accuracy')
    
    return {
        'mean_score': scores.mean(),
        'std_score': scores.std(),
        'scores': scores
    }
```

### 2. 情感分析特征工程
```python
def sentiment_analysis_features(texts):
    """
    情感分析专用特征工程
    """
    features = []
    
    for text in texts:
        # 基础特征
        char_count = len(text)
        word_count = len(text.split())
        
        # 情感词典特征
        positive_words = ['good', 'great', 'excellent', 'amazing', 'wonderful']
        negative_words = ['bad', 'terrible', 'awful', 'horrible', 'disgusting']
        
        positive_count = sum(1 for word in text.lower().split() if word in positive_words)
        negative_count = sum(1 for word in text.lower().split() if word in negative_words)
        
        # 情感比例
        positive_ratio = positive_count / word_count if word_count > 0 else 0
        negative_ratio = negative_count / word_count if word_count > 0 else 0
        
        # 情感强度
        sentiment_intensity = abs(positive_ratio - negative_ratio)
        
        features.append([
            char_count, word_count,
            positive_count, negative_count,
            positive_ratio, negative_ratio,
            sentiment_intensity
        ])
    
    return np.array(features)
```

## 总结

文本数据特征工程是NLP项目的核心环节，需要：

1. **多维度特征提取**：结合统计、语义、结构等多种特征
2. **智能特征选择**：选择对任务最有用的特征
3. **适当特征变换**：将特征转换为模型友好的形式
4. **持续优化**：根据模型性能调整特征工程策略

通过系统化的特征工程，可以显著提升NLP模型的性能和效果。
