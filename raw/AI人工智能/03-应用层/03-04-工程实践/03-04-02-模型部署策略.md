# 模型部署策略

## 概述

模型部署是将训练好的机器学习模型投入生产环境的关键步骤。它涉及模型打包、环境配置、服务部署、监控等多个环节。本章将介绍模型部署的核心策略、技术方案和最佳实践。

## 核心概念

### 1. 模型部署定义
- **目标**：将训练好的模型投入生产环境，提供稳定的预测服务
- **关键要素**：模型版本管理、环境一致性、服务可用性、性能监控
- **部署模式**：批处理、实时服务、边缘部署、云端部署
- **挑战**：模型版本控制、环境差异、性能优化、监控告警

### 2. 部署架构模式
- **单体部署**：单一服务实例
- **微服务部署**：独立服务组件
- **容器化部署**：Docker容器
- **无服务器部署**：Serverless函数

## 模型打包与版本管理

### 1. 模型打包

#### 模型序列化
```python
import pickle
import joblib
import json
import os
from datetime import datetime
import hashlib

class ModelPackager:
    def __init__(self):
        self.model_metadata = {}
        self.model_artifacts = {}
        
    def package_model(self, model, model_name, model_version, metadata=None):
        """
        打包模型
        """
        # 创建模型包目录
        package_dir = f"models/{model_name}_{model_version}"
        os.makedirs(package_dir, exist_ok=True)
        
        # 序列化模型
        model_path = os.path.join(package_dir, "model.pkl")
        joblib.dump(model, model_path)
        
        # 创建模型元数据
        model_metadata = {
            'name': model_name,
            'version': model_version,
            'created_at': datetime.now().isoformat(),
            'model_type': type(model).__name__,
            'model_size': os.path.getsize(model_path),
            'checksum': self._calculate_checksum(model_path),
            'metadata': metadata or {}
        }
        
        # 保存元数据
        metadata_path = os.path.join(package_dir, "metadata.json")
        with open(metadata_path, 'w') as f:
            json.dump(model_metadata, f, indent=2)
        
        # 创建模型配置文件
        config_path = os.path.join(package_dir, "config.json")
        self._create_model_config(config_path, model_metadata)
        
        # 创建依赖文件
        requirements_path = os.path.join(package_dir, "requirements.txt")
        self._create_requirements_file(requirements_path)
        
        print(f"模型 {model_name} v{model_version} 打包完成")
        return package_dir
    
    def _calculate_checksum(self, file_path):
        """
        计算文件校验和
        """
        with open(file_path, 'rb') as f:
            return hashlib.md5(f.read()).hexdigest()
    
    def _create_model_config(self, config_path, metadata):
        """
        创建模型配置文件
        """
        config = {
            'model': {
                'name': metadata['name'],
                'version': metadata['version'],
                'type': metadata['model_type']
            },
            'deployment': {
                'framework': 'scikit-learn',
                'python_version': '3.8',
                'dependencies': [
                    'scikit-learn',
                    'pandas',
                    'numpy'
                ]
            },
            'inference': {
                'input_schema': {
                    'type': 'array',
                    'items': {
                        'type': 'number'
                    }
                },
                'output_schema': {
                    'type': 'array',
                    'items': {
                        'type': 'number'
                    }
                }
            }
        }
        
        with open(config_path, 'w') as f:
            json.dump(config, f, indent=2)
    
    def _create_requirements_file(self, requirements_path):
        """
        创建依赖文件
        """
        requirements = [
            'scikit-learn>=1.0.0',
            'pandas>=1.3.0',
            'numpy>=1.21.0',
            'flask>=2.0.0',
            'gunicorn>=20.0.0'
        ]
        
        with open(requirements_path, 'w') as f:
            f.write('\n'.join(requirements))
    
    def load_model(self, package_dir):
        """
        加载模型包
        """
        # 加载元数据
        metadata_path = os.path.join(package_dir, "metadata.json")
        with open(metadata_path, 'r') as f:
            metadata = json.load(f)
        
        # 验证模型文件
        model_path = os.path.join(package_dir, "model.pkl")
        if not os.path.exists(model_path):
            raise FileNotFoundError(f"模型文件不存在: {model_path}")
        
        # 验证校验和
        current_checksum = self._calculate_checksum(model_path)
        if current_checksum != metadata['checksum']:
            raise ValueError("模型文件校验和不匹配")
        
        # 加载模型
        model = joblib.load(model_path)
        
        return model, metadata
    
    def validate_model_package(self, package_dir):
        """
        验证模型包
        """
        validation_results = {
            'valid': True,
            'errors': [],
            'warnings': []
        }
        
        # 检查必需文件
        required_files = ['model.pkl', 'metadata.json', 'config.json', 'requirements.txt']
        for file in required_files:
            file_path = os.path.join(package_dir, file)
            if not os.path.exists(file_path):
                validation_results['errors'].append(f"缺少必需文件: {file}")
                validation_results['valid'] = False
        
        # 验证元数据
        if validation_results['valid']:
            try:
                metadata_path = os.path.join(package_dir, "metadata.json")
                with open(metadata_path, 'r') as f:
                    metadata = json.load(f)
                
                # 检查元数据字段
                required_fields = ['name', 'version', 'created_at', 'model_type']
                for field in required_fields:
                    if field not in metadata:
                        validation_results['errors'].append(f"元数据缺少字段: {field}")
                        validation_results['valid'] = False
                
            except Exception as e:
                validation_results['errors'].append(f"元数据解析失败: {e}")
                validation_results['valid'] = False
        
        return validation_results

class ModelVersionManager:
    def __init__(self):
        self.model_registry = {}
        self.version_history = {}
        
    def register_model(self, model_name, model_version, package_dir, metadata):
        """
        注册模型版本
        """
        if model_name not in self.model_registry:
            self.model_registry[model_name] = {}
            self.version_history[model_name] = []
        
        self.model_registry[model_name][model_version] = {
            'package_dir': package_dir,
            'metadata': metadata,
            'registered_at': datetime.now().isoformat(),
            'status': 'registered'
        }
        
        self.version_history[model_name].append({
            'version': model_version,
            'registered_at': datetime.now().isoformat(),
            'status': 'registered'
        })
        
        print(f"模型 {model_name} v{model_version} 已注册")
    
    def get_model_version(self, model_name, version=None):
        """
        获取模型版本
        """
        if model_name not in self.model_registry:
            raise ValueError(f"模型 {model_name} 未注册")
        
        if version is None:
            # 获取最新版本
            versions = list(self.model_registry[model_name].keys())
            if not versions:
                raise ValueError(f"模型 {model_name} 没有版本")
            version = max(versions)
        
        if version not in self.model_registry[model_name]:
            raise ValueError(f"模型 {model_name} 版本 {version} 不存在")
        
        return self.model_registry[model_name][version]
    
    def list_model_versions(self, model_name):
        """
        列出模型版本
        """
        if model_name not in self.model_registry:
            return []
        
        versions = []
        for version, info in self.model_registry[model_name].items():
            versions.append({
                'version': version,
                'status': info['status'],
                'registered_at': info['registered_at'],
                'metadata': info['metadata']
            })
        
        return sorted(versions, key=lambda x: x['version'], reverse=True)
    
    def promote_model_version(self, model_name, version, target_stage):
        """
        提升模型版本阶段
        """
        if model_name not in self.model_registry:
            raise ValueError(f"模型 {model_name} 未注册")
        
        if version not in self.model_registry[model_name]:
            raise ValueError(f"模型 {model_name} 版本 {version} 不存在")
        
        self.model_registry[model_name][version]['stage'] = target_stage
        self.model_registry[model_name][version]['promoted_at'] = datetime.now().isoformat()
        
        print(f"模型 {model_name} v{version} 已提升到 {target_stage} 阶段")
    
    def deprecate_model_version(self, model_name, version):
        """
        废弃模型版本
        """
        if model_name not in self.model_registry:
            raise ValueError(f"模型 {model_name} 未注册")
        
        if version not in self.model_registry[model_name]:
            raise ValueError(f"模型 {model_name} 版本 {version} 不存在")
        
        self.model_registry[model_name][version]['status'] = 'deprecated'
        self.model_registry[model_name][version]['deprecated_at'] = datetime.now().isoformat()
        
        print(f"模型 {model_name} v{version} 已废弃")
```

### 2. 模型服务化

#### 模型服务封装
```python
from flask import Flask, request, jsonify
import numpy as np
import pandas as pd
from functools import wraps
import logging

class ModelService:
    def __init__(self, model, metadata):
        self.model = model
        self.metadata = metadata
        self.app = Flask(__name__)
        self.setup_routes()
        self.setup_logging()
        
    def setup_routes(self):
        """
        设置路由
        """
        @self.app.route('/health', methods=['GET'])
        def health_check():
            return jsonify({
                'status': 'healthy',
                'model_name': self.metadata['name'],
                'model_version': self.metadata['version'],
                'timestamp': datetime.now().isoformat()
            })
        
        @self.app.route('/predict', methods=['POST'])
        def predict():
            try:
                # 获取输入数据
                input_data = request.get_json()
                
                # 验证输入
                if not input_data:
                    return jsonify({'error': '缺少输入数据'}), 400
                
                # 预处理输入
                processed_input = self._preprocess_input(input_data)
                
                # 执行预测
                prediction = self._predict(processed_input)
                
                # 后处理输出
                output = self._postprocess_output(prediction)
                
                return jsonify({
                    'prediction': output,
                    'model_name': self.metadata['name'],
                    'model_version': self.metadata['version'],
                    'timestamp': datetime.now().isoformat()
                })
                
            except Exception as e:
                logging.error(f"预测失败: {e}")
                return jsonify({'error': str(e)}), 500
        
        @self.app.route('/model_info', methods=['GET'])
        def model_info():
            return jsonify({
                'name': self.metadata['name'],
                'version': self.metadata['version'],
                'model_type': self.metadata['model_type'],
                'created_at': self.metadata['created_at'],
                'input_schema': self._get_input_schema(),
                'output_schema': self._get_output_schema()
            })
        
        @self.app.route('/batch_predict', methods=['POST'])
        def batch_predict():
            try:
                # 获取批量输入数据
                input_data = request.get_json()
                
                if not input_data or 'data' not in input_data:
                    return jsonify({'error': '缺少批量输入数据'}), 400
                
                # 处理批量数据
                predictions = []
                for item in input_data['data']:
                    try:
                        processed_input = self._preprocess_input(item)
                        prediction = self._predict(processed_input)
                        output = self._postprocess_output(prediction)
                        predictions.append(output)
                    except Exception as e:
                        predictions.append({'error': str(e)})
                
                return jsonify({
                    'predictions': predictions,
                    'model_name': self.metadata['name'],
                    'model_version': self.metadata['version'],
                    'timestamp': datetime.now().isoformat()
                })
                
            except Exception as e:
                logging.error(f"批量预测失败: {e}")
                return jsonify({'error': str(e)}), 500
    
    def setup_logging(self):
        """
        设置日志
        """
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        self.logger = logging.getLogger(__name__)
    
    def _preprocess_input(self, input_data):
        """
        预处理输入数据
        """
        # 转换为DataFrame
        if isinstance(input_data, dict):
            df = pd.DataFrame([input_data])
        elif isinstance(input_data, list):
            df = pd.DataFrame(input_data)
        else:
            df = input_data
        
        # 数据类型转换
        for column in df.columns:
            if df[column].dtype == 'object':
                try:
                    df[column] = pd.to_numeric(df[column])
                except:
                    pass
        
        return df
    
    def _predict(self, input_data):
        """
        执行预测
        """
        try:
            prediction = self.model.predict(input_data)
            return prediction
        except Exception as e:
            self.logger.error(f"模型预测失败: {e}")
            raise
    
    def _postprocess_output(self, prediction):
        """
        后处理输出
        """
        # 转换为可序列化格式
        if isinstance(prediction, np.ndarray):
            return prediction.tolist()
        elif isinstance(prediction, pd.Series):
            return prediction.tolist()
        else:
            return prediction
    
    def _get_input_schema(self):
        """
        获取输入模式
        """
        return {
            'type': 'object',
            'properties': {
                'feature1': {'type': 'number'},
                'feature2': {'type': 'number'},
                'feature3': {'type': 'number'}
            },
            'required': ['feature1', 'feature2', 'feature3']
        }
    
    def _get_output_schema(self):
        """
        获取输出模式
        """
        return {
            'type': 'array',
            'items': {'type': 'number'}
        }
    
    def run(self, host='0.0.0.0', port=5000, debug=False):
        """
        运行服务
        """
        self.logger.info(f"启动模型服务: {self.metadata['name']} v{self.metadata['version']}")
        self.app.run(host=host, port=port, debug=debug)

class ModelServiceManager:
    def __init__(self):
        self.services = {}
        self.service_configs = {}
        
    def deploy_model_service(self, model_name, model_version, config):
        """
        部署模型服务
        """
        try:
            # 加载模型
            model_registry = ModelVersionManager()
            model_info = model_registry.get_model_version(model_name, model_version)
            
            model_packager = ModelPackager()
            model, metadata = model_packager.load_model(model_info['package_dir'])
            
            # 创建服务
            service = ModelService(model, metadata)
            
            # 配置服务
            service_config = {
                'model_name': model_name,
                'model_version': model_version,
                'host': config.get('host', '0.0.0.0'),
                'port': config.get('port', 5000),
                'workers': config.get('workers', 1),
                'timeout': config.get('timeout', 30)
            }
            
            self.service_configs[f"{model_name}_{model_version}"] = service_config
            
            # 启动服务
            service.run(
                host=service_config['host'],
                port=service_config['port'],
                debug=config.get('debug', False)
            )
            
            self.services[f"{model_name}_{model_version}"] = service
            
            print(f"模型服务 {model_name} v{model_version} 部署成功")
            
        except Exception as e:
            print(f"模型服务部署失败: {e}")
            raise
    
    def stop_model_service(self, model_name, model_version):
        """
        停止模型服务
        """
        service_key = f"{model_name}_{model_version}"
        
        if service_key in self.services:
            # 停止服务
            service = self.services[service_key]
            # 这里需要实现停止逻辑
            
            del self.services[service_key]
            del self.service_configs[service_key]
            
            print(f"模型服务 {model_name} v{model_version} 已停止")
        else:
            print(f"模型服务 {model_name} v{model_version} 未运行")
    
    def list_services(self):
        """
        列出所有服务
        """
        services = []
        for service_key, service in self.services.items():
            services.append({
                'service_key': service_key,
                'config': self.service_configs[service_key],
                'status': 'running'
            })
        
        return services
```

#### Docker构建脚本
```python
import docker
import os
import json
from datetime import datetime

class DockerModelDeployer:
    def __init__(self):
        self.client = docker.from_env()
        self.images = {}
        
    def build_model_image(self, model_name, model_version, package_dir):
        """
        构建模型镜像
        """
        try:
            # 创建Dockerfile
            dockerfile_path = self._create_dockerfile(package_dir)
            
            # 构建镜像
            image_name = f"{model_name}:{model_version}"
            image_tag = f"{model_name}:latest"
            
            image, build_log = self.client.images.build(
                path=package_dir,
                tag=image_name,
                dockerfile=dockerfile_path,
                rm=True
            )
            
            # 添加标签
            image.tag(image_tag)
            
            self.images[image_name] = {
                'image_id': image.id,
                'created_at': datetime.now().isoformat(),
                'size': image.attrs['Size']
            }
            
            print(f"模型镜像 {image_name} 构建成功")
            return image_name
            
        except Exception as e:
            print(f"模型镜像构建失败: {e}")
            raise
    
    def _create_dockerfile(self, package_dir):
        """
        创建Dockerfile
        """
        dockerfile_content = """
FROM python:3.8-slim

WORKDIR /app

RUN apt-get update && apt-get install -y \\
    gcc \\
    g++ \\
    curl \\
    && rm -rf /var/lib/apt/lists/*

COPY requirements.txt .

RUN pip install --no-cache-dir -r requirements.txt

COPY . .

ENV FLASK_APP=app.py
ENV FLASK_ENV=production

EXPOSE 5000

HEALTHCHECK --interval=30s --timeout=10s --start-period=5s --retries=3 \\
    CMD curl -f http://localhost:5000/health || exit 1

CMD ["gunicorn", "--bind", "0.0.0.0:5000", "--workers", "4", "--timeout", "30", "app:app"]
"""
        
        dockerfile_path = os.path.join(package_dir, "Dockerfile")
        with open(dockerfile_path, 'w') as f:
            f.write(dockerfile_content)
        
        return dockerfile_path
    
    def run_model_container(self, image_name, container_name, port_mapping, environment_vars=None):
        """
        运行模型容器
        """
        try:
            # 容器配置
            container_config = {
                'image': image_name,
                'name': container_name,
                'ports': {5000: port_mapping},
                'environment': environment_vars or {},
                'restart_policy': {'Name': 'unless-stopped'},
                'healthcheck': {
                    'test': ['CMD', 'curl', '-f', 'http://localhost:5000/health'],
                    'interval': 30000000000,  # 30 seconds
                    'timeout': 10000000000,   # 10 seconds
                    'retries': 3
                }
            }
            
            # 创建并启动容器
            container = self.client.containers.run(**container_config, detach=True)
            
            print(f"模型容器 {container_name} 启动成功")
            return container
            
        except Exception as e:
            print(f"模型容器启动失败: {e}")
            raise
    
    def stop_model_container(self, container_name):
        """
        停止模型容器
        """
        try:
            container = self.client.containers.get(container_name)
            container.stop()
            container.remove()
            
            print(f"模型容器 {container_name} 已停止")
            
        except Exception as e:
            print(f"停止模型容器失败: {e}")
    
    def list_containers(self):
        """
        列出所有容器
        """
        containers = []
        for container in self.client.containers.list(all=True):
            containers.append({
                'name': container.name,
                'image': container.image.tags[0] if container.image.tags else container.image.id,
                'status': container.status,
                'created_at': container.attrs['Created']
            })
        
        return containers
    
    def get_container_logs(self, container_name, tail=100):
        """
        获取容器日志
        """
        try:
            container = self.client.containers.get(container_name)
            logs = container.logs(tail=tail).decode('utf-8')
            return logs
        except Exception as e:
            print(f"获取容器日志失败: {e}")
            return None
```

```python
import torch
import torch.quantization as quantization
from sklearn.preprocessing import StandardScaler
import numpy as np

class ModelOptimizer:
    def __init__(self):
        self.optimization_techniques = {}
        
    def quantize_model(self, model, calibration_data):
        """
        量化模型
        """
        try:
            # 设置量化配置
            model.qconfig = quantization.get_default_qconfig('fbgemm')
            
            # 准备量化
            model_prepared = quantization.prepare(model)
            
            # 校准
            model_prepared.eval()
            with torch.no_grad():
                for data in calibration_data:
                    model_prepared(data)
            
            # 转换为量化模型
            quantized_model = quantization.convert(model_prepared)
            
            print("模型量化完成")
            return quantized_model
            
        except Exception as e:
            print(f"模型量化失败: {e}")
            raise
    
    def prune_model(self, model, pruning_ratio=0.2):
        """
        剪枝模型
        """
        try:
            # 获取模型参数
            parameters_to_prune = []
            for name, module in model.named_modules():
                if isinstance(module, torch.nn.Linear):
                    parameters_to_prune.append((module, 'weight'))
            
            # 应用剪枝
            quantization.prune.global_unstructured(
                parameters_to_prune,
                pruning_method=quantization.prune.L1Unstructured,
                amount=pruning_ratio,
            )
            
            print(f"模型剪枝完成，剪枝比例: {pruning_ratio}")
            return model
            
        except Exception as e:
            print(f"模型剪枝失败: {e}")
            raise
    
    def optimize_inference(self, model, input_shape):
        """
        优化推理性能
        """
        try:
            # 设置模型为评估模式
            model.eval()
            
            # 使用torch.jit.script优化
            scripted_model = torch.jit.script(model)
            
            # 使用torch.jit.optimize_for_inference优化
            optimized_model = torch.jit.optimize_for_inference(scripted_model)
            
            print("模型推理优化完成")
            return optimized_model
            
        except Exception as e:
            print(f"模型推理优化失败: {e}")
            raise
    
    def benchmark_model(self, model, input_data, num_runs=100):
        """
        模型性能基准测试
        """
        try:
            model.eval()
            
            # 预热
            with torch.no_grad():
                for _ in range(10):
                    model(input_data)
            
            # 性能测试
            import time
            start_time = time.time()
            
            with torch.no_grad():
                for _ in range(num_runs):
                    model(input_data)
            
            end_time = time.time()
            
            # 计算性能指标
            total_time = end_time - start_time
            avg_time = total_time / num_runs
            throughput = 1 / avg_time
            
            benchmark_results = {
                'total_time': total_time,
                'avg_inference_time': avg_time,
                'throughput': throughput,
                'num_runs': num_runs
            }
            
            print(f"模型性能基准测试完成:")
            print(f"  平均推理时间: {avg_time:.4f}s")
            print(f"  吞吐量: {throughput:.2f} requests/s")
            
            return benchmark_results
            
        except Exception as e:
            print(f"模型性能基准测试失败: {e}")
            raise
```

#### 负载均衡
```python
import requests
import random
import time
from threading import Lock

class LoadBalancer:
    def __init__(self):
        self.services = []
        self.service_weights = {}
        self.service_health = {}
        self.lock = Lock()
        
    def add_service(self, service_url, weight=1):
        """
        添加服务
        """
        with self.lock:
            self.services.append(service_url)
            self.service_weights[service_url] = weight
            self.service_health[service_url] = True
    
    def remove_service(self, service_url):
        """
        移除服务
        """
        with self.lock:
            if service_url in self.services:
                self.services.remove(service_url)
                del self.service_weights[service_url]
                del self.service_health[service_url]
    
    def health_check(self, service_url):
        """
        健康检查
        """
        try:
            response = requests.get(f"{service_url}/health", timeout=5)
            return response.status_code == 200
        except:
            return False
    
    def update_service_health(self):
        """
        更新服务健康状态
        """
        with self.lock:
            for service_url in self.services:
                self.service_health[service_url] = self.health_check(service_url)
    
    def get_healthy_services(self):
        """
        获取健康服务
        """
        with self.lock:
            return [service for service in self.services if self.service_health[service]]
    
    def round_robin_select(self):
        """
        轮询选择
        """
        healthy_services = self.get_healthy_services()
        if not healthy_services:
            return None
        
        # 简单的轮询实现
        if not hasattr(self, '_current_index'):
            self._current_index = 0
        
        service = healthy_services[self._current_index % len(healthy_services)]
        self._current_index += 1
        
        return service
    
    def weighted_round_robin_select(self):
        """
        加权轮询选择
        """
        healthy_services = self.get_healthy_services()
        if not healthy_services:
            return None
        
        # 计算总权重
        total_weight = sum(self.service_weights[service] for service in healthy_services)
        
        # 随机选择
        random_value = random.uniform(0, total_weight)
        current_weight = 0
        
        for service in healthy_services:
            current_weight += self.service_weights[service]
            if random_value <= current_weight:
                return service
        
        return healthy_services[0]
    
    def least_connections_select(self):
        """
        最少连接选择
        """
        healthy_services = self.get_healthy_services()
        if not healthy_services:
            return None
        
        # 简化的最少连接实现
        # 实际应用中需要跟踪每个服务的连接数
        return random.choice(healthy_services)
    
    def send_request(self, endpoint, data, method='POST'):
        """
        发送请求
        """
        service_url = self.weighted_round_robin_select()
        if not service_url:
            raise Exception("没有可用的服务")
        
        try:
            url = f"{service_url}{endpoint}"
            if method == 'POST':
                response = requests.post(url, json=data, timeout=30)
            else:
                response = requests.get(url, timeout=30)
            
            response.raise_for_status()
            return response.json()
            
        except Exception as e:
            # 标记服务为不健康
            with self.lock:
                self.service_health[service_url] = False
            
            # 重试其他服务
            return self.send_request(endpoint, data, method)

class ServiceMonitor:
    def __init__(self):
        self.metrics = {}
        self.alerts = []
        
    def record_metric(self, service_name, metric_name, value, timestamp=None):
        """
        记录指标
        """
        if timestamp is None:
            timestamp = time.time()
        
        if service_name not in self.metrics:
            self.metrics[service_name] = {}
        
        if metric_name not in self.metrics[service_name]:
            self.metrics[service_name][metric_name] = []
        
        self.metrics[service_name][metric_name].append({
            'value': value,
            'timestamp': timestamp
        })
    
    def get_metric_stats(self, service_name, metric_name, time_window=3600):
        """
        获取指标统计
        """
        if service_name not in self.metrics or metric_name not in self.metrics[service_name]:
            return None
        
        current_time = time.time()
        recent_metrics = [
            m for m in self.metrics[service_name][metric_name]
            if current_time - m['timestamp'] <= time_window
        ]
        
        if not recent_metrics:
            return None
        
        values = [m['value'] for m in recent_metrics]
        
        return {
            'count': len(values),
            'min': min(values),
            'max': max(values),
            'avg': sum(values) / len(values),
            'latest': values[-1]
        }
    
    def check_alerts(self, service_name):
        """
        检查告警
        """
        alerts = []
        
        # 检查响应时间
        response_time_stats = self.get_metric_stats(service_name, 'response_time')
        if response_time_stats and response_time_stats['avg'] > 5.0:
            alerts.append({
                'type': 'high_response_time',
                'message': f"服务 {service_name} 平均响应时间过高: {response_time_stats['avg']:.2f}s",
                'severity': 'warning'
            })
        
        # 检查错误率
        error_rate_stats = self.get_metric_stats(service_name, 'error_rate')
        if error_rate_stats and error_rate_stats['avg'] > 0.1:
            alerts.append({
                'type': 'high_error_rate',
                'message': f"服务 {service_name} 错误率过高: {error_rate_stats['avg']:.2%}",
                'severity': 'critical'
            })
        
        return alerts
    
    def get_service_health(self, service_name):
        """
        获取服务健康状态
        """
        health = {
            'status': 'healthy',
            'metrics': {},
            'alerts': []
        }
        
        # 获取指标统计
        for metric_name in ['response_time', 'error_rate', 'throughput']:
            stats = self.get_metric_stats(service_name, metric_name)
            if stats:
                health['metrics'][metric_name] = stats
        
        # 检查告警
        alerts = self.check_alerts(service_name)
        health['alerts'] = alerts
        
        # 确定健康状态
        if any(alert['severity'] == 'critical' for alert in alerts):
            health['status'] = 'critical'
        elif any(alert['severity'] == 'warning' for alert in alerts):
            health['status'] = 'warning'
        
        return health
```
