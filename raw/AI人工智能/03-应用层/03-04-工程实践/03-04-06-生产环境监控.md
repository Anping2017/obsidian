# 生产环境监控

## 概述

生产环境监控是确保AI系统稳定运行的关键环节，涉及系统性能、模型质量、业务指标等多个维度的实时监控。通过系统化的监控体系，可以及时发现和解决问题，保障AI服务的可用性和可靠性。本章将介绍生产环境监控的核心概念、技术方案和最佳实践。

## 核心概念

### 1. 监控系统定义
- **目标**：实时监控AI系统的运行状态，确保服务稳定可靠
- **监控维度**：系统性能、模型质量、业务指标、用户体验
- **关键指标**：延迟、吞吐量、错误率、资源使用率
- **监控层次**：基础设施、应用服务、业务逻辑、用户体验

### 2. 监控架构模式
- **集中式监控**：单一监控中心
- **分布式监控**：多节点监控
- **混合监控**：集中式+分布式
- **云原生监控**：基于云平台的监控

```python
import psutil
import time
import threading
from collections import deque
import json
import logging

class SystemMonitor:
    def __init__(self, monitoring_interval=60):
        self.monitoring_interval = monitoring_interval
        self.metrics_history = {
            'cpu': deque(maxlen=1000),
            'memory': deque(maxlen=1000),
            'disk': deque(maxlen=1000),
            'network': deque(maxlen=1000)
        }
        self.alerts = []
        self.monitoring_thread = None
        self.is_monitoring = False
        
    def start_monitoring(self):
        """
        开始监控
        """
        if self.is_monitoring:
            return
        
        self.is_monitoring = True
        self.monitoring_thread = threading.Thread(target=self._monitor_loop)
        self.monitoring_thread.daemon = True
        self.monitoring_thread.start()
        
        print("系统监控已启动")
    
    def stop_monitoring(self):
        """
        停止监控
        """
        self.is_monitoring = False
        if self.monitoring_thread:
            self.monitoring_thread.join(timeout=5)
        
        print("系统监控已停止")
    
    def _monitor_loop(self):
        """
        监控循环
        """
        while self.is_monitoring:
            try:
                # 收集系统指标
                metrics = self._collect_system_metrics()
                
                # 存储历史数据
                self._store_metrics(metrics)
                
                # 检查告警
                self._check_alerts(metrics)
                
                # 等待下次监控
                time.sleep(self.monitoring_interval)
                
            except Exception as e:
                print(f"监控错误: {e}")
                time.sleep(self.monitoring_interval)
    
    def _collect_system_metrics(self):
        """
        收集系统指标
        """
        timestamp = time.time()
        
        # CPU指标
        cpu_percent = psutil.cpu_percent(interval=1)
        cpu_count = psutil.cpu_count()
        cpu_freq = psutil.cpu_freq()
        
        # 内存指标
        memory = psutil.virtual_memory()
        swap = psutil.swap_memory()
        
        # 磁盘指标
        disk = psutil.disk_usage('/')
        disk_io = psutil.disk_io_counters()
        
        # 网络指标
        network_io = psutil.net_io_counters()
        
        metrics = {
            'timestamp': timestamp,
            'cpu': {
                'percent': cpu_percent,
                'count': cpu_count,
                'frequency': cpu_freq.current if cpu_freq else None
            },
            'memory': {
                'total': memory.total,
                'available': memory.available,
                'used': memory.used,
                'percent': memory.percent,
                'swap_total': swap.total,
                'swap_used': swap.used,
                'swap_percent': swap.percent
            },
            'disk': {
                'total': disk.total,
                'used': disk.used,
                'free': disk.free,
                'percent': (disk.used / disk.total) * 100,
                'read_bytes': disk_io.read_bytes if disk_io else 0,
                'write_bytes': disk_io.write_bytes if disk_io else 0
            },
            'network': {
                'bytes_sent': network_io.bytes_sent if network_io else 0,
                'bytes_recv': network_io.bytes_recv if network_io else 0,
                'packets_sent': network_io.packets_sent if network_io else 0,
                'packets_recv': network_io.packets_recv if network_io else 0
            }
        }
        
        return metrics
    
    def _store_metrics(self, metrics):
        """
        存储指标
        """
        for metric_type, metric_data in metrics.items():
            if metric_type != 'timestamp' and metric_type in self.metrics_history:
                self.metrics_history[metric_type].append(metric_data)
    
    def _check_alerts(self, metrics):
        """
        检查告警
        """
        alerts = []
        
        # CPU告警
        if metrics['cpu']['percent'] > 80:
            alerts.append({
                'type': 'cpu_high',
                'message': f"CPU使用率过高: {metrics['cpu']['percent']:.1f}%",
                'severity': 'warning',
                'timestamp': metrics['timestamp']
            })
        
        # 内存告警
        if metrics['memory']['percent'] > 85:
            alerts.append({
                'type': 'memory_high',
                'message': f"内存使用率过高: {metrics['memory']['percent']:.1f}%",
                'severity': 'critical',
                'timestamp': metrics['timestamp']
            })
        
        # 磁盘告警
        if metrics['disk']['percent'] > 90:
            alerts.append({
                'type': 'disk_high',
                'message': f"磁盘使用率过高: {metrics['disk']['percent']:.1f}%",
                'severity': 'critical',
                'timestamp': metrics['timestamp']
            })
        
        # 存储告警
        self.alerts.extend(alerts)
        
        # 发送告警通知
        for alert in alerts:
            self._send_alert(alert)
    
    def _send_alert(self, alert):
        """
        发送告警
        """
        print(f"告警: {alert['message']}")
        # 这里可以实现邮件、短信、Slack等通知方式
    
    def get_metrics_summary(self, time_window=3600):
        """
        获取指标摘要
        """
        current_time = time.time()
        start_time = current_time - time_window
        
        summary = {}
        
        for metric_type, history in self.metrics_history.items():
            if not history:
                continue
            
            # 过滤时间窗口内的数据
            recent_data = [
                data for data in history 
                if data.get('timestamp', 0) >= start_time
            ]
            
            if not recent_data:
                continue
            
            # 计算统计指标
            if metric_type == 'cpu':
                values = [data['percent'] for data in recent_data]
                summary[metric_type] = {
                    'avg': sum(values) / len(values),
                    'max': max(values),
                    'min': min(values),
                    'current': values[-1] if values else 0
                }
            elif metric_type == 'memory':
                values = [data['percent'] for data in recent_data]
                summary[metric_type] = {
                    'avg': sum(values) / len(values),
                    'max': max(values),
                    'min': min(values),
                    'current': values[-1] if values else 0
                }
            elif metric_type == 'disk':
                values = [data['percent'] for data in recent_data]
                summary[metric_type] = {
                    'avg': sum(values) / len(values),
                    'max': max(values),
                    'min': min(values),
                    'current': values[-1] if values else 0
                }
        
        return summary
    
    def get_alerts(self, severity=None, time_window=3600):
        """
        获取告警
        """
        current_time = time.time()
        start_time = current_time - time_window
        
        filtered_alerts = [
            alert for alert in self.alerts
            if alert['timestamp'] >= start_time
        ]
        
        if severity:
            filtered_alerts = [
                alert for alert in filtered_alerts
                if alert['severity'] == severity
            ]
        
        return filtered_alerts

class ContainerMonitor:
    def __init__(self):
        self.container_metrics = {}
        
    def monitor_container(self, container_id):
        """
        监控容器
        """
        try:
            import docker
            
            client = docker.from_env()
            container = client.containers.get(container_id)
            
            # 获取容器统计信息
            stats = container.stats(stream=False)
            
            # 解析统计信息
            metrics = self._parse_container_stats(stats)
            
            self.container_metrics[container_id] = metrics
            
            return metrics
            
        except Exception as e:
            print(f"容器监控失败: {e}")
            return None
    
    def _parse_container_stats(self, stats):
        """
        解析容器统计信息
        """
        # CPU使用率
        cpu_usage = stats['cpu_stats']['cpu_usage']['total_usage']
        cpu_system = stats['cpu_stats']['system_cpu_usage']
        cpu_percent = (cpu_usage / cpu_system) * 100 if cpu_system > 0 else 0
        
        # 内存使用
        memory_usage = stats['memory_stats']['usage']
        memory_limit = stats['memory_stats']['limit']
        memory_percent = (memory_usage / memory_limit) * 100 if memory_limit > 0 else 0
        
        # 网络统计
        network_stats = stats['networks']
        network_rx = sum(net['rx_bytes'] for net in network_stats.values())
        network_tx = sum(net['tx_bytes'] for net in network_stats.values())
        
        return {
            'timestamp': time.time(),
            'cpu_percent': cpu_percent,
            'memory_usage': memory_usage,
            'memory_limit': memory_limit,
            'memory_percent': memory_percent,
            'network_rx': network_rx,
            'network_tx': network_tx
        }
    
    def get_container_health(self, container_id):
        """
        获取容器健康状态
        """
        if container_id not in self.container_metrics:
            return None
        
        metrics = self.container_metrics[container_id]
        
        health_status = 'healthy'
        issues = []
        
        if metrics['cpu_percent'] > 80:
            health_status = 'warning'
            issues.append('CPU使用率过高')
        
        if metrics['memory_percent'] > 85:
            health_status = 'critical'
            issues.append('内存使用率过高')
        
        return {
            'status': health_status,
            'issues': issues,
            'metrics': metrics
        }
```

```python
import time
import threading
from collections import defaultdict, deque
import statistics

class ApplicationMonitor:
    def __init__(self):
        self.request_metrics = defaultdict(lambda: deque(maxlen=1000))
        self.error_metrics = defaultdict(lambda: deque(maxlen=1000))
        self.performance_metrics = defaultdict(lambda: deque(maxlen=1000))
        self.business_metrics = defaultdict(lambda: deque(maxlen=1000))
        
    def record_request(self, endpoint, method, response_time, status_code):
        """
        记录请求指标
        """
        timestamp = time.time()
        
        request_data = {
            'timestamp': timestamp,
            'endpoint': endpoint,
            'method': method,
            'response_time': response_time,
            'status_code': status_code
        }
        
        self.request_metrics[endpoint].append(request_data)
        
        # 记录性能指标
        self.performance_metrics[endpoint].append({
            'timestamp': timestamp,
            'response_time': response_time,
            'status_code': status_code
        })
        
        # 记录错误指标
        if status_code >= 400:
            self.error_metrics[endpoint].append({
                'timestamp': timestamp,
                'status_code': status_code,
                'response_time': response_time
            })
    
    def record_business_metric(self, metric_name, value, tags=None):
        """
        记录业务指标
        """
        timestamp = time.time()
        
        business_data = {
            'timestamp': timestamp,
            'value': value,
            'tags': tags or {}
        }
        
        self.business_metrics[metric_name].append(business_data)
    
    def get_endpoint_metrics(self, endpoint, time_window=3600):
        """
        获取端点指标
        """
        current_time = time.time()
        start_time = current_time - time_window
        
        # 过滤时间窗口内的数据
        recent_requests = [
            req for req in self.request_metrics[endpoint]
            if req['timestamp'] >= start_time
        ]
        
        if not recent_requests:
            return None
        
        # 计算指标
        response_times = [req['response_time'] for req in recent_requests]
        status_codes = [req['status_code'] for req in recent_requests]
        
        # 计算统计指标
        metrics = {
            'total_requests': len(recent_requests),
            'avg_response_time': statistics.mean(response_times),
            'min_response_time': min(response_times),
            'max_response_time': max(response_times),
            'p95_response_time': self._calculate_percentile(response_times, 95),
            'p99_response_time': self._calculate_percentile(response_times, 99),
            'error_rate': sum(1 for code in status_codes if code >= 400) / len(status_codes),
            'success_rate': sum(1 for code in status_codes if code < 400) / len(status_codes)
        }
        
        return metrics
    
    def _calculate_percentile(self, data, percentile):
        """
        计算百分位数
        """
        if not data:
            return 0
        
        sorted_data = sorted(data)
        index = int(len(sorted_data) * percentile / 100)
        return sorted_data[min(index, len(sorted_data) - 1)]
    
    def get_system_metrics(self, time_window=3600):
        """
        获取系统指标
        """
        current_time = time.time()
        start_time = current_time - time_window
        
        all_requests = []
        for endpoint_requests in self.request_metrics.values():
            all_requests.extend([
                req for req in endpoint_requests
                if req['timestamp'] >= start_time
            ])
        
        if not all_requests:
            return None
        
        # 计算系统级指标
        response_times = [req['response_time'] for req in all_requests]
        status_codes = [req['status_code'] for req in all_requests]
        
        # 计算吞吐量
        time_span = current_time - start_time
        throughput = len(all_requests) / time_span if time_span > 0 else 0
        
        metrics = {
            'total_requests': len(all_requests),
            'throughput': throughput,
            'avg_response_time': statistics.mean(response_times),
            'p95_response_time': self._calculate_percentile(response_times, 95),
            'p99_response_time': self._calculate_percentile(response_times, 99),
            'error_rate': sum(1 for code in status_codes if code >= 400) / len(status_codes),
            'success_rate': sum(1 for code in status_codes if code < 400) / len(status_codes)
        }
        
        return metrics
    
    def get_business_metrics(self, metric_name, time_window=3600):
        """
        获取业务指标
        """
        current_time = time.time()
        start_time = current_time - time_window
        
        if metric_name not in self.business_metrics:
            return None
        
        # 过滤时间窗口内的数据
        recent_metrics = [
            metric for metric in self.business_metrics[metric_name]
            if metric['timestamp'] >= start_time
        ]
        
        if not recent_metrics:
            return None
        
        values = [metric['value'] for metric in recent_metrics]
        
        return {
            'count': len(recent_metrics),
            'sum': sum(values),
            'avg': statistics.mean(values),
            'min': min(values),
            'max': max(values),
            'latest': values[-1] if values else 0
        }
    
    def check_performance_alerts(self, endpoint=None, time_window=3600):
        """
        检查性能告警
        """
        alerts = []
        
        if endpoint:
            metrics = self.get_endpoint_metrics(endpoint, time_window)
            if metrics:
                alerts.extend(self._check_endpoint_alerts(endpoint, metrics))
        else:
            system_metrics = self.get_system_metrics(time_window)
            if system_metrics:
                alerts.extend(self._check_system_alerts(system_metrics))
        
        return alerts
    
    def _check_endpoint_alerts(self, endpoint, metrics):
        """
        检查端点告警
        """
        alerts = []
        
        # 响应时间告警
        if metrics['avg_response_time'] > 5.0:  # 5秒阈值
            alerts.append({
                'type': 'high_response_time',
                'endpoint': endpoint,
                'message': f"端点 {endpoint} 平均响应时间过高: {metrics['avg_response_time']:.2f}s",
                'severity': 'warning'
            })
        
        # 错误率告警
        if metrics['error_rate'] > 0.05:  # 5%阈值
            alerts.append({
                'type': 'high_error_rate',
                'endpoint': endpoint,
                'message': f"端点 {endpoint} 错误率过高: {metrics['error_rate']:.2%}",
                'severity': 'critical'
            })
        
        return alerts
    
    def _check_system_alerts(self, metrics):
        """
        检查系统告警
        """
        alerts = []
        
        # 系统响应时间告警
        if metrics['avg_response_time'] > 3.0:  # 3秒阈值
            alerts.append({
                'type': 'system_high_response_time',
                'message': f"系统平均响应时间过高: {metrics['avg_response_time']:.2f}s",
                'severity': 'warning'
            })
        
        # 系统错误率告警
        if metrics['error_rate'] > 0.03:  # 3%阈值
            alerts.append({
                'type': 'system_high_error_rate',
                'message': f"系统错误率过高: {metrics['error_rate']:.2%}",
                'severity': 'critical'
            })
        
        return alerts

class DatabaseMonitor:
    def __init__(self, connection_string):
        self.connection_string = connection_string
        self.metrics_history = deque(maxlen=1000)
        
    def monitor_database(self):
        """
        监控数据库
        """
        try:
            import psycopg2
            
            conn = psycopg2.connect(self.connection_string)
            cursor = conn.cursor()
            
            # 获取数据库统计信息
            metrics = self._collect_database_metrics(cursor)
            
            self.metrics_history.append(metrics)
            
            cursor.close()
            conn.close()
            
            return metrics
            
        except Exception as e:
            print(f"数据库监控失败: {e}")
            return None
    
    def _collect_database_metrics(self, cursor):
        """
        收集数据库指标
        """
        timestamp = time.time()
        
        # 获取连接数
        cursor.execute("SELECT count(*) FROM pg_stat_activity")
        active_connections = cursor.fetchone()[0]
        
        # 获取数据库大小
        cursor.execute("SELECT pg_database_size(current_database())")
        database_size = cursor.fetchone()[0]
        
        # 获取表统计信息
        cursor.execute("""
            SELECT schemaname, tablename, n_tup_ins, n_tup_upd, n_tup_del
            FROM pg_stat_user_tables
            ORDER BY n_tup_ins + n_tup_upd + n_tup_del DESC
            LIMIT 10
        """)
        table_stats = cursor.fetchall()
        
        # 获取慢查询
        cursor.execute("""
            SELECT query, mean_time, calls
            FROM pg_stat_statements
            ORDER BY mean_time DESC
            LIMIT 5
        """)
        slow_queries = cursor.fetchall()
        
        return {
            'timestamp': timestamp,
            'active_connections': active_connections,
            'database_size': database_size,
            'table_stats': table_stats,
            'slow_queries': slow_queries
        }
    
    def get_database_health(self):
        """
        获取数据库健康状态
        """
        if not self.metrics_history:
            return None
        
        latest_metrics = self.metrics_history[-1]
        
        health_status = 'healthy'
        issues = []
        
        # 检查连接数
        if latest_metrics['active_connections'] > 100:
            health_status = 'warning'
            issues.append('活跃连接数过多')
        
        # 检查数据库大小
        if latest_metrics['database_size'] > 10 * 1024 * 1024 * 1024:  # 10GB
            health_status = 'warning'
            issues.append('数据库大小过大')
        
        return {
            'status': health_status,
            'issues': issues,
            'metrics': latest_metrics
        }
```

## 模型质量监控

### 1. 模型性能监控

#### 模型指标监控
```python
import numpy as np
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import pandas as pd

class ModelMonitor:
    def __init__(self):
        self.model_metrics = {}
        self.prediction_history = deque(maxlen=10000)
        self.performance_history = deque(maxlen=1000)
        
    def record_prediction(self, model_name, input_data, prediction, actual=None):
        """
        记录预测结果
        """
        timestamp = time.time()
        
        prediction_record = {
            'timestamp': timestamp,
            'model_name': model_name,
            'input_data': input_data,
            'prediction': prediction,
            'actual': actual
        }
        
        self.prediction_history.append(prediction_record)
        
        # 如果有实际值，计算性能指标
        if actual is not None:
            self._calculate_performance_metrics(model_name, prediction, actual, timestamp)
    
    def _calculate_performance_metrics(self, model_name, prediction, actual, timestamp):
        """
        计算性能指标
        """
        # 计算准确率
        accuracy = accuracy_score([actual], [prediction])
        
        # 计算精确率、召回率、F1分数
        precision = precision_score([actual], [prediction], average='weighted')
        recall = recall_score([actual], [prediction], average='weighted')
        f1 = f1_score([actual], [prediction], average='weighted')
        
        performance_record = {
            'timestamp': timestamp,
            'model_name': model_name,
            'accuracy': accuracy,
            'precision': precision,
            'recall': recall,
            'f1_score': f1
        }
        
        self.performance_history.append(performance_record)
    
    def get_model_performance(self, model_name, time_window=3600):
        """
        获取模型性能
        """
        current_time = time.time()
        start_time = current_time - time_window
        
        # 过滤时间窗口内的数据
        recent_performance = [
            perf for perf in self.performance_history
            if perf['model_name'] == model_name and perf['timestamp'] >= start_time
        ]
        
        if not recent_performance:
            return None
        
        # 计算平均性能指标
        avg_accuracy = np.mean([perf['accuracy'] for perf in recent_performance])
        avg_precision = np.mean([perf['precision'] for perf in recent_performance])
        avg_recall = np.mean([perf['recall'] for perf in recent_performance])
        avg_f1 = np.mean([perf['f1_score'] for perf in recent_performance])
        
        return {
            'model_name': model_name,
            'time_window': time_window,
            'avg_accuracy': avg_accuracy,
            'avg_precision': avg_precision,
            'avg_recall': avg_recall,
            'avg_f1_score': avg_f1,
            'sample_count': len(recent_performance)
        }
    
    def detect_model_drift(self, model_name, time_window=3600, reference_window=3600):
        """
        检测模型漂移
        """
        current_time = time.time()
        
        # 获取当前时间窗口的性能
        current_start = current_time - time_window
        current_performance = [
            perf for perf in self.performance_history
            if perf['model_name'] == model_name and perf['timestamp'] >= current_start
        ]
        
        # 获取参考时间窗口的性能
        reference_start = current_time - time_window - reference_window
        reference_end = current_time - time_window
        reference_performance = [
            perf for perf in self.performance_history
            if perf['model_name'] == model_name and 
            reference_start <= perf['timestamp'] <= reference_end
        ]
        
        if not current_performance or not reference_performance:
            return None
        
        # 计算性能差异
        current_accuracy = np.mean([perf['accuracy'] for perf in current_performance])
        reference_accuracy = np.mean([perf['accuracy'] for perf in reference_performance])
        
        accuracy_drift = abs(current_accuracy - reference_accuracy)
        
        # 检测漂移
        drift_detected = accuracy_drift > 0.05  # 5%的阈值
        
        return {
            'model_name': model_name,
            'drift_detected': drift_detected,
            'accuracy_drift': accuracy_drift,
            'current_accuracy': current_accuracy,
            'reference_accuracy': reference_accuracy,
            'drift_threshold': 0.05
        }
    
    def get_prediction_distribution(self, model_name, time_window=3600):
        """
        获取预测分布
        """
        current_time = time.time()
        start_time = current_time - time_window
        
        # 过滤时间窗口内的数据
        recent_predictions = [
            pred for pred in self.prediction_history
            if pred['model_name'] == model_name and pred['timestamp'] >= start_time
        ]
        
        if not recent_predictions:
            return None
        
        # 计算预测分布
        predictions = [pred['prediction'] for pred in recent_predictions]
        
        if isinstance(predictions[0], (int, float)):
            # 数值预测
            distribution = {
                'mean': np.mean(predictions),
                'std': np.std(predictions),
                'min': np.min(predictions),
                'max': np.max(predictions),
                'median': np.median(predictions)
            }
        else:
            # 分类预测
            unique_values, counts = np.unique(predictions, return_counts=True)
            distribution = {
                'unique_values': unique_values.tolist(),
                'counts': counts.tolist(),
                'probabilities': (counts / len(predictions)).tolist()
            }
        
        return {
            'model_name': model_name,
            'time_window': time_window,
            'distribution': distribution,
            'sample_count': len(recent_predictions)
        }
    
    def check_model_alerts(self, model_name, time_window=3600):
        """
        检查模型告警
        """
        alerts = []
        
        # 获取模型性能
        performance = self.get_model_performance(model_name, time_window)
        if not performance:
            return alerts
        
        # 性能下降告警
        if performance['avg_accuracy'] < 0.8:  # 80%阈值
            alerts.append({
                'type': 'low_accuracy',
                'model_name': model_name,
                'message': f"模型 {model_name} 准确率过低: {performance['avg_accuracy']:.3f}",
                'severity': 'warning'
            })
        
        # 检测模型漂移
        drift_result = self.detect_model_drift(model_name, time_window)
        if drift_result and drift_result['drift_detected']:
            alerts.append({
                'type': 'model_drift',
                'model_name': model_name,
                'message': f"模型 {model_name} 检测到漂移: {drift_result['accuracy_drift']:.3f}",
                'severity': 'critical'
            })
        
        return alerts

class DataDriftDetector:
    def __init__(self):
        self.reference_data = {}
        self.drift_threshold = 0.1
        
    def set_reference_data(self, model_name, reference_data):
        """
        设置参考数据
        """
        self.reference_data[model_name] = reference_data
        
    def detect_drift(self, model_name, current_data):
        """
        检测数据漂移
        """
        if model_name not in self.reference_data:
            return None
        
        reference_data = self.reference_data[model_name]
        
        # 计算分布差异
        drift_scores = {}
        
        for column in current_data.columns:
            if column in reference_data.columns:
                # 计算KS检验
                from scipy import stats
                
                reference_values = reference_data[column].dropna()
                current_values = current_data[column].dropna()
                
                if len(reference_values) > 0 and len(current_values) > 0:
                    statistic, p_value = stats.ks_2samp(reference_values, current_values)
                    drift_scores[column] = {
                        'ks_statistic': statistic,
                        'p_value': p_value,
                        'drift_detected': statistic > self.drift_threshold
                    }
        
        # 计算总体漂移
        drift_detected = any(
            score['drift_detected'] for score in drift_scores.values()
        )
        
        return {
            'model_name': model_name,
            'drift_detected': drift_detected,
            'drift_scores': drift_scores,
            'drift_threshold': self.drift_threshold
        }
    
    def update_reference_data(self, model_name, new_data, update_method='replace'):
        """
        更新参考数据
        """
        if update_method == 'replace':
            self.reference_data[model_name] = new_data
        elif update_method == 'append':
            if model_name in self.reference_data:
                self.reference_data[model_name] = pd.concat([
                    self.reference_data[model_name], new_data
                ])
            else:
                self.reference_data[model_name] = new_data
        elif update_method == 'rolling':
            if model_name in self.reference_data:
                # 滚动更新，保留最新的数据
                combined_data = pd.concat([
                    self.reference_data[model_name], new_data
                ])
                self.reference_data[model_name] = combined_data.tail(10000)  # 保留最新10000条
            else:
                self.reference_data[model_name] = new_data
```

## 总结

生产环境监控是确保AI系统稳定运行的关键：

1. **系统性能监控**：CPU、内存、磁盘、网络等基础设施监控
2. **应用性能监控**：请求指标、响应时间、错误率等应用层监控
3. **模型质量监控**：模型性能、数据漂移、预测质量等模型层监控
4. **告警与通知**：实时告警、多渠道通知、告警管理

通过系统化的监控体系，可以及时发现和解决问题，保障AI服务的可用性和可靠性。
