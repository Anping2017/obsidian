# MLOps实践

## 概述

MLOps（Machine Learning Operations）是机器学习工程化的核心实践，它将DevOps的理念应用到机器学习项目中，实现ML模型的持续集成、持续部署和持续监控。本章将介绍MLOps的核心概念、实践方法和工具链。

## 核心概念

### 1. MLOps定义
- **目标**：实现ML模型的自动化、可重复和可扩展的部署与管理
- **核心要素**：版本控制、自动化流水线、模型监控、A/B测试
- **关键原则**：可重现性、可观测性、可扩展性、可维护性
- **价值**：提高模型质量、加速部署速度、降低运维成本

### 2. MLOps成熟度模型
- **Level 0：手动流程**：完全手动的ML工作流
- **Level 1：ML流水线自动化**：自动化的训练和部署
- **Level 2：CI/CD流水线**：持续集成和持续部署
- **Level 3：持续训练**：自动化的模型重训练

```python
import dvc
import pandas as pd
import numpy as np
from datetime import datetime
import hashlib
import json
import os

class DataVersionManager:
    def __init__(self, repo_path="."):
        self.repo_path = repo_path
        self.data_registry = {}
        self.data_lineage = {}
        
    def register_dataset(self, dataset_name, data_path, metadata=None):
        """
        注册数据集
        """
        # 计算数据哈希
        data_hash = self._calculate_data_hash(data_path)
        
        # 创建数据集记录
        dataset_record = {
            'name': dataset_name,
            'path': data_path,
            'hash': data_hash,
            'size': self._get_data_size(data_path),
            'created_at': datetime.now().isoformat(),
            'metadata': metadata or {},
            'version': self._get_next_version(dataset_name)
        }
        
        # 注册到DVC
        self._register_with_dvc(dataset_name, data_path)
        
        # 保存到注册表
        self.data_registry[dataset_name] = dataset_record
        
        # 更新数据血缘
        self._update_data_lineage(dataset_name, dataset_record)
        
        print(f"数据集 {dataset_name} v{dataset_record['version']} 已注册")
        return dataset_record
    
    def _calculate_data_hash(self, data_path):
        """
        计算数据哈希
        """
        if os.path.isfile(data_path):
            with open(data_path, 'rb') as f:
                return hashlib.md5(f.read()).hexdigest()
        elif os.path.isdir(data_path):
            # 计算目录下所有文件的哈希
            hashes = []
            for root, dirs, files in os.walk(data_path):
                for file in files:
                    file_path = os.path.join(root, file)
                    with open(file_path, 'rb') as f:
                        hashes.append(hashlib.md5(f.read()).hexdigest())
            return hashlib.md5(''.join(hashes).encode()).hexdigest()
        else:
            raise ValueError(f"数据路径不存在: {data_path}")
    
    def _get_data_size(self, data_path):
        """
        获取数据大小
        """
        if os.path.isfile(data_path):
            return os.path.getsize(data_path)
        elif os.path.isdir(data_path):
            total_size = 0
            for root, dirs, files in os.walk(data_path):
                for file in files:
                    file_path = os.path.join(root, file)
                    total_size += os.path.getsize(file_path)
            return total_size
        else:
            return 0
    
    def _get_next_version(self, dataset_name):
        """
        获取下一个版本号
        """
        if dataset_name not in self.data_registry:
            return "1.0.0"
        
        current_version = self.data_registry[dataset_name]['version']
        major, minor, patch = map(int, current_version.split('.'))
        
        if patch < 99:
            patch += 1
        elif minor < 99:
            minor += 1
            patch = 0
        else:
            major += 1
            minor = 0
            patch = 0
        
        return f"{major}.{minor}.{patch}"
    
    def _register_with_dvc(self, dataset_name, data_path):
        """
        注册到DVC
        """
        try:
            # 添加数据到DVC
            dvc.add(data_path)
            
            # 提交到Git
            dvc.commit(f"Add dataset {dataset_name}")
            
            print(f"数据集 {dataset_name} 已添加到DVC")
        except Exception as e:
            print(f"DVC注册失败: {e}")
    
    def _update_data_lineage(self, dataset_name, dataset_record):
        """
        更新数据血缘
        """
        self.data_lineage[dataset_name] = {
            'current_version': dataset_record['version'],
            'history': [dataset_record],
            'dependencies': [],
            'dependents': []
        }
    
    def get_dataset_version(self, dataset_name, version=None):
        """
        获取数据集版本
        """
        if dataset_name not in self.data_registry:
            raise ValueError(f"数据集 {dataset_name} 未注册")
        
        if version is None:
            return self.data_registry[dataset_name]
        
        # 查找指定版本
        for record in self.data_lineage[dataset_name]['history']:
            if record['version'] == version:
                return record
        
        raise ValueError(f"数据集 {dataset_name} 版本 {version} 不存在")
    
    def compare_datasets(self, dataset_name, version1, version2):
        """
        比较数据集版本
        """
        try:
            record1 = self.get_dataset_version(dataset_name, version1)
            record2 = self.get_dataset_version(dataset_name, version2)
            
            comparison = {
                'dataset_name': dataset_name,
                'version1': version1,
                'version2': version2,
                'size_diff': record2['size'] - record1['size'],
                'hash_diff': record1['hash'] != record2['hash'],
                'metadata_diff': self._compare_metadata(record1['metadata'], record2['metadata'])
            }
            
            return comparison
            
        except Exception as e:
            print(f"数据集比较失败: {e}")
            return None
    
    def _compare_metadata(self, metadata1, metadata2):
        """
        比较元数据
        """
        diff = {}
        
        all_keys = set(metadata1.keys()) | set(metadata2.keys())
        
        for key in all_keys:
            if key not in metadata1:
                diff[key] = {'action': 'added', 'value': metadata2[key]}
            elif key not in metadata2:
                diff[key] = {'action': 'removed', 'value': metadata1[key]}
            elif metadata1[key] != metadata2[key]:
                diff[key] = {
                    'action': 'changed',
                    'old_value': metadata1[key],
                    'new_value': metadata2[key]
                }
        
        return diff

class DataPipeline:
    def __init__(self):
        self.steps = []
        self.data_manager = DataVersionManager()
        
    def add_step(self, step_name, step_func, dependencies=None):
        """
        添加流水线步骤
        """
        self.steps.append({
            'name': step_name,
            'function': step_func,
            'dependencies': dependencies or [],
            'outputs': []
        })
    
    def run_pipeline(self, input_data, pipeline_name="default"):
        """
        运行流水线
        """
        pipeline_results = {
            'name': pipeline_name,
            'start_time': datetime.now().isoformat(),
            'steps': [],
            'final_output': None
        }
        
        current_data = input_data
        
        for step in self.steps:
            try:
                print(f"执行步骤: {step['name']}")
                
                # 执行步骤
                step_result = step['function'](current_data)
                
                # 记录步骤结果
                step_record = {
                    'name': step['name'],
                    'status': 'success',
                    'output_type': type(step_result).__name__,
                    'timestamp': datetime.now().isoformat()
                }
                
                pipeline_results['steps'].append(step_record)
                
                # 更新当前数据
                current_data = step_result
                
                print(f"步骤 {step['name']} 执行成功")
                
            except Exception as e:
                print(f"步骤 {step['name']} 执行失败: {e}")
                
                step_record = {
                    'name': step['name'],
                    'status': 'failed',
                    'error': str(e),
                    'timestamp': datetime.now().isoformat()
                }
                
                pipeline_results['steps'].append(step_record)
                break
        
        pipeline_results['end_time'] = datetime.now().isoformat()
        pipeline_results['final_output'] = current_data
        
        return pipeline_results

def data_validation_step(data):
    """
    数据验证步骤
    """
    if isinstance(data, pd.DataFrame):
        # 检查数据完整性
        missing_count = data.isnull().sum().sum()
        if missing_count > 0:
            print(f"警告: 发现 {missing_count} 个缺失值")
        
        # 检查数据类型
        for column in data.columns:
            if data[column].dtype == 'object':
                try:
                    data[column] = pd.to_numeric(data[column])
                except:
                    pass
    
    return data

def data_cleaning_step(data):
    """
    数据清洗步骤
    """
    if isinstance(data, pd.DataFrame):
        # 处理缺失值
        data = data.fillna(data.median())
        
        # 移除重复值
        data = data.drop_duplicates()
        
        # 异常值处理
        for column in data.select_dtypes(include=[np.number]).columns:
            Q1 = data[column].quantile(0.25)
            Q3 = data[column].quantile(0.75)
            IQR = Q3 - Q1
            lower_bound = Q1 - 1.5 * IQR
            upper_bound = Q3 + 1.5 * IQR
            data = data[(data[column] >= lower_bound) & (data[column] <= upper_bound)]
    
    return data

def data_transformation_step(data):
    """
    数据转换步骤
    """
    if isinstance(data, pd.DataFrame):
        # 标准化
        from sklearn.preprocessing import StandardScaler
        scaler = StandardScaler()
        
        numeric_columns = data.select_dtypes(include=[np.number]).columns
        data[numeric_columns] = scaler.fit_transform(data[numeric_columns])
    
    return data
```

### 2. 模型训练流水线

#### 自动化训练流程
```python
import mlflow
import mlflow.sklearn
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import joblib
import json

class MLTrainingPipeline:
    def __init__(self, experiment_name="default"):
        self.experiment_name = experiment_name
        self.models = {}
        self.training_history = {}
        
        # 设置MLflow实验
        mlflow.set_experiment(experiment_name)
    
    def train_model(self, model_name, model_class, data, target_column, 
                   hyperparameters=None, validation_split=0.2):
        """
        训练模型
        """
        try:
            # 准备数据
            X = data.drop(columns=[target_column])
            y = data[target_column]
            
            # 分割数据
            X_train, X_val, y_train, y_val = train_test_split(
                X, y, test_size=validation_split, random_state=42
            )
            
            # 设置超参数
            if hyperparameters:
                model = model_class(**hyperparameters)
            else:
                model = model_class()
            
            # 开始MLflow运行
            with mlflow.start_run(run_name=f"{model_name}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
                # 记录参数
                mlflow.log_params(hyperparameters or {})
                
                # 训练模型
                model.fit(X_train, y_train)
                
                # 验证模型
                y_pred = model.predict(X_val)
                
                # 计算指标
                metrics = self._calculate_metrics(y_val, y_pred)
                
                # 记录指标
                mlflow.log_metrics(metrics)
                
                # 记录模型
                mlflow.sklearn.log_model(model, "model")
                
                # 保存训练历史
                training_record = {
                    'model_name': model_name,
                    'model_class': model_class.__name__,
                    'hyperparameters': hyperparameters,
                    'metrics': metrics,
                    'training_data_size': len(X_train),
                    'validation_data_size': len(X_val),
                    'timestamp': datetime.now().isoformat()
                }
                
                self.training_history[model_name] = training_record
                self.models[model_name] = model
                
                print(f"模型 {model_name} 训练完成")
                print(f"验证准确率: {metrics['accuracy']:.4f}")
                
                return model, metrics
                
        except Exception as e:
            print(f"模型训练失败: {e}")
            raise
    
    def _calculate_metrics(self, y_true, y_pred):
        """
        计算评估指标
        """
        metrics = {
            'accuracy': accuracy_score(y_true, y_pred),
            'precision': precision_score(y_true, y_pred, average='weighted'),
            'recall': recall_score(y_true, y_pred, average='weighted'),
            'f1_score': f1_score(y_true, y_pred, average='weighted')
        }
        
        return metrics
    
    def hyperparameter_tuning(self, model_name, model_class, data, target_column,
                             param_grid, cv_folds=5):
        """
        超参数调优
        """
        try:
            from sklearn.model_selection import GridSearchCV
            
            # 准备数据
            X = data.drop(columns=[target_column])
            y = data[target_column]
            
            # 创建基础模型
            base_model = model_class()
            
            # 网格搜索
            grid_search = GridSearchCV(
                base_model,
                param_grid,
                cv=cv_folds,
                scoring='accuracy',
                n_jobs=-1
            )
            
            # 开始MLflow运行
            with mlflow.start_run(run_name=f"{model_name}_tuning_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
                # 执行搜索
                grid_search.fit(X, y)
                
                # 记录最佳参数
                mlflow.log_params(grid_search.best_params_)
                
                # 记录最佳分数
                mlflow.log_metric("best_cv_score", grid_search.best_score_)
                
                # 记录模型
                mlflow.sklearn.log_model(grid_search.best_estimator_, "model")
                
                tuning_results = {
                    'model_name': model_name,
                    'best_params': grid_search.best_params_,
                    'best_score': grid_search.best_score_,
                    'cv_results': grid_search.cv_results_,
                    'timestamp': datetime.now().isoformat()
                }
                
                print(f"超参数调优完成")
                print(f"最佳参数: {grid_search.best_params_}")
                print(f"最佳分数: {grid_search.best_score_:.4f}")
                
                return grid_search.best_estimator_, tuning_results
                
        except Exception as e:
            print(f"超参数调优失败: {e}")
            raise
    
    def cross_validation(self, model_name, model_class, data, target_column,
                        hyperparameters=None, cv_folds=5):
        """
        交叉验证
        """
        try:
            # 准备数据
            X = data.drop(columns=[target_column])
            y = data[target_column]
            
            # 创建模型
            if hyperparameters:
                model = model_class(**hyperparameters)
            else:
                model = model_class()
            
            # 执行交叉验证
            cv_scores = cross_val_score(model, X, y, cv=cv_folds, scoring='accuracy')
            
            cv_results = {
                'model_name': model_name,
                'cv_scores': cv_scores.tolist(),
                'mean_score': cv_scores.mean(),
                'std_score': cv_scores.std(),
                'cv_folds': cv_folds,
                'timestamp': datetime.now().isoformat()
            }
            
            print(f"交叉验证完成")
            print(f"平均分数: {cv_scores.mean():.4f} (+/- {cv_scores.std() * 2:.4f})")
            
            return cv_results
            
        except Exception as e:
            print(f"交叉验证失败: {e}")
            raise
    
    def model_comparison(self, models_config, data, target_column):
        """
        模型比较
        """
        try:
            comparison_results = {}
            
            for model_name, config in models_config.items():
                model_class = config['class']
                hyperparameters = config.get('hyperparameters', {})
                
                # 训练模型
                model, metrics = self.train_model(
                    model_name, model_class, data, target_column, hyperparameters
                )
                
                comparison_results[model_name] = {
                    'metrics': metrics,
                    'model': model
                }
            
            # 找出最佳模型
            best_model = max(comparison_results.items(), 
                           key=lambda x: x[1]['metrics']['accuracy'])
            
            print(f"最佳模型: {best_model[0]}")
            print(f"最佳准确率: {best_model[1]['metrics']['accuracy']:.4f}")
            
            return comparison_results, best_model
            
        except Exception as e:
            print(f"模型比较失败: {e}")
            raise
    
    def save_model(self, model_name, model, metadata=None):
        """
        保存模型
        """
        try:
            # 创建模型目录
            model_dir = f"models/{model_name}"
            os.makedirs(model_dir, exist_ok=True)
            
            # 保存模型
            model_path = os.path.join(model_dir, "model.pkl")
            joblib.dump(model, model_path)
            
            # 保存元数据
            if metadata:
                metadata_path = os.path.join(model_dir, "metadata.json")
                with open(metadata_path, 'w') as f:
                    json.dump(metadata, f, indent=2)
            
            print(f"模型 {model_name} 已保存")
            
        except Exception as e:
            print(f"模型保存失败: {e}")
            raise
    
    def load_model(self, model_name):
        """
        加载模型
        """
        try:
            model_path = f"models/{model_name}/model.pkl"
            model = joblib.load(model_path)
            
            # 加载元数据
            metadata_path = f"models/{model_name}/metadata.json"
            metadata = None
            if os.path.exists(metadata_path):
                with open(metadata_path, 'r') as f:
                    metadata = json.load(f)
            
            return model, metadata
            
        except Exception as e:
            print(f"模型加载失败: {e}")
            raise
```

```python
import docker
import subprocess
import yaml
from kubernetes import client, config

class MLDeploymentPipeline:
    def __init__(self):
        self.deployment_configs = {}
        self.deployed_models = {}
        
    def create_deployment_config(self, model_name, model_version, config):
        """
        创建部署配置
        """
        deployment_config = {
            'model_name': model_name,
            'model_version': model_version,
            'deployment_type': config.get('type', 'container'),
            'replicas': config.get('replicas', 3),
            'resources': config.get('resources', {
                'requests': {'memory': '512Mi', 'cpu': '250m'},
                'limits': {'memory': '1Gi', 'cpu': '500m'}
            }),
            'environment': config.get('environment', {}),
            'health_check': config.get('health_check', {
                'path': '/health',
                'interval': 30,
                'timeout': 10
            })
        }
        
        self.deployment_configs[f"{model_name}_{model_version}"] = deployment_config
        
        return deployment_config
    
    def deploy_model(self, model_name, model_version, deployment_config):
        """
        部署模型
        """
        try:
            deployment_type = deployment_config['deployment_type']
            
            if deployment_type == 'container':
                return self._deploy_container(model_name, model_version, deployment_config)
            elif deployment_type == 'kubernetes':
                return self._deploy_kubernetes(model_name, model_version, deployment_config)
            elif deployment_type == 'serverless':
                return self._deploy_serverless(model_name, model_version, deployment_config)
            else:
                raise ValueError(f"不支持的部署类型: {deployment_type}")
                
        except Exception as e:
            print(f"模型部署失败: {e}")
            raise
    
    def _deploy_container(self, model_name, model_version, config):
        """
        容器部署
        """
        try:
            # 构建Docker镜像
            image_name = f"{model_name}:{model_version}"
            dockerfile_path = f"models/{model_name}/Dockerfile"
            
            # 构建镜像
            client = docker.from_env()
            image, build_log = client.images.build(
                path=f"models/{model_name}",
                tag=image_name,
                dockerfile=dockerfile_path
            )
            
            # 运行容器
            container_name = f"{model_name}-{model_version}"
            container = client.containers.run(
                image_name,
                name=container_name,
                ports={5000: config.get('port', 8080)},
                environment=config['environment'],
                detach=True,
                restart_policy={'Name': 'unless-stopped'}
            )
            
            self.deployed_models[f"{model_name}_{model_version}"] = {
                'type': 'container',
                'container_id': container.id,
                'status': 'running'
            }
            
            print(f"模型 {model_name} v{model_version} 容器部署成功")
            return container
            
        except Exception as e:
            print(f"容器部署失败: {e}")
            raise
    
    def _deploy_kubernetes(self, model_name, model_version, config):
        """
        Kubernetes部署
        """
        try:
            # 加载Kubernetes配置
            try:
                config.load_incluster_config()
            except:
                config.load_kube_config()
            
            # 创建Deployment
            deployment = self._create_k8s_deployment(model_name, model_version, config)
            apps_v1 = client.AppsV1Api()
            apps_v1.create_namespaced_deployment(
                namespace='default',
                body=deployment
            )
            
            # 创建Service
            service = self._create_k8s_service(model_name, config)
            core_v1 = client.CoreV1Api()
            core_v1.create_namespaced_service(
                namespace='default',
                body=service
            )
            
            self.deployed_models[f"{model_name}_{model_version}"] = {
                'type': 'kubernetes',
                'deployment_name': f"{model_name}-service",
                'service_name': f"{model_name}-service",
                'status': 'deployed'
            }
            
            print(f"模型 {model_name} v{model_version} Kubernetes部署成功")
            
        except Exception as e:
            print(f"Kubernetes部署失败: {e}")
            raise
    
    def _create_k8s_deployment(self, model_name, model_version, config):
        """
        创建Kubernetes Deployment
        """
        deployment = client.V1Deployment(
            metadata=client.V1ObjectMeta(
                name=f"{model_name}-service",
                labels={"app": f"{model_name}-service"}
            ),
            spec=client.V1DeploymentSpec(
                replicas=config['replicas'],
                selector=client.V1LabelSelector(
                    match_labels={"app": f"{model_name}-service"}
                ),
                template=client.V1PodTemplateSpec(
                    metadata=client.V1ObjectMeta(
                        labels={"app": f"{model_name}-service"}
                    ),
                    spec=client.V1PodSpec(
                        containers=[
                            client.V1Container(
                                name=f"{model_name}-service",
                                image=f"{model_name}:{model_version}",
                                ports=[client.V1ContainerPort(container_port=5000)],
                                env=[
                                    client.V1EnvVar(name=k, value=v) 
                                    for k, v in config['environment'].items()
                                ],
```

```python
import time
import threading
from collections import deque
import statistics

class ModelMonitor:
    def __init__(self):
        self.metrics = {}
        self.alerts = []
        self.monitoring_threads = {}
        
    def start_monitoring(self, model_name, model_version, monitoring_config):
        """
        开始监控
        """
        monitor_key = f"{model_name}_{model_version}"
        
        # 初始化指标存储
        self.metrics[monitor_key] = {
            'predictions': deque(maxlen=1000),
            'response_times': deque(maxlen=1000),
            'errors': deque(maxlen=100),
            'throughput': deque(maxlen=100),
            'data_drift': deque(maxlen=100),
            'model_drift': deque(maxlen=100)
        }
        
        # 启动监控线程
        monitor_thread = threading.Thread(
            target=self._monitor_model,
            args=(monitor_key, monitoring_config)
        )
        monitor_thread.daemon = True
        monitor_thread.start()
        
        self.monitoring_threads[monitor_key] = monitor_thread
        
        print(f"模型 {model_name} v{model_version} 监控已启动")
    
    def _monitor_model(self, monitor_key, config):
        """
        监控模型
        """
        while True:
            try:
                # 收集指标
                self._collect_metrics(monitor_key, config)
                
                # 检查告警
                self._check_alerts(monitor_key, config)
                
                # 等待下次检查
                time.sleep(config.get('check_interval', 60))
                
            except Exception as e:
                print(f"监控错误: {e}")
                time.sleep(60)
    
    def _collect_metrics(self, monitor_key, config):
        """
        收集指标
        """
        try:
            # 获取模型服务状态
            service_url = config.get('service_url')
            if service_url:
                response = requests.get(f"{service_url}/health", timeout=5)
                if response.status_code == 200:
                    health_data = response.json()
                    
                    # 记录响应时间
                    response_time = response.elapsed.total_seconds()
                    self.metrics[monitor_key]['response_times'].append(response_time)
                    
                    # 记录吞吐量
                    throughput = health_data.get('throughput', 0)
                    self.metrics[monitor_key]['throughput'].append(throughput)
            
            # 数据漂移检测
            if config.get('data_drift_detection', False):
                drift_score = self._detect_data_drift(monitor_key, config)
                self.metrics[monitor_key]['data_drift'].append(drift_score)
            
            # 模型漂移检测
            if config.get('model_drift_detection', False):
                model_drift_score = self._detect_model_drift(monitor_key, config)
                self.metrics[monitor_key]['model_drift'].append(model_drift_score)
                
        except Exception as e:
            print(f"指标收集失败: {e}")
    
    def _detect_data_drift(self, monitor_key, config):
        """
        检测数据漂移
        """
        try:
            # 获取当前数据样本
            current_data = self._get_current_data_sample(config)
            
            # 获取参考数据
            reference_data = self._get_reference_data(config)
            
            # 计算分布差异
            drift_score = self._calculate_distribution_difference(
                current_data, reference_data
            )
            
            return drift_score
            
        except Exception as e:
            print(f"数据漂移检测失败: {e}")
            return 0.0
    
    def _detect_model_drift(self, monitor_key, config):
        """
        检测模型漂移
        """
        try:
            # 获取模型预测结果
            predictions = list(self.metrics[monitor_key]['predictions'])
            
            if len(predictions) < 100:
                return 0.0
            
            # 计算预测分布变化
            recent_predictions = predictions[-50:]
            older_predictions = predictions[-100:-50]
            
            drift_score = self._calculate_prediction_drift(
                recent_predictions, older_predictions
            )
            
            return drift_score
            
        except Exception as e:
            print(f"模型漂移检测失败: {e}")
            return 0.0
    
    def _calculate_distribution_difference(self, data1, data2):
        """
        计算分布差异
        """
        try:
            from scipy import stats
            
            # 计算KS检验
            statistic, p_value = stats.ks_2samp(data1, data2)
            
            return statistic
            
        except:
            return 0.0
    
    def _calculate_prediction_drift(self, predictions1, predictions2):
        """
        计算预测漂移
        """
        try:
            # 计算预测分布差异
            mean1 = statistics.mean(predictions1)
            mean2 = statistics.mean(predictions2)
            
            std1 = statistics.stdev(predictions1) if len(predictions1) > 1 else 0
            std2 = statistics.stdev(predictions2) if len(predictions2) > 1 else 0
            
            # 计算漂移分数
            drift_score = abs(mean1 - mean2) / (std1 + std2 + 1e-8)
            
            return drift_score
            
        except:
            return 0.0
    
    def _check_alerts(self, monitor_key, config):
        """
        检查告警
        """
        try:
            metrics = self.metrics[monitor_key]
            
            # 检查响应时间告警
            if len(metrics['response_times']) > 0:
                avg_response_time = statistics.mean(metrics['response_times'])
                if avg_response_time > config.get('response_time_threshold', 5.0):
                    self._create_alert(
                        monitor_key,
                        'high_response_time',
                        f"平均响应时间过高: {avg_response_time:.2f}s",
                        'warning'
                    )
            
            # 检查错误率告警
            if len(metrics['errors']) > 0:
                error_rate = len(metrics['errors']) / max(len(metrics['predictions']), 1)
                if error_rate > config.get('error_rate_threshold', 0.05):
                    self._create_alert(
                        monitor_key,
                        'high_error_rate',
                        f"错误率过高: {error_rate:.2%}",
                        'critical'
                    )
            
            # 检查数据漂移告警
            if len(metrics['data_drift']) > 0:
                latest_drift = metrics['data_drift'][-1]
                if latest_drift > config.get('data_drift_threshold', 0.1):
                    self._create_alert(
                        monitor_key,
                        'data_drift',
                        f"数据漂移检测: {latest_drift:.4f}",
                        'warning'
                    )
            
            # 检查模型漂移告警
            if len(metrics['model_drift']) > 0:
                latest_model_drift = metrics['model_drift'][-1]
                if latest_model_drift > config.get('model_drift_threshold', 0.1):
                    self._create_alert(
                        monitor_key,
                        'model_drift',
                        f"模型漂移检测: {latest_model_drift:.4f}",
                        'critical'
                    )
                    
        except Exception as e:
            print(f"告警检查失败: {e}")
    
    def _create_alert(self, monitor_key, alert_type, message, severity):
        """
        创建告警
        """
        alert = {
            'monitor_key': monitor_key,
            'type': alert_type,
            'message': message,
            'severity': severity,
            'timestamp': datetime.now().isoformat()
        }
        
        self.alerts.append(alert)
        
        # 发送告警通知
        self._send_alert_notification(alert)
        
        print(f"告警: {message}")
    
    def _send_alert_notification(self, alert):

        发送告警通知
        """

    def get_monitoring_metrics(self, monitor_key):
        """
        获取监控指标
        """
        if monitor_key not in self.metrics:
            return None
        
        metrics = self.metrics[monitor_key]
        
        return {
            'predictions_count': len(metrics['predictions']),
            'avg_response_time': statistics.mean(metrics['response_times']) if metrics['response_times'] else 0,
            'error_count': len(metrics['errors']),
            'avg_throughput': statistics.mean(metrics['throughput']) if metrics['throughput'] else 0,

        if monitor_key:
```

```python
import random
import json
from datetime import datetime, timedelta

class ABTestFramework:
    def __init__(self):
        self.experiments = {}
        self.traffic_splits = {}
        self.results = {}
        
    def create_experiment(self, experiment_name, model_a, model_b, 
                         traffic_split=0.5, duration_days=7):
        """
        创建A/B测试实验
        """
        experiment = {
            'name': experiment_name,
            'model_a': model_a,
            'model_b': model_b,
            'traffic_split': traffic_split,
            'start_date': datetime.now().isoformat(),
            'end_date': (datetime.now() + timedelta(days=duration_days)).isoformat(),
            'status': 'active',
            'metrics': {
                'model_a': {'requests': 0, 'successes': 0, 'errors': 0},
                'model_b': {'requests': 0, 'successes': 0, 'errors': 0}
            }
        }
        
        self.experiments[experiment_name] = experiment
        
        print(f"A/B测试实验 {experiment_name} 已创建")
        return experiment
    
    def route_request(self, experiment_name, request_data):
        """
        路由请求到模型
        """
        if experiment_name not in self.experiments:
            raise ValueError(f"实验 {experiment_name} 不存在")
        
        experiment = self.experiments[experiment_name]
        
        if experiment['status'] != 'active':
            raise ValueError(f"实验 {experiment_name} 未激活")
        
        # 检查实验是否过期
        if datetime.now().isoformat() > experiment['end_date']:
            experiment['status'] = 'expired'
            raise ValueError(f"实验 {experiment_name} 已过期")
        
        # 随机分配流量
        if random.random() < experiment['traffic_split']:
            model_name = experiment['model_a']
            variant = 'A'
        else:
            model_name = experiment['model_b']
            variant = 'B'
        
        # 记录请求
        experiment['metrics'][model_name]['requests'] += 1
        
        return model_name, variant
    
    def record_result(self, experiment_name, variant, success, response_time=None, error=None):
        """
        记录实验结果
        """
        if experiment_name not in self.experiments:
            return
        
        experiment = self.experiments[experiment_name]
        
        if variant == 'A':
            model_name = experiment['model_a']
        else:
            model_name = experiment['model_b']
        
        # 更新指标
        if success:
            experiment['metrics'][model_name]['successes'] += 1
        else:
            experiment['metrics'][model_name]['errors'] += 1
        
        # 记录详细结果
        if experiment_name not in self.results:
            self.results[experiment_name] = []
        
        result = {
            'variant': variant,
            'model': model_name,
            'success': success,
            'response_time': response_time,
            'error': error,
            'timestamp': datetime.now().isoformat()
        }
        
        self.results[experiment_name].append(result)
    
    def get_experiment_results(self, experiment_name):
        """
        获取实验结果
        """
        if experiment_name not in self.experiments:
            return None
        
        experiment = self.experiments[experiment_name]
        
        # 计算统计指标
        model_a_metrics = experiment['metrics'][experiment['model_a']]
        model_b_metrics = experiment['metrics'][experiment['model_b']]
        
        # 成功率
        model_a_success_rate = (model_a_metrics['successes'] / 
                              max(model_a_metrics['requests'], 1))
        model_b_success_rate = (model_b_metrics['successes'] / 
                              max(model_b_metrics['requests'], 1))
        
        # 错误率
        model_a_error_rate = (model_a_metrics['errors'] / 
                            max(model_a_metrics['requests'], 1))
        model_b_error_rate = (model_b_metrics['errors'] / 
                            max(model_b_metrics['requests'], 1))
        
        # 统计显著性检验
        significance = self._calculate_significance(experiment_name)
        
        results = {
            'experiment_name': experiment_name,
            'status': experiment['status'],
            'duration': self._calculate_duration(experiment),
            'model_a': {
                'name': experiment['model_a'],
                'requests': model_a_metrics['requests'],
                'success_rate': model_a_success_rate,
                'error_rate': model_a_error_rate
            },
            'model_b': {
                'name': experiment['model_b'],
                'requests': model_b_metrics['requests'],
                'success_rate': model_b_success_rate,
                'error_rate': model_b_error_rate
            },
            'significance': significance,
            'recommendation': self._get_recommendation(experiment_name)
        }
        
        return results
    
    def _calculate_significance(self, experiment_name):
        """
        计算统计显著性
        """
        if experiment_name not in self.results:
            return None
        
        results = self.results[experiment_name]
        
        # 分离A/B组结果
        group_a = [r for r in results if r['variant'] == 'A']
        group_b = [r for r in results if r['variant'] == 'B']
        
        if len(group_a) < 30 or len(group_b) < 30:
            return {'significant': False, 'reason': '样本量不足'}
        
        # 计算成功率
        success_a = sum(1 for r in group_a if r['success'])
        success_b = sum(1 for r in group_b if r['success'])
        
        rate_a = success_a / len(group_a)
        rate_b = success_b / len(group_b)
        
        # 简化的显著性检验
        # 实际应用中应该使用更严格的统计检验
        difference = abs(rate_a - rate_b)
        
        if difference > 0.05:  # 5%的差异阈值
            return {
                'significant': True,
                'difference': difference,
                'confidence': 'high' if difference > 0.1 else 'medium'
            }
        else:
            return {
                'significant': False,
                'difference': difference,
                'reason': '差异不显著'
            }
    
    def _calculate_duration(self, experiment):
        """
        计算实验持续时间
        """
        start_date = datetime.fromisoformat(experiment['start_date'])
        end_date = datetime.fromisoformat(experiment['end_date'])
        
        return (end_date - start_date).days
    
    def _get_recommendation(self, experiment_name):
        """
        获取推荐
        """
        results = self.get_experiment_results(experiment_name)
        
        if not results:
            return '无法生成推荐'
        
        model_a_success = results['model_a']['success_rate']
        model_b_success = results['model_b']['success_rate']
        
        if results['significance']['significant']:
            if model_a_success > model_b_success:
                return f"推荐使用模型 {results['model_a']['name']}"
            else:
                return f"推荐使用模型 {results['model_b']['name']}"
        else:
            return '两个模型性能相当，建议继续收集数据'
    
    def stop_experiment(self, experiment_name):
        """
        停止实验
        """
        if experiment_name in self.experiments:
            self.experiments[experiment_name]['status'] = 'stopped'
            print(f"实验 {experiment_name} 已停止")
    
    def list_experiments(self):
        """
        列出所有实验
        """
        return list(self.experiments.keys())
    
    def get_experiment_status(self, experiment_name):
        """
        获取实验状态
        """
        if experiment_name not in self.experiments:
            return None
        
        return self.experiments[experiment_name]['status']
```
