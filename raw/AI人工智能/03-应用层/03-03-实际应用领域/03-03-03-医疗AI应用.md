# 医疗AI应用

## 概述

医疗AI是人工智能在医疗健康领域的重要应用，通过机器学习、深度学习、自然语言处理等技术，辅助医生进行诊断、治疗、药物发现等医疗活动。它正在革命性地改变医疗行业，提高诊断准确率、降低医疗成本、改善患者体验。

## 核心概念

### 1. 医疗AI定义
- **目标**：辅助医疗决策，提高医疗质量和效率
- **技术栈**：计算机视觉、自然语言处理、机器学习、深度学习
- **应用领域**：医学影像、临床决策、药物发现、健康管理
- **伦理要求**：安全性、可解释性、隐私保护、公平性

### 2. 医疗AI分类
- **诊断辅助**：影像诊断、病理分析、症状识别
- **治疗辅助**：个性化治疗、手术规划、药物推荐
- **预防保健**：健康监测、风险评估、早期预警
- **药物研发**：分子设计、临床试验、副作用预测

```python
import cv2
import numpy as np
from scipy import ndimage
import matplotlib.pyplot as plt

class MedicalImagePreprocessor:
    def __init__(self):
        self.normalization_methods = ['z_score', 'min_max', 'histogram']
        
    def preprocess_image(self, image, method='z_score'):
        """
        医学影像预处理
        """
        # 1. 去噪
        denoised = self._denoise_image(image)
        
        # 2. 增强对比度
        enhanced = self._enhance_contrast(denoised)
        
        # 3. 标准化
        normalized = self._normalize_image(enhanced, method)
        
        # 4. 尺寸调整
        resized = self._resize_image(normalized)
        
        return resized
    
    def _denoise_image(self, image):
        """
        图像去噪
        """
        # 使用高斯滤波去噪
        denoised = cv2.GaussianBlur(image, (5, 5), 0)
        
        # 使用中值滤波去除椒盐噪声
        denoised = cv2.medianBlur(denoised, 5)
        
        return denoised
    
    def _enhance_contrast(self, image):
        """
        对比度增强
        """
        # 直方图均衡化
        if len(image.shape) == 3:
            # 彩色图像
            enhanced = cv2.cvtColor(image, cv2.COLOR_BGR2YUV)
            enhanced[:,:,0] = cv2.equalizeHist(enhanced[:,:,0])
            enhanced = cv2.cvtColor(enhanced, cv2.COLOR_YUV2BGR)
        else:
            # 灰度图像
            enhanced = cv2.equalizeHist(image)
        
        return enhanced
    
    def _normalize_image(self, image, method='z_score'):
        """
        图像标准化
        """
        if method == 'z_score':
            # Z-score标准化
            mean = np.mean(image)
            std = np.std(image)
            normalized = (image - mean) / std
        elif method == 'min_max':
            # Min-Max标准化
            min_val = np.min(image)
            max_val = np.max(image)
            normalized = (image - min_val) / (max_val - min_val)
        elif method == 'histogram':
            # 直方图标准化
            normalized = cv2.normalize(image, None, 0, 255, cv2.NORM_MINMAX)
        
        return normalized
    
    def _resize_image(self, image, target_size=(224, 224)):
        """
        图像尺寸调整
        """
        resized = cv2.resize(image, target_size)
        return resized

class MedicalImageAnalyzer:
    def __init__(self):
        self.preprocessor = MedicalImagePreprocessor()
        
    def analyze_xray(self, xray_image):
        """
        X光片分析
        """
        # 预处理
        processed_image = self.preprocessor.preprocess_image(xray_image)
        
        # 肺部区域分割
        lung_mask = self._segment_lung(processed_image)
        
        # 异常检测
        abnormalities = self._detect_abnormalities(processed_image, lung_mask)
        
        # 生成报告
        report = self._generate_xray_report(abnormalities)
        
        return {
            'processed_image': processed_image,
            'lung_mask': lung_mask,
            'abnormalities': abnormalities,
            'report': report
        }
    
    def analyze_ct_scan(self, ct_image):
        """
        CT扫描分析
        """
        # 预处理
        processed_image = self.preprocessor.preprocess_image(ct_image)
        
        # 器官分割
        organ_masks = self._segment_organs(processed_image)
        
        # 病变检测
        lesions = self._detect_lesions(processed_image, organ_masks)
        
        # 3D重建
        volume_data = self._reconstruct_3d(processed_image)
        
        return {
            'processed_image': processed_image,
            'organ_masks': organ_masks,
            'lesions': lesions,
            'volume_data': volume_data
        }
    
    def analyze_mri(self, mri_image):
        """
        MRI分析
        """
        # 预处理
        processed_image = self.preprocessor.preprocess_image(mri_image)
        
        # 脑部结构分割
        brain_structures = self._segment_brain_structures(processed_image)
        
        # 异常检测
        anomalies = self._detect_brain_anomalies(processed_image, brain_structures)
        
        return {
            'processed_image': processed_image,
            'brain_structures': brain_structures,
            'anomalies': anomalies
        }
    
    def _segment_lung(self, image):
        """
        肺部区域分割
        """
        # 阈值分割
        _, binary = cv2.threshold(image, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
        
        # 形态学操作
        kernel = np.ones((5,5), np.uint8)
        binary = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, kernel)
        binary = cv2.morphologyEx(binary, cv2.MORPH_OPEN, kernel)
        
        # 连通域分析
        num_labels, labels = cv2.connectedComponents(binary)
        
        # 选择最大的连通域作为肺部
        largest_component = 0
        max_area = 0
        
        for i in range(1, num_labels):
            area = np.sum(labels == i)
            if area > max_area:
                max_area = area
                largest_component = i
        
        lung_mask = (labels == largest_component).astype(np.uint8) * 255
        
        return lung_mask
    
    def _detect_abnormalities(self, image, lung_mask):
        """
        异常检测
        """
        # 在肺部区域内检测异常
        lung_region = cv2.bitwise_and(image, image, mask=lung_mask)
        
        # 使用边缘检测
        edges = cv2.Canny(lung_region, 50, 150)
        
        # 查找轮廓
        contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        
        # 过滤异常区域
        abnormalities = []
        for contour in contours:
            area = cv2.contourArea(contour)
            if area > 100:  # 面积阈值
                # 计算轮廓特征
                perimeter = cv2.arcLength(contour, True)
                circularity = 4 * np.pi * area / (perimeter * perimeter)
                
                # 判断是否为异常
                if circularity < 0.7:  # 圆形度阈值
                    abnormalities.append({
                        'contour': contour,
                        'area': area,
                        'circularity': circularity
                    })
        
        return abnormalities
    
    def _generate_xray_report(self, abnormalities):
        """
        生成X光片报告
        """
        report = {
            'total_abnormalities': len(abnormalities),
            'abnormality_details': []
        }
        
        for i, abnormality in enumerate(abnormalities):
            report['abnormality_details'].append({
                'id': i + 1,
                'area': abnormality['area'],
                'circularity': abnormality['circularity'],
                'severity': 'high' if abnormality['area'] > 500 else 'medium'
            })
        
        return report
```

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
from torchvision import models

class MedicalImageCNN(nn.Module):
    def __init__(self, num_classes=2, pretrained=True):
        super(MedicalImageCNN, self).__init__()
        
        # 使用预训练的ResNet作为骨干网络
        self.backbone = models.resnet50(pretrained=pretrained)
        
        # 修改第一层以适应医学影像
        self.backbone.conv1 = nn.Conv2d(1, 64, kernel_size=7, stride=2, padding=3, bias=False)
        
        # 修改分类层
        self.backbone.fc = nn.Sequential(
            nn.Dropout(0.5),
            nn.Linear(2048, 512),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(512, num_classes)
        )
        
    def forward(self, x):
        return self.backbone(x)

class MedicalImageClassifier:
    def __init__(self, num_classes=2, learning_rate=0.001):
        self.model = MedicalImageCNN(num_classes)
        self.optimizer = torch.optim.Adam(self.model.parameters(), lr=learning_rate)
        self.criterion = nn.CrossEntropyLoss()
        
    def train(self, train_loader, val_loader, epochs=50):
        """
        训练医学影像分类模型
        """
        self.model.train()
        
        for epoch in range(epochs):
            train_loss = 0
            train_acc = 0
            
            for batch_images, batch_labels in train_loader:
                self.optimizer.zero_grad()
                
                # 前向传播
                outputs = self.model(batch_images)
                loss = self.criterion(outputs, batch_labels)
                
                # 反向传播
                loss.backward()
                self.optimizer.step()
                
                train_loss += loss.item()
                _, predicted = torch.max(outputs.data, 1)
                train_acc += (predicted == batch_labels).sum().item()
            
            # 验证
            val_loss, val_acc = self.evaluate(val_loader)
            
            print(f'Epoch {epoch+1}/{epochs}:')
            print(f'Train Loss: {train_loss/len(train_loader):.4f}, Train Acc: {train_acc/len(train_loader.dataset):.4f}')
            print(f'Val Loss: {val_loss:.4f}, Val Acc: {val_acc:.4f}')
            print('-' * 50)
    
    def evaluate(self, data_loader):
        """
        评估模型
        """
        self.model.eval()
        total_loss = 0
        correct = 0
        
        with torch.no_grad():
            for batch_images, batch_labels in data_loader:
                outputs = self.model(batch_images)
                loss = self.criterion(outputs, batch_labels)
                
                total_loss += loss.item()
                _, predicted = torch.max(outputs.data, 1)
                correct += (predicted == batch_labels).sum().item()
        
        avg_loss = total_loss / len(data_loader)
        accuracy = correct / len(data_loader.dataset)
        
        return avg_loss, accuracy
    
    def predict(self, image):
        """
        预测单张图像
        """
        self.model.eval()
        
        with torch.no_grad():
            # 预处理图像
            processed_image = self._preprocess_image(image)
            
            # 预测
            output = self.model(processed_image)
            probabilities = F.softmax(output, dim=1)
            predicted_class = torch.argmax(probabilities, dim=1)
            
            return {
                'predicted_class': predicted_class.item(),
                'probabilities': probabilities.numpy()[0]
            }
    
    def _preprocess_image(self, image):
        """
        预处理图像
        """
        # 转换为张量
        if len(image.shape) == 3:
            image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        
        # 调整尺寸
        image = cv2.resize(image, (224, 224))
        
        # 标准化
        image = image.astype(np.float32) / 255.0
        
        # 转换为张量
        tensor = torch.FloatTensor(image).unsqueeze(0).unsqueeze(0)
        
        return tensor
```

```python
import pandas as pd

        """
        训练症状分析模型
        """

        # 划分训练集和测试集

        self.model.fit(X_train, y_train)
        
        # 评估模型

        print("症状分析模型评估结果:")
        print(classification_report(y_test, y_pred))
        print("\n混淆矩阵:")
        print(confusion_matrix(y_test, y_pred))
        
        return self.model
    
    def predict_disease(self, symptoms):
        """
        根据症状预测疾病
        """
        # 预处理症状数据
        processed_symptoms = self._preprocess_symptoms(symptoms)
        
        # 预测疾病
        predicted_disease = self.model.predict([processed_symptoms])[0]
        
        # 获取预测概率
        probabilities = self.model.predict_proba([processed_symptoms])[0]
        
        # 获取疾病名称
        disease_name = self.disease_encoder.inverse_transform([predicted_disease])[0]
        
        return {
            'predicted_disease': disease_name,
            'confidence': max(probabilities),
            'all_probabilities': dict(zip(
                self.disease_encoder.classes_, 
                probabilities
            ))
        }
    
    def _preprocess_data(self, symptoms_data, diseases_data):
        """
        预处理训练数据
        """
        # 创建症状编码器
        from sklearn.preprocessing import LabelEncoder
        
        self.symptom_encoder = LabelEncoder()
        self.disease_encoder = LabelEncoder()
        
        # 编码症状
        encoded_symptoms = []
        for symptoms in symptoms_data:
            encoded_symptom = self.symptom_encoder.fit_transform(symptoms)
            encoded_symptoms.append(encoded_symptom)
        
        # 编码疾病
        encoded_diseases = self.disease_encoder.fit_transform(diseases_data)
        
        # 创建特征矩阵
        X = self._create_feature_matrix(encoded_symptoms)
        y = encoded_diseases
        
        return X, y
    
    def _create_feature_matrix(self, encoded_symptoms):
        """
        创建特征矩阵
        """
        # 获取所有症状
        all_symptoms = set()
        for symptoms in encoded_symptoms:
            all_symptoms.update(symptoms)
        
        # 创建特征矩阵
        feature_matrix = []
        for symptoms in encoded_symptoms:
            features = [1 if symptom in symptoms else 0 
                      for symptom in all_symptoms]
            feature_matrix.append(features)
        
        return feature_matrix
    
    def _preprocess_symptoms(self, symptoms):
        """
        预处理症状数据
        """
        # 编码症状
        encoded_symptoms = self.symptom_encoder.transform(symptoms)

        all_symptoms = set(self.symptom_encoder.classes_)
        features = [1 if symptom in symptoms else 0 
                  for symptom in all_symptoms]
        
        return features
```

```python
class DrugRecommender:
    def __init__(self):
        self.drug_database = {}
        self.interaction_checker = DrugInteractionChecker()
        self.allergy_checker = AllergyChecker()
        
    def recommend_drugs(self, disease, medical_history):
        """
        推荐药物
        """
        # 获取疾病相关药物
        candidate_drugs = self._get_candidate_drugs(disease)
        
        # 检查药物相互作用
        safe_drugs = self._check_drug_interactions(
            candidate_drugs, medical_history['current_medications'])
        
        # 检查过敏反应
        safe_drugs = self._check_allergies(
            safe_drugs, medical_history['allergies'])
        
        # 考虑患者特征
        personalized_drugs = self._personalize_drugs(
            safe_drugs, medical_history['patient_profile'])
        
        # 排序推荐
        ranked_drugs = self._rank_drugs(personalized_drugs, disease)
        
        return ranked_drugs
    
    def _get_candidate_drugs(self, disease):
        """
        获取候选药物
        """
        # 模拟药物数据库查询
        disease_drugs = {
            'hypertension': ['amlodipine', 'lisinopril', 'metoprolol'],
            'diabetes': ['metformin', 'insulin', 'glipizide'],
            'asthma': ['albuterol', 'budesonide', 'montelukast'],
            'depression': ['sertraline', 'fluoxetine', 'citalopram']
        }
        
        return disease_drugs.get(disease, [])
    
    def _check_drug_interactions(self, drugs, current_medications):
        """
        检查药物相互作用
        """
        safe_drugs = []
        
        for drug in drugs:
            is_safe = True
            
            for current_drug in current_medications:
                if self.interaction_checker.check_interaction(drug, current_drug):
                    is_safe = False
                    break
            
            if is_safe:
                safe_drugs.append(drug)
        
        return safe_drugs
    
    def _check_allergies(self, drugs, allergies):
        """
        检查过敏反应
        """
        safe_drugs = []
        
        for drug in drugs:
            if not self.allergy_checker.check_allergy(drug, allergies):
                safe_drugs.append(drug)
        
        return safe_drugs
    
    def _personalize_drugs(self, drugs, patient_profile):
        """
        个性化药物推荐
        """
        personalized_drugs = []
        
        for drug in drugs:
            # 考虑年龄
            if patient_profile['age'] > 65:
                # 老年人可能需要调整剂量
                drug_info = {'name': drug, 'dose_adjustment': 'reduce'}
            else:
                drug_info = {'name': drug, 'dose_adjustment': 'normal'}
            
            # 考虑肾功能
            if patient_profile['kidney_function'] < 60:
                drug_info['kidney_adjustment'] = True
            
            # 考虑肝功能
            if patient_profile['liver_function'] < 60:
                drug_info['liver_adjustment'] = True
            
            personalized_drugs.append(drug_info)
        
        return personalized_drugs
    
    def _rank_drugs(self, drugs, disease):
        """
        药物排序
        """
        # 根据疗效、副作用、成本等因素排序
        ranked_drugs = sorted(drugs, key=lambda x: self._calculate_drug_score(x, disease))
        
        return ranked_drugs
    
    def _calculate_drug_score(self, drug, disease):
        """
        计算药物评分
        """
        # 模拟评分计算
        efficacy_score = 0.8  # 疗效评分
        safety_score = 0.9    # 安全性评分
        cost_score = 0.7      # 成本评分
        
        total_score = 0.5 * efficacy_score + 0.3 * safety_score + 0.2 * cost_score
        
        return total_score

class DrugInteractionChecker:
    def __init__(self):
        self.interaction_database = {
            ('warfarin', 'aspirin'): 'increased_bleeding_risk',
            ('digoxin', 'furosemide'): 'increased_digoxin_toxicity',
            ('metformin', 'contrast_dye'): 'increased_lactic_acidosis_risk'
        }
    
    def check_interaction(self, drug1, drug2):
        """
        检查药物相互作用
        """
        # 检查直接相互作用
        if (drug1, drug2) in self.interaction_database:
            return True
        
        if (drug2, drug1) in self.interaction_database:
            return True
        
        return False

class AllergyChecker:
    def __init__(self):
        self.allergy_database = {
            'penicillin': ['amoxicillin', 'ampicillin', 'piperacillin'],
            'sulfa': ['sulfamethoxazole', 'trimethoprim'],
            'aspirin': ['ibuprofen', 'naproxen']
        }
    
    def check_allergy(self, drug, allergies):
        """
        检查过敏反应
        """
        for allergy in allergies:
            if allergy in self.allergy_database:
                if drug in self.allergy_database[allergy]:
                    return True
        
        return False
```

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
from rdkit import Chem
from rdkit.Chem import Descriptors

class MolecularGenerator(nn.Module):
    def __init__(self, vocab_size=100, hidden_size=256, num_layers=3):
        super(MolecularGenerator, self).__init__()
        
        self.vocab_size = vocab_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        
        # 嵌入层
        self.embedding = nn.Embedding(vocab_size, hidden_size)
        
        # LSTM层
        self.lstm = nn.LSTM(hidden_size, hidden_size, num_layers, 
                           batch_first=True, dropout=0.2)
        
        # 输出层
        self.output_layer = nn.Linear(hidden_size, vocab_size)
        
    def forward(self, input_sequence, hidden=None):
        # 嵌入
        embedded = self.embedding(input_sequence)
        
        # LSTM
        output, hidden = self.lstm(embedded, hidden)
        
        # 输出
        output = self.output_layer(output)
        
        return output, hidden
    
    def generate_molecule(self, start_token, max_length=100):
        """
        生成分子
        """
        self.eval()
        
        with torch.no_grad():
            # 初始化
            current_token = start_token
            generated_sequence = [current_token]
            hidden = None
            
            for _ in range(max_length):
                # 前向传播
                input_tensor = torch.LongTensor([[current_token]])
                output, hidden = self.forward(input_tensor, hidden)
                
                # 选择下一个token
                probabilities = F.softmax(output[0, -1, :], dim=0)
                current_token = torch.multinomial(probabilities, 1).item()
                
                # 检查结束条件
                if current_token == 0:  # 假设0是结束token
                    break
                
                generated_sequence.append(current_token)
            
            return generated_sequence

class DrugDiscoveryAI:
    def __init__(self):
        self.molecular_generator = MolecularGenerator()
        self.property_predictor = PropertyPredictor()
        self.synthesis_planner = SynthesisPlanner()
        
    def discover_drug(self, target_properties):
        """
        药物发现
        """
        # 生成候选分子
        candidate_molecules = self._generate_candidate_molecules(target_properties)
        
        # 预测分子性质
        evaluated_molecules = self._evaluate_molecules(candidate_molecules, target_properties)
        
        # 选择最优分子
        best_molecules = self._select_best_molecules(evaluated_molecules)
        
        # 规划合成路线
        synthesis_plans = self._plan_synthesis(best_molecules)
        
        return {
            'candidate_molecules': candidate_molecules,
            'evaluated_molecules': evaluated_molecules,
            'best_molecules': best_molecules,
            'synthesis_plans': synthesis_plans
        }
    
    def _generate_candidate_molecules(self, target_properties):
        """
        生成候选分子
        """
        candidate_molecules = []
        
        # 使用分子生成模型生成候选分子
        for _ in range(100):  # 生成100个候选分子
            molecule_smiles = self.molecular_generator.generate_molecule_smiles()
            
            if molecule_smiles:
                candidate_molecules.append(molecule_smiles)
        
        return candidate_molecules
    
    def _evaluate_molecules(self, molecules, target_properties):
        """
        评估分子
        """
        evaluated_molecules = []
        
        for molecule_smiles in molecules:
            try:
                # 解析分子
                molecule = Chem.MolFromSmiles(molecule_smiles)
                
                if molecule is not None:
                    # 预测性质
                    properties = self.property_predictor.predict_properties(molecule)
                    
                    # 计算与目标性质的匹配度
                    match_score = self._calculate_match_score(properties, target_properties)
                    
                    evaluated_molecules.append({
                        'smiles': molecule_smiles,
                        'properties': properties,
                        'match_score': match_score
                    })
            
            except Exception as e:
                print(f"Error evaluating molecule {molecule_smiles}: {e}")
                continue
        
        return evaluated_molecules
    
    def _calculate_match_score(self, properties, target_properties):
        """
        计算匹配分数
        """
        score = 0
        
        for prop_name, target_value in target_properties.items():
            if prop_name in properties:
                actual_value = properties[prop_name]
                
                # 计算相对误差
                if target_value != 0:
                    relative_error = abs(actual_value - target_value) / abs(target_value)
                    score += 1 - relative_error
                else:
                    score += 1 if actual_value == 0 else 0
        
        return score / len(target_properties)
    
    def _select_best_molecules(self, evaluated_molecules, top_k=10):
        """
        选择最优分子
        """
        # 按匹配分数排序
        sorted_molecules = sorted(evaluated_molecules, 
                                key=lambda x: x['match_score'], 
                                reverse=True)
        
        return sorted_molecules[:top_k]
    
    def _plan_synthesis(self, molecules):

        synthesis_plans = []

                'synthesis_plan': plan
            })
        
        return synthesis_plans
```

```python
        设计临床试验
        """
        # 确定试验类型
        trial_type = self._determine_trial_type(drug_info)
        
        # 计算样本量
        sample_size = self._calculate_sample_size(drug_info, target_population)
        
        # 设计随机化方案
        randomization_scheme = self._design_randomization(sample_size)

        endpoints = self._determine_endpoints(drug_info, trial_type)
        
        # 设计数据收集计划
        data_collection_plan = self._design_data_collection(endpoints)
        
        return {
            'trial_type': trial_type,
            'sample_size': sample_size,
            'randomization_scheme': randomization_scheme,
            'endpoints': endpoints,
            'data_collection_plan': data_collection_plan
        }
    
    def _determine_trial_type(self, drug_info):
        """
        确定试验类型
        """
        if drug_info['phase'] == 1:
            return 'safety_trial'
        elif drug_info['phase'] == 2:
            return 'efficacy_trial'
        elif drug_info['phase'] == 3:
            return 'confirmatory_trial'
        else:
            return 'post_marketing_trial'
    
    def _calculate_sample_size(self, drug_info, target_population):
        """
        计算样本量
        """
        # 基于效应量计算样本量
        effect_size = drug_info.get('expected_effect_size', 0.5)
        power = 0.8
        alpha = 0.05
        
        # 使用统计公式计算样本量
        z_alpha = stats.norm.ppf(1 - alpha/2)
        z_beta = stats.norm.ppf(power)
        
        n = 2 * ((z_alpha + z_beta) / effect_size) ** 2
        
        # 考虑脱落率
        dropout_rate = 0.1
        adjusted_n = int(n / (1 - dropout_rate))
        
        return adjusted_n
    
    def _design_randomization(self, sample_size):
        """
        设计随机化方案
        """
        # 简单随机化
        treatment_assignment = np.random.choice(
            ['treatment', 'control'], 
            size=sample_size, 
            p=[0.5, 0.5]
        )
        
        return {
            'method': 'simple_randomization',
            'treatment_assignment': treatment_assignment.tolist()
        }
    
    def _determine_endpoints(self, drug_info, trial_type):
        """
        确定终点指标
        """
        if trial_type == 'safety_trial':
            endpoints = {
                'primary': ['adverse_events', 'serious_adverse_events'],

            }
        else:
            endpoints = {
                'primary': ['efficacy_measures'],
                'secondary': ['safety_measures']
            }

class PatientRecruiter:

                eligible_patients.append(patient_id)

class EndpointAnalyzer:
    def __init__(self):
        self.endpoint_models = {}
        
    def analyze_endpoint(self, endpoint_data, endpoint_type):
        """
        分析终点指标
        """
        if endpoint_type == 'binary':
            return self._analyze_binary_endpoint(endpoint_data)
        elif endpoint_type == 'continuous':
            return self._analyze_continuous_endpoint(endpoint_data)
        elif endpoint_type == 'time_to_event':
            return self._analyze_time_to_event_endpoint(endpoint_data)
        else:
            return self._analyze_generic_endpoint(endpoint_data)
    
    def _analyze_binary_endpoint(self, data):
        """
        分析二分类终点
        """
        # 计算响应率
        response_rate = np.mean(data)
        
        # 计算置信区间
        n = len(data)
        p = response_rate
        se = np.sqrt(p * (1 - p) / n)
        
        ci_lower = p - 1.96 * se
        ci_upper = p + 1.96 * se
        
        return {
            'response_rate': response_rate,
            'confidence_interval': [ci_lower, ci_upper],
            'sample_size': n
        }
    
    def _analyze_continuous_endpoint(self, data):
        """
        分析连续终点
        """
        # 描述性统计
        mean = np.mean(data)
        std = np.std(data)
        median = np.median(data)
        
        # 置信区间
        n = len(data)
        se = std / np.sqrt(n)
        ci_lower = mean - 1.96 * se
        ci_upper = mean + 1.96 * se
        
        return {
            'mean': mean,
            'std': std,
            'median': median,
            'confidence_interval': [ci_lower, ci_upper],
            'sample_size': n
        }
    
    def _analyze_time_to_event_endpoint(self, data):
        """
        分析时间事件终点
        """
        # 使用Kaplan-Meier估计
        from lifelines import KaplanMeierFitter
        
        kmf = KaplanMeierFitter()
        kmf.fit(data['time'], data['event'])
        
        return {
            'median_survival': kmf.median_survival_time_,
            'survival_curve': kmf.survival_function_,
            'hazard_ratio': None  # 需要对照组数据
        }

class StatisticalAnalyzer:
    def __init__(self):
        self.test_methods = {}
        
    def perform_statistical_test(self, data, test_type):
        """
        执行统计检验
        """
        if test_type == 't_test':
            return self._perform_t_test(data)
        elif test_type == 'chi_square':
            return self._perform_chi_square_test(data)
        elif test_type == 'anova':
            return self._perform_anova(data)
        else:
            return self._perform_generic_test(data)
    
    def _perform_t_test(self, data):
        """
        执行t检验
        """
        group1, group2 = data['group1'], data['group2']
        
        t_stat, p_value = stats.ttest_ind(group1, group2)
        
        return {
            'test_statistic': t_stat,
            'p_value': p_value,
            'significant': p_value < 0.05
        }
    
    def _perform_chi_square_test(self, data):
        """
        执行卡方检验
        """
        observed = data['observed']
        expected = data['expected']
        
        chi2_stat, p_value = stats.chisquare(observed, expected)
        
        return {
            'test_statistic': chi2_stat,
            'p_value': p_value,
            'significant': p_value < 0.05
        }
    
    def _perform_anova(self, data):
        """
        执行ANOVA
        """
        groups = data['groups']
        
        f_stat, p_value = stats.f_oneway(*groups)
        
        return {
            'test_statistic': f_stat,
            'p_value': p_value,
            'significant': p_value < 0.05
        }
```

```python
            risk_score += 2
        
        # 生活方式
        if patient_data.get('high_salt_diet'):
            risk_score += 1
        
        return self._categorize_risk(risk_score)
    
    def _categorize_risk(self, risk_score):
        """
        风险分类
        """
        if risk_score >= 8:
            return 'high'
        elif risk_score >= 4:
            return 'medium'
        else:
            return 'low'
    
    def _calculate_overall_risk(self, individual_risks):
        """
        计算总体风险
        """
        high_risks = sum(1 for risk in individual_risks if risk == 'high')
        medium_risks = sum(1 for risk in individual_risks if risk == 'medium')
        
        if high_risks >= 2:
            return 'high'
        elif high_risks >= 1 or medium_risks >= 2:
            return 'medium'
        else:
            return 'low'
    
    def _identify_risk_factors(self, patient_data, vital_analysis):
        """
        识别风险因素
        """
        risk_factors = []
        
        # 血压风险
        if vital_analysis['blood_pressure']['status'] == 'high':
            risk_factors.append('hypertension')
        
        # 心率风险
        if vital_analysis['heart_rate']['status'] == 'abnormal':
            risk_factors.append('arrhythmia')
        
        # 体温风险
        if vital_analysis['temperature']['status'] == 'fever':
            risk_factors.append('infection')
        
        # 生活方式风险
        if patient_data.get('smoking_history'):
            risk_factors.append('smoking')
        
        if patient_data.get('bmi', 25) > 30:
            risk_factors.append('obesity')
        
        return risk_factors

class AlertSystem:
    def __init__(self):
        self.alert_thresholds = {
            'critical': {
                'blood_pressure': {'systolic': 180, 'diastolic': 110},
                'heart_rate': {'min': 40, 'max': 120},
                'temperature': {'min': 35, 'max': 40},
                'oxygen_saturation': {'min': 90}
            },
            'warning': {
                'blood_pressure': {'systolic': 160, 'diastolic': 100},
                'heart_rate': {'min': 50, 'max': 110},
                'temperature': {'min': 36, 'max': 39},
                'oxygen_saturation': {'min': 95}
            }
        }
    
    def check_alerts(self, vital_analysis, risk_assessment):
        """
        检查警报
        """
        alerts = []
        
        # 检查生命体征警报
        vital_alerts = self._check_vital_alerts(vital_analysis)
        alerts.extend(vital_alerts)
        
        # 检查风险警报
        risk_alerts = self._check_risk_alerts(risk_assessment)
        alerts.extend(risk_alerts)
        
        return alerts
    
    def _check_vital_alerts(self, vital_analysis):
        """
        检查生命体征警报
        """
        alerts = []
        
        # 血压警报
        bp = vital_analysis['blood_pressure']
        if bp['systolic'] > self.alert_thresholds['critical']['blood_pressure']['systolic']:
            alerts.append({
                'type': 'critical',
                'message': '血压过高，需要立即就医',
                'vital_sign': 'blood_pressure'
            })
        
        # 心率警报
        hr = vital_analysis['heart_rate']
        if hr['value'] > self.alert_thresholds['critical']['heart_rate']['max']:
            alerts.append({
                'type': 'critical',
                'message': '心率过快，需要立即就医',
                'vital_sign': 'heart_rate'
            })
        
        # 体温警报
        temp = vital_analysis['temperature']
        if temp['value'] > self.alert_thresholds['critical']['temperature']['max']:
            alerts.append({
                'type': 'critical',
                'message': '体温过高，需要立即就医',
                'vital_sign': 'temperature'
            })
        
        return alerts
    
    def _check_risk_alerts(self, risk_assessment):
        """
        检查风险警报
        """
        alerts = []
        
        if risk_assessment['overall_risk'] == 'high':
            alerts.append({
                'type': 'warning',
                'message': '健康风险较高，建议及时就医',
                'risk_type': 'overall'
            })
        
        return alerts
```
