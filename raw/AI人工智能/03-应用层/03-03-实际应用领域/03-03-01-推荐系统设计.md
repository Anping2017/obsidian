# 推荐系统设计

## 概述

推荐系统是信息过滤技术的一个分支，旨在预测用户对物品的评分或偏好，从而为用户推荐可能感兴趣的物品。它是现代互联网应用的核心技术之一，广泛应用于电商、社交媒体、内容平台等领域。

## 核心概念

### 1. 推荐系统定义
- **目标**：为用户推荐可能感兴趣的物品
- **输入**：用户行为数据、物品特征、用户特征
- **输出**：个性化推荐列表
- **评价**：准确率、覆盖率、多样性、新颖性

### 2. 推荐系统类型
- **协同过滤**：基于用户行为相似性
- **内容过滤**：基于物品特征相似性
- **混合推荐**：结合多种方法
- **深度学习推荐**：基于神经网络的推荐

## 协同过滤算法

### 1. 基于用户的协同过滤

#### 算法原理
```python
import numpy as np
from scipy.spatial.distance import cosine
from sklearn.metrics.pairwise import cosine_similarity

class UserBasedCF:
    def __init__(self):
        self.user_item_matrix = None
        self.user_similarity = None
    
    def fit(self, user_item_matrix):
        """
        训练基于用户的协同过滤模型
        """
        self.user_item_matrix = user_item_matrix
        
        # 计算用户相似度矩阵
        self.user_similarity = cosine_similarity(user_item_matrix)
        
        return self
    
    def predict(self, user_id, item_id, k=5):
        """
        预测用户对物品的评分
        """
        # 找到与目标用户最相似的k个用户
        user_similarities = self.user_similarity[user_id]
        similar_users = np.argsort(user_similarities)[::-1][1:k+1]
        
        # 计算加权平均评分
        weighted_sum = 0
        similarity_sum = 0
        
        for similar_user in similar_users:
            if self.user_item_matrix[similar_user, item_id] != 0:
                similarity = user_similarities[similar_user]
                rating = self.user_item_matrix[similar_user, item_id]
                
                weighted_sum += similarity * rating
                similarity_sum += abs(similarity)
        
        if similarity_sum == 0:
            return 0
        
        return weighted_sum / similarity_sum
    
    def recommend(self, user_id, n_recommendations=10):
        """
        为用户推荐物品
        """
        user_ratings = self.user_item_matrix[user_id]
        unrated_items = np.where(user_ratings == 0)[0]
        
        predictions = []
        for item_id in unrated_items:
            pred_rating = self.predict(user_id, item_id)
            predictions.append((item_id, pred_rating))
        
        # 按预测评分排序
        predictions.sort(key=lambda x: x[1], reverse=True)
        
        return predictions[:n_recommendations]
```

## 矩阵分解算法

### 1. 奇异值分解（SVD）

#### 算法实现
```python
from scipy.sparse.linalg import svds
import numpy as np

class SVDRecommender:
    def __init__(self, n_factors=50):
        self.n_factors = n_factors
        self.user_factors = None
        self.item_factors = None
        self.global_bias = None
        self.user_bias = None
        self.item_bias = None
    
    def fit(self, user_item_matrix):
        """
        训练SVD模型
        """
        # 计算全局偏置
        self.global_bias = np.mean(user_item_matrix[user_item_matrix != 0])
        
        # 计算用户和物品偏置
        self.user_bias = np.zeros(user_item_matrix.shape[0])
        self.item_bias = np.zeros(user_item_matrix.shape[1])
        
        for i in range(user_item_matrix.shape[0]):
            user_ratings = user_item_matrix[i]
            rated_items = user_ratings != 0
            if np.any(rated_items):
                self.user_bias[i] = np.mean(user_ratings[rated_items]) - self.global_bias
        
        for j in range(user_item_matrix.shape[1]):
            item_ratings = user_item_matrix[:, j]
            rated_users = item_ratings != 0
            if np.any(rated_users):
                self.item_bias[j] = np.mean(item_ratings[rated_users]) - self.global_bias
        
        # 中心化矩阵
        centered_matrix = user_item_matrix.copy()
        for i in range(user_item_matrix.shape[0]):
            for j in range(user_item_matrix.shape[1]):
                if user_item_matrix[i, j] != 0:
                    centered_matrix[i, j] = (user_item_matrix[i, j] - 
                                           self.global_bias - 
                                           self.user_bias[i] - 
                                           self.item_bias[j])
        
        # SVD分解
        U, sigma, Vt = svds(centered_matrix, k=self.n_factors)
        
        # 重构因子矩阵
        self.user_factors = U
        self.item_factors = Vt.T
        
        return self
    
    def predict(self, user_id, item_id):
        """
        预测用户对物品的评分
        """
        if self.user_factors is None or self.item_factors is None:
            return 0
        
        # 计算预测评分
        pred_rating = (self.global_bias + 
                      self.user_bias[user_id] + 
                      self.item_bias[item_id] + 
                      np.dot(self.user_factors[user_id], self.item_factors[item_id]))
        
        return pred_rating
    
    def recommend(self, user_id, n_recommendations=10):
        """
        为用户推荐物品
        """
        user_ratings = self.user_item_matrix[user_id]
        unrated_items = np.where(user_ratings == 0)[0]
        
        predictions = []
        for item_id in unrated_items:
            pred_rating = self.predict(user_id, item_id)
            predictions.append((item_id, pred_rating))
        
        # 按预测评分排序
        predictions.sort(key=lambda x: x[1], reverse=True)
        
        return predictions[:n_recommendations]
```

```python
from sklearn.decomposition import NMF

class NMFRecommender:

        为用户推荐物品
```

## 深度学习推荐系统

### 1. 深度协同过滤

#### 神经网络实现
```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

class DeepCF(nn.Module):
    def __init__(self, n_users, n_items, n_factors=50, hidden_dims=[100, 50]):
        super(DeepCF, self).__init__()
        
        self.n_users = n_users
        self.n_items = n_items
        self.n_factors = n_factors
        
        # 用户和物品嵌入层
        self.user_embedding = nn.Embedding(n_users, n_factors)
        self.item_embedding = nn.Embedding(n_items, n_factors)
        
        # 深度网络
        layers = []
        input_dim = n_factors * 2
        
        for hidden_dim in hidden_dims:
            layers.append(nn.Linear(input_dim, hidden_dim))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(0.2))
            input_dim = hidden_dim
        
        layers.append(nn.Linear(input_dim, 1))
        layers.append(nn.Sigmoid())
        
        self.deep_layers = nn.Sequential(*layers)
        
    def forward(self, user_ids, item_ids):
        # 获取用户和物品嵌入
        user_emb = self.user_embedding(user_ids)
        item_emb = self.item_embedding(item_ids)
        
        # 拼接嵌入向量
        concat_emb = torch.cat([user_emb, item_emb], dim=1)
        
        # 通过深度网络
        output = self.deep_layers(concat_emb)
        
        return output.squeeze()

class DeepCFRecommender:
    def __init__(self, n_users, n_items, n_factors=50, hidden_dims=[100, 50]):
        self.n_users = n_users
        self.n_items = n_items
        self.model = DeepCF(n_users, n_items, n_factors, hidden_dims)
        self.optimizer = optim.Adam(self.model.parameters(), lr=0.001)
        self.criterion = nn.MSELoss()
    
    def fit(self, user_item_matrix, epochs=100, batch_size=256):
        """
        训练深度协同过滤模型
        """
        # 准备训练数据
        user_ids = []
        item_ids = []
        ratings = []
        
        for i in range(user_item_matrix.shape[0]):
            for j in range(user_item_matrix.shape[1]):
                if user_item_matrix[i, j] != 0:
                    user_ids.append(i)
                    item_ids.append(j)
                    ratings.append(user_item_matrix[i, j])
        
        # 转换为张量
        user_ids = torch.LongTensor(user_ids)
        item_ids = torch.LongTensor(item_ids)
        ratings = torch.FloatTensor(ratings)
        
        # 创建数据加载器
        dataset = TensorDataset(user_ids, item_ids, ratings)
        dataloader = DataLoader(dataset, batch_size=batch_size, shuffle=True)
        
        # 训练模型
        self.model.train()
        for epoch in range(epochs):
            total_loss = 0
            for batch_user_ids, batch_item_ids, batch_ratings in dataloader:
                self.optimizer.zero_grad()
                
                # 前向传播
                predictions = self.model(batch_user_ids, batch_item_ids)
                
                # 计算损失
                loss = self.criterion(predictions, batch_ratings)
                
                # 反向传播
                loss.backward()
                self.optimizer.step()
                
                total_loss += loss.item()
            
            if epoch % 10 == 0:
                print(f'Epoch {epoch}, Loss: {total_loss/len(dataloader):.4f}')
    
    def predict(self, user_id, item_id):
        """
        预测用户对物品的评分
        """
        self.model.eval()
        with torch.no_grad():
            user_tensor = torch.LongTensor([user_id])
            item_tensor = torch.LongTensor([item_id])
            prediction = self.model(user_tensor, item_tensor)
            return prediction.item()
    
    def recommend(self, user_id, n_recommendations=10):
        """
        为用户推荐物品
        """
        self.model.eval()
        predictions = []
        
        with torch.no_grad():
            for item_id in range(self.n_items):
                pred_rating = self.predict(user_id, item_id)
                predictions.append((item_id, pred_rating))
        
        # 按预测评分排序
        predictions.sort(key=lambda x: x[1], reverse=True)
        
        return predictions[:n_recommendations]
```

```python
class NeuralCF(nn.Module):
    def __init__(self, n_users, n_items, n_factors=50, hidden_dims=[100, 50]):
        super(NeuralCF, self).__init__()
        
        self.n_users = n_users
        self.n_items = n_items
        self.n_factors = n_factors
        
        # 用户和物品嵌入层
        self.user_embedding = nn.Embedding(n_users, n_factors)
        self.item_embedding = nn.Embedding(n_items, n_factors)
        
        # 广义矩阵分解（GMF）
        self.gmf_layer = nn.Linear(n_factors, n_factors)
        
        # 多层感知机（MLP）
        mlp_layers = []
        input_dim = n_factors * 2
        
        for hidden_dim in hidden_dims:
            mlp_layers.append(nn.Linear(input_dim, hidden_dim))
            mlp_layers.append(nn.ReLU())
            mlp_layers.append(nn.Dropout(0.2))
            input_dim = hidden_dim
        
        self.mlp_layers = nn.Sequential(*mlp_layers)
        
        # 最终预测层
        self.prediction_layer = nn.Linear(n_factors + hidden_dims[-1], 1)
        self.sigmoid = nn.Sigmoid()
        
    def forward(self, user_ids, item_ids):
        # 获取用户和物品嵌入
        user_emb = self.user_embedding(user_ids)
        item_emb = self.item_embedding(item_ids)
        
        # GMF部分
        gmf_output = self.gmf_layer(user_emb * item_emb)
        
        # MLP部分
        mlp_input = torch.cat([user_emb, item_emb], dim=1)
        mlp_output = self.mlp_layers(mlp_input)
        
        # 拼接GMF和MLP输出
        concat_output = torch.cat([gmf_output, mlp_output], dim=1)
        
        # 最终预测
        prediction = self.prediction_layer(concat_output)
        prediction = self.sigmoid(prediction)
        
        return prediction.squeeze()
```

## 推荐系统评估

### 1. 评估指标

#### 准确率指标
```python
import numpy as np
from sklearn.metrics import mean_squared_error, mean_absolute_error

def evaluate_recommendation_system(model, test_data, user_item_matrix):
    """
    评估推荐系统性能
    """
    predictions = []
    actual_ratings = []
    
    for user_id, item_id, rating in test_data:
        pred_rating = model.predict(user_id, item_id)
        predictions.append(pred_rating)
        actual_ratings.append(rating)
    
    # 计算评估指标
    mse = mean_squared_error(actual_ratings, predictions)
    rmse = np.sqrt(mse)
    mae = mean_absolute_error(actual_ratings, predictions)
    
    return {
        'MSE': mse,
        'RMSE': rmse,
        'MAE': mae
    }

def precision_at_k(recommendations, relevant_items, k=10):
    """
    计算Precision@K
    """
    if len(recommendations) == 0:
        return 0
    
    recommended_items = [item for item, _ in recommendations[:k]]
    relevant_recommended = len(set(recommended_items) & set(relevant_items))
    
    return relevant_recommended / k

def recall_at_k(recommendations, relevant_items, k=10):
    """
    计算Recall@K
    """
    if len(relevant_items) == 0:
        return 0
    
    recommended_items = [item for item, _ in recommendations[:k]]
    relevant_recommended = len(set(recommended_items) & set(relevant_items))
    
    return relevant_recommended / len(relevant_items)

def ndcg_at_k(recommendations, relevant_items, k=10):
    """
    计算NDCG@K
    """
    if len(recommendations) == 0:
        return 0
    
    # 计算DCG
    dcg = 0
    for i, (item, score) in enumerate(recommendations[:k]):
        if item in relevant_items:
            dcg += score / np.log2(i + 2)
    
    # 计算IDCG
    idcg = 0
    for i in range(min(k, len(relevant_items))):
        idcg += 1 / np.log2(i + 2)
    
    return dcg / idcg if idcg > 0 else 0
```

### 2. 评估框架

#### 完整评估流程
```python
class RecommendationEvaluator:
    def __init__(self):
        self.metrics = {}
    
    def evaluate_model(self, model, test_data, user_item_matrix, k_values=[5, 10, 20]):
        """
        全面评估推荐模型
        """
        results = {}
        
        # 准确率评估
        accuracy_metrics = evaluate_recommendation_system(model, test_data, user_item_matrix)
        results['accuracy'] = accuracy_metrics
        
        # 排名评估
        ranking_metrics = {}
        for k in k_values:
            precision_scores = []
            recall_scores = []
            ndcg_scores = []
            
            for user_id in range(user_item_matrix.shape[0]):
                # 获取用户的实际评分物品
                user_ratings = user_item_matrix[user_id]
                relevant_items = np.where(user_ratings > 3)[0]  # 假设评分>3为相关
                
                if len(relevant_items) > 0:
                    # 获取推荐列表
                    recommendations = model.recommend(user_id, n_recommendations=k)
                    
                    # 计算指标
                    precision = precision_at_k(recommendations, relevant_items, k)
                    recall = recall_at_k(recommendations, relevant_items, k)
                    ndcg = ndcg_at_k(recommendations, relevant_items, k)
                    
                    precision_scores.append(precision)
                    recall_scores.append(recall)
                    ndcg_scores.append(ndcg)
            
            ranking_metrics[f'precision@{k}'] = np.mean(precision_scores)
            ranking_metrics[f'recall@{k}'] = np.mean(recall_scores)
            ranking_metrics[f'ndcg@{k}'] = np.mean(ndcg_scores)
        
        results['ranking'] = ranking_metrics
        
        return results
```

```python
        self.item_cf.fit(user_item_matrix)
        self.svd_model.fit(user_item_matrix)
        
        # 如果有用户和物品特征，训练深度学习模型
        if user_features is not None and item_features is not None:
            n_users, n_items = user_item_matrix.shape
            self.deep_cf = DeepCFRecommender(n_users, n_items)
            self.deep_cf.fit(user_item_matrix)
    
    def recommend(self, user_id, n_recommendations=10, method='ensemble'):
        """
        生成推荐列表
        """
        if method == 'ensemble':
            # 集成推荐
            user_cf_recs = self.user_cf.recommend(user_id, n_recommendations)
            item_cf_recs = self.item_cf.recommend(user_id, n_recommendations)
            svd_recs = self.svd_model.recommend(user_id, n_recommendations)
            
            # 加权融合
            ensemble_recs = {}
            for recs, weight in [(user_cf_recs, 0.3), (item_cf_recs, 0.3), (svd_recs, 0.4)]:
                for item_id, score in recs:
                    if item_id not in ensemble_recs:
                        ensemble_recs[item_id] = 0
                    ensemble_recs[item_id] += weight * score
            
            # 排序并返回
            sorted_recs = sorted(ensemble_recs.items(), key=lambda x: x[1], reverse=True)
            return sorted_recs[:n_recommendations]
        
        elif method == 'user_cf':
            return self.user_cf.recommend(user_id, n_recommendations)
        elif method == 'item_cf':
            return self.item_cf.recommend(user_id, n_recommendations)
        elif method == 'svd':
            return self.svd_model.recommend(user_id, n_recommendations)
        elif method == 'deep_cf' and self.deep_cf is not None:
            return self.deep_cf.recommend(user_id, n_recommendations)
        else:
            raise ValueError("Invalid recommendation method")
```

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np

class ContentBasedRecommender:
    def __init__(self):
        self.vectorizer = TfidfVectorizer()
        self.item_features = None
        self.item_similarity = None
    
    def fit(self, item_descriptions):
        """
        训练基于内容的推荐模型
        """
        # 提取文本特征
        self.item_features = self.vectorizer.fit_transform(item_descriptions)
        
        # 计算物品相似度
        self.item_similarity = cosine_similarity(self.item_features)
        
        return self
    
    def recommend(self, user_profile, n_recommendations=10):
        """
        基于用户画像推荐内容

        user_scores = np.dot(self.item_features, user_profile)
        
        # 排序并返回推荐
        item_scores = list(enumerate(user_scores))
        item_scores.sort(key=lambda x: x[1], reverse=True)
        
        return item_scores[:n_recommendations]
```

```python
class ColdStartHandler:
    def __init__(self):
        self.popularity_model = None
        self.content_model = None
    
    def handle_new_user(self, user_features, item_features):
        """
        处理新用户冷启动
        """
        # 基于内容的推荐
        content_recs = self.content_model.recommend(user_features)
        
        # 热门物品推荐
        popular_recs = self.popularity_model.get_popular_items()
        
        # 混合推荐
        mixed_recs = self._mix_recommendations(content_recs, popular_recs)
        
        return mixed_recs
    
    def handle_new_item(self, item_features, user_features):
        """
        处理新物品冷启动
        """
        # 基于物品特征找到相似物品
        similar_items = self.content_model.find_similar_items(item_features)
        
        # 基于相似物品的用户偏好推荐
        recommendations = self._recommend_by_similar_items(similar_items)
        
        return recommendations
    
    def _mix_recommendations(self, content_recs, popular_recs, alpha=0.7):
        """
        混合推荐结果
        """
        mixed_recs = {}
        
        # 内容推荐权重
        for item_id, score in content_recs:
            mixed_recs[item_id] = alpha * score
        
        # 热门推荐权重
        for item_id, score in popular_recs:
            if item_id in mixed_recs:
                mixed_recs[item_id] += (1 - alpha) * score
            else:
                mixed_recs[item_id] = (1 - alpha) * score
        
        # 排序并返回
        sorted_recs = sorted(mixed_recs.items(), key=lambda x: x[1], reverse=True)
        return sorted_recs
```

```python
class OnlineRecommender:
    def __init__(self, base_model):
        self.base_model = base_model
        self.user_profiles = {}
        self.item_profiles = {}
    
    def update_user_profile(self, user_id, interaction_data):
        """
        实时更新用户画像
        """
        if user_id not in self.user_profiles:
            self.user_profiles[user_id] = {}
        
        # 更新用户偏好
        for item_id, rating, timestamp in interaction_data:
            self.user_profiles[user_id][item_id] = {
                'rating': rating,
                'timestamp': timestamp
            }
    
    def get_realtime_recommendations(self, user_id, n_recommendations=10):
        """
        获取实时推荐
        """
        # 基于最新用户画像生成推荐
        user_profile = self.user_profiles.get(user_id, {})
        
        # 结合历史推荐和实时更新
        recommendations = self.base_model.recommend(user_id, n_recommendations)
        
        # 根据实时反馈调整推荐
        adjusted_recs = self._adjust_recommendations(recommendations, user_profile)
        
        return adjusted_recs
    
    def _adjust_recommendations(self, recommendations, user_profile):
        """
        根据实时反馈调整推荐
        """
        adjusted_recs = []
        
        for item_id, score in recommendations:
            # 如果用户最近有负面反馈，降低推荐分数
            if item_id in user_profile:
                recent_rating = user_profile[item_id]['rating']
                if recent_rating < 3:  # 负面反馈
                    score *= 0.5
            
            adjusted_recs.append((item_id, score))
        
        # 重新排序
        adjusted_recs.sort(key=lambda x: x[1], reverse=True)
        
        return adjusted_recs
```
