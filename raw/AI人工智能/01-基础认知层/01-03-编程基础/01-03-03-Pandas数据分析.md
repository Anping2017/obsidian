# Pandas数据分析

## 🐼 Pandas核心概念

### 1. 数据结构

**Series（序列）：**
```python
import pandas as pd
import numpy as np

# 创建Series
s = pd.Series([1, 2, 3, 4, 5])
s_with_index = pd.Series([1, 2, 3, 4, 5], index=['a', 'b', 'c', 'd', 'e'])

# Series操作
print(s.values)        # 获取值
print(s.index)         # 获取索引
print(s.dtype)         # 数据类型
print(s.describe())    # 统计描述
```

**DataFrame（数据框）：**
```python
# 创建DataFrame
data = {
    'Name': ['Alice', 'Bob', 'Charlie', 'David'],
    'Age': [25, 30, 35, 28],
    'City': ['Beijing', 'Shanghai', 'Guangzhou', 'Shenzhen'],
    'Salary': [50000, 60000, 70000, 55000]
}
df = pd.DataFrame(data)

# DataFrame属性
print(df.shape)        # (4, 4) - 形状
print(df.columns)       # 列名
print(df.index)         # 索引
print(df.dtypes)        # 数据类型
print(df.info())        # 详细信息
```

### 2. 数据读取和写入

**读取数据：**
```python
# 读取CSV文件
df = pd.read_csv('data.csv')

# 读取Excel文件
df = pd.read_excel('data.xlsx', sheet_name='Sheet1')

# 读取JSON文件
df = pd.read_json('data.json')

# 读取数据库
import sqlite3
conn = sqlite3.connect('database.db')
df = pd.read_sql_query("SELECT * FROM table_name", conn)
```

**写入数据：**
```python
# 写入CSV文件
df.to_csv('output.csv', index=False)

# 写入Excel文件
df.to_excel('output.xlsx', sheet_name='Sheet1', index=False)

# 写入JSON文件
df.to_json('output.json', orient='records')

# 写入数据库
df.to_sql('table_name', conn, if_exists='replace', index=False)
```

## 📊 数据操作

### 1. 数据选择

**基本选择：**
```python
# 选择列
df['Name']              # 单列
df[['Name', 'Age']]     # 多列

# 选择行
df.iloc[0]              # 第一行
df.iloc[0:3]            # 前3行
df.loc[0]               # 索引为0的行
df.loc[0:2]             # 索引0到2的行

# 条件选择
df[df['Age'] > 30]      # 年龄大于30
df[(df['Age'] > 30) & (df['Salary'] > 60000)]  # 复合条件
```

**高级选择：**
```python
# 使用query方法
df.query('Age > 30 and Salary > 60000')

# 使用isin方法
df[df['City'].isin(['Beijing', 'Shanghai'])]

# 使用contains方法
df[df['Name'].str.contains('a')]

# 使用between方法
df[df['Age'].between(25, 35)]
```

### 2. 数据修改

**添加列：**
```python
# 添加新列
df['Bonus'] = df['Salary'] * 0.1

# 使用assign方法
df = df.assign(Bonus=df['Salary'] * 0.1)

# 使用insert方法
df.insert(2, 'Department', ['IT', 'HR', 'IT', 'HR'])

# 使用apply方法
df['Age_Group'] = df['Age'].apply(lambda x: 'Young' if x < 30 else 'Old')
```

**修改数据：**
```python
# 修改特定值
df.loc[df['Name'] == 'Alice', 'Salary'] = 55000

# 批量修改
df['Salary'] = df['Salary'] * 1.1  # 涨薪10%

# 使用replace方法
df['City'] = df['City'].replace({'Beijing': 'BJ', 'Shanghai': 'SH'})

# 使用map方法
city_mapping = {'Beijing': 'BJ', 'Shanghai': 'SH'}
df['City_Code'] = df['City'].map(city_mapping)
```

### 3. 数据清洗

**处理缺失值：**
```python
# 检查缺失值
print(df.isnull().sum())        # 每列缺失值数量
print(df.isnull().any())        # 是否有缺失值

# 删除缺失值
df_dropna = df.dropna()         # 删除有缺失值的行
df_dropna_col = df.dropna(axis=1)  # 删除有缺失值的列

# 填充缺失值
df_fillna = df.fillna(0)        # 用0填充
df_fillna_mean = df.fillna(df.mean())  # 用均值填充
df_fillna_ffill = df.fillna(method='ffill')  # 前向填充
```

**处理重复值：**
```python
# 检查重复值
print(df.duplicated().sum())    # 重复行数量

# 删除重复值
df_unique = df.drop_duplicates()

# 基于特定列删除重复值
df_unique_col = df.drop_duplicates(subset=['Name'])
```

## 📈 数据分析

### 1. 描述性统计

```python
# 基本统计
print(df.describe())            # 数值列统计
print(df.describe(include='all'))  # 所有列统计

# 特定统计
print(df['Salary'].mean())      # 均值
print(df['Salary'].median())    # 中位数
print(df['Salary'].std())       # 标准差
print(df['Salary'].quantile([0.25, 0.75]))  # 四分位数

# 分组统计
print(df.groupby('City')['Salary'].mean())  # 按城市分组求平均薪资
```

### 2. 数据分组

**基本分组：**
```python
# 单列分组
grouped = df.groupby('City')
print(grouped['Salary'].sum())   # 按城市求和
print(grouped['Salary'].mean())  # 按城市求平均

# 多列分组
grouped_multi = df.groupby(['City', 'Department'])
print(grouped_multi['Salary'].mean())

# 分组后应用函数
def custom_func(x):
    return x.max() - x.min()

print(grouped['Salary'].apply(custom_func))
```

**高级分组：**
```python
# 使用agg方法
result = df.groupby('City').agg({
    'Salary': ['mean', 'std', 'count'],
    'Age': ['min', 'max']
})

# 使用transform方法
df['Salary_Mean'] = df.groupby('City')['Salary'].transform('mean')

# 使用filter方法
filtered = df.groupby('City').filter(lambda x: len(x) > 1)
```

### 3. 数据透视表

```python
# 基本透视表
pivot = df.pivot_table(
    values='Salary',
    index='City',
    columns='Department',
    aggfunc='mean'
)

# 复杂透视表
pivot_complex = df.pivot_table(
    values=['Salary', 'Age'],
    index='City',
    columns='Department',
    aggfunc={'Salary': 'mean', 'Age': 'max'},
    fill_value=0
)

# 使用crosstab
cross_tab = pd.crosstab(df['City'], df['Department'], values=df['Salary'], aggfunc='mean')
```

## 🔄 数据转换

### 1. 数据重塑

**长宽格式转换：**
```python
# 宽格式转长格式
df_long = df.melt(
    id_vars=['Name', 'Age'],
    value_vars=['Salary', 'Bonus'],
    var_name='Type',
    value_name='Amount'
)

# 长格式转宽格式
df_wide = df_long.pivot(
    index=['Name', 'Age'],
    columns='Type',
    values='Amount'
)
```

**数据堆叠：**
```python
# 堆叠
stacked = df.set_index(['Name', 'City']).stack()

# 解堆叠
unstacked = stacked.unstack()
```

### 2. 数据合并

**基本合并：**
```python
# 创建第二个DataFrame
df2 = pd.DataFrame({
    'Name': ['Alice', 'Bob', 'Eve'],
    'Department': ['IT', 'HR', 'Finance'],
    'Manager': ['John', 'Jane', 'Mike']
})

# 内连接
inner_join = pd.merge(df, df2, on='Name', how='inner')

# 左连接
left_join = pd.merge(df, df2, on='Name', how='left')

# 外连接
outer_join = pd.merge(df, df2, on='Name', how='outer')
```

**高级合并：**
```python
# 多列连接
multi_join = pd.merge(df, df2, on=['Name', 'Department'], how='inner')

# 索引连接
index_join = df.join(df2.set_index('Name'), on='Name')

# 纵向合并
concat_df = pd.concat([df, df2], ignore_index=True)
```

## 🎯 时间序列分析

### 1. 时间索引

```python
# 创建时间序列
dates = pd.date_range('2023-01-01', periods=100, freq='D')
ts = pd.Series(np.random.randn(100), index=dates)

# 时间索引操作
print(ts['2023-01-01'])         # 特定日期
print(ts['2023-01'])            # 特定月份
print(ts['2023-01-01':'2023-01-10'])  # 日期范围

# 时间重采样
daily = ts.resample('D').mean()  # 日平均
weekly = ts.resample('W').mean()  # 周平均
monthly = ts.resample('M').mean()  # 月平均
```

### 2. 时间序列分析

```python
# 移动平均
ts_ma = ts.rolling(window=7).mean()

# 指数移动平均
ts_ema = ts.ewm(span=7).mean()

# 时间差分
ts_diff = ts.diff()

# 季节性分解
from statsmodels.tsa.seasonal import seasonal_decompose
decomposition = seasonal_decompose(ts, model='additive')
```

## 🔧 性能优化

### 1. 数据类型优化

```python
# 检查数据类型
print(df.dtypes)

# 优化数据类型
df['Age'] = df['Age'].astype('int8')      # 8位整数
df['Salary'] = df['Salary'].astype('float32')  # 32位浮点数
df['City'] = df['City'].astype('category')     # 分类数据

# 检查内存使用
print(df.memory_usage(deep=True))
```

### 2. 查询优化

```python
# 使用query方法（通常更快）
result = df.query('Age > 30 and Salary > 60000')

# 使用isin方法
result = df[df['City'].isin(['Beijing', 'Shanghai'])]

# 使用categorical类型
df['City'] = df['City'].astype('category')
result = df[df['City'] == 'Beijing']  # 更快的比较
```

### 3. 并行处理

```python
# 使用multiprocessing
from multiprocessing import Pool

def process_chunk(chunk):
    return chunk.groupby('City')['Salary'].mean()

# 分块处理
chunks = [df[i:i+1000] for i in range(0, len(df), 1000)]
with Pool() as pool:
    results = pool.map(process_chunk, chunks)
```
