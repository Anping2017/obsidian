# 数学工具速查

## 🔧 常用数学工具

### 1. 线性代数工具

**矩阵运算：**
```python
import numpy as np

# 矩阵创建
A = np.array([[1, 2], [3, 4]])
B = np.array([[5, 6], [7, 8]])

# 基本运算
C = A + B          # 矩阵加法
D = A @ B          # 矩阵乘法
E = A.T            # 转置
F = np.linalg.inv(A)  # 逆矩阵

# 特征值分解
eigenvals, eigenvecs = np.linalg.eig(A)

# SVD分解
U, S, Vt = np.linalg.svd(A)
```

**向量运算：**
```python
# 向量操作
v1 = np.array([1, 2, 3])
v2 = np.array([4, 5, 6])

dot_product = np.dot(v1, v2)    # 点积
cross_product = np.cross(v1, v2)  # 叉积
norm = np.linalg.norm(v1)       # 范数
```

### 2. 概率统计工具

**概率分布：**
```python
import scipy.stats as stats

# 正态分布
normal = stats.norm(loc=0, scale=1)
pdf = normal.pdf(x)    # 概率密度
cdf = normal.cdf(x)    # 累积分布
rvs = normal.rvs(1000) # 随机采样

# 二项分布
binomial = stats.binom(n=10, p=0.5)
pmf = binomial.pmf(k)  # 概率质量
cdf = binomial.cdf(k)  # 累积分布

# 泊松分布
poisson = stats.poisson(mu=3)
pmf = poisson.pmf(k)
```

**统计检验：**
```python
from scipy import stats

# t检验
t_stat, p_value = stats.ttest_ind(group1, group2)

# 卡方检验
chi2_stat, p_value = stats.chi2_contingency(contingency_table)

# 正态性检验
statistic, p_value = stats.normaltest(data)
```

### 3. 微积分工具

**数值微分：**
```python
from scipy.misc import derivative

# 数值导数
f = lambda x: x**2
df = derivative(f, x0, dx=1e-6)

# 梯度计算
from scipy.optimize import approx_fprime
gradient = approx_fprime(x0, func, epsilon=1e-6)
```

**数值积分：**
```python
from scipy.integrate import quad, trapz

# 定积分
result, error = quad(func, a, b)

# 梯形积分
result = trapz(y, x)

# 蒙特卡洛积分
from scipy.stats import uniform
samples = uniform.rvs(a, b-a, size=10000)
result = (b-a) * np.mean(func(samples))
```

## 🎯 优化工具

### 1. 数值优化

```python
from scipy.optimize import minimize, minimize_scalar

# 无约束优化
def objective(x):
    return x**2 + 2*x + 1

result = minimize(objective, x0=0)
print(f"Optimal x: {result.x}")
print(f"Optimal value: {result.fun}")

# 约束优化
from scipy.optimize import LinearConstraint, NonlinearConstraint

constraints = [{'type': 'ineq', 'fun': lambda x: x[0] - 1}]
result = minimize(objective, x0=[0], constraints=constraints)
```

### 2. 线性规划

```python
from scipy.optimize import linprog

# 线性规划
c = [-1, -2]  # 目标函数系数
A = [[1, 1], [2, 1]]  # 约束矩阵
b = [3, 4]  # 约束右端
bounds = [(0, None), (0, None)]  # 变量边界

result = linprog(c, A_ub=A, b_ub=b, bounds=bounds)
print(f"Optimal solution: {result.x}")
print(f"Optimal value: {result.fun}")
```

## 🧮 符号计算

x, y = sp.symbols('x y')

## 📈 信号处理工具

### 1. 傅里叶变换

```python
from scipy.fft import fft, ifft, fftfreq

# 信号生成
t = np.linspace(0, 1, 1000)
signal = np.sin(2*np.pi*50*t) + 0.5*np.sin(2*np.pi*120*t)

# 傅里叶变换
fft_signal = fft(signal)
frequencies = fftfreq(len(signal), t[1]-t[0])

# 功率谱
power_spectrum = np.abs(fft_signal)**2
plt.plot(frequencies[:len(frequencies)//2], power_spectrum[:len(power_spectrum)//2])
plt.title('Power Spectrum')
plt.show()
```

### 2. 滤波器设计

```python
from scipy.signal import butter, filtfilt

# 低通滤波器
b, a = butter(4, 0.1, btype='low')
filtered_signal = filtfilt(b, a, signal)

plt.plot(t, signal, label='Original')
plt.plot(t, filtered_signal, label='Filtered')
plt.legend()
plt.show()
```
